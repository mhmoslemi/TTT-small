Goal: implement kernel as:

```
@triton.jit
def batched_gemm_outgoing_kernel(A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # idx
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N
    # pointer to C tile
    c_ptrs = C_ptr + batch * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K
        # A
        a_ptrs = A_ptr + batch * stride_az + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # B (notice stride ordering: K dimension then N dimension)
        b_ptrs = B_ptr + batch * stride_bz + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # convert to float32 for accumulation
        acc += tl.dot(a, b)
    # store
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps=1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Convert to half for speed
    x_h = x.to(torch.float16)

    # Linear weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Projections
    left = torch.nn.functional.linear(x_h, left_proj_w)   # shape B,N,N,hidden
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # Mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x_h, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute contraction using Triton
    B, N, _, H = left.shape   # H = hidden_dim
    # allocate output
    out_half = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # For right (same shape)
    rs_b, rs_i, rs_k, rs_h = right.stride()

    # Output strides
    cs_b, cs_i, cs_j, cs_h = out_half.stride()

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

"""
TriMul (outgoing) forward pass.

The heavy part is the 3‑D tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication (A @ Bᵀ) for each (b,d) pair.
We keep all preparatory ops (LayerNorm, Linear, gating, mask) in PyTorch and
offload only this contraction to a custom Triton kernel. The kernel works on
half‑precision tensors, accumulates in fp32 and writes the result back in fp16.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trmul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    # strides for A (left)
    stride_A_batch, stride_A_h, stride_A_i, stride_A_k,
    # strides for B (right, transposed as (k,j))
    stride_B_batch, stride_B_h, stride_B_k, stride_B_j,
    # strides for C (output)
    stride_C_batch, stride_C_h, stride_C_i, stride_C_j,
    M, N, K,                # all equal to sequence length
    H: tl.constexpr,        # hidden dimension (num heads)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ   where
      A : (B, H, M, K)   (M=i, K=k)
      B : (B, H, K, N)   (K=k, N=j)
      C : (B, H, M, N)   (i,j)
    The first program_id (axis 0) indexes the flattened (batch, head) dimension.
    """
    pid_flat = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row   (i)
    pid_n = tl.program_id(2)   # block col   (j)

    # recover batch index b and head index h from flattened id
    b = pid_flat // H
    h = pid_flat % H

    # offsets inside the M/N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (b,h) slice
    a_base = A_ptr + b * stride_A_batch + h * stride_A_h
    b_base = B_ptr + b * stride_B_batch + h * stride_B_h
    c_base = C_ptr + b * stride_C_batch + h * stride_C_h

    # pointer to output tile
    c_ptrs = c_base + offs_m[:, None] * stride_C_i + offs_n[None, :] * stride_C_j
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_A_i + offs_k[None, :] * stride_A_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # load B tile (K x N)   (rows = k, cols = j)
        b_ptrs = b_base + offs_k[:, None] * stride_B_k + offs_n[None, :] * stride_B_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write result (convert back to fp16)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward of the outgoing TriMul module.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensors containing layer‑norm and linear weights
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------------- LayerNorm on input ----------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------- Projections & Gates (half) -------------------
    x_h = x.to(torch.float16)

    w_left = weights["left_proj.weight"].to(torch.float16)          # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)          # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, w_right)

    # ---- optional mask -------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ---- gating ---------------------------------------------------------
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()        # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------- Contraction with Triton -------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    out_half = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # Strides of the tensors (as they are laid out in memory)
    s_b, s_i, s_k, s_h = left.stride()    # left: (B, i, k, H)
    rs_b, rs_i, rs_k, rs_h = right.stride()  # right: same layout

    cs_b, cs_i, cs_j, cs_h = out_half.stride()   # out: (B, i, j, H)

    # block sizes – chosen for good occupancy on B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                     # flattened (batch, head)
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
    )

    _trmul_outgoing_kernel[grid](
        left,
        right,
        out_half,
        # A strides
        s_b, s_h, s_i, s_k,
        # B strides (transposed as k,j)
        rs_b, rs_h, rs_k, rs_i,
        # C strides
        cs_b, cs_h, cs_i, cs_j,
        N, N, N,                     # M, N, K
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------- Output LayerNorm & Projection -------------------
    # Convert to fp32 for the remaining ops
    out_fp32 = out_half.to(torch.float32)

    # to_out_norm
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    mean_o = out_fp32.mean(dim=-1, keepdim=True)
    var_o = out_fp32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_fp32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * w_outnorm + b_outnorm

    # apply out gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to original dim
    w_to_out = weights["to_out.weight"]
    output = F.linear(out_norm, w_to_out)   # [B, N, N, dim] float32

    return output
