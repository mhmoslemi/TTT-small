def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights['norm.weight'],
                     bias=weights['norm.bias'],
                     eps=1e-5)

    B,N,_,D = x.shape

    # Linear projections and gates
    x_flat = x.view(-1, dim)

    left = F.linear(x_flat, weights['left_proj.weight']).view(B,N,N,hidden_dim)
    right = F.linear(x_flat, weights['right_proj.weight']).view(B,N,N,hidden_dim)

    left_gate = F.linear(x_flat, weights['left_gate.weight']).view(B,N,N,hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights['right_gate.weight']).view(B,N,N,hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights['out_gate.weight']).view(B,N,N,hidden_dim).sigmoid()

    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # bfloat16 for contraction
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # reshape for batched GEMM
    left_bf_ = left_bf.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_bf_ = right_bf.permute(0,3,2,1).reshape(B*hidden_dim, N, N)

    out_bf = torch.bmm(left_bf_, right_bf_)  # shape (B*H,N,N)
    out = out_bf.reshape(B, hidden_dim, N, N).permute(0,2,3,1).to(x.dtype)

    # hidden dim LayerNorm via Triton
    out_norm = torch.empty_like(out)
    s_b, s_i, s_j, s_h = out.stride()
    total = B * N * N
    grid = (total,)
    BLOCK_H = 384  # enough
    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights['to_out_norm.weight'],
        weights['to_out_norm.bias'],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,

    out_scaled = out_norm * out_gate

"""
Triton‑accelerated outgoing TriMul forward.

Algorithm
---------
1. LayerNorm over the last (dim) axis (PyTorch).
2. Linear projections → left / right and three gating vectors (PyTorch).
3. Apply optional pairwise mask and elementwise gates.
4. Contract over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched GEMM:
       left_h = left[..., :, :, h]      (N×N)
       right_hᵀ = right[..., :, :, h]ᵀ (N×N)
   All heads are merged into a single batch dimension (B·H) and computed
   with a single torch.bmm in bfloat16.
5. LayerNorm over the hidden dimension (custom Triton kernel).
6. Multiply by the outgoing gate, then a final linear projection back to
   the original dimension (PyTorch).

Only the hidden‑dim LayerNorm is written in Triton; the expensive O(N³)
contraction is performed by an optimized batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,          # *float32  [B,N,N,H]
    out_ptr,          # *float32  [B,N,N,H]
    weight_ptr,       # *float32  [H]
    bias_ptr,         # *float32  [H]
    B,                # tl.int32
    N,                # tl.int32
    H,                # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,# tl.int32
    eps: tl.constexpr,                                   # const float
    BLOCK_H: tl.constexpr                               # compile‑time block size = hidden_dim
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode program id to (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets along H dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base pointers for the (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load H‑dimensional vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    )
    # ------------------------------------------------------------------
    # Compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    # ------------------------------------------------------------------
    # Compute variance
    diff = x - mean
    sq = diff * diff
    sum_sq = tl.sum(sq, axis=0)
    var = sum_sq / H

    # ------------------------------------------------------------------
    # Normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ------------------------------------------------------------------
    # Apply learned gain / bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # ------------------------------------------------------------------
    # Store result
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be ignored)
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2. Linear projections & gating (float32)
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # --------------------------------------------------------------
    # 3. Mask & gate application
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Pairwise contraction via batched GEMM (bfloat16)
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Bring hidden dim into batch dimension for GEMM
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right_bf.permute(0, 3, 2, 1).contiguous() # [B,H,N,N] (transpose N‑axes)

    left_batch = left_perm.view(B * hidden_dim, N, N)   # (B·H,N,N)
    right_batch = right_perm.view(B * hidden_dim, N, N) # (B·H,N,N)

    # Batched matrix multiplication
    out_bf_batch = torch.bmm(left_batch, right_batch)   # (B·H,N,N)

    # Reshape back to [B,N,N,H]
    out_bf = out_bf_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_bf.to(dtype)  # back to float32

    # --------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm (Triton)
    out_norm = torch.empty_like(out)
    s_b, s_i, s_j, s_h = out.stride()
    total_progs = B * N * N
    grid = (total_progs,)

    # BLOCK_H is a compile‑time constant; set to hidden_dim for exact sizing
    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=hidden_dim,
    )

    # --------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
