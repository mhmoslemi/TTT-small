"""
High‑performance Triton implementation of the outgoing TriMul operation.

Algorithm
---------
1. Layer‑norm the input tensor over its channel dimension (`dim`).
2. Linear projections → `left`, `right` (shape [B,N,N,H]) and three
   gate vectors (`left_gate`, `right_gate`, `out_gate`) using the provided
   weight matrices.  Apply the optional pairwise mask and the gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
3. Compute the pairwise contraction over the sequence dimension `k`:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication:
       out[b,:,:,h] = left[b,:,:,h] @ right[b,:,:,h].T
   The multiplication is performed in bfloat16 with `torch.bmm`.
4. Hidden‑dim LayerNorm (per (b,i,j) vector) is carried out by a custom
   Triton kernel to avoid the reduction overhead of the PyTorch version.
5. Multiply by `out_gate`, then apply the final linear projection back to
   the original channel dimension `dim`.

Only the hidden‑dim LayerNorm runs on Triton; the rest uses
high‑throughput PyTorch kernels (including a batched GEMM), which together
yield a substantial speed‑up over the reference `einsum` implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: Layer‑norm over the hidden dimension (the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,               # *float32  [B,N,N,H]
    out_ptr,               # *float32  [B,N,N,H]
    weight_ptr,            # *float32  [H]
    bias_ptr,              # *float32  [H]
    B,                     # tl.int32
    N,                     # tl.int32
    H,                     # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # tl.int32
    eps: tl.constexpr,                                   # const float
    BLOCK_H: tl.constexpr                               # compile‑time block size (≥ max hidden_dim)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets along hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Base address for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # Load the H‑dimensional vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    )

    # ---- mean ---------------------------------------------------------
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    # ---- variance -----------------------------------------------------
    diff = x - mean
    sq = diff * diff
    sum_sq = tl.sum(sq, axis=0)
    var = sum_sq / H

    # ---- normalize ----------------------------------------------------
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ---- apply affine parameters ---------------------------------------
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # ---- store ---------------------------------------------------------
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h
    )

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (optional, bool/int)
            - weights: dict of torch.Tensor parameters
            - config: dict containing "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]            # input channel dimension
    hidden_dim = config["hidden_dim"]  # hidden dimension (H)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over last dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gating (float32)
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) and gates
    if not nomask:
        # mask may be bool/int; convert to float for multiplication
        mask_exp = mask.to(dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction over k (batched GEMM)
    # Use bfloat16 for bandwidth‑limited matmul
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # reshape to (B*H, N, N)
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous()
    left_batch = left_perm.view(B * hidden_dim, N, N)
    right_batch = right_perm.view(B * hidden_dim, N, N)

    # Batched matmul: out_batch[b*h] = left_batch[b*h] @ right_batch[b*h].T
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

    # Reshape back to [B, N, N, hidden_dim] and cast to float32
    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype).contiguous()

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)  # allocate output
    s_b, s_i, s_j, s_h = out.stride()  # strides in element units

    total_programs = B * N * N
    grid = (total_programs,)

    # BLOCK_H must be a compile‑time constant ≥ hidden_dim; set to 128
    BLOCK_H = 128

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)        # [B*N*N, H]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
