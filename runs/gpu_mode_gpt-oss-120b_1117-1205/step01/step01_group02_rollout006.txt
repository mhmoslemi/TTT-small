"""
TriMul (outgoing) forward pass fused with a Triton batched GEMM.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Apply two linear projections (left / right) and three sigmoid gates
   (left, right, out) using provided weights (all in fp16).
3. Optionally mask the projected tensors (mask is broadcast over the hidden dim).
4. Fuse the heavy N³ computation:
      out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   via a Triton kernel that performs a batched GEMM
   (left @ rightᵀ) for each hidden head.
5. Layer‑norm the result over the hidden dimension, apply the output gate,
   and linearly project back to the original channel dimension.

The kernel accumulates in float32 for numerical stability and stores the
matrix in fp16 before casting back to fp32 for the remaining PyTorch ops.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_am, stride_ak,          # A: row stride, col stride
    stride_bk, stride_bn,          # B: row stride, col stride
    stride_cm, stride_cn,          # C: row stride, col stride
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  for many independent batches.

    A : [batch, M, K] (fp16)
    B : [batch, K, N] (fp16)
    C : [batch, M, N] (fp16, accumulation in fp32)
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch index (batch * hidden)

    # Offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # A block pointer (batch, row, col)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        # B block pointer (batch, row, col)
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot accumulates in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store C
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor containing all linear / LN parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (over channel dimension)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) and gate logits
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden]
    right = torch.nn.functional.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )

    left_gate = torch.nn.functional.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Apply element‑wise gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM: left @ rightᵀ   (out shape [B,N,N,hidden])
    # -----------------------------------------------------------------
    # Permute to [B, hidden, N, N] for easy batching
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose last two dims of right so we can do A @ B
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N] (K,N)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dims
    A = left_perm.view(total_batch, N, N)
    B_mat = right_T.view(total_batch, N, N)
    C = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_a_batch, stride_am, stride_ak = A.stride()
    stride_b_batch, stride_bk, stride_bn = B_mat.stride()
    stride_c_batch, stride_cm, stride_cn = C.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )

    _trimul_gemm_kernel[grid](
        A,
        B_mat,
        C,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = (
        C.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)        # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)

    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # [B, N, N, dim]

    return output
