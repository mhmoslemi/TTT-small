"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input over its last dimension.
2. Linear projections (left/right) and three gating projections (left_gate,
   right_gate, out_gate) using the supplied weight matrices.
3. Apply an optional pairwise mask, then multiply by the corresponding gates.
4. Compute the pairwise contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is performed as a batched matrix multiplication using torch.bmm on
   BF16 tensors (≈2× bandwidth reduction).
5. Apply a hidden‑dimension LayerNorm implemented in Triton (one program per
   (b,i,j) slice).
6. Multiply by the outgoing gate and linearly project back to the original
   dimensionality.
The function returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def hidden_layernorm_kernel(
    inp_ptr,                # *float32  [B,N,N,H]
    out_ptr,                # *float32  [B,N,N,H]
    weight_ptr,             # *float32  [H]
    bias_ptr,               # *float32  [H]
    B, N, H,                # program constants (int32)
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # input strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # output strides
    eps: tl.constexpr,      # epsilon for numeric stability
    BLOCK_H: tl.constexpr   # block size along hidden dimension (≥ H)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # (b,i,j) coordinate of this instance
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets within the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base pointers for this slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load the H‑dimensional vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    )
    # ------------------------------------------------------------------
    # Compute mean
    sum_x = tl.sum(x, axis=0)          # sum over hidden dim
    mean = sum_x / H

    # ------------------------------------------------------------------
    # Compute variance
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / H

    # ------------------------------------------------------------------
    # Normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ------------------------------------------------------------------
    # Apply learned gain/bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h,   mask=mask_h, other=0.0)
    out = norm * w + b_

    # ------------------------------------------------------------------
    # Store result
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be unused)
            - weights: dict of weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Ensure contiguity for later view/reshape ops
    x = x.contiguous()

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (float32)
    x_flat = x.view(-1, dim)  # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) and gate multiplications
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] -> float32
        mask_exp = mask.to(dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction via batched BF16 GEMM
    # Cast to bfloat16 to reduce memory traffic
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Free the FP32 versions early to save memory
    del left, right, left_gate, right_gate

    # Reshape to (B*H, N, N) for batched matmul
    left_mat = left_bf.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_bf.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_bf: (B*H, N, N) = left_mat @ right_mat^T
    out_bf = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, hidden_dim] and cast to FP32
    out = out_bf.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype)

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm implemented in Triton
    out_norm = torch.empty_like(out)

    # Strides in element units
    s_b, s_i, s_j, s_h = out.stride()
    # Launch one program per (b,i,j) slice
    total_programs = B * N * N
    grid = (total_programs,)

    # Choose a BLOCK_H that covers hidden_dim (multiple of 32)
    BLOCK_H = 128
    if hidden_dim > BLOCK_H:
        # round up to next multiple of 32
        BLOCK_H = ((hidden_dim + 31) // 32) * 32

    hidden_layernorm_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,   # input strides
        s_b, s_i, s_j, s_h,   # output strides
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)  # [B*N*N, hidden_dim]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
