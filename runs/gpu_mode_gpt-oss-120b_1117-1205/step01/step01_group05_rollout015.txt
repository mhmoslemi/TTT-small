@triton.jit
def _batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          stride_az, stride_am, stride_ak,
                          stride_bz, stride_bk, stride_bn,
                          stride_cz, stride_cm, stride_cn,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # row tile
    pid_n = tl.program_id(2)   # col tile

    # Compute offsets within the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to output tile
    c_ptrs = (C_ptr + batch_id * stride_cz
              + offs_m[:, None] * stride_cm
              + offs_n[None, :] * stride_cn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = (A_ptr + batch_id * stride_az
                  + offs_m[:, None] * stride_am
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],

The expensive contraction

    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]
    All tensors are row‑major.  A: (batch, M, K), B: (batch, K, N), C: (batch, M, N)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile column

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # load B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # store result (down‑cast to fp16)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask         : torch.Tensor [B, N, N] (bool/float) or None
            weights      : dict of torch.Tensors for all parameters
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    x = input_tensor  # float32
    # torch.nn.functional.layer_norm works on any shape
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for projection & gating
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear weight tensors (fp16)
    w_left = weights["left_proj.weight"].to(torch.float16)     # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Project to hidden space
    # ------------------------------------------------------------------
    left = F.linear(x_h, w_left)      # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool; convert to same dtype as the projections
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Batched GEMM (the heavy contraction)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N) and make RHS transposed (k dimension as rows)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    left_mat = left_perm.view(B * H, N, N)                     # (B*H, M=N, K=N)

    right_perm = right.permute(0, 3, 1, 2).contiguous()        # (B, H, N, N)
    right_mat = right_perm.permute(0, 1, 3, 2).contiguous()   # (B, H, N, N) == (B, H, K, N)
    right_mat = right_mat.view(B * H, N, N)                    # (B*H, K=N, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides for Triton (row‑major)
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tiling parameters (tuned for NVIDIA Hopper B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                           # batch dimension
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N)         # cols
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (broadcasted)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original dimension
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_weight)  # [B, N, N, dim] float32

    return output
