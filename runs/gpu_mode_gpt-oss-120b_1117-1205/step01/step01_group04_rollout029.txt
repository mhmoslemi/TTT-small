"""
TriMul (outgoing) forward pass with Triton-accelerated core contraction.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections `left_proj` and `right_proj` (no bias).
3. Optional mask multiplication (applied before gating).
4. Gating: three sigmoid‑activated linear layers (`left_gate`,
   `right_gate`, `out_gate`) multiplied element‑wise to the projected tensors.
5. Core contraction:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication:
      out = left @ rightᵀ  for each (batch, hidden) pair.
   Implemented as a Triton kernel that computes many independent GEMMs
   (batch = B * hidden_dim) with tiles (BLOCK_M, BLOCK_N, BLOCK_K).
6. Layer‑norm over the hidden dimension, multiplicative output gate,
   and final linear projection back to the original channel dimension.

The heavy O(N³·H) contraction is performed entirely on the GPU using
float‑16 inputs and float‑32 accumulation, achieving high throughput on
modern NVIDIA GPUs (e.g. B200).  All other operations are kept in PyTorch
for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM computing C = A @ Bᵀ.

    A : [batch, M, K]   (float16)
    B : [batch, N, K]   (float16) – loaded transposed as (K, N)
    C : [batch, M, N]   (float16)

    The kernel tiles over M, N and loops over K.
    """
    batch_id = tl.program_id(2)          # independent GEMM per (batch, hidden)
    pid_m = tl.program_id(0)             # tile row
    pid_n = tl.program_id(1)             # tile column

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the current batch
    a_base = A_ptr + batch_id * stride_a_batch
    b_base = B_ptr + batch_id * stride_b_batch
    c_base = C_ptr + batch_id * stride_c_batch

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of tiles along K dimension
    num_tiles_k = tl.cdiv(K, BLOCK_K)

    for k in range(0, num_tiles_k):
        k_offset = k * BLOCK_K
        mask_k = (k_offset + offs_k) < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + (k_offset + offs_k)[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile transposed on‑the‑fly: (BLOCK_K, BLOCK_N)
        # B is stored as (batch, N, K); we need (K, N) for the transpose.
        b_ptrs = b_base + (k_offset + offs_k)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate the matrix product
        acc += tl.dot(a, b)

    # Write out the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: contains all linear and layer‑norm parameters
            - config_dict : must contain "dim" and "hidden_dim".
                           Optional key "nomask" (bool, default True).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------ #
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Cast projected tensors to half‑precision for the heavy GEMM
    # ------------------------------------------------------------------ #
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 6. Reshape to (batch_h, N, N) where batch_h = B * hidden_dim
    # ------------------------------------------------------------------ #
    #   (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_batch = left_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_batch = right_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    batch_h = left_batch.shape[0]  # B * hidden_dim

    # ------------------------------------------------------------------ #
    # 7. Allocate output buffer for the GEMM results (half‑precision)
    # ------------------------------------------------------------------ #
    out_fp16 = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------ #
    # 8. Triton launch parameters
    # ------------------------------------------------------------------ #
    # Tuned tile sizes: M/N = 64, K = 128 (reduces K-loop iterations)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    grid = (num_blocks_m, num_blocks_n, batch_h)

    # Strides in elements (not bytes)
    stride_a_batch = left_batch.stride(0)
    stride_a_m = left_batch.stride(1)
    stride_a_k = left_batch.stride(2)

    stride_b_batch = right_batch.stride(0)
    stride_b_n = right_batch.stride(1)   # second dim (N)
    stride_b_k = right_batch.stride(2)   # third dim (K)

    stride_c_batch = out_fp16.stride(0)
    stride_c_m = out_fp16.stride(1)
    stride_c_n = out_fp16.stride(2)

    # ------------------------------------------------------------------ #
    # 9. Launch the Triton kernel (C = A @ Bᵀ)
    # ------------------------------------------------------------------ #
    tri_mul_gemm_kernel[grid](
        left_batch,
        right_batch,
        out_fp16,
        N, N, N,                               # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 10. Reshape back to [B, N, N, hidden_dim] and convert to float32
    # ------------------------------------------------------------------ #
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------ #
    # 11. Output gate, post‑norm, final linear projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
