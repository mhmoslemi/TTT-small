```python
import torch
import triton
import triton.language as tl

@triton.jit
def fuse_gate_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    size,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    left_out = left * left_gate
    right_out = right * right_gate

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Steps:
    1. Layer‑norm ``x`` over the last dimension.
    2. Linear projections left/right and three gating projections.
    3. Optional pair mask is applied.
    4. Fuse elementwise gate multiplication for left/right with a Triton kernel.
    5. Batched matrix multiplication ``left @ rightᵀ`` for every head
       using torch.bmm (FP16, FP32 accumulation).
    6. Output LayerNorm, output gate, and final linear projection back to ``dim``.
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    import math

    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1. LayerNorm over last dim (float32)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # 2. Linear projections (float32)
    left = torch.nn.functional.linear(
        x_norm, weights["left_proj.weight"]
    )  # [B, N, N, H]
    right = torch.nn.functional.linear(
        x_norm, weights["right_proj.weight"]
    )
    left_gate = torch.nn.functional.linear(
        x_norm, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm, weights["out_gate.weight"]
    ).sigmoid()

    # 3. Apply optional mask (broadcast over hidden dim)
    if mask is not None:
        # mask may be bool or float; ensure float32
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 4. Cast to fp16 and fuse gate multiplication
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()
    out_gate_fp16 = out_gate.to(torch.float16).contiguous()

    left_out = torch.empty_like(left_fp16)
    right_out = torch.empty_like(right_fp16)

    total_elements = left_fp16.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    fuse_gate_kernel[grid](
        left_fp16,
        left_gate_fp16,
        right_fp16,
        right_gate_fp16,
        left_out,
        right_out,
        total_elements,
        BLOCK=BLOCK,
    )

    # 5. Batched GEMM: left_out @ right_outᵀ for each head
    B, N, _, H = left_out.shape
    # Permute to [B, H, N, N]
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()

    # Flatten batch and head dimension
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Matrix multiply (FP16 input, FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] fp16

    # 6. Output layer‑norm, output gate, final linear projection
    # Cast norm weights/bias to fp16
    to_out_norm_weight_fp16 = weights["to_out_norm.weight"].to(out.dtype)
    to_out_norm_bias_fp16 = weights["to_out_norm.bias"].to(out.dtype)

    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight_fp16,
        bias=to_out_norm_bias_fp16,
        eps=eps,
    )

    out_norm = out_norm * out_gate_fp16

    # Final linear projection back to dim
    to_out_weight_fp16 = weights["to_out.weight"].to(out.dtype)
    output_fp16 = torch.nn.functional.linear(out_norm, to_out_weight_fp16)

    # Return float32 tensor
    return output_fp16.to(torch.float32)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch module:

1. Layer‑norm the input over the last dimension.
2. Linear projections (left/right) and three gating projections.
3. Optional pair‑mask is applied (broadcast over the hidden dimension).
4. A tiny Triton kernel fuses the elementwise gate multiplication
   `left = left * left_gate` and `right = right * right_gate`.
5. Batched matrix multiplication `left @ rightᵀ` is performed with
   `torch.bmm` on FP16 tensors (FP32 accumulation).
6. The result is reshaped, a second Layer‑norm and the output gate are
   applied, and a final linear projection maps back to `dim`.
The function returns a float32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def fuse_gate_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    size,
    BLOCK: tl.constexpr,
):
    """
    Element‑wise fusion:
        left_out  = left  * left_gate
        right_out = right * right_gate
    All tensors are contiguous FP16 vectors of length `size`.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    left_out  = left * left_gate
    right_out = right * right_gate

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with all linear / layer‑norm parameters
            - config: dict containing at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, same device)
    """
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------ #
    # 1. Layer‑norm over the channel dimension (float32)
    # ------------------------------------------------------------------ #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------ #
    # 2. Linear projections and gating (float32)
    # ------------------------------------------------------------------ #
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------ #
    # 3. Apply optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------ #
    if mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 4. Fuse gate multiplication with a Triton kernel (FP16)
    # ------------------------------------------------------------------ #
    # Cast tensors to FP16 and make them contiguous
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    left_out = torch.empty_like(left_fp16)
    right_out = torch.empty_like(right_fp16)

    total_elem = left_fp16.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elem, BLOCK),)

    fuse_gate_kernel[grid](
        left_fp16,
        left_gate_fp16,
        right_fp16,
        right_gate_fp16,
        left_out,
        right_out,
        total_elem,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------ #
    # 5. Batched GEMM: left_out @ right_outᵀ  (FP16 input, FP32 acc)
    # ------------------------------------------------------------------ #
    B, N, _, H = left_out.shape                # H == hidden_dim
    # reshape to [B, H, N, N] for batched matmul
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)                # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # matrix multiply with transpose on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, output gate, final linear projection (float32)
    # ------------------------------------------------------------------ #
    out_fp32 = out.to(torch.float32)                # cast for high‑precision LN
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate                  # both float32
    output = torch.nn.functional.linear(
        out_norm,
        weights["to_out.weight"],
    )                                               # [B, N, N, dim] float32

    return output
