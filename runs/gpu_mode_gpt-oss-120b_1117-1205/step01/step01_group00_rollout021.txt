def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # cast to fp16
    x_fp16 = x.to(torch.float16)

    # linear proj
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # gates
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # prepare for batched matmul: permute
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # batched matmul using torch.matmul (fp16)
    out_flat = torch.matmul(left_flat, right_flat)   # (B*H, N, N)

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # to_out_norm (float32)
    out_norm = F.layer_norm(
        out.to(torch.float32),
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # cast to fp16 for multiplication
    out_norm_fp16 = out_norm.to(torch.float16).contiguous()
    out_gate_fp16 = out_gate.to(torch.float16).contiguous()

    # elementwise multiplication using Triton kernel
    size = out_norm_fp16.numel()
    BLOCK = 65536  # can be tuned
    grid = (triton.cdiv(size, BLOCK),)
    elementwise_mul_kernel[grid](
        out_norm_fp16,
        out_gate_fp16,
        size,
        BLOCK=BLOCK,
    )

    # final linear projection
    out_fp16 = F.linear(out_norm_fp16, weights["to_out.weight"].to(torch.float16))
    out = out_fp16.to(torch.float32)
    return out
```

"""
TriMul (outgoing) forward pass with a fused Triton element‑wise kernel.

Algorithm
---------
1. Layer‑normalize the input tensor over the channel dimension (float32).
2. Cast to float16 and compute the left/right linear projections.
3. Optionally apply a binary mask (broadcast over the hidden dimension).
4. Compute left/right gates (float32 → float16) and multiply them into the
   projected tensors.
5. Perform the heavy N³ contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication of shape
   (B·H, N, N) @ (B·H, N, N)^T.  This is executed with torch.matmul on
   float16 tensors (leveraging cuBLAS/TensorCores).
6. Apply a second LayerNorm over the hidden dimension (float32).
7. Multiply the normalized result with the output gate using a small
   Triton kernel that fuses the element‑wise product.
8. Final linear projection back to the original channel dimension
   (float16 → float32 output).

The Triton kernel is only used for the final element‑wise multiplication,
ensuring at least one custom kernel is exercised while the dominant N³
computation benefits from highly‑optimized cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication (fp16)
# ----------------------------------------------------------------------
@triton.jit
def mul_inplace_kernel(out_ptr, mul_ptr, size, BLOCK: tl.constexpr):
    """
    out_ptr[0:size] *= mul_ptr[0:size]   (in‑place)

    Arguments
    ---------
    out_ptr : pointer to the output tensor (float16)
    mul_ptr : pointer to the multiplier tensor (float16)
    size    : total number of elements
    BLOCK   : compile‑time constant, number of elements processed per program
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    a = tl.load(out_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(mul_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool or numeric)
        - weights_dict : mapping of parameter names to torch.Tensor
        - config_dict  : must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and compute linear projections
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gates (float32 computation → cast to fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_w).sigmoid()      # float32
    right_gate = F.linear(x, right_gate_w).sigmoid()    # float32
    out_gate = F.linear(x, out_gate_w).sigmoid()        # float32

    # Cast gates to fp16 for the element‑wise products
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)
    out_gate_fp16 = out_gate.to(torch.float16)

    # Apply gates to the projected tensors
    left = left * left_gate_fp16
    right = right * right_gate_fp16

    # ------------------------------------------------------------------
    # 5. Core contraction (batched GEMM) – fp16 matmul
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) matrices
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication, fp16 → fp16
    out_flat = torch.matmul(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out.to(torch.float32),
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # (B, N, N, H) float32

    # ------------------------------------------------------------------
    # 7. Multiply with out_gate using Triton kernel (fp16)
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel
    out_norm_fp16 = out_norm.to(torch.float16).contiguous()
    out_gate_fp16 = out_gate_fp16.contiguous()

    total_elems = out_norm_fp16.numel()
    BLOCK_SIZE = 1024  # 32‑warp friendly
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    mul_inplace_kernel[grid](
        out_norm_fp16,
        out_gate_fp16,
        total_elems,
        BLOCK=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out_fp16 = F.linear(out_norm_fp16, to_out_w)  # (B, N, N, dim) fp16
    out_final = out_fp16.to(torch.float32)

    return out_final
