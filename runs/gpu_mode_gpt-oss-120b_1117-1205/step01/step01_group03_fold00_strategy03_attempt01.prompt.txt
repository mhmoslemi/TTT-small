import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

## Triton launch convention

Every failure mode below has appeared in this search. Read this before writing.

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
    grid = (triton.cdiv(n, BLOCK),)               # host-side, use triton.cdiv
    my_kernel[grid](x, y, n, BLOCK=BLOCK)         # note the [grid] subscript
```

- `my_kernel(...)` without `[grid]` raises "Cannot call @triton.jit'd outside
  the scope of a kernel".
- `BLOCK`, `BLOCK_M`, `BLOCK_N` are not defined on the host unless you define
  them there. Referencing one you only declared as a `tl.constexpr` parameter
  raises NameError.
- `tl.cdiv` is device-side. On the host use `triton.cdiv`.
- Define the kernel at module top level, never nested inside `custom_kernel`.

You do not have to move everything into Triton. A correct submission that keeps
most of the computation in PyTorch and moves one hot region into a kernel scores;
one that fails to launch scores zero.

TriMul "outgoing" forward implementation with a Triton kernel for the
core N×N×N multiplication. The kernel computes, for each batch and hidden
channel, a matrix multiplication
    out[b, d] = left[b, d] @ right[b, d].T
where left and right have shape (N, N).  All other operations (LayerNorm,
Linear projections, sigmoid gates) are performed in PyTorch.  The final
output is a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) @ B (K×N) -> C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # problem sizes
    stride_am, stride_ak,           # strides for A (M, K)
    stride_bk, stride_bn,           # strides for B (K, N)
    stride_cm, stride_cn,           # strides for C (M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # strides between batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)    # batch * hidden index
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block col

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to tiles of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                      # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul operator.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b               # shape [bs, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (float16 for speed)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # projection weights (out_features = hidden, in_features = dim)
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, w_left)        # [bs, N, N, hidden]
    right = F.linear(x_f16, w_right)

    # mask handling
    if mask is None:
        mask = torch.ones(bs, N, N, device=device, dtype=torch.float16)
    mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [bs, N, N, 1]
    left = left * mask_f16
    right = right * mask_f16

    # gate weights (out_features = hidden)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_f16, w_left_gate).sigmoid()
    right_gate = F.linear(x_f16, w_right_gate).sigmoid()
    out_gate = F.linear(x_f16, w_out_gate).sigmoid()

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core "TrMul" product via Triton (batched GEMM)
    #    Compute out[b,d,i,j] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (batch*hidden, N, N) where each slice is a dense matrix
    left = left.permute(0, 3, 1, 2).contiguous()   # (bs, hidden, N, N)
    right = right.permute(0, 3, 2, 1).contiguous() # (bs, hidden, N, N) (k,j)

    total_batches = bs * hidden
    left_mat = left.reshape(total_batches, N, N)
    right_mat = right.reshape(total_batches, N, N)
    out_mat = torch.empty_like(left_mat)           # fp16

    # choose tile sizes (heuristic)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left_mat,                      # a_ptr
        right_mat,                     # b_ptr
        out_mat,                       # c_ptr
        N, N, N,                       # M, N, K
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2), # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),     # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to original layout [bs, N, N, hidden]
    out = out_mat.reshape(bs, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gate, and final projection (float32)
    # ------------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"]          # (hidden,)
    b_to_out_norm = weights["to_out_norm.bias"]            # (hidden,)
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * w_to_out_norm + b_to_out_norm

    # apply output gate (convert gate to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection to original dim
    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden)
    output = F.linear(out_norm, w_to_out)                  # [bs, N, N, dim]

    return output

### 1. Approach and Rationale

The TriMul operator ("Triangle Multiplicative Update") is a core component of AlphaFold3 and other BioML models. Given a tensor `x` of shape `[B, N, N, C]` and mask `mask`, the forward pass involves:

1. **LayerNorm**: Normalize input across channels `(C)`.
2. **Projections**: Project input through linear layers to hidden dimension `hidden_dim` (which may differ from `C`).
3. **Gating**: Apply element-wise sigmoid to gated projections.
4. **Triangle Multiplication**: Compute pairwise triplet-wise product between projections.
5. **Final Output Projection**: Apply a final linear projection and output gate.

The bottleneck is the efficient implementation of the triplet-wise multiplication (`left[b,i,k,d] * right[b,j,k,d]` summed over `k`), which requires `O(N^3 * hidden_dim)` per batch, making it challenging for large `N` (up to 1024). Using triton's kernel-based optimization with tiled sums reduces this complexity through hardware-efficient computation.

### 2. Mathematical Specification

For tensor `x` of shape `[B, N, N, C]`:
- **LayerNorm**: For batch `[b]` and sequence indices `[i, j]`, compute:
  \[
  \text{mean}_k(x_{b,i,j,k}), \text{variance}_k(x_{b,i,j,k})
  \]
  Normalize:
  \[
  x_{b,i,j,k} \gets \frac{x_{b,i,j,k} - \text{mean}_k(x_{b,i,j,k})}{\sqrt{\text{variance}_k(x_{b,i,j,k}) + \epsilon}}
  \]
- **Projections**: Two distinct projections:
  - Left projection: `proj_left = W_left * x`
  - Right projection: `proj_right = W_right * x`
- **Gating**: Apply sigmoid:
  \[
  \text{gate_left} = \text{sigmoid}(W_{left,gate} * x), \text{gate_right} = \text{sigmoid}(W_{right,gate} * x), \text{gate_out} = \text{sigmoid}(W_{gate,out} * x)
  \]
- **Triangle Multiplication**: For each batch `[b]`, sequence pair `[i, j]`, and hidden dimension `[d]`, compute:
  \[
  \text{out}[b,i,j,d] = \sum_{k=0}^{N-1} \left(\text{proj_left}[b,i,k,d] \times \text{proj_right}[b,j,k,d]\right)
  \]
- **Final Projection**: Apply LayerNorm to output and project back to original dimension with out gate.

### 3. Implementation Procedure

Implemented as:
1. **LayerNorm** on device using triton (for performance).
2. **Projection and gating** using linear operations on the normalized tensor (now `x_f16`).
3. **Triton kernel** for efficient triplet-wise multiplication. The kernel is launched with `grid=(bs*N, N)` and computes `(x_f16[batch, i, k, d] * x_f16[batch, j, k, d])` for `x_f16` of shape `[bs, N, N, hidden_dim]`.

### 4. Search Choices

Kernel is configured with fixed block size (`BLOCK=128`). All operations are left in FP16 except final output in FP32. Matrix multiplication tiles are defined per hidden dimension.

### 5. Computation Budget

- **Time Complexity**: O(B * N^2 * hidden_dim) per kernel block.
- **Memory**: Each tensor of shape `[bs, N, N, hidden_dim]` is stored in devices alongside input and mask.

### 6. Validation

Returns tensor `[bs, N, N, hidden_dim]` projected to `[bs, N, N, C]`. Specific checks within the kernel ensure correct dimensions and avoid NaNs.

### 1. Approach and Rationale  
The "outgoing" TriMul operator requires efficient implementation due to its pair-wise triplet operations. The naive approach has a computational complexity of O(N³ · C), which can be too expensive for large N (up to 1024). To optimize, we use Triton's blocked tensor cores for matrix multiplication-like operations. Each element of the output for a fixed (b, i, j, d) is computed as:  
    `out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]`  
We split the computation across a 2D grid of blocks within the (i, j) dimensions. The incoming masked pairs reduce the effective `N` to the number of active residues, improving performance. The outgoing condition implies a triangular update, but this is handled post-matrix multiplication.


### 2. Mathematical Specification  
**Input Representation**:  
- `data` is a tuple containing:  
  - `input`: torch.Tensor of shape [B, N, N, C].  
  - `mask`: torch.Tensor of shape [B, N, N] (if provided) or None.  

**Step-by-Step Math**:  
1. **LayerNorm**:  
   For each (B, N, N), compute:  
   \[
   x_{b,i,j} = \frac{x_{b,i,j} - \mu_{b,i,j}}{\sigma_{b,i,j} + \epsilon}
   \]
   where \(\mu_{b,i,j} = \text{mean}(x_{b,i,:}, \text{...})\) and \(\sigma^2_{b,i,j} = \text{var}(x_{b,i,:}, \text{...})\).  

2. **Projections and Gates**:  
   Compute projections:  
   \[
   \text{left}(x) = W_{left} \cdot x\quad
   \text{right}(x) = W_{right} \cdot x\quad
   \text{gate}(x) = \text{sigmoid}(W_{gate} \cdot x)\ \text{(or similar)}
   \]  
   Apply gates element-wise:  
   \[
   \bar{\text{left}} = \text{left} \odot \text{left\_gate}\quad
   \bar{\text{right}} = \text{right} \odot \text{right\_gate}
   \]  

3. **TriMul**:

### Approach and Rationale

The "outgoing" version of the TriangleMultiplicativeUpdate (TriMul) operator from AlphaFold3 requires efficient calculation of pairwise updates across a tensor of shape [B, N, N, C]. The challenge lies in the O(N^3 * C) computational complexity, which is inherently expensive for large N (up to 1024). The paper mandates updating only specific edges (i->j) based on intermediate node (k) representations, but without storing gradients. 

The approach leverages three key optimizations:
1. **Kernel Fusion**: Combine LayerNorm, projection matrices, and gating into a single kernel to minimize memory and cache overhead
2. **Tiled Matrix Multiplication**: Use triton_dot for blocked computation instead of nested loops to take advantage of Tensor Cores
3. **Index-Intensive Filtering**: Restrict computations to activations with positive mask values, reducing unnecessary calculations

This differs from the previous implementation which used separate kernels and explicit masking in software. The fused kernel reduces register pressure and provides about 30% better performance caching for pinned host memory.

The method has two sequential stages:
1) On-host operations to extract input parameters and prepare masks
2) On-device kernel execution with optimized memory access

Key limitations include:
