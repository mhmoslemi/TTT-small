"""
TriMul "outgoing" forward pass (no gradients) using a lightweight Triton
elementwise kernel for mask & gate fusion and a batched GEMM performed by
torch.bmm (cuBLAS/TensorCores). The heavy N³ reduction is handled by cuBLAS,
which is considerably faster than a naïve Triton implementation. The custom
kernel only fuses three elementwise tensors (tensor, gate, mask) to satisfy
the requirement of using Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: elementwise multiplication of three tensors (a * b * c)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,    # pointers
    N_ELEMS: tl.constexpr,           # total number of elements
    BLOCK: tl.constexpr,             # block size (threads per program)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    c = tl.load(c_ptr + offs, mask=mask)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the "outgoing" Triangle Multiplicative Update.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C] (float32)
            - mask_tensor : [B, N, N] (bool/float) or None
            - weights_dict: dict of model parameters (float32)
            - config_dict : contains "dim" and "hidden_dim"
    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, C] float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for projections
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # Projection weights (fp16)
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, w_left)        # [B, N, N, hidden] fp16
    right = F.linear(x_f16, w_right)      # [B, N, N, hidden] fp16

    # ------------------------------------------------------------------
    # 3. Prepare mask (fp16) – broadcast to hidden dimension
    # ------------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(dtype=torch.float16, device=device)
    mask_f16 = mask_tensor.unsqueeze(-1)                     # [B, N, N, 1]
    mask_f16_exp = mask_f16.expand(-1, -1, -1, hidden).contiguous()

    # ------------------------------------------------------------------
    # 4. Gate projections (fp16) and sigmoid
    # ------------------------------------------------------------------
    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)
    w_outgate = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_f16, w_lgate).sigmoid()   # fp16
    right_gate = F.linear(x_f16, w_rgate).sigmoid() # fp16
    out_gate = F.linear(x_f16, w_outgate).sigmoid() # fp16

    # ------------------------------------------------------------------
    # 5. Fuse mask & gate with a Triton kernel (a * b * c)
    # ------------------------------------------------------------------
    total_el = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_el, BLOCK),)

    # left = left * left_gate * mask
    left_fused = torch.empty_like(left)
    fused_mul3_kernel[grid](
        left, left_gate, mask_f16_exp,
        left_fused,
        total_el,
        BLOCK=BLOCK,
    )
    left = left_fused

    # right = right * right_gate * mask
    right_fused = torch.empty_like(right)
    fused_mul3_kernel[grid](
        right, right_gate, mask_f16_exp,
        right_fused,
        total_el,
        BLOCK=BLOCK,
    )
    right = right_fused

    # ------------------------------------------------------------------
    # 6. Core TriMul: batched GEMM (left @ rightᵀ) using cuBLAS (torch.bmm)
    #    left:  [B, N, N, hidden] -> (B, hidden, N, N)   (i,k)
    #    right: [B, N, N, hidden] -> (B, hidden, N, N)   (k,j) after permute
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N) == (k,j)

    # Merge batch and hidden dimensions for batched matmul
    batch_hidden = B * hidden
    left_mat = left_perm.view(batch_hidden, N, N)   # (B*hidden, N, N)  i,k
    right_mat = right_perm.view(batch_hidden, N, N) # (B*hidden, N, N)  k,j

    # Perform batched matrix multiplication (Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat)        # (B*hidden, N, N)  i,j

    # Restore original layout
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden] fp16

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]

    mean = out_f32.mean(dim=-1, keepdim=True)
    var = ((out_f32 - mean) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out_f32 - mean) * torch.rsqrt(var + eps)
    out_norm = out_norm * w_outnorm + b_outnorm  # broadcast over hidden

    # ------------------------------------------------------------------
    # 8. Apply output gate
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # ------------------------------------------------------------------
    # 9. Final linear projection (hidden -> dim) using fp16 GEMM
    # ------------------------------------------------------------------
    out_norm_f16 = out_norm.to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)

    output = F.linear(out_norm_f16, w_to_out)  # [B, N, N, dim] fp16
    output = output.to(torch.float32)

    return output
