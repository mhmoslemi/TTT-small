@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to tiles
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    num_k_iter = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k_iter):
        # load
        a = tl.load(a_ptrs,
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0)
        b = tl.load(b_ptrs,
                mask=mask_k[:, None] & mask_n[None, :],
                other=0.0)
        # dot
        acc += tl.dot(a, b)  # using K dimension automatically
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
            mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a Triton kernel.

Algorithm:
1. Layer‑norm the input over the last dimension (dim).
2. Linear projections (dim → hidden) for left/right and three gates.
3. Apply the pairwise mask (if provided) to left/right.
4. Fuse the gates: left *= left_gate, right *= right_gate.
5. Core O(N³) operation:
   out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   which is a batched matrix multiplication left @ rightᵀ.
   Implemented with a Triton GEMM kernel that processes (B·hidden) batches.
6. Layer‑norm over the hidden dimension, multiply by out_gate, and a final
   linear projection back to dim.
All heavy computation (step 5) is performed in FP16 on Tensor‑Cores,
while reductions (layer‑norm) are done in FP32 for stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # matrix sizes
    stride_am, stride_ak,                # strides for A (M, K)
    stride_bk, stride_bn,                # strides for B (K, N)  (B stored as N×K)
    stride_cm, stride_cn,                # strides for C (M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # strides between batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  with A[M,K] and B[N,K] (i.e. sum over K)."""
    pid_batch = tl.program_id(0)   # batch × hidden index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to the tiles of A and B (B is accessed as K×N)
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)                             # [B, N, N, dim] (fp32)

    # ------------------------------------------------------------
    # 2. Projections & gates (cast to fp16 for speed)
    # ------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # linear projections (dim -> hidden)
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, w_left)          # [B, N, N, hidden] fp16
    right = F.linear(x_f16, w_right)

    # optional mask
    if mask_tensor is None or config.get("nomask", True):
        mask = torch.ones(B, N, N, device=device, dtype=torch.float16)
    else:
        mask = mask_tensor.to(torch.float16)
    mask = mask.unsqueeze(-1)                # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # gating (dim -> hidden) and sigmoid
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_f16, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_f16, w_out_gate))

    # apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 3. Core TriMul via Triton (batched GEMM left @ rightᵀ)
    # ------------------------------------------------------------
    # reshape to (B*hidden, N, N) where each slice is a matrix
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    total_batches = B * hidden
    left_mat = left.view(total_batches, N, N)      # [B*H, N, N]
    right_mat = right.view(total_batches, N, N)    # [B*H, N, N]
    out_mat = torch.empty_like(left_mat)           # fp16 accumulator

    # choose block sizes heuristically
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_mat,                           # a_ptr
        right_mat,                          # b_ptr
        out_mat,                            # c_ptr
        N, N, N,                            # M, N, K
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        right_mat.stride(2), right_mat.stride(1), # stride_bk, stride_bn (B stored as N×K)
        out_mat.stride(1), out_mat.stride(2),     # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden] and cast to float32 for the next norm
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,hidden]

    # ------------------------------------------------------------
    # 4. Output LayerNorm, out_gate gating and final projection
    # ------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=w_to_out_norm,
                       bias=b_to_out_norm,
                       eps=eps)                     # [B,N,N,hidden] fp32

    # apply the output gate (cast gate to fp32)
    out = out * out_gate.to(torch.float32)

    # final linear projection hidden → dim
    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden)
    output = F.linear(out, w_to_out)                        # [B, N, N, dim]

    return output
