Let's define kernel as:

```
@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    x = tl.load(inp_ptr + inp_base + offs_h * stride_in_h, mask=mask_h, other=0.0)

    # Compute mean
    mean = tl.sum(x) / H

    # Compute variance
    diff = x - mean
    var = tl.sum(diff * diff) / H

    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)

    out = norm * w + b_

    tl.store(out_ptr + out_base + offs_h * stride_out_h, out, mask=mask_h)
```

"""
TriMul (outgoing) forward pass with Triton-accelerated hidden‑dim LayerNorm
and a batched‑matrix‑multiplication (BMM) based contraction.

Algorithm
---------
1. Layer‑norm over the input channel dimension (dim) – PyTorch.
2. Linear projections (left/right) and gates (left, right, out) – PyTorch.
3. Apply optional pairwise mask and gates.
4. For each hidden channel h, compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication:
       left(b,h)   : N×N matrix M  (M_{i,k}=left[b,i,k,h])
       right(b,h)  : N×N matrix N  (N_{j,k}=right[b,j,k,h])
       out(b,h) = M @ Nᵀ
   The contraction is performed in bfloat16 using torch.matmul for
   high throughput.
5. Hidden‑dim LayerNorm (over `hidden_dim`) via a custom Triton kernel.
6. Multiply by the out‑gate, then a final linear projection back to `dim`.

Only the LayerNorm over the hidden dimension is written in Triton; all
other heavy operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,               # *float32  [B, N, N, H]
    out_ptr,               # *float32  [B, N, N, H]
    weight_ptr,            # *float32  [H]
    bias_ptr,              # *float32  [H]
    B,                     # tl.int32
    N,                     # tl.int32
    H,                     # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # tl.int32
    eps: tl.constexpr,                                   # const float
    BLOCK_H: tl.constexpr                                 # compile‑time block size (≥ H)
):
    """Layer‑norm over the last dimension H for a 4‑D tensor."""
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base pointer for the (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # load H‑dimensional vector
    x = tl.load(inp_ptr + inp_base + offs_h * stride_in_h,
                mask=mask_h,
                other=0.0)

    # mean over H
    mean = tl.sum(x) / H

    # variance over H
    diff = x - mean
    var = tl.sum(diff * diff) / H

    # normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # apply learned scale and bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # store result
    tl.store(out_ptr + out_base + offs_h * stride_out_h,
             out,
             mask=mask_h)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------------
    # 1. Input LayerNorm over the last channel dimension "dim"
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x = x.contiguous()
    B, N, _, _ = x.shape

    # ---------------------------------------------------------------
    # 2. Linear projections & gates (all in float32)
    x_flat = x.view(-1, dim)                      # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ---------------------------------------------------------------
    # 3. Apply mask (if any) and gates
    if not nomask and mask is not None:
        # mask shape: [B, N, N] -> broadcast to [B, N, N, 1]
        mask_exp = mask.to(dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 4. Pairwise contraction over sequence index k using BMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Compute as left(b,h) @ right(b,h).T  for each hidden channel.
    left_b = left.to(torch.bfloat16).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_b = right.to(torch.bfloat16).permute(0, 3, 1, 2).contiguous()

    # batched matrix multiplication (M @ Nᵀ) for each (b,h)
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))            # [B, H, N, N]

    # reshape back to [B, N, N, H] and cast to float32
    out = out_b.permute(0, 2, 3, 1).contiguous().to(dtype)            # [B, N, N, H]

    # ---------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)
    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (total_programs,)

    BLOCK_H = 128  # hidden_dim ≤ 128 in all test configurations

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B, N, hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ---------------------------------------------------------------
    # 6. Multiply by outgoing gate
    out_scaled = out_norm * out_gate

    # ---------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)               # [B*N*N, H]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
