@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          M, N, K,
                          stride_am, stride_ak,
                          stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          batch_stride_a, batch_stride_b, batch_stride_c,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to start of the batch:
    a_batch_ptr = a_ptr + batch_id * batch_stride_a
    b_batch_ptr = b_ptr + batch_id * batch_stride_b
    c_batch_ptr = c_ptr + batch_id * batch_stride_c

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (offs_k < K)),
                    other=0.0)
        # Load B tile
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=((offs_k < K)[:, None] & mask_n[None, :]),
                    other=0.0)
        # Dot: a shape (BLOCK_M,BLOCK_K), b shape (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # Write out
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The heavy N³ contraction:
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
is implemented as a batched matrix multiplication
    (B·D, N, N) @ (B·D, N, N)ᵀ
using a high‑performance Triton kernel (FP16 inputs, FP32 accumulator,
FP16 output).  All other operations (LayerNorm, linear projections,
gating, masking) are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where A: (batch, M, K), B: (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Rank‑K update
        acc += tl.dot(a, b)

    # Write back the tile
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B
    A: (batch, M, K)    FP16, contiguous
    B: (batch, K, N)    FP16, contiguous
    Returns C: (batch, M, N) FP16
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tuning parameters (good for FP16 Tensor‑Core GEMM on modern GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool/int) – ignored if config["nomask"] is True
            - weights_dict: mapping parameter name → torch.Tensor
            - config_dict: must contain "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load parameters (convert to appropriate dtype)
    # ------------------------------------------------------------------
    # LayerNorm over input channels
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projections
    left_proj_w = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, non_blocking=True)

    # Gating weights
    left_gate_w = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    # Output layers
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) – keep high precision for statistics
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16) – no bias
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, left_proj_w)          # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Compute gates (FP16) and apply
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # shape [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange tensors so the batched GEMM can be applied:
    #   left : (B, H, N, N)  where left[b,d,i,k]
    #   right: (B, H, N, N) where right[b,d,k,j] (i.e. transposed on last two dims)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch & channel dimensions for the batched GEMM
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Compute C = left_flat @ right_flat   (both FP16)
    out_flat = triton_matmul(left_flat, right_flat)           # (B*H, N, N) FP16

    # Reshape back to (B, N, N, H) and promote to FP32 for the remaining ops
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w)   # final projection back to dim (bias=False)

    return out
