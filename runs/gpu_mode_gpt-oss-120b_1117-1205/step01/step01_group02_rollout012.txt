import torch
import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(...):

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # LayerNorm on input
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm_fp16 = x_norm.to(torch.float16)

    # Projections
    left = torch.nn.functional.linear(x_norm_fp16, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_norm_fp16, weights["right_proj.weight"].to(torch.float16))

    # Gating
    left_gate = torch.nn.functional.linear(x_norm_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Optional mask
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for GEMM: reshape to (B, hidden_dim, N, N)
    B, N, _, _ = left.shape
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # right_T = right_perm.transpose(2,3) (swap last two dims). The right_perm shape [B, hidden_dim, N, N]; we want (K,N) shape, so we need to transpose last two dims: right_T = right_perm.transpose(-2,-1).contiguous()

    right_T = right_perm.transpose(-2, -1).contiguous()

    # flatten batch and head dims
    total_batch = B * hidden_dim
    left_flat = left_perm.view(total_batch, N, N)   # shape [batch*head, N, N]
    right_flat = right_T.view(total_batch, N, N)   # shape [batch*head, N, N]
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Get strides
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # Set block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # typical

    # Launch grid
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )
    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,  # M
        N,  # N
        N,  # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back to original format: [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # apply out_gate
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32

    # Final linear
    output = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float32))

    return output

"""TriMul (outgoing) forward pass with a fused Triton GEMM kernel.

The heavy part of TriMul is a batched matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batch of GEMMs left @ rightᵀ for each hidden
dimension d. This implementation:
  1. Layer‑norms and gated linear projections are computed in PyTorch.
  2. left and right are reshaped to (B·H, N, N) (H = hidden_dim) and
     multiplied by a custom Triton kernel that accumulates in FP32.
  3. The result is reshaped back, passed through the output LayerNorm,
     gated by out_gate and linearly projected to the original channel
     dimension.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers to A, B, C
    M, N, K,                            # matrix dimensions
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: M×K, B: K×N) for many independent batches."""
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch index (B·H)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # Pointers with batch stride
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Load tiles (apply masks)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back result (FP16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor (torch.Tensor): [B, N, N, dim], float32, on CUDA
            - mask (torch.Tensor or None): [B, N, N], bool/float, optional
            - weights (dict): model parameters (see function body)
            - config (dict): contains "dim" and "hidden_dim"
    Returns:
        torch.Tensor: [B, N, N, dim], float32 on the same device
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm_fp16 = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_norm_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = torch.nn.functional.linear(
        x_norm_fp16, weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3. Gating (fp16)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Prepare for batched GEMM: reshape to (B·H, N, N)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden dimension to second axis and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose right so that we can compute A @ B (instead of A @ Bᵀ)
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    total_batch = B * hidden_dim
    left_flat = left_perm.view(total_batch, N, N)   # [B·H, N, N]
    right_flat = right_T.view(total_batch, N, N)    # [B·H, N, N]
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 7. Strides (in elements)
    # -----------------------------------------------------------------
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # -----------------------------------------------------------------
    # 8. Launch Triton GEMM kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )
    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 9. Reshape back to [B, N, N, H] and convert to float32
    # -----------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)   # [B, H, N, N]
        .permute(0, 2, 3, 1)                # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # -----------------------------------------------------------------
    # 10. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)  # apply output gate

    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # [B, N, N, dim]

    return output
