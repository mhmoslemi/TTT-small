def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # load parameters
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5

    # weights dict access:
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]; left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]; left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]; to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]; to_out_weight = weights["to_out.weight"]

    # 1) LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # cast to fp16
    x_norm_f16 = x_norm.to(torch.float16)

    # 2) Linear projections
    left = torch.nn.functional.linear(x_norm_f16, left_proj_weight.to(torch.float16))
    right = torch.nn.functional.linear(x_norm_f16, right_proj_weight.to(torch.float16))

    # gates
    left_gate = torch.nn.functional.linear(x_norm_f16, left_gate_weight.to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_f16, right_gate_weight.to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_f16, out_gate_weight.to(torch.float16)).sigmoid()

    # 3) Apply mask if exists
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1) # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 4) Apply elementwise gates
    left = left * left_gate
    right = right * right_gate

    # 5) Prepare for GEMM: permute to (B, H, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_T.view(total_batch, N, N)
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # load strides (in elements)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # 6) Triton kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,  # M
        N,  # N
        N,  # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 7) Reshape back to (B,N,N,H)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)  # [B,N,N,H]
        .contiguous()
        .to(torch.float32)
    )

    # 8) Output LN and gate
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)

    # 9) Final linear
    output = torch.nn.functional.linear(
        out,
        to_out_weight.to(torch.float32),
        bias=None,
    )
    return output

"""
TriMul (outgoing) forward pass with a fused Triton GEMM kernel.

Algorithm
---------
1. Layer‑norm the input tensor over its channel dimension.
2. Compute left/right projections and their sigmoid gates (half‑precision).
3. Apply optional pairwise mask, then element‑wise gates.
4. Fuse the O(N³) “left @ rightᵀ” operation with a custom Triton kernel.
   The kernel computes a batched GEMM for each (batch, hidden) pair:
        out[b·h, i, j] = Σₖ left[b·h, i, k] * rightᵀ[b·h, k, j]
5. Reshape the result back to [B, N, N, hidden_dim] and cast to fp32.
6. Apply the output LayerNorm, output gate and final linear projection.
The implementation keeps the heavy N³ work on the GPU in fp16 with
float‑32 accumulation for numerical stability.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ B) where
#   A : [total_batch, M, K]   (fp16)
#   B : [total_batch, K, N]   (fp16)
#   C : [total_batch, M, N]   (fp16)
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # tensor pointers
    M, N, K,                        # matrix dimensions (int)
    stride_am, stride_ak,           # A strides (row, col)
    stride_bk, stride_bn,           # B strides (row, col)
    stride_cm, stride_cn,           # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_b = tl.program_id(2)  # batch index (batch * hidden)

    # -------------------------------------------------
    # Tile offsets
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------
    # Accumulator (FP32 for stability)
    # -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over K dimension in tiles
    # -------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # Pointers for the current tile (including batch offset)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Load tiles with masking (zero padding for OOB)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Store the result (cast back to fp16)
    # -------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------------- #
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensors containing all linear and LN params
            - config: dict with at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, same device)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # Load model parameters (already on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over channel dimension
    # --------------------------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # Cast to half‑precision for the heavy path
    x_norm_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Linear projections + sigmoid gates (all half‑precision)
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(
        x_norm_h, left_proj_weight.to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(
        x_norm_h, right_proj_weight.to(torch.float16)
    )

    left_gate = torch.nn.functional.linear(
        x_norm_h, left_gate_weight.to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm_h, right_gate_weight.to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm_h, out_gate_weight.to(torch.float16)
    ).sigmoid()

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Reshape for batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    # Move hidden dimension before the two sequence dims and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right so that we can compute A @ B
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    left_flat = left_perm.view(total_batch, N, N)   # [total_batch, N, N]
    right_flat = right_T.view(total_batch, N, N)   # [total_batch, N, N]
    out_flat = torch.empty(
        (total_batch, N, N), dtype=torch.float16, device=device
    )  # result buffer

    # -------------------------------------------------
    # Strides in element units (not bytes)
    # -------------------------------------------------
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # -------------------------------------------------
    # 6. Triton GEMM launch
    # -------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        total_batch,               # batch * hidden
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim] and cast to fp32
    # -------------------------------------------------
    out = (
        out_flat.view(B, H, N, N)       # [B, H, N, N]
        .permute(0, 2, 3, 1)            # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # -------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)   # apply output gate

    output = torch.nn.functional.linear(
        out,
        to_out_weight.to(torch.float32),
        bias=None,
    )  # [B, N, N, dim]

    return output
