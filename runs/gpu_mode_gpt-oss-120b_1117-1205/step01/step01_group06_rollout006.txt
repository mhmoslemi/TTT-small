def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expecting float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    B, N, _, _ = x.shape

    # flatten for linear projections
    x_flat = x.view(-1, dim)

    # linear projections
    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # broadcast
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # Cast to bfloat16 for matmul
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Reshape for batched GEMM
    # left: (B,N,N,H) -> (B*H,N,N)
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    # flatten batches
    left_reshape = left_perm.view(B * hidden_dim, N, N)  # (B*H,N,N)
    right_reshape = right_perm.view(B * hidden_dim, N, N)  # (B*H,N,N)

    # compute out
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))  # (B*H,N,N)

    # reshape back
    out_bf = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    out = out_bf.to(dtype).contiguous()

    # 5. Hidden-dim layernorm via Triton
    out_norm = torch.empty_like(out)

    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (total_programs,)
    BLOCK_H = hidden_dim  # <= 128

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # Apply out_gate
    out_scaled = out_norm * out_gate

    # Final linear projection
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Input LayerNorm over the last dimension (dim).
2. Linear projections:
       left  = Linear(input, left_proj.weight)          # shape (B,N,N,H)
       right = Linear(input, right_proj.weight)
   Gates (sigmoid):
       left_gate  = Linear(input, left_gate.weight).sigmoid()
       right_gate = Linear(input, right_gate.weight).sigmoid()
       out_gate   = Linear(input, out_gate.weight).sigmoid()
3. Apply optional pair‑wise mask and gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
4. Contraction over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (B*H matrices of size N×N) in bfloat16.
5. LayerNorm over the hidden dimension H (custom Triton kernel).
6. Multiply by out_gate (element‑wise).
7. Final Linear projection back to dim.
All heavy compute (step 4) uses cuBLAS via torch.bmm; the hidden‑dim
LayerNorm is the Triton‑accelerated component.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def hidden_layernorm_kernel(
    inp_ptr,                # *float32  [B, N, N, H]
    out_ptr,                # *float32  [B, N, N, H]
    weight_ptr,             # *float32  [H]
    bias_ptr,               # *float32  [H]
    B, N, H,                # int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # int32
    eps: tl.constexpr,                                   # float
    BLOCK_H: tl.constexpr                                # compile‑time block size ≥ H
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Base pointer for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # Load hidden vector
    x = tl.load(inp_ptr + inp_base + offs_h * stride_in_h,
                mask=mask_h, other=0.0)

    # Compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    # Compute variance
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / H

    # Normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # Apply affine parameters
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h,   mask=mask_h, other=0.0)
    out = norm * w + b_

    # Store result
    tl.store(out_ptr + out_base + offs_h * stride_out_h,
             out, mask=mask_h)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dict with all linear / layer‑norm parameters
            - config: dict containing "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (float32)
    x_flat = x.view(-1, dim)                         # (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Mask & gate application
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)      # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction over k using batched GEMM in bfloat16
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # (B, N, N, H) -> (B, H, N, N) and flatten batch dimension
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)    # (B*H, N, N)

    # Compute out = left @ right^T for each hidden slice
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out_bf = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)
    out = out_bf.to(dtype).contiguous()

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)

    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (total_programs,)

    hidden_layernorm_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,          # input strides
        s_b, s_i, s_j, s_h,          # output strides (same layout)
        eps=1e-5,
        BLOCK_H=hidden_dim,
    )

    # ------------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)    # (B*N*N, H)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
