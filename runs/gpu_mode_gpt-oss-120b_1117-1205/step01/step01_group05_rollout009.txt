    M, N, K,

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy part of TriMul is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication:
    out = left @ rightᵀ
for each (batch, hidden_dim) pair, i.e. (B·H) independent N×N matmuls.
The kernel below performs this batched GEMM in fp16 with fp32 accumulation
using Tensor‑cores.  All other operations (LayerNorms, linear projections,
gating, final Linear) are carried out in PyTorch, but the O(B·H·N³) work
is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: M×K, B: K×N) for many batches."""
    batch_id = tl.program_id(0)          # batch*head index
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (M × K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )  # fp16

        # B tile: (K × N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )  # fp16

        acc += tl.dot(a, b)   # fp16 dot, fp32 accumulator

    # write back
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim] (float32)
            mask   : torch.Tensor [B, N, N] (bool/float) or None
            weights: dict of torch.Tensors for all linear / norm parameters
            config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)

    x = input_tensor.to(torch.float32)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections + gating (all in fp16)
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    w_left = weights["left_proj.weight"].to(torch.float16)       # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)          # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # optional mask
    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    left_gate = F.linear(x_h, w_left_gate).sigmoid()   # fp16
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()    # used later

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Batched GEMM: out = left @ rightᵀ
    #    left: (B, N, N, H) -> (B*H, N, N)   rows = i, cols = k
    #    rightᵀ: (B, N, N, H) -> (B*H, N, N) rows = k, cols = j
    # ------------------------------------------------------------------
    # reshape for the kernel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch_size * hidden_dim, N, N)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(batch_size * hidden_dim, N, N)

    out_mat = torch.empty_like(left_mat)   # fp16, same device

    # strides for the kernel (batch, row, col)
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat_T.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tiling parameters – chosen for good occupancy on Hopper B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_size * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(batch_size, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_f32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 4) Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Apply output gate (cast to fp32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear back to original dim
    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)                  # [B, N, N, dim]

    return output
```
