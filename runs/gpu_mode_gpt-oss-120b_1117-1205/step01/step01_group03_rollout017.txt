def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1. LayerNorm on input: we will compute using torch.nn.functional.layer_norm for efficiency.
    # shape last dim dim
    # using weight, bias from weights dict
    # views = input_tensor reshape? Actually we need layernorm across last dim (dim)
    # shape [bs, N, N, dim] => treat as [bs * N * N, dim] for layernorm.
    # Use torch.nn.functional.layer_norm (for each "row" across last dimension)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # 2. Linear projections: left_proj = x @ w_left.T? Wait F.linear expects weight shape (out_features, in_features), weight multiplied by x (the last dimension). So left = F.linear(x, left_proj_weight)
    # Cast to half for speed
    x_f16 = x.to(torch.float16)
    left = torch.nn.functional.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # mask handling
    if mask is None:
        mask_f16 = torch.ones_like(left, dtype=torch.float16)
    else:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
    left = left * mask_f16
    right = right * mask_f16

    # Gates
    left_gate = torch.nn.functional.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # 3. Core TriMul via Triton
    # left shape [B, N, N, hidden]
    # right shape [B, N, N, hidden]

    # Permute to [B, hidden, N, N] for each.
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N = k,j)

    B, H, N, N_check = left_mat.shape
    assert N == N_check

    total_batches = B * H
    # Reshape to (total_batches, N, N)
    left_mat = left_mat.view(total_batches, N, N)
    right_mat = right_mat.view(total_batches, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16

    # Choose block sizes heuristically based on N
    # Use two sets for BLOCK_K = max(16, 32)
    # Should be constexpr for kernel
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # Launch kernel
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(1), out_mat.stride(2),
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute masks for OOB
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # compute base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # compute pointer for tiles
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    num_iters = tl.cdiv(K, BLOCK_K)
    for i in range(num_iters):
        a = tl.load(a_ptrs, mask=(mask_m[:, None] & mask_k[None, :]), other=0.0)
        b = tl.load(b_ptrs, mask=(mask_k[:, None] & mask_n[None, :]), other=0.0)
        # dot product
        acc += tl.dot(a, b)
        # move pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Layernorm across dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Cast to half
    x_h = x.to(torch.float16)

    # Projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    if mask is None:
        mask_f16 = torch.ones_like(left, dtype=torch.float16)
    else:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # shape B,N,N,1
    left = left * mask_f16
    right = right * mask_f16

    # Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # TriMul core:
    left_mat = left.permute(0,3,1,2).contiguous()
    right_mat = right.permute(0,3,2,1).contiguous()

    B, H, N, _ = left_mat.shape
    total_batches = B * H
    left_mat = left_mat.view(total_batches, N, N)
    right_mat = right_mat.view(total_batches, N, N)

    out_mat = torch.empty_like(left_mat)  # same dtype float16

    if N <= 256:
        BLOCK_M = 64; BLOCK_N = 64; BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = 128; BLOCK_N = 128; BLOCK_K = 32
    else:
        BLOCK_M = 128; BLOCK_N = 128; BLOCK_K = 64

    grid = (total_batches, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(1), out_mat.stride(2),
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]
    out = out.to(torch.float32)

    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

"""
Outgoing Triangle Multiplicative Update (TriMul) implementation.

The heavy N³ computation is performed by a custom Triton kernel that
executes a batched matrix multiplication:
    out[b, d] = left[b, d] @ right[b, d].T
where left/right have shape (N, N) for each hidden channel d.
All other operations (LayerNorm, Linear projections, sigmoid gates) are
handled in PyTorch (half precision where possible). The kernel uses
FP16 for inputs, FP32 accumulation and stores the result in FP16,
which is later cast back to FP32 for the final LayerNorm and projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM   A (M×K) @ B (K×N) -> C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # problem sizes
    stride_am, stride_ak,         # strides for A (row, col)
    stride_bk, stride_bn,         # strides for B (row, col)
    stride_cm, stride_cn,         # strides for C (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance computes a BLOCK_M×BLOCK_N tile of the output
    for a single (batch, hidden) pair.
    """
    pid_batch = tl.program_id(0)       # index over (batch * hidden)
    pid_m = tl.program_id(1)           # block row index
    pid_n = tl.program_id(2)           # block column index

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # compute pointers to the start of the tiles
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)            # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: [B, N, N, dim] (float32)
              - mask_tensor:  [B, N, N] (float32 or bool) or None
              - weights_dict: mapping of weight names to torch.Tensor
              - config_dict: contains "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]  # called hidden_dim in the paper

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections & gating (FP16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ----- Projections -------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, hidden]
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ----- Mask handling ------------------------------------------------
    if mask is None:
        mask_h = torch.ones_like(left, dtype=torch.float16)
    else:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask_h
    right = right * mask_h

    # ----- Gates --------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core TriMul via Triton (batched GEMM)
    #    out[b, d] = left[b, d] @ right[b, d].T
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, k, j)

    B, H, N, _ = left_mat.shape
    total_batches = B * H

    # Collapse batch & hidden into a single dimension for the kernel
    left_mat = left_mat.view(total_batches, N, N)
    right_mat = right_mat.view(total_batches, N, N)

    out_mat = torch.empty_like(left_mat)   # FP16 buffer for result

    # Heuristic block sizes
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat.stride(1),
        right_mat.stride(2),
        out_mat.stride(1),
        out_mat.stride(2),
        left_mat.stride(0),
        right_mat.stride(0),
        out_mat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, output gate, final linear projection (float32)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate.to(torch.float32)   # apply output gate

    # Final linear projection back to original dimensionality
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
