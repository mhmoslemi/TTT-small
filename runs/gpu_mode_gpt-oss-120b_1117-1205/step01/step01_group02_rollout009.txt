"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.
The heavy O(N³) part out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d] is
implemented as a batched matrix multiplication left @ rightᵀ for each
(hidden) head using a custom Triton kernel.  All other operations
(LayerNorm, linear projections, gates, final LayerNorm and linear) are
performed with PyTorch (fp16 for the heavy tensors, fp32 for the final
output).
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for many independent (M=N) × (K=N) products.
# A: [B*H, M, K] (fp16)
# B: [B*H, K, N] (fp16)   (right matrix already transposed)
# C: [B*H, M, N] (fp16)   (output)
# Accumulation is in fp32 for stability.
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,           # pointers
    M, N, K,                       # matrix dimensions
    stride_am, stride_ak,          # A strides (row, col)
    stride_bk, stride_bn,          # B strides (row, col)
    stride_cm, stride_cn,          # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch (B*H)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # Pointers to the current block of A and B (including batch offset)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Load tiles (masked)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back the result (convert to fp16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------------- #
# Entry point required by the harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor containing all linear / ln params
            - config: dict containing at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    import torch.nn.functional as F

    # --------------------------------------------------------------------- #
    # Unpack
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the channel dimension
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to fp16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias) and gating weights
    # --------------------------------------------------------------------- #
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------------- #
    # 3. Apply mask (if any) – broadcast over hidden dimension
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------------- #
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Prepare tensors for the fused GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    # Rearrange to [B, H, N, N] for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right to get rightᵀ
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dimensions for the kernel
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_T.view(total_batch, N, N)
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # --------------------------------------------------------------------- #
    # 6. Launch the Triton batched GEMM kernel
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        total_batch,                  # batch dimension (B * hidden_dim)
    )
    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, H] (fp32) and apply output LN & gate
    # --------------------------------------------------------------------- #
    out = (
        out_flat.view(B, H, N, N)      # [B, H, N, N]
        .permute(0, 2, 3, 1)           # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )
    # Output LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the output gate (cast to fp32 for the multiplication)
    out = out * out_gate.to(torch.float32)

    # --------------------------------------------------------------------- #
    # 8. Final linear projection back to `dim`
    # --------------------------------------------------------------------- #
    output = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return output
