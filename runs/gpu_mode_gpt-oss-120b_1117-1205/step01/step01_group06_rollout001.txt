"""
TriMul (outgoing) forward pass – optimized Triton/PyTorch implementation.

Algorithm
---------
1. Input LayerNorm over the last dimension (dim).
2. Linear projections → left/right (hidden_dim) and three gate vectors
   (left_gate, right_gate, out_gate) using supplied weights (no bias).
3. Apply optional mask (broadcast on the hidden dimension) and the gates:

       right = right * mask * right_gate
4. Pairwise contraction over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication:
       out = (left_permuted) @ (right_permuted).T
   The multiplication is performed in bfloat16 for Tensor‑core speed‑up.
5. LayerNorm over the hidden dimension (h) using a custom Triton kernel.
6. Multiply by out_gate.
7. Final linear projection back to `dim`.

The kernel in step 5 is required so that some work is performed in Triton.
All other heavy work uses highly‑optimized Torch kernels (batched GEMM,
linear, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,               # *float32  [B,N,N,H]
    out_ptr,               # *float32  [B,N,N,H]
    weight_ptr,            # *float32  [H]
    bias_ptr,              # *float32  [H]
    B,                     # tl.int32
    N,                     # tl.int32
    H,                     # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,# tl.int32
    eps: tl.constexpr,                                   # const float
    BLOCK_H: tl.constexpr                               # compile‑time block size (≥ max hidden_dim)
):
    """
    Compute LayerNorm over the hidden dimension `H` for each (b,i,j) slice.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode pid -> (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Base pointer for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # Load the H‑dimensional vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0,
    )

    # ---- mean ----
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    # ---- variance ----
    diff = x - mean
    sq = diff * diff
    var = tl.sum(sq, axis=0) / H

    # ---- normalize ----
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ---- apply affine parameters ----
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # Store result
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h,
    )


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dict of model weight tensors
            - config: dict containing "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (float32)
    x_flat = x.view(-1, dim)  # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) and gates
    if not nomask:
        # mask shape: [B,N,N] -> [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction over k using batched GEMM (bf16)
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Bring hidden dimension to batch axis: (B, H, N, N)
    left_bf = left_bf.permute(0, 3, 1, 2).contiguous()
    right_bf = right_bf.permute(0, 3, 1, 2).contiguous()

    # Flatten batch & head
    left_flat = left_bf.view(B * hidden_dim, N, N)
    right_flat = right_bf.view(B * hidden_dim, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Back to (B, N, N, H) in float32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype)

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)

    # Strides are given in element units
    s_b, s_i, s_j, s_h = out.stride()

    total_programs = B * N * N
    grid = (total_programs,)

    BLOCK_H = 128  # hidden_dim ≤ 128 for all test configs

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,   # input strides
        s_b, s_i, s_j, s_h,   # output strides
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Apply out_gate (elementwise)
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
