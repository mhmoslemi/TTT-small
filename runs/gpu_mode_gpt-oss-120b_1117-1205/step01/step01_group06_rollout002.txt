def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # step1: layernorm on dim
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # step2: linear projections
    # flatten for F.linear
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B,N,N,hidden_dim)

    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()

    # mask
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # step3: contraction using matmul with bfloat16
    # cast to bfloat16 for performance
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Permute: left (B,N,N,hidden) -> (B, hidden, N, N) ; right similarly but we need transpose on last two dims (j,k)
    left_perm = left_bf.permute(0, 3, 1, 2)   # (B, hidden, i, k)
    right_perm = right_bf.permute(0, 3, 2, 1) # (B, hidden, k, j) (transpose)
    # batch matmul for each hidden dimension
    out_bf = torch.matmul(left_perm, right_perm)  # (B, hidden, i, j)
    # permute back to (B,N,N,hidden)
    out = out_bf.permute(0, 2, 3, 1).contiguous()  # .contiguous needed for stride.

    out = out.to(dtype)  # back to float32 for layernorm

    # step4: hidden-dim LayerNorm via Triton
    # allocate output
    out_norm = torch.empty_like(out)

    # Get strides
    s_b, s_i, s_j, s_h = out.stride()
    # Launch kernel
    total_programs = B * N * N
    grid = (total_programs, )
    BLOCK_H = 128  # equal to hidden_dim max

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # step5: apply out_gate
    out_scaled = out_norm * out_gate

    # step6: final linear
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = torch.nn.functional.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final

@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # compute base offset
    inp_offset = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_offset = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # load
    x = tl.load(inp_ptr + inp_offset + offs_h * stride_in_h, mask=mask_h, other=0.0).to(tl.float32)
    # compute mean and variance
    # Note: tl.mean not available; use sum
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    diff = x - mean
    sq = diff * diff
    var = tl.sum(sq, axis=0) / H

    inv_std = 1.0 / tl.sqrt(var + eps)
    norm = diff * inv_std

    # scale and bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    y = norm * w + b_

    # store
    tl.store(out_ptr + out_offset + offs_h * stride_out_h, y, mask=mask_h)

TriMul (outgoing) forward pass.

Algorithm
---------
1. Apply LayerNorm on the last dimension (dim) of the input tensor.
2. Compute linear projections (`left`, `right`) and three gating vectors

3. If a mask is present, multiply `left` and `right` by the mask (broadcasted).
4. Apply the gates:
        left  = left  * left_gate
        right = right * right_gate
5. Compute the pairwise contraction over the sequence dimension `k`:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of shape (B, hidden_dim, N, N)
   performed in bfloat16 for higher throughput.
6. Perform a LayerNorm across the hidden dimension `hidden_dim` using a small
   Triton kernel (one program per (b,i,j) slice).
7. Multiply the normalized tensor by `out_gate`.
8. Final linear projection back to the original `dim`.

Only the hidden‑dimension LayerNorm is implemented in Triton; the remaining
operations use highly‑optimized PyTorch kernels (including a batched GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: LayerNorm over the hidden dimension for each (b,i,j) slice.
# ----------------------------------------------------------------------------- #
@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,               # *float32  [B,N,N,H]
    out_ptr,               # *float32  [B,N,N,H]
    weight_ptr,            # *float32  [H]
    bias_ptr,              # *float32  [H]
    B,                     # tl.int32
    N,                     # tl.int32
    H,                     # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # tl.int32
    eps: tl.constexpr,                                   # const float
    BLOCK_H: tl.constexpr                               # compile‑time block size (≥ max hidden_dim)
):
    pid = tl.program_id(0)

    total = B * N * N
    if pid >= total:
        return

    # Decode the (b,i,j) coordinates from the 1‑D program id
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Base offsets for the slice (b,i,j,:)
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load the H‑dimensional vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    ).to(tl.float32)

    # Compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    # Compute variance
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / H

    # Normalize
    inv_std = 1.0 / tl.sqrt(var + eps)
    norm = diff * inv_std

    # Apply gain and bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h,   mask=mask_h, other=0.0)
    y = norm * w + b_

    # Store result
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        y,
        mask=mask_h
    )

# ----------------------------------------------------------------------------- #
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor, shape [B, N, N, dim]
        - mask        : torch.Tensor, shape [B, N, N] (may be unused)
        - weights     : dict of tensors containing all model parameters
        - config      : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # assumed to be torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Ensure a contiguous layout for later reshapes
    x = x.contiguous()

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------ #
    # 2. Linear projections and gating vectors
    # ------------------------------------------------------------------ #
    # Collapse the first three dims for a single linear call
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate  = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate   = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------ #
    # 3. Apply mask (if present) and gates
    # ------------------------------------------------------------------ #
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 4. Pairwise contraction over k (batched matrix multiplication)
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------ #
    # Cast to bfloat16 to reduce bandwidth
    left_bf  = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Reshape to (B, hidden_dim, N, N) for batched GEMM
    left_perm  = left_bf.permute(0, 3, 1, 2)   # (B, H, i, k)
    right_perm = right_bf.permute(0, 3, 2, 1)  # (B, H, k, j)   ← transpose last two dims

    # Batched matrix multiplication (uses cuBLAS)
    out_bf = torch.matmul(left_perm, right_perm)   # (B, H, i, j)

    # Back to original layout and to float32
    out = out_bf.permute(0, 2, 3, 1).contiguous().to(dtype)   # (B, N, N, H)

    # ------------------------------------------------------------------ #
    # 5. LayerNorm over the hidden dimension (Triton)
    # ------------------------------------------------------------------ #
    out_norm = torch.empty_like(out)

    # Strides are expressed in element counts
    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (total_programs,)

    # Hidden dimension never exceeds 384 in the test suite; use a safe block size
    BLOCK_H = 384

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------ #
    # 6. Apply outgoing gate
    # ------------------------------------------------------------------ #
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------ #
    # 7. Final linear projection back to `dim`
    # ------------------------------------------------------------------ #
    out_scaled_flat = out_scaled.view(-1, hidden_dim)      # [B*N*N, H]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
```
