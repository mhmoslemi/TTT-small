"""
TriMul (outgoing) forward pass fused with a Triton batched GEMM kernel.

Algorithm
---------
1. LayerNorm over the last dimension of the input tensor.
2. Linear projections to a hidden dimension and sigmoid gating (left/right/out).
3. Optional mask is applied to the projected values before gating.
4. The core operation is a batched matrix multiplication:
       out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
   which is equivalent to left @ rightᵀ for each (batch, head) pair.
   This is performed by a Triton kernel that processes all (batch*head)
   matrices in parallel, using fp16 arithmetic with fp32 accumulation.
5. The result is reshaped back, LayerNorm’ed, multiplied by the out‑gate,
   and finally projected back to the original dimension.
All heavy arithmetic is kept in fp16; the final output is fp32.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM A @ Bᵀ  (A: M×K, B: N×K -> C: M×N)
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers to A, B, C
    M, N, K,                      # matrix dimensions (int)
    stride_am, stride_ak,         # A strides (row, col)
    stride_bk, stride_bn,         # B strides (row, col)  (B is already transposed)
    stride_cm, stride_cn,         # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch (b * head)

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)

        # mask for current K tile
        mask_k = (k_offset + offs_k) < K

        # compute pointers for the current tile (including batch offset)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # load with masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # multiply‑accumulate
        acc += tl.dot(a, b)

    # write result back (cast to FP16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# --------------------------------------------------------------------------- #
# Entry point used by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Perform the outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensors containing all linear/LN parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # 1️⃣  LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to fp16 for the heavy ops
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2️⃣  Linear projections & gating (all in fp16)
    # ------------------------------------------------------------------- #
    left = torch.nn.functional.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = torch.nn.functional.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )

    left_gate = torch.nn.functional.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------- #
    # 3️⃣  Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------- #
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4️⃣  Apply element‑wise gates
    # ------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣  Prepare tensors for the fused GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------- #
    # 5a. Permute to bring the head dimension upfront and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # 5b. Transpose the last two axes of right so that we can compute A @ B
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse the batch & head dimensions for the Triton kernel
    left_flat = left_perm.view(total_batch, N, N)   # [B*H, N, N]
    right_flat = right_T.view(total_batch, N, N)    # [B*H, N, N]
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------- #
    # 6️⃣  Launch the Triton kernel
    # ------------------------------------------------------------------- #
    # Strides in *elements* (not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (rows)
        triton.cdiv(N, BLOCK_N),   # blocks along N (cols)
        total_batch,               # one program per (batch, head)
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 7️⃣  Reshape back to [B, N, N, hidden_dim] (fp32)
    # ------------------------------------------------------------------- #
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------- #
    # 8️⃣  Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)  # apply the sigmoid‑gated mask

    # Linear projection back to `dim`
    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # shape [B, N, N, dim]

    return output
