def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # cast to half
    x_h = x_norm.to(torch.float16)

    # Linear proj
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)  # shape B,N,N,hidden
    right = F.linear(x_h, w_right)

    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)
        left = left * mask_exp
        right = right * mask_exp

    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # prepare for gemm
    B, N, _, H = left.shape

    # permute and view
    left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0,3,2,1).contiguous().view(B*H, N, N)  # transposed

    # output placeholder
    out_mat = torch.empty((B*H, N, N), dtype=torch.float16, device=device)

    # strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        #optional: num_stages=3
    )

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # shape B,N,N,H
    out_f32 = out.to(torch.float32)

    # Output layernorm
    gamma_out = weights["to_out_norm.weight"]
    beta_out = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_f32, (hidden_dim,), weight=gamma_out, bias=beta_out, eps=eps)

    # out gate (float32)
    out_gate_f32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # final linear
    w_to_out = weights["to_out.weight"]
    output = F.linear(out_norm, w_to_out)  # shape B,N,N,dim

    return output

"""
TriMul forward kernel (outgoing version).

The module performs LayerNorm, linear projections, gating, and the core
contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    out_d = left_d @ right_dᵀ
for each hidden dimension d. The heavy matrix multiplication is implemented
as a custom Triton kernel; all other operations stay in PyTorch (using
mixed‑precision where beneficial). The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel: C = A @ B   (A: (M,K), B: (K,N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch * hidden_dim
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to output tile
    c_ptrs = (
        C_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B tile (K x N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: mapping from weight names to torch.Tensor
            - config_dict : contains "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Cast to fp16 for projections/gates
    # --------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # Linear weights (fp16)
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3. Projections
    # --------------------------------------------------------------
    left = F.linear(x_h, w_left)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # --------------------------------------------------------------
    # 4. Optional mask (applied before gating)
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 5. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N) where each row‑matrix corresponds to a hidden slice
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right transposed on the second sequence axis: (b, d, k, j)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides for the view‑shaped tensors
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16
    out_f32 = out.to(torch.float32)

    # --------------------------------------------------------------
    # 7. Output LayerNorm (float32)
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 8. Apply out gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)
    output = F.linear(out_norm, weights["to_out.weight"])  # [B, N, N, dim] (float32)

    return output
