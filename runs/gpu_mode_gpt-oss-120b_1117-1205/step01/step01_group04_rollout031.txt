import torch
import torch.nn.functional as F

def custom_kernel(data):
    inp, mask, weights, cfg = data
    # Get dims
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # Layernorm
    x = F.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Cast to half for later
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    if not nomask:
        # mask shape [B,N,N] may be bool, convert to float16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        # Expand mask to hidden_dim
        mask_fp16 = mask_fp16.expand(-1, -1, -1, hidden_dim).contiguous()
        # Allocate output tensors
        left_out = torch.empty_like(left_fp16)
        right_out = torch.empty_like(right_fp16)

        # Determine kernel launch parameters
        total = left_fp16.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        mask_gate_kernel[grid](
            left_fp16, left_gate_fp16, mask_fp16, left_out,
            total,
            BLOCK=BLOCK,
        )
        mask_gate_kernel[grid](
            right_fp16, right_gate_fp16, mask_fp16, right_out,
            total,
            BLOCK=BLOCK,
        )
        left = left_out
        right = right_out
    else:
        # No mask
        left = left_fp16 * left_gate_fp16
        right = right_fp16 * right_gate_fp16

    # Prepare for batched GEMM
    left_batch = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # Compute batched matrix multiplication
    # out_batch shape (B*hidden_dim, N, N)
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

    # Reshape back to [B, N, N, hidden_dim]
    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # Output norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) implementation with Triton acceleration.

The heavy O(N³) contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
is expressed as a batched matrix multiplication
   out_batch = left_batch @ right_batchᵀ
where each batch corresponds to a pair (batch, hidden_dim).
We use PyTorch's highly‑optimized cuBLAS `torch.bmm` for this GEMM and
a tiny Triton kernel to fuse the optional mask with the gating
multiplications.  All other linear layers and LayerNorms are kept in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(src_ptr, gate_ptr, mask_ptr, dst_ptr,
                     total_elements, BLOCK: tl.constexpr):
    """Element‑wise: dst = src * gate * mask   (all fp16)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    src = tl.load(src_ptr + offs, mask=m, other=0.0, dtype=tl.float16)
    gate = tl.load(gate_ptr + offs, mask=m, other=0.0, dtype=tl.float16)
    mask = tl.load(mask_ptr + offs, mask=m, other=1.0, dtype=tl.float16)

    out = src * gate * mask
    tl.store(dst_ptr + offs, out, mask=m)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input, mask, weights, config)
            input  – torch.Tensor of shape [B, N, N, dim] (float32)
            mask   – torch.Tensor of shape [B, N, N] (bool or float) or None
            weights – dict containing the model parameters
            config – dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 1. Layer‑norm on input
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 3. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------ #
    # 4. Cast projections & gates to fp16
    # ------------------------------------------------------------------ #
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 5. Apply mask & gates (fused via Triton when a mask is present)
    # ------------------------------------------------------------------ #
    if not nomask and mask is not None:
        # mask → fp16, broadcast to hidden_dim, make contiguous
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)                 # [B, N, N, 1]
        mask_fp16 = mask_fp16.expand(-1, -1, -1, hidden_dim).contiguous()  # [B, N, N, hidden_dim]

        # allocate output buffers
        left = torch.empty_like(left_fp16)
        right = torch.empty_like(right_fp16)

        total = left_fp16.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        # fused mask * gate * proj for left
        mask_gate_kernel[grid](
            left_fp16, left_gate_fp16, mask_fp16, left,
            total,
            BLOCK=BLOCK,
        )
        # same for right
        mask_gate_kernel[grid](
            right_fp16, right_gate_fp16, mask_fp16, right,
            total,
            BLOCK=BLOCK,
        )
    else:
        # no mask → simple element‑wise multiplication in PyTorch
        left = left_fp16 * left_gate_fp16
        right = right_fp16 * right_gate_fp16

    # ------------------------------------------------------------------ #
    # 6. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    (equivalent to left_batch @ right_batchᵀ)
    # ------------------------------------------------------------------ #
    # reshape to (B * hidden_dim, N, N)
    left_batch = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # [B*H, N, N]
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # [B*H, N, N]

    # cuBLAS batched GEMM (fp16 * fp16 → fp16)
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))      # [B*H, N, N]

    # reshape back to [B, N, N, hidden_dim] and promote to fp32 for subsequent ops
    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------ #
    # 7. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])      # [B, N, N, dim]

    return out
