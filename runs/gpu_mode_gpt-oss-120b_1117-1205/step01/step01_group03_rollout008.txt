@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_m, stride_left_k,
    stride_right_k, stride_right_n,
    stride_out_m, stride_out_n,
    stride_batch_left, stride_batch_right, stride_batch_out,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # base pointers for this batch
    left = left_ptr + pid_batch * stride_batch_left
    right = right_ptr + pid_batch * stride_batch_right
    out = out_ptr + pid_batch * stride_batch_out

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    a_ptrs = left + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
    b_ptrs = right + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    c_ptrs = out + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul "outgoing" forward pass.

The kernel `trimul_kernel` computes a batched matrix multiplication
    out[b, :, :] = left[b, :, :] @ right[b, :, :].T
for each batch * hidden channel (total_batches = batch * hidden_dim).
All other operations (LayerNorm, linear projections, sigmoid gates, final
normalisation and projection) are performed with PyTorch.  The
computation uses fp16 for the large GEMM and fp32 accumulation,
mirroring the behavior of the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    M, N, K,                               # matrix sizes
    stride_left_m, stride_left_k,          # left matrix strides (M, K)
    stride_right_k, stride_right_n,        # right matrix strides (K, N) – note transposed layout
    stride_out_m, stride_out_n,            # output matrix strides (M, N)
    stride_batch_left, stride_batch_right, stride_batch_out,  # strides between batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B^T with fp16 inputs and fp32 accumulation."""
    pid_batch = tl.program_id(0)        # batch * hidden index
    pid_m = tl.program_id(1)            # block row
    pid_n = tl.program_id(2)            # block column

    # Base pointers for this batch
    left = left_ptr + pid_batch * stride_batch_left
    right = right_ptr + pid_batch * stride_batch_right
    out = out_ptr + pid_batch * stride_batch_out

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds accesses
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the current tiles
    a_ptrs = left + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
    b_ptrs = right + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                 # (BLOCK_M,BLOCK_K)·(BLOCK_K,BLOCK_N) → (BLOCK_M,BLOCK_N)
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # Write back the result in fp16
    c_ptrs = out + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the Triangle Multiplicative Update (outgoing) operator.

    Args:
        data: Tuple of (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [bs, N, N, dim]
            - mask_tensor: torch.Tensor of shape [bs, N, N] (may be None)
            - weights: dict with the required weight tensors
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]            # (dim,)
    norm_b = weights["norm.bias"]              # (dim,)

    mean = input_tensor.mean(-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x_norm * norm_w + norm_b                # [bs, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections & mask (fp16)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)    # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)      # [bs, N, N, hidden]
    right = F.linear(x_f16, right_proj_w)

    # Mask handling – if mask is None treat as all‑ones
    if mask is None:
        mask_f16 = torch.ones((bs, N, N), device=device, dtype=torch.float16)
    else:
        mask_f16 = mask.to(torch.float16)
    mask_f16 = mask_f16.unsqueeze(-1)        # broadcast over hidden
    left = left * mask_f16
    right = right * mask_f16

    # ------------------------------------------------------------------
    # 3. Gates (fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_f16, left_gate_w).sigmoid()
    right_gate = F.linear(x_f16, right_gate_w).sigmoid()
    out_gate = F.linear(x_f16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core TriMul: out = left @ right^T   (batched GEMM via Triton)
    # ------------------------------------------------------------------
    # Rearrange tensors to (batch * hidden, N, N) where right is already transposed
    left = left.permute(0, 3, 1, 2).contiguous()      # [bs, hidden, N, N]  (i,k)
    right = right.permute(0, 3, 2, 1).contiguous()    # [bs, hidden, N, N]  (k,j)

    total_batches = bs * hidden
    left_mat = left.reshape(total_batches, N, N)
    right_mat = right.reshape(total_batches, N, N)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Heuristic block sizes
    if N <= 128:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                 # M, N, K
        left_mat.stride(1), left_mat.stride(2),  # stride_left_m, stride_left_k
        right_mat.stride(1), right_mat.stride(2),# stride_right_k, stride_right_n
        out_mat.stride(1), out_mat.stride(2),    # stride_out_m, stride_out_n
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [bs, N, N, hidden] and cast to fp32 for the remaining ops
    out = out_mat.reshape(bs, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gating and final linear projection (float32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]   # (hidden,)
    to_out_norm_b = weights["to_out_norm.bias"]     # (hidden,)

    mean_out = out.mean(-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(-1, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Apply output gate (cast to fp32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final projection back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float32)   # (dim, hidden)
    output = F.linear(out_norm, to_out_w)                  # [bs, N, N, dim]

    return output
