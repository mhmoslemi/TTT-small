```python
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to c tile
    c_ptrs = C_ptr + pid_batch * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + pid_batch * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + pid_batch * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Cast back to fp16
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

Make sure to have `tl.program_id` up to 3 dimensions. In Triton, we need to set grid shape as 3D tuple.

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    
    # LayerNorm over dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Convert to half precision for projection
    x_h = x.half()
    
    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].half())
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].half())

    # Apply mask if provided (mask shape [B,N,N])
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, weights["out_gate.weight"].half()))

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # B, N, N, hidden_dim

    # reshape for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Allocate output
    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # Strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # Tiling parameters

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B  (A:B shape: (batch, M, K) x (batch, K, N))"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to C tile
    c_ptrs = C_ptr + pid_batch * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + pid_batch * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + pid_batch * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
    

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    The heavy tensor contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is performed as a batched GEMM with a custom Triton kernel.
    All other linear layers and LayerNorms are executed in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- Input LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ---------- Projections & Gates (half precision) ----------
    x_h = x.half()  # cast once

    w_left = weights["left_proj.weight"].half()
    w_right = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()

    left = F.linear(x_h, w_left)          # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)  # broadcast to hidden_dim
        left = left * mask_exp
        right = right * mask_exp

    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    # ---------- Batched GEMM via Triton ----------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (B*H, N, N) for A and (B*H, N, N) for B (transposed)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # transpose k<->j

    C = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides
    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bk, stride_bn = B_mat.stride()
    stride_cz, stride_cm, stride_cn = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        A,
        B_mat,
        C,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ---------- Reshape back ----------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]
    out_f32 = out.float()

    # ---------- Output LayerNorm ----------
    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ---------- Apply out gate ----------
    out_norm = out_norm * out_gate.float()

    # ---------- Final linear projection ----------
    output = F.linear(out_norm, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass.

The core contraction 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realized as a batched GEMM:
    out = (left reshaped) @ (right reshaped).T
A custom Triton kernel does the GEMM in fp16 with fp32 accumulation.
All other linear layers and LayerNorms are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B
    A: (batch, M, K)   fp16
    B: (batch, K, N)   fp16
    C: (batch, M, N)   fp16  (stored as fp16, accumulated in fp32)
    """
    pid_batch = tl.program_id(0)          # batch*hidden_dim dimension
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to output tile
    c_ptrs = (
        C_ptr
        + pid_batch * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,

        b_ptrs = (
            B_ptr

            + offs_k[:, None] * stride_bk

            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

    # store result as fp16
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (bool/float or None)
        - weights      : dict of torch.Tensors containing all linear / norm params
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Input LayerNorm (over last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Linear projections and gates (half precision)
    # ------------------------------------------------------------------
    x_h = x.half()  # cast once for all following ops

    # Linear weights (cast to half)
    w_left = weights["left_proj.weight"].half()
    w_right = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()

    # Projections
    left = F.linear(x_h, w_left)    # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # Optional mask (broadcast over hidden_dim)
    if mask is not None:

        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Batched GEMM: out = left @ right^T   (over the k dimension)
    # ------------------------------------------------------------------

    B_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)    # (batch, K, N)

    C = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bk, stride_bn = B_mat.stride()
    stride_cz, stride_cm, stride_cn = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        A,
        B_mat,
        C,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and convert to fp32
    # ------------------------------------------------------------------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]
    out_f32 = out.float()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
