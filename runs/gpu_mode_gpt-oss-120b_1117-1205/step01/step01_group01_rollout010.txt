@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)  # row block index
    pid_n = tl.program_id(1)  # col block index
    pid_batch = tl.program_id(2)  # batch index (B*H)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute number of K blocks
    num_k_blocks = tl.cdiv(K, BLOCK_K)

    for k in range(num_k_blocks):
        # offsets for K block
        k_offset = k * BLOCK_K
        # pointers
        a_ptrs = a_ptr \
                + pid_batch * stride_batch_a \
                + offs_m[:, None] * stride_am \
                + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak

        b_ptrs = b_ptr \
                + pid_batch * stride_batch_b \
                + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                + offs_n[None, :] * stride_bn

        # mask
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr \
            + pid_batch * stride_batch_c \
            + offs_m[:, None] * stride_cm \
            + offs_n[None, :] * stride_cn

    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

That's fine.

"""
TriMul (outgoing) forward pass with a Triton‑accelerated cubic contraction.

The module first applies LayerNorm, two linear projections and three sigmoid
gates (all in FP32 for numerical stability).  The expensive contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

is implemented as a batched matrix multiplication using a custom Triton
kernel.  For each batch * hidden dimension we multiply an (N×N) matrix
with the transpose of another (N×N) matrix, accumulating in FP32 while the
operands are stored in FP16.  The result is then processed with a second
LayerNorm, an output gate and a final linear projection back to the input
dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,                # [BH, M, K] matrix A (FP16)
    b_ptr,                # [BH, K, N] matrix B (FP16), already transposed
    c_ptr,                # [BH, M, N] output matrix C (FP16)
    M, N, K,              # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak, # strides for A: row (M), col (K)
    stride_bk, stride_bn, # strides for B: row (K), col (N)
    stride_cm, stride_cn, # strides for C: row (M), col (N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C[batch] = A[batch] @ B[batch]  (M×K)·(K×N) → (M×N)."""
    pid_m = tl.program_id(0)          # tile row index
    pid_n = tl.program_id(1)          # tile column index
    pid_batch = tl.program_id(2)      # batch * hidden index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_off = k * BLOCK_K
        # Pointers to A and B tiles
        a_ptrs = a_ptr \
                 + pid_batch * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak

        b_ptrs = b_ptr \
                 + pid_batch * stride_batch_b \
                 + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & ((k_off + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_off + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # dot accumulates in FP32
        acc += tl.dot(a, b)

    # write back result
    c_ptrs = c_ptr \
             + pid_batch * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn

    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool) or None
            - weights: dict of model parameters (see module definition)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B, N, N, dim]

    # --------------------------------------------------------------------
    # 2. Linear projections (float32, bias=False)
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------------
    # 3. Optional mask (broadcasted, still float32)
    # --------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------
    # 4. Gating (float32)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (Triton, FP16 inputs)
    # --------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Convert to FP16 just before the kernel to keep most work in half precision
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange tensors so each (batch, hidden) pair becomes an independent matrix
    A = left_fp16.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N) -> rows=i, cols=k
    Bmat = right_fp16.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) -> rows=k, cols=j

    BH = B * H
    A_flat = A.view(BH, N, N)   # (BH, M, K)
    B_flat = Bmat.view(BH, N, N)  # (BH, K, N)
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)  # (BH, M, N)

    # Block sizes – tuned for typical N up to 1024 on an NVIDIA H100 (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        BH                          # batch * hidden
    )

    batched_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,                                 # M, N, K
        A_flat.stride(1), A_flat.stride(2),      # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),      # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),      # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden_dim]
    out = C_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm on hidden dim, output gate, final proj)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)  # back to FP32 for stable LayerNorm

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the output gate (still float32)
    out_norm = out_norm * out_gate

    to_out_w = weights["to_out.weight"].to(device)   # shape (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)            # final shape [B, N, N, dim]

    return output
