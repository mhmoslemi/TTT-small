"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

The operator works on a 4‑D tensor [B, N, N, C] (C = dim) and consists of:
1. LayerNorm over the channel dimension.
2. Linear projections (left/right) and gating (left/right/out) in half precision.
3. Optional element‑wise mask.
4. Application of the gates.
5. Core contraction:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which is a batched matrix multiplication of shape (B·D, N, N) @ (B·D, N, N)^T.
   Implemented with a custom Triton kernel (float16 inputs, float32 accumulation).
6. Output LayerNorm, out‑gate, and final linear projection back to the original
   channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (C = A @ B) where
#   A: (batch, M, K)  B: (batch, K, N)
# Uses float16 inputs and float32 accumulation.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load tiles
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Compute block‑wise dot product (uses Tensor Cores)
        acc += tl.dot(a, b)

    # Write result back (convert to FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B.

    A: (batch, M, K)   B: (batch, K, N)      both float16 and contiguous
    Returns C: (batch, M, N) float16.
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes (tuned for H100/B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2) Convert to half‑precision for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    )

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, hidden, N, N) layout expected by the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) = (B, H, K, J)

    Bsz, H, N, _ = left_perm.shape
    # Collapse the batch and hidden dimensions
    left_flat = left_perm.view(Bsz * H, N, N)
    right_flat = right_perm.view(Bsz * H, N, N)

    # Batched matmul: each (batch*hidden) computes (N,N) @ (N,N)^T
    out_flat = triton_batched_matmul(left_flat, right_flat)   # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)   # broadcast over hidden channel
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
