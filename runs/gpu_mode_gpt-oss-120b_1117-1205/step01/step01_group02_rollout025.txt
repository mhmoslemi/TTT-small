"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input over the feature dimension.
2. Linear projections (left/right) and three element‑wise sigmoid gates.
3. (Optional) apply the pairwise mask to the projected tensors.
4. Fuse the heavy N³ computation with a Triton batched GEMM:
       out[b, h] = left[b, h] @ right[b, h].T
   where left/right have shape (B, H, N, N) and we treat the (B·H) axis
   as independent batches.
5. Reshape back to (B, N, N, H), cast to FP32, apply output LayerNorm,
   multiply by the output gate, and a final linear projection to the
   original feature dimension.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dims
    stride_am, stride_ak,         # A strides (row, col)
    stride_bk, stride_bn,         # B strides (row, col)
    stride_cm, stride_cn,         # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for many (batch * heads) matrices:
       C = A @ B   where A: (M, K), B: (K, N)
       Each independent batch is addressed by program_id(2).
    """
    pid_b = tl.program_id(2)   # batch * head index
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # Pointers to the current tiles of A and B
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Load with masking
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of outgoing TriMul using a fused Triton GEMM.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor containing all linear / LN parameters
            - config: dict with at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor: [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. Input layer‑norm over the feature dimension
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_fp16 = x.to(torch.float16)  # fp16 for the heavy ops

    # --------------------------------------------------------------------- #
    # 2. Linear projections and gates (all fp16)
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]

    right = torch.nn.functional.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]

    left_gate = torch.nn.functional.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, H]

    right_gate = torch.nn.functional.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, H]

    out_gate = torch.nn.functional.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Prepare tensors for the fused GEMM (A @ Bᵀ)
    # --------------------------------------------------------------------- #
    # Permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose last two axes of right so we can multiply A @ B
    right_t = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dimensions
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_t.view(total_batch, N, N)

    # Output buffer
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # --------------------------------------------------------------------- #
    # 6. Launch Triton kernel
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        total_batch,               # batch * head
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, H] and convert to FP32
    # --------------------------------------------------------------------- #
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # --------------------------------------------------------------------- #
    # 8. Output LayerNorm, gate, and final projection
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)  # apply output gate

    # Final linear projection back to `dim`
    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # [B, N, N, dim]

    return output
