"""
Highly‑optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑normalize the input tensor over the channel dimension (dim).
2. Compute left/right projections (Linear, no bias) in fp16.
3. Compute three sigmoid gates (left, right, out) in fp32 from the
   normalized full‑precision tensor.
4. Apply optional mask to the projected tensors, then apply the elementwise
   gates (cast to fp16).
5. Fuse the core O(N³·H) computation:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (A @ Bᵀ) with
   batch = B·H.  A custom Triton kernel performs the GEMM in fp16
   (accumulating in fp32) without any additional transposes.
6. Layer‑normalize the result over the hidden dimension, multiply by the
   output gate, and linearly project back to the original channel size.
7. Return a float32 tensor of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel: C = A @ B   where A: (M,K), B: (K,N)
# The third grid dimension indexes the independent batch (B*H).
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_am, stride_ak,         # A strides (row, col)
    stride_bk, stride_bn,         # B strides (row, col)
    stride_cm, stride_cn,         # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch index (B * H)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # A tile
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # B tile
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)

    # Store C
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask_tensor, weight_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weight_dict : dict containing all Linear / LayerNorm parameters
            - config_dict : dict with at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs and config
    # ------------------------------------------------------------------
    input_tensor, mask, weights, cfg = data
    device = input_tensor.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16) and gating (fp32)
    # ------------------------------------------------------------------
    # Cast normalized tensor to fp16 for the heavy matmul part
    x_norm_fp16 = x_norm.half()                               # [B,N,N,dim] fp16

    # Projection weights (fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_norm_fp16, left_proj_w)   # [B,N,N,hidden_dim] fp16
    right = torch.nn.functional.linear(x_norm_fp16, right_proj_w)  # fp16

    # Gate weights stay in fp32 (more stable sigmoid)
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()   # fp32
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid() # fp32
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()     # fp32

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask is [B, N, N]; broadcast to [B, N, N, 1] and convert to fp16
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Apply element‑wise gates (cast gates to fp16)
    # ------------------------------------------------------------------
    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the fused GEMM (A @ Bᵀ)
    #    We want: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    This is equivalent to (left_perm @ right_T) for each (b,h)
    # ------------------------------------------------------------------
    # Permute to bring hidden dimension to the batch axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]

    # Transpose last two axes of right so we can do a simple GEMM
    right_T = right_perm.transpose(-2, -1).contiguous()       # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dimensions for the kernel
    left_flat = left_perm.view(total_batch, N, N)   # [total_batch, M, K]
    right_flat = right_T.view(total_batch, N, N)   # [total_batch, K, N]
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        total_batch,               # batch dimension (B*H)
    )

    _trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )   # --> shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, and final projection to dim
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [B, N, N, hidden_dim] float32

    out = out * out_gate                     # broadcasted element‑wise (fp32)

    # Final linear projection back to original dimension
    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] float32

    return output
