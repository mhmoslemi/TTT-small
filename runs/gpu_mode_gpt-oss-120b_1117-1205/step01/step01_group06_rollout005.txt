"""
TriMul outgoing operator (forward only).

Algorithm
---------
1. Layer‑norm over the last dimension ``dim`` (torch.nn.functional.layer_norm).
2. Linear projections to hidden dimension ``hidden_dim`` for left/right
   and three gating vectors (left_gate, right_gate, out_gate).
3. Apply the optional mask to left/right, then element‑wise multiply by
   the corresponding gates.
4. Contract over the sequence dimension `k`:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of matrix multiplications:
       left_h @ right_hᵀ for each hidden channel `h`.
   We reshape the tensors to (B·H,N,N) and use ``torch.bmm`` in bfloat16.
5. Layer‑norm over the hidden dimension (custom Triton kernel).
6. Multiply by ``out_gate`` and apply the final linear projection back to
   ``dim``.
All heavy compute (step 4 and step 5) runs on the GPU; the rest uses
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: Layer‑norm over the hidden dimension (H)
@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,               # *float32  [B,N,N,H]
    out_ptr,               # *float32  [B,N,N,H]
    weight_ptr,            # *float32  [H]
    bias_ptr,              # *float32  [H]
    B,                     # int32
    N,                     # int32
    H,                     # int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # int32
    eps: tl.constexpr,                                   # float constant
    BLOCK_H: tl.constexpr                               # compile‑time block size (≥ hidden_dim)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for the H‑axis
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base pointer for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load hidden vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    )
    # ------------------------------------------------------------------
    # Compute mean
    sum_x = tl.sum(x, axis=0)               # sum over H
    mean = sum_x / H

    # ------------------------------------------------------------------
    # Compute variance
    diff = x - mean
    sq = diff * diff
    sum_sq = tl.sum(sq, axis=0)
    var = sum_sq / H

    # ------------------------------------------------------------------
    # Normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ------------------------------------------------------------------
    # Apply learned scale (weight) and bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h,   mask=mask_h, other=0.0)
    out = norm * w + b_

    # ------------------------------------------------------------------
    # Store result
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h
    )

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim]
            - mask: torch.Tensor[B, N, N] (may be ignored)
            - weights: dict of torch.Tensor parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input layer‑norm over last dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2. Linear projections & gates (all in float32)
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # --------------------------------------------------------------
    # 3. Mask + gate application
    if not nomask:
        # mask shape: [B, N, N] -> [B,N,N,1]
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Pairwise contraction via batched GEMM (bfloat16)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # bring hidden dim to leading position: (B, H, N, N)
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous()

    # transpose right on the last two axes (k ↔ j) → (B, H, N, N)
    right_T = right_perm.transpose(-2, -1).contiguous()

    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_T.view(B * hidden_dim, N, N)

    # batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat)      # bf16, shape (B*H, N, N)

    # reshape back to (B, N, N, H) and cast to float32
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)

    # strides in element units
    s_b, s_i, s_j, s_h = out.stride()
    # grid: one program per (b,i,j)
    total_programs = B * N * N
    grid = (total_programs,)

    BLOCK_H = 128                     # hidden_dim ≤ 128 for all test configs
    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 6. Apply out_gate
    out_scaled = out_norm * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)   # [B*N*N, hidden_dim]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
