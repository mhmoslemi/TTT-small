"""
TriMul (outgoing) forward pass – heavy cubic contraction is accelerated with
cuBLAS batched GEMM while the cheap elementwise mask‑gate multiplication
is fused in a tiny Triton kernel.  All LayerNorms and Linear projections are
performed in PyTorch (FP16 where possible); the final output is returned in
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_mul_kernel(
    src_ptr,            # float16*  (B,N,N,H) flattened
    gate_ptr,           # float16*  same shape as src
    mask_ptr,           # float16*  (B,N,N) flattened
    dst_ptr,            # float16*  output, same shape as src
    total_elements,    # total number of elements in src/gate (B*N*N*H)
    hidden_dim,         # H, used for mask broadcasting
    BLOCK: tl.constexpr,
):
    """Apply src = src * gate * mask  (mask broadcast over hidden_dim)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK, dtype=tl.int64)
    mask = offs < total_elements

    src = tl.load(src_ptr + offs, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # mask is broadcast over the hidden dimension: each mask element repeats H times.
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    tl.store(dst_ptr + offs, src * gate * mask_val, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32) – may be all‑ones
            - weights: dict of model parameters (see module definition)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B, N, N, dim]

    # --------------------------------------------------------------------
    # 2. Cast to fp16 for the expensive path
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_w) # [B, N, N, hidden_dim]

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid) – still fp16
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------------
    # 5. Apply mask and gating with a tiny Triton kernel
    # --------------------------------------------------------------------
    # Ensure contiguous tensors for simple 1‑D indexing
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if mask is None:
        mask_fp16 = torch.ones_like(left[..., 0], dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()

    B, N, _, H = left.shape
    total = B * N * N * H                       # number of fp16 elements
    mask_flat = mask_fp16.view(-1)               # shape (B*N*N,)

    left_masked = torch.empty_like(left)
    right_masked = torch.empty_like(right)

    BLOCK = 65536
    grid = (triton.cdiv(total, BLOCK),)

    # left = left * left_gate * mask
    mask_gate_mul_kernel[grid](
        left,
        left_gate,
        mask_flat,
        left_masked,
        total,
        H,
        BLOCK=BLOCK,
    )
    # right = right * right_gate * mask
    mask_gate_mul_kernel[grid](
        right,
        right_gate,
        mask_flat,
        right_masked,
        total,
        H,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------------
    # 6. Cubic contraction via batched fp16 GEMM (leveraging cuBLAS)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------------
    # (B,H,N,N) layout; right needs the (k,j) ordering
    left_perm = left_masked.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) – rows=i, cols=k
    right_perm = right_masked.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) – rows=k, cols=j

    BH = B * H
    left_mat = left_perm.view(BH, N, N)    # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)  # (BH, N, N)

    # Batched matrix multiply – fp16, TensorCore‑accelerated
    out_mat = torch.bmm(left_mat, right_mat)  # (BH, N, N) fp16

    # Restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------------
    # 7. Post‑processing (LayerNorm on hidden dim, final gate, output proj)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)           # final [B, N, N, dim] float32

    return output
