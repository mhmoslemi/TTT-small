"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched matrix multiplication.
The heavy O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM:
    out_d = left_d @ right_d.T
for each hidden channel d. All surrounding LayerNorms, linear projections,
gating and final projection are performed in PyTorch (mixed‐precision).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where
#   A: (BH, M, K)   row‑major
#   B: (BH, K, N)   row‑major
#   C: (BH, M, N)   output
# ---------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers to A, B, C
    M, N, K,                      # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak,         # strides for A (row, col)
    stride_bk, stride_bn,         # strides for B (row, col)
    stride_cm, stride_cn,         # strides for C (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block index over M
    pid_n = tl.program_id(1)          # block index over N
    pid_batch = tl.program_id(2)      # which (batch×hidden) matrix

    # Compute offsets for the block of output C we are producing
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K
        # Offsets for tiles of A and B
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers for the current K‑tile
        a_ptrs = a_ptr + pid_batch * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_batch_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Load tiles
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate partial dot product
        acc += tl.dot(a, b)

    # Write back results to C
    c_ptrs = c_ptr + pid_batch * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ---------------------------------------------------------------
# Host entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module (outgoing version).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:         torch.Tensor [B, N, N] (float32) or None
            - weights:      dict of model parameters (see module definition)
            - config:       dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # -----------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B, N, N, dim]

    # Work in FP16 for heavy work
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)   # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # -----------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # [B, N, N, H]

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (Triton)
    #    Compute out_d = left_d @ right_d.T for each hidden channel.
    # -----------------------------------------------------------
    B, N, _, H = left.shape  # left shape: [B, N, N, H]

    # Re‑order to (B, H, N, N) where rows=i, cols=k for left
    left_perm = left.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)
    # Re‑order right to (B, H, N, N) where rows=k, cols=j (i.e. transpose last two dims)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N)

    BH = B * H
    A_flat = left_perm.reshape(BH, N, N)   # (BH, M, K)
    B_flat = right_perm.reshape(BH, N, N)  # (BH, K, N)

    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)  # (BH, M, N)

    # Block sizes – chosen to give good occupancy on NVIDIA B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        BH                         # batch*hidden
    )

    batched_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,                                  # M, N, K
        A_flat.stride(1), A_flat.stride(2),       # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),       # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),       # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),   # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = C_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -----------------------------------------------------------
    # 6. Post‑processing (LayerNorm on hidden dim, final gate, output proj)
    # -----------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)   # apply final gate

    to_out_w = weights["to_out.weight"].to(device)    # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)            # [B, N, N, dim] (float32)

    return output
