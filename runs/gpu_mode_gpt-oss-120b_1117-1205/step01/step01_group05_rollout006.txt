Implementation of kernel for batched GEMM:

```
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Determine valid offsets within matrix dimensions
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute pointers for the C tile
    c_ptrs = C_ptr + pid_batch * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    # Initialize accumulation with zeros in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + pid_batch * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + pid_batch * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # Convert accumulation to fp16 (or keep fp32?) We'll store as half for memory.
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

"""
TriMul (outgoing) forward pass.

The heavy part of the operation is the contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is exactly a batched matrix multiplication:
    C = A @ B   where
    A (batch, M, K) = left[b, d, i, k]   (i as rows, k as cols)
    B (batch, K, N) = right[b, d, k, j]  (k as rows, j as cols)

We launch a Triton kernel that computes this GEMM for all
(batch = batch_size * hidden_dim) simultaneously. All other
operations (LayerNorm, linear projections, gates) are performed
in PyTorch (mixed FP16/FP32). The final output is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B
    A: (batch, M, K)
    B: (batch, K, N)
    C: (batch, M, N)
    All tensors are row‑major.
    """
    pid_batch = tl.program_id(0)   # batch index (batch = bs * hidden_dim)
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # ---- tile offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- pointers to the output tile ----
    c_ptrs = (
        C_ptr
        + pid_batch * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension in tiles ----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B tile (K x N)
        b_ptrs = (
            B_ptr
            + pid_batch * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate into fp32 accumulator
        acc += tl.dot(a, b)

    # store result as fp16 (saves memory)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim] (float32)
            mask   : torch.Tensor [B, N, N] (float32, bool or None)
            weights: dict of torch.Tensors with model parameters
            config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------------
    # Unpack arguments
    # ---------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------------------------------------------------------------
    # Input LayerNorm (over the last dimension)
    # ---------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)  # [B, N, N, dim]

    # ---------------------------------------------------------------
    # FP16 projection + gating
    # ---------------------------------------------------------------
    x_h = x.to(torch.float16)  # half‑precision for heavy ops

    # Linear weight tensors (convert once)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Projections
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] (fp16)
    right = F.linear(x_h, right_proj_w)

    # Optional mask (broadcasted over hidden_dim)
    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, hidden_dim] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # Reshape for batched GEMM: C = A @ B   (A: left, B: right^T)
    # ---------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # left : (B, N, N, H) -> (B, H, N, N)  (rows = i, cols = k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)   # (batch, M, K)

    # right : (B, N, N, H) -> (B, H, N, N) with k/j swapped -> (B, H, K, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # swap the last two N axes
    right_mat = right_perm.view(B * H, N, N)   # (batch, K, N)

    # Output buffer (fp16)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # ---------------------------------------------------------------
    # Launch Triton kernel for the batched GEMM
    # ---------------------------------------------------------------
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,          # M
        N,          # N
        N,          # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ---------------------------------------------------------------
    # Reshape back to original layout and continue in FP32
    # ---------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim] fp16
    out_f32 = out.to(torch.float32)

    # ---------------------------------------------------------------
    # Output LayerNorm over hidden_dim
    # ---------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_f32, (hidden_dim,), weight=to_out_norm_w,
                            bias=to_out_norm_b, eps=eps)

    # Apply output gate (cast gate to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # ---------------------------------------------------------------
    # Final linear projection back to original dimension
    # ---------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    output = F.linear(out_norm, to_out_w)   # [B, N, N, dim]  (float32)

    return output
