"""
Outgoing TriMul (Triangular Multiplicative Update) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over its last dimension (dim) using provided
   weight and bias.
2. Apply three linear projections (left, right, gates) without bias.
3. If a mask is present, multiply `left` and `right` by the mask.
4. Apply gating (sigmoid) to `left` and `right`.
5. Compute the pairwise contraction over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix multiplication:
       out_d = left_d @ right_dᵀ  (for every hidden‑dim d).
   The multiplication is performed in bfloat16 using a single
   `torch.bmm` over a batch of size B·hidden_dim.
6. Layer‑norm over the hidden dimension `hidden_dim` using a small Triton
   kernel (one program per (b,i,j) slice).
7. Multiply the normalized tensor by the `out_gate`.
8. Final linear projection back to `dim`.

Only the hidden‑dim LayerNorm is implemented in Triton; the heavy
contraction is done with an efficient batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,                # *float32  [B,N,N,H]
    out_ptr,                # *float32  [B,N,N,H]
    weight_ptr,             # *float32  [H]
    bias_ptr,               # *float32  [H]
    B,                      # i32
    N,                      # i32
    H,                      # i32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # i32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # i32
    eps: tl.constexpr,                                      # f32
    BLOCK_H: tl.constexpr                                   # compile‑time block size
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets inside the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base offsets for the 4‑D tensor
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load the hidden‑dim slice
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    )
    # ------------------------------------------------------------------
    # mean over hidden dim
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    # variance over hidden dim
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / H

    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ------------------------------------------------------------------
    # Apply learned gain (weight) and bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # ------------------------------------------------------------------
    # Store the normalized slice
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be ignored when nomask=True)
            - weights: dict of weight tensors
                * norm.weight, norm.bias
                * left_proj.weight, right_proj.weight
                * left_gate.weight, right_gate.weight, out_gate.weight
                * to_out_norm.weight, to_out_norm.bias
                * to_out.weight
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool, optional)

    Returns:
        torch.Tensor: processed tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    ##################################################################
    # 1. Input LayerNorm
    ##################################################################
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    B, N, _, _ = x.shape

    ##################################################################
    # 2. Linear projections & gating (all in float32)
    ##################################################################
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = (
        F.linear(x_flat, weights["left_gate.weight"])
        .view(B, N, N, hidden_dim)
        .sigmoid()
    )
    right_gate = (
        F.linear(x_flat, weights["right_gate.weight"])
        .view(B, N, N, hidden_dim)
        .sigmoid()
    )
    out_gate = (
        F.linear(x_flat, weights["out_gate.weight"])
        .view(B, N, N, hidden_dim)
        .sigmoid()
    )

    ##################################################################
    # 3. Apply pairwise mask (if any) and gating
    ##################################################################
    if not nomask:
        mask_exp = mask.to(dtype).unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    ##################################################################
    # 4. Pairwise contraction over k via batched GEMM (bfloat16)
    ##################################################################
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # bring hidden dimension to the 2nd axis and make tensors contiguous
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # reshape to (B*H, N, N) for batched matmul
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(dtype)
    )  # [B, N, N, H] in float32

    ##################################################################
    # 5. Hidden‑dim LayerNorm via Triton
    ##################################################################
    out_norm = torch.empty_like(out)

    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (total_programs,)

    BLOCK_H = hidden_dim  # compile‑time constant for the kernel

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b,
        s_i,
        s_j,
        s_h,
        s_b,
        s_i,
        s_j,
        s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    ##################################################################
    # 6. Apply outgoing gate
    ##################################################################
    out_scaled = out_norm * out_gate  # [B, N, N, H]

    ##################################################################
    # 7. Final linear projection back to `dim`
    ##################################################################
    out_scaled_flat = out_scaled.view(-1, hidden_dim)  # [B*N*N, H]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
