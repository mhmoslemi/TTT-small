import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Program IDs
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    pid_bh = tl.program_id(2) # batch*hidden block
    
    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # offsets in K dimension
    offs_k = tl.arange(0, BLOCK_K)
    
    # Linear offsets for a and b for current block pair. Accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Compute number of K blocks
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        
        # a pointers
        a_ptrs = a_ptr + pid_bh * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + (k_offset + offs_k)[None, :] * stride_ak
        # b pointers
        b_ptrs = b_ptr + pid_bh * stride_batch_b \
                 + (k_offset + offs_k)[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        
        # Mask for out of bounds
        a_mask = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        b_mask = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)
        
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)
        
        # dot product
        acc += tl.dot(a, b)
    
    # Store
    c_ptrs = c_ptr + pid_bh * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)
```

"""
TriMul (outgoing) forward pass.

The heavy cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM where each pair (batch, hidden) forms an
independent N×N matrix multiply.  All surrounding LayerNorms,
linear projections and gating are performed in PyTorch; the contraction
is offloaded to a Triton kernel that tiles the matrices in
BLOCK_M×BLOCK_N×BLOCK_K blocks and accumulates in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (all = seq_len)
    stride_am, stride_ak,                    # A strides: row, col
    stride_bk, stride_bn,                    # B strides: row, col
    stride_cm, stride_cn,                    # C strides: row, col
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where A,B,C are [BH, M, K], [BH, K, N], [BH, M, N]"""
    pid_m = tl.program_id(0)          # block row index
    pid_n = tl.program_id(1)          # block col index
    pid_bh = tl.program_id(2)         # batch*hidden index

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_off = k * BLOCK_K

        # Pointers for the current tile of A and B
        a_ptrs = a_ptr + pid_bh * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + (k_off + offs_k)[None, :] * stride_ak

        b_ptrs = b_ptr + pid_bh * stride_batch_b \
                 + (k_off + offs_k)[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Bounds checking
        a_mask = (offs_m[:, None] < M) & ((k_off + offs_k)[None, :] < K)
        b_mask = ((k_off + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Block matrix multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write back results
    c_ptrs = c_ptr + pid_bh * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of model parameters (see module definition)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---- 1. Input LayerNorm (float32) ----
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B, N, N, dim]

    # ---- 2. Cast to fp16 for the heavy part ----
    x_fp16 = x.to(torch.float16)

    # ---- 3. Linear projections (no bias) ----
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # ---- 4. Optional mask (broadcast) ----
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ---- 5. Gating ----
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))      # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ---- 6. Batched GEMM for the cubic contraction (Triton) ----
    B, N, _, H = left.shape  # H == hidden_dim

    # Rearrange tensors to (B, H, N, N) layout
    A = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)  i,k
    Bmat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  k,j

    BH = B * H
    A_flat = A.reshape(BH, N, N)   # (BH, M=N, K=N)
    B_flat = Bmat.reshape(BH, N, N)  # (BH, K=N, N)

    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)  # output buffer

    # Block sizes (tuned for Ampere GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # columns
        BH,                        # batch*hidden
    )

    batched_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,                              # M, N, K
        A_flat.stride(1), A_flat.stride(2),   # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),   # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),   # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = C_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ---- 7. Post‑processing (LayerNorm on hidden dim, final gate, output proj) ----
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gate (broadcasted)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # shape (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)           # [B, N, N, dim] float32

    return output
