import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)  # fp16
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=1.0)  # default 1.0

    left = left * m * lg
    right = right * m * rg

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Implements the core pairwise multiplication using torch.bmm (cuBLAS) and
    fuses mask and gating multiplication in a simple Triton kernel.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = F.linear(x, left_proj_weight)  # float32
    right = F.linear(x, right_proj_weight)

    # Cast to half for compute
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # keep float32

    # Prepare mask (half) if needed
    if nomask:
        mask = torch.ones_like(left, dtype=torch.float16, device=device)
    else:
        # mask_tensor shape [B,N,N]; broadcast to hidden dimension
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        mask = mask.expand_as(left)  # [B,N,N,hidden_dim]

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    # Fuse mask and gating with a Triton kernel
    total = left.numel()
    BLOCK = 65536  # 64K elements per thread block (tune)
    grid = (triton.cdiv(total, BLOCK),)
    mask_gate_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask.data_ptr(),
        total,
        BLOCK=BLOCK,
    )

    # Batched matmul using torch.bmm (half precision)
    B, N, _, _ = left.shape
    # reshape for batched multiplication
    left_mat = left.permute(0, 3, 1, 2).contiguous()    # (B, hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden_dim, N, N) where third dim is k, fourth is j

    P = B * hidden_dim
    left_mat = left_mat.view(P, N, N)
    right_mat = right_mat.view(P, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (P, N, N) half precision

    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, hidden_dim)
    out = out.to(torch.float32)  # bring back to float32 for following layers

    # Final LayerNorm over hidden dimension
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out gate and final linear projection
    out = out * out_gate  # broadcast over last dim
    out = F.linear(out, to_out_weight)  # (B,N,N,dim)

    return out

    left_ptr, right_ptr,          # half*  (output tensors, also inputs)
    left_gate_ptr, right_gate_ptr,# half*  (gate tensors)
    mask_ptr,                     # half*  (broadcasted mask)
    total_elems: tl.int32,        # total number of elements in left/right
    BLOCK: tl.constexpr,
):
    """
    In‑place ``left = left * mask * left_gate`` and ``right = right * mask * right_gate``.
    All tensors are flattened to 1‑D and have the same number of elements.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load values (half precision)
    l  = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs,mask=mask, other=0.0)
    m  = tl.load(mask_ptr + offs,      mask=mask, other=1.0)

    # Fuse mask and gates
    l = l * m * lg
    r = r * m * rg

    # Write back
    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool or 0/1 float) – ignored when `nomask` is True
            - weights: dict of model weight tensors (float32)
            - config: dict with keys:
                * "dim"          – input channel dimension
                * "hidden_dim"   – projection dimension
                * "nomask"       – bool, whether the mask should be applied

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Load model weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,),
                     weight=norm_weight, bias=norm_bias, eps=eps)   # float32

    # -----------------------------------------------------------------
    # Linear projections (float32 -> cast to float16)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_weight).to(torch.float16)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight).to(torch.float16)

    # -----------------------------------------------------------------
    # Gates (computed in float32, then cast)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))          # float32, kept for later

    # -----------------------------------------------------------------
    # Apply mask and gates
    # -----------------------------------------------------------------
    if nomask:
        # Simple elementwise multiplication when no mask is required
        left  = left  * left_gate
        right = right * right_gate
    else:
        # Broadcast mask to hidden_dim, convert to half
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)               # [B,N,N,1]
        mask = mask.expand(-1, -1, -1, hidden_dim).contiguous()          # [B,N,N,hidden_dim]

        # Ensure contiguity of all operands for the kernel
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask      = mask.contiguous()

        total = left.numel()
        BLOCK = 65536                          # tuneable; ~64K elements per thread block
        grid  = (triton.cdiv(total, BLOCK),)

        mask_gate_kernel[grid](
            left.data_ptr(),
            right.data_ptr(),
            left_gate.data_ptr(),
            right_gate.data_ptr(),
            mask.data_ptr(),
            total,
            BLOCK=BLOCK,
        )
        # After kernel left/right already contain mask*gate product

    # -----------------------------------------------------------------
    # Batched matrix multiplication: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # This is performed as a batched half‑precision GEMM via torch.bmm.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # shape = [B, N, N, hidden_dim]
    # Re‑arrange tensors to (B*hidden_dim, N, N) where the inner dimensions are (i,k) and (k,j)
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N) -> (i,k)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden_dim, N, N) -> (k,j)

    P = B * hidden_dim
    left_mat  = left_mat.view(P, N, N)   # (P, N, N)
    right_mat = right_mat.view(P, N, N)  # (P, N, N)

    # Half‑precision batched GEMM (Tensor‑Core friendly)
    out_mat = torch.bmm(left_mat, right_mat)          # (P, N, N), dtype=half

    # Reshape back to [B, N, N, hidden_dim] and convert to float32 for the rest
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # Output LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # -----------------------------------------------------------------
    # Apply final gating and projection
    # -----------------------------------------------------------------
    out = out * out_gate                     # broadcast over last dim (float32)
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
