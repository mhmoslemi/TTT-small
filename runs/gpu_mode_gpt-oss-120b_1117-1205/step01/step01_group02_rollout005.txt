"""TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The operation is:
    x = LayerNorm(x)                                    # [B, N, N, C]
    left  = Linear(x, left_proj)    # [B, N, N, H]
    right = Linear(x, right_proj)   # [B, N, N, H]
    left  = left * mask * sigmoid(Linear(x, left_gate))
    right = right * mask * sigmoid(Linear(x, right_gate))
    out   = Σ_k left[..., i, k, d] * right[..., j, k, d]   # i‑j matrix for each d
           = (left @ rightᵀ)   (batched over B·H)
    out   = LayerNorm(out) * sigmoid(Linear(x, out_gate))
    out   = Linear(out, to_out)   # back to C
The heavy Σ_k is a batched matrix multiplication performed by a Triton
kernel that works on fp16 data with fp32 accumulation.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ B) where B is already the transposed matrix
#   A : [B*H, N, N]   (fp16)
#   B : [B*H, N, N]   (fp16)   (rightᵀ)
#   C : [B*H, N, N]   (fp16)
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (int)
    stride_am, stride_ak,          # A strides (row, col)
    stride_bk, stride_bn,          # B strides (row, col)
    stride_cm, stride_cn,          # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch index (b·h)

    # tile indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulation buffer in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # A block pointer: (batch, row, col)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        # B block pointer (already transposed)
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # store result in fp16
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point compatible with the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32, CUDA)
            - mask       : torch.Tensor of shape [B, N, N] or None
            - weights    : dict of model parameters (torch.Tensors)
            - config     : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # 1. LayerNorm over the last dimension (float32 for stability)
    # ------------------------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------- #
    # 2. Cast to fp16 for the heavy linear projections
    # ------------------------------------------------------------------- #
    x_norm_fp16 = x_norm.to(torch.float16)

    # projection weights (fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_norm_fp16, left_proj_w)   # [B,N,N,H] fp16
    right = torch.nn.functional.linear(x_norm_fp16, right_proj_w)  # [B,N,N,H] fp16

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4. Gates (computed in float32, then cast to fp16 for multiplication)
    # ------------------------------------------------------------------- #
    left_gate = torch.nn.functional.linear(
        x_norm, weights["left_gate.weight"]
    ).sigmoid().to(torch.float16)           # [B,N,N,H] fp16

    right_gate = torch.nn.functional.linear(
        x_norm, weights["right_gate.weight"]
    ).sigmoid().to(torch.float16)           # [B,N,N,H] fp16

    out_gate_fp32 = torch.nn.functional.linear(
        x_norm, weights["out_gate.weight"]
    ).sigmoid()                             # [B,N,N,H] float32

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Prepare tensors for the batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------- #
    # reshape to [B, H, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose the last two dims of right (so we can do A @ B)
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B,H,N,N]

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    left_flat = left_perm.view(total_batch, N, N)   # [B*H, N, N]
    right_flat = right_T.view(total_batch, N, N)    # [B*H, N, N]
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # strides in elements
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # ------------------------------------------------------------------- #
    # 6. Launch the Triton GEMM kernel
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # block rows
        triton.cdiv(N, BLOCK_N),   # block cols
        total_batch,               # batch dimension
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, H] (float32) for the output LN and linear
    # ------------------------------------------------------------------- #
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------- #
    # 8. Output LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate_fp32  # broadcasted element‑wise gate (float32)

    # final linear back to dim
    output = torch.nn.functional.linear(
        out, weights["to_out.weight"]
    )  # [B, N, N, C] float32

    return output
