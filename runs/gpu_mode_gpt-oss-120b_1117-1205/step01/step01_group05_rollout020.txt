"""
TriMul (outgoing) forward implementation with a fused Triton batched GEMM.
The heavy tensor contraction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed as a batched matrix multiplication (left @ rightᵀ) using a
custom Triton kernel. All other operations (LayerNorms, linear projections,
gates, masking) are carried out in PyTorch (half‑precision where possible).
The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for A (batch, M, K) * B (batch, K, N) -> C (batch, M, N)
    All tensors are assumed to be in row‑major layout and stored as fp16.
    Accumulation is performed in fp32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to C tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Load B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # Tensor‑core dot (fp16 → fp32 accumulation)
        acc += tl.dot(a, b)

    # Store C tile as fp16
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    TriMul forward (outgoing) implementation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], float32
            - mask_tensor  : torch.Tensor or None, shape [B, N, N]
            - weights_dict : dict of model parameters
            - config_dict  : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------- unpack --------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------- LayerNorm (input) --------------------
    gamma = weights["norm.weight"].to(torch.float32).to(device)
    beta = weights["norm.bias"].to(torch.float32).to(device)

    # compute mean / var over last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta                # [B, N, N, dim]

    # half‑precision work for the heavy paths
    x_h = x_norm.to(torch.float16)

    # -------------------- Linear projections & gates --------------------
    w_left = weights["left_proj.weight"].to(torch.float16).to(device)      # (hidden, dim)
    w_right = weights["right_proj.weight"].to(torch.float16).to(device)

    w_left_gate = weights["left_gate.weight"].to(torch.float16).to(device)
    w_right_gate = weights["right_gate.weight"].to(torch.float16).to(device)
    w_out_gate = weights["out_gate.weight"].to(torch.float16).to(device)

    left = F.linear(x_h, w_left)          # [B, N, N, hidden]
    right = F.linear(x_h, w_right)

    # optional mask (broadcasted on hidden dim)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # gates (sigmoid in fp16)
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()   # later applied after to_out_norm

    left = left * left_gate
    right = right * right_gate

    # -------------------- Batched GEMM: out = left @ rightᵀ --------------------
    B, N, _, H = left.shape   # H == hidden_dim
    # reshape to (batch*hidden, N, N) for GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # transpose k dimension

    out_mat = torch.empty_like(left_mat)  # fp16 output

    # strides (in elements)
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat_T.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tiling parameters – chosen for B200 tensor‑core efficiency
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # -------------------- Output LayerNorm --------------------
    gamma_out = weights["to_out_norm.weight"].to(torch.float32).to(device)
    beta_out = weights["to_out_norm.bias"].to(torch.float32).to(device)

    out_f32 = out.to(torch.float32)
    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # Apply the output gate (broadcasted)
    out_norm = out_norm * out_gate.to(torch.float32)

    # -------------------- Final linear projection --------------------
    w_to_out = weights["to_out.weight"].to(torch.float32).to(device)   # (dim, hidden)
    output = F.linear(out_norm, w_to_out)   # [B, N, N, dim], float32

    return output
