@triton.jit
def _batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = A_ptr + batch_id * stride_az + offs_m[:, None] * stride_am + 0 * stride_ak # placeholder
    b_ptrs = B_ptr + batch_id * stride_bz + 0 * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + batch_id * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + batch_id * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    c_ptrs = C_ptr + batch_id * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)  # shape [B,N,N,dim]

    # Convert to half for speed
    x_h = x.to(torch.float16)

    # Linear projections
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    left = F.linear(x_h, w_left)   # shape [B,N,N,hidden_dim]
    right = F.linear(x_h, w_right)

    # Apply mask if needed
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)  # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid)
    left_gate = F.linear(x_h, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare matrices for batched GEMM
    # left: [B,N,N,H] -> [B*H, N, N]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # right_T: transpose last two dimensions of right to align K
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Allocate output
    out_mat = torch.empty((bs * hidden_dim, N, N), dtype=torch.float16, device=device)

    # strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # choose 32 maybe better compute to memory ratio

    grid = (
        bs * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # Reshape back
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous() # [B,N,N,H] in fp16

    # Output LayerNorm over hidden_dim
    out_f32 = out.to(torch.float32)
    gamma_out = weights["to_out_norm.weight"]
    beta_out = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_f32, (hidden_dim,), weight=gamma_out, bias=beta_out, eps=eps)

    # Apply out gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to dim
    w_to_out = weights["to_out.weight"]
    output = F.linear(out_norm, w_to_out)  # shape [B,N,N,dim]
    return output
```

"""
TriMul outgoing forward pass (AlphaFold3) with Triton acceleration.

The heavy N³ contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM for each hidden dimension d:
    left_mat (B*H, N, N) @ right_mat (B*H, N, N).T

All other operations (LayerNorms, Linear projections, gates) are performed
in PyTorch (mixed fp16/fp32). The contraction uses a custom Triton kernel
with 64×64 tiles and fp16 inputs accumulating in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ B  (A: [batch, M, K], B: [batch, K, N], C: [batch, M, N])
    All tensors are row‑major. Uses fp16 for A/B, fp32 accumulation.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input, mask, weights, config)
            input : torch.Tensor [B, N, N, dim] (float32)
            mask  : torch.Tensor [B, N, N] (bool/float) or None
            weights: dict of torch.Tensors with model parameters
            config : dict containing "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32) – TriMul forward output.
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for projections/gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear projections (no bias)
    left = F.linear(
        x_h, weights["left_proj.weight"].to(torch.float16)
    )  # [B,N,N,hidden_dim] fp16
    right = F.linear(
        x_h, weights["right_proj.weight"].to(torch.float16)
    )  # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 3. Apply mask if present (mask broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_h, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_h, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM for the N³ contraction (Trimus outgoing)
    #    left   : (B, N, N, H) -> (B*H, N, N)
    #    right_T: (B, N, N, H) with dimensions swapped -> (B*H, N, N)
    # ------------------------------------------------------------------
    # reshape + transpose
    left_mat = (
        left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    )  # (B*H, M, K)
    right_mat = (
        right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    )  # (B*H, K, N)

    out_mat = torch.empty(
        (B * hidden_dim, N, N), dtype=torch.float16, device=device
    )

    # strides for Triton
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tiling
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply out gate (converted to fp32)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # ------------------------------------------------------------------
    output = F.linear(out_norm, weights["to_out.weight"])  # [B, N, N, dim] float32

    return output
