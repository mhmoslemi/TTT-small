```python
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr = 2,
):
    pid = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_batch = tl.program_id(axis=2)

    # offsets
    off_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute start offsets for each K block
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # a block pointers
        a_ptrs = a_ptr + pid_batch * stride_batch_a \
                 + off_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        # b block pointers
        b_ptrs = b_ptr + pid_batch * stride_batch_b \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + off_n[None, :] * stride_bn

        # masks for out-of-bound
        a_mask = (off_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (off_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)  # fp32 accumulation

    # Store result
    c_ptrs = c_ptr + pid_batch * stride_batch_c \
             + off_m[:, None] * stride_cm \
             + off_n[None, :] * stride_cn
    c_mask = (off_m[:, None] < M) & (off_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # 1. Input layernorm
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # Cast to half for heavy ops
    x = x.to(torch.float16)

    # 2. Projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w) # same

    # 3. Mask
    if mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # 4. Gates
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # 5. Contract via Triton kernel
    B, N, _, H = left.shape
    # Permute to bring hidden dim as second dimension
    A = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) => rows=i, cols=k
    Bmat = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) => rows=k, cols=j

    # Flatten batch and hidden
    BH = B * H
    A_flat = A.reshape(BH, N, N)

"""Triton‑accelerated forward pass for the outgoing TriMul operation.

The heavy cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM where each (batch, hidden) pair computes
a matrix multiplication A @ Bᵀ with A and B of shape (N, N).  All
pre‑ and post‑processing (LayerNorm, linear projections, gating) is performed
with PyTorch; only the O(N³·hidden_dim) core is executed on the GPU via
a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr,  # [BH, M, K]   (fp16)
    b_ptr,  # [BH, K, N]   (fp16)   (B is already transposed)
    c_ptr,  # [BH, M, N]   (fp16)
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B  (A: MxK, B: KxN).

    The kernel treats the leading dimension BH as the batch dimension and
    computes independent GEMMs for each batch element.
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_b = tl.program_id(2)  # batch (BH)

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the current batch
    a_batch_ptr = a_ptr + pid_b * stride_batch_a
    b_batch_ptr = b_ptr + pid_b * stride_batch_b
    c_batch_ptr = c_ptr + pid_b * stride_batch_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Pointers for the current K‑tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Compute masks for partial tiles
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        acc += tl.dot(a, b)   # fp32 accumulation

    # Write back
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            input: torch.Tensor [B, N, N, dim] (float32)
            mask:  torch.Tensor [B, N, N]   (float32) – may be all‑ones or None
            weights: dict of model parameters (see module definition)
            config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # float32

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)          # (dim, hidden_dim)
    right_proj_w = weights["right_proj.weight"].to(device)        # (dim, hidden_dim)

    left = F.linear(x, left_proj_w)    # [B, N, N, H]  float32
    right = F.linear(x, right_proj_w)  # [B, N, N, H]  float32

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Gating (computed in float32, then cast)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H] float32
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,H] float32
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))      # [B,N,N,H] float32

    # Cast to half‑precision for the heavy contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM via Triton: out = left @ rightᵀ
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # left shape: [B, N, N, H]

    # Rearrange so that hidden dimension becomes the second axis
    A = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N) – rows i, cols k
    Bmat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – rows k, cols j (transposed)

    # Collapse batch and hidden dimensions → independent matrices
    BH = B * H
    A_flat = A.reshape(BH, N, N)
    B_flat = Bmat.reshape(BH, N, N)

    # Output buffer
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)

    # Tuned block sizes (powers of two, fit well into shared memory)
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 64 if N >= 64 else 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # columns
        BH,                        # batch*hidden
    )

    batched_gemm_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,
        A_flat.stride(1), A_flat.stride(2),   # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),   # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),   # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Restore original layout: [B, N, N, H]
    out = C_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -----------------------------------------------------------------
    # 6. Post‑processing: LayerNorm on hidden dim, gating, final linear
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate  # element‑wise gating (float32)

    to_out_w = weights["to_out.weight"].to(device)  # (hidden_dim, dim)
    output = F.linear(out_norm, to_out_w)          # [B, N, N, dim] float32

    return output

    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # col block
    pid_b = tl.program_id(2)   # batch (BH)

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the current batch
    a_batch = a_ptr + pid_b * stride_batch_a
    b_batch = b_ptr + pid_b * stride_batch_b
    c_batch = c_ptr + pid_b * stride_batch_c

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        a_ptrs = a_batch + offs_m[:, None] * stride_am + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a_mask = (offs_m[:, None] < M) & ((k_off + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_off + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        acc += tl.dot(a, b)   # result accumulated in fp32

    # Write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            input : torch.Tensor of shape [B, N, N, dim] (float32)
            mask  : torch.Tensor of shape [B, N, N] (float32) – may be None or all‑ones
            weights: dict of model parameters
            config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # float32

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)          # (dim, hidden_dim)
    right_proj_w = weights["right_proj.weight"].to(device)        # (dim, hidden_dim)

    left = F.linear(x, left_proj_w)    # [B, N, N, H] float32
    right = F.linear(x, right_proj_w)  # [B, N, N, H] float32

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Gating (computed in float32, then cast)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H] float32
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,H] float32
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # [B,N,N,H] float32

    # Cast to half‑precision for the expensive contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM via Triton (out = left @ rightᵀ)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape                     # left: [B, N, N, H]

    # Move hidden dimension next to batch for efficient flattening
    A = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N) – rows i, cols k
    Bmat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – rows k, cols j (transposed)

    BH = B * H
    A_flat = A.reshape(BH, N, N)
    B_flat = Bmat.reshape(BH, N, N)

    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)

    # Simple heuristic for block sizes
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 64 if N >= 64 else 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # columns
        BH,                        # batch * hidden
    )

    batched_gemm_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,
        A_flat.stride(1), A_flat.stride(2),   # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),   # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),   # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Restore shape [B, N, N, H]
    out = C_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -----------------------------------------------------------------
    # 6. Post‑processing: LayerNorm on hidden dim, final gating, projection
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate                      # element‑wise gating (float32)

    to_out_w = weights["to_out.weight"].to(device)      # (hidden_dim, dim)
    output = F.linear(out_norm, to_out_w)              # [B, N, N, dim] float32

    return output
