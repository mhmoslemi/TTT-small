    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    x_f16 = x.to(torch.float16)

    # Compute projections
    left = torch.nn.functional.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # Apply mask if provided
    if mask is not None:

        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_f16
        right = right * mask_f16

    # compute gates (float16)
    left_gate = torch.nn.functional.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

"""
TriMul "outgoing" forward implementation using Triton.

The core operation is the tri‑multiplicative update:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

We compute left/right projections and their sigmoid gates in PyTorch (half precision)
and then perform a batched GEMM (A @ B) for every (batch, hidden) pair via a
custom Triton kernel. The result is normalized, multiplied by an output gate,
and projected back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) @ B (K×N) → C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # problem sizes
    stride_am, stride_ak,                    # strides for A (M, K)
    stride_bk, stride_bn,                    # strides for B (K, N)
    stride_cm, stride_cn,                    # strides for C (M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # which batch (batch * hidden)
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # Pointers to the tiles of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] or None
            - weights: dict of torch.Tensors with model weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections and gates (half precision)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # Projections
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, hidden]
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # Optional mask
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # Gates (computed from the original normalized tensor)
    left_gate = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()          # float32

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core TriMul via batched GEMM (Triton)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) where the last two dims are (i,k) and (k,j)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N) == (i,k)
    right = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N) == (k,j)

    B, H, N, _ = left.shape
    total_batches = B * H

    # Collapse batch and hidden dimensions for the kernel
    left_mat = left.view(total_batches, N, N)
    right_mat = right.view(total_batches, N, N)
    out_mat = torch.empty_like(left_mat)  # FP16 output

    # Choose block sizes based on N
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,          # M
        N,          # N
        N,          # K
        left_mat.stride(1),   # stride_am (row stride)
        left_mat.stride(2),   # stride_ak (col stride)
        right_mat.stride(1),  # stride_bk (row stride of B)
        right_mat.stride(2),  # stride_bn (col stride of B)
        out_mat.stride(1),    # stride_cm
        out_mat.stride(2),    # stride_cn
        left_mat.stride(0),   # batch stride A
        right_mat.stride(0),  # batch stride B
        out_mat.stride(0),    # batch stride C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden] and convert to FP32 for the rest
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, output gate, and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the output gate (broadcasted)
    out_norm = out_norm * out_gate

    # Final linear projection back to original dim
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
