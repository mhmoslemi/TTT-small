"""
TriMul outgoing forward pass with a Triton kernel for the core
N×N×N multiplication.

Algorithm
---------
1. Layer‑norm over the channel dimension of the input tensor.
2. Linear projections (left/right) and per‑position gating (left/right/out).
   All projections are performed in fp16 for speed.  The optional mask
   (if provided) is applied after the projections.
3. Core TriMul: for each batch and hidden channel compute
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication A @ Bᵀ where
        A = left.permute(0,3,1,2).reshape(B*H,N,N)
        B = right.permute(0,3,2,1).reshape(B*H,N,N)
   A Triton GEMM kernel performs the multiplication in fp16 with
   fp32 accumulation.
4. Layer‑norm over the hidden dimension, gated by the out‑gate,
   and a final linear projection back to the original channel size.
   The final output is fp32 with shape [B, N, N, dim].

The kernel processes many (batch*hidden) independent GEMMs in parallel.
Tile sizes are chosen heuristically based on sequence length.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ B   (A: M×K, B: K×N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # which (batch, hidden) pair
    pid_m   = tl.program_id(1)   # block row
    pid_n   = tl.program_id(2)   # block col

    # Pointers to the current batch slice
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k):
        # Global k offset for this tile
        k_offset = k_iter * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + k_offset
        mask_k = offs_k < K

        # Pointers to the tiles of A and B
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)            # (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N)

    # Store the result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 0. Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)

    # Cast to fp16 for the heavy lifting
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, hidden]
    right = F.linear(x_f16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (applied after projection)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f16
        right = right * mask_f16

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = F.linear(x_f16, left_gate_w).sigmoid()
    right_gate = F.linear(x_f16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_f16, out_gate_w).sigmoid()   # shape [B, N, N, hidden]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul via Triton GEMM
    #    Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B, hidden, N, N) and then merge first two dims
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N) where dim2=k, dim3=j

    total_batches = B * hidden
    left_mat  = left_perm.reshape(total_batches, N, N)
    right_mat = right_perm.reshape(total_batches, N, N)
    out_mat   = torch.empty_like(left_mat)   # fp16

    # Block sizes (heuristic)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides for the flattened matrices
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_a_batch = left_mat.stride(0)

    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_b_batch = right_mat.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_c_batch = out_mat.stride(0)

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                     # M, N, K
        stride_am, stride_ak, stride_a_batch,
        stride_bk, stride_bn, stride_b_batch,
        stride_cm, stride_cn, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden] in fp32
    out = out_mat.reshape(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # Apply out‑gate (cast to fp32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to original dimension
    to_out_w = weights["to_out.weight"]            # (dim, hidden)
    output = F.linear(out, to_out_w)               # [B, N, N, dim]

    return output
