import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_kernel(out_ptr, gate_ptr, total_elems, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    o = tl.load(out_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, o * g, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.
    Implements all dense operations in PyTorch for speed, and uses a small
    Triton kernel to apply the final gating (out_norm * out_gate) in place.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # shape [B,N,N,dim]

    B, N, _, _ = x.shape

    # 2. Linear projections (float32)
    x_flat = x.reshape(-1, dim)  # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # 3. Apply mask (if present) and gates
    if not nomask:
        mask_exp = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # 4. Pairwise contraction over k using batched matrix multiply (bf16)
    # Cast to bfloat16 for bandwidth reduction
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # bring hidden dimension to batch dimension
    left_perm = left_bf.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H,N,N)
    right_perm = right_bf.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N) # (B*H,N,N)

    # bmm: left @ right^T
    out_bf = torch.bmm(left_perm, right_perm.transpose(1, 2))

"""
Triton‑accelerated forward pass for the outgoing TriMul module.

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension (`dim`).
2. Linear projections (left/right) and three gating vectors (left_gate,
   right_gate, out_gate) using the supplied weight matrices.
3. Optional pairwise mask is applied, then the gates are fused.
4. Pairwise contraction over the sequence dimension `k`:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication per hidden channel.
   The projection tensors are cast to bfloat16 before the multiplication to
   reduce memory traffic, then cast back to float32.
5. Layer‑norm over the hidden dimension (`hidden_dim`) using the fast
   torch implementation.
6. Apply the outgoing gate (`out_gate`) with a tiny Triton kernel
   (in‑place elementwise multiplication).
7. Final linear projection back to the original feature dimension `dim`.

Only the final gating step is executed on Triton; the remaining dense
operations are performed with highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_kernel(out_ptr, gate_ptr, total_elems, BLOCK: tl.constexpr):
    """In‑place multiplication out_ptr *= gate_ptr."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    o = tl.load(out_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, o * g, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored.
            - weights: dict of weight tensors needed by the module.
            - config: dict with keys "dim", "hidden_dim", "nomask".

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gating (float32)
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(
        B, N, N, hidden_dim
    ).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(
        B, N, N, hidden_dim
    ).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(
        B, N, N, hidden_dim
    ).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) and gates
    if not nomask:
        mask_exp = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction over k (batched GEMM, BF16 for bandwidth)
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Rearrange to (B, H, N, N) and collapse batch*hidden into one dim
    left_perm = left_bf.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_perm = right_bf.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_bf = torch.bmm(left_perm, right_perm.transpose(1, 2))

    # Restore shape [B, N, N, H] and cast back to float32
    out = (
        out_bf.to(dtype)
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm (fast PyTorch implementation)
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    ).contiguous()

    # ------------------------------------------------------------------
    # 6. Apply the outgoing gate using a tiny Triton kernel (in‑place)
    out_gate = out_gate.contiguous()
    total_elements = out_norm.numel()
    BLOCK = 128
    grid = (triton.cdiv(total_elements, BLOCK),)
    mul_gate_kernel[grid](
        out_norm,
        out_gate,
        total_elements,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_norm.view(-1, hidden_dim)          # (B*N*N, H)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
