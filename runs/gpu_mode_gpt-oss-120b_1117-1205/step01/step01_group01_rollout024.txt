@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # pid for M block
    pid_n = tl.program_id(1)  # pid for N block
    pid_batch = tl.program_id(2)  # batch index

    # offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute pointers for batch offset
    a_ptr = a_ptr + pid_batch * stride_batch_a
    b_ptr = b_ptr + pid_batch * stride_batch_b
    c_ptr = c_ptr + pid_batch * stride_batch_c

    # Compute base pointers for a and b tiles
    # We'll compute per k tile inside loop

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over tiles of K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # a_tile ptr
        a_tile_ptr = a_ptr + (offs_m[:, None] * stride_am) + ((k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        # b_tile ptr (note B shape is K,N)
        b_tile_ptr = b_ptr + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # masks
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_tile_ptr, mask=a_mask, other=0.0)
        b = tl.load(b_tile_ptr, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)
    # Write back
    c_tile_ptr = c_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=c_mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Convert to fp16 for compute
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left = torch.nn.functional.linear(x_fp16, left_proj_w)  # shape [B,N,N,H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    # Mask
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gates
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Build matrices for batched GEMM
    B, N, _, H = left.shape
    # Permute to (B,H,N,N) for A and B_T
    A = left.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)
    Bmat = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) (k,j)
    BH = B * H
    A_flat = A.view(BH, N, N)
    B_flat = Bmat.view(BH, N, N)
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)

    # Kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)
    batched_gemm_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,
        A_flat.stride(1), A_flat.stride(2),
        B_flat.stride(1), B_flat.stride(2),
        C_flat.stride(1), C_flat.stride(2),
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape to original layout
    out = C_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # Post-processing
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)   # out_gate currently half

    to_out_w = weights["to_out.weight"].to(device)  # shape (dim, hidden_dim)
    output = torch.nn.functional.linear(out_norm, to_out_w)

    return output

"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

The expensive contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batch of independent N×N matrix multiplications (one per batch×head).
We implement this with a custom Triton kernel (blocked GEMM) while the
remaining LayerNorms, linear projections and gates stay in PyTorch.
All heavy work uses fp16 arithmetic with fp32 accumulation,
the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr,               # [BH, M, K]  (fp16)
    b_ptr,               # [BH, K, N]  (fp16)
    c_ptr,               # [BH, M, N]  (fp16)
    M, N, K,             # matrix dimensions (int)
    stride_am, stride_ak,           # A: row stride, col stride
    stride_bk, stride_bn,           # B: row (k) stride, col (n) stride
    stride_cm, stride_cn,           # C: row stride, col stride
    stride_batch_a, stride_batch_b, stride_batch_c,   # batch stride for each tensor
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # row block index
    pid_n = tl.program_id(1)            # col block index
    pid_b = tl.program_id(2)            # batch*head index

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # add batch offset to base pointers
    a_ptr = a_ptr + pid_b * stride_batch_a
    b_ptr = b_ptr + pid_b * stride_batch_b
    c_ptr = c_ptr + pid_b * stride_batch_c

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # pointers for the A and B tiles
        a_tile_ptr = a_ptr + (offs_m[:, None] * stride_am) \
                     + ((k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        b_tile_ptr = b_ptr + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk) \
                     + (offs_n[None, :] * stride_bn)

        # masks for out‑of‑bounds
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_tile_ptr, mask=a_mask, other=0.0)
        b = tl.load(b_tile_ptr, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # write back
    c_tile_ptr = c_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]     (float32)
            - mask:         torch.Tensor [B, N, N] (float32) or None
            - weights: dict of model parameters (torch.Tensors)
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)   # [B, N, N, dim]

    # Use fp16 for the heavy path
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_w) # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM via Triton: out = left @ right^T over k
    # -----------------------------------------------------------------
    B, N, _, H = left.shape                      # shapes: [B, N, N, H]
    # reshape to (B, H, N, N) where rows=i (or k) and cols=k (or j)
    A = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N) : (i,k)
    Bmat = right.permute(0, 3, 2, 1).contiguous()      # (B, H, N, N) : (k,j)

    BH = B * H
    A_flat = A.view(BH, N, N)          # (BH, M=N, K=N)
    B_flat = Bmat.view(BH, N, N)       # (BH, K=N, N)
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)

    # Blocking parameters (tuned for B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)
    batched_gemm_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,
        A_flat.stride(1), A_flat.stride(2),   # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),   # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),   # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore layout: [B, N, N, hidden_dim] (float32)
    out = C_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Post‑processing: LayerNorm on hidden_dim, gate, final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate (already fp32 after casting)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)            # [B, N, N, dim]

    return output
