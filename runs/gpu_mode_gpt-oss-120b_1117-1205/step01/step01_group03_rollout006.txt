@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Determine masks
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Compute base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # Pointers to tile of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)  # A (M,K), B(K,N) -> (M,N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # 1. LayerNorm
    # norm_weight = weights["norm.weight"] shape (dim,)
    # norm_bias = weights["norm.bias"] shape (dim,)
    # Use torch's layernorm? We'll implement manually: x_normalized = (x - mean)/sqrt(var+eps) * w + b
    # Compute mean and var across dim axis.
    # Cast weight and bias to appropriate dtype and device.

    # It's necessary to compute mean and var per last dimension (dim). Return shape (B,N,N,1)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x = (input_tensor - mean) * inv_std
    x = x * weights["norm.weight"] + weights["norm.bias"]
    # x shape (B,N,N,dim)

    # 2. Projections and gates using FP16
    x_f16 = x.to(torch.float16)

    # weight casting
    left_proj_w = weights["left_proj.weight"].to(torch.float16)  # shape (hidden, dim) or (dim, hidden)? 
    # In nn.Linear(dim, hidden) the weight shape is (hidden, dim). In torch.nn.functional.linear expects weight shape (out_features, in_features)
    # So shape (hidden, dim). We'll keep as is.

    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # left and right via F.linear
    left = F.linear(x_f16, left_proj_w)  # shape (B,N,N,hidden)
    right = F.linear(x_f16, right_proj_w)

    # mask
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
    else:
        mask_f16 = torch.ones((B,N,N,1), dtype=torch.float16, device=device)

    left = left * mask_f16
    right = right * mask_f16

    # gates
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_f16, left_gate_w).sigmoid()
    right_gate = F.linear(x_f16, right_gate_w).sigmoid()
    out_gate = F.linear(x_f16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # 3. Core multiplication via Triton
    # Permute to (B, hidden, N, N) for both left and right
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    total_batches = B * hidden
    left_mat = left.reshape(total_batches, N, N)
    right_mat = right.reshape(total_batches, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)  # allocate output in fp16

    # Choose block sizes based on N (we need to produce values as constants)
    if N <= 128:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:  # N <= 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Launch grid
    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_gemm_kernel[grid](
        left_mat,  # a_ptr
        right_mat, # b_ptr
        out_mat,   # c_ptr
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        # For B transposed: stride_bk = right_mat.stride(2), stride_bn = right_mat.stride(1)
        right_mat.stride(2), right_mat.stride(1),
        out_mat.stride(1), out_mat.stride(2),
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape output back to (B,N,N,hidden)
    out = out_mat.reshape(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # 4. Output LayerNorm
    # to_out_norm weight and bias shapes (hidden,)
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(dim=-1, keepdim=True)

"""
TriMul "outgoing" forward pass with a fused Triton batched GEMM.

The core operation is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the projected and gated tensors of shape
[batch, N, N, hidden_dim].  This can be expressed as a batched matrix
multiplication left @ right.T for each hidden channel.  The kernel
computes a GEMM for every `(batch * hidden_dim)` slice using fp16
inputs and fp32 accumulation (TensorCore friendly).  All other
operations (LayerNorm, linear projections, gates, and the final
LayerNorm & linear) are performed with PyTorch (fp16/fp32 where
appropriate).

The implementation is fully self‑contained:
  * `batched_gemm_kernel` – Triton kernel that computes C = A @ Bᵀ.
  * `custom_kernel` – entry point matching the required signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # problem sizes (all = seq_len)
    stride_am, stride_ak,               # strides of A (M, K)
    stride_bk, stride_bn,               # strides of B (K, N) – note transposed access
    stride_cm, stride_cn,               # strides of C (M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # stride between batch slices
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM computing C = A @ Bᵀ for each batch slice."""
    pid_batch = tl.program_id(0)    # batch * hidden index
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block col

    # base pointers for this batch slice
    a_base = a_ptr + pid_batch * stride_batch_a
    b_base = b_ptr + pid_batch * stride_batch_b
    c_base = c_ptr + pid_batch * stride_batch_c

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to the current tiles of A and B (B is accessed transposed)
    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write result back (store as fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] or None
            - weights_dict: mapping from weight name to torch.Tensor (already on device)
            - config_dict: contains at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack arguments
    inp, mask, weights, config = data
    device = inp.device
    dtype = inp.dtype
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (per last dimension)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"].to(dtype).to(device)
    b_norm = weights["norm.bias"].to(dtype).to(device)
    mean = inp.mean(dim=-1, keepdim=True)
    var = ((inp - mean) ** 2).mean(dim=-1, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x = (inp - mean) * inv_std
    x = x * w_norm + b_norm                      # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections & masks (use fp16 for speed)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # projection weights (shape [hidden, dim])
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, w_left)   # [B, N, N, hidden]
    right = F.linear(x_f16, w_right)

    # mask handling
    if mask is None:
        mask_f16 = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
    left = left * mask_f16
    right = right * mask_f16

    # gating weights (shape [hidden, dim])
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_f16, w_left_gate).sigmoid()
    right_gate = F.linear(x_f16, w_right_gate).sigmoid()
    out_gate = F.linear(x_f16, w_out_gate).sigmoid()

    # apply elementwise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core TriMul via batched GEMM: out = left @ rightᵀ
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) for the kernel
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    total_batches = B * hidden
    left_mat = left.reshape(total_batches, N, N)
    right_mat = right.reshape(total_batches, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # Simple heuristics for tile sizes
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_mat,                         # a_ptr
        right_mat,                        # b_ptr (will be read transposed)
        out_mat,                          # c_ptr
        N, N, N,                          # M, N, K
        left_mat.stride(1), left_mat.stride(2),               # stride_am, stride_ak
        # transposed access: stride over K is the last dimension, over N is the middle one
        right_mat.stride(2), right_mat.stride(1),              # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),                  # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden] and convert to fp32
    out = out_mat.reshape(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gate, and final projection
    # ------------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"].to(dtype).to(device)
    b_out_norm = weights["to_out_norm.bias"].to(dtype).to(device)

    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(dim=-1, keepdim=True)
    inv_std_out = torch.rsqrt(var_out + eps)
    out_norm = (out - mean_out) * inv_std_out
    out_norm = out_norm * w_out_norm + b_out_norm

    # apply the output gate (already sigmoid-ed)
    out_norm = out_norm * out_gate.to(dtype)

    # final linear projection back to original dimension
    w_to_out = weights["to_out.weight"].to(dtype).to(device)  # shape (dim, hidden)
    output = F.linear(out_norm, w_to_out)   # [B, N, N, dim]

    return output
