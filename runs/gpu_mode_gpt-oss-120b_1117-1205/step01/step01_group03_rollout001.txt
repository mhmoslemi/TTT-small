def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # x_norm shape [bs,N,N,dim] float32
    # Use F.layer_norm with normalized_shape = (dim,)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Switch to float16 for speed
    x_f16 = x.to(torch.float16)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(torch.float16) # shape (hidden, dim) in pytorch linear weight is (out_features, in_features)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_weight) # gives (bs,N,N,hidden)
    right = F.linear(x_f16, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_f16, left_gate_weight).sigmoid()
    right_gate = F.linear(x_f16, right_gate_weight).sigmoid()
    out_gate = F.linear(x_f16, out_gate_weight).sigmoid()

    # Mask handling
    if mask is None:
        mask_expand = torch.ones(bs, N, N, hidden, dtype=torch.float16, device=device)
    else:
        # mask shape [bs,N,N], may be bool or float
        mask_f = mask.to(torch.float16)
        mask_expand = mask_f.unsqueeze(-1).expand(-1, -1, -1, hidden).contiguous()

    # Ensure all tensors contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_expand = mask_expand.contiguous()

    # Apply mask and gates via Triton kernel
    total = left.numel()  # bs*N*N*hidden
    BLOCK = 1024  # can be larger but keep moderate
    grid = (triton.cdiv(total, BLOCK),)
    apply_mask_gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_expand,
        total,
        BLOCK=BLOCK,
    )

    # Core TriMul using batched matmul via torch.bmm
    # left, right are now gated and masked
    # reshape: (bs*hidden, N, N)
    left_mat = left.view(bs * hidden, N, N)
    right_mat = right.view(bs * hidden, N, N)

    # Compute out_mat = left_mat @ right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [bs, N, N, hidden]
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate (convert to float32 maybe)
    out_gate = out_gate.to(out.dtype)  # out_gate is float16 currently (or maybe fp16)
    out = out * out_gate

    # Final linear projection (fp32)
    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden)
    # Ensure out shape is [bs,N,N,hidden] and dtype fp32
    out_fp32 = out.to(torch.float32)
    output = F.linear(out_fp32, to_out_weight)  # bias none

    return output

"""
Custom Triton‑accelerated implementation of the outgoing TriMul operation.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the channel dimension.
2. Project `x` to a hidden dimension (fp16) with two linear layers -> left/right.
3. Compute three sigmoid gates (fp16): left_gate, right_gate, out_gate.
4. Apply mask (if any) and the two gates to left/right in a fused Triton kernel.
5. Perform the core triangular multiplication:
       out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication `left @ right.T`.
   Use highly‑optimized cuBLAS via `torch.bmm`.
6. Layer‑norm the result over the hidden dimension, multiply by out_gate,
   and project back to the original channel dimension.

Only the element‑wise masking + gating step is implemented in Triton; the
large GEMM uses cuBLAS for maximal performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------
# Triton kernel: fuse mask and gating for left/right tensors
# -----------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    left_ptr,          # fp16 * (total,)
    right_ptr,         # fp16 * (total,)
    left_gate_ptr,     # fp16 * (total,)
    right_gate_ptr,    # fp16 * (total,)
    mask_ptr,          # fp16 * (total,)
    total_elems: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=m, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=m, other=0.0)
    l_gate    = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    r_gate    = tl.load(right_gate_ptr+ offs, mask=m, other=0.0)
    mask_val  = tl.load(mask_ptr      + offs, mask=m, other=0.0)

    left_out  = left  * l_gate * mask_val
    right_out = right * r_gate * mask_val

    tl.store(left_ptr  + offs, left_out,  mask=m)
    tl.store(right_ptr + offs, right_out, mask=m)


# -----------------------------------------------------------
# Entry point
# -----------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor [B, N, N, C] (float32)
              - mask_tensor: torch.Tensor [B, N, N] or None
              - weights_dict: contains all linear / layer‑norm parameters
              - config_dict: must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Project + Gates (fp16)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    left  = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden]
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    left_gate  = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Mask handling (expanded to hidden dim) – fp16
    # ------------------------------------------------------------------
    if mask is None:
        mask_exp = torch.ones(B, N, N, hidden, dtype=torch.float16, device=device)
    else:
        # mask: [B, N, N] → [B, N, N, hidden] (broadcast)
        mask_exp = mask.to(torch.float16).unsqueeze(-1).expand(-1, -1, -1, hidden).contiguous()

    # Ensure tensors are contiguous for the flattening used in the kernel
    left       = left.contiguous()
    right      = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_exp   = mask_exp.contiguous()

    # ------------------------------------------------------------------
    # 4. Fuse mask & gates with Triton
    # ------------------------------------------------------------------
    total = left.numel()  # B * N * N * hidden
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # Flatten to 1‑D pointers (stride‑1)
    apply_mask_gate_kernel[grid](
        left.view(-1),
        right.view(-1),
        left_gate.view(-1),
        right_gate.view(-1),
        mask_exp.view(-1),
        total,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Core TriMul via batched GEMM (cuBLAS)
    #    out = left @ rightᵀ for each (batch, hidden) slice
    # ------------------------------------------------------------------
    left_mat  = left.view(B * hidden, N, N)                      # (B*hidden, N, N)
    right_mat = right.view(B * hidden, N, N)                     # (B*hidden, N, N)

    # GEMM: (B*hidden, N, N) @ (B*hidden, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape [B, N, N, hidden]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate, and final projection (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden] float32

    # Apply output gate (convert gate to float32)
    out = out * out_gate.to(out.dtype)

    # Final linear projection to original dim (no bias)
    output = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return output
