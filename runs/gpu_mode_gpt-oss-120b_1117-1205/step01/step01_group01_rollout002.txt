"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel computes the cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left and right are linear projections of the input tensor,
optionally masked and gated.  The contraction is performed as a batch
of independent matrix multiplications (one per (batch, hidden) pair)
using a tiled GEMM kernel in Triton, while all surrounding
normalizations, gating, and linear layers are executed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers to A, B, C
    M, N, K,                                 # matrix dimensions (all = seq_len)
    stride_am, stride_ak,                    # strides for A: (row, col)
    stride_bk, stride_bn,                    # strides for B: (row, col)   (B is already transposed)
    stride_cm, stride_cn,                    # strides for C: (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,   # batch stride for each tensor
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B where A:[BH,M,K], B:[BH,K,N], C:[BH,M,N]"""
    pid_m = tl.program_id(0)          # block row index within a matrix
    pid_n = tl.program_id(1)          # block column index within a matrix
    pid_b = tl.program_id(2)          # which (batch * hidden) pair

    # Compute the start offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Offsets for the K dimension (inner product)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the beginning of the batch
    a_batch_ptr = a_ptr + pid_b * stride_batch_a
    b_batch_ptr = b_ptr + pid_b * stride_batch_b
    c_batch_ptr = c_ptr + pid_b * stride_batch_c

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K

        # Compute pointers for current tile
        a_ptrs = a_batch_ptr \
                 + (offs_m[:, None] * stride_am) \
                 + ((k_offset + offs_k)[None, :] * stride_ak)
        b_ptrs = b_batch_ptr \
                 + ((k_offset + offs_k)[:, None] * stride_bk) \
                 + (offs_n[None, :] * stride_bn)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        b_mask = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Compute partial product (fp16 * fp16 -> fp32 accumulation)
        acc += tl.dot(a, b)

    # Write result back to C
    c_ptrs = c_batch_ptr \
             + (offs_m[:, None] * stride_cm) \
             + (offs_n[None, :] * stride_cn)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict with the required Linear/LayerNorm parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # float32

    # Cast to FP16 for the heavy path
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cubic contraction via Triton batched GEMM
    # --------------------------------------------------------------
    B, N_seq, _, H = left.shape  # left: [B, N, N, H]

    # Rearrange tensors to (B, H, N, N) so each (batch, hidden) pair is an N×N matrix
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_t = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)  <-- transposed for matmul

    BH = B * H
    A_flat = left_t.reshape(BH, N_seq, N_seq)
    B_flat = right_t.reshape(BH, N_seq, N_seq)

    # Output buffer
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)

    # Tile sizes (tuned for A100 / RTX‑B200 style GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N_seq, BLOCK_M),
        triton.cdiv(N_seq, BLOCK_N),
        BH,
    )

    batched_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N_seq, N_seq, N_seq,                     # M, N, K
        A_flat.stride(1), A_flat.stride(2),      # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),      # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),      # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = C_flat.reshape(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, final gate, output projection
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)   # to fp32 for LayerNorm

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by the output gate (converted to fp32)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # shape (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)           # final shape [B, N, N, dim]

    return output
