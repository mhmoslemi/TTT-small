```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm over dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections
    # input shape (B,N,N,dim)
    # weights are shape (hidden_dim, dim)
    left = torch.nn.functional.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # 3. Mask
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # 4. Gates
    left_gate = torch.nn.functional.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # 5. Contraction via Triton
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right for gemm
    right_T = right_perm.transpose(-2, -1).contiguous()
    B, D, N, _ = left_perm.shape
    # flatten batch and hidden dim
    left_flat = left_perm.view(B * D, N, N)
    right_flat = right_T.view(B * D, N, N)

    # Ensure half dtype
    left_flat = left_flat.to(torch.float16)
    right_flat = right_flat.to(torch.float16)

    out_flat = batched_matmul(left_flat, right_flat)  # returns fp16

    # reshape back
    out = out_flat.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()
    # Convert to float32 for subsequent layers
    out = out.to(torch.float32)

    # 6. Output layer_norm and gating
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # out_gate is still half; convert same dtype
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32

    # final linear back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication:
    left_ik (N×N) @ (right_jk)ᵀ (N×N) → (N×N)
for each batch and hidden dimension d. The rest of the
TriMul pipeline (LayerNorms, linear projections, gating,
masking, final projection) is performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) for half‑precision inputs
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for a batch of matrices.
    A: (batch, M, K)  half
    B: (batch, K, N)  half
    C: (batch, M, N)  half (accumulated in fp32)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute coordinates of the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the batched GEMM kernel
# ----------------------------------------------------------------------
def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B.
    A: (batch, M, K)   half
    B: (batch, K, N)   half
    Returns C: (batch, M, N) half
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Block sizes – tuned for typical N (≤1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half for the heavy linear/gate work
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half‑precision, bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4. Gates (half‑precision, sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM: left @ rightᵀ
    # --------------------------------------------------------------
    # Bring hidden dimension to the second axis for easy batching
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose the last two axes of right so that we compute A @ Bᵀ
    right_T = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N)

    B_batch, H, N, _ = left_perm.shape
    # Flatten batch and hidden dimensions
    left_flat = left_perm.view(B_batch * H, N, N)
    right_flat = right_T.view(B_batch * H, N, N)

    # Ensure half precision for the kernel
    left_flat = left_flat.to(torch.float16)
    right_flat = right_flat.to(torch.float16)

    # Batched GEMM via Triton
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B_batch, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # remaining ops use fp32

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gate, and final projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
