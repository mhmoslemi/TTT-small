@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ak,
                          stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          stride_batch,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(axis=0) # m block
    pid_n = tl.program_id(axis=1) # n block
    pid_batch = tl.program_id(axis=2) # batch*hidden

    # Compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute linear offsets for C location to store later
    c_ptrs = C_ptr + pid_batch * stride_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Load A tile
        a_ptrs = A_ptr + pid_batch * stride_batch + offs_m[:, None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        # A mask
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile
        b_ptrs = B_ptr + pid_batch * stride_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # Store
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm
    norm_w = weights['norm.weight']
    norm_b = weights['norm.bias']
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for speed
    x_fp16 = x.half()

    # Projections
    left_proj_w = weights['left_proj.weight'].half()
    right_proj_w = weights['right_proj.weight'].half()
    left = F.linear(x_fp16, left_proj_w)
    right = F.linear(x_fp16, right_proj_w)

    # Apply mask
    if mask is not None:
        mask_f = mask.unsqueeze(-1).half()
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate_w = weights['left_gate.weight'].half()
    right_gate_w = weights['right_gate.weight'].half()
    out_gate_w = weights['out_gate.weight'].half()

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Reshape for batched matmul
    B, N, _, H = left.shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Flatten batch*hidden dimension
    BH = B * H
    A = left_perm.reshape(BH, N, N)
    Bmat = right_perm.reshape(BH, N, N)

    # Allocate output buffer
    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # Determine block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Compute grid
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    # Launch kernel
    batched_matmul_kernel[grid](
        A,
        Bmat,
        C,
        N, N, N,
        A.stride(1), A.stride(2),   # stride_am, stride_ak
        Bmat.stride(1), Bmat.stride(2),  # stride_bk, stride_bn
        C.stride(1), C.stride(2),   # stride_cm, stride_cn
        A.stride(0), Bmat.stride(0), C.stride(0),   # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape output back
    out = C.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # Apply to_out_norm and to_out
    out_fp32 = out.float()

    to_out_norm_w = weights['to_out_norm.weight']
    to_out_norm_b = weights['to_out_norm.bias']
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply with out_gate (now out_gate in fp16, need cast)
    out_norm = out_norm * out_gate.float()

    to_out_w = weights['to_out.weight']
    output = F.linear(out_norm, to_out_w)

    return output

"""
TriMul (outgoing) forward pass optimized with a custom Triton kernel.

Algorithm:
1. Layer‑normalize the input (float32).
2. Cast to fp16 and apply three linear projections:
   - left  = x @ left_proj_weight
   - right = x @ right_proj_weight
   - left_gate, right_gate, out_gate = sigmoids of linear projections on x
3. Apply optional mask and gating:
   left  *= mask * left_gate
   right *= mask * right_gate
4. Compute the cubic contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix multiplication left (i,k) @ rightᵀ (k,j) for every
   (b,d) pair.  The kernel works directly on the original tensors,
   avoiding any costly permute/copy.
5. Post‑process:
   - Layer‑norm over the hidden dimension,
   - element‑wise multiply with out_gate,
   - final linear projection back to the original channel dimension.
All heavy arithmetic is performed in fp16 with fp32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,                     # left  tensor [B,N,N,H] (fp16)
    b_ptr,                     # right tensor [B,N,N,H] (fp16)
    c_ptr,                     # output tensor [B,N,N,H] (fp16)
    M, N, K,                   # all equal to seq_len
    stride_am, stride_ak,      # A row stride (i) and column stride (k)
    stride_bk, stride_bn,      # B row stride (k) and column stride (j) – transposed view
    stride_cm, stride_cn,      # C row and column strides
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch‑hidden stride
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H: tl.constexpr           # hidden dimension (used to split pid into batch & channel)
):
    pid_m = tl.program_id(0)          # row‑tile index
    pid_n = tl.program_id(1)          # col‑tile index
    pid_bn = tl.program_id(2)         # combined batch*hidden index

    # Decompose pid_bn into batch index b and hidden channel d
    b = pid_bn // H
    d = pid_bn % H

    # Base pointers for this (b,d) matrix
    a_base = a_ptr + b * stride_batch_a + d
    b_base = b_ptr + b * stride_batch_b + d
    c_base = c_ptr + b * stride_batch_c + d

    # Offsets within the matrix tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Pointers for A tile (rows i, cols k)
        a_ptrs = a_base + offs_m[:, None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Pointers for B tile (rows k, cols j) – note transposed layout
        b_ptrs = b_base + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Compute partial product
        acc += tl.dot(a, b)

    # Write back the result (fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module (outgoing).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict containing all model weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the expensive part
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left = F.linear(x_fp16, left_proj_w)          # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w)        # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(device).half()
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid on linear projections of x)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as left (i,k) @ rightᵀ (k,j) for each (b,d).
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty_like(left, dtype=torch.float16, device=device)  # [B,N,N,H]

    # Block sizes – tuned for FP16 Tensor‑Core matmul
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * hidden_dim,            # batch × hidden
    )

    batched_matmul_kernel[grid](
        left,                      # a_ptr
        right,                     # b_ptr
        out,                       # c_ptr
        N, N, N,                   # M, N, K
        left.stride(1), left.stride(2),          # stride_am, stride_ak  (i, k)
        right.stride(2), right.stride(1),        # stride_bk, stride_bn (k, j) – transposed view
        out.stride(1), out.stride(2),            # stride_cm, stride_cn
        left.stride(0), right.stride(0), out.stride(0),  # batch‑hidden strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H=hidden_dim,
    )

    # ------------------------------------------------------------------
    # 7. Post‑processing (LayerNorm on hidden dim, gating, final projection)
    # ------------------------------------------------------------------
    out_fp32 = out.float()

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate (cast to float32 for stability)
    out_norm = out_norm * out_gate.float()

    to_out_w = weights["to_out.weight"].to(device)      # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)               # [B,N,N,dim]

    return output
```
