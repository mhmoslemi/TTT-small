@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # tile offsets for M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to tile of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b) # dot: (BLOCK_M,BLOCK_K) * (BLOCK_K,BLOCK_N) -> (BLOCK_M,BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul "outgoing" forward implementation with a Triton kernel for the
core N×N×N multiplication:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All surrounding operations (LayerNorm, Linear projections, gates,
output LayerNorm and final projection) are performed in PyTorch.
The heavy triple‑product is reduced to a batched matrix multiplication
A (i,k) @ Bᵀ (k,j) for each hidden channel d and batch b and executed
by a tiled Triton kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) @ B (K×N) -> C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes
    stride_am, stride_ak,              # A strides (row, col)
    stride_bk, stride_bn,              # B strides (row, col)
    stride_cm, stride_cn,              # C strides (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D program ID: batch, block‑row, block‑col
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointer arithmetic for tiles of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul operator (outgoing variant).
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, D]  (float32)
            - mask_tensor:  [B, N, N] or None
            - weights_dict: contains all model weights as torch tensors
            - config_dict:  must contain "dim" and "hidden_dim"
    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, D = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # mean / var over the last dimension (D)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b                     # [B, N, N, D]

    # --------------------------------------------------------------
    # 2. Linear projections & gates (float16 for speed)
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # projection weights (out_features = hidden, in_features = D)
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, w_left)   # [B, N, N, hidden]
    right = F.linear(x_f16, w_right)

    # optional mask
    if mask is None:
        mask = torch.ones(B, N, N, device=device, dtype=x.dtype)
    mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
    left = left * mask_f16
    right = right * mask_f16

    # gate weights
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_f16, w_left_gate).sigmoid()
    right_gate = F.linear(x_f16, w_right_gate).sigmoid()
    out_gate = F.linear(x_f16, w_out_gate).sigmoid()   # [B, N, N, hidden]

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3. Core TriMul via Triton (batched GEMM)
    #    out[b, d] = left[b, d] @ right[b, d].T
    # --------------------------------------------------------------
    # reshape to (B * hidden, N, N) with appropriate transposition for right
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N_i, N_k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N_k, N_j)

    total_batches = B * hidden
    left_mat = left_perm.reshape(total_batches, N, N)
    right_mat = right_perm.reshape(total_batches, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # block sizes – simple heuristic
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_mat,               # a_ptr
        right_mat,              # b_ptr
        out_mat,                # c_ptr
        N, N, N,                # M, N, K
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2), # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),     # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out = out_mat.reshape(B, hidden, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden]
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 4. Output LayerNorm, final gate, final projection (float32)
    # --------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]

    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * w_out_norm + b_out_norm

    # apply output gate (convert to float32 if needed)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to original dimension D
    w_to_out = weights["to_out.weight"].to(torch.float32)   # (D, hidden)
    output = F.linear(out_norm, w_to_out)                  # [B, N, N, D]

    return output
