"""
TriMul (outgoing) forward pass with Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Apply left/right linear projections and three gating linear layers.
3. (Optional) mask the projected tensors before gating.
4. Fuse left/right into a batched GEMM:
   For every batch * hidden index we compute
       out[b,h,i,j] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix multiplication
       A @ Bᵀ  where  A = left[b,h,:,:] , B = right[b,h,:,:] .
   This is performed with a custom Triton kernel that works in float16
   and accumulates in float32.
5. Reshape the result back to [B, N, N, hidden_dim].
6. Apply a second LayerNorm, the output gate, and a final linear projection
   back to the original channel dimension.

All operations outside the heavy contraction are kept in PyTorch for
clarity; the core O(N³) part is fully fused in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM computing C = A @ Bᵀ.

    Shapes (per batch):
        A : [M, K]   (float16)
        B : [N, K]   (float16) – loaded transposed as (K, N)
        C : [M, N]   (float16)
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    batch_id = tl.program_id(2)  # batch (B * hidden_dim)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = A_ptr + batch_id * stride_a_batch
    b_base = B_ptr + batch_id * stride_b_batch
    c_base = C_ptr + batch_id * stride_c_batch

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_tiles_k):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile transposed on‑the‑fly: (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_base + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask  : torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # 1. Unpack inputs
    # ------------------------------------------------------------------ #
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 2. Input layer‑norm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 4. Optional masking (applied before gating)
    # ------------------------------------------------------------------ #
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] broadcast over hidden_dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 5. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 6. Cast to half‑precision for the heavy contraction
    # ------------------------------------------------------------------ #
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 7. Reshape to (batch_h, N, N) where batch_h = B * hidden_dim
    # ------------------------------------------------------------------ #
    #   Permute to bring hidden_dim to batch axis, then flatten batch & hidden
    left_batch = left_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_batch = right_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    batch_h = left_batch.shape[0]  # B * hidden_dim

    # ------------------------------------------------------------------ #
    # 8. Allocate output buffer (float16)
    # ------------------------------------------------------------------ #
    out_fp16 = torch.empty((batch_h, N, N), dtype=torch.float16,
                           device=inp.device)

    # ------------------------------------------------------------------ #
    # 9. Kernel launch configuration
    # ------------------------------------------------------------------ #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    grid = (num_blocks_m, num_blocks_n, batch_h)

    # Strides in elements
    stride_a_batch = left_batch.stride(0)
    stride_a_m = left_batch.stride(1)
    stride_a_k = left_batch.stride(2)

    stride_b_batch = right_batch.stride(0)
    stride_b_n = right_batch.stride(1)   # row dimension
    stride_b_k = right_batch.stride(2)   # column dimension (K)

    stride_c_batch = out_fp16.stride(0)
    stride_c_m = out_fp16.stride(1)
    stride_c_n = out_fp16.stride(2)

    # ------------------------------------------------------------------ #
    # 10. Run the batched GEMM kernel
    # ------------------------------------------------------------------ #
    tri_mul_gemm_kernel[grid](
        left_batch,
        right_batch,
        out_fp16,
        N, N, N,                         # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 11. Reshape back to [B, N, N, hidden_dim] and convert to float32
    # ------------------------------------------------------------------ #
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)          # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------ #
    # 12. Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B, N, N, dim]

    return out
