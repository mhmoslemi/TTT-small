"""
TriMul outgoing kernel: layernorm, linear proj, gating, pairwise
contraction via block batched matrix multiplication, hidden‑dim LayerNorm done
in Triton, final gating and output projection in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base offsets
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # load
    x = tl.load(inp_ptr + inp_base + offs_h * stride_in_h,
                mask=mask_h, other=0.0)
    # mean
    sum_x = tl.sum(x)
    mean = sum_x / H

    # variance
    diff = x - mean
    sq = diff * diff
    sum_sq = tl.sum(sq)
    var = sum_sq / H

    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    tl.store(out_ptr + out_base + offs_h * stride_out_h,
             out, mask=mask_h)

    hidden_dim = config["hidden_dim"]

        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],

    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # 3. Apply mask (if any) and gates
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    right = right * right_gate

    # 4. Pairwise contraction via batched matrix multiplication (bfloat16)
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # reshape to (B*H, N, N)
    left_b = left_bf.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_b = right_bf.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # matmul: left @ right^T
    out_b = torch.bmm(left_b, right_b.transpose(-2, -1))

    # reshape back to (B, N, N, H)
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N

        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    out_scaled = out_norm * out_gate

    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

"""
Outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the last (dim) axis.
2. Linear projections → left/right and three gating vectors (left_gate,
   right_gate, out_gate) using the supplied weight matrices.
3. Apply optional pairwise mask and element‑wise gates.
4. Compute the pairwise contraction over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is performed as a batched matrix multiplication per hidden
   dimension (cast to bfloat16 for higher bandwidth).
5. Apply a LayerNorm over the hidden dimension (hidden_dim) with a custom
   Triton kernel that computes per‑(b,i,j) mean/variance.
6. Multiply the normalized tensor by out_gate.
7. Final linear projection back to the original dimensionality.

Only the hidden‑dim LayerNorm is written in Triton; all other heavy
operations leverage highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,           # *float32  [B,N,N,H]
    out_ptr,           # *float32  [B,N,N,H]
    weight_ptr,        # *float32  [H]
    bias_ptr,          # *float32  [H]
    B,                 # tl.int32
    N,                 # tl.int32
    H,                 # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # tl.int32
    eps: tl.constexpr,                                   # const float
    BLOCK_H: tl.constexpr                               # compile‑time block size (≥ max hidden_dim)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (b, i, j) from a linear program id
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets along the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Base address for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # Load the hidden‑dim vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0,
    )

    # ---- mean -------------------------------------------------------
    sum_x = tl.sum(x)
    mean = sum_x / H

    # ---- variance ---------------------------------------------------
    diff = x - mean
    sq = diff * diff
    sum_sq = tl.sum(sq)
    var = sum_sq / H

    # ---- normalize --------------------------------------------------
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ---- apply learned weight & bias --------------------------------
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # ---- store -------------------------------------------------------
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h,
    )
