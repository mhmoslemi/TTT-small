Define kernel:

@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # compute b,i,j coordinates
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offset for each (b,i,j)
    inp_off = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_off = b * stride_out_b + i * stride_out_i + j * stride_out_j

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    x = tl.load(inp_ptr + inp_off + offs_h * stride_in_h, mask=mask_h, other=0.0)

    sum_x = tl.sum(x)
    mean = sum_x / H

    diff = x - mean
    var = tl.sum(diff * diff) / H

    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)

    out = norm * w + b_

    tl.store(out_ptr + out_off + offs_h * stride_out_h, out, mask=mask_h)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    # Norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    B,N,N1,_ = x.shape ; assert N==N1

    # left_proj
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B,N,N,hidden_dim)

    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B,N,N,hidden_dim)

    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm over the channel dimension `dim`.
2. Linear projections to a hidden dimension `hidden_dim` and three gates
   (left, right, out).  Gates are passed through a sigmoid.
3. Apply the pairwise mask (if provided) and the left/right gates.
4. Compute the pairwise contraction over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batch of matrix‑multiplications:
   for each hidden channel h we multiply
       left_h (N×N) with right_hᵀ (N×N) .
   The matrices are cast to bfloat16 so that Tensor‑cores are used.
5. Perform a Layer‑norm over the hidden dimension using a small Triton kernel.
6. Multiply by the out‑gate.
7. Final linear projection back to the original dimension `dim`.

Only the hidden‑dim LayerNorm is written in Triton; the rest uses
high‑performance PyTorch kernels (batched GEMM in bfloat16).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,                # *float32  [B,N,N,H]
    out_ptr,                # *float32  [B,N,N,H]
    weight_ptr,             # *float32  [H]
    bias_ptr,               # *float32  [H]
    B,                      # tl.int32
    N,                      # tl.int32
    H,                      # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # tl.int32
    eps: tl.constexpr,                                   # const float
    BLOCK_H: tl.constexpr                               # compile‑time block size (≥ hidden_dim)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # 3‑D coordinates for this program instance
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Base offset for the (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # Offsets along the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Load the hidden‑dim vector
    x = tl.load(inp_ptr + inp_base + offs_h * stride_in_h,
                mask=mask_h, other=0.0)

    # ----- mean -----
    sum_x = tl.sum(x)
    mean = sum_x / H

    # ----- variance -----
    diff = x - mean
    var = tl.sum(diff * diff) / H

    # ----- normalize -----
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ----- apply learned weight & bias -----
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h,   mask=mask_h, other=0.0)
    out = norm * w + b_

    # Store the result
    tl.store(out_ptr + out_base + offs_h * stride_out_h,
             out, mask=mask_h)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension `dim`
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (float32)
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) and gating
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction over k (batched GEMM in bfloat16)
    left_bf  = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # reshape to (B*hidden_dim, N, N) where the last axis is the k‑dimension
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # batched matrix multiplication: (i,k) @ (j,k)^T  -> (i,j)
    out_bf = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H] and cast to float32
    out = out_bf.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)

    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (total_programs,)
    BLOCK_H = 256  # larger than any hidden_dim used in the tests (max 128)

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Apply the outgoing gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    final = F.linear(out_scaled.view(-1, hidden_dim), weights["to_out.weight"])
    final = final.view(B, N, N, dim)

    return final
