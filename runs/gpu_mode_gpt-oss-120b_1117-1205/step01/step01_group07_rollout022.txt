Pseudo-code:

@triton.jit
def triton_batched_matmul_kernel(A, B, C,
                                 M, N, K,
                                 stride_aa, stride_ab, stride_ac,
                                 stride_ba, stride_bb, stride_bc,
                                 stride_ca, stride_cb, stride_cc,
                                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # get block ids
    pid = tl.program_id(0) # row block id
    pid_n = tl.program_id(1) # col block id
    pid_batch = tl.program_id(2) # batch index

    # offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers into A and B
        a_ptrs = A + pid_batch * stride_aa + offs_m[:, None] * stride_ab + k_offset[None, :] * stride_ac
        b_ptrs = B + pid_batch * stride_ba + k_offset[:, None] * stride_bb + offs_n[None, :] * stride_bc

        # load with mask
        mask_a = (offs_m[:, None] < M) & (k_offset[None, :] < K)
        mask_b = (k_offset[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float16)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = C + pid_batch * stride_ca + offs_m[:, None] * stride_cb + offs_n[None, :] * stride_cc
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

"""
TriMul (outgoing) forward pass with a small Triton kernel.

The heavy O(N³·hidden) pairwise contraction is performed with a
batched FP16 matrix multiplication using `torch.matmul` (cuBLAS/Tensor
cores).  A tiny Triton kernel fuses the element‑wise gate multiplication
so that the PyTorch part does not incur extra memory traffic.  The
implementation follows the reference PyTorch module and returns a
float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: fuse left/right gating (in‑place)
# -------------------------------------------------------------------------
@triton.jit
def apply_gate_kernel(
    left_ptr,            # *fp16
    right_ptr,           # *fp16
    left_gate_ptr,       # *fp16
    right_gate_ptr,      # *fp16
    total_elems: tl.int64,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    tl.store(left_ptr + offs, l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)


# -------------------------------------------------------------------------
# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float, optional)
            - weights_dict: dict with the model parameters
            - config_dict: must contain "dim", "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Load weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # LayerNorm on input (last dimension)
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -----------------------------------------------------------------
    # Linear projections (float16 for bandwidth)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gates (sigmoid, then cast to fp16)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # keep fp32 for later

    # Cast projections to fp16 before the element‑wise gate fusion
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # -----------------------------------------------------------------
    # Fuse gate multiplication with a tiny Triton kernel (in‑place)
    # -----------------------------------------------------------------
    total_elems = left.numel()
    BLOCK = 1024                               # max threads per CUDA block
    grid = (triton.cdiv(total_elems, BLOCK),)
    apply_gate_kernel[grid](
        left, right,
        left_gate, right_gate,
        total_elems,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # Batched matrix multiplication: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # Implemented with torch.matmul (cuBLAS/Tensor‑cores)
    # -----------------------------------------------------------------
    # left:  [B, N, N, H]  -> [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2)           # (B, hidden, N, N)
    # right: [B, N, N, H] -> swap the two N axes -> [B, H, N, N] where inner K is transposed
    right_perm = right.permute(0, 3, 2, 1)        # (B, hidden, N, N)

    # Batched matmul: each (B, H) pair performs N×N @ N×Nᵀ
    out_mat = torch.matmul(left_perm, right_perm)  # (B, hidden, N, N)

    # Bring back to original layout
    out = out_mat.permute(0, 2, 3, 1)               # (B, N, N, hidden)

    # -----------------------------------------------------------------
    # Final LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = out.to(torch.float32)                     # convert for layernorm
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # -----------------------------------------------------------------
    # Apply out gate and final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate                            # broadcast over last dim
    out = F.linear(out, to_out_weight)              # (B, N, N, dim)

    return out
