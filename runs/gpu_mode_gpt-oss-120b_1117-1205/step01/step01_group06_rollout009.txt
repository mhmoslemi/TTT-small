"""
TriMul outgoing forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the input tensor over its last dimension.
2. Compute three linear projections (left/right) and three gates (left, right, out)
   using the provided weight matrices (no bias).
3. Apply optional pairwise mask and multiply by the gates.
4. Pairwise contraction over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is realised as a batched matrix‑multiply:
       left_reshape = (B·H, N, N)
       right_reshape = (B·H, N, N)
       out_reshape = left_reshape @ right_reshapeᵀ
   The mats are performed in bfloat16 for higher throughput.
5. Layer‑norm over the hidden dimension (H) using a small Triton kernel.
6. Multiply by the out‑gate.
7. Final linear projection back to the original channel dimension.

Only step 5 is implemented in Triton; all other heavy lifting uses
high‑performance PyTorch kernels (cublas / cuDNN).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,           # *float32  [B,N,N,H]
    out_ptr,           # *float32  [B,N,N,H]
    weight_ptr,        # *float32  [H]
    bias_ptr,          # *float32  [H]
    B, N, H,           # int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,  # int32
    eps: tl.constexpr,                               # const float
    BLOCK_H: tl.constexpr                           # compile‑time block size (≥ hidden_dim)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (b,i,j) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Base pointers for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # Load the hidden‑dim vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    )

    # ---- mean ----
    sum_x = tl.sum(x)               # sum over H
    mean = sum_x / H

    # ---- variance ----
    diff = x - mean
    sq = diff * diff
    var = tl.sum(sq) / H

    # ---- normalize ----
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ---- apply scale & bias ----
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # Store
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None if nomask)
            - weights: dict of weight tensors (all on the same device as input)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input layer‑norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # --------------------------------------------------------------
    # 2. Linear projections & gates (float32)
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                         # (B·N·N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # --------------------------------------------------------------
    # 3. Mask & gating
    if not nomask:
        # mask may be bool/int – cast to float for multiplication
        mask_exp = mask.to(dtype).unsqueeze(-1)      # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Pairwise contraction via batched GEMM (bfloat16)
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # (B, N, N, H) -> (B, H, N, N) -> (B·H, N, N)
    left_bf = left_bf.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_bf = right_bf.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiply: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_bf = torch.bmm(left_bf, right_bf.transpose(1, 2))

    # Restore shape (B, N, N, H) and convert back to float32
    out = out_bf.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # --------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm (implemented in Triton)
    out_norm = torch.empty_like(out)
    s_b, s_i, s_j, s_h = out.stride()

    total_programs = B * N * N
    grid = (total_programs,)
    BLOCK_H = 128  # hidden_dim ≤ 128 in all test configs

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B, N, hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # F.linear works on the last dimension directly
    final = F.linear(out_scaled, weights["to_out.weight"])

    return final
