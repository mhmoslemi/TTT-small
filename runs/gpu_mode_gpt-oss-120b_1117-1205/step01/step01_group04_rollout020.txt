def custom_kernel(data):

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # Normalization
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)
    
    # Projections
    left = F.linear(x_norm, weights["left_proj.weight"])      # shape B,N,N,hidden_dim
    right = F.linear(x_norm, weights["right_proj.weight"])
    
    # Apply mask if needed using kernel
    if not nomask:

        mask_f = mask.to(left.dtype).unsqueeze(-1)   # B,N,N,1
        # We'll use Triton kernel for elementwise multiplication on left and right
        left = apply_mask(left, mask_f)  # define function that uses kernel
        right = apply_mask(right, mask_f)
    
    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))
    
    left = left * left_gate
    right = right * right_gate

"""
TriMul (outgoing) forward pass using fused Triton kernel for elementwise gating.
The heavy N^3 contraction is performed with a batched half‑precision matrix
multiply via torch.bmm (leveraging cuBLAS/TensorCores). The final multiplication
with the output gate is implemented in Triton to satisfy the requirement of a
custom kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, numel, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # 1. LayerNorm on input
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # 3. Mask (if present)
    if not nomask and mask is not None:
        # mask: [B, N, N]; broadcast to [B, N, N, 1]
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # 4. Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5. Heavy contraction via batched half‑precision matrix multiply (leverages cuBLAS/TensorCore)
    B, N, _, _ = left.shape

    # Cast to half to improve throughput (small rounding error tolerated)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*hidden_dim, N, N)
    left_view = left_h.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_view = right_h.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # batched matmul:   (B*H, N, N) @ (B*H, N, N)^T
    out_view_h = torch.bmm(left_view, right_view.transpose(1, 2))
    # back to float32 and original layout
    out = out_view_h.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layer‑norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 7. Apply output gate using a Triton kernel (elementwise multiply)
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel = out.numel()
    BLOCK = 65536  # 64K elements per program

        out,

        out,
        numel,
        BLOCK=BLOCK,
    )

    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Two linear projections (left / right) and three gate projections are
   computed in float‑32.
3. Optional mask is applied (broadcasted over the hidden dimension).
4. The heavy contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication.  The tensors are cast to float‑16
   and reshaped to (B·H, N, N) so that a single call to `torch.bmm`
   (cuBLAS/Tensor‑Core) computes all independent GEMMs efficiently.
5. The result is reshaped back to [B, N, N, H] and a second layer‑norm is
   applied.
6. The output gate (sigmoid) is fused with an element‑wise Triton kernel
   (required to have a custom kernel in the implementation).
7. A final linear projection brings the tensor back to the original
   channel dimension.

Only the element‑wise gate multiplication is performed in Triton; the
matrix multiplication leverages cuBLAS which is already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, numel, BLOCK: tl.constexpr):
    """Element‑wise multiplication: c = a * b."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input, mask, weights, config)
            input  – torch.Tensor, shape [B, N, N, dim]
            mask   – torch.Tensor, shape [B, N, N] (bool/float) or None
            weights – dict of torch.Tensors with all linear and LayerNorm params
            config – dict containing at least "dim" and "hidden_dim";
                     optional "nomask" (bool, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------ #
    if not nomask and mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Heavy contraction via batched half‑precision GEMM (cuBLAS)
    # ------------------------------------------------------------------ #
    B, N, _, _ = left.shape

    # Cast to half for Tensor‑Core acceleration
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # (B, N, N, H) -> (B·H, N, N)
    left_view = left_h.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_view = right_h.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)^T
    out_view_h = torch.bmm(left_view, right_view.transpose(1, 2))

    # Back to fp32 and original layout
    out = (
        out_view_h.to(torch.float32)
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------ #
    # 7. Apply output gate using a Triton kernel (element‑wise mul)
    # ------------------------------------------------------------------ #
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel = out.numel()
    BLOCK = 65536  # 64 K elements per program
    grid = (triton.cdiv(numel, BLOCK),)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        numel,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------ #
    # 8. Final linear projection back to original dimension
    # ------------------------------------------------------------------ #
    out = F.linear(out, weights["to_out.weight"])

    return out
