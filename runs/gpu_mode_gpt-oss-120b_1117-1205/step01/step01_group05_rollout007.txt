@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        stride_az, stride_am, stride_ak,
                        stride_bz, stride_bk, stride_bn,
                        stride_cz, stride_cm, stride_cn,
                        M, N, K,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
                        num_stages: tl.constexpr, num_warps: tl.constexpr):
    # bats
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to C tile
    c_ptrs = C_ptr + pid_batch * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_batch * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + pid_batch * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

"""
TriMul (outgoing) forward kernel.

The main computational bottleneck is the 4‑D contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix multiplication
    out = (left reshaped) @ (rightᵀ reshaped)
where each batch*hidden slice is an N×N GEMM.

We perform LayerNorms, linear projections and gating in PyTorch (mixed
precision) and offload the massive GEMM to a custom Triton kernel that
computes many independent N×N products in a single launch.
All tensors are kept on the GPU and the final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, num_warps: tl.constexpr,
):
    """
    Batched GEMM C = A @ B  (A: [batch, M, K], B: [batch, K, N]).
    All tensors are row‑major (contiguous), compute in FP16 → FP32 accumulation.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)    # block row
    pid_n = tl.program_id(2)    # block col

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current C tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (K×N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back in FP16
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

    Forward pass of the outgoing TriMul operator.

    # ------------------------------------------------------------------

    eps = 1e-5

    dim = config["dim"]

    # ------------------------------------------------------------------

    x = input_tensor.to(torch.float32)                     # [B, N, N, dim]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)

    x_norm = x_norm * norm_weight + norm_bias               # [B, N, N, dim]

    # Cast to fp16 for the heavy linear & gating stages
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Projections
    left = F.linear(x_h, w_left)          # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

        left = left * mask_exp
        right = right * mask_exp

    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()     # will be used after the second LayerNorm

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim

    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)      # [batch, M, K]
    right_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)     # [batch, K, N]

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # Strides for the contiguous views
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_T.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # Tiling parameters (tuned for square N up to 1024 on B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                              # batch dimension (program_id 0)
        triton.cdiv(N, BLOCK_M),           # row tiles   (program_id 1)
        triton.cdiv(N, BLOCK_N),           # col tiles   (program_id 2)
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_T,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, hidden_dim] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden‑dim wise)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Apply the output gate (broadcasted)
    out_norm = out_norm * out_gate.to(torch.float32)

    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)                  # [B, N, N, dim]
