def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm input
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # Cast to half for speed
    x = x.to(torch.float16)

    # Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # Apply mask if provided
    if mask_tensor is not None:
        # ensure mask is float16
        mask_fp = mask_tensor.unsqueeze(-1).to(torch.float16).to(device)
        left = left * mask_fp
        right = right * mask_fp

    # Gating
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Core contraction using Triton
    B, N, _, H = left.shape  # B,N,N,H
    # Permute to (B,H,N,N) for gemm
    A = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N) rows i, cols k
    Bmat = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) rows k, cols j

    BH = B * H
    A_flat = A.reshape(BH, N, N)
    B_flat = Bmat.reshape(BH, N, N)
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)

    # Triton block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,
        A_flat.stride(1), A_flat.stride(2),
        B_flat.stride(1), B_flat.stride(2),
        C_flat.stride(1), C_flat.stride(2),
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,H)
    out = C_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # Post-processing
    out_fp32 = out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

The heavy operation is the cubic contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is equivalent to a per‑channel matrix multiply
    left_d @ right_d.T  for each hidden channel d.
We fuse this N×N × N×N multiplication across the (batch × hidden_dim) axes
using a custom Triton kernel.  All surrounding LayerNorms, linear projections,
and gating are performed in PyTorch (FP16 where possible).  The kernel
accumulates in FP32 and writes FP16 results.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched half‑precision matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers to A, B, C
    M, N, K,                        # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak,           # A strides: (row, col)
    stride_bk, stride_bn,           # B strides: (row, col)
    stride_cm, stride_cn,           # C strides: (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)       # block row index
    pid_n = tl.program_id(1)       # block column index
    pid_b = tl.program_id(2)       # which (batch × hidden) matrix

    # --------------------------------------------------------------
    # Compute offsets for the current block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for A and B
        a_ptrs = a_ptr + pid_b * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_batch_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # masked loads
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product for the tile
        acc += tl.dot(a, b)   # (BLOCK_M,BLOCK_K)·(BLOCK_K,BLOCK_N) → (BLOCK_M,BLOCK_N)

    # write back
    c_ptrs = c_ptr + pid_b * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N]  (float32 or bool)
            - weights_dict: mapping str → torch.Tensor containing all model parameters
            - config_dict:  contains at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # Work in FP16 for the heavy steps
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        # mask shape: [B,N,N] -> [B,N,N,1]
        mask_fp16 = mask_tensor.unsqueeze(-1).to(torch.float16).to(device)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (Triton)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange tensors to (B, H, N, N) where the last two axes are the matrix
    A = left.permute(0, 3, 1, 2).contiguous()          # rows = i, cols = k
    Bmat = right.permute(0, 3, 2, 1).contiguous()      # rows = k, cols = j (transposed)

    # Collapse (B, H) into a single batch dimension for the kernel
    BH = B * H
    A_flat = A.reshape(BH, N, N)
    B_flat = Bmat.reshape(BH, N, N)

    # Output buffer
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)

    # Block sizes (tuned for RTX/A100‑class GPUs; work well up to N=1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )

    batched_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,                                   # M, N, K
        A_flat.stride(1), A_flat.stride(2),        # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),        # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),        # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout: [B,N,N,hidden_dim]
    out = C_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm on hidden dim, final gate, output proj)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    # LayerNorm over hidden dimension
    out_norm = F.layer_norm(out_fp32, (hidden_dim,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b)

    # Apply the output gate (cast to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(device)   # shape (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)            # [B, N, N, dim] float32

    return output
