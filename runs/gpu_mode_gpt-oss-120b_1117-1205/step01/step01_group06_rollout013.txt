@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    mask_ptr,  # optional maybe null if nomask
    has_mask: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    eps: tl.constexpr    # maybe not needed
):
    pid = tl.program_id(0)
    total = B * N * N  # (b,i,j)
    if pid >= total:
        return
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # allocate accumulator
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # Loop over k in tiles
    num_tiles = tl.cdiv(N, BLOCK_K)
    for tile in range(num_tiles):
        # compute k indices
        k_offset = tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # compute pointers
        # left: left_ptr + b*sb + i*si + k_offset*sk + offs_h*sh
        left_ptrs = left_ptr + b*stride_left_b + i*stride_left_i + k_offset[:, None]*stride_left_k + offs_h[None, :]*stride_left_h
        # load left tile
        left_tile = tl.load(left_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # right: right_ptr + b*sb + j*sj + k_offset*sk + offs_h*sh
        right_ptrs = right_ptr + b*stride_right_b + j*stride_right_j + k_offset[:, None]*stride_right_k + offs_h[None, :]*stride_right_h
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # elementwise multiply and sum across k dimension:
        # For each h, sum_{k} left_tile[k, h]*right_tile[k, h]
        # We can multiply and then reduce sum across k axis.
        prod = left_tile * right_tile  # shape (BLOCK_K, BLOCK_H)
        # sum across k axis
        # Use tl.sum across k dimension
        acc += tl.sum(prod, axis=0)

    # after loop, save to out
    # out_ptr offset: b*sb_out + i*si_out + j*sj_out + offs_h*sh_out
    out_ptrs = out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j + offs_h*stride_out_h
    tl.store(out_ptrs, acc, mask=mask_h)

"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor over its last dimension (dim).
2. Linear projections → left/right and three gates (left_gate, right_gate, out_gate)
   using the supplied weight matrices (no bias).
3. Apply optional mask and per‑gate multiplicative gating.
4. Pairwise contraction over the sequence dimension k:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   Implemented as a batched matrix multiplication on BF16 tensors.
5. Hidden‑dim LayerNorm (over H) – performed by a small Triton kernel.
6. Multiply by out_gate.
7. Final linear projection back to the original dimension (dim).

Only the hidden‑dim LayerNorm is executed in Triton; the remaining
operations use highly‑optimized PyTorch kernels (including a batched
matrix‑multiply for the cubic contraction).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,               # *float32  [B,N,N,H]
    out_ptr,               # *float32  [B,N,N,H]
    weight_ptr,            # *float32  [H]
    bias_ptr,              # *float32  [H]
    B,                     # int32
    N,                     # int32
    H,                     # int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # int32
    eps: tl.constexpr,                                   # float32
    BLOCK_H: tl.constexpr                                 # compile‑time block size >= max hidden dim
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets along H dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base pointers for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load hidden‑dim vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0,
    )
    # ------------------------------------------------------------------
    # Compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / tl.float32(H)

    # ------------------------------------------------------------------
    # Compute variance
    diff = x - mean
    sq = diff * diff
    var = tl.sum(sq, axis=0) / tl.float32(H)

    # ------------------------------------------------------------------
    # Normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ------------------------------------------------------------------
    # Apply learned scale (weight) and bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # ------------------------------------------------------------------
    # Store result
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be unused)
            - weights: dict of model weight tensors
            - config: dict containing "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x = x.contiguous()

    B, N, _, _ = x.shape  # _ is dim

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (all in float32)
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) and gates
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction over k (batched GEMM on BF16)
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # bring hidden dim to the batch dimension: [B, H, N, N]
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous()

    # reshape to batch = B*H
    left_mat = left_perm.view(-1, N, N)                     # [B*H, N, N]
    right_mat = right_perm.view(-1, N, N)                   # [B*H, N, N]

    # batched matrix multiplication: (i,k) * (j,k)ᵀ  -> (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # reshape back to [B, N, N, H] and cast to float32
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)

    # strides in element units (not bytes)
    s_b, s_i, s_j, s_h = out.stride()
    s_b_o, s_i_o, s_j_o, s_h_o = out_norm.stride()

    total_programs = B * N * N
    grid = (total_programs,)

    # BLOCK_H must be >= hidden_dim; we pick a safe compile‑time constant
    BLOCK_H = 384  # covers max hidden_dim in the provided configs

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b_o, s_i_o, s_j_o, s_h_o,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)        # [B*N*N, H]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)
