def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm over dim
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    B, N, _, _ = x.shape

    # Flatten for linear projections
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    # Compute left, right
    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # Compute gates
    left_gate = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim))
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim))
    out_gate = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim))

    # Apply mask if exists
    if not nomask:
        mask_exp = mask[:, :, :, None].to(x.dtype)  # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Reshape for batched GEMM
    # (B,N,N,H) -> (B*H,N,N)
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # Transpose right for B^T
    right_mat_T = right_mat.permute(0, 2, 1).contiguous()  # shape (B*H, N, N)

    # Allocate output tensor in float16
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Launch batched gemm kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    batched_gemm_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        B * hidden_dim,
        N,
        N,

        right_mat_T.stride(0), right_mat_T.stride(1), right_mat_T.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape out_mat back to (B,N,N,H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)
    out = out.to(torch.float32)  # convert to float32

    # Hidden dim layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # Apply out_gate
    out = out * out_gate

    out_flat = out.view(-1, hidden_dim)
    final = F.linear(out_flat, weights["to_out.weight"])
    final = final.view(B, N, N, dim)

    return final

"""
TriMul (outgoing) forward pass with a custom Triton batched GEMM kernel.
The kernel computes the contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
as a batched matrix multiplication (B*H independent GEMMs of size N×N).
All other operations (LayerNorm, linear projections, gating) are performed
with PyTorch functions. This satisfies the requirement of using a Triton
kernel while keeping the overall runtime low.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr,                      # *fp16   [batch, M, K]
    b_ptr,                      # *fp16   [batch, K, N]
    c_ptr,                      # *fp16   [batch, M, N]
    M, N, K,                    # matrix dimensions
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bk, stride_bn,   # strides for B
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: M×K, B: K×N) for each batch."""
    pid_z = tl.program_id(2)            # batch index
    pid_m = tl.program_id(0)            # block row
    pid_n = tl.program_id(1)            # block col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # range checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch offsets
    a_batch_offset = pid_z * stride_az
    b_batch_offset = pid_z * stride_bz
    c_batch_offset = pid_z * stride_cz

    # accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # pointers to current tiles
        a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        b_ptrs = b_ptr + b_batch_offset + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load with masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product of tiles
        acc += tl.dot(a, b)   # a,b are fp16, acc is fp32

    # store result (cast back to fp16)
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored if config["nomask"] is True)
            - weights: dict containing all model weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) and gating vectors
    x_flat = x.view(-1, dim)                      # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # --------------------------------------------------------------
    # 3. Apply mask (if present) and gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Batched GEMM: compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # Cast to fp16 for bandwidth
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Merge batch and hidden dimensions: (B*H, N, N)
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Transpose right for A @ B^T pattern
    right_mat_T = right_mat.permute(0, 2, 1).contiguous()

    # Allocate output tensor (fp16)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),          # blocks along M (i)
        triton.cdiv(N, BLOCK_N),          # blocks along N (j)
        B * hidden_dim,                   # batch dimension (b * h)
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        N,               # M
        N,               # N
        N,               # K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat_T.stride(0), right_mat_T.stride(1), right_mat_T.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 5. Reshape back to [B, N, N, hidden_dim] and promote to fp32
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # --------------------------------------------------------------
    # 6. LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 7. Apply outgoing gate
    out = out * out_gate

    # --------------------------------------------------------------
    # 8. Final linear projection back to dim
    out_flat = out.view(-1, hidden_dim)               # [B*N*N, hidden_dim]
    final = F.linear(out_flat, weights["to_out.weight"])
    final = final.view(B, N, N, dim)

    return final
