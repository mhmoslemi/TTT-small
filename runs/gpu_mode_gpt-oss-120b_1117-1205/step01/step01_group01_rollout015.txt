    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK: tl.constexpr
):
   pid = tl.program_id(0)
   offs = pid * BLOCK + tl.arange(0, BLOCK)
   total = B * N * N * H
   mask = offs < total
   # Compute indices
   # using integer division
   pos = offs
   b = pos // (N * N * H)
   pos1 = pos % (N * N * H)
   i = pos1 // (N * H)
   pos2 = pos1 % (N * H)
   j = pos2 // H
   h = pos2 % H
   # compute offsets
   left_off = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
   right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
   left_gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
   right_gate_off = left_gate_off  # same shape
   mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

   # load
   left_val = tl.load(left_ptr + left_off, mask=mask, other=tl.float16(0.0))
   right_val = tl.load(right_ptr + right_off, mask=mask, other=tl.float16(0.0))
   left_gate_val = tl.load(left_gate_ptr + left_gate_off, mask=mask, other=tl.float16(0.0))
   right_gate_val = tl.load(right_gate_ptr + left_gate_off, mask=mask, other=tl.float16(0.0))
   mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))

   left_out = left_val * left_gate_val * mask_val
   right_out = right_val * right_gate_val * mask_val

   tl.store(left_ptr + left_off, left_out, mask=mask)

"""
TriMul (outgoing) forward pass optimized with cuBLAS batched GEMM for the cubic
contraction and a tiny Triton kernel for the element‑wise mask & gate fusion.
All heavy arithmetic is performed in float16; final output is cast to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_gate_mask_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] tensors (in‑place output)
    left_gate_ptr, right_gate_ptr,      # [B,N,N,H] gating tensors
    mask_ptr,                           # [B,N,N] mask tensor (broadcasted over H)

    B: tl.int32, N: tl.int32, H: tl.int32,

    # strides for left/right (B, i, j, h)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    # strides for gates (same layout as left/right)
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    # strides for mask (B, i, j)
    stride_mask_b, stride_mask_i, stride_mask_j,

    BLOCK: tl.constexpr,
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ----- decode linear offset into (b,i,j,h) ---------------------------------
    pos = offs
    b = pos // (N * N * H)
    pos1 = pos % (N * N * H)
    i = pos1 // (N * H)
    pos2 = pos1 % (N * H)
    j = pos2 // H
    h = pos2 % H

    # ----- compute flat pointers -------------------------------------------------
    left_off  = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h

    gate_off  = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    mask_off  = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # ----- load, compute, store -------------------------------------------------
    left_val  = tl.load(left_ptr + left_off, mask=mask, other=tl.float16(0.0))
    right_val = tl.load(right_ptr + right_off, mask=mask, other=tl.float16(0.0))

    left_g    = tl.load(left_gate_ptr + gate_off, mask=mask, other=tl.float16(0.0))
    right_g   = tl.load(right_gate_ptr + gate_off, mask=mask, other=tl.float16(0.0))

    mask_val  = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))

    left_out  = left_val * left_g * mask_val
    right_out = right_val * right_g * mask_val

    tl.store(left_ptr + left_off, left_out, mask=mask)
    tl.store(right_ptr + right_off, right_out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, C] (float32)
            - mask: torch.Tensor, shape [B, N, N] (float32) or None
            - weights: dict of model parameters (see module definition)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- 1. LayerNorm on the input ------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # [B,N,N,dim]

    # ---------- 2. Cast to fp16 & linear projections -------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)   # (H, C)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)  # (H, C)

    left  = F.linear(x_fp16, left_proj_w)   # [B,N,N,H]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,H]

    # ---------- 3. Gating linear layers ------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,H]

    # ---------- 4. Optional mask (broadcasted over H) -----------------------
    if mask is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # ---------- 5. Fuse mask & gates via a tiny Triton kernel ---------------
    total_elements = B * N * N * hidden_dim
    BLOCK = 1024  # good balance between occupancy & integer‑division cost
    grid = (triton.cdiv(total_elements, BLOCK),)

    _apply_gate_mask_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_fp16,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        BLOCK=BLOCK,
    )

    # ---------- 6. Cubic contraction via cuBLAS batched GEMM (FP16) -------
    # left: [B,N,N,H] -> (B,H,N,N) where H is the hidden dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = (B,H,k,j)

    BH = B * hidden_dim
    left_flat  = left_perm.reshape(BH, N, N)   # (BH, N, N)
    right_flat = right_perm.reshape(BH, N, N)  # (BH, N, N)

    # Batched matrix multiplication: (i,k) @ (k,j) -> (i,j)
    out_flat = torch.bmm(left_flat, right_flat)   # (BH, N, N)

    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ---------- 7. Post‑processing (norm, final gate, projection) ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)   # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)     # (H,)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply final gate (all tensors are fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (C, H)
    output = F.linear(out, to_out_w)   # [B,N,N,C] (fp16)

    # Cast back to float32 as required by the API
    return output.to(torch.float32)
