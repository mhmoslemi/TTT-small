@triton.jit
def batched_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_left_batch, stride_left_hidden, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_hidden, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_hidden, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(2)  # batch*hidden index
    pid_i = tl.program_id(0)  # tile row (i)
    pid_j = tl.program_id(1)  # tile col (j)

    # Compute b and d (hidden index)
    d = pid % hidden_dim
    b = pid // hidden_dim

    # Compute base offsets for the current batch/hidden slice
    offset_left_base = b * stride_left_batch + d * stride_left_hidden
    offset_right_base = b * stride_right_batch + d * stride_right_hidden
    offset_out_base = b * stride_out_batch + d * stride_out_hidden

    # Compute tile offsets in i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bound indices
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute addresses
        a_ptrs = left_ptr + offset_left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        b_ptrs = right_ptr + offset_right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

        # Load with masks
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write to output
    out_ptrs = out_ptr + offset_out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

"""
Optimized outgoing TriMul forward pass using Triton.

The heavy O(N³·hidden) reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is implemented as a batched matrix multiplication:
    left (B, D, i, k)  @  right_T (B, D, k, j) → out (B, D, i, j)
All other operations (LayerNorms, Linear projections, gates) are performed
with PyTorch.  The matrix multiplication kernel works on half‑precision inputs
and accumulates in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_left_batch, stride_left_hidden, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_hidden, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_hidden, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul for TriMul: out = left @ right_T.

    left : (B, D, i, k)  [fp16]
    right: (B, D, k, j)  [fp16]  (already transposed)
    out  : (B, D, i, j)  [fp16]

    Arguments:
        N            – sequence length (both M and K dimensions)
        hidden_dim   – number of hidden channels (D)
        stride_*     – element strides for the respective tensors
        BLOCK_*      – tile sizes
    """
    pid = tl.program_id(2)          # flattened batch*hidden index
    pid_i = tl.program_id(0)        # tile row (i)
    pid_j = tl.program_id(1)        # tile col (j)

    # recover batch and hidden indices
    d = pid % hidden_dim
    b = pid // hidden_dim

    # base offsets for this (b, d) slice
    offset_left_base  = b * stride_left_batch  + d * stride_left_hidden
    offset_right_base = b * stride_right_batch + d * stride_right_hidden
    offset_out_base   = b * stride_out_batch   + d * stride_out_hidden

    # tile offsets within the i/j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers with appropriate strides
        a_ptrs = left_ptr + offset_left_base \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k

        b_ptrs = right_ptr + offset_right_base \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_j

        # masked loads (default 0.0 for OOB)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)          # fp16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)          # fp16

        # dot product accumulates in FP32
        acc += tl.dot(a, b)

    # write back the result
    out_ptrs = out_ptr + offset_out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool / 0‑1 float)
            - weights: dict of model weights (tensors)
            - config: dict with keys "dim", "hidden_dim", ["nomask"]

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Load model weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # Linear projections (cast to fp16 for memory bandwidth)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight).to(torch.float16)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight).to(torch.float16)

    # -----------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gated activations
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # keep FP32

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Prepare tensors for the Triton kernel
    # -----------------------------------------------------------------
    # left: (B, N, N, hidden) -> (B, D, N, N)  (i, k)
    # right: (B, N, N, hidden) -> (B, D, N, N) where axes are (k, j)
    left_perm = left.permute(0, 3, 1, 2)    # (B, D, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, D, N, N) = right_T

    B, N, _, _ = left.shape

    out_perm = torch.empty(
        (B, hidden_dim, N, N), dtype=torch.float16, device=device
    )  # (B, D, N, N)

    # strides (in elements)
    stride_left_batch = left_perm.stride(0)
    stride_left_hidden = left_perm.stride(1)
    stride_left_i = left_perm.stride(2)
    stride_left_k = left_perm.stride(3)

    stride_right_batch = right_perm.stride(0)
    stride_right_hidden = right_perm.stride(1)
    stride_right_k = right_perm.stride(2)
    stride_right_j = right_perm.stride(3)

    stride_out_batch = out_perm.stride(0)
    stride_out_hidden = out_perm.stride(1)
    stride_out_i = out_perm.stride(2)
    stride_out_j = out_perm.stride(3)

    # -----------------------------------------------------------------
    # Triton batched matrix multiplication
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_outgoing_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        hidden_dim,
        stride_left_batch,
        stride_left_hidden,
        stride_left_i,
        stride_left_k,
        stride_right_batch,
        stride_right_hidden,
        stride_right_k,
        stride_right_j,
        stride_out_batch,
        stride_out_hidden,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Rearrange to original layout and continue with the remaining ops
    # -----------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B, N, N, hidden)

    # Final LayerNorm + gating + output projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # (B, N, N, dim)

    return out
