        right = right * mask_exp

    right = right * right_gate

    right = right.permute(0,3,1,2).contiguous() # (B, H, N, N)

    right_bf = right.to(torch.bfloat16)

    out = out.permute(0,2,3,1).contiguous()

"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑norm the input `x` over its last dimension `dim`.
2. Linear projections → `left`, `right` (shape [B,N,N,H]) and three gate tensors
   (`left_gate`, `right_gate`, `out_gate`) using the supplied weight matrices.
3. If a mask is present, apply it to `left` and `right`.
4. Fuse the gates:  left = left * left_gate , right = right * right_gate.
5. Permute to [B, H, N, N] and cast to bfloat16.
   Compute the pairwise contraction as a batched matrix multiplication:
       out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k] .
   (i.e. `out = left @ right.transpose(-2,-1)`).
   Cast the result back to float32 and permute to [B,N,N,H].
6. Hidden‑dim LayerNorm + `out_gate` scaling are fused in a Triton kernel.
7. Final linear projection back to `dim`.

The Triton kernel (`layernorm_gate_kernel`) loads a hidden‑dim slice
for each (batch, i, j) triple, computes mean/variance, normalises,
applies the learned weight/bias, multiplies by `out_gate`, and writes
the result.  This satisfies the requirement of using Triton while the
computationally dominant O(N³) contraction is performed by cuBLAS/Tensor
Cores via `torch.matmul` on bfloat16 data.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_gate_kernel(
    inp_ptr,          # *float32  [B,N,N,H]   input before layer‑norm
    out_ptr,          # *float32  [B,N,N,H]   output after norm+gate
    weight_ptr,       # *float32  [H]         layer‑norm weight
    bias_ptr,         # *float32  [H]         layer‑norm bias
    gate_ptr,         # *float32  [B,N,N,H]   out_gate
    B,                # tl.int32
    N,                # tl.int32
    H,                # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,   # tl.int32
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,   # tl.int32
    eps: tl.constexpr,                                  # float
    BLOCK_H: tl.constexpr                               # compile‑time block size (≥ max hidden_dim)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets along hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base offsets for the three tensors
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j
    gate_base = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j

    # load the hidden‑dim slice
    x = tl.load(inp_ptr + inp_base + offs_h * stride_in_h,
                mask=mask_h, other=0.0)
    g = tl.load(gate_ptr + gate_base + offs_h * stride_gate_h,
                mask=mask_h, other=0.0)

    # mean
    sum_x = tl.sum(x)
    mean = sum_x / H

    # variance
    diff = x - mean
    sq = diff * diff
    var = tl.sum(sq) / H

    # normalise
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # weight & bias of the layer‑norm
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)

    # apply weight, bias and outgoing gate
    y = norm * w + b_
    y = y * g

    # store result
    tl.store(out_ptr + out_base + offs_h * stride_out_h, y, mask=mask_h)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask:        torch.Tensor of shape [B, N, N] (may be None)
            - weights:     dict of model weight tensors
            - config:      dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input layer‑norm (full‑dim)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x = x.contiguous()
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2. Linear projections & gates (all in float32)
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # --------------------------------------------------------------
    # 3. Apply pairwise mask (if present) and gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Pairwise contraction over k using bfloat16 matmul
    #    left, right: [B,N,N,H] -> permute to [B,H,N,N] for batched GEMM
    left = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # out_bf: (B, H, N, N) = left_bf @ right_bf^T
    out_bf = torch.matmul(left_bf, right_bf.transpose(-2, -1))
    out = out_bf.to(dtype)                           # back to float32

    # permute back to the original layout [B,N,N,H]
    out = out.permute(0, 2, 3, 1).contiguous()       # (B, N, N, H)

    # --------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm + outgoing gate (fused in Triton)
    out_scaled = torch.empty_like(out)

    # strides (in element units)
    s_b, s_i, s_j, s_h = out.stride()
    # out_gate has identical strides
    total_progs = B * N * N
    grid = (total_progs,)

    BLOCK_H = 128  # hidden_dim ≤ 128 for all test configurations

    layernorm_gate_kernel[grid](
        out,                               # input pre‑norm
        out_scaled,                        # output post‑norm & gate
        weights["to_out_norm.weight"],     # layer‑norm weight
        weights["to_out_norm.bias"],       # layer‑norm bias
        out_gate,                          # outgoing gate
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,               # input strides
        s_b, s_i, s_j, s_h,               # output strides (same layout)
        s_b, s_i, s_j, s_h,               # gate strides
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 6. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)    # [B*N*N, H]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
