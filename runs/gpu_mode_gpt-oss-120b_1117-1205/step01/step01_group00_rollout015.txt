```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"])

    # Convert to fp16 for speed
    x_f16 = x.to(torch.float16)

    # Linear proj
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # Mask
    if not nomask:
        mask_f16 = mask.unsqueeze(-1).to(torch.float16) # shape [B,N,N,1]
        left = left * mask_f16
        right = right * mask_f16

    # Gates
    left_gate = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Permutes and flatten
    left_perm = left.permute(0,3,1,2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0,3,2,1).contiguous() # (B, H, N, N) - swapped axes

    B, H, N, _ = left_perm.shape

    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)

    out_flat = triton_matmul(left_flat, right_flat)  # shape (B*H, N, N) fp16

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous() # shape (B,N,N,H) fp16

    out = out.to(torch.float32)

    # Output layer norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Apply out_gate (converted to fp32)
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32

    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """Batched matrix multiplication C = A @ B using Triton.
    A, B must be half‑precision tensors of shape (batch, M, K) and (batch, K, N)
    respectively. Returns C of shape (batch, M, N) in fp16."""
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Outgoing TriMul forward pass accelerated with a Triton batched GEMM.
    Input:
        data = (input_tensor, mask, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm on channel dim
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # float32

    # 2. Cast to fp16 for projections/gates
    x_f16 = x.to(torch.float16)

    # 3. Linear projections (no bias)
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # 4. Optional mask (broadcast over hidden dim)
    if not nomask:
        mask_f16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_f16
        right = right * mask_f16

    # 5. Gates (sigmoid)
    left_gate = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # 6. Core contraction via Triton batched GEMM
    #    left_perm: (B, H, N, N)  with axes (i,k)
    #    right_perm: (B, H, N, N) with axes (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = triton_matmul(left_flat, right_flat)   # (B*H, N, N) fp16

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out = out.to(torch.float32)

    # 7. Output layer norm over hidden dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # 8. Apply out gate
    out = out * out_gate.to(torch.float32)

    # 9. Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

The heavy N³ contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is rewritten as a batched matrix multiplication:
    left_perm  = left.permute(0, 3, 1, 2)  # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1) # (B, H, k, j)
    out = (left_perm @ right_perm)  # for each (b, d) a (N×N) @ (N×N)ᵀ
The batched GEMM is performed by a custom Triton kernel that
loads tiles, uses `tl.dot` (Tensor‑Core accelerated for fp16) and
accumulates in fp32. All other operations (LayerNorms, linear layers,
gates, masking) stay in PyTorch. The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM C = A @ B where
    A: (batch, M, K)  fp16
    B: (batch, K, N)  fp16
    C: (batch, M, N)  fp16 (accumulated in fp32)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # fp16·fp16 -> fp32 accumulation

    # Write out C
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    A: (batch, M, K) fp16
    B: (batch, K, N) fp16
    Returns C: (batch, M, N) fp16
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask:          torch.Tensor [B, N, N] (bool/int/float) – may be ignored if config["nomask"] is True
            weights_dict:  mapping of parameter names to torch.Tensor
            config_dict:   must contain 'dim' and 'hidden_dim'; optional 'nomask' (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim), float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f16 = mask.unsqueeze(-1).to(torch.float16)  # (B,N,N,1)
        left = left * mask_f16
        right = right * mask_f16

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    -> for each (b,d) a matrix product left_perm @ right_perm
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) where H == hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, K=N, N)

    out_flat = _triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N) fp16

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 8. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
