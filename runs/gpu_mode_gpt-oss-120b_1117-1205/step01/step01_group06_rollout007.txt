"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. LayerNorm over the input channel dimension `dim`.
2. Linear projections (`left`, `right`) and three gates
   (`left_gate`, `right_gate`, `out_gate`) using the provided weight
   matrices (biases are omitted as in the reference).
3. Optional mask application on `left` and `right`.
4. Fuse the gates into the projections.
5. Pairwise contraction over the sequence dimension `k`:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batch of matrix multiplications (BF16) using
   `torch.bmm` after reshaping (`hidden_dim` is treated as a batch
   dimension).
6. LayerNorm over the hidden dimension (`hidden_dim`) with a custom
   Triton kernel (one program per output position (b,i,j)).
7. Apply the outgoing gate (`out_gate`) and final linear projection
   back to `dim`.

The kernel implements step 6; all other heavy work uses highly‑optimized
CUDA kernels (cuBLAS / cuDNN). The implementation supports optional
masking and works for all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over the hidden dimension (C = hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,            # *float32  [B,N,N,H]
    out_ptr,            # *float32  [B,N,N,H]
    weight_ptr,         # *float32  [H]
    bias_ptr,           # *float32  [H]
    B, N, H,            # int32 scalars
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,  # epsilon for stability
    BLOCK_H: tl.constexpr,  # compile‑time block size ≥ hidden_dim
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (b,i,j) from the linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets for the hidden (h) axis
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Compute base pointer for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load the vector of length H
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0,
    )
    # ------------------------------------------------------------------
    # Compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    # ------------------------------------------------------------------
    # Compute variance
    diff = x - mean
    sq = diff * diff
    sum_sq = tl.sum(sq, axis=0)
    var = sum_sq / H

    # ------------------------------------------------------------------
    # Normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ------------------------------------------------------------------
    # Apply learned scale (weight) and shift (bias)
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # ------------------------------------------------------------------
    # Store the normalized vector
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h,
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Ensure contiguous layout for later view/reshape ops
    x = x.contiguous()
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (float32)
    x_flat = x.view(-1, dim)  # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Optional mask
    if not nomask and mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # 4. Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Pairwise contraction over k (N dimension) using batched GEMM
    # Cast to bfloat16 for higher throughput
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Reshape to (B * hidden_dim, N, N)
    left_view = left_bf.view(B * hidden_dim, N, N)
    right_view = right_bf.view(B * hidden_dim, N, N)

    # Compute out = left @ right.T for each hidden slice
    # right.T corresponds to swapping the last two dimensions
    out_bf = torch.bmm(left_view, right_view.transpose(1, 2))

    # Reshape back to [B, N, N, hidden_dim] (float32)
    out = out_bf.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype).contiguous()

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm (Triton)
    out_norm = torch.empty_like(out)
    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (total_programs,)

    # hidden_dim ≤ 128 for all test configs → BLOCK_H = 128 works
    BLOCK_H = 128

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 7. Outgoing gate and final linear projection
    out_scaled = out_norm * out_gate

    out_scaled_flat = out_scaled.view(-1, hidden_dim)  # [B*N*N, hidden_dim]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])  # no bias
    final = final_flat.view(B, N, N, dim)

    return final
