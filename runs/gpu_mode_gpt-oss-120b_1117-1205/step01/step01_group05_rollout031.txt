def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype # likely float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # float32

    # Convert to half for projection
    x_h = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    if mask is not None:

        left = left * mask_h
        right = right * mask_h

    # Gates
    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Contract via Triton
    B, N, _, _ = left.shape
    # reshape
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # view
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # strides

    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_bz = right_mat.stride(0)
    stride_bj = right_mat.stride(1)
    stride_bk = right_mat.stride(2)
    stride_cz = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

        N, N, N,

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,

```python
"""
Triton implementation of the outgoing triangular multiplicative update (TriMul).

The kernel computes, for each batch and hidden channel d,
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

This is a batched matrix multiplication C = A @ Bᵀ where
    A[b*d] has shape (N, N) indexed by (i, k)
    B[b*d] has shape (N, N) indexed by (j, k)

All linear projections, gating, and layer‑norms are performed in PyTorch.
Only the O(N³) contraction is executed in Triton, using a tiled GEMM
with 64×64 tiles and 32‑element K‑tiles.  The kernel works for arbitrary
batch size (1‑2), sequence length up to 1024 and hidden dimension 128.
The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bj, stride_bk,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ
    A : (batch, M, K)   (row‑major, dtype fp16)
    B : (batch, N, K)   (row‑major, dtype fp16) – accessed transposed as (K, N)
    C : (batch, M, N)   (row‑major, dtype fp16)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile transposed (K × N)
        # Original B layout is (batch, N, K); we need elements (k, j)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # Store the result
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]      (float32)
            mask:        torch.Tensor [B, N, N] or None   (bool/float)
            weights: dict of model parameters (float32 tensors)
            config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    eps = 1e-5
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2. Linear projections & gating (half precision)
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Projections
    left = F.linear(x_h, w_left)    # [B, N, N, hidden]
    right = F.linear(x_h, w_right)

    # Optional mask (broadcasted over the hidden dim)
    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # Gates
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    # Apply gates

    right = right * right_gate

    # -----------------------------------------------------------------

    stride_cm = out_mat.stride(1)

    BLOCK_N = 64
    BLOCK_K = 32

        N,

        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -----------------------------------------------------------------
    # 4. Output layer‑norm, gating and final projection
    # -----------------------------------------------------------------

        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden]

    out_norm = out_norm * out_gate.to(torch.float32)

    output = F.linear(out_norm, weights["to_out.weight"])  # [B, N, N, dim]

    return output

"""
Triton implementation of the outgoing triangular multiplicative update (TriMul).

The kernel computes for each batch and hidden channel d:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

This is a batched GEMM C = A @ Bᵀ where
    A has shape (batch*hidden, N, N) indexed as (i, k)
    B has shape (batch*hidden, N, N) indexed as (j, k).

All linear projections, gating, and layer‑norms are done with PyTorch.
Only the heavy O(N³·hidden) contraction is executed in Triton.
The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bj, stride_bk,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ
    A : (batch, M, K)   (row‑major, fp16)
    B : (batch, N, K)   (row‑major, fp16) – accessed transposed as (K, N)
    C : (batch, M, N)   (row‑major, fp16)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Output tile pointer
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M × K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B tile transposed (K × N)
        # original B layout is (batch, N, K); we need elements (k, j)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
