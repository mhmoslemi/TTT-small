"""
TriMul (outgoing) forward pass with a fused Triton GEMM kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension.
2. Linear projections `left` and `right` (no bias) are computed, together with three
   element‑wise sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
3. An optional mask is applied to `left` and `right` before gating.
4. For each batch and each hidden dimension we need
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication `left @ rightᵀ`.  This is performed by a
   custom Triton kernel that processes all `(batch * hidden_dim)` GEMMs in one launch.
5. The result is reshaped, layer‑normed, multiplied by `out_gate` and finally projected
   back to the original channel dimension.
All heavy arithmetic is carried out in fp16 (accumulated in fp32) while the final
output is returned in fp32.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ B) where
#   A : [total_batch, M, K]   (fp16)
#   B : [total_batch, K, N]   (fp16)   (already transposed)
#   C : [total_batch, M, N]   (fp16)
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_am, stride_ak,              # A strides (row, col)
    stride_bk, stride_bn,              # B strides (row, col)
    stride_cm, stride_cn,              # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch (b * hidden)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # Pointer arithmetic for A and B tiles (including batch offset)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back (convert to fp16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------------- #
# Entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor containing all model parameters
            - config: dict with keys "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------- #
    # 1. LayerNorm over the channel dimension
    # ----------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # Cast to fp16 for the heavy part
    x_norm_fp16 = x_norm.to(torch.float16)

    # ----------------------------------------------------- #
    # 2. Projections and gates (all fp16, no bias)
    # ----------------------------------------------------- #
    left = torch.nn.functional.linear(
        x_norm_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(
        x_norm_fp16, weights["right_proj.weight"].to(torch.float16)
    )

    left_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ----------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------- #
    if not config.get("nomask", False) and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ----------------------------------------------------- #
    # 4. Apply element‑wise gates
    # ----------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------- #
    # 5. Prepare for the fused GEMM: left @ rightᵀ
    # ----------------------------------------------------- #
    # Rearrange to [B, hidden_dim, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    # Transpose the last two axes of right so that we can do A @ B
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head into a single GEMM batch dimension
    left_flat = left_perm.view(total_batch, N, N)   # [total_batch, M=N, K=N]
    right_flat = right_T.view(total_batch, N, N)   # [total_batch, K=N, N]
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # ----------------------------------------------------- #
    # 6. Launch Triton kernel
    # ----------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        total_batch,               # batch dimension
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------- #
    # 7. Reshape back to [B, N, N, hidden_dim] (float32)
    # ----------------------------------------------------- #
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # ----------------------------------------------------- #
    # 8. Output LayerNorm, output gate, final projection
    # ----------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)   # apply output gate

    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # [B, N, N, dim]

    return output
