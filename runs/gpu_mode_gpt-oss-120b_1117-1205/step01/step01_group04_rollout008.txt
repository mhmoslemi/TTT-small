"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy O(N³) contraction
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
is performed as a batched GEMM (A @ Bᵀ) where each hidden head `h`
and each batch entry are treated as an independent matrix‑multiply.
All surrounding operations (layer‑norm, linear projections, gating)
are done in PyTorch; only the contraction is off‑loaded to Triton.

The kernel works on half‑precision tensors and accumulates in
float32 for numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM computing C = A @ Bᵀ.
    A : [batch, M, K]   (float16)
    B : [batch, N, K]   (float16) – loaded transposed as (K, N)
    C : [batch, M, N]   (float16)
    """
    batch_id = tl.program_id(2)          # third axis -> independent batch
    pid_m = tl.program_id(0)             # block row
    pid_n = tl.program_id(1)             # block col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this batch
    a_base = A_ptr + batch_id * stride_a_batch
    b_base = B_ptr + batch_id * stride_b_batch
    c_base = C_ptr + batch_id * stride_c_batch

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for t in range(num_tiles_k):
        k_offset = t * BLOCK_K
        # Boolean mask for the K dimension of this tile
        mask_k = (k_offset + offs_k) < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + (k_offset + offs_k)[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile transposed on‑the‑fly: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + (k_offset + offs_k)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate FP32 product
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask  : torch.Tensor, shape [B, N, N] (bool or float)
            - weights: dict of torch.Tensors with keys
                ["norm.weight","norm.bias",
                 "left_proj.weight","right_proj.weight",
                 "left_gate.weight","right_gate.weight","out_gate.weight",
                 "to_out_norm.weight","to_out_norm.bias","to_out.weight"]
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # 0. Unpack inputs
    # ------------------------------------------------------------------ #
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 3. Optional masking (applied before gating)
    # ------------------------------------------------------------------ #
    if not nomask:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 4. Gating (sigmoid of linear on the normalized input)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Cast to half‑precision for the heavy contraction
    # ------------------------------------------------------------------ #
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 6. Reshape to (batch_h, N, N) where batch_h = B * hidden_dim
    #    Each hidden head becomes an independent GEMM batch.
    # ------------------------------------------------------------------ #
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_batch = left_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_batch = right_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    batch_h = left_batch.shape[0]  # B * hidden_dim

    # ------------------------------------------------------------------ #
    # 7. Allocate output buffer (half)
    # ------------------------------------------------------------------ #
    out_fp16 = torch.empty((batch_h, N, N), dtype=torch.float16, device=inp.device)

    # ------------------------------------------------------------------ #
    # 8. Triton launch parameters
    # ------------------------------------------------------------------ #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_h,                   # independent GEMMs
    )

    # Strides in elements (not bytes)
    stride_a_batch = left_batch.stride(0)
    stride_a_m = left_batch.stride(1)
    stride_a_k = left_batch.stride(2)

    stride_b_batch = right_batch.stride(0)
    stride_b_n = right_batch.stride(1)   # row stride (N)
    stride_b_k = right_batch.stride(2)   # col stride (K)

    stride_c_batch = out_fp16.stride(0)
    stride_c_m = out_fp16.stride(1)
    stride_c_n = out_fp16.stride(2)

    # ------------------------------------------------------------------ #
    # 9. Run the kernel: out = left_batch @ right_batchᵀ
    # ------------------------------------------------------------------ #
    tri_mul_gemm_kernel[grid](
        left_batch,
        right_batch,
        out_fp16,
        N, N, N,                         # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 10. Reshape back to [B, N, N, hidden_dim] and up‑cast to fp32
    # ------------------------------------------------------------------ #
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 11. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
