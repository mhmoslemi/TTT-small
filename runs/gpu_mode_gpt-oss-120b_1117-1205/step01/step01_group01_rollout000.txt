"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

The heavy O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is expressed as a batch of independent matrix multiplications:
    (i,k) @ (k,j) → (i,j)
for each hidden dimension d.  After reshaping we obtain B·H independent
N×N matrix products which are computed in a single Triton kernel
using 64×64 shared‑memory tiles and fp16 arithmetic with fp32 accumulation.
All surrounding LayerNorms, linear projections and gating are performed in
standard PyTorch (mixed precision) to keep the kernel simple yet fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr,                      # [BH, M, K] matrix A (fp16)
    b_ptr,                      # [BH, K, N] matrix B (fp16), already transposed
    c_ptr,                      # [BH, M, N] output matrix C (fp16)
    M, N, K,                    # all equal to seq_len
    stride_batch_a, stride_am, stride_ak,   # strides for A
    stride_batch_b, stride_bk, stride_bn,   # strides for B
    stride_batch_c, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block index in M dimension
    pid_n = tl.program_id(1)          # block index in N dimension
    pid_bh = tl.program_id(2)         # batch·head index

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers to current tile of A and B
        a_ptrs = a_ptr + pid_bh * stride_batch_a \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_bh * stride_batch_b \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K), fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N), fp16

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulation

    # Write back result (cast to fp16)
    c_ptrs = c_ptr + pid_bh * stride_batch_c \
               + offs_m[:, None] * stride_cm \
               + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (float32) or None
            - weights:     dict of model parameters (see module definition)
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # fp32

    # --------------------------------------------------------------------
    # 2. Mixed‑precision linear projections
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H] fp16

    # --------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dim)
    # --------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,H] fp16
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,H] fp16

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5. Batched GEMM via Triton (cubic contraction)
    # --------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape: [B, N, N, H]

    # Rearrange for GEMM: left (i,k), right (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, k, j)

    # Collapse batch and head dimensions -> independent matrices
    BH = B * H
    A = left_perm.reshape(BH, N, N).contiguous()   # (BH, M, K)
    Bmat = right_perm.reshape(BH, N, N).contiguous()  # (BH, K, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16, device=device)  # (BH, M, N)

    # Tile sizes (tuned for NVIDIA B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        BH                         # batch*head dimension
    )

    batched_gemm_kernel[grid](
        A,                     # a_ptr
        Bmat,                  # b_ptr
        C,                     # c_ptr
        N, N, N,               # M, N, K
        A.stride(0), A.stride(1), A.stride(2),   # batch, row, col strides for A
        Bmat.stride(0), Bmat.stride(1), Bmat.stride(2),   # for B
        C.stride(0), C.stride(1), C.stride(2),   # for C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore [B, N, N, H] layout
    out_fp16 = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # --------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, final gate, output projection
    # --------------------------------------------------------------------
    out_fp32 = out_fp16.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gate (out_gate computed in fp16, cast to fp32)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)          # [B, N, N, dim] fp32

    return output
