    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)

        right = right * right_gate

"""
Optimized outgoing TriMul forward pass.

- LayerNorms, linear projections and the large batched matrix multiplication are
  performed with PyTorch (cuBLAS) for peak performance.
- Two tiny Triton kernels are used to fuse element‑wise operations that would
  otherwise require separate kernels:
    1) `apply_gate_kernel` multiplies the projected tensors by their respective
       gates (and optionally a mask) in a single pass.
    2) `apply_mul_fp32_kernel` multiplies the post‑norm tensor by the output
       gate before the final projection.
  Both kernels operate on flattened contiguous buffers.
- Mixed precision (FP16 for the heavy GEMM) reduces memory traffic while the
  remaining operations stay in FP32 for numerical fidelity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_gate_kernel(
    left_ptr,               # float16 *
    right_ptr,              # float16 *
    left_gate_ptr,          # float16 *
    right_gate_ptr,         # float16 *
    total_elements,         # int (runtime)
    BLOCK_SIZE: tl.constexpr
):
    """In‑place elementwise multiplication:
       left  = left  * left_gate
       right = right * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs,   mask=mask, other=0.0)
    r = tl.load(right_ptr + offs,  mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    l = l * lg
    r = r * rg

    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


@triton.jit
def apply_mul_fp32_kernel(
    out_ptr,                # float32 *
    gate_ptr,               # float32 *
    total_elements,         # int (runtime)
    BLOCK_SIZE: tl.constexpr
):
    """In‑place multiplication out = out * gate (both FP32)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    o = tl.load(out_ptr + offs,   mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs,  mask=mask, other=0.0)

    o = o * g

    tl.store(out_ptr + offs, o, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or float)
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # extract weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # LayerNorm on the input tensor (over the last dimension)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # Linear projections to hidden_dim (keep them in FP16 for the heavy GEMM)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight).to(torch.float16)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight).to(torch.float16) # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # mask Tensor: [B, N, N] -> [B, N, N, 1] -> broadcast
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # half precision view
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gates (sigmoid after a linear projection)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))                       # [B,N,N,hidden_dim] float32

    # -----------------------------------------------------------------
    # Fuse per‑element gating (and mask) using a tiny Triton kernel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    total = B * N * N * H
    BLOCK = 65536
    grid = (triton.cdiv(total, BLOCK),)
    # ensure tensors are contiguous for the flat view the kernel expects
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    apply_gate_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        total,
        BLOCK_SIZE=BLOCK,
    )

    # -----------------------------------------------------------------
    # Batched matrix multiplication: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # This is a batch of (B*hidden_dim) GEMMs of size N×N.
    # -----------------------------------------------------------------
    # reshape to (Batch*Hidden, N, N) where each batch corresponds to one hidden channel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (P, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (P, N, N) -> transpose on k
    out_mat = torch.bmm(left_mat, right_mat)  # FP16 GEMM, result (P, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim] FP16
    out = out.to(torch.float32)  # convert before the remaining FP32 ops

    # -----------------------------------------------------------------
    # Final LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B,N,N,hidden_dim] float32

    # -----------------------------------------------------------------
    # Apply the output gate via a Triton kernel (FP32)
    # -----------------------------------------------------------------
    total_out = out.numel()
    grid_out = (triton.cdiv(total_out, BLOCK),)
    out_gate = out_gate.contiguous()
    apply_mul_fp32_kernel[grid_out](
        out.data_ptr(),
        out_gate.data_ptr(),
        total_out,
        BLOCK_SIZE=BLOCK,
    )

    # -----------------------------------------------------------------
    # Linear projection back to the original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim] float32

    return out
