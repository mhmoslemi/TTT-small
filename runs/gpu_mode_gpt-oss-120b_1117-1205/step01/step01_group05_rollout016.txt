def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dims = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm (input)
    gamma = weights["norm.weight"].to(device) # shape [dim]
    beta = weights["norm.bias"].to(device)
    # x as fp32
    x = input_tensor
    # compute mean and var along dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta

    # cast to fp16 for projections/gates
    x_h = x_norm.to(torch.float16)

    # Linear weights (fp16)
    w_left = weights["left_proj.weight"].to(torch.float16) # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float32) # final projection often fp32

    # Left and right projections
    left = F.linear(x_h, w_left) # shape (B,N,N,H)
    right = F.linear(x_h, w_right)

    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))  # shape (B,N,N,H)

    right = right * right_gate

    # right_T matrix (transpose last two dims)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    stride_cz, stride_cm, stride_cn = out_mat.stride()

    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _batched_matmul_kernel[grid](
        left_mat,
        right_T,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp16
    out_f32 = out.to(torch.float32)

    # Output LayerNorm
    gamma_out = weights["to_out_norm.weight"].to(torch.float32)
    beta_out = weights["to_out_norm.bias"].to(torch.float32)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # Apply out gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection
    output = F.linear(out_norm, w_to_out)  # shape (B,N,N,dim)

    return output

"""
Optimized Triton kernel for the outgoing Triangle Multiplicative Update (TriMul).
The kernel computes the contraction:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
where left and right are linear projections of the normalized input.
All other operations (layer‑norm, gates, final projection) are performed in
PyTorch; the heavy O(N³) contraction is performed by a batched GEMM kernel in
Triton that processes (batch*hidden_dim) independent (N,N) @ (N,N) matrix
multiplications.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B, where A: (batch, M, K), B: (batch, K, N).
    A, B, C are stored in row‑major order. Accumulates in fp32 and writes
    fp16 result.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row index
    pid_n = tl.program_id(2)   # block column index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Store result (convert back to fp16)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Entry point for the outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool or float)
        - weights      : dict of torch.Tensors containing all linear / norm params
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm over the last dimension ----------
    gamma = weights["norm.weight"].to(device)  # shape [dim]
    beta = weights["norm.bias"].to(device)    # shape [dim]

    x = input_tensor  # float32
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta                           # [B,N,N,dim]

    # ---------- Cast to half precision for the heavy work ----------
    x_h = x_norm.to(torch.float16)

    # ---------- Linear weights (half precision where possible) ----------
    w_left = weights["left_proj.weight"].to(torch.float16)       # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    w_to_out = weights["to_out.weight"].to(torch.float32)       # final projection stays fp32

    # ---------- Projections ----------
    left = F.linear(x_h, w_left)        # [B,N,N,hidden_dim]
    right = F.linear(x_h, w_right)

    # ---------- Optional mask ----------
    if mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    # ---------- Batched matrix multiplication (the O(N³) part) ----------
    B, N, _, H = left.shape   # H == hidden_dim

    # reshape to (B*H, N, N) for A
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose the second N dimension of right and reshape similarly
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16
    out_f32 = out.to(torch.float32)

    # ---------- Output LayerNorm ----------
    gamma_out = weights["to_out_norm.weight"].to(torch.float32)   # (hidden_dim)
    beta_out = weights["to_out_norm.bias"].to(torch.float32)    # (hidden_dim)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # ---------- Apply output gate ----------
    out_norm = out_norm * out_gate.to(torch.float32)

    # ---------- Final linear projection ----------
    output = F.linear(out_norm, w_to_out)   # [B,N,N,dim] float32

    return output

"""
Optimized Triton kernel for the outgoing Triangle Multiplicative Update (TriMul).

The heavy part of the TriMul forward pass is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be viewed as a batched matrix multiplication of (N,N) @ (N,N)
repeated for every hidden dimension slice.  This kernel performs that
batched GEMM in half‑precision (accumulating in fp32) and the surrounding
operations (layer‑norms, gating, final projection) are handled in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B, where A: (batch, M, K), B: (batch, K, N).

    All tensors are stored in row‑major order.  The kernel loops over the K
    dimension in tiles, accumulates in fp32, and writes the result back as fp16.
    """
    batch_id = tl.program_id(0)          # which (batch * hidden) matrix
    pid_m = tl.program_id(1)            # tile row index
    pid_n = tl.program_id(2)            # tile column index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to the output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B tile (K x N) – note B is already transposed on the k‑axis
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # store result as fp16
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor, shape [B, N, N] (bool or float) or None
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (float32)
    # ------------------------------------------------------------------
    gamma = weights["norm.weight"].to(device)   # (dim,)
    beta  = weights["norm.bias"].to(device)    # (dim,)

    x = input_tensor                         # [B,N,N,dim] fp32
    mean = x.mean(dim=-1, keepdim=True)
    var  = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta            # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Cast to half‑precision for projections and gates
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # Linear weights (half‑precision where possible)
    w_left      = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    w_right     = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate= weights["right_gate.weight"].to(torch.float16)
    w_out_gate  = weights["out_gate.weight"].to(torch.float16)

    w_to_out    = weights["to_out.weight"].to(torch.float32)     # final projection

    # ------------------------------------------------------------------
    # 3. Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left)   # [B,N,N,hidden_dim]
    right = F.linear(x_h, w_right)

    # ------------------------------------------------------------------
    # 4. Apply optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))  # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Batched matrix multiplication (the O(N³) part)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim

    # reshape left → (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose right on its second N dimension and reshape similarly:
    # right[b,i,k,d] -> right_T[b,d,k,i]
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tiling parameters (tuned for the B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16
    out_f32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    gamma_out = weights["to_out_norm.weight"].to(torch.float32)   # (hidden_dim,)
    beta_out  = weights["to_out_norm.bias"].to(torch.float32)    # (hidden_dim,)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o  = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # ------------------------------------------------------------------
    # 8. Apply output gate (broadcasted)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    output = F.linear(out_norm, w_to_out)   # [B,N,N,dim] fp32

    return output
