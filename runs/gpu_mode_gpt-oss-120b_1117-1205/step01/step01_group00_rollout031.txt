    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load tiles
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ B.
    A, B: (batch, M, K) and (batch, K, N) in torch.float16, contiguous.
    Returns C: (batch, M, N) torch.float16.
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # tile size heuristics (simple static choice)
    BLOCK_M = 64 if M >= 64 else 32
    BLOCK_N = 64 if N >= 64 else 32
    BLOCK_K = 32 if K >= 32 else 16

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: contains 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16),
    )  # [B, N, N, hidden_dim], fp16
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16),
    )  # same shape

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gates (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16),
    ).sigmoid()                                 # [B, N, N, hidden_dim], fp16
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16),
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16),
    ).sigmoid()                                 # [B, N, N, hidden_dim], fp16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # permute to (B, H, N, N) – required layout for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # reshape to (B*H, N, N) for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # batched matrix multiply (half‑precision)
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N), fp16

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # float32 for subsequent ops

    # free intermediate tensors early
    del left, right, left_perm, right_perm, left_flat, right_flat, out_flat

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float32),
    )
    return out
