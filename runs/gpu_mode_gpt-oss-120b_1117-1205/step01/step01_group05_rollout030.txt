    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)  # shape [B,N,N,dim]; float32

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elem_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) operator.
    All linear / layer‑norm ops are performed in PyTorch,
    while the final element‑wise gating of the output is fused in a tiny Triton kernel.
    The expensive pairwise contraction is realized with a batched GEMM
    (torch.bmm) after reshaping the tensors, which leverages cuBLAS/TensorCores.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- Input layer‑norm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)   # [B,N,N,dim] fp32

    # Cast to half for the bulk of the linear work
    x_h = x.to(torch.float16)

    # ---------- Linear projections ----------
    w_left = weights["left_proj.weight"].to(torch.float16)       # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)          # [B,N,N,hidden_dim]
    right = F.linear(x_h, w_right)

    # ---------- Gates ----------
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()           # [B,N,N,hidden_dim]

    # ---------- Optional mask ----------
    if mask is not None and not config.get("nomask", False):
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ---------- Apply gating ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Batched matrix multiplication  (left @ rightᵀ) ----------
    B, N, _, H = left.shape      # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (batch, M, K)
    right_mat = right.permute(0, 3, 1, 2).contiguous()                    # (B, H, N, N)
    # right transpose: (B*H, K, N)
    right_mat_T = right_mat.permute(0, 1, 3, 2).contiguous().view(B * H, N, N)

    # Compute batched GEMM with cuBLAS (fp16 inputs, fp32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat_T)      # (B*H, N, N) fp16

    # Reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16
    out_f32 = out.to(torch.float32)

    # ---------- Output layer‑norm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_f32, (hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=eps)

    # ---------- Fuse out‑gate using Triton ----------
    out_gate_f32 = out_gate.to(torch.float32)
    out_norm_flat = out_norm.contiguous().view(-1)
    out_gate_flat = out_gate_f32.contiguous().view(-1)
    out_gated_flat = torch.empty_like(out_norm_flat)

    size = out_norm_flat.numel()
    BLOCK = 65536
    grid = (triton.cdiv(size, BLOCK),)
    _elem_mul_kernel[grid](out_norm_flat, out_gate_flat, out_gated_flat, size,
                           BLOCK=BLOCK, num_warps=4)

    out_gated = out_gated_flat.view_as(out_norm)   # float32

    # ---------- Final linear ----------
    w_to_out = weights["to_out.weight"]
    output = F.linear(out_gated, w_to_out)   # [B,N,N,dim] float32

    return output

"""
TriMul (outgoing) forward pass.

The heavy pair‑wise contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM using torch.bmm (cuBLAS/TensorCores).
All LayerNorms and Linear layers run in PyTorch; a tiny Triton kernel
fuses the final element‑wise multiplication with the output gate.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elem_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    """Element‑wise a * b → out, 1‑D flattened tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: Tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim] (float32)
            mask   : torch.Tensor [B, N, N] or None
            weights: dict of torch.Tensors for all Linear/LayerNorm parameters
            config : dict with keys "dim", "hidden_dim", optionally "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # Cast to fp16 for the bulk of the linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim) – no bias
    # ------------------------------------------------------------------
    w_left = weights["left_proj.weight"].to(torch.float16)      # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)      # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, w_right)

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)      # [B, N, N, 1] fp16
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Batched matrix multiplication: left @ rightᵀ
    # Shape gymnastics: (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (batch, M, K)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)      # (batch, K, N)

    # fp16 @ fp16 → fp16 (cuBLAS uses Tensor Cores, accumulates in fp32 internally)
    out_mat = torch.bmm(left_mat, right_mat_T)                                   # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()             # fp16
    out_f32 = out.to(torch.float32)                                              # fp32 for LayerNorm

    # ------------------------------------------------------------------
    # Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B, N, N, hidden_dim] float32

    # ------------------------------------------------------------------
    # Fuse out‑gate using a tiny Triton kernel (element‑wise multiplication)
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.to(torch.float32)
    out_norm_flat = out_norm.contiguous().view(-1)
    out_gate_flat = out_gate_f32.contiguous().view(-1)
    out_gated_flat = torch.empty_like(out_norm_flat)

    size = out_norm_flat.numel()
    BLOCK = 65536
    grid = (triton.cdiv(size, BLOCK),)
    _elem_mul_kernel[grid](
        out_norm_flat, out_gate_flat, out_gated_flat, size,
        BLOCK=BLOCK, num_warps=4,
    )
    out_gated = out_gated_flat.view_as(out_norm)   # back to [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Final projection back to original dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"]
    output = F.linear(out_gated, w_to_out)  # [B, N, N, dim] float32

    return output
