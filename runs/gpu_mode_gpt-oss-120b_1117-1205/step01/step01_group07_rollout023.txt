"""
Optimized outgoing TriMul forward kernel.

The heavy part of the TriMul module is the 3‑D tensor contraction
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication (per batch & per hidden channel).
This implementation
  * performs LayerNorms and gating in PyTorch,
  * casts the projected tensors to fp16 for bandwidth savings,
  * uses a Triton kernel that streams tiles of A (i×k) and B (k×j) and
    accumulates in fp32.
No memory copies of the permuted tensors are performed; we operate on
strided views and flatten the (batch, head) dimensions via a program id.
The final result is cast back to fp32 and processed by the remaining
LayerNorm, output gating and linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul contraction
#   A : (B, H, M, K)   fp16   where M = i, K = k
#   B : (B, H, K, N)   fp16   where N = j (transposed order)
#   C : (B, H, M, N)   fp16   output
# -------------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_head, stride_am, stride_ak,
    stride_b_batch, stride_b_head, stride_bk, stride_bn,
    stride_c_batch, stride_c_head, stride_cm, stride_cn,
    HEAD_DIM,                         # hidden dimension (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --- program ids -------------------------------------------------------
    pid_m = tl.program_id(0)               # tile row index (i)
    pid_n = tl.program_id(1)               # tile col index (j)
    pid   = tl.program_id(2)               # combined (batch, head)

    # recover batch and head from combined id
    batch = pid // HEAD_DIM
    head  = pid - batch * HEAD_DIM        # pid % HEAD_DIM

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this (batch, head) pair
    a_base = A_ptr + batch * stride_a_batch + head * stride_a_head
    b_base = B_ptr + batch * stride_b_batch + head * stride_b_head
    c_base = C_ptr + batch * stride_c_batch + head * stride_c_head

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers with current k‑offset
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # out‑of‑bounds masks
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        acc += tl.dot(a, b)                           # fp32 accumulation

    # write back the result
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# -------------------------------------------------------------------------
# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, S, S, D] (float32)
            - mask_tensor: torch.Tensor [B, S, S] (bool or float) – may be ignored.
            - weights: dict of parameters (see code for required keys)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor [B, S, S, D] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------
    # LayerNorm on input
    # ------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                            # [B, S, S, D] float32

    # ------------------------------------------------------------
    # Linear projections (half precision)
    # ------------------------------------------------------------
    left = F.linear(x, left_proj_w).to(torch.float16)   # [B, S, S, H]
    right = F.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, S, S, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))     # keep fp32

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # Triton batched matmul: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ----------------------------------------------------------------
    B, S, _, _ = left.shape  # S == seq_len, _ hidden dim == hidden_dim

    # Permute to (batch, head, M, K) and (batch, head, K, N)
    left_perm  = left.permute(0, 3, 1, 2)          # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1)         # (B, H, k, j)

    # Output tensor (batch, head, i, j) – fp16
    out_perm = torch.empty((B, hidden_dim, S, S),
                           dtype=torch.float16,
                           device=device)

    # Strides (in elements)
    stride_a_batch, stride_a_head, stride_am, stride_ak = left_perm.stride()
    stride_b_batch, stride_b_head, stride_bk, stride_bn = right_perm.stride()
    stride_c_batch, stride_c_head, stride_cm, stride_cn = out_perm.stride()

    # Block sizes – tuned for typical seq_len up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(S, BLOCK_M),
            triton.cdiv(S, BLOCK_N),
            B * hidden_dim)

    trimul_outgoing_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        S, S, S,                         # M, N, K
        stride_a_batch, stride_a_head, stride_am, stride_ak,
        stride_b_batch, stride_b_head, stride_bk, stride_bn,
        stride_c_batch, stride_c_head, stride_cm, stride_cn,
        hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Rearrange back to (B, S, S, H) and cast to fp32 for the rest of the module
    out = out_perm.permute(0, 2, 3, 1).to(torch.float32)   # [B, S, S, H]

    # ------------------------------------------------------------
    # Final LayerNorm, gating and projection
    # ------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    out = out * out_gate                     # broadcast over last dim
    out = F.linear(out, to_out_w)           # [B, S, S, D]

    return out
