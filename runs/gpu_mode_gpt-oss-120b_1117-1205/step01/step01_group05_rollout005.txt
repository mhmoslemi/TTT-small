def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim, hidden_dim = config['dim'], config['hidden_dim']

    # LayerNorm
    norm_weight = weights['norm.weight'].to(torch.float32)
    norm_bias = weights['norm.bias'].to(torch.float32)
    eps=1e-5

    x = input_tensor
    # compute mean and variance over last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # half conversion for projection speed
    x_h = x_norm.to(torch.float16)

    # Linear projections
    left_proj_weight = weights['left_proj.weight'].to(torch.float16)
    right_proj_weight = weights['right_proj.weight'].to(torch.float16)
    left = F.linear(x_h, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x_h, right_proj_weight)

    # Mask
    if mask is not None:
        # mask could be bool or float
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # Gates
    left_gate_weight = weights['left_gate.weight'].to(torch.float16)
    right_gate_weight = weights['right_gate.weight'].to(torch.float16)
    out_gate_weight = weights['out_gate.weight'].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_weight).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # contraction via Triton
    B, N, _, hd = left.shape
    # reshape to B*hd, N, N
    left_mat = left.permute(0,3,1,2).contiguous().view(B*hd, N, N)   # (M= N, K=N)
    # right transposed for matmul
    right_mat = right.permute(0,3,1,2).contiguous().permute(0,1,3,2).contiguous().view(B*hd, N, N) # (K,N)
    # Note right_mat is after permute result (K,N)

    out_mat = torch.empty((B*hd, N, N), dtype=torch.float16, device=device)

    # compute strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # maybe smaller for register use (but okay)
    # Choose warps maybe
    grid = (B*hd, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape out_mat back to [B,N,N,hd]
    out = out_mat.view(B, hd, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hd] in half

    # Convert to float32 for further operations

    # to_out_norm
    to_out_norm_weight = weights['to_out_norm.weight'].to(torch.float32)

"""
TriMul outgoing kernel (forward only).

The module normalizes the input, projects to a hidden dimension, applies a mask
and sigmoid gates, performs the core contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication over the sequence dimension N.
The heavy contraction is implemented with a custom Triton kernel; all other
operations remain in PyTorch. The final output has shape [B, N, N, dim] and
dtype float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where
#   A : (batch, M, K)
#   B : (batch, K, N)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # store result (cast back to FP16)
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask       : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights    : dict of model parameters
            - config     : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # LayerNorm on input (over channel dim)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"].to(torch.float32)
    b_norm = weights["norm.bias"].to(torch.float32)

    x = input_tensor  # [B, N, N, dim]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * w_norm + b_norm  # [B, N, N, dim]

    # --------------------------------------------------------------
    # Projections and gates (use FP16 for speed)
    # --------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # Linear weights (FP16)
    w_left = weights["left_proj.weight"].to(torch.float16)      # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)
    w_outgate = weights["out_gate.weight"].to(torch.float16)

    # Project
    left = F.linear(x_h, w_left)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # Optional mask (applied after projection, before gating)
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # Gates
    left_gate = F.linear(x_h, w_lgate).sigmoid()
    right_gate = F.linear(x_h, w_rgate).sigmoid()
    out_gate = F.linear(x_h, w_outgate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Core contraction via Triton batched GEMM
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape left to (B*H, N, N) where each matrix is (i,k)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # reshape right and transpose so that each matrix is (k,j)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # output buffer
    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides for Triton
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # Tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, hidden_dim] (FP16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_f32 = out.to(torch.float32)

    # --------------------------------------------------------------
    # Output LayerNorm over hidden_dim
    # --------------------------------------------------------------
    w_outnorm = weights["to_out_norm.weight"].to(torch.float32)
    b_outnorm = weights["to_out_norm.bias"].to(torch.float32)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * w_outnorm + b_outnorm

    # Apply output gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # --------------------------------------------------------------
    # Final linear projection back to original dim
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)  # [B, N, N, dim]

    return output
