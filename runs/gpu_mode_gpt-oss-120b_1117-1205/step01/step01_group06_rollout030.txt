@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_y = tl.program_id(0)
    pid_n = tl.program_id(1)
    # compute batch idx, tile_m
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_idx = pid_y // num_tiles_m
    tile_m = pid_y % num_tiles_m
    # offsets
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # compute base pointer offsets for a and b per batch
    a_batch_offset = batch_idx * stride_a_batch
    b_batch_offset = batch_idx * stride_b_batch
    c_batch_offset = batch_idx * stride_c_batch
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # load a: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # load b: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + b_batch_offset + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        # dot
        acc += tl.dot(a, b)
    # write output
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16) if tl.is_fp16(c_ptr) else acc, mask=mask_m[:, None] & mask_n[None, :])
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_y = tl.program_id(0)
    pid_n = tl.program_id(1)
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_idx = pid_y // num_tiles_m
    tile_m = pid_y % num_tiles_m
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N
    a_batch_offset = batch_idx * stride_a_batch
    b_batch_offset = batch_idx * stride_b_batch
    c_batch_offset = batch_idx * stride_c_batch
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + b_batch_offset + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.bfloat16).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.bfloat16).to(tl.float32)
        acc += tl.dot(a, b)
    c = acc.to(tl.bfloat16)
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass optimized with Triton.
    Steps:
    1) LayerNorm over input channels.
    2) Linear projections (left/right) and gates.
    3) Apply mask and gates.
    4) Batched BF16 matrix multiplication (left @ right^T) via Triton.
    5) Hidden‑dim LayerNorm (torch).
    6) Apply out_gate.
    7) Final linear projection to original dim.
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input layer norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    B, N, _, _ = x.shape

    # 2. Linear projections (float32)
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # 3. Mask and gating
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # Cast to BF16 for matmul
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # free unused tensors (optional)
    del left, right, left_gate, right_gate, x_flat, x

    # 4. Batched BF16 matmul (left @ right^T) via Triton
    # Rearrange dimensions to (batch*head, N, N)
    # left: (B,N,N,H) -> (B,H,N,N) -> (B*H,N,N)
    left_mat = left_bf.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right transpose: (B,N,N,H) -> (B,H,N,N) where last two dims are (k,j)
    right_mat = right_bf.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.bfloat16, device=device)

    # strides in element units
    s_a_batch, s_a_m, s_a_k = left_mat.stride()
    s_b_batch, s_b_k, s_b_n = right_mat.stride()
    s_c_batch, s_c_m, s_c_n = out_mat.stride()

    # launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_y = (B * hidden_dim) * num_tiles_m
    grid = (total_y, num_tiles_n)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * hidden_dim,
        N,
        N,
        N,
        s_a_batch, s_a_m, s_a_k,
        s_b_batch, s_b_k, s_b_n,
        s_c_batch, s_c_m, s_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 5. Convert back to float32 and reshape
    out = out_mat.to(dtype).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # 5b. Hidden-dim LayerNorm (torch)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 6. Apply out_gate (float32) – note out_gate is already float32 from earlier
    out = out * out_gate

    # 7. Final linear projection back to dim
    out_flat = out.view(-1, hidden_dim)
    final_flat = F.linear(out_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final

"""
Outgoing TriMul forward pass (AlphaFold3) accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input over its channel dimension.
2. Compute left/right projections and three gate vectors with linear layers.
3. Apply optional pairwise mask and element‑wise gates.
4. Perform a batched BF16 matrix multiplication
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel (batch = B * hidden_dim).
5. Layer‑norm over the hidden dimension (torch implementation).
6. Multiply by the output gate.
7. Final linear projection back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,               # *bfloat16  [batch, M, K]
    b_ptr,               # *bfloat16  [batch, K, N]
    c_ptr,               # *bfloat16  [batch, M, N]
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_y = tl.program_id(0)                # encodes (batch, tile_m)
    pid_n = tl.program_id(1)                # tile_n

    # Number of tiles along M dimension
    num_tiles_m = tl.cdiv(M, BLOCK_M)

    # Decode batch index and tile index along M
    batch_idx = pid_y // num_tiles_m
    tile_m = pid_y % num_tiles_m

    # Offsets within the output tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base offsets for this batch
    a_batch_offset = batch_idx * stride_a_batch
    b_batch_offset = batch_idx * stride_b_batch
    c_batch_offset = batch_idx * stride_c_batch

    # Accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [BLOCK_M, BLOCK_K]
        a_ptrs = a_ptr + a_batch_offset \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.bfloat16).to(tl.float32)

        # Load B tile [BLOCK_K, BLOCK_N]
        b_ptrs = b_ptr + b_batch_offset \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.bfloat16).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back in BF16
    c = acc.to(tl.bfloat16)
    c_ptrs = c_ptr + c_batch_offset \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor, shape [B, N, N] (may be ignored)
            - weights: dict of torch.Tensor containing all model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2) Linear projections and gates (float32)
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # --------------------------------------------------------------
    # 3) Mask and gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4) Batched BF16 matmul: left @ right^T using Triton
    # Cast to BF16 to reduce memory traffic
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Free no‑longer‑needed FP32 tensors early
    del left, right, left_gate, right_gate, x, x_flat

    # Rearrange to (batch*head, N, N) where head = hidden_dim
    left_mat = left_bf.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_bf.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.bfloat16, device=device)

    # Strides (in elements) for Triton
    s_a_batch, s_a_m, s_a_k = left_mat.stride()
    s_b_batch, s_b_k, s_b_n = right_mat.stride()
    s_c_batch, s_c_m, s_c_n = out_mat.stride()

    # Launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_y = (B * hidden_dim) * num_tiles_m
    grid = (total_y, num_tiles_n)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * hidden_dim,
        N,
        N,
        N,
        s_a_batch, s_a_m, s_a_k,
        s_b_batch, s_b_k, s_b_n,
        s_c_batch, s_c_m, s_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 5) Convert back to FP32 and reshape to (B,N,N,hidden_dim)
    out = out_mat.to(dtype).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # 5b) Hidden‑dim LayerNorm (torch)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 6) Apply out gate
    out = out * out_gate

    # --------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    out_flat = out.view(-1, hidden_dim)                     # (B*N*N, hidden_dim)
    final_flat = F.linear(out_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
