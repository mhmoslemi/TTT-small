"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm:
1. Layer‑norm the input tensor across the last channel dimension.
2. Project the normalized tensor to a hidden dimension (left/right) and compute
   three gating vectors (left, right, out) with a sigmoid.
3. Apply the mask (if present) and the left/right gates.
4. Perform the core triangle multiplication:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM where each hidden channel d is an independent
   matrix multiplication (N×N)·(N×N)ᵀ.  The computation is carried out by a
   Triton kernel that tiles over M, N and K dimensions.
5. Layer‑norm the result over the hidden dimension, apply the output gate,
   and project back to the original channel dimension.

All heavy arithmetic (the N³ reduction) runs in FP16 with FP32 accumulation.
The final output is FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) @ B (K×N) -> C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes
    stride_am, stride_ak,              # A strides (row, col)
    stride_bk, stride_bn,              # B strides (row, col)
    stride_cm, stride_cn,              # C strides (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Grid: (batch, block_row, block_col)
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Base pointer for this batch (hidden channel)
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the tiles of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(0, num_k_tiles):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul operator (outgoing version).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (float32) or None
            - weights_dict: contains all model weights as torch.Tensors
            - config_dict: contains 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm over channel dimension
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b                               # [B, N, N, C] (float32)

    # --------------------------------------------------------------
    # 2. Projections & gating (FP16 for speed)
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # Linear projections to hidden dim
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, hidden]
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # Apply mask if present; mask is broadcast over the hidden dim
    if mask is None:
        mask_f16 = torch.ones(B, N, N, dtype=torch.float16, device=device).unsqueeze(-1)
    else:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)                     # [B, N, N, 1]
    left = left * mask_f16
    right = right * mask_f16

    # Gates (sigmoid)
    left_gate = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply left/right gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3. Core TriMul: batched matrix multiplication via Triton
    #    out[b, d] = left[b, d] @ right[b, d].T
    # --------------------------------------------------------------
    # reshape to (batch*hidden, N, N) where each slice is a dense matrix
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N) -> rows i, cols k
    right = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N) -> rows k, cols j (transposed)

    total_batches = B * hidden
    left_mat = left.reshape(total_batches, N, N)
    right_mat = right.reshape(total_batches, N, N)

    # Allocate output in FP16
    out_mat = torch.empty_like(left_mat)

    # Choose tile sizes based on sequence length
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                   # M, N, K
        left_mat.stride(1), left_mat.stride(2),    # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2),  # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),      # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),   # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back: [B, N, N, hidden] (FP32)
    out = out_mat.reshape(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 4. Output LayerNorm, output gate, and final projection
    # --------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = ((out - mean_out) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * w_out_norm + b_out_norm

    # Apply output gate (convert gate to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to original dimension C
    w_to_out = weights["to_out.weight"].to(torch.float32)   # (C, hidden)
    output = F.linear(out_norm, w_to_out)                  # [B, N, N, C] (float32)

    return output
