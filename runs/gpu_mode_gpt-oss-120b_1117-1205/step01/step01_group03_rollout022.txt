def gated_mask_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    total_elements, hidden,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_idx = offs // hidden
    mask_valid = offs < total_elements
    mask_valid = mask_valid & (mask_idx < total_elements // hidden)  # ensure mask_idx valid
    mask = tl.load(mask_ptr + mask_idx, mask=mask_idx < (total_elements // hidden), other=0.0)
    left = tl.load(left_ptr + offs, mask=mask_valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_valid, other=0.0)

    tl.store(left_out_ptr + offs, left_out, mask=mask_valid)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    eps = 1e-5

    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    # compute mean and var across dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias   # shape (B,N,N,dim)

    # FP16 conversion
    x_f16 = x.to(torch.float16)

    # Linear projections
    w_left = weights['left_proj.weight'].to(torch.float16)
    w_right = weights['right_proj.weight'].to(torch.float16)

    left_proj = torch.nn.functional.linear(x_f16, w_left)   # (B,N,N,hidden)
    right_proj = torch.nn.functional.linear(x_f16, w_right)

    # Gates
    w_left_gate = weights['left_gate.weight'].to(torch.float16)
    w_right_gate = weights['right_gate.weight'].to(torch.float16)
    w_out_gate = weights['out_gate.weight'].to(torch.float16)

    right_gate = torch.nn.functional.linear(x_f16, w_right_gate).sigmoid()
    out_gate = torch.nn.functional.linear(x_f16, w_out_gate).sigmoid()

    # Mask handling (broadcast)
    if mask is None:

    else:

    # Ensure contiguous

    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f16 = mask_f16.contiguous()

    right = torch.empty_like(right_proj)

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gated_mask_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    total, hidden, mask_total,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    valid = offs < total

    mask_idx = offs // hidden
    mask_valid = mask_idx < mask_total

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=0.0)

    left_out = left * mask_val * left_gate
    right_out = right * mask_val * right_gate

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    The heavy O(N³·hidden) pairwise multiplication is performed with torch.bmm
    (cuBLAS), while a tiny Triton kernel fuses mask application and the two
    element‑wise gates on the projected tensors.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. FP16 projection + gate computation (torch)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    left_proj = F.linear(x_f16, w_left)   # (B,N,N,hidden)
    right_proj = F.linear(x_f16, w_right)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_f16, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_f16, w_out_gate))

    # ------------------------------------------------------------------
    # 3. Apply mask and element‑wise gates with a Triton kernel
    # ------------------------------------------------------------------
    if mask is None:
        mask_f16 = torch.ones(bs, N, N, dtype=torch.float16, device=device)
    else:
        mask_f16 = mask.to(torch.float16)
    # make everything contiguous 1‑D compatible
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f16 = mask_f16.contiguous()

    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_el = left_proj.numel()
    mask_el = mask_f16.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_el, BLOCK),)

    gated_mask_kernel[grid](
        left_proj, right_proj,
        mask_f16,
        left_gate, right_gate,
        left, right,
        total_el, hidden, mask_el,
        BLOCK=BLOCK
    )

    # ------------------------------------------------------------------
    # 4. Core triangle multiplication: batched GEMM via cuBLAS
    #    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) and multiply left @ right.T

    right_mat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat)          # fp16 result

    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gate, final linear projection (float32)
    # ------------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * w_out_norm + b_out_norm

    out_norm = out_norm * out_gate.to(out_norm.dtype)

    w_to_out = weights["to_out.weight"].to(out_norm.dtype)
    output = F.linear(out_norm, w_to_out)   # (B,N,N,dim)

    return output

"""
TriMul (outgoing) forward pass with Triton acceleration.

The computational bottleneck – an N×N×N tensor contraction – is expressed as a
batched matrix multiplication (left @ rightᵀ) which is executed with cuBLAS via
torch.bmm.  A tiny Triton kernel fuses the mask and the two element‑wise gates
(left‑gate and right‑gate) on the projected tensors, satisfying the requirement
to use Triton while keeping the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gate multiplications.
# ----------------------------------------------------------------------
@triton.jit
def gated_mask_kernel(
    left_ptr, right_ptr,               # projected tensors (fp16)
    mask_ptr,                          # mask (fp16), shape [B, N, N]
    left_gate_ptr, right_gate_ptr,     # gate tensors (fp16)
    left_out_ptr, right_out_ptr,       # results (fp16)
    total, hidden, mask_total,         # scalar arguments
    BLOCK: tl.constexpr,               # thread block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear offsets

    # valid range for the main tensors
    valid = offs < total

    # mask index – broadcast over the hidden dimension
    mask_idx = offs // hidden
    mask_valid = mask_idx < mask_total

    # load data (mask is broadcasted)
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=0.0)

    # apply mask and gates
    left_out = left * mask_val * left_gate
    right_out = right * mask_val * right_gate

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: mapping of weight names to torch.Tensors
            - config_dict: contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b                     # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. FP16 projections + gate computation (torch)
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left_proj = F.linear(x_f16, w_left)         # (B, N, N, hidden)
    right_proj = F.linear(x_f16, w_right)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_f16, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_f16, w_out_gate))

    # --------------------------------------------------------------
    # 3. Apply mask and element‑wise gates via a Triton kernel
    # --------------------------------------------------------------
    if mask is None:
        mask_f16 = torch.ones(bs, N, N, dtype=torch.float16, device=device)
    else:
        mask_f16 = mask.to(torch.float16)

    # make tensors contiguous for 1‑D indexing inside the kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f16 = mask_f16.contiguous()

    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_el = left_proj.numel()
    mask_el = mask_f16.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_el, BLOCK),)

    gated_mask_kernel[grid](
        left_proj, right_proj,
        mask_f16,
        left_gate, right_gate,
        left, right,
        total_el, hidden, mask_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4. Core N³ contraction: batched GEMM via cuBLAS (torch.bmm)
    #    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) where each slice is a matrix
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat)                     # fp16

    # back to (B, N, N, hidden) in float32
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 5. Output LayerNorm, output gate, and final projection (float32)
    # --------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * w_out_norm + b_out_norm

    out_norm = out_norm * out_gate.to(out_norm.dtype)

    w_to_out = weights["to_out.weight"].to(out_norm.dtype)   # (dim, hidden)
    output = F.linear(out_norm, w_to_out)                    # (B, N, N, dim)

    return output
