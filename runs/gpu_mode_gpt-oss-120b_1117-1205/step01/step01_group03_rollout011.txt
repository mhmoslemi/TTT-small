@triton.jit
def mask_mul_kernel(
    left_ptr, right_ptr, mask_ptr, left_out_ptr, right_out_ptr,
    size,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0))
    m = tl.load(mask_ptr + offs, mask=mask, other=tl.float16(1))
    # mask could be 0 or 1; broadcast already applied to shape; we can multiply:
    l = l * m
    r = r * m
    tl.store(left_out_ptr + offs, l, mask=mask)
    tl.store(right_out_ptr + offs, r, mask=mask)

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_mul_kernel(left_ptr, right_ptr, mask_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0))
    m = tl.load(mask_ptr + offs, mask=mask, other=tl.float16(0))
    tl.store(left_ptr + offs, l * m, mask=mask)
    tl.store(right_ptr + offs, r * m, mask=mask)

def custom_kernel(data):
    """
    TriMul outgoing forward implementation.
    Performs LayerNorm, linear projections, gating, mask application using a
    Triton kernel, then batched matrix multiplication with torch.bmm, final
    LayerNorm, gating and projection.
    Returns a float32 tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # 1. LayerNorm on input
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b   # shape [bs, N, N, dim]

    # Cast to fp16 for projection & gate ops
    x_f16 = x.to(torch.float16)

    # 2. Linear projections
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # 3. Gates
    left_gate = torch.sigmoid(F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # 4. Mask handling (if mask is None treat as all 1s)
    if mask is None:
        mask_f16 = torch.ones(bs, N, N, dtype=torch.float16, device=device)
    else:

    # Broadcast mask to hidden dimension
    mask_exp = mask_f16.unsqueeze(-1).expand_as(left).contiguous()

    # Ensure tensors are contiguous
    left = left.contiguous()
    right = right.contiguous()
    mask_exp = mask_exp.contiguous()

    # Triton kernel: elementwise mask multiplication (in-place)
    total_elems = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)
    mask_mul_kernel[grid](left, right, mask_exp, total_elems, BLOCK=BLOCK)

    # 5. Core triangle multiplication via batched GEMM (cuBLAS)
    # Reshape to (bs*hidden, N, N) for left, and (bs*hidden, N, N) for right transposed.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden, N, N)  # (k,j) order

    out_mat = torch.bmm(left_mat, right_mat)  # [bs*hidden, N, N] FP16

    # Reshape back to [bs, N, N, hidden]
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # 6. Output LayerNorm (float32)
    out_f32 = out.to(torch.float32)
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    mean_out = out_f32.mean(dim=-1, keepdim=True)

    out_norm = (out_f32 - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * w_to_out_norm + b_to_out_norm

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_weight = weights["to_out.weight"].to(torch.float32)
    output = F.linear(out_norm, to_out_weight)  # shape [bs, N, N, dim]

    return output

    mask_mul_kernel[grid](
        left, right, mask_expanded, total_elems, BLOCK=BLOCK
    )  # modifies left/right in place

    # ------------------------------------------------------------------
    # 4. Core triangle multiplication: batched GEMM (cuBLAS)
    #    out[b, d] = left[b, d] @ right[b, d].T
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) where each slice is a dense matrix
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # (i,k)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)  # (k,j)
    out_mat = torch.bmm(left_mat, right_mat)                                 # (i,j)

    # reshape back to [B, N, N, hidden] (still FP16)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_out = out_f32.mean(dim=-1, keepdim=True)
    var_out = ((out_f32 - mean_out) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out_f32 - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * w_out_norm + b_out_norm

    # ------------------------------------------------------------------
    # 6. Apply output gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)

    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden)
    output = F.linear(out_norm, w_to_out)                  # [B, N, N, dim] float32

    return output
```
