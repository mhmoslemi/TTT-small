import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_trimul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_i, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    M, N, K,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # decode batch and head
    batch_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch index and head index
    batch = batch_head // hidden_dim
    head = batch_head % hidden_dim

    # compute base pointers for this (batch, head)
    a_base = A_ptr + batch * stride_a_batch + head * stride_a_h
    b_base = B_ptr + batch * stride_b_batch + head * stride_b_h
    c_base = C_ptr + batch * stride_c_batch + head * stride_c_h

    # compute tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile (K x N) (row = k, col = j)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

    # store result
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    gamma = weights["norm.weight"]
    beta = weights["norm.bias"]
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta

    # Convert to fp16
    x_h = x_norm.to(torch.float16)

    # Linear projections
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_h, w_left)   # shape (B,N,N,hidden)
    right = F.linear(x_h, w_right)

    # Apply mask if provided
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Allocate output tensor
    out = torch.empty_like(left)  # fp16

    B, N, _, H = left.shape

    # Strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h = out.stride()

    # Prepare kernel launch parameters
    # Choose block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_trimul_kernel[grid](
        left,
        right,
        out,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_k,  # note: row stride for B is k dimension
        stride_b_i,  # col stride for B is i dimension
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_k,  # note: col stride for output is k dimension (=H?) Wait output col stride should be stride_c_k (original third dimension)
        stride_c_h,
        N, N, N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # out now (B,N,N,H) fp16
    out_f32 = out.to(torch.float32)

    # Output LayerNorm over hidden_dim
    gamma_out = weights["to_out_norm.weight"]
    beta_out = weights["to_out_norm.bias"]
    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # Apply out gate (broadcast)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection
    w_to_out = weights["to_out.weight"]
    output = F.linear(out_norm, w_to_out)

    return output

"""
TriMul (outgoing) forward pass optimized with Triton.

The heavy part is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication (left @ rightᵀ) for every
hidden dimension `d`.  The kernel computes this GEMM in FP16 (accumulated
in FP32) while all other operations (LayerNorm, linear projections,
gates) remain in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_batch_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_i, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    M, N, K,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:
        C = A @ Bᵀ
    A, B, C have shape (batch, M, K), (batch, K, N), (batch, M, N) respectively.
    The logical batch size is `batch = B * hidden_dim` where `B` is the
    original batch dimension.  Inside the kernel we recover the original
    batch index and the hidden head index to compute proper pointers.
    """
    batch_head = tl.program_id(0)      # combined (batch * hidden_dim)
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # decode original batch and hidden head
    batch = batch_head // hidden_dim
    head  = batch_head % hidden_dim

    # base pointers for this (batch, head)
    a_base = A_ptr + batch * stride_a_batch + head * stride_a_h
    b_base = B_ptr + batch * stride_b_batch + head * stride_b_h
    c_base = C_ptr + batch * stride_c_batch + head * stride_c_h

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # B tile (K x N) – note that rows of B are the original K dimension
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)   # result shape (BLOCK_M, BLOCK_N)

    # write result
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    TriMul forward (outgoing) implementation.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32/bool) or None
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on input ----------
    gamma = weights["norm.weight"].to(torch.float32)
    beta = weights["norm.bias"].to(torch.float32)
    x = input_tensor  # [B, N, N, dim]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta               # [B, N, N, dim]

    # ---------- Cast to half for speed ----------
    x_h = x_norm.to(torch.float16)

    # ---------- Linear projections ----------
    w_left = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)       # [B, N, N, hidden]
    right = F.linear(x_h, w_right)

    # ---------- Optional mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()       # used later

    left = left * left_gate
    right = right * right_gate

    # ---------- Batched GEMM (left @ rightᵀ) ----------
    B, N, _, H = left.shape   # H == hidden_dim
    out = torch.empty_like(left)   # fp16 container for result

    # Strides (batch, i, k, h) for the original tensors
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h = out.stride()

    # Tile sizes – tuned for NVIDIA B200 (SM 80+)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                              # combined batch*head dimension
        triton.cdiv(N, BLOCK_M),            # rows
        triton.cdiv(N, BLOCK_N),            # cols
    )
    _trimul_batch_gemm_kernel[grid](
        left, right, out,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_k, stride_b_i, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_k, stride_c_h,
        N, N, N,                           # M, N, K
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    out_f32 = out.to(torch.float32)   # [B, N, N, hidden]

    # ---------- Output LayerNorm ----------
    gamma_out = weights["to_out_norm.weight"].to(torch.float32)
    beta_out = weights["to_out_norm.bias"].to(torch.float32)
    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # ---------- Apply output gate ----------
    out_norm = out_norm * out_gate.to(torch.float32)

    # ---------- Final linear projection ----------
    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden)
    output = F.linear(out_norm, w_to_out)                  # [B, N, N, dim]

    return output
