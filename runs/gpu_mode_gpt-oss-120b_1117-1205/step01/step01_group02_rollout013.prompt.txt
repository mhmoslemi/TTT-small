```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

## Triton launch convention

Every failure mode below has appeared in this search. Read this before writing.

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
    grid = (triton.cdiv(n, BLOCK),)               # host-side, use triton.cdiv
    my_kernel[grid](x, y, n, BLOCK=BLOCK)         # note the [grid] subscript
```

- `my_kernel(...)` without `[grid]` raises "Cannot call @triton.jit'd outside
  the scope of a kernel".
- `BLOCK`, `BLOCK_M`, `BLOCK_N` are not defined on the host unless you define
  them there. Referencing one you only declared as a `tl.constexpr` parameter
  raises NameError.
- `tl.cdiv` is device-side. On the host use `triton.cdiv`.
- Define the kernel at module top level, never nested inside `custom_kernel`.

You do not have to move everything into Triton. A correct submission that keeps
most of the computation in PyTorch and moves one hot region into a kernel scores;
one that fails to launch scores zero.

"""
TriMul (outgoing) forward pass implemented with a fused Triton GEMM kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Compute left/right projections and their sigmoid gates using the provided
   linear weights (all in fp16 for speed).
3. Apply the optional mask and the element‑wise gates.
4. For every batch b and hidden dimension h we need
        out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   which is a batched matrix multiplication of shape (N,N) with its transpose.
   The heavy N³ work is performed by a custom Triton kernel that computes
   left @ rightᵀ for all (b·h) batches in one launch.
5. The result (still fp16) is reshaped, cast to fp32 and passed through the
   output LayerNorm, the output gate and the final linear projection.

The kernel computes A (M×K) × B (K×N) → C (M×N) for many independent
batches (total_batch = batch_size * hidden_dim).  Accumulation is done in
float32 for numerical stability; the final matrix is stored as fp16.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) where
#   A : [total_batch, M, K]   (fp16)
#   B : [total_batch, K, N]   (fp16)   (already transposed, i.e. B = rightᵀ)
#   C : [total_batch, M, N]   (fp16)
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # tensor pointers
    M, N, K,                           # matrix dimensions (int)
    stride_am, stride_ak,              # A strides (row, col)
    stride_bk, stride_bn,              # B strides (row, col)
    stride_cm, stride_cn,              # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)               # block row index
    pid_n = tl.program_id(1)               # block col index
    pid_b = tl.program_id(2)               # batch (b·h) index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑bounds masks for rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulation buffer in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)

        # mask for the current K‑tile
        mask_k = (k_offset + offs_k) < K

        # pointers to the current block of A and B (including batch offset)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # load tiles (apply masks)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # write the result back in FP16
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

# Entry point required by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask: torch.Tensor [B, N, N] (bool or float) or None
            - weights: dict of torch.Tensor containing all linear / ln parameters
            - config: dict containing at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    import math

    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the last dimension
    # --------------------------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to fp16 for all subsequent heavy ops
    x_norm_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias) and gating
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(
        x_norm_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = torch.nn.functional.linear(
        x_norm_fp16, weights["right_proj.weight"].to(torch.float16)
    )

    left_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # --------------------------------------------------------------------- #
    # 3. Apply optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Prepare tensors for the fused GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    # Permute to [B, H, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right so that we can do A @ B (no extra transpose)
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dimensions for the kernel
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_T.view(total_batch, N, N)
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # --------------------------------------------------------------------- #
    # 6. Launch the Triton kernel
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, H] and continue in FP32
    # --------------------------------------------------------------------- #
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # --------------------------------------------------------------------- #
    # 8. Output LayerNorm, output gate and final linear projection
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)  # apply output gate

    # Final linear projection back to `dim`
    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # shape [B, N, N, dim]

    return output
