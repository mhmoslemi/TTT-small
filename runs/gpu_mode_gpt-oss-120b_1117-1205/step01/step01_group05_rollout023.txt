@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute tile start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute c tile pointer
    c_ptrs = (
        C_ptr
        + batch * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile
        a_ptrs = (
            A_ptr
            + batch * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # a cast to fp32
        a = a.to(tl.float32)

        # B tile
        b_ptrs = (
            B_ptr
            + batch * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # write to C in fp16
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_batch, stride_i, stride_k,
    hidden_dim,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch, hidden
    b = pid // hidden_dim
    h = pid - b * hidden_dim

    # offset within the flattened tensor
    offset = b * stride_batch + h

    # tile offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

@triton.jit
def _tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,
    stride_batch, stride_i, stride_k, stride_h,
    hidden_dim,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden indices
    b = pid // hidden_dim
    h = pid - b * hidden_dim

    # base offset for current (b,h)
    offset = b * stride_batch + h * stride_h

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (i,k)
        a_ptrs = left_ptr + offset + offs_m[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (k,j) - note we need right transposed
        b_ptrs = right_ptr + offset + offs_k[:, None] * stride_k + offs_n[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc = acc + tl.dot(a, b)

    # Store result
    c_ptrs = out_ptr + offset + offs_m[:, None] * stride_i + offs_n[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    gamma = weights["norm.weight"].to(torch.float32)
    beta = weights["norm.bias"].to(torch.float32)
    x = input_tensor.to(torch.float32)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta

    # Move to half
    x_h = x_norm.to(torch.float16)

    # Linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))

    # Apply gates and mask
    left = left * left_gate
    right = right * right_gate

    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Contraction via Triton
    B, N, _, H = left.shape  # H == hidden_dim
    # Output tensor
    out = torch.empty_like(left)

    # Strides (in elements)
    stride_batch = left.stride(0)
    stride_i = left.stride(1)
    stride_k = left.stride(2)
    stride_h = left.stride(3)  # 1

    # Kernel tiling sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_kernel[grid](
        out,
        left,
        right,
        stride_batch,
        stride_i,
        stride_k,
        stride_h,
        H,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    # Continue with rest of pipeline
    out_f32 = out.to(torch.float32)

    # Output LayerNorm
    gamma_out = weights["to_out_norm.weight"].to(torch.float32)
    beta_out = weights["to_out_norm.bias"].to(torch.float32)
    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # Apply out_gate (converted to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection
    w_to_out = weights["to_out.weight"].to(torch.float32)
    output = F.linear(out_norm, w_to_out)

    return output

"""
TriMul (outgoing) forward implementation.

The heavy tensor contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batched matrix multiplication for each (b,h) pair:
    out_{b,h} = left_{b,h} @ right_{b,h}.T
where left_{b,h} and right_{b,h} are N×N matrices.
We implement this contraction with a custom Triton kernel that works
directly on the original [B, N, N, H] layout (no extra copies or
transposes) and then finish the remaining operations in PyTorch.
All other operations (LayerNorms, Linear projections, gates) are
performed with PyTorch’s highly‑optimized kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,
    stride_batch, stride_i, stride_k, stride_h,
    hidden_dim,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction.

    Arguments
    ---------
    out_ptr, left_ptr, right_ptr : pointers to tensors of shape
        [B, N, N, H] (row‑major, contiguous).
    stride_batch, stride_i, stride_k, stride_h : element strides of the
        input tensors (as returned by .stride()).
    hidden_dim   : size of the hidden dimension H.
    M, N, K      : all equal to seq_len (N).  M = rows (i), N = cols (j),
                   K = reduction dimension (k).
    BLOCK_M/N/K  : compile‑time tile sizes.
    """
    pid = tl.program_id(0)           # combined batch * hidden index
    pid_m = tl.program_id(1)         # tile row
    pid_n = tl.program_id(2)         # tile column

    # decode batch and hidden indices from pid
    b = pid // hidden_dim
    h = pid - b * hidden_dim          # same as pid % hidden_dim

    # base offset for the current (b, h) slice
    offset = b * stride_batch + h * stride_h

    # tile offsets within the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp16 (sufficient for hidden_dim=128)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # iterate over the reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: left[b,i,k,h]  (rows i, cols k)
        a_ptrs = left_ptr + offset + offs_m[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile: right[b,j,k,h] transposed => rows k, cols j
        b_ptrs = right_ptr + offset + offs_k[:, None] * stride_k + offs_n[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc = acc + tl.dot(a, b)      # accumulate in fp16

    # write the result tile to out[b,i,j,h]
    c_ptrs = out_ptr + offset + offs_m[:, None] * stride_i + offs_n[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    """
    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32)
            - mask  : torch.Tensor [B, N, N]       (float32, bool, or None)
            - weights: dict of torch.Tensors for all linear / norm parameters
            - config : dict with keys "dim", "hidden_dim", "nomask" (optional)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Input LayerNorm (over last dim)
    # ------------------------------------------------------------------
    gamma = weights["norm.weight"].to(torch.float32)
    beta = weights["norm.bias"].to(torch.float32)

    x = input_tensor.to(torch.float32)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta          # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # Projections and gates (half precision for speed)
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # Linear weights (fp16)
    w_left = weights["left_proj.weight"].to(torch.float16)      # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Projections
    left = F.linear(x_h, w_left)          # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Apply mask if present and not disabled
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Triton kernel: out = Σ_k left[...,k] * right[...,k]ᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    out = torch.empty_like(left)  # allocate output in fp16

    # Strides (in elements)
    stride_batch = left.stride(0)   # B stride = N*N*H
    stride_i = left.stride(1)       # stride over i dimension = N*H
    stride_k = left.stride(2)       # stride over k dimension = H
    stride_h = left.stride(3)       # = 1

    # Tiling parameters – chosen to fit B200 SM resources
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * H,                          # one program for each (batch, hidden) pair
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # columns
    )
    _tri_mul_kernel[grid](
        out,
        left,
        right,
        stride_batch,
        stride_i,
        stride_k,
        stride_h,
        H,               # hidden_dim as a runtime argument
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)
    gamma_out = weights["to_out_norm.weight"].to(torch.float32)
    beta_out = weights["to_out_norm.bias"].to(torch.float32)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # Apply the output gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # Final linear projection back to input dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float32)
    output = F.linear(out_norm, w_to_out)   # shape [B, N, N, dim]

    return output
```
