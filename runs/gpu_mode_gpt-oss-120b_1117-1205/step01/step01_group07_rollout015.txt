@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(2)  # batch*hidden
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute batch and head index
    b = pid_bh // H
    h = pid_bh % H

    # block start offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # offsets for row and column indices, mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers offsets base for this b,h
    base_a = left_ptr + b * stride_b + h * stride_h
    base_b = right_ptr + b * stride_b + h * stride_h
    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile
        a_ptrs = base_a + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        b_ptrs = base_b + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i

        # mask for loads
        mask_a = mask_i[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_j[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # store out
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_i
    mask_c = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # get shapes
    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # load weight tensors: each weight shape consistent
    # Use F.layer_norm for the first LN.
    eps=1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections: left and right (float16)
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # Apply mask if present
    if not nomask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask
        right = right * mask

    # Gates compute in fp32
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # stays fp32

    right = right * right_gate

    out = torch.empty_like(left, dtype=torch.float16)  # will store [B,N,N,hidden_dim]

    # same strides for right and out
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch_head = tl.program_id(2)

    # compute batch and head indices
    batch = pid_batch_head // H
    head = pid_batch_head % H

    # compute block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for this batch and head
    base_a = left_ptr + batch * stride_b + head * stride_h
    base_b = right_ptr + batch * stride_b + head * stride_h
    base_c = out_ptr + batch * stride_out_b + head * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for A and B tiles
        a_ptrs = base_a + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        b_ptrs = base_b + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i

        # load with mask
        mask_a = mask_i[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_j[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # write result
    c_ptrs = base_c + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_k
    mask_c = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)
```

"""
Highly‑optimized Triton implementation of the outgoing TriMul operator
(AlphaFold3). The kernel computes the core tensor contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for every batch and hidden head without any extra copies. LayerNorms,
linear projections, and gating are performed in PyTorch; the expensive
N³ contraction is offloaded to a custom Triton GEMM that works directly
on the original (B, N, N, H) layout.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,                         # pointers
    B, N, H,                                              # B: batch, N: seq len, H: hidden dim
    stride_b, stride_i, stride_k, stride_h,               # strides of left/right (same layout)
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 2‑D tile indices + combined (batch, head) index
    pid_i = tl.program_id(0)          # tile row   → i dimension
    pid_j = tl.program_id(1)          # tile col   → j dimension
    pid_bh = tl.program_id(2)         # batch * hidden index

    # -> batch and head indices
    batch = pid_bh // H
    head = pid_bh % H

    # ------------------------------------------------------------------
    # Offsets for the current tile (with bounds masks)
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch, head) pair
    # ------------------------------------------------------------------
    base_a = left_ptr + batch * stride_b + head * stride_h
    base_b = right_ptr + batch * stride_b + head * stride_h
    base_c = out_ptr + batch * stride_out_b + head * stride_out_h

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main reduction over the K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: (i, k)
        a_ptrs = base_a + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        # B tile: (k, j) – note the transposed access pattern
        b_ptrs = base_b + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i

        mask_a = mask_i[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_j[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulation

    # ------------------------------------------------------------------
    # Write back the result tile (fp32 → fp16)
    # ------------------------------------------------------------------
    c_ptrs = base_c + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_k
    mask_c = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Triton entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Config & weight extraction
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Linear projections to hidden dimension (float16 for bandwidth)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight).to(torch.float16)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight).to(torch.float16)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (computed in float32 then cast)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # keep in float32

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Ensure contiguity (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # Strides in element units (not bytes)
    stride_b, stride_i, stride_k, stride_h = left.stride()
    # Output tensor uses the exact same layout
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out.stride()

    # Block sizes – 64 is a good trade‑off for most N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # tiles over i
        triton.cdiv(N, BLOCK_N),          # tiles over j
        B * H,                            # batch * hidden
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        stride_out_b,
        stride_out_i,
        stride_out_k,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 for the remaining layers
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # Final LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply output gate and final projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
