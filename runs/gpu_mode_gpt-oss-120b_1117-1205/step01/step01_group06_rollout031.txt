def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # 1. Layer norm on dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # 2. Linear projections and gates
    B,N,_,_ = x.shape  # dim as last dimension
    # Flatten
    x_flat = x.view(-1, dim)
    # Projections
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B,N,N,hidden_dim)
    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()

    # Apply mask if not nomask
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to bf16
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # Allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides
    stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_h = left_bf16.stride()
    stride_rhs_b, stride_rhs_j, stride_rhs_k, stride_rhs_h = right_bf16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    # Launch kernel
    tri_mul_kernel[grid](
        left_bf16,
        right_bf16,
        out,
        B,
        N,
        hidden_dim,
        stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_h,
        stride_rhs_b, stride_rhs_j, stride_rhs_k, stride_rhs_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Now out is [B,N,N,H]; apply hidden layernorm (to_out_norm) using Torch's layer_norm (fast)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    out_flat = out.view(-1, hidden_dim)
    final = torch.nn.functional.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps=1e-5

    # layernorm input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    B, N, _, _ = x.shape

    # linear projections
    x_flat = x.view(-1, dim)
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # mask
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to BF16 for kernel
    left_bf16 = left.to(torch.bfloat16).contiguous()
    right_bf16 = right.to(torch.bfloat16).contiguous()

    # Output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides
    lsb, lsi, lsk, lsh = left_bf16.stride()
    rsb, rsj, rsk, rsh = right_bf16.stride()
    csb, csi, csj, csh = out.stride()

    # block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_bf16,

"""
TriMul outgoing implementation (forward only).

Algorithm
---------
1. LayerNorm on the input channel dimension.
2. Linear projections (left/right) and three gate projections (left_gate, right_gate,
   out_gate) using the supplied weight matrices.
3. Apply the optional pairwise mask and element‑wise gates:
        left  = left  * mask  * left_gate
        right = right * mask  * right_gate
4. Contraction over the sequence dimension k:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for every hidden slice h.
   A custom Triton kernel computes the whole N×N×H contraction in a single
   launch (grid over i‑tiles, j‑tiles and the combined batch‑hidden index).
   The kernel works on bfloat16 inputs and accumulates in float32.
5. LayerNorm over the hidden dimension (torch implementation, fast enough).
6. Multiply by out_gate and apply the final linear projection back to `dim`.

Only the O(N³·H) contraction is performed in Triton; all other
operations use PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    lhs_ptr,                # *bfloat16  [B,N,N,H]
    rhs_ptr,                # *bfloat16  [B,N,N,H] (to be accessed as B[j,k,h])
    out_ptr,                # *float32   [B,N,N,H]
    B,                      # int32
    N,                      # int32  sequence length
    H,                      # int32  hidden dimension
    stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_h,
    stride_rhs_b, stride_rhs_j, stride_rhs_k, stride_rhs_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in k dimension
):
    pid_m = tl.program_id(0)                # i‑tile
    pid_n = tl.program_id(1)                # j‑tile
    pid_z = tl.program_id(2)                # combined batch & hidden index

    # decode batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # start offsets of the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h) slice
    lhs_base = lhs_ptr + b * stride_lhs_b + h * stride_lhs_h
    rhs_base = rhs_ptr + b * stride_rhs_b + h * stride_rhs_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # A tile: [BLOCK_M, BLOCK_K] -> left[b,i,k,h]
        a_ptrs = lhs_base + offs_i[:, None] * stride_lhs_i + cur_k[None, :] * stride_lhs_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # B tile (transposed): [BLOCK_K, BLOCK_N] -> right[b,j,k,h] as (k, j)
        b_ptrs = rhs_base + cur_k[:, None] * stride_rhs_k + offs_j[None, :] * stride_rhs_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # write the result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be unused)
            - weights: dict of model weight tensors
            - config: dict containing "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    B, N, _, _ = x.shape

    # 2. Linear projections & gates (float32)
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # 3. Apply mask (if present) and element‑wise gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)    # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # 4. Contraction over k using Triton
    # Cast to bfloat16 to reduce bandwidth
    left_bf16 = left.to(torch.bfloat16).contiguous()
    right_bf16 = right.to(torch.bfloat16).contiguous()

    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements)
    lsb, lsi, lsk, lsh = left_bf16.stride()
    rsb, rsj, rsk, rsh = right_bf16.stride()
    csb, csi, csj, csh = out.stride()

    # Tile sizes – chosen to stay within shared‑memory limits
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_bf16,
        right_bf16,
        out,
        B,
        N,
        hidden_dim,
        lsb, lsi, lsk, lsh,
        rsb, rsj, rsk, rsh,
        csb, csi, csj, csh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 5. Hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 6. Apply outgoing gate
    out = out * out_gate

    # 7. Final linear projection back to `dim`
    out_flat = out.view(-1, hidden_dim)      # [B*N*N, hidden_dim]
    final = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return final
