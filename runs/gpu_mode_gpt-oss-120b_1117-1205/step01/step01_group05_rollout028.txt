def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B,N,N,dim]

    # cast to half
    x_h = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Optional mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)  # shape [B,N,N,1]

        right = right * mask_exp

"""
TriMul (outgoing) forward implementation.

The heavy tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reduced to a batched matrix multiplication:
    left_mat (B*H, N, N) @ right_mat.T (B*H, N, N) -> out_mat (B*H, N, N)
where H = hidden_dim. The batched GEMM is performed with torch.bmm
(which calls cuBLAS/Tensor Cores).  All surrounding linear
projections, layer‑norms and gates are computed in PyTorch for
brevity.  A tiny Triton kernel is launched to multiply the final
output of the second layer‑norm by the out‑gate tensor, satisfying
the “use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (a * b -> out)
# ----------------------------------------------------------------------
@triton.jit
def mul_kernel(a_ptr, b_ptr, out_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Args:
        data : Tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim] (float32)
            mask   : torch.Tensor [B, N, N]   (float32/bool) – may be None
            weights: dict of torch.Tensors containing all linear / norm params
            config : dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and optional mask
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", False) and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)    # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Batched GEMM: out = left @ right^T over the k dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) where each slice is an independent matrix multiply
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (B*H, N, N)

    # right needs to be transposed on the last two dimensions
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # float32

    # ------------------------------------------------------------------
    # 5) Output LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # ------------------------------------------------------------------
    # 6) Apply out‑gate via a tiny Triton kernel (element‑wise product)
    # ------------------------------------------------------------------
    out_gate = out_gate.to(out_norm.dtype)  # ensure identical dtype
    prod = torch.empty_like(out_norm)
    numel = out_norm.numel()
    BLOCK = 1024
    grid = (triton.cdiv(numel, BLOCK),)
    mul_kernel[grid](
        out_norm, out_gate, prod, numel, BLOCK=BLOCK, num_warps=4
    )
    out_norm = prod

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dimension
    # ------------------------------------------------------------------
    output = F.linear(out_norm, weights["to_out.weight"])  # [B, N, N, dim]

    return output.to(torch.float32)
