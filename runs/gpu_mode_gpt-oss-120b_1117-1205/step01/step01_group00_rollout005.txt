"""
Implementation of the outgoing TriMul operator using Triton for the core
contraction. The kernel computes for each batch and hidden channel a matrix
multiplication:
    out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}
which is equivalent to left @ right^T.
All other operations (LayerNorm, linear projections, gating, mask handling)
are performed in PyTorch (mixed precision). The contraction is performed in
half‑precision with float32 accumulation for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,  # pointers
    M, N, K,               # matrix dims
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B  where:
        A: (batch, M, K)  (fp16)
        B: (batch, K, N)  (fp16)
        C: (batch, M, N)  (fp16)
    Each program processes one block of size BLOCK_M x BLOCK_N for a single
    batch index.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute k offsets
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base \
            + offs_m[:, None] * stride_a_m \
            + offs_k[None, :] * stride_a_k
        a_mask = (mask_m[:, None] &
                  (offs_k[None, :] < K))
        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)

        # Load B tile (K x N)
        b_ptrs = b_base \
            + offs_k[:, None] * stride_b_k \
            + offs_n[None, :] * stride_b_n
        b_mask = ((offs_k[:, None] < K) &
                  mask_n[None, :])
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_base \
        + offs_m[:, None] * stride_c_m \
        + offs_n[None, :] * stride_c_n
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast to half for the expensive ops
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: (B, N, N) -> (B, N, N, 1)
        mask_fp = mask.to(dtype=torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gates (half‑precision)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden_dim)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()    # (B,N,N,hidden_dim)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Bring hidden dimension to batch axis for the batched matmul.

    out_flat = triton_matmul(left_flat, right_flat)   # (B*H, N, N), fp16

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Final projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

"""
Outgoing TriMul forward pass with a Triton‑accelerated core contraction.

The heavy N³ operation
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
is realized as a batched matrix multiplication
    out_{b,:, :, d} = left_{b,:, :, d} @ right_{b,:, :, d}ᵀ
for each batch b and hidden channel d.  All other steps
(LayerNorm, linear projections, gating, mask handling) are performed
in PyTorch (mixed precision).  The contraction runs in half‑precision
with float‑32 accumulation for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                 # tensor pointers
    M, N, K,                             # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B for a batch of matrices.
    A: (batch, M, K)   FP16
    B: (batch, K, N)   FP16
    C: (batch, M, N)   FP16 (stored)
    Each program processes one BLOCK_M × BLOCK_N tile for a single batch.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M × K)
        a_ptrs = a_base \
            + offs_m[:, None] * stride_a_m \
            + offs_k[None, :] * stride_a_k
        a_mask = (mask_m[:, None] &
                  (offs_k[None, :] < K))
        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)

        # Load B tile (K × N)
        b_ptrs = b_base \
            + offs_k[:, None] * stride_b_k \
            + offs_n[None, :] * stride_b_n
        b_mask = ((offs_k[:, None] < K) &
                  mask_n[None, :])
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Store result back to C (cast to FP16)
    c_ptrs = c_base \
        + offs_m[:, None] * stride_c_m \
        + offs_n[None, :] * stride_c_n
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.

    Args:
        A: (batch, M, K)  FP16 contiguous tensor
        B: (batch, K, N)  FP16 contiguous tensor

    Returns:
        C: (batch, M, N) FP16 tensor
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – chosen for good occupancy on B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict mapping parameter names to torch.Tensors
        - config_dict  : dict containing at least 'dim', 'hidden_dim',
                         optional 'nomask' (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half for the heavy path
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp = mask.to(dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = rightᵀ

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = triton_matmul(left_flat, right_flat)    # (B*H, N, N), fp16

    # reshape back to (B, N, N, H) and promote to float32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
