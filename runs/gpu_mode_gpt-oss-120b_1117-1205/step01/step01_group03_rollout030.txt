TriMul (outgoing) forward pass optimized with a Triton batched GEMM kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32) using torch.nn.functional.layer_norm.

      left  = left  * left_gate
      right = right * right_gate
4. Core O(N³·H) update:
   For every hidden channel d compute
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication `left @ rightᵀ`.
   The multiplication is performed by a custom Triton kernel that
   processes a batch×hidden slice as a dense (N×N) matrix.
5. Layer‑norm the result over the hidden dimension,
   multiply with the output gate, and project back to `dim`.
6. Return the final tensor in float32: shape [B, N, N, dim].

The kernel uses tile sizes (BLOCK_M, BLOCK_N, BLOCK_K) chosen
based on the sequence length to keep blocks well‑filled while
leveraging Tensor‑cores (float16 compute, float32 accumulation).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: M×K, Bᵀ: K×N → C: M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes
    stride_am, stride_ak,              # A strides (row, col)
    stride_bk, stride_bn,              # Bᵀ strides (row, col)
    stride_cm, stride_cn,              # C strides (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to tiles of A and Bᵀ
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)                     # [B,N,N,dim] float32

    # ------------------------------------------------------------------
    # 2. Projections & gates (float16 for speed)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # Linear projections (no bias)
    w_left = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, w_left)      # [B,N,N,hidden]
    right = F.linear(x_f16, w_right)

    # Optional mask
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f16
        right = right * mask_f16

    # Gates (sigmoid, no bias)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_f16, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_f16, w_out_gate))

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core TriMul product via Triton (batched GEMM)
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as left @ rightᵀ for each hidden slice.
    # ------------------------------------------------------------------
    # Prepare tensors: (B, hidden, N, N) – contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = transpose of last two axes

    total_batches = bs * hidden
    left_mat = left.reshape(total_batches, N, N)    # (B*H, N, N)  FP16
    right_mat = right.reshape(total_batches, N, N)  # (B*H, N, N)  FP16

    out_mat = torch.empty_like(left_mat)            # FP16 output

    # Heuristic tile sizes
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_mat,                     # a_ptr
        right_mat,                    # b_ptr (already transposed)
        out_mat,                      # c_ptr
        N, N, N,                      # M, N, K
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2), # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),     # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),   # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_mat.reshape(bs, hidden, N, N).permute(0, 2, 3, 1)   # FP16

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gate, final projection (float16 → float32)
    # ------------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"].to(torch.float16)   # (hidden,)
    b_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden,),
                            weight=w_out_norm,
                            bias=b_out_norm,
                            eps=eps)                     # FP16

    out_norm = out_norm * out_gate                         # apply output gate (FP16)

    w_to_out = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    output_fp16 = F.linear(out_norm, w_to_out)              # [B,N,N,dim] FP16
    output = output_fp16.to(torch.float32)

    return output
```
