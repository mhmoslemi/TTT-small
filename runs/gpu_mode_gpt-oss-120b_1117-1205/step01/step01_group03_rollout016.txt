```
@triton.jit
def batched_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-bound offsets
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Compute base pointers
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # cast to higher precision if needed?
        # a is fp16, b is fp16. Use tl.dot that returns fp32 accumulation.
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) * (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # store fp16 back
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
custom_kernel: Triton-accelerated implementation of the "outgoing" TriMul
operator (AlphaFold3). The kernel performs the core cubic contraction
out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d] which is a
batched matrix multiplication (left @ rightᵀ) for each hidden channel d.
All other linear, LayerNorm and gating operations are done in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (M x K) @ (K x N) → (M x N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # problem sizes
    stride_am, stride_ak,               # strides for A (M, K)
    stride_bk, stride_bn,               # strides for B (K, N)
    stride_cm, stride_cn,               # strides for C (M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to the tiles of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulation buffer in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result (cast back to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[:, None])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract shapes / config
    # ------------------------------------------------------------------
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                              # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (float16 for speed)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --- projections ---
    w_left = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, w_left)     # [B, N, N, hidden]
    right = F.linear(x_f16, w_right)

    # --- mask handling (optional) ---
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)    # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # --- gating ---
    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)
    w_outgate = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, w_lgate))
    right_gate = torch.sigmoid(F.linear(x_f16, w_rgate))
    out_gate = torch.sigmoid(F.linear(x_f16, w_outgate))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core TriMul via Triton: batched GEMM for each hidden channel
    #    out = left @ rightᵀ   (shape [B, hidden, N, N])
    # ------------------------------------------------------------------
    # Rearrange to (batch, hidden, M, K) and (batch, hidden, K, N)
    # left: (i, k)   -> rows=i, cols=k
    left = left.permute(0, 3, 1, 2).contiguous()          # [B, hidden, N, N]
    # right needs (k, j) ordering
    right = right.permute(0, 3, 2, 1).contiguous()       # [B, hidden, N, N] (k as row)

    total_batches = bs * hidden_dim                         # fused batch*hidden
    left_mat = left.view(total_batches, N, N)               # [total_batches, N, N]
    right_mat = right.view(total_batches, N, N)             # [total_batches, N, N]

    out_mat = torch.empty_like(left_mat)                     # fp16 output buffer

    # Heuristic tile sizes
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_gemm_kernel[grid](
        left_mat,                       # a_ptr
        right_mat,                      # b_ptr
        out_mat,                        # c_ptr
        N, N, N,                        # M, N, K
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2), # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),     # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to [B, N, N, hidden] and convert to fp32 for the next layers
    out = (out_mat.view(bs, hidden_dim, N, N)
                 .permute(0, 2, 3, 1)
                 .to(torch.float32))

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gating, final projection (float32)
    # ------------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=w_to_out_norm,
                       bias=b_to_out_norm,
                       eps=eps)

    out = out * out_gate.to(torch.float32)

    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden)
    output = F.linear(out, w_to_out)                       # [B, N, N, dim]

    return output
