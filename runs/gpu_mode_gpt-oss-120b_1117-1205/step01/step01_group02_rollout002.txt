"""TriMul (outgoing) forward pass with a fused Triton GEMM.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension.
2. Compute left/right linear projections and their sigmoid gates
   (all in fp16 for speed).  Apply the optional pairwise mask.
3. Fuse the heavy O(N³) update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication left @ rightᵀ.
   A custom Triton kernel computes it for all (batch, head) pairs in
   fp16 with float32 accumulation.
4. Apply the output layer‑norm, the output gate and the final linear
   projection back to the original channel dimension.
The final tensor is returned in float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix sizes
    stride_am, stride_ak,           # A strides (row, col)
    stride_bk, stride_bn,           # B strides (row, col)
    stride_cm, stride_cn,           # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where
       A : (total_batch, M, K)
       B : (total_batch, K, N)
       C : (total_batch, M, N)
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch index (batch * hidden_dim)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # Pointers with batch offset
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Load tiles, applying masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store result (FP16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32, CUDA)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with tensors for all linear / LN parameters
            - config: dict with at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32, CUDA)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1. LayerNorm over the channel dimension
    # ------------------------------------------------------------------ #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Work in half‑precision for the heavy part
    x_norm_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 2. Linear projections + gates (all bias‑free)
    # ------------------------------------------------------------------ #
    # Cast weights once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_norm_fp16, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_norm_fp16, right_proj_w)

    left_gate = torch.nn.functional.linear(x_norm_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_fp16, out_gate_w).sigmoid()

    # ------------------------------------------------------------------ #
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------ #
    if mask is not None:
        # mask may be bool or float; ensure float16 for multiplication
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 4. Apply element‑wise gates
    # ------------------------------------------------------------------ #
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Prepare tensors for the fused GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------ #
    # Move hidden dim to second axis and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose last two axes so that we can multiply A @ B (no extra transpose)
    right_T = right_perm.transpose(-2, -1).contiguous()   # [B,H,N,N] (k,j)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dimensions
    left_flat = left_perm.view(total_batch, N, N)   # (batch, M, K)
    right_flat = right_T.view(total_batch, N, N)    # (batch, K, N)
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides in elements
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # ------------------------------------------------------------------ #
    # 6. Launch the Triton kernel
    # ------------------------------------------------------------------ #
    # Tile sizes – chosen to give good occupancy across the tested N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        total_batch,               # batch*head
    )
    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 7. Reshape back to [B, N, N, H] (float32)
    # ------------------------------------------------------------------ #
    out = (
        out_flat.view(B, H, N, N)      # [B,H,N,N]
        .permute(0, 2, 3, 1)          # [B,N,N,H]
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------ #
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------ #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply output gate (broadcast to hidden dim)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # [B, N, N, dim]

    return output
