"""
TriMul (outgoing) forward pass with a Triton kernel.
The heavy N³ computation of the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched matrix multiplication:
    out_{b,d} = left_{b,d} @ right_{b,d}.T
where left/right are (N,N) matrices per hidden channel.
All other ops (LayerNorm, linear projections, sigmoid gates,
output LayerNorm, gating and final projection) are done in PyTorch.
The kernel launches a 3‑D grid (batch*hidden, row block, col block)
and tiles the GEMM with masks for out‑of‑bounds handling.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix sizes (all equal to seq_len)
    stride_am, stride_ak,           # strides for A (row, col)
    stride_bk, stride_bn,           # strides for B (row, col)  (B is K×N)
    stride_cm, stride_cn,           # strides for C (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)       # row block
    pid_n = tl.program_id(2)       # col block

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the K‑tile for this block
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # current k‑range mask
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float32),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b                     # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and mask handling
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if mask is None or config.get("nomask", False):
        mask_f = torch.ones(bs, N, N, device=device, dtype=x.dtype)
    else:
        mask_f = mask.to(x.dtype)
    mask_f = mask_f.unsqueeze(-1)               # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) using the normalized input
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core TriMul: batched GEMM (left @ right^T) via Triton
    # ------------------------------------------------------------------
    # reshape to (batch*hidden, N, N) where each slice is a matrix
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()       # (B, H, N, N) -> rows=j, cols=k
    total_batches = bs * hidden

    left_mat = left_perm.reshape(total_batches, N, N)
    right_mat = right_perm.reshape(total_batches, N, N)
    out_mat = torch.empty_like(left_mat)                      # fp32

    # Heuristic tile sizes
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 16
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        stride_batch_a=left_mat.stride(0),
        stride_batch_b=right_mat.stride(0),
        stride_batch_c=out_mat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out = out_mat.reshape(bs, hidden, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm + output gate
    # ------------------------------------------------------------------
    w_out_ln = weights["to_out_norm.weight"]
    b_out_ln = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * w_out_ln + b_out_ln

    out_norm = out_norm * out_gate  # out_gate is already float32

    # ------------------------------------------------------------------
    # 6. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    output = F.linear(out_norm, to_out_w)  # (B, N, N, dim)

    return output
