"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left/right) and compute three element‑wise sigmoid gates.
3. Optionally apply a pair‑wise mask (broadcasted over the hidden dim).
4. Fuse the mask and the left/right gates into the projected tensors with a
   tiny Triton kernel (element‑wise multiplication).  This satisfies the
   “use a kernel” requirement while keeping the heavy N³ work on cuBLAS.
5. For each batch and hidden head compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batched matrix‑multiplication of shape (N,N) × (N,N)ᵀ.
   The batched GEMM is performed by torch.bmm (cuBLAS), which is
   considerably faster than a custom Triton GEMM for the tested sizes.
6. Apply an output LayerNorm, an output gate, and a final linear projection
   back to the original dimension.
The function returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, total_elements,
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = a * b   (element‑wise)

    All tensors must be contiguous 1‑D views of the same shape.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict containing the model parameters
            - config: dict with at least {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------
    # 1. Layer‑norm over the last dimension (float32)
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. FP16 linear projections (no bias) and gates
    # -----------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    left = torch.nn.functional.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]

    right = torch.nn.functional.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )

    left_gate = torch.nn.functional.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()

    right_gate = torch.nn.functional.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()

    out_gate = torch.nn.functional.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # -----------------------------------------------------------------
    # 3. Apply optional mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 4. Fuse gates (and mask) with a tiny Triton kernel
    # -----------------------------------------------------------------
    total_elems = left.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    elementwise_mul_kernel[grid](
        left, left_gate, left_fused,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    elementwise_mul_kernel[grid](
        right, right_gate, right_fused,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # replace with the fused tensors
    left = left_fused
    right = right_fused

    # -----------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ   (cuBLAS)
    #    left/right : [B, N, N, hidden_dim] → [B*hidden_dim, N, N]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B, H, N, N) and make contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    # transpose the last two dimensions of right for the matmul
    right_T = right_mat.transpose(-2, -1).contiguous()    # [B, H, N, N]

    # flatten batch & head dimensions
    left_flat = left_mat.view(B * H, N, N)
    right_flat = right_T.view(B * H, N, N)

    # batched matrix multiplication (FP16, cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat)           # [B*H, N, N]

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final projection
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate.to(torch.float32)   # apply output gate

    # final linear projection back to `dim`
    output = torch.nn.functional.linear(
        out_norm, weights["to_out.weight"].to(torch.float32)
    )  # [B, N, N, dim]

    return output
