def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # Should be torch.float32
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    gamma = weights["norm.weight"]
    beta = weights["norm.bias"]
    x = input_tensor
    # Compute mean and var across last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta

    # cast to fp16 for further compute
    x_fp16 = x_norm.to(torch.float16)

    # Linear weights
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    # Proj
    left = F.linear(x_fp16, w_left)   # shape [B,N,N,hidden_dim]
    right = F.linear(x_fp16, w_right)

    # Apply mask if present
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # Gates
    left_gate = torch.sigmoid(F.linear(x_fp16, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_fp16, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_fp16, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape
    # reshape for batched GEMM
    # left shape: (B,N,N,H) -> permute to (B,H,N,N)
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    left_mat = left_mat.view(B * H, N, N)   # (batch, M, K)

    # right transpose K dimension with N: (B,N,N,H) -> (B,H,N,N) then transpose last two dims
    right_mat = right.permute(0, 3, 2, 1).contiguous()
    right_mat = right_mat.view(B * H, N, N)   # (batch, K, N)

    # allocate output
    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # get strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # block sizes (tuned)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,

    )

    out_f32 = out.to(torch.float32)

    # to_out_norm

    beta_out = weights["to_out_norm.bias"]
    mean_o = out_f32.mean(dim=-1, keepdim=True)

    w_to_out = weights["to_out.weight"].to(torch.float32)
    output = F.linear(out_norm, w_to_out)  # shape [B,N,N,dim]

"""
TriMul (outgoing) forward implementation.

The heavy contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication for every head d:
    left_mat  = left.permute(0,3,1,2) → (B, H, N, N)
    right_mat = right.permute(0,3,2,1) → (B, H, N, N)  (k and j swapped)
    out_mat   = (left_mat @ right_mat)  for each (B,H) pair.

We fuse this batched GEMM into a single Triton kernel.
All other operations (LayerNorm, Linear projections, gating) are performed
in PyTorch (mostly in fp16) for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ B for a batch of (M×K) @ (K×N) matrices.
    A : (batch, M, K)
    B : (batch, K, N)
    C : (batch, M, N)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate the block product
        acc += tl.dot(a, b)

    # write back with fp16 truncation
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (float32/bool) or None
            weights_dict : mapping of weight names to torch.Tensors (float32)
            config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Input LayerNorm (over last dim)
    # --------------------------------------------------------------
    gamma = weights["norm.weight"]
    beta = weights["norm.bias"]
    x = input_tensor  # [B, N, N, dim] float32

    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta   # still float32

    # --------------------------------------------------------------
    # Cast to fp16 for the heavy compute
    # --------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # Linear projections (no bias) and gating projections
    # --------------------------------------------------------------
    w_left = weights["left_proj.weight"].to(torch.float16)          # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # project
    left = F.linear(x_fp16, w_left)    # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, w_right)

    # optional mask
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # gates
    left_gate = torch.sigmoid(F.linear(x_fp16, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_fp16, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_fp16, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Batched GEMM via Triton: out = left @ right^T   (over the K axis)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape for batched matmul
    # left: (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    left_mat = left_mat.view(B * H, N, N)

    # right: need to transpose the K dimension with the second axis
    # right: (B, N, N, H) -> (B, H, N, N) -> (B, H, N, N) with K and N swapped
    right_mat = right.permute(0, 3, 2, 1).contiguous()
    right_mat = right_mat.view(B * H, N, N)

    # output buffer
    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides for Triton (in element units)
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tiling parameters – chosen for good occupancy on A100/B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                         # batch dimension
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and convert to fp32
    # --------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]
    out_f32 = out.to(torch.float32)

    # --------------------------------------------------------------
    # Output LayerNorm over hidden_dim
    # --------------------------------------------------------------
    gamma_out = weights["to_out_norm.weight"]
    beta_out = weights["to_out_norm.bias"]

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # --------------------------------------------------------------
    # Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)

    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)   # [B, N, N, dim], float32

    return output
