"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over its last dimension.
2. Compute left/right linear projections and three sigmoidal gates
   (left, right, out) from the normalized input.
3. Apply the optional pairwise mask to the projected tensors.
4. Apply element‑wise gates to the masked projections.
5. Fuse the heavy N³ work with a batched GEMM kernel:
       out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
   i.e. left @ rightᵀ for each (batch, hidden) pair.
   The kernel accumulates in FP32 while storing FP16.
6. Layer‑norm the GEMM result, apply the output gate and a final linear
   projection back to the original dimension.

Only the matrix‑multiplication (step 5) is implemented in Triton; the rest
uses PyTorch (mostly in FP32, casting to FP16 only for the GEMM).
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (batch, head) pairs.
#   A : [total_batch, M, K]   (FP16)
#   B : [total_batch, K, N]   (FP16)   (right_T)
#   C : [total_batch, M, N]   (FP16)
# Accumulation is performed in FP32.
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims
    stride_am, stride_ak,              # A strides (row, col)
    stride_bk, stride_bn,              # B strides (row, col)
    stride_cm, stride_cn,              # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col
    pid_b = tl.program_id(2)               # batch index (batch * hidden)

    # offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # pointers for the current tile (including batch offset)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result (convert back to FP16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------------- #
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor containing the model parameters
            - config: dict with at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # ----------------------------------------------------------------- #
    # Unpack inputs
    # ----------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------------------- #
    # 2. Linear projections (float32) and optional mask
    # ----------------------------------------------------------------- #
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if mask is not None:
        mask_f = mask.to(torch.float32).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------- #
    # 3. Gating (float32)
    # ----------------------------------------------------------------- #
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------- #
    # 4. Cast to FP16 for the heavy GEMM
    # ----------------------------------------------------------------- #
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to [B, hidden, N, N] and make contiguous
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    # Transpose right so that we can perform A @ B (no extra transpose inside kernel)
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dimensions for the kernel
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_T.view(total_batch, N, N)
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # ----------------------------------------------------------------- #
    # 5. Launch the fused GEMM kernel (left @ rightᵀ)
    # ----------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------- #
    # 6. Reshape back to [B, N, N, hidden] (float32)
    # ----------------------------------------------------------------- #
    out = (
        out_flat.view(B, H, N, N)          # [B, hidden, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, hidden]
        .contiguous()
        .to(torch.float32)
    )

    # ----------------------------------------------------------------- #
    # 7. Output LayerNorm, output gate and final projection
    # ----------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # out_gate is already float32

    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
