import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                           stride_az, stride_am, stride_ak,
                           stride_bz, stride_bk, stride_bn,
                           stride_cz, stride_cm, stride_cn,
                           M, N, K,
                           BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)   # batch id
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to output tile
    c_ptrs = C_ptr + pid * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + pid * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        b_ptrs = B_ptr + pid * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------------- LayerNorm on input ----------------
    # use provided weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # input tensor shape: [B, N, N, dim]
    x = input_tensor
    # torch.nn.functional.layer_norm expects normalized shape
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # cast to half for projection
    x_h = x.to(torch.float16)

    # ---------------- Linear projections ----------------
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)      # [B,N,N,hidden_dim]
    right = F.linear(x_h, w_right)

    # Apply mask if supplied
    if mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------------- Contraction via Triton ----------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape for batched GEMM
    # Left: (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Right: transpose the last two axes before view
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tiling parameters (tuned for B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ---------------- Output LayerNorm ----------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_f32 = out.to(torch.float32)

    out_norm = torch.nn.functional.layer_norm(out_f32, (hidden_dim,),
                                              weight=to_out_norm_weight,
                                              bias=to_out_norm_bias,
                                              eps=eps)

    # Apply output gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to dim
    w_to_out = weights["to_out.weight"].to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)  # [B, N, N, dim]

    return output
```

TriMul (outgoing) forward pass with a Triton‐accelerated contraction.

The heavy operation is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of H independent GEMMs of size (N,N) per batch.
We reshape the tensors so that each GEMM is processed by a custom
Triton kernel that tiles the M, N dimensions (128×128) and the
reduction K dimension (32).  All other linear layers and LayerNorms
are performed in PyTorch (mixed fp16/fp32) for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """C = A @ B   (A: (M,K), B: (K,N)) for a batch of matrices.

    A, B, C are stored in row‑major order.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to the output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (K × N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # store fp16 result
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim]  (float32)
            mask   : torch.Tensor [B, N, N]       (float32/bool) or None
            weights: dict of parameter tensors
            config : dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------
    # Input LayerNorm  (over last dimension)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -------------------------------------------------
    # Linear projections (fp16) + optional mask
    # -------------------------------------------------
    x_h = x.to(torch.float16)

    w_left = weights["left_proj.weight"].to(torch.float16)      # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)      # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    if mask is not None:
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # gates (sigmoid)
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Triton batched GEMM: out = left @ rightᵀ over the K dim
    # -------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N) where the last two dims are (i,k) for left
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # for right we need (k,j) ordering before flattening
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides (batch, rows, cols)
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tiling parameters (tuned for NVIDIA B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, hidden_dim] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # Output LayerNorm + out_gate
    # -------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_fp32 = out.to(torch.float32)

    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate.to(torch.float32)

    # -------------------------------------------------
    # Final linear projection back to dim
    # -------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)  # [B, N, N, dim] float32

    return output
