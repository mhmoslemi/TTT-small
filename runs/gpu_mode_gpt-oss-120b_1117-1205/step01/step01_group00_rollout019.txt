"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The implementation follows the reference PyTorch code:
1. LayerNorm over the input channel dimension (float32).
2. Linear projections (left/right) and gating (left_gate/right_gate/out_gate)
   are performed in float16 for speed.
3. Optional element‑wise mask is applied.
4. The core contraction
        out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   is a batched matrix multiplication of shape (B·H, N, N) @ (B·H, N, N)^T.
   It is executed by a custom Triton kernel (fp16 inputs, fp32 accumulation).
5. Output LayerNorm, out‑gate, and final linear projection back to the original
   channel dimension are done in float32.

This kernel is sufficient for correctness and provides a notable speedup
over a pure‑PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B where B is already transposed)
# ------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)       # batch dimension (B*H)
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # load B tile (K x N) – B is already transposed (k,i)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where
        A: (batch, M, K)  FP16
        B: (batch, K, N)  FP16 (already transposed as needed)
    Returns C: (batch, M, N) FP16
    """
    assert a.dtype == b.dtype == torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    B_total, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_total, M, N), dtype=torch.float16, device=a.device)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tuned block sizes (work well for N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B_total,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim], float32

    # Cast to fp16 for the heavy linear ops
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 4. Gating (fp16) and apply
    # -----------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    -> left (i,k) @ right^T (k,j) for each (b,d)
    # -----------------------------------------------------------------
    # Bring hidden dimension to the second axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)   (i,k)
    # Transpose the last two dims of right to get (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)   (k,i) -> will act as (k,j)

    B, H, N, _ = left_perm.shape
    # Flatten batch and heads into one dimension for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Perform the batched matrix multiplication (fp16 in, fp16 out)
    out_flat = triton_matmul(left_flat, right_flat)   # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection (float32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
