def triton_matmul(A, B, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32):
    B_total, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2
    C = torch.empty((B_total, M, N), dtype=A.dtype, device=A.device)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()
    grid = (B_total, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](A, B, C, M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K
    )
    return C

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Compute mask for out-of-bound
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for the batch
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # Init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for K tile
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k_a = offs_k < K
        mask_k_b = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k_a[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k_b[:, None] & mask_n[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Key steps:
1. LayerNorm over the channel dimension (float32).
2. Linear projections (left/right) and gates (left/right/out) via torch.nn.functional.
3. Optional element‑wise mask applied to the projected tensors.
4. Core contraction:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   is implemented as a batched matrix multiplication
   (B·hidden_dim, N, N) @ (B·hidden_dim, N, N)^T using a custom Triton kernel.
5. Output LayerNorm, out‑gate multiplication and final linear projection back
   to the original dimension.

The heavy N³ contraction is off‑loaded to Triton, everything else stays in
PyTorch. The final output is float32 with shape [bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B where A: (batch, M, K), B: (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_matmul(A: torch.Tensor,
                  B: torch.Tensor,
                  BLOCK_M: int = 128,
                  BLOCK_N: int = 128,
                  BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B (float16 inputs, float32 accumulation).
    A: (batch, M, K)  float16, contiguous
    B: (batch, K, N)  float16, contiguous
    Returns C: (batch, M, N) float16
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    grid = (batch,
            triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(input_tensor,
                          (dim,),
                          weight=norm_weight,
                          bias=norm_bias)  # [B, N, N, dim] float32

    # Cast to fp16 for the cheap linear proj/gates
    x_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half‑precision, no bias)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim] fp16

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4. Gates (computed in float32, cast to fp16 for multiplication)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Gates are derived from the *float32* normalized tensor
    left_gate = F.linear(x_norm, left_gate_w).sigmoid()   # fp32
    right_gate = F.linear(x_norm, right_gate_w).sigmoid()  # fp32
    out_gate = F.linear(x_norm, out_gate_w).sigmoid()    # fp32, shape [B,N,N,hidden_dim]

    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    left = left * left_gate_fp16
    right = right * right_gate_fp16

    # --------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # --------------------------------------------------------------
    # Rearrange dimensions so that the hidden dimension is folded into the batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) -> (i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (k, i)

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    # Flatten hidden dimension into the batch dimension
    A = left_perm.view(batch_total, N, N)   # (batch_total, M=N, K=N)
    Bmat = right_perm.view(batch_total, N, N)  # (batch_total, K=N, N=N)

    # Perform batched matrix multiplication with Triton
    C_mat = triton_matmul(A, Bmat)  # (batch_total, N, N) fp16

    # Reshape back to [B, N, N, hidden_dim]
    out = C_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16
    out = out.to(torch.float32)  # cast to fp32 for the remaining ops

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gate, and final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)    # fp32

    out = out * out_gate  # fp32 multiplication with out‑gate

    to_out_w = weights["to_out.weight"].to(torch.float32)  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)  # fp32, final shape [B, N, N, dim]

    return out
