def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # assume float32
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Input layer norm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=1e-5)

    B,N,_,_ = x.shape

    # Linear projections
    x_flat = x.reshape(-1, dim) # shape (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # Gates
    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()

    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # Pairwise contraction: bf16 matmul
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # reshape to (B, hidden_dim, N, N)
    left_bf = left_bf.permute(0,3,1,2) # (B, H, N, N)
    right_bf = right_bf.permute(0,3,1,2) # (B, H, N, N)

    out = torch.matmul(left_bf, right_bf.transpose(-1,-2)) # (B, H, N, N)
    out = out.permute(0,2,3,1) # (B, N, N, H)
    out = out.to(dtype).contiguous()

    # Hidden dim layernorm via Triton
    # allocate output tensor
    out_norm = torch.empty_like(out)
    s_b, s_i, s_j, s_h = out.stride()
    # total kernels:
    total_programs = B * N * N
    grid = (total_programs,)
    # Determine BLOCK_H
    BLOCK_H = ((hidden_dim + 63)//64)*64  # round up to multiple of 64 for performance
    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # Apply out_gate
    out_scaled = out_norm * out_gate

    # Final linear proj
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final

@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    base_in = b * stride_in_b + i * stride_in_i + j * stride_in_j
    base_out = b * stride_out_b + i * stride_out_i + j * stride_out_j

    x = tl.load(inp_ptr + base_in + offs_h * stride_in_h,
                mask=mask_h,
                other=0.0)

    # Compute mean over H
    x_masked = tl.where(mask_h, x, 0.0)
    sum_x = tl.sum(x_masked)
    mean = sum_x / H

    # Compute variance
    diff = x - mean
    sq = diff * diff
    sum_sq = tl.sum(tl.where(mask_h, sq, 0.0))
    var = sum_sq / H

    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_
    tl.store(out_ptr + base_out + offs_h * stride_out_h,
             out, mask=mask_h)
```

"""Implementation of TriMul forward (outgoing version) with a Triton kernel for the hidden-dim layernorm.
...
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b, i, j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # compute base offsets
    base_in = b * stride_in_b + i * stride_in_i + j * stride_in_j
    base_out = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # load slice
    x = tl.load(inp_ptr + base_in + offs_h * stride_in_h,
                mask=mask_h,
                other=0.0)

    # mean
    sum_x = tl.sum(tl.where(mask_h, x, 0.0))
    mean = sum_x / H

    # variance
    diff = x - mean
    sq = diff * diff
    var = tl.sum(tl.where(mask_h, sq, 0.0)) / H

    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)

    out = norm * w + b_
    tl.store(out_ptr + base_out + offs_h * stride_out_h,
             out,
             mask=mask_h)

def custom_kernel(data):
    """TriMul outgoing forward pass.

    Params:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (ignored when config['nomask'] is True)
            - weights: dict of weight tensors
            - config: dict containing keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input layer norm (dim last)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    B, N, _, _ = x.shape

    # 2. Linear projections (project dim -> hidden_dim)
    # Flatten batch and pair dimensions for broadcasting
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # 3. Apply mask (if present) and gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # shape (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # 4. Pairwise contraction over k using batched matrix multiplication in bfloat16
    # Convert to bfloat16 to reduce bandwidth
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # reshape to (B, hidden_dim, N, N) for matmul
    left_bf = left_bf.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_bf = right_bf.permute(0, 3, 1, 2) # (B, H, N, N)

    # out shape (B, H, N, N)
    out = torch.matmul(left_bf, right_bf.transpose(-1, -2))
    # back to original layout (B, N, N, H) and float32
    out = out.permute(0, 2, 3, 1).to(dtype).contiguous()

    # 5. Hidden‑dim LayerNorm using a Triton kernel
    out_norm = torch.empty_like(out)
    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (total_programs,)

    # Choose block size that covers hidden_dim (multiple of 64 for good occupancy)
    BLOCK_H = ((hidden_dim + 63) // 64) * 64

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final

"""
TriMul outgoing forward pass (AlphaFold3 style).

The implementation follows the reference PyTorch code but speeds up the
most expensive part (pairwise contraction) with a batched matrix
multiply in bfloat16 and offloads the hidden‑dimension LayerNorm to a
custom Triton kernel.  The kernel processes one (batch, i, j) slice
at a time, computes mean/variance over the hidden dimension, applies
the learned gain and bias, and writes the result back.

Steps
-----
1. Layer‑norm the input over its last dimension (`dim`).
2. Linear projections into the hidden dimension (`hidden_dim`) for
   `left`, `right` and three gates (`left_gate`, `right_gate`,
   `out_gate`).
3. Apply optional pairwise mask and the gate sigmoids.
4. Contraction over the sequence dimension `k`:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication
   (L @ Rᵀ) in bfloat16.
5. Layer‑norm over the hidden dimension using a Triton kernel.
6. Multiply by `out_gate`.
7. Final linear projection back to `dim`.

All tensors stay on the GPU; only the hidden‑dim layernorm is written
in Triton, satisfying the requirement for a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    """LayerNorm over the hidden dimension H for each (b,i,j) slice."""
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (batch, i, j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets along H
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Base pointers for this slice
    base_in = b * stride_in_b + i * stride_in_i + j * stride_in_j
    base_out = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # Load H‑dim slice
    x = tl.load(
        inp_ptr + base_in + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0,
    )

    # ---- mean ----------------------------------------------------
    sum_x = tl.sum(tl.where(mask_h, x, 0.0))
    mean = sum_x / H

    # ---- variance ------------------------------------------------
    diff = x - mean
    sq = diff * diff
    var = tl.sum(tl.where(mask_h, sq, 0.0)) / H

    # ---- normalize ------------------------------------------------
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ---- apply learned weight & bias --------------------------------
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # Store result
    tl.store(
        out_ptr + base_out + offs_h * stride_out_h,
        out,
        mask=mask_h,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored if config['nomask'])
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------

    x_flat = x.view(-1, dim)  # (B * N * N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = (
        F.linear(x_flat, weights["left_gate.weight"])
        .view(B, N, N, hidden_dim)
        .sigmoid()
    )
    right_gate = (
        F.linear(x_flat, weights["right_gate.weight"])
        .view(B, N, N, hidden_dim)
        .sigmoid()
    )
    out_gate = (
        F.linear(x_flat, weights["out_gate.weight"])
        .view(B, N, N, hidden_dim)
        .sigmoid()
    )

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) and gate activations
    if not nomask:
        # mask shape: (B, N, N) -> (B, N, N, 1)
        mask_exp = mask.unsqueeze(-1).to(dtype)

        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction over the sequence dimension k
    #    Use bfloat16 for reduced bandwidth and a batched matmul.
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    left_bf = left_bf.permute(0, 3, 1, 2)
    right_bf = right_bf.permute(0, 3, 1, 2)

    # out shape (B, H, N, N)
    out = torch.matmul(left_bf, right_bf.transpose(-1, -2))

    out = out.permute(0, 2, 3, 1).to(dtype).contiguous()

    # ------------------------------------------------------------------

    out_norm = torch.empty_like(out)

    BLOCK_H = ((hidden_dim + 63) // 64) * 64

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b,
        s_i,
        s_j,
        s_h,
        s_b,
        s_i,
        s_j,
        s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)
