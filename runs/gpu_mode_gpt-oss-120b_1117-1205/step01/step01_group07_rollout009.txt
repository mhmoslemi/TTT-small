    device = input_tensor.device

    # weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5) # (B,N,N,dim), FP32

    right = F.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))

        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

"""
Optimized outgoing TriMul forward pass.

- LayerNorms and linear projections are computed with PyTorch (FP32/FP16).
- The expensive N‑size batched matrix multiplication
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is performed with a single torch.bmm on FP16 tensors (leveraging cuBLAS).
- A tiny Triton kernel is used for the final element‑wise multiplication
  with the out‑gate, satisfying the requirement of using Triton.
- All tensors are kept on the input CUDA device; the final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: element‑wise multiplication (used for out_gate)
# --------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    """
    out[i] = a[i] * b[i]   for i in [0, size)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool or float) – may be ignored.
            - weights: dict of parameter tensors (all on the same CUDA device)
            - config: dict containing at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Load model weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim], FP32

    # -----------------------------------------------------------------
    # 2) Linear projections to hidden_dim (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim], FP32
    right = F.linear(x, right_proj_weight)  # same shape

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # keep FP32 for later

    # -----------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the heavy matmul
    # -----------------------------------------------------------------
    left = (left.to(torch.float16) * left_gate)
    right = (right.to(torch.float16) * right_gate)

    # -----------------------------------------------------------------
    # 6) Batched matrix multiplication: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched GEMM of shape (B*hidden, N, N) @ (B*hidden, N, N)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # _ == hidden_dim
    P = B * hidden_dim

    # Rearrange to (B, hidden, N, N) and make the second operand transposed on
    # the last two dimensions so that we can use torch.bmm directly.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(P, N, N)      # (P, N, K)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(P, N, N)    # (P, K, N)

    # Batched GEMM (FP16) – cuBLAS does the heavy lifting
    out_mat = torch.bmm(left_mat, right_mat)  # (P, N, N), FP16

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 7) Final LayerNorm over hidden_dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B, N, N, hidden_dim], FP32

    # -----------------------------------------------------------------
    # 8) Multiply by out_gate using a tiny Triton kernel (element‑wise)
    # -----------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    size = out.numel()
    BLOCK = 1024
    grid = (triton.cdiv(size, BLOCK),)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        size,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 9) Final linear projection back to original dim
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim], FP32

    return out
