def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,

):
    pid_m = tl.program_id(0)   # i block
    pid_n = tl.program_id(1)   # j block
    pid_b = tl.program_id(2)   # batch index

    # Compute global offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create masks for i/j based on N
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):

        mask_k = offs_k < K

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # tile over rows (i)
    pid_j = tl.program_id(1)  # tile over cols (j)
    pid_b = tl.program_id(2)  # batch index (M dimension)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_b * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_b * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)  # dot produces FP16, added to FP32 accumulator

    # Store result
    c_ptrs = c_ptr + pid_b * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    Triton-accelerated implementation of the outgoing Triangle Multiplicative Update (TriMul).
    The heavy contraction `out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]` is performed
    by a custom Triton kernel, while all other linear, bias, gating and normalization
    operations are executed with PyTorch (float16 for speed, final output cast to float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    compute_dtype = torch.float16

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Cast inputs to compute dtype
    x = input_tensor.to(compute_dtype)

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"].to(compute_dtype)
    norm_bias = weights["norm.bias"].to(compute_dtype)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    left_proj_w = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w = weights["right_proj.weight"].to(compute_dtype)
    left = F.linear(x, left_proj_w, bias=None)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w, bias=None)

    # ---------- Optional mask ----------
    if not config.get("nomask", False):
        m = mask.to(compute_dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # ---------- Gating ----------
    left_gate_w = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w = weights["out_gate.weight"].to(compute_dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w, bias=None))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w, bias=None))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w, bias=None))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core contraction using Triton ----------
    # Rearrange tensors to (batch*hidden_dim, N, N) and transpose right to (M, K, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()         # (B, hidden_dim, N, N)
    right_T_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, hidden_dim, N, N) with swapped inner dims

    M = bs * hidden_dim
    left_flat = left_perm.view(M, N, N)       # (M, i, k)
    right_flat = right_T_perm.view(M, N, N)   # (M, k, j)

    out_flat = torch.empty_like(left_flat)    # (M, i, j)

    # Strides
    a_stride_m, a_stride_i, a_stride_k = left_flat.stride()
    b_stride_m, b_stride_k, b_stride_j = right_flat.stride()
    c_stride_m, c_stride_i, c_stride_j = out_flat.stride()

    # Triton block sizes (tuned for typical GPUs)
    BLOCK_M = 64   # rows per block
    BLOCK_N = 64   # cols per block
    BLOCK_K = 32   # reduction block size

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M)
    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N, N,
        a_stride_m, a_stride_i, a_stride_k,
        b_stride_m, b_stride_k, b_stride_j,
        c_stride_m, c_stride_i, c_stride_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ---------- Output Normalization ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by output gate
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(compute_dtype)
    out = F.linear(out, to_out_w, bias=None)   # [B, N, N, dim]

Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·hidden_dim) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction (k) dimension
    for k in range(0, K, BLOCK_K):

    c_ptrs = c_ptr + pid_b * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]   (float32)
            - mask:          [B, N, N]       (bool or float)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        output: Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    compute_dtype = torch.float16          # use half precision for all heavy ops

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------------------------------
    # 1. LayerNorm on the input
    # -------------------------------------------------------------------------
    x = input_tensor.to(compute_dtype)
    norm_w = weights["norm.weight"].to(compute_dtype)
    norm_b = weights["norm.bias"].to(compute_dtype)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # -------------------------------------------------------------------------
    # 2. Linear projections (left/right)
    # -------------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w = weights["right_proj.weight"].to(compute_dtype)
    left = F.linear(x, left_proj_w, bias=None)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w, bias=None)

    # -------------------------------------------------------------------------
    # 3. Optional masking (mask has shape [B, N, N])
    # -------------------------------------------------------------------------
    if not config.get("nomask", False):
        m = mask.to(compute_dtype).unsqueeze(-1)    # [B, N, N, 1]
        left = left * m
        right = right * m

    # -------------------------------------------------------------------------
    # 4. Gating
    # -------------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w = weights["out_gate.weight"].to(compute_dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w, bias=None))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w, bias=None))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w, bias=None))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    # -------------------------------------------------------------------------
    # Rearrange to (batch*hidden_dim, N, N) and transpose right for GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, hidden_dim, N, N)
    right_T_perm = right.permute(0, 3, 2, 1).contiguous()     # (B, hidden_dim, N, N) with inner axes swapped

    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)      # (M, i, k)
    right_flat = right_T_perm.view(M, N, N)  # (M, k, j)

    out_flat = torch.empty_like(left_flat)   # (M, i, j)

    # Strides for the flattened tensors
    a_stride_m, a_stride_i, a_stride_k = left_flat.stride()
    b_stride_m, b_stride_k, b_stride_j = right_flat.stride()
    c_stride_m, c_stride_i, c_stride_j = out_flat.stride()

    # Kernel launch parameters (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M)

    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N, N,
        a_stride_m, a_stride_i, a_stride_k,
        b_stride_m, b_stride_k, b_stride_j,
        c_stride_m, c_stride_i, c_stride_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # -------------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # -------------------------------------------------------------------------
    # 7. Final linear projection back to original dimension
    # -------------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(compute_dtype)
    out = F.linear(out, to_out_w, bias=None)   # [B, N, N, dim]

    # Return in float32 as required
    return out.to(torch.float32)
