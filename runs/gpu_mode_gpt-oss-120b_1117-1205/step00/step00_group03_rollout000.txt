"""
TriMul forward pass (outgoing) with a fused Triton kernel for the
core N×N inner‑product reduction.
The heavy part:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is implemented as a batched GEMM where the right operand is accessed
in a transposed view (k, j).  All other ops (LayerNorms, linear
projections, gating) are performed with PyTorch which already provides
high‑performance kernels.
"""

import torch
import triton
import triton.language as tl
from torch import nn, functional as F

# --------------------------------------------------------------
# Triton kernel: batch‑wise matmul with transposed right operand
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    B, N, H,                                          # batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    batch = pid_z // H
    h = pid_z % H

    # start indices of this tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_i = i < N
    mask_j = j < N

    # base pointers for this (batch, h) slice
    left_base = left_ptr + batch * stride_l_b + h * stride_l_h
    right_base = right_ptr + batch * stride_r_b + h * stride_r_h
    out_base = out_ptr + batch * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for tile_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = tile_k * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left[i, k, h]   -> shape (BLOCK_M, BLOCK_K)
        left_off = left_base + i[:, None] * stride_l_i + k[None, :] * stride_l_k
        left = tl.load(left_off, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right_transpose[k, j, h] where right_transpose = right[:, :, :, h].transpose(1,2)
        # --> shape (BLOCK_K, BLOCK_N)
        right_off = right_base + k[:, None] * stride_r_k + j[None, :] * stride_r_i
        right = tl.load(right_off, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)

    # write result (convert back to half)
    out_off = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_off, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# Custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or 0/1)
            - weights_dict: mapping of weight names to torch.Tensors
            - config_dict: contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half and launch Triton kernel for the N×N inner‑product sum
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h = torch.empty_like(left_h)          # [B, N, N, hidden_dim]

    B, N, _, H = left_h.shape

    # strides for the tensors (int64)
    stride_l = left_h.stride()
    stride_r = right_h.stride()
    stride_o = out_h.stride()

    # tile sizes – tuned for B200, works for all test sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    out = out_h.to(torch.float32)            # back to FP32

    # ------------------------------------------------------------------
    # Post‑processing: norm, gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
