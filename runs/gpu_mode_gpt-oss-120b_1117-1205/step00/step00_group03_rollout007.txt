"""
TriMul (outgoing) forward pass with the heavy N³ contraction off‑loaded to Triton.

The implementation follows the reference PyTorch module:
- LayerNorm on the input (dim)
- Linear projections (left/right) and gating (left/right/out)
- Masking and element‑wise gating
- Core contraction:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  implemented as a batched matrix multiplication (left @ rightᵀ) with a
  custom Triton kernel for maximum throughput.
- LayerNorm on the hidden dimension, final gating and linear projection back
  to the original feature dimension.

All tensors are kept on the GPU; the kernel works on FP16 inputs and accumulates
in FP32 for numerical stability. The final output is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ Bᵀ  for a batch of matrices.
    A: (batch, M, K)   (float16)
    B: (batch, K, N)   (float16)   (already transposed)
    C: (batch, M, N)   (float32 accumulation)
    """
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    # number of output blocks in M dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)

    batch_id = pid // num_blocks_m
    pid_m = pid % num_blocks_m

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for the tails
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = (
            a_ptr
            + batch_id * batch_stride_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = (
            b_ptr
            + batch_id * batch_stride_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = (
        c_ptr
        + batch_id * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor : torch.Tensor of shape [B, N, N]
            - weights      : dict of model weight tensors
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------
    # Helper to move weight tensors onto the same device + dtype
    # ------------------------------------------------------------
    def _to(t):
        return t.to(device=device, dtype=input_tensor.dtype)

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = _to(weights["norm.weight"])
    norm_bias = _to(weights["norm.bias"])
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = _to(weights["left_proj.weight"])
    right_proj_w = _to(weights["right_proj.weight"])
    left = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Apply mask (if present)
    # ------------------------------------------------------------------
    mask = mask_tensor.unsqueeze(-1).to(dtype=left.dtype)  # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating networks (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = _to(weights["left_gate.weight"])
    right_gate_w = _to(weights["right_gate.weight"])
    out_gate_w = _to(weights["out_gate.weight"])

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # Cast to fp16 for the N³ contraction
    # ------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # N = seq_len

    # ------------------------------------------------------------------
    # Rearrange tensors so that each (batch, hidden) pair is a separate
    # matrix multiplication:   left @ rightᵀ
    # ------------------------------------------------------------------
    # left -> (B, H, N, N)   where inner dims are (i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right -> (B, H, N, N) where inner dims are (k, i) (i.e. transposed)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Flatten batch & hidden into a single batch dimension
    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)   # (batch_h, M, K)
    right_flat = right_perm.view(batch_h, N, N)  # (batch_h, K, N)

    # Output tensor (FP32 accumulation)
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------
    # Strides (in elements, not bytes)
    # ------------------------------------------------------------
    a_stride_batch, a_stride_m, a_stride_k = left_flat.stride()
    b_stride_batch, b_stride_k, b_stride_n = right_flat.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_flat.stride()

    # ------------------------------------------------------------
    # Launch parameters
    # ------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)

    # grid: (batch_h * num_blocks_m, num_blocks_n)
    grid = (batch_h * num_blocks_m, num_blocks_n)

    _batched_matmul_transpose_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        batch_stride_a=a_stride_batch,
        batch_stride_b=b_stride_batch,
        batch_stride_c=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()       # (B, N, N, H)

    # ------------------------------------------------------------------
    # Hidden‑dim LayerNorm, final gating and linear projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_w = _to(weights["to_out_norm.weight"])
    to_out_norm_b = _to(weights["to_out_norm.bias"])
    to_out_w = _to(weights["to_out.weight"])

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate  # broadcast over i,j

    # final linear projection (no bias)
    output = F.linear(out, to_out_w)  # [B, N, N, dim], float32

    return output
```
