    input_tensor (B,N,N,dim), mask (B,N,N), weights dict, config dict.

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    norm_weight, norm_bias = weights["norm.weight"], weights["norm.bias"]

    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

import triton
import triton.language as tl

@triton.jit
def batch_gemm_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bn, stride_bk,
    stride_ch, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # tile row index
    pid_n = tl.program_id(1)  # tile col index
    pid_b = tl.program_id(2)  # batch index

    # Compute base offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for m and n within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_b * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B_T tile (using B original layout): shape (BLOCK_K, BLOCK_N)
        # B_T element at (k, n) = B[n, k]
        b_ptrs = B_ptr + pid_b * stride_bh + offs_n[None, :] * stride_bn + offs_k[:, None] * stride_bk
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = C_ptr + pid_b * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul "outgoing" implementation with a fused Triton GEMM kernel.

The heavy part of TriMul is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication of shape (B*D, N, N) @ (B*D, N, N).T.
We fuse this computation in a Triton kernel (block‑wise fp16 × fp16 → fp32
accumulator → fp16 store). All other operations (LayerNorm, linear projections,
gating) are performed with PyTorch for simplicity.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the "outgoing" TriMul contraction.
# A : (batch, M, K)   – left tensor   (row‑major)
# B : (batch, N, K)   – right tensor (row‑major, we read its transpose)
# C : (batch, M, N)   – output
# The kernel computes C = A @ B^T block‑wise.
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ah, stride_am, stride_ak,      # A: batch, row (M), col (K)
    stride_bh, stride_bn, stride_bk,      # B: batch, row (N), col (K)  (we read as B^T)
    stride_ch, stride_cm, stride_cn,      # C: batch, row (M), col (N)
    M, N, K,                              # matrix dimensions (runtime)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)      # tile row index
    pid_n = tl.program_id(1)      # tile col index
    pid_b = tl.program_id(2)      # batch (B*hidden) index

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_b * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B^T tile: (BLOCK_K, BLOCK_N)
        # B_T[k, n] == B[n, k]  → address = row_offset*n*stride_bn + col_offset*k*stride_bk
        b_ptrs = B_ptr + pid_b * stride_bh + offs_n[None, :] * stride_bn + offs_k[:, None] * stride_bk
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write result tile
    c_ptrs = C_ptr + pid_b * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data = (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32)
            - mask         : [B, N, N]        (bool/int/float)
            - weights      : dict of model parameters (torch tensors)
            - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Unpack weights (all are float32, bias may be missing for linear layers)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Apply mask (if any) – mask is broadcast over the hidden dimension
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – gates are fused in fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid().to(torch.float16)
    right_gate = F.linear(x, right_gate_weight).sigmoid().to(torch.float16)
    out_gate = F.linear(x, out_gate_weight).sigmoid()   # keep fp32 for later

    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM: out = left @ right^T  (over the sequence dimension)
    #    Shape: (B, N, N, hidden_dim) -> (B*hidden_dim, N, N) @ (B*hidden_dim, N, N)^T
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape
    # Permute to bring hidden_dim next to batch for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch * hidden_dim into a single leading dimension
    A = left_perm.view(-1, N_seq, N_seq).contiguous()   # [B*H, N, N]
    B_mat = right_perm.view(-1, N_seq, N_seq).contiguous()

    # Output placeholder
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides for kernel (int64 values)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B_mat.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Block sizes – tuned for Hopper (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_m = triton.cdiv(N_seq, BLOCK_M)
    total_n = triton.cdiv(N_seq, BLOCK_N)
    total_batches = A.shape[0]                     # B * hidden_dim

    grid = (total_m, total_n, total_batches)

    # Launch Triton kernel
    batch_gemm_outgoing_kernel[grid](
        A, B_mat, C,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bn, stride_bk,
        stride_ch, stride_cm, stride_cn,
        M=N_seq, N=N_seq, K=N_seq,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden_dim] and promote to fp32
    # ------------------------------------------------------------------
    out = C.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Final LayerNorm + out gating
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)
    out = out * out_gate   # out_gate is fp32, matches out

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ------------------------------------------------------------------
    output = F.linear(out, to_out_weight)   # bias is None

    return output
