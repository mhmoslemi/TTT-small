def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # create mask for valid indices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = A_ptr + pid_batch * stride_a_batch
    b_batch_ptr = B_ptr + pid_batch * stride_b_batch   # need stride_b_batch argument as well!
    c_batch_ptr = C_ptr + pid_batch * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for k dimension
        mask_k = offs_k < K

        # compute pointers for A and B
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # dot: a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # ensure same dtype
    dtype = input_tensor.dtype
    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)
    # mask (if present)
    if mask is not None:
        # convert mask to same dtype (float)
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f
    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    # TriMul via batched GEMM
    B, N, _, _ = left.shape  # left shape [B,N,N,hidden]
    # Permute to (B, hidden, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch dim
    batch = B * hidden_dim
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)
    out_flat = torch.empty_like(left_flat)
    # Strides
    a_batch_stride = left_flat.stride(0)
    a_m_stride = left_flat.stride(1)
    a_k_stride = left_flat.stride(2)
    b_batch_stride = right_flat.stride(0)
    b_k_stride = right_flat.stride(2)   # k dimension stride in original (col)
    b_n_stride = right_flat.stride(1)   # n dimension stride (row)
    c_batch_stride = out_flat.stride(0)
    c_m_stride = out_flat.stride(1)
    c_n_stride = out_flat.stride(2)
    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # Launch grid
    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    # out_gate multiplication
    out = out * out_gate
    # final linear
    output = torch.nn.functional.linear(out, to_out_weight)
    return output

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy part of the TriMul module is the tensor contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is equivalent to a batch of independent matrix multiplications:
    out_d = left_d @ right_d.T   for each hidden dimension d.
We fuse the N³ reduction across the sequence dimension using a
batched GEMM kernel written in Triton. The rest of the module
(LayerNorm, Linear projections and gating) is performed with PyTorch
operations.
"""
import torch
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes (M=N, K=N)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each batch compute C = A @ B.T  (A:MxK, B:NxK)"""
    pid_batch = tl.program_id(0)        # batch = B * hidden_dim
    pid_i = tl.program_id(1)            # block row
    pid_j = tl.program_id(2)            # block col

    # offsets inside the output tile
    offs_m = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    a_batch_ptr = A_ptr + pid_batch * stride_a_batch
    b_batch_ptr = B_ptr + pid_batch * stride_b_batch
    c_batch_ptr = C_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m
                                + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B block (transposed view): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k
                                + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m
                            + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    TriMul forward pass: input tensor [B, N, N, dim],
    mask [B, N, N], model weights and config.
    Returns output tensor [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ----- unpack weights -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm on input -----
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )

    # ----- Linear projections -----
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----- Apply mask (if present) -----
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gates -----
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- TriMul via batched GEMM -----
    # bring hidden dimension to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch = B * hidden_dim
    left_flat = left_perm.view(batch, N, N)             # [batch, N, N]
    right_flat = right_perm.view(batch, N, N)
    out_flat = torch.empty_like(left_flat)              # allocate result

    # strides for the flattened tensors
    a_batch_stride = left_flat.stride(0)
    a_m_stride = left_flat.stride(1)
    a_k_stride = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_k_stride = right_flat.stride(2)   # stride of K (last dim)
    b_n_stride = right_flat.stride(1)   # stride of N (row dim)

    c_batch_stride = out_flat.stride(0)
    c_m_stride = out_flat.stride(1)
    c_n_stride = out_flat.stride(2)

    # block sizes (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        a_batch_stride,
        a_m_stride,
        a_k_stride,
        b_batch_stride,
        b_k_stride,
        b_n_stride,
        c_batch_stride,
        c_m_stride,
        c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- Output normalisation, gate and final linear -----
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, to_out_weight)

    return output
