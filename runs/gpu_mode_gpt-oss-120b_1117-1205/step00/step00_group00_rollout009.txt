@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          batch, M, N, K,
                          stride_ab, stride_ak, stride_a_batch,
                          stride_bk, stride_bn, stride_b_batch,
                          stride_cm, stride_cn, stride_c_batch,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for M,N: row start, col start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # K dimension pointer offsets
    offs_k = tl.arange(0, BLOCK_K)

    # Define pointers
    a_ptrs = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_ab + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=offs_m < M, other=0.0)
        b = tl.load(b_ptrs, mask=offs_n < N, other=0.0)
        # Compute dot
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # Store with mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

    device = input_tensor.device

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_ab, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Batched matrix multiplication: C = A @ B
    A: [batch, M, K]  (float16)
    B: [batch, K, N]  (float16)
    C: [batch, M, N]  (float16)
    The kernel computes each block of C per batch.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # compute pointers
        a_ptrs = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_ab + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # masks for OOB loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing TriMul operator.
    Expects data = (input_tensor, mask_tensor, weights_dict, config_dict).
    Returns tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Extract weights and ensure proper dtype
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # LayerNorm over dim
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim]

    # Projections (float32)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Optional mask
    if not nomask:
        # mask shape [B,N,N] -> broadcast to last dim
        mask_unsq = mask.unsqueeze(-1).type_as(left)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates (float32)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half precision for the expensive N^3 contraction
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Prepare matrices for batched matmul:
    # left: [B,N,N,hidden] -> [B, hidden, N, N]
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()
    # right: transpose the last two dims before permute
    right_perm = right_f16.permute(0, 3, 2, 1).contiguous()

    B, N, _, _ = input_tensor.shape  # B,N,N,dim
    batch = B * hidden_dim

    A = left_perm.view(batch, N, N)
    B_mat = right_perm.view(batch, N, N)

    # Allocate output tensor for the matmul
    C = torch.empty_like(A)  # fp16

    # Compute strides (in elements)
    stride_a_batch, stride_a_m, stride_a_n = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tiling parameters (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        batch,
        N,
        N,
        N,
        stride_ab=stride_a_m,
        stride_ak=stride_a_n,
        stride_a_batch=stride_a_batch,
        stride_bk=stride_b_k,
        stride_bn=stride_b_n,
        stride_b_batch=stride_b_batch,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        stride_c_batch=stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,hidden] and cast to float32 for following ops
    out_fp16 = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)

    # to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply out gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    # Ensure output type is float32
    out = out.to(torch.float32)
    return out

"""
Fast Triton implementation of the outgoing TriMul operator (AlphaFold3).

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reduced to a batched matrix multiplication:
    C = A @ B
where A = left (shape [B, hidden, N, N]) and B = right transposed.
The batched matmul kernel is written in Triton; the surrounding
operations (layer‑norm, linear projections, gating) are performed in
PyTorch, using float16 for the contraction to save bandwidth.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_ab, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[batch, M, N] = A[batch, M, K] @ B[batch, K, N]   (FP16 inputs, FP32 accumulation)."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        a_ptrs = A_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_ab \
                 + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)   # FP16·FP16 → FP32

    # write back
    c_ptrs = C_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of TriMul.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]  (float32, on CUDA)
            - mask_tensor:  [B, N, N] (float32 or bool)
            - weights_dict: model parameters (all float32 tensors)
            - config_dict: contains "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- unpack weights -----
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    to_out_weight = weights["to_out.weight"].to(device)

    # ----- LayerNorm over channel dim -----
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]

    # ----- Linear projections (float32) -----
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # ----- Optional mask -----
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).type_as(left)   # broadcast
        left = left * mask_unsq
        right = right * mask_unsq

    # ----- Gating (float32) -----
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to FP16 for the N³ contraction -----
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # ----- Prepare batched matmul (A @ B) -----
    # left:   [B, N, N, hidden] -> [B, hidden, N, N]
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()
    # right:  [B, N, N, hidden] -> transpose (k,j) -> [B, hidden, N, N]
    right_perm = right_f16.permute(0, 3, 2, 1).contiguous()

    B, N, _, _ = input_tensor.shape
    batch = B * hidden_dim

    A = left_perm.view(batch, N, N)          # [batch, N, N]  FP16
    B_mat = right_perm.view(batch, N, N)     # [batch, N, N]  FP16
    C = torch.empty_like(A)                  # result, FP16

    # ----- Strides (in elements) -----
    stride_a_batch, stride_a_m, stride_a_n = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # ----- Triton launch parameters -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        batch,
        N,
        N,
        N,
        stride_a_m,          # stride_ab
        stride_a_n,          # stride_ak
        stride_a_batch,      # stride_a_batch
        stride_b_k,          # stride_bk  (row stride of B)
        stride_b_n,          # stride_bn  (col stride of B)
        stride_b_batch,      # stride_b_batch
        stride_c_m,          # stride_cm
        stride_c_n,          # stride_cn
        stride_c_batch,      # stride_c_batch
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Reshape back to [B, N, N, hidden] -----
    out_fp16 = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)

    # ----- Output LayerNorm over hidden dim -----
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ----- Apply final gate and projection -----
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # → [B, N, N, dim]

    return out.to(torch.float32)
