Let's examine the TriMul forward:

- x shape [B, N, N, dim]
- norm: LayerNorm(dim) applied per last dimension.

- left = left_proj(x) shape [B,N,N,hidden_dim]
- right = right_proj(x) shape [B,N,N,hidden_dim]

- mask shape [B,N,N], unsqueeze -1, multiply left, right elementwise.

- left_gate = left_gate(x).sigmoid() shape [B,N,N,hidden_dim]
- right_gate = right_gate(x).sigmoid() same
- out_gate = out_gate(x).sigmoid() shape [B,N,N,hidden_dim]

- left = left * left_gate
- right = right * right_gate

- out = einsum('... i k d, ... j k d -> ... i j d', left, right)  (sum over k dimension)

Effectively, for each batch b, for each i,j,d: out[b,i,j,d] = sum_{k=0}^{N-1} left[b,i,k,d] * right[b,j,k,d]

Note that left is [B, N, N, H]; inner dimension N stands at third index k. (b, i, k, d). Right is (b, j, k, d). So we sum over k.

Then apply to_out_norm (LayerNorm(hidden_dim)), out = out * out_gate, then final Linear to_out (hidden_dim->dim), output shape [B,N,N,dim].

    N,  # dimension of square matrices

    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # Compute offsets for each tile
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # offsets along k for tile iteration
    # Load A tile [M, K] and B tile [K, N]

@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Offsets in a, b, c
    a_batch_ptr = a_ptr + batch * stride_a_batch
    b_batch_ptr = b_ptr + batch * stride_b_batch
    c_batch_ptr = c_ptr + batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # masks
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # load a tile (M x K)
        a = tl.load(
            a_batch_ptr + (offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col),
            mask=a_mask,
            other=0.0
        )
        # load b tile (K x N)
        b = tl.load(
            b_batch_ptr + (offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col),
            mask=b_mask,
            other=0.0
        )
        acc += tl.dot(a, b)  # automatically uses float32 accumulation
    # store to c
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    c = acc.to(tl.float16)  # store as half, could also keep float32
    tl.store(
        c_batch_ptr + (offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col),
        c,
        mask=c_mask
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape # input shape includes dim as last dim
    hidden_dim = config["hidden_dim"]
    # Cast to float32 for speeds.
    x = input_tensor
    
    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections:
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    mask_unsq = mask.unsqueeze(-1).to(x.dtype) # broadcast
    left = left * mask_unsq
    right = right * mask_unsq

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Permute and reshape for kernel
    # left: [B,N,N,H]; need shape [B,H,N,N] then flatten batch
    left_perm = left_half.permute(0, 3, 1, 2).contiguous()
    right_perm = right_half.permute(0, 3, 2, 1).contiguous() # note transpose

    B, H, N, N_check = left_perm.shape
    assert N == N_check, "N mismatch"

    # flatten batch dimension: P = B * H
    P = B * H
    left_flat = left_perm.view(P, N, N)
    right_flat = right_perm.view(P, N, N)

    out_flat = torch.empty((P, N, N), dtype=torch.float16, device=left_flat.device)

"""
TriMul forward kernel (outgoing version) with Triton accelerated contraction.

The heavy operation is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each batch and hidden head `d` this is a matrix multiplication
    (N×N) @ (N×N)^T .
We fuse this batched GEMM into a Triton kernel, while keeping the
surrounding LayerNorms, linear projections and gates in PyTorch.
All tensors are kept on the GPU; the contraction uses float16
for bandwidth and float32 accumulation for numerical safety.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B  (A: [M,K], B: [K,N])
# Batch dimension is the first program id.
# --------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    N,                                 # square matrix size
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # load tiles (A: M×K, B: K×N)
        a = tl.load(
            a_batch
            + (offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col),
            mask=a_mask,
            other=0.0,
        )
        b = tl.load(
            b_batch
            + (offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col),
            mask=b_mask,
            other=0.0,
        )
        # fused dot product, accumulates in FP32
        acc += tl.dot(a, b)

    # write results (store as FP16 to save memory)
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    c = acc.to(tl.float16)
    tl.store(
        c_batch
        + (offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col),
        c,
        mask=c_mask,
    )

# --------------------------------------------------------------
# Entry point used by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask: torch.Tensor [B, N, N] (binary or float)
            weights: dict of torch.Tensors containing all model parameters
            config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -------- LayerNorm on the input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias
    )  # shape [B,N,N,dim]

    # -------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # -------- Mask ----------
    # mask may be bool or float; ensure same dtype for multiplication
    mask_ = mask.to(dtype).unsqueeze(-1)        # [B,N,N,1]
    left = left * mask_
    right = right * mask_

    # -------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Triton-accelerated contraction (left @ right^T)
    # -------------------------------------------------
    # Convert to FP16 for the matmul kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute so that the hidden dimension becomes a batch dimension.
    # left: [B,N,N,H] -> [B,H,N,N] (row i, col k)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on the last two axes: (j,k) -> (k,j)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()

    # Flatten batch*head dimension for the kernel
    P = B * hidden_dim
    left_flat = left_perm.view(P, N, N)
    right_flat = right_perm.view(P, N, N)
    out_flat = torch.empty((P, N, N), dtype=torch.float16, device=device)

    # Strides (row‑major layout)
    s_a_batch, s_a_row, s_a_col = left_flat.stride()
    s_b_batch, s_b_row, s_b_col = right_flat.stride()
    s_c_batch, s_c_row, s_c_col = out_flat.stride()

    # Tile sizes (tuned for 1024² matrices)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        P,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        s_a_batch,
        s_a_row,
        s_a_col,
        s_b_batch,
        s_b_row,
        s_b_col,
        s_c_batch,
        s_c_row,
        s_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,hidden_dim] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -------- Post‑contraction LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )

    # -------- Output gate ----------
    out = out * out_gate

    # -------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
