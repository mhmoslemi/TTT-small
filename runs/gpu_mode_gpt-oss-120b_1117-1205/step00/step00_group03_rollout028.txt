def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over last dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask
    mask = mask.unsqueeze(-1).to(left.dtype)  # broadcast
    left = left * mask
    right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to bfloat16 for matmul
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # Permute for batched matmul
    left_perm = left_bf16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_bf16.permute(0, 3, 2, 1).contiguous()

    B, D, N, _ = left_perm.shape  # D = hidden_dim
    # reshape to (B*D, N, N)
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.bfloat16)

    # get strides
    a_ptr = left_mat
    b_ptr = right_mat
    c_ptr = out_mat
    # compute stride values as ints (in elements)
    stride_a_batch = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_b_batch = right_mat.stride(0)
    stride_bk = right_mat.stride(1)  # corresponds to k (rows) bc B*D,N,K
    stride_bn = right_mat.stride(2)
    stride_c_batch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Setup block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), left_mat.shape[0])

    triton_matmul_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        N, N, N,
        stride_am, stride_ak, stride_a_batch,
        stride_bk, stride_bn, stride_b_batch,
        stride_cm, stride_cn, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Reshape back
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).to(torch.float32)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply with out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input.
2. Linear projections to a hidden dimension, apply mask and sigmoid gates.
3. Cast the gated tensors to float16 and transpose them so that
   left  : (B, D, N, N) contains the (i, k) matrix for each hidden dim.
   right : (B, D, N, N) contains the (k, j) matrix (i.e. the transposed
            second matrix of the einsum).
   The two tensors are viewed as a batch of size B*D, each a (N×N) matrix.
4. Run a batched GEMM in Triton:
   out[b·D+ d] = left[b·D+ d] @ right[b·D+ d]   (float16 inputs,
   accumulation in float32, result stored as float16).
5. Reshape / permute the result back to (B, N, N, D) and cast to float32.
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   apply the final linear projection back to the original `dim`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,          # A: row, col, batch
    stride_bk, stride_bn, stride_bb,          # B: row (k), col (n), batch
    stride_cm, stride_cn, stride_cb,          # C: row, col, batch
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication for the TriMul core.

    A: (batch, M, K)   – stored as (batch, row, col)
    B: (batch, K, N)
    C: (batch, M, N)   – result
    """
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch (B*D)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile for this batch
    a_ptrs = (
        a_ptr
        + pid_b * stride_ab
        + offs_m[:, None] * stride_am
        + offs_k[None, :] * stride_ak
    )
    b_ptrs = (
        b_ptr
        + pid_b * stride_bb
        + offs_k[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k_offset = k_iter * BLOCK_K
        cur_k_valid = (offs_k + k_offset) < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & cur_k_valid[None, :],
            other=tl.float16(0.0),
        )
        b = tl.load(
            b_ptrs,
            mask=cur_k_valid[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )

        # Cast to float32 for numerically stable accumulation
        a_f32 = tl.cast(a, tl.float32)
        b_f32 = tl.cast(b, tl.float32)
        acc += tl.dot(a_f32, b_f32)

        # Advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back the block to C
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        tl.cast(acc, tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation using a Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] (0/1)
            - weights      : dict of model weights (torch.Tensors)
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ----- Load model parameters ------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- Layer‑norm -----------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections ----------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Mask (optional) -------------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ----- Gates -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to float16 for the heavy multiplication -------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Permute so that the hidden dimension becomes a batch dimension
    # (B, D, N, N) where D = hidden_dim
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_f16.permute(0, 3, 2, 1).contiguous()  # transpose second N

    # Collapse batch & hidden dim => (B*D, N, N)
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)

    # Output placeholder
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # ----- Prepare strides for Triton -------------------------------------------
    stride_a_batch = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # ----- Triton launch --------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        left_mat.shape[0],  # batch = B * hidden_dim
    )
    _tri_mul_batch_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Reshape back to (B, N, N, hidden_dim) -------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ----- Output layer‑norm, gating and final projection -----------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)
    return out
