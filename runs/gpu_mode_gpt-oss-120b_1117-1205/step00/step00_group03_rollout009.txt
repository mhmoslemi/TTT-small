   @triton.jit
   def tri_mul_kernel(
           left_ptr, right_ptr, out_ptr,
           B, N, H,
           stride_left_b, stride_left_i, stride_left_k, stride_left_h,
           stride_right_b, stride_right_i, stride_right_k, stride_right_h,
           stride_out_b, stride_out_i, stride_out_j, stride_out_h,
           BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
       ):
       pid_i = tl.program_id(0)
       pid_j = tl.program_id(1)
       pid_hb = tl.program_id(2)  # maps to batch * heads

       # Compute head and batch index.
       num_heads = H // BLOCK_H
       # Actually we set BLOCK_H=1, so num_heads = H.
       # Compute overall index
       total_h = pid_hb
       # Derive batch b and head h
       b = total_h // H   # integer division
       h = total_h % H

       # Compute start indices
       i_start = pid_i * BLOCK_M
       j_start = pid_j * BLOCK_N

       # Define offsets within tile
       offs_i = i_start + tl.arange(0, BLOCK_M)
       offs_j = j_start + tl.arange(0, BLOCK_N)

       # Mask out-of-bounds
       mask_i = offs_i < N
       mask_j = offs_j < N

       # Initialize accumulator
       acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)   # using fp32 accumulation

       # Loop over k
       num_k_blocks = tl.cdiv(N, BLOCK_K)
       for k_iter in range(num_k_blocks):
           k_start = k_iter * BLOCK_K
           offs_k = k_start + tl.arange(0, BLOCK_K)
           mask_k = offs_k < N

           # Load left tile: shape (BLOCK_M, BLOCK_K)
           left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
           left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

           # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
           # We want right element at (k, j). So index with k along row, j along column.
           right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + offs_j[None, :] * stride_right_i + offs_k[:, None] * stride_right_k
           # This yields shape (BLOCK_K, BLOCK_N). careful: off_j broadcast first dimension
           right = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0).to(tl.float32)

           # Compute dot product: for each i row and j column, sum over k => tl.dot over dim K
           acc += tl.dot(left, right)  # left (M,K), right (K,N) -> (M,N)

       # Write output tile
       out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
       tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # dtype: we assume float32, else cast.
    # Normalize input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    
    # Linear projections
    # left_proj_weight shape (hidden_dim, dim)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    # Apply mask if exists
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp
    
    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate
    
    # Cast to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    
    B, N, _, H = left_fp16.shape
    
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)  # same shape
    
    # prepare strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp16.stride()
    
    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    
    # Compute grid dims
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_hb = B * H  # each program for batch-head pair
    
    tri_mul_kernel[ (grid_i, grid_j, grid_hb) ](
        left_fp16, right_fp16, out_fp16,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=1,
    )
    
    out = out_fp16.to(torch.float32)
    
    # Output layer norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    
    # out_gate
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    out = out * out_gate
    
    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    
    return out

"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy part of TriMul is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be seen as a batched GEMM for each hidden dimension d.
We implement this reduction with a Triton kernel that tiles the i‑j
output matrix (blocks M×N) and the reduction axis k (block K).  The
kernel processes a single head (d) and batch element per program and
loops over k‑blocks accumulating in FP32.  All other operations
(LayerNorm, Linear projections, sigmoids, and final projection) are
performed with PyTorch for simplicity.

The kernel is launched with a 3‑D grid:
    (num_i_blocks, num_j_blocks, B * hidden_dim)
where each program handles a distinct (batch, head) pair.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the O(N³) contraction: out = Σ_k left[...,k,:] * right[...,k,:]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)               # i‑block
    pid_j = tl.program_id(1)               # j‑block
    pid_hb = tl.program_id(2)              # combined batch‑head index

    # decode batch and head from pid_hb
    b = pid_hb // H                        # batch index
    h = pid_hb % H                         # head index

    # start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # block offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # bounds checking
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)

    # loop over k
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)  (i, k)
        left_offset = (
            b * stride_left_b + h * stride_left_h +
            offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile **transposed**: shape (BLOCK_K, BLOCK_N)  (k, j)
        # we read right[b, j, k, h] but present it as (k, j)
        right_offset = (
            b * stride_right_b + h * stride_right_h +
            offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        )
        right = tl.load(right_ptr + right_offset,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # dot product over k dimension
        acc += tl.dot(left, right)          # (M, N)

    # store result tile
    out_offset = (
        b * stride_out_b + h * stride_out_h +
        offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptr + out_offset,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation with Triton acceleration.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (float, 0/1)
            - weights: dict with required weight tensors
            - config: dict with keys "dim" (C) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------- unpack -------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", False)

    # ------------------- LayerNorm on input -------------------
    # LayerNorm over the last dim (C)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------- Linear projections -------------------
    # left & right: shape [B, N, N, H]
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------- Mask (optional) -------------------
    if not nomask:
        # mask shape [B,N,N] -> broadcast over hidden dim
        mask_exp = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------- Gating -------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # ------------------- Triton contraction -------------------
    # Ensure tensors are contiguous and on the same device
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=device)  # final result in FP32

    # Strides (int) for pointer arithmetic
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tile sizes – chosen to fit typical GPU shared memory and registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_hb = B * H   # each program processes one (batch, head) pair

    # Launch kernel
    tri_mul_kernel[
        (grid_i, grid_j, grid_hb)
    ](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # modest number of warps (adjustable)
    )

    # ------------------- Output layer norm -------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # ------------------- Output gate -------------------
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    out = out * out_gate

    # ------------------- Final linear projection -------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
