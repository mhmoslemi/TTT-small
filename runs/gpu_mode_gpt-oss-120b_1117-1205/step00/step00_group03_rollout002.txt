```python
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_b, stride_i, stride_k, stride_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # instance index (batch * hidden_dim)
    pid1 = tl.program_id(1)  # i block
    pid2 = tl.program_id(2)  # j block

    # Compute batch and hidden index
    b = pid0 // hidden_dim
    hd = pid0 % hidden_dim

    # offsets for i and j
    i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # instance offset
    offset_instance = b * stride_b + hd * stride_h

    # create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for i,k and j,k
        mask_i = i < N
        mask_j = j < N
        mask_k = k < N

        # offsets for left: [i,k]
        left_ptrs = left_ptr + offset_instance \
                    + i[:, None] * stride_i \
                    + k[None, :] * stride_k

        # offsets for right: [j,k]
        right_ptrs = right_ptr + offset_instance \
                     + j[None, :] * stride_r_i \
                     + k[:, None] * stride_r_k

        # load tiles
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0)

        # convert to fp32 for accumulation
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        # dot product
        acc += tl.dot(left, right)  # (BLOCK_M, BLOCK_N)

    # store result
    out_ptrs = out_ptr + offset_instance \
               + i[:, None] * stride_o_i \
               + j[None, :] * stride_o_j
    mask_store = (i[:, None] < N) & (j[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)
```

"""
TriMul forward kernel (outgoing version).

The heavy O(N³·C) operation is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We fuse this as a batched matrix multiplication for each hidden
dimension `d` (and batch `b`).  The Triton kernel computes
C = A @ Bᵀ for each (b, d) instance where
    A[i,k] = left[b,i,k,d]   (size N×N)
    B[j,k] = right[b,j,k,d]  (size N×N)
The result C[i,j] is stored in `out`.

All other operations (LayerNorm, Linear projections, gates) are
performed with PyTorch.  The kernel works in FP16 for memory‑bandwidth
efficiency and accumulates in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    # left strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h,
    # right strides
    right_stride_b, right_stride_i, right_stride_k, right_stride_h,
    # out strides
    out_stride_b, out_stride_i, out_stride_j, out_stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid0 = tl.program_id(0)               # instance = batch * hidden_dim
    pid_i = tl.program_id(1)               # i‑tile
    pid_j = tl.program_id(2)               # j‑tile

    # decode batch and hidden indices
    b = pid0 // hidden_dim
    hd = pid0 % hidden_dim

    # base offsets for this instance
    offset_left  = b * left_stride_b  + hd * left_stride_h
    offset_right = b * right_stride_b + hd * right_stride_h
    offset_out   = b * out_stride_b   + hd * out_stride_h

    # tile coordinates
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range of rows/cols inside the tile
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows/cols (used when storing)
    i_mask = i < N
    j_mask = j < N
    ij_mask = i_mask[:, None] & j_mask[None, :]

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # -------- load left tile (i,k) --------
        # shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + offset_left
            + i[:, None] * left_stride_i
            + k[None, :] * left_stride_k
        )
        left_mask = i_mask[:, None] & k_mask[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # -------- load right tile (j,k) --------
        # shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + offset_right
            + j[None, :] * right_stride_i   # row index uses stride_i (same as left's i‑stride)
            + k[:, None] * right_stride_k
        )
        right_mask = j_mask[None, :] & k_mask[:, None]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # -------- accumulate dot product --------
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # -------- store result (FP16) --------
    out_ptrs = (
        out_ptr
        + offset_out
        + i[:, None] * out_stride_i
        + j[None, :] * out_stride_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=ij_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) module.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C]
            - mask: torch.Tensor [B, N, N] (float or bool)
            - weights: dict of model weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on the input (over the last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Apply mask if provided (mask may be all‑ones for nomask=True)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()

    # ------------------------------------------------------------------
    # Triton kernel for the Einstein summation (batched matmul)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)    # shape [B,N,N,hidden_dim]

    # strides (in elements) for the three tensors
    ls_b, ls_i, ls_k, ls_h = left_fp16.stride()
    rs_b, rs_i, rs_k, rs_h = right_fp16.stride()
    os_b, os_i, os_j, os_h = out_fp16.stride()

    # tile sizes – chosen to fit typical H100/B200 SM resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                            # instance dimension (batch × hidden)
        (N + BLOCK_M - 1) // BLOCK_M,              # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,              # j‑tiles
    )

    tri_mul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # left strides
        ls_b,
        ls_i,
        ls_k,
        ls_h,
        # right strides
        rs_b,
        rs_i,
        rs_k,
        rs_h,
        # out strides
        os_b,
        os_i,
        os_j,
        os_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert the result back to FP32
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # Output processing: LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
```
