@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, stride_a_m, stride_a_k,
    batch_stride_b, stride_b_n, stride_b_k,
    batch_stride_c, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # batch
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # compute batch offset
    batch = pid
    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = A_ptr + batch * batch_stride_a + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (N x K) as we need B^T (K x N): B is originally [N,K] row-major where rows are j and cols are k. To multiply A @ B.T we need B.T shape (K,N). So we can load B as (N,K) and then use tl.dot with trans_b=True, which expects B as (N,K) and transposes internally.

        b_ptrs = B_ptr + batch * batch_stride_b + offs_n[:, None] * stride_b_n + offs_k[None, :] * stride_b_k
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

    out = out_flat.view(B, H, N, N).permute(0,2,3,1)

"""
TriMul GPU kernel implementation (outgoing update).

The kernel computes the core O(N^3) contraction:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
for each batch b and hidden channel h (hidden_dim).  The rest of the TriMul
module (layer-norms, linear projections, gating) is performed in PyTorch.
The contraction is expressed as a batched matrix multiplication with
`right` transposed; it is implemented as a Triton kernel using block‑tiling
and FP16 arithmetic with FP32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM with B*H batches, each multiplying
#   A (M x K) with B^T (K x N)  ->  C (M x N)
#   Here M = N = K = sequence length.
# -------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_am, stride_ak,   # strides for A: batch, M, K
    stride_bz, stride_bn, stride_bk,   # strides for B: batch, N, K
    stride_cz, stride_cm, stride_cn,   # strides for C: batch, M, N
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(2)   # batch*hidden index
    pid_m = tl.program_id(0)   # output block row (i)
    pid_n = tl.program_id(1)   # output block col (j)

    # Offsets within the M/N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = a_ptr + pid_z * stride_az + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (N x K) – will be multiplied as B^T
        b_ptrs = b_ptr + pid_z * stride_bz + offs_n[:, None] * stride_bn + cur_k[None, :] * stride_bk
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Compute contribution to the output block.
        # tl.dot with trans_b=True treats B as (K,N) internally.
        acc += tl.dot(a, b, trans_b=True)

    # Store the result
    c_ptrs = c_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the Triangle Multiplicative Update (outgoing) using a
    Triton kernel for the heavy O(N^3) contraction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (bool/int/float)
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    torch._C._set_grad_enabled(False)  # ensure no autograd overhead
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm on input ----------
    # LayerNorm over last dimension (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Cast to half for the rest of the pipeline
    x_norm = x_norm_fp32.half()

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"].half()
    right_proj_weight = weights["right_proj.weight"].half()
    left_gate_weight = weights["left_gate.weight"].half()
    right_gate_weight = weights["right_gate.weight"].half()
    out_gate_weight = weights["out_gate.weight"].half()
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Project to hidden dimension (half precision)
    left = F.linear(x_norm, left_proj_weight)       # [B,N,N,hidden]
    right = F.linear(x_norm, right_proj_weight)    # [B,N,N,hidden]

    # ---------- Mask ----------
    if nomask:
        mask_half = torch.ones_like(mask, dtype=torch.float16, device=mask.device).unsqueeze(-1)
    else:
        mask_half = mask.to(torch.float16).unsqueeze(-1)   # broadcast over hidden dim

    left = left * mask_half
    right = right * mask_half

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # half
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight)) # half
    left = left * left_gate
    right = right * right_gate

    # ---------- Core contraction via Triton ----------
    B, N, _, _ = left.shape  # N is sequence length
    # Rearrange tensors to (B, hidden, N, N) and flatten batch*hidden dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    P = B * hidden_dim
    left_flat = left_perm.view(P, N, N)    # [P, N, N]
    right_flat = right_perm.view(P, N, N)

    out_flat = torch.empty_like(left_flat)   # half precision output

    # Grid configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j)
        P,                         # batch*hidden
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Reshape back to original layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden]

    # Convert to float32 for the remaining operations
    out = out.float()

    # ---------- Post‑contraction LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ---------- Output gating ----------
    # out_gate is computed from the original (float32) normalized input
    out_gate = torch.sigmoid(F.linear(x_norm_fp32, out_gate_weight))   # [B,N,N,hidden]
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
