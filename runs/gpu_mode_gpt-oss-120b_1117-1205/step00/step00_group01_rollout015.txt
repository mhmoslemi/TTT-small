def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr, mask_ptr,
    B, N, hidden_dim,
    stride_b_l, stride_i_l, stride_k_l, stride_d_l,
    stride_b_r, stride_i_r, stride_k_r, stride_d_r,
    stride_b_o, stride_i_o, stride_j_o, stride_d_o,
    stride_b_m, stride_i_m, stride_j_m,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    batch = pid_bh // hidden_dim
    d = pid_bh % hidden_dim
    
    # block offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    
    i_valid = i_offset < N
    j_valid = j_offset < N
    
    # base pointers for the selected hidden slice
    left_base = left_ptr + batch * stride_b_l + d * stride_d_l
    right_base = right_ptr + batch * stride_b_r + d * stride_d_r
    out_base = out_ptr + batch * stride_b_o + d * stride_d_o
    
    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # optional mask base
    if HAS_MASK:
        mask_base = mask_ptr + batch * stride_b_m
    
    # loop over k dimension in tiles
    for k_base in range(0, N, BLOCK_K):
        k_offset = k_base + tl.arange(0, BLOCK_K)
        k_valid = k_offset < N
        
        # Load left tile (M,K)
        left_ptrs = left_base + i_offset[:, None] * stride_i_l + k_offset[None, :] * stride_k_l
        a = tl.load(left_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)
        if HAS_MASK:
            mask_left_ptrs = mask_base + i_offset[:, None] * stride_i_m + k_offset[None, :] * stride_j_m
            mask_left = tl.load(mask_left_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)
            a = a * mask_left

        # Load right tile transposed (K,N)

        right_ptrs = right_base + k_offset[:, None] * stride_k_r + j_offset[None, :] * stride_i_r
        b = tl.load(right_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)
        if HAS_MASK:
            # mask for right (j,k) transposed to (k,j)
            mask_right_ptrs = mask_base + k_offset[:, None] * stride_j_m + j_offset[None, :] * stride_i_m
            mask_right = tl.load(mask_right_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)
            b = b * mask_right
        
        # dot product over K dimension, accumulate
        # a shape (M,K), b shape (K,N) -> result (M,N)
        acc += tl.dot(a, b)
    
    # write out result

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_b_l, stride_i_l, stride_k_l, stride_d_l,
    stride_b_r, stride_i_r, stride_k_r, stride_d_r,
    stride_b_o, stride_i_o, stride_j_o, stride_d_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    batch = pid_bh // hidden_dim
    d = pid_bh % hidden_dim

    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    i_valid = i_offset < N
    j_valid = j_offset < N

    # base pointers
    left_base = left_ptr + batch * stride_b_l + d * stride_d_l
    right_base = right_ptr + batch * stride_b_r + d * stride_d_r
    out_base = out_ptr + batch * stride_b_o + d * stride_d_o

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_base in range(0, N, BLOCK_K):
        k_offset = k_base + tl.arange(0, BLOCK_K)
        k_valid = k_offset < N

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy part of the module is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where `left` and `right` are the result of linear projections (with optional mask
and gating).  This is equivalent to a batch of matrix‑multiplications
    out_d = left_d @ right_d.T   for each hidden dimension d.
The kernel below computes this contraction using a tiled GEMM.  The rest of
the TriMul pipeline (LayerNorms, linear layers, sigmoids) is performed in
plain PyTorch for simplicity.

The kernel works on a fused batch‑hidden dimension:
    program_id(0) → (batch, hidden_dim) pair,
    program_id(1) → i‑tile,
    program_id(2) → j‑tile.
Each program accumulates a BLOCK_M × BLOCK_N tile in fp32 and stores the
result in fp16.  Masking (if required) is applied in the PyTorch code before
the kernel, so the kernel itself contains the core matmul only.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: computes out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const fp16 *
    right_ptr,              # const fp16 *
    out_ptr,                # fp16 *
    B, N, hidden_dim,       # int32 scalar arguments
    stride_b_l, stride_i_l, stride_k_l, stride_d_l,   # left strides
    stride_b_r, stride_i_r, stride_k_r, stride_d_r,   # right strides
    stride_b_o, stride_i_o, stride_j_o, stride_d_o,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------
    # 0‑D program IDs
    # ------------------------------------------------------
    pid_bh = tl.program_id(0)          # batch * hidden_dim
    pid_i  = tl.program_id(1)          # i‑tile
    pid_j  = tl.program_id(2)          # j‑tile

    batch = pid_bh // hidden_dim
    d     = pid_bh % hidden_dim

    # ------------------------------------------------------
    # 1‑D offsets inside the tile
    # ------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # ------------------------------------------------------
    # Base pointers for the selected (batch, d) slice
    # ------------------------------------------------------
    left_base = left_ptr  + batch * stride_b_l + d * stride_d_l
    right_base = right_ptr + batch * stride_b_r + d * stride_d_r
    out_base   = out_ptr   + batch * stride_b_o + d * stride_d_o

    # ------------------------------------------------------
    # Accumulator in fp32
    # ------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------
    # Loop over k dimension (inner dimension of the GEMM)
    # ------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        k_offset = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # ---- left tile (i,k) -------------------------------------------------
        left_ptrs = left_base + \
                    i_offset[:, None] * stride_i_l + \
                    k_offset[None, :] * stride_k_l
        a = tl.load(left_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)                 # fp16

        # ---- right tile transposed (k,j) ------------------------------------
        # right is stored as (batch, j, k, d). We need (k, j) for the dot.
        right_ptrs = right_base + \
                     k_offset[:, None] * stride_k_r + \
                     j_offset[None, :] * stride_i_r
        b = tl.load(right_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0)                 # fp16

        # ---- dot‑product over K, accumulate in fp32 -------------------------
        acc += tl.dot(a, b)                     # (M,N) += (M,K)·(K,N)

    # ------------------------------------------------------
    # Store the result (cast back to fp16)
    # ------------------------------------------------------
    out_ptrs = out_base + \
               i_offset[:, None] * stride_i_o + \
               j_offset[None, :] * stride_j_o
    store_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)


# --------------------------------------------------------------
# Public entry point used by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (may be None)
        - weights_dict : mapping of parameter names to torch.Tensors
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if (mask_tensor is None) or nomask:
        # no masking – use a tensor of ones
        mask_fp = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=left.dtype,
            device=device,
        )
    else:
        mask_fp = mask_tensor.to(dtype=left.dtype)

    # unsqueeze to broadcast over hidden dimension
    mask_unsq = mask_fp.unsqueeze(-1)   # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and make tensors contiguous for the kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    # the mask is no longer needed inside the kernel
    # ------------------------------------------------------------------
    # 6. Allocate output for the contracted tensor (fp16)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    out_contract = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float16, device=device
    )

    # ------------------------------------------------------------------
    # 7. Launch Triton kernel (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (in elements) of the tensors
    stride_b_l, stride_i_l, stride_k_l, stride_d_l = left.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_d_r = right.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_d_o = out_contract.stride()

    grid = (
        B * hidden_dim,                # batch * hidden_dim
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
    )

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_contract.data_ptr(),
        B, N, hidden_dim,
        stride_b_l, stride_i_l, stride_k_l, stride_d_l,
        stride_b_r, stride_i_r, stride_k_r, stride_d_r,
        stride_b_o, stride_i_o, stride_j_o, stride_d_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Remaining TriMul post‑processing in PyTorch
    # ------------------------------------------------------------------
    # Convert contracted result to float32 for the remaining ops
    out = out_contract.to(torch.float32)

    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    output = F.linear(out, weights["to_out.weight"])

    return output
