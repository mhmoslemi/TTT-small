@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    N, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    pid_hb = tl.program_id(1)
    # Compute batch and hidden indices
    batch_idx = pid_hb // hidden_dim
    hidden_idx = pid_hb % hidden_dim

    # Compute block coordinates
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_i = (pid // num_blocks_n) * BLOCK_M
    block_j = (pid % num_blocks_n) * BLOCK_N

    # Pointers to start of A, B, C for this batch and hidden
    a_ptr_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_ptr_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_h
    c_ptr_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_h

    # Offsets for row (i) and col (j)
    a_offset = block_i * stride_a_i
    b_offset = block_j * stride_b_j
    c_offset = block_i * stride_c_i + block_j * stride_c_j

    # Compute pointers
    a_ptr = a_ptr_base + a_offset
    b_ptr = b_ptr_base + b_offset
    c_ptr = c_ptr_base + c_offset

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for cur_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = cur_k * BLOCK_K

        # Offsets within A: (i,k)
        a_tile_ptrs = a_ptr + (tl.arange(0, BLOCK_M)[:, None] * stride_a_i + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_a_k)

        # Offsets within B: (k,j) (since B is transposed)
        b_tile_ptrs = b_ptr + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_b_k + tl.arange(0, BLOCK_N)[None, :] * stride_b_j)

        # Load tiles
        a_tile = tl.load(a_tile_ptrs, mask=(tl.arange(0, BLOCK_M)[:, None] < N - block_i) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < N - k_offset), other=0.0)
        b_tile = tl.load(b_tile_ptrs, mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < N - k_offset) & (tl.arange(0, BLOCK_N)[None, :] < N - block_j), other=0.0)

        # Convert to float32 for accumulation
        a_tile = a_tile.to(tl.float32)
        b_tile = b_tile.to(tl.float32)

        # Dot product
        acc += tl.dot(a_tile, b_tile)

        # Increment pointers for next k block
        a_ptr += BLOCK_K * stride_a_k
        b_ptr += BLOCK_K * stride_b_k

    # Store
    c_tile_ptrs = c_ptr + tl.arange(0, BLOCK_M)[:, None] * stride_c_i + tl.arange(0, BLOCK_N)[None, :] * stride_c_j
    mask = (tl.arange(0, BLOCK_M)[:, None] < N - block_i) & (tl.arange(0, BLOCK_N)[None, :] < N - block_j)
    tl.store(c_tile_ptrs, acc.to(tl.float16), mask=mask)

    right_proj_weight = weights["right_proj.weight"]

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # Ensure weights are on same device & dtype float32
    # Possibly not needed but cast

    norm_bias = norm_bias.to(device)
    left_proj_weight = left_proj_weight.to(device)
    right_proj_weight = right_proj_weight.to(device)
    left_gate_weight = left_gate_weight.to(device)
    right_gate_weight = right_gate_weight.to(device)
    out_gate_weight = out_gate_weight.to(device)
    to_out_norm_weight = to_out_norm_weight.to(device)
    to_out_norm_bias = to_out_norm_bias.to(device)
    to_out_weight = to_out_weight.to(device)

    x = torch.nn.functional.layer_norm(

    right = right * right_gate

"""TriMul (outgoing) forward pass with a Triton kernel.

The heavy part of the operation is
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which can be written as a batched matrix multiplication:
    out[b, d] = left[b, d] @ right[b, d].transpose(-2, -1)

We implement this with a custom Triton GEMM kernel that processes one
(hidden) slice `d` for a given batch `b` at a time.  The kernel works on
tiles of (BLOCK_M, BLOCK_N) and accumulates over K‑tiles.  All other
operations (LayerNorm, linear projections, gating) are done with
standard PyTorch primitives.  The kernel uses float16 for the inputs and
stores the result in float32, keeping the final precision identical to
the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    a_ptr,          # left  [B, N, N, H] (float16)
    b_ptr,          # right [B, N, N, H] (float16)
    c_ptr,          # out   [B, N, N, H] (float32)

    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_j, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,

    N, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B^T for each (batch, hidden) pair."""
    pid = tl.program_id(0)          # tiles over i,j
    pid_hb = tl.program_id(1)       # combined batch * hidden index

    # ---- decode batch and hidden indices ---------------------------------
    batch_idx = pid_hb // hidden_dim
    hidden_idx = pid_hb % hidden_dim

    # ---- tile coordinates -------------------------------------------------
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    i0 = (pid // num_blocks_j) * BLOCK_M
    j0 = (pid % num_blocks_j) * BLOCK_N

    # ---- base pointers for this batch / hidden slice ----------------------
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_h
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_h

    # ---- offset to the beginning of the current tile -----------------------
    a_ptr_tile = a_base + i0 * stride_a_i
    b_ptr_tile = b_base + j0 * stride_b_j
    c_ptr_tile = c_base + i0 * stride_c_i + j0 * stride_c_j

    # ---- accumulator -------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ranges for the tile dimensions
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # ---- loop over K dimension --------------------------------------------
    for cur_k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = cur_k * BLOCK_K

        # pointers to sub‑tiles
        a_tile_ptrs = a_ptr_tile + i_range[:, None] * stride_a_i + (k0 + k_range)[None, :] * stride_a_k
        b_tile_ptrs = b_ptr_tile + (k0 + k_range)[:, None] * stride_b_k + j_range[None, :] * stride_b_j

        # masks for out‑of‑bounds loads
        mask_a = (i_range[:, None] < N - i0) & (k_range[None, :] < N - k0)
        mask_b = (k_range[:, None] < N - k0) & (j_range[None, :] < N - j0)

        # load (still in float16), promote to float32 for the dot
        a_tile = tl.load(a_tile_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b_tile = tl.load(b_tile_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a_tile, b_tile)

    # ---- store the result ---------------------------------------------------
    mask_c = (i_range[:, None] < N - i0) & (j_range[None, :] < N - j0)
    c_tile_ptrs = c_ptr_tile + i_range[:, None] * stride_c_i + j_range[None, :] * stride_c_j
    tl.store(c_tile_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or float)
            - weights_dict: contains all needed linear / norm parameters
            - config_dict: must contain "dim", "hidden_dim", and optionally "nomask"

    Returns:
        output torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- extract weights ---------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # ----- LayerNorm ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ----- Linear projections ------------------------------------------------
    left = F.linear(x, left_proj_weight)       # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Optional mask ------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----- Gates -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Convert to float16 for the gemm kernel ----------------------------
    left_h = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)

    # ----- Allocate output (float32) -----------------------------------------
    out_fp32 = torch.empty_like(left_h, dtype=torch.float32, device=device)

    B, N, _, _ = left_h.shape

    # ----- Strides (in elements) --------------------------------------------
    stride_a_batch = left_h.stride(0)
    stride_a_i = left_h.stride(1)
    stride_a_k = left_h.stride(2)
    stride_a_h = left_h.stride(3)

    stride_b_batch = right_h.stride(0)
    stride_b_j = right_h.stride(1)
    stride_b_k = right_h.stride(2)
    stride_b_h = right_h.stride(3)

    stride_c_batch = out_fp32.stride(0)
    stride_c_i = out_fp32.stride(1)
    stride_c_j = out_fp32.stride(2)
    stride_c_h = out_fp32.stride(3)

    # ----- Tile sizes --------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_i * num_j, B * hidden_dim)

    # ----- Launch Triton kernel ------------------------------------------------
    trimul_gemm_kernel[grid](
        left_h, right_h, out_fp32,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_k, stride_b_j, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing (LayerNorm, gating, final linear) -----------------
    # to_out_norm operates over the hidden dimension
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # back to original dim

    return output
