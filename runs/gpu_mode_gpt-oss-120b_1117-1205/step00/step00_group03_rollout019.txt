def custom_kernel(data):
    import torch, triton, triton.language as tl
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, N, _, _ = input_tensor.shape

    # extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]  # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # layernorm x
    eps = 1e-5
    # Use torch.nn.functional.layer_norm:
    # normalized_shape = dim
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # linear projections (no bias)
    # ensure weight dtype matches compute dtype
    # x_norm dtype may be float32; we will compute in fp16 for speed
    x_norm_fp16 = x_norm.to(torch.float16)

    left = torch.nn.functional.linear(x_norm_fp16, left_proj_weight.to(torch.float16))  # shape [B,N,N,H] fp16
    right = torch.nn.functional.linear(x_norm_fp16, right_proj_weight.to(torch.float16))

    # gating
    left_gate = torch.nn.functional.linear(x_norm_fp16, left_gate_weight.to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_fp16, right_gate_weight.to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_fp16, out_gate_weight.to(torch.float16)).sigmoid()

    # mask
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # prepare shapes for gemm
    # left shape: [B,N,N,H] -> permute to [B,H,N,N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose right for gemm: shape [B,H,N,N] -> rows k, cols j => right_T shape [B,H,N,N] where last two dims are (K,N); Actually we need right_T = right_perm.transpose(-2,-1)
    right_T = right_perm.transpose(-2, -1).contiguous()

    total_batch = batch_size * hidden_dim
    M = N
    K = N
    # reshape to 3D for kernel
    left_flat = left_perm.view(total_batch, M, K)
    right_flat = right_T.view(total_batch, K, N)

    # allocate output fp16 buffer
    out_flat = torch.empty((total_batch, M, N), dtype=torch.float16, device=device)

    # Compute strides
    # left_flat contiguous: stride for dim0 (batch) = left_flat.stride(0) * elementsize? Actually PyTorch stride returns number of elements. So we can get as int.
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)  # row stride = K
    stride_ak = left_flat.stride(2)  # col stride = 1

    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(1)  # row stride = N
    stride_bn = right_flat.stride(2)  # col stride = 1

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)  # row stride = N
    stride_cn = out_flat.stride(2)  # col stride = 1

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid
    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N), total_batch)

    # Launch kernel
    trimul_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape output back
    out = out_flat.view(batch_size, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # convert to float32
    out = out.to(torch.float32)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # apply out_gate (need to upcast out_gate dtype)
    out = out * out_gate.to(torch.float32)

    # final linear
    output = torch.nn.functional.linear(out, to_out_weight.to(torch.float32))

    return output

"""
TriMul (outgoing) forward pass implemented with a fused Triton GEMM kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Compute left/right projections and their sigmoid gates using the provided
   linear weights (all in fp16 for speed).
3. Apply the optional mask and the element‑wise gates.
4. For every batch b and hidden dimension h we need
        out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   which is a batched matrix multiplication of shape (N,N) with its transpose.
   The heavy N³ work is performed by a custom Triton kernel that computes
   left @ rightᵀ for all (b·h) batches in one launch.
5. The result (still fp16) is reshaped, cast to fp32 and passed through the
   output LayerNorm, the output gate and the final linear projection.

The kernel computes A (M×K) × B (K×N) → C (M×N) for many independent
batches (total_batch = batch_size * hidden_dim).  Accumulation is done in
float32 for numerical stability; the final matrix is stored as fp16.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) where
#   A : [total_batch, M, K]   (fp16)
#   B : [total_batch, K, N]   (fp16)   (already transposed, i.e. B = rightᵀ)
#   C : [total_batch, M, N]   (fp16)
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # tensor pointers
    M, N, K,                           # matrix dimensions (int)
    stride_am, stride_ak,              # A strides (row, col)
    stride_bk, stride_bn,              # B strides (row, col)
    stride_cm, stride_cn,              # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)               # block row index
    pid_n = tl.program_id(1)               # block col index
    pid_b = tl.program_id(2)               # batch (b·h) index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑bounds masks for rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulation buffer in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)

        # mask for the current K‑tile
        mask_k = (k_offset + offs_k) < K

        # pointers to the current block of A and B (including batch offset)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # load tiles (apply masks)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # write the result back in FP16
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------------- #
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask: torch.Tensor [B, N, N] (bool or float) or None
            - weights: dict of torch.Tensor containing all linear / ln parameters
            - config: dict containing at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    import math

    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the last dimension
    # --------------------------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to fp16 for all subsequent heavy ops
    x_norm_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias) and gating
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(
        x_norm_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = torch.nn.functional.linear(
        x_norm_fp16, weights["right_proj.weight"].to(torch.float16)
    )

    left_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # --------------------------------------------------------------------- #
    # 3. Apply optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Prepare tensors for the fused GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    # Permute to [B, H, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right so that we can do A @ B (no extra transpose)
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dimensions for the kernel
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_T.view(total_batch, N, N)
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # --------------------------------------------------------------------- #
    # 6. Launch the Triton kernel
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, H] and continue in FP32
    # --------------------------------------------------------------------- #
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # --------------------------------------------------------------------- #
    # 8. Output LayerNorm, output gate and final linear projection
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)  # apply output gate

    # Final linear projection back to `dim`
    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # shape [B, N, N, dim]

    return output
