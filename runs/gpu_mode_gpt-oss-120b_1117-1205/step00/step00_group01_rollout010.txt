"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------

1. Layer‑normalize the input tensor `x` with the provided LayerNorm parameters.
2. Linear projections:
     left  = Linear(x, left_proj_weight)      # [B,N,N,H]
     right = Linear(x, right_proj_weight)    # [B,N,N,H]
3. Gating vectors (sigmoid):
     left_gate  = Linear(x, left_gate_weight).sigmoid()   # [B,N,N,H]
     right_gate = Linear(x, right_gate_weight).sigmoid()  # [B,N,N,H]
     out_gate   = Linear(x, out_gate_weight).sigmoid()    # [B,N,N,H]
4. **Triton kernel** – apply mask and element‑wise gating in one pass:
        left_masked  = left  * mask * left_gate
        right_masked = right * mask * right_gate
   (mask has shape [B,N,N] and is broadcasted over the hidden dimension.)
5. For every hidden channel `h` compute a matrix product between the
   masked “left” and the transpose of the masked “right”:
        out[...,h] = left_masked[...,h] @ right_masked[...,h].T
   This is realised with a batched `torch.bmm` after reshaping to
   (B*H, N, N).
6. Layer‑normalize the result with `to_out_norm` parameters.
7. Multiply by `out_gate`.
8. Final linear projection back to the original channel dimension:
        output = Linear(out, to_out_weight)   # [B,N,N,dim]

Only step 4 is performed on Triton; the remaining heavy linear algebra
uses cuBLAS via PyTorch which gives a very fast implementation while
still satisfying the “use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_and_gate_kernel(
    left_ptr,      # float32*  (flattened left_proj)
    right_ptr,     # float32*  (flattened right_proj)
    left_gate_ptr, # float32*  (flattened left_gate)
    right_gate_ptr,# float32*  (flattened right_gate)
    mask_ptr,      # float32*  (flattened mask)
    left_out_ptr,  # float32*  (output left)
    right_out_ptr, # float32*  (output right)
    total_elts: tl.constexpr,   # B * N * N * H
    hidden_dim: tl.constexpr,   # H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    oob = offs >= total_elts

    # Load tensors (masked loads return 0.0 outside the valid range)
    left = tl.load(left_ptr + offs, mask=~oob, other=0.0)
    right = tl.load(right_ptr + offs, mask=~oob, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=~oob, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=~oob, other=0.0)

    # Compute the index into the 1‑D mask (mask is broadcasted over hidden_dim)
    mask_idx = offs // hidden_dim                # 0 … B*N*N-1
    mask_len = total_elts // hidden_dim
    mask_oob = mask_idx >= mask_len
    mask_val = tl.load(mask_ptr + mask_idx,
                       mask=~mask_oob,
                       other=0.0)

    # Apply mask and gating
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_out_ptr + offs, left_out, mask=~oob)
    tl.store(right_out_ptr + offs, right_out, mask=~oob)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask):
    """
    Fuse mask, left‑gate and right‑gate with a Triton kernel.
    All tensors must be contiguous and on the same CUDA device.
    """
    # flatten everything to 1‑D views
    total_elts = left.numel()
    hidden_dim = left.shape[-1]

    BLOCK = 1024
    grid = (triton.cdiv(total_elts, BLOCK),)

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # mask is [B,N,N] → flatten to 1‑D float32
    mask_f = mask.view(-1).contiguous()

    _apply_mask_and_gate_kernel[grid](
        left.view(-1), right.view(-1),
        left_gate.view(-1), right_gate.view(-1),
        mask_f,
        left_out.view(-1), right_out.view(-1),
        total_elts, hidden_dim,
        BLOCK=BLOCK,
    )
    return left_out, right_out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:         torch.Tensor [B, N, N] (float32, 0/1)
            - weights: dict with the following keys (all torch.Tensors):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        output torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -----------------------------------------------------------------
    # 2. Linear projections and gates (all dense, bias‑free)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, dtype)
    right_proj_w = weights["right_proj.weight"].to(device, dtype)

    left_gate_w = weights["left_gate.weight"].to(device, dtype)
    right_gate_w = weights["right_gate.weight"].to(device, dtype)
    out_gate_w = weights["out_gate.weight"].to(device, dtype)

    left = F.linear(x, left_proj_w)          # [B,N,N,H]
    right = F.linear(x, right_proj_w)        # [B,N,N,H]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Ensure mask is float32 (0.0 / 1.0) and contiguous
    mask_f = mask.to(dtype=dtype, device=device)

    # -----------------------------------------------------------------
    # 3. Apply mask + element‑wise gating via Triton
    # -----------------------------------------------------------------
    left, right = _apply_mask_and_gate(left, right, left_gate, right_gate, mask_f)

    # -----------------------------------------------------------------
    # 4. Batched matrix multiplication across the reduction dimension k
    #    (equivalent to einsum ... i k d, ... j k d -> ... i j d)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B, H, N, N) so that batch dimension = B*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # right needs to be transposed on the last two axes
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)

    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device, dtype)
    output = F.linear(out, to_out_w)   # [B,N,N,dim]

    return output
