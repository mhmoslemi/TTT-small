@triton.jit
def triton_batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ah, stride_ak, stride_ac,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_ci, stride_cj,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute batch index and block coords
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_i = block_idx // num_blocks_n
    block_j = block_idx % num_blocks_n

    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Pointers for A and B
    a_ptrs = A_ptr + batch_idx * stride_ah + offs_i[:, None] * stride_ak + tl.arange(0, BLOCK_K)[None, :] * stride_ac
    b_ptrs = B_ptr + batch_idx * stride_bh + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_j[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)  # but maybe treat all elements
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K - k*BLOCK_K) & mask_j[None, :], other=0.0)
        # compute dot
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_ac
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = C_ptr + batch_idx * stride_ch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']
    # optional mask if nomask param: maybe config has "nomask"? not in provided dict. But we can check.
    # Use torch.nn.functional.layer_norm, etc.

    # flatten device.

    # compute layernorm
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=dim, weight=weights['norm.weight'], bias=weights['norm.bias'])
    # It will automatically broadcast over all dimensions except last one.

    # left
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])
    # mask handling
    if mask_tensor is not None:
        mask = mask_tensor[..., None].to(left.dtype)
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # convert left/right to half to feed into kernel (optional)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # permute and reshape
    B, N, _, _ = left.shape
    # left: [B,N,N,H] -> [B,H,N,N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

    batch_h = B * hidden_dim
    lhs = left_perm.view(batch_h, N, N)
    rhs = right_perm.view(batch_h, N, N)

    # allocate output tensor
    out_flat = torch.empty_like(lhs, dtype=torch.float32)  # will hold result

    # define block sizes based on N
    # heuristics
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64  # maybe 64 for efficiency
    # ensure they are constexpr values: but we set them as host constants to pass as kernel args.
    # compute grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_h * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    # compute strides
    stride_lhs_batch = lhs.stride(0)
    stride_lhs_m = lhs.stride(1)
    stride_lhs_k = lhs.stride(2)
    stride_rhs_batch = rhs.stride(0)
    stride_rhs_k = rhs.stride(1)
    stride_rhs_n = rhs.stride(2)
    stride_out_batch = out_flat.stride(0)
    stride_out_i = out_flat.stride(1)
    stride_out_j = out_flat.stride(2)

    triton_batch_matmul_kernel[grid](
        lhs, rhs, out_flat,
        stride_lhs_batch, stride_lhs_m, stride_lhs_k,
        stride_rhs_batch, stride_rhs_k, stride_rhs_n,
        stride_out_batch, stride_out_i, stride_out_j,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape out to original shape
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()
    # apply to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, normalized_shape=hidden_dim,
                                         weight=weights['to_out_norm.weight'],
                                         bias=weights['to_out_norm.bias'])
    # multiply out_gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights['to_out.weight'])
    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        stride_ah, stride_ak, stride_ac,
                        stride_bh, stride_bk, stride_bn,
                        stride_ch, stride_ci, stride_cj,
                        M, N, K,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # compute block coordinates
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n
    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    block_i = block_idx // num_blocks_n
    block_j = block_idx % num_blocks_n

    # offsets within block
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # base pointers for this block (k dimension starts at 0)
    a_base = A_ptr + batch_idx * stride_ah + offs_i[:, None] * stride_ak
    b_base = B_ptr + batch_idx * stride_bh + offs_j[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = cur_k + offs_k < K

        a = tl.load(a_base + cur_k * stride_ac,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_base + cur_k * stride_bk,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = C_ptr + batch_idx * stride_ch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator using a custom Triton kernel
    to compute the heavy ternary matrix multiplication.
    The kernel computes, for each batch and hidden channel:
        out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
    The rest of the operations (layer norm, linear projections, gating)
    are performed with PyTorch.
    """
    import torch
    import torch.nn.functional as F
    import triton

    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm over last dim
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Optional mask (mask may be boolean or float)
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating signals (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half precision for the heavy matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # Rearrange to [B, hidden_dim, N, N] then flatten batch+channel
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim

    lhs = left_perm.view(batch_h, N, N)
    rhs = right_perm.view(batch_h, N, N)

    # Output buffer
    out_flat = torch.empty_like(lhs, dtype=torch.float32)

    # Block sizes (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_h * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    # Strides (in elements)
    stride_ah = lhs.stride(0)
    stride_ak = lhs.stride(1)
    stride_ac = lhs.stride(2)

    stride_bh = rhs.stride(0)
    stride_bk = rhs.stride(1)   # k dimension
    stride_bn = rhs.stride(2)   # n dimension (j)

    stride_ch = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    batch_matmul_kernel[grid](
        lhs, rhs, out_flat,
        stride_ah, stride_ak, stride_ac,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_ci, stride_cj,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to original layout [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Final LayerNorm over hidden_dim
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # Apply output gate (still fp32)
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy part of the module is the ternary matrix multiplication
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to a batched matrix multiplication where the second
matrix is transposed (rows of `right` are dotted with rows of `left`).
The kernel processes each (batch, hidden) pair as a separate
matrix‑multiply using block‑wise tiling and half‑precision arithmetic.
All other operations (LayerNorm, linear projections, gating) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matmul of shape (B*H, N, N) @ (B*H, N, N).T
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ah, stride_ak, stride_ac,        # A: (batch_h, M, K)
    stride_bh, stride_bk, stride_bn,        # B_T: (batch_h, K, N)  (original B is (batch_h, N, K))
    stride_ch, stride_ci, stride_cj,        # C: (batch_h, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------ block indices ------------------------------------------ #
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    block_i = block_idx // num_blocks_n
    block_j = block_idx % num_blocks_n

    # ------------ offsets within the tile -------------------------------- #
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # ------------ base pointers for A and B_T (k starts at 0) ------------- #
    a_base = A_ptr + batch_idx * stride_ah + offs_i[:, None] * stride_ak          # (BLOCK_M, K)
    b_base = B_ptr + batch_idx * stride_bh + offs_j[None, :] * stride_bn          # (K, BLOCK_N)

    # ------------ accumulator --------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------ loop over k dimension --------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):

            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,

            b_base + cur_k * stride_bk,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_N)

    # ------------ write back ---------------------------------------------- #
    c_ptrs = C_ptr + batch_idx * stride_ch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
