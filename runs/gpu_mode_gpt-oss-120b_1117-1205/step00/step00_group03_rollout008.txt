def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    B, N, _, C = input_tensor.shape
    assert C == dim

    # Load weights
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # LayerNorm over last dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    left_proj = torch.nn.functional.linear(x_flat, left_proj_weight)  # (B*N*N, hidden_dim)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_weight)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_weight))

    # reshape to (B,N,N,hidden_dim)
    left_proj = left_proj.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Ensure contiguous
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    # mask to float32
    mask_f = mask.to(torch.float32)

    # Allocate output buffers for masked gated tensors
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    # Triton kernel
    @triton.jit
    def mask_gate_fuse_kernel(
        left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
        mask_ptr,
        out_left_ptr, out_right_ptr,
        B, N, H,
        BLOCK_SIZE: tl.constexpr
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        total = B * N * N * H
        mask = offs < total

        # load values
        left = tl.load(left_ptr + offs, mask=mask, other=0.0)
        left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
        right = tl.load(right_ptr + offs, mask=mask, other=0.0)
        right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

        # compute indices b,i,j,d
        # In integer arithmetic:
        temp = offs
        b = temp // (N * N * H)
        temp = temp % (N * N * H)
        i = temp // (N * H)
        temp = temp % (N * H)
        j = temp // H
        # mask offset
        mask_idx = b * (N * N) + i * N + j
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

        out_left = left * mask_val * left_gate
        out_right = right * mask_val * right_gate

        tl.store(out_left_ptr + offs, out_left, mask=mask)
        tl.store(out_right_ptr + offs, out_right, mask=mask)

    BLOCK_SIZE = 1024
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    # Get pointers
    mask_f_contig = mask_f.contiguous()
    mask_f_ptr = mask_f_contig.data_ptr()
    left_ptr = left_proj.data_ptr()
    left_gate_ptr = left_gate.data_ptr()
    right_ptr = right_proj.data_ptr()
    right_gate_ptr = right_gate.data_ptr()
    left_out_ptr = left_out.data_ptr()
    right_out_ptr = right_out.data_ptr()

    # launch kernel
    mask_gate_fuse_kernel[grid](
        left_ptr, left_gate_ptr,
        right_ptr, right_gate_ptr,
        mask_f_ptr,
        left_out_ptr, right_out_ptr,
        B, N, hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE
    )

    # Now we have left_out and right_out ready.
    # Compute contraction using batch matmul
    # Permute to bring hidden dim to batch dim
    left_perm = left_out.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 2, 1)  # (B, H, N, N) - transpose N dims

    # Optionally use half precision for matmul
    # Use torch.float16 for speed:
    left_perm_fp16 = left_perm.to(torch.float16)
    right_perm_fp16 = right_perm.to(torch.float16)
    out_mat = torch.matmul(left_perm_fp16, right_perm_fp16)  # (B, H, N, N) in fp16
    out_mat = out_mat.to(torch.float32)  # back to fp32

    out = out_mat.permute(0, 2, 3, 1)  # (B, N, N, H)

    # Apply to_out_norm
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear
    out_flat = out_norm.reshape(-1, hidden_dim)
    output = torch.nn.functional.linear(out_flat, to_out_weight)  # (B*N*N, dim)
    output = output.view(B, N, N, dim)

    return output

TriMul forward (outgoing) implementation with a fused Triton kernel.

Algorithm:
1. LayerNorm the input tensor (shape [B, N, N, C]).
2. Linear projections to hidden dimension:
   - left_proj, right_proj : Linear(dim -> hidden_dim)
   - left_gate, right_gate, out_gate : Linear(dim -> hidden_dim) + sigmoid
3. Fuse mask, left/right gating into a single Triton kernel
   (left = left_proj * mask * left_gate, right = right_proj * mask * right_gate).
4. Compute the contraction out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched matrix multiplication: for each hidden slice
   out[..., d] = left[..., d] @ (right[..., d])ᵀ.
5. Apply LayerNorm on the hidden dimension, multiply by out_gate, and
   project back to the original dimension with a final linear layer.
All heavy tensor algebra (step 4) uses torch.matmul; the masking &
gating step (3) is performed by a custom Triton kernel.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def mask_gate_fuse_kernel(
    left_ptr,                # float32* left projection (input & output)
    left_gate_ptr,           # float32* left gate
    right_ptr,               # float32* right projection (input & output)
    right_gate_ptr,          # float32* right gate
    mask_ptr,                # float32* mask (flattened B*N*N)
    out_left_ptr,            # float32* left output (same as left_ptr, can alias)
    out_right_ptr,           # float32* right output (same as right_ptr, can alias)
    B, N, H,                 # int64 runtime constants
    BLOCK_SIZE: tl.constexpr # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Load data (masked loads return 0.0 outside the range)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Decode 4‑D indices from the flat offset
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    # d = tmp % H   # not needed for mask lookup

    # Mask is stored as a flat (B, N, N) tensor.
    mask_idx = b * (N * N) + i * N + j
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Apply mask and gates
    out_left = left * mask_val * left_gate
    out_right = right * mask_val * right_gate

    # Store results (in‑place writes are safe because we read before writing)
    tl.store(out_left_ptr + offs, out_left, mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (float or bool)
            - weights: dict of named weight tensors
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, C = input_tensor.shape
    assert C == dim, "Input dim mismatch with config"

    # -------------------- 1. LayerNorm --------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -------------------- 2. Linear projections & gates --------------------
    # Flatten the first three dims for a single matmul per projection
    x_flat = x.view(-1, dim)  # shape (B*N*N, dim)

    left_proj = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right_proj = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = torch.sigmoid(
        F.linear(x_flat, weights["left_gate.weight"])
    ).view(B, N, N, hidden_dim)

    right_gate = torch.sigmoid(
        F.linear(x_flat, weights["right_gate.weight"])
    ).view(B, N, N, hidden_dim)

    out_gate = torch.sigmoid(
        F.linear(x_flat, weights["out_gate.weight"])
    ).view(B, N, N, hidden_dim)

    # Ensure contiguous layout for the kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_float = mask.to(torch.float32).contiguous()
    mask_flat = mask_float.view(-1)  # shape (B*N*N,)

    # -------------------- 3. Fuse mask and gating (Triton) --------------------
    total_elements = B * N * N * hidden_dim
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    # In‑place update: left_proj/right_proj become the masked‑gated tensors
    mask_gate_fuse_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask_flat.data_ptr(),
        left_proj.data_ptr(),   # out pointer (can alias input)
        right_proj.data_ptr(),  # out pointer (can alias input)
        B, N, hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE
    )

    # -------------------- 4. Contraction via batched matmul --------------------
    # Bring hidden dim to the batch dimension for matmul
    left_perm = left_proj.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 2, 1)  # (B, H, N, N) = rightᵀ

    # Use float16 (or bfloat16) for the heavy matmul
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)
    contracted = torch.matmul(left_fp16, right_fp16)  # (B, H, N, N) in fp16
    contracted = contracted.to(torch.float32)        # back to fp32

    out = contracted.permute(0, 2, 3, 1)  # (B, N, N, H)

    # -------------------- 5. Post‑processing --------------------
    # LayerNorm on hidden dimension
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    out_flat = out_norm.view(-1, hidden_dim)
    output = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return output
