import torch
import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(inp_ptr, gate_ptr, mask_ptr, out_ptr,
                           B, N, H,
                           total_elems,
                           BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # compute indices (b,i,j,d)
    d = offs % H
    rem = offs // H
    j = rem % N
    rem = rem // N
    i = rem % N
    b = rem // N

    # compute offset into mask (no hidden dim)
    mask_off = b * N * N + i * N + j

    # load values
    x = tl.load(inp_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))

    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) operator.
    It performs LayerNorm, linear projections + gating, fused mask‑gate elementwise
    multiplication in a Triton kernel, a batched matrix multiplication via torch.matmul,
    followed by a second LayerNorm, gating, and linear projection back to the input dimension.
    """
    input_tensor, mask_tensor, weights, config = data

    # Determine shape
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Handle optional mask (nomask flag)
    if config.get("nomask", False):
        mask_tensor = torch.ones((B, N, N), dtype=input_tensor.dtype, device=input_tensor.device)
    # Ensure mask is float16 for kernel
    mask_fp16 = mask_tensor.to(torch.float16).contiguous()

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Cast to half for the heavy part
    x_fp16 = x_norm.to(torch.float16)

    # Linear projections (half precision)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)          # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # Fuse mask and gate with Triton
    total_elements = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    left_processed = torch.empty_like(left)
    right_processed = torch.empty_like(right)

    # Launch kernel for left
    apply_mask_gate_kernel[grid](
        left,
        left_gate,
        mask_fp16,
        left_processed,
        B, N, hidden_dim,
        total_elements,
        BLOCK=BLOCK,
    )
    # Launch kernel for right
    apply_mask_gate_kernel[grid](
        right,
        right_gate,
        mask_fp16,
        right_processed,
        B, N, hidden_dim,
        total_elements,
        BLOCK=BLOCK,
    )

    # Batched matrix multiplication across the shared N dimension
    # shape -> B, hidden_dim, N, N
    left_perm = left_processed.permute(0, 3, 1, 2).contiguous()
    right_perm = right_processed.permute(0, 3, 1, 2).contiguous()
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # B,N,N,hidden_dim

    # Second LayerNorm (half precision)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply output gate
    out_gated = out_norm * out_gate

    # Final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    final = torch.nn.functional.linear(out_gated, to_out_w)

    # Return float32 tensor
    return final.to(torch.float32)

    # first LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # linear projections & gating (half precision)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)          # [B,N,N,H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # ------------------------------------------------------------------
    # fuse mask + gate with Triton (out = proj * gate * mask)
    # ------------------------------------------------------------------
    total_elements = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    left_fused  = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # left side
    apply_mask_gate_kernel[grid](
        left,
        left_gate,
        mask_fp16,
        left_fused,
        B, N, hidden_dim,
        total_elements,
        BLOCK=BLOCK,
    )
    # right side
    apply_mask_gate_kernel[grid](
        right,
        right_gate,
        mask_fp16,
        right_fused,
        B, N, hidden_dim,
        total_elements,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # batched matmul: out_{i,j} = sum_k left_{i,k} * right_{j,k}
    # ------------------------------------------------------------------
    # shape -> (B, H, N, N)
    left_perm  = left_fused.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous()

    # matmul over the shared N dimension
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))   # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()                 # (B, N, N, H)

    # ------------------------------------------------------------------
    # second LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out_gated = out_norm * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    final = torch.nn.functional.linear(out_gated, to_out_w)  # (B, N, N, dim)

    # return float32 as required
    return final.to(torch.float32)
