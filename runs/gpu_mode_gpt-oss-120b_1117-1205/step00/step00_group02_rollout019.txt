@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start indices
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    b = pid_b

    # Compute offset for each tile
    # Pointer base offsets
    left_base = left_ptr + b * stride_left_b
    right_base = right_ptr + b * stride_right_b
    out_base = out_ptr + b * stride_out_b

    # Create offsets arrays
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = tl.arange(0, BLOCK_D)

    # Build masks for out-of-bound rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    for k_start in range(0, N, BLOCK_K):
        k_end = k_start + BLOCK_K
        cur_k = tl.minimum(offs_k, k_end - k_start - 1)  # Actually need mask for last block
        # Compute mask for k dimension
        mask_k = (k_start + offs_k) < N

        # Load left tile
        a_ptrs = left_base \
            + (offs_i[:, None, None] * stride_left_i) \
            + ((k_start + offs_k)[None, :, None] * stride_left_k) \
            + (offs_d[None, None, :] * stride_left_d)
        a = tl.load(a_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)  # shape (BM,BK,BD)

        # Load right tile transposed: (BK,BN,BD)
        b_ptrs = right_base \
            + ((k_start + offs_k)[:, None, None] * stride_right_k) \
            + (offs_j[None, :, None] * stride_right_j) \
            + (offs_d[None, None, :] * stride_right_d)
        b = tl.load(b_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None], other=0.0)  # shape (BK,BN,BD)

        # Compute dot for each d
        for d in range(0, BLOCK_D):
            a_slice = a[:, :, d]  # (BM,BK)
            b_slice = b[:, :, d]  # (BK,BN)
            acc[:, :, d] += tl.dot(a_slice, b_slice)

    # Write out
    out_ptrs = out_base \
        + (offs_i[:, None, None] * stride_out_i) \
        + (offs_j[None, :, None] * stride_out_j) \
        + (offs_d[None, None, :] * stride_out_d)
    # Write only within bounds
    mask_out = mask_i[:, None, None] & mask_j[None, :, None]
    tl.store(out_ptrs, acc, mask=mask_out)
```

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if provided and not None
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).type_as(x)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Triton contraction
    out = torch.empty_like(left)  # shape B,N,N,hidden_dim

    B, N, _, H = left.shape
    # Cast to float16 for compute? Not necessary.
    # Get strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Kernel block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32  # must be <= hidden_dim, adjust later if needed

    # Adjust BLOCK_D to not exceed hidden_dim
    if hidden_dim < BLOCK_D:
        BLOCK_D = hidden_dim

    grid = (B, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )
    # Now apply final norm, gate, linear

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate

"""
TriMul (outgoing) implementation with a fused Triton kernel for the
pairwise contraction:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

The surrounding LayerNorms, linear projections, and gate sigmoids are
performed in PyTorch.  The heavy O(N³·C) contraction is executed by a
custom Triton kernel that tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N)
and the channel axis (BLOCK_D).  Each program computes a tile of the
output matrix for a batch element, looping over the reduction axis k in
blocks (BLOCK_K).  For every channel sub‑block the kernel uses
`tl.dot` (i.e. a small matmul) to accumulate the result, providing a
significant speed‑up over a naïve implementation.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                     # *float32  [B, N, N, C]
    right_ptr,                    # *float32  [B, N, N, C]
    out_ptr,                      # *float32  [B, N, N, C]
    B, N, C,                      # shape infos
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    """Compute the outgoing triangle multiplicative update.

    Grid: (B, ceil(N/BLOCK_M), ceil(N/BLOCK_N)).
    Each program processes a tile (i_range, j_range) for a single batch.
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # ------------------------------------------------------------------
    #  Offsets for the tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # 1‑D index ranges
    i_idx = i_base + tl.arange(0, BLOCK_M)
    j_idx = j_base + tl.arange(0, BLOCK_N)
    k_idx = tl.arange(0, BLOCK_K)
    c_idx = tl.arange(0, BLOCK_C)

    # Masks for out‑of‑bounds
    mask_i = i_idx < N
    mask_j = j_idx < N
    mask_c = (pid_b * 0) + (c_idx < C)  # placeholder; actual C‑mask applied later

    # Pointers to the start of the current batch
    left_batch_ptr  = left_ptr  + pid_b * stride_left_b
    right_batch_ptr = right_ptr + pid_b * stride_right_b
    out_batch_ptr   = out_ptr   + pid_b * stride_out_b

    # ------------------------------------------------------------------
    #  Accumulator for a tile (BLOCK_M, BLOCK_N, BLOCK_C)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_C), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over the reduction axis k
    # ------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        # k_range for this iteration
        cur_k = k_base + k_idx
        mask_k = cur_k < N

        # --------------------------------------------------------------
        # Load left tile: shape (BLOCK_M, BLOCK_K, BLOCK_C)
        # --------------------------------------------------------------
        left_ptrs = left_batch_ptr \
            + (i_idx[:, None, None] * stride_left_i) \
            + (cur_k[None, :, None] * stride_left_k) \
            + (c_idx[None, None, :] * stride_left_c)
        left_mask = mask_i[:, None, None] & mask_k[None, :, None]
        a = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # --------------------------------------------------------------
        # Load right tile (transposed version):
        #   we need right[b, j, k, c] accessed as (k, j, c)
        #   shape (BLOCK_K, BLOCK_N, BLOCK_C)
        # --------------------------------------------------------------
        right_ptrs = right_batch_ptr \
            + (cur_k[:, None, None] * stride_right_k) \
            + (j_idx[None, :, None] * stride_right_j) \
            + (c_idx[None, None, :] * stride_right_c)
        right_mask = mask_k[:, None, None] & mask_j[None, :, None]
        b = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # --------------------------------------------------------------
        # Accumulate: for each channel slice c do a dot-product
        # --------------------------------------------------------------
        for c in range(0, BLOCK_C):
            a_slice = a[:, :, c]      # (BLOCK_M, BLOCK_K)
            b_slice = b[:, :, c]      # (BLOCK_K, BLOCK_N)
            acc[:, :, c] += tl.dot(a_slice, b_slice)

    # ------------------------------------------------------------------
    #  Write the output tile
    # ------------------------------------------------------------------
    # Apply channel mask for the tail case
    mask_c = (c_idx + 0) < C  # true for valid channels
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :]

    out_ptrs = out_batch_ptr \
        + (i_idx[:, None, None] * stride_out_i) \
        + (j_idx[None, :, None] * stride_out_j) \
        + (c_idx[None, None, :] * stride_out_c)

    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor [B, N, N, dim]
              - mask: torch.Tensor [B, N, N] (may be all ones)
              - weights: dict of pre‑loaded weight tensors
              - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    #  LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    #  Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    #  Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(input_tensor.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    #  Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #  Triton contraction
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)  # will hold [B, N, N, hidden_dim]

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_c = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_c = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_c = out.stride()

    # Tile sizes – chosen to fit in registers for typical GPUs
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_C = 8                     # channel sub‑tile (must divide hidden_dim reasonably)

    # Adjust BLOCK_C for small hidden dimensions
    if hidden_dim < BLOCK_C:
        BLOCK_C = hidden_dim

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_c,
        stride_r_b, stride_r_j, stride_r_k, stride_r_c,
        stride_o_b, stride_o_i, stride_o_j, stride_o_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_C=BLOCK_C,
    )

    # ------------------------------------------------------------------
    #  Final LayerNorm, gating and output linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
