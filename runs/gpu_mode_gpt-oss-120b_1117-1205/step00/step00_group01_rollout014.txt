import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_batch, stride_row, stride_col,
    M, N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ right^T for batch of matrices.
    left/right have shape (B*H, M, N). The kernel treats the first dimension
    as independent batches and performs a GEMM for each batch with accumulation
    in fp32. Supports M,N up to ~1024.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers
        left_tile_ptr = left_ptr + pid_batch * stride_batch + offs_m[:, None] * stride_row + offs_k[None, :] * stride_col
        right_tile_ptr = right_ptr + pid_batch * stride_batch + offs_k[:, None] * stride_row + offs_n[None, :] * stride_col

        # load tiles (fp16) and promote to fp32
        a = tl.load(left_tile_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(right_tile_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    out_tile_ptr = out_ptr + pid_batch * stride_batch + offs_m[:, None] * stride_row + offs_n[None, :] * stride_col
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.
    Steps:
    1. Layer‑norm the input tensor.
    2. Linear projections + mask.
    3. Gating (sigmoid of linear layers) applied element‑wise.
    4. Core O(N³) operation (`out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`)
       implemented as a batched GEMM with a custom Triton kernel.
    5. Post‑norm, gating and final linear projection back to `dim`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on the channel dimension -----
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype=device),
        bias=weights["norm.bias"].to(dtype=device),
        eps=eps,
    )  # shape [B,N,N,dim]

    # ----- Linear projections -----
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(dtype=device))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(dtype=device))

    # ----- Apply mask -----
    # mask shape [B,N,N] -> [B,N,N,1] for broadcasting
    mask_exp = mask.to(dtype=device).unsqueeze(-1)
    left = left * mask_exp
    right = right * mask_exp

    # ----- Gating -----
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"].to(dtype=device))
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"].to(dtype=device))
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"].to(dtype=device))
    )
    left = left * left_gate
    right = right * right_gate

    # ----- Prepare tensors for the Triton GEMM -----
    # Convert to fp16 for the compute kernel to exploit tensor cores
    left_fp16 = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # [B, H, N, N]
    right_fp16 = right.permute(0, 3, 2, 1).contiguous().to(torch.float16)  # [B, H, N, N] (transposed on the last two dims)

    B, N, _, _ = left.shape  # N already defined, B batch size
    # flatten batch*hidden dimension
    batch_h = B * hidden_dim
    left_flat = left_fp16.view(batch_h, N, N)
    right_flat = right_fp16.view(batch_h, N, N)

    # output buffer (fp32)
    out_fp32 = torch.empty((batch_h, N, N), dtype=torch.float32, device=device)

    # ----- Triton kernel launch -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_h, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    stride_batch = left_flat.stride(0)  # N*N
    stride_row = left_flat.stride(1)    # N
    stride_col = left_flat.stride(2)    # 1

    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_fp32,
        stride_batch,
        stride_row,
        stride_col,
        N,          # M (rows)
        N,          # N (cols)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Reshape back to original layout -----
    out = (
        out_fp32.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # ----- Post‑norm, gating and final projection -----
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype=device),
        bias=weights["to_out_norm.bias"].to(dtype=device),
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(dtype=device))

    return out

    """
