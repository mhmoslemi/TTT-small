```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, C,
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # batch_hd index
    b_hd = tl.program_id(2) # range 0 .. B*C-1
    # decode batch and hidden index
    batch = b_hd // C
    c = b_hd % C

    # get block coordinates
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute offsets for each block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate result
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block: shape (BLOCK_M, BLOCK_K)
        # left_ptr offset: (batch, i, k, c)
        left_ptrs = left_ptr + batch * stride_left_b + c * stride_left_c + \
                    offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right block: shape (BLOCK_N, BLOCK_K)
        # we need right transposed: we need right[j,k,c] element with k as second axis.
        # For matmul left(i,k)*right_T(j,k) we need right[j,k,c], but we want to multiply left * right_T.
        # In matmul, we compute sum_k left[i,k] * right_T[j,k] = sum_k left[i,k] * right[k,j]? Wait right_T[j,k] is right[k,j].
        # But we want sum_k left[i,k] * right[j,k] = left[i,k] * right_T[j,k]? Actually right_T[j,k] = right[k,j], not right[j,k].
        # So we should use right without transpose (j,k) -> shape (BLOCK_N, BLOCK_K)
        # To achieve sum_k left[i,k] * right[j,k], we can load right block with offsets (j, k) and treat it as (BLOCK_N, BLOCK_K) and multiply elementwise, summing over k dimension.
        # In matrix multiplication formula: C[i,j] = sum_k A[i,k] * B[k,j]. We have B[k,j] rather than B[j,k]. However, we could load right as B_T = right.transpose(-2, -1), but then shape (N,N) and would give B_T[k,j] = right[j,k].
        # Let's compute left * right_T: left[i,k] * right_T[k,j] = left[i,k] * right[j,k] which matches our formula! Because right_T[k,j] = right[j,k].
        # So we need right_T block: we need elements where first index is k (row) and second index is j (col). That is, we need to load the transpose of right: stride for first dimension = stride_right_k (which originally is for k), second dimension = stride_right_j.
        # So we set base pointer for right_T: right_ptr + batch * stride_right_b + c * stride_right_c + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        # This yields shape (BLOCK_K, BLOCK_N) which after transpose is (BLOCK_N, BLOCK_K). But we can load as shape (BLOCK_K, BLOCK_N) and then compute dot accordingly: we need left (BLOCK_M, BLOCK_K) dot right_T (BLOCK_K, BLOCK_N) -> result (BLOCK_M, BLOCK_N). So we load right_T as shape (BLOCK_K, BLOCK_N).
        right_ptrs = right_ptr + batch * stride_right_b + c * stride_right_c + \
                     offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right_T = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute product: we want sum over k: left[i,k] * right_T[k,j]
        # Use tl.dot over k dimension for each i/j. But tl.dot expects two vectors. We'll compute using tl.dot for each pair maybe:
        # But we can use classic matmul approach: accumulate via tl.dot for each block
        acc += tl.dot(left, right_T)

    # store result
    out_ptrs = out_ptr + batch * stride_out_b + c * stride_out_c + \
               offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16) if acc.dtype == tl.float16 else acc, mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Convert to float32 for calculations (makes any half issues simpler)
    x = input_tensor.to(torch.float32)

    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    eps = 1e-5 # typical for LayerNorm

    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections left and right
    # Left projection
    left_proj_weight = weights['left_proj.weight']  # shape (hidden_dim, dim)
    # Perform linear: out = x @ weight.T? PyTorch Linear does x @ weight.t() + bias
    # Since bias=False, we can compute torch.einsum('... d, hd d -> ... h')
    # Actually weight shape is (hidden_dim, dim) -> we can compute left = torch.einsum('... d, hd d -> ... h')
    left = torch.einsum('... d, hd d -> ... h', x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    # Similarly right
    right_proj_weight = weights['right_proj.weight']
    right = torch.einsum('... d, hd d -> ... h', x, right_proj_weight)

    # Masking: mask shape [B,N,N], unsqueeze -1 to broadcast to hidden_dim
    if mask is not None:
        mask = mask.to(torch.float32).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates: left_gate, right_gate, out_gate
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']

    left_gate = torch.sigmoid(torch.einsum('... d, hd d -> ... h', x, left_gate_weight))
    right_gate = torch.sigmoid(torch.einsum('... d, hd d -> ... h', x, right_gate_weight))
    out_gate = torch.sigmoid(torch.einsum('... d, hd d -> ... h', x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Now compute core contraction: out = left @ right^T per hidden dim.
    # Implement with Triton
    # Flatten batch and hidden dims
    B = bs
    H = hidden_dim
    # Reshape left and right to shape (B*H, N, N)
    left_2d = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_2d = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    # Need right_T = right_2d.transpose(1,2) for multiplication.

    # Allocate output tensor
    out_2d = torch.empty_like(left_2d)  # shape (B*H, N, N)

    # Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Determine grid: (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B*H)
    tri_mul_kernel[grid](
        left_2d, right_2d, out_2d,
        N, H,
        left_2d.stride(0), left_2d.stride(1), left_2d.stride(2), left_2d.stride(3),
        right_2d.stride(0), right_2d.stride(1), right_2d.stride(2), right_2d.stride(3),
        out_2d.stride(0), out_2d.stride(1), out_2d.stride(2), out_2d.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Reshape back to original order
    out = out_2d.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # Now out is shape [B,N,N,hidden_dim] (float32). Apply to_out_norm, out_gate, to_out.

    # to_out_norm
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    out = out * out_gate

    # to_out linear
    to_out_weight = weights['to_out.weight']  # shape (dim, hidden_dim)
    out = torch.einsum('... h, d h -> ... d', out, to_out_weight)  # [B,N,N,dim]

    return out

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(2)  # batch * hidden index
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute global offsets for this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K)
        left_ptrs = left_ptr + bh * stride_left_batch + \
                    offs_i[:, None] * stride_left_i + \
                    offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right transposed block: (K, N)
        # right has shape (batch, i, j) -> we need right_T with rows = k (original j), cols = j (original i)
        right_ptrs = right_ptr + bh * stride_right_batch + \
                     offs_k[:, None] * stride_right_j + \
                     offs_j[None, :] * stride_right_i
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        acc += tl.dot(left, right_T)  # (M,N)

    # store result
    out_ptrs = out_ptr + bh * stride_out_batch + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])
```

```python
def custom_kernel(data):
    """
    TriMul forward pass with Triton-accelerated O(N^3) contraction.
    Steps:
      1. LayerNorm the input.
      2. Linear projections (left/right) and gate values.
      3. Optional masking.
      4. Apply gates.
      5. Compute out = left @ right.T per hidden channel using a Triton kernel.
      6. LayerNorm + output gate.
      7. Final linear projection back to input dimension.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    # Cast everything to float32 for numerical stability.
    x = input_tensor.to(torch.float32)

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on last dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Optional mask (mask: [B, N, N])
    if not nomask:
        # ensure mask is on same device and dtype
        mask_f = mask.to(torch.float32).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # Gate values (sigmoid)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Triton contraction: out = left @ right.T per hidden channel
    # -------------------------------------------------
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim

    # Prepare tensors for batched GEMM: merge batch and hidden_dim
    # left_2d shape (B*hidden_dim, N, N)
    left_2d = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_2d = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_2d = torch.empty_like(left_2d)   # allocate output buffer

    # Strides for kernel (int64)
    stride_left_batch = left_2d.stride(0)
    stride_left_i = left_2d.stride(1)
    stride_left_k = left_2d.stride(2)

    stride_right_batch = right_2d.stride(0)
    stride_right_i = right_2d.stride(1)
    stride_right_j = right_2d.stride(2)

    stride_out_batch = out_2d.stride(0)
    stride_out_i = out_2d.stride(1)
    stride_out_j = out_2d.stride(2)

    # Block sizes (tuneable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M),   # blocks over rows (i)
            triton.cdiv(N, BLOCK_N),   # blocks over cols (j)
            B * hidden_dim)            # batch * hidden dimension

    tri_mul_kernel[grid](
        left_2d, right_2d, out_2d,
        N,
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_i, stride_right_j,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_2d.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm on hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dim
    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden_dim)
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```

"""
TriMul forward pass with a Triton kernel for the O(N³) contraction.

Algorithm:
1. Layer‑norm the input tensor over the feature dimension.
2. Linear projections `left_proj` / `right_proj` (no bias) and three sigmoid gates.
3. Optional masking of the projected tensors.
4. Apply element‑wise gates to the projected tensors.
5. Compute the core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiply `left @ right.T`
   for each hidden channel.  This is performed by a Triton kernel.
6. Second Layer‑norm over the hidden dimension, multiply by the output gate.
7. Final linear projection back to the original feature dimension.
All heavy arithmetic is carried out in float32; the kernel works on
contiguous tensors reshaped to (batch*hidden, N, N).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # batch*hidden index
    bh = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # global offsets of the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction (k) dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (M, K)
        left_ptrs = left_ptr + bh * stride_left_batch + \
                    offs_i[:, None] * stride_left_i + \
                    offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right transposed block: shape (K, N)
        # we need right_T[k, j] = right[j, k]
        right_ptrs = right_ptr + bh * stride_right_batch + \
                     offs_k[:, None] * stride_right_j + \
                     offs_j[None, :] * stride_right_i
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        acc += tl.dot(left, right_T)   # (M, N)

    # write the tile back
    out_ptrs = out_ptr + bh * stride_out_batch + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward computation of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] torch.Tensor
            - mask:          [B, N, N] torch.Tensor (bool or 0/1)
            - weights: dict of model weights (see reference implementation)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # Cast everything to float32 for stability
    x = input_tensor.to(dtype)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- Load and cast weights ------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)

    left_proj_weight = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)

    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)

    to_out_weight = weights["to_out.weight"].to(dtype)   # shape (dim, hidden_dim)

    # ----- 1. LayerNorm over the feature dimension -----------------------------
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ----- 2. Linear projections ------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- 3. Optional mask -----------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----- 4. Gates (sigmoid) ---------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- 5. Triton‑accelerated contraction ------------------------------------
    B, N, _, _ = left.shape

    # Merge batch and hidden dimensions to get independent GEMMs
    left_2d = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_2d = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_2d = torch.empty_like(left_2d)

    # Strides for kernel (int64)
    stride_left_batch = left_2d.stride(0)
    stride_left_i = left_2d.stride(1)
    stride_left_k = left_2d.stride(2)

    stride_right_batch = right_2d.stride(0)
    stride_right_i = right_2d.stride(1)
    stride_right_j = right_2d.stride(2)

    stride_out_batch = out_2d.stride(0)
    stride_out_i = out_2d.stride(1)
    stride_out_j = out_2d.stride(2)

    # Tuneable block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim                    # batch * hidden
    )

    tri_mul_kernel[grid](
        left_2d, right_2d, out_2d,
        N,
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_i, stride_right_j,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_2d.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- 6. Second LayerNorm and output gate ---------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate

    # ----- 7. Final linear projection back to input dim -------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
