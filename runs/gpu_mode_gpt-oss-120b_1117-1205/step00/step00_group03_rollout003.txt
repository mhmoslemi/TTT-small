left: shape [B,N,N,H] where H=hidden_dim.
right: shape [B,N,N,H].

@triton.jit
def batch_matmul_kernel( a_ptr, b_ptr, c_ptr,
                         M, N, K,
                         stride_aa, stride_ai, stride_ak,
                         stride_bb, stride_bk, stride_bj,
                         stride_cc, stride_ci, stride_cj,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0) # batch
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        # Compute k size
        cur_k = tl.select(k + offs_k < K, offs_k, 0) # but we can just mask in load
        # compute pointer offsets
        a_ptrs = a_ptr + pid_b * stride_aa + offs_m[:, None] * stride_ai + (k + offs_k)[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_bb + (k + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bj

        # Load with mask
        a = tl.load(a_ptrs, mask=(mask_m[:, None] & ((k + offs_k) < K)), other=0.0)
        b = tl.load(b_ptrs, mask=(((k + offs_k) < K)[:, None] & mask_n[None, :]), other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + pid_b * stride_cc + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    mask = mask_m[:, None] & mask_n[None, :]

"""
TriMul forward kernel (outgoing version) with a custom Triton GEMM.

The heavy part of the TriMul operation is the contraction:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

We implement this as a batched matrix multiplication using Triton:
    A = left  -> (batch*hidden, N, N)   (M=N, K=N)
    B = right.T -> (batch*hidden, N, N) (K=N, N)
    C = A @ B  -> (batch*hidden, N, N)

The rest of the module (LayerNorms, linear projections, gating) is done
with PyTorch for simplicity. The kernel works on half‑precision data
(most memory‑intensive tensors are cast to fp16) while accumulating in
float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aa, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bj,
    stride_cc, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B  where A[b,m,k], B[b,k,n] -> C[b,m,n]"""
    pid_batch = tl.program_id(0)   # batch index (combined batch*hidden)
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for output tiles
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = tl.arange(0, BLOCK_K) + k
        mask_k = offs_k < K

        # pointers for A and B
        a_ptrs = a_ptr + pid_batch * stride_aa + \
                 offs_m[:, None] * stride_ai + \
                 offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_bb + \
                 offs_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bj

        # load with masks (A: fp16, B: fp16)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product (fp16 inputs, fp32 accumulation)
        acc += tl.dot(a, b)

    # write result back in fp16
    c_ptrs = c_ptr + pid_batch * stride_cc + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    high‑cost contraction. Returns a float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------
    # Load model parameters (all assumed to be float32)
    # ------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------
    # 3) Optional masking
    # ------------------------------------------------------------
    if not config.get("nomask", False):
        # mask shape [B, N, N] -> [B, N, N, 1] broadcast
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 5) Contraction via Triton batched GEMM (fp16 for memory efficiency)
    # ------------------------------------------------------------
    # Cast to half for the kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange tensors so that each (batch, hidden) pair is a separate GEMM
    # A: (batch*H, N, N)   (M=N, K=N)
    # B: (batch*H, N, N)   (K=N, N)  -> we need the K dimension first
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (k, j)

    batch = bs * hidden_dim
    A = left_perm.view(batch, N, N)      # (batch, M, K)
    B = right_perm.view(batch, N, N)     # (batch, K, N)

    C = torch.empty((batch, N, N), dtype=torch.float16, device=device)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batch_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = C.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out = out.to(torch.float32)

    # ------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # shape [B, N, N, dim]

    return out
```
