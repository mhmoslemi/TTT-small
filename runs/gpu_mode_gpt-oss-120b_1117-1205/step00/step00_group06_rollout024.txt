@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # tile row
    pid_n = tl.program_id(2)  # tile col

    # compute offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)

    # Compute pointers offsets for the batch
    a_batch_ptr = a_ptr + pid_b * batch_stride_a
    b_batch_ptr = b_ptr + pid_b * batch_stride_b
    c_batch_ptr = c_ptr + pid_b * batch_stride_c

    # Load a tile: shape (BLOCK_M, BLOCK_K)
    a_ptrs = a_batch_ptr + off_m[:, None] * stride_am + off_k[None, :] * stride_ak
    # However if off_m may exceed M, we mask later.

    # For B, we treat as B_T: row = k (off_k) column = off_n
    b_ptrs = b_batch_ptr + off_k[:, None] * stride_bk + off_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(off_m < M) & (off_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(off_k < K) & (off_n < N), other=0.0)
        acc += tl.dot(a, b)

        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
        off_k += BLOCK_K

    # Write output
    c_ptrs = c_batch_ptr + off_m[:, None] * stride_cm + off_n[None, :] * stride_cn
    mask = (off_m[:, None] < M) & (off_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16) if acc.dtype == tl.float16 else acc, mask=mask)
```

    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # Linear projections: left_proj, right_proj
    # Use torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    mask = mask.unsqueeze(-1).to(x.dtype)  # shape [B,N,N,1], broadcast
    left = left * mask
    right = right * mask

    # Gate computations:
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert left and right to shape suitable for kernel
    # Permute to bring hidden_dim as second dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, hidden, N, _ = left_perm.shape
    batchM = B * hidden

    # reshape to (batchM, N, N)
    left_mat = left_perm.reshape(batchM, N, N)
    right_mat = right_perm.reshape(batchM, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)  # same dtype float32

    # Compute strides
    a_ptr = left_mat
    b_ptr = right_mat
    c_ptr = out_mat

    stride_batch_a = a_ptr.stride(0)
    stride_am = a_ptr.stride(1)
    stride_ak = a_ptr.stride(2)
    stride_batch_b = b_ptr.stride(0)
    stride_bk = b_ptr.stride(2) # original column stride = 1
    stride_bn = b_ptr.stride(1) # original row stride = N
    stride_batch_c = c_ptr.stride(0)
    stride_cm = c_ptr.stride(1)
    stride_cn = c_ptr.stride(2)

    # Choose block sizes based on N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # Slightly adjust for larger N: ensure BLOCK_K <= K
    # Not necessary.

    grid = (batchM, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    trimul_gemm_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        stride_batch_a, stride_am, stride_ak,
        stride_batch_b, stride_bk, stride_bn,
        stride_batch_c, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back
    out = out_mat.reshape(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # to_out_norm
    # LayerNorm over hidden dimension (last dim)
    eps_out = 1e-5
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps_out)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # Multiply by out_gate
    out_gated = out_norm * out_gate

    # Final linear
    out_final = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return out_final

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm:
1. Apply LayerNorm on the input tensor (last dimension) using provided weight/bias.
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left_proj, right_proj) and compute three gating projections (left_gate,
   right_gate, out_gate) followed by sigmoid activations.
3. Apply the mask (if any) and element‑wise gate the left/right projections.
4. The core contraction is:
       out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   which is a batched matrix multiplication C = A @ Bᵀ for each hidden
   channel d.  This is computed with a Triton GEMM kernel that tiles the
   i‑ and j‑dimensions and sums over k.
5. Reshape the result back, apply a second LayerNorm (to_out_norm) and the
   out_gate, and finish with a linear projection back to the original
   feature dimension.
All operations are performed in float32; the Triton kernel accumulates in
float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_am, stride_ak,
    stride_batch_b, stride_bk, stride_bn,
    stride_batch_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A: (batch, M, K)   (row‑major)
    B: (batch, N, K)   (row‑major, will be treated as transposed inside)
    C: (batch, M, N)
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_base = a_ptr + pid_batch * stride_batch_a
    b_base = b_ptr + pid_batch * stride_batch_b
    c_base = c_ptr + pid_batch * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers for this tile
        a_ptrs = a_base + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        b_ptrs = b_base + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load with bounds checking
        mask_a = (offs_m[:, None] < M) & (k_offset[None, :] < K)
        mask_b = (k_offset[:, None] < K) & (offs_n[None, :] < N)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the TriMul (outgoing) module with a Triton‑accelerated contraction.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] float32
            - mask:         [B, N, N] (bool or float) – will be broadcasted
            - weights: dict with all linear/layer‑norm parameters
            - config:  {"dim": int, "hidden_dim": int}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on input ----------
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ---------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # ---------- Gating ----------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for Triton GEMM ----------
    # Bring hidden_dim to second axis and make them contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batchM = B * H  # total number of independent GEMMs

    # Collapse batch & hidden dim
    left_mat = left_perm.reshape(batchM, N, N)
    right_mat = right_perm.reshape(batchM, N, N)

    # Allocate output buffer
    out_mat = torch.empty_like(left_mat)

    # Strides (in elements)
    stride_batch_a = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_batch_b = right_mat.stride(0)
    # For Bᵀ we treat rows = k, cols = j.
    stride_bk = right_mat.stride(2)   # original column stride (==1)
    stride_bn = right_mat.stride(1)   # original row stride (==N)

    stride_batch_c = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batchM,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        stride_batch_a, stride_am, stride_ak,
        stride_batch_b, stride_bk, stride_bn,
        stride_batch_c, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ---------- Second LayerNorm ----------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ---------- Apply out gate ----------
    out_gated = out_norm * out_gate

    # ---------- Final linear projection ----------
    output = F.linear(out_gated, weights["to_out.weight"])

    return output
