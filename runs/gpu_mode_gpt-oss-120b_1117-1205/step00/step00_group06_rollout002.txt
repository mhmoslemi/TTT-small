def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          stride_ah, stride_ak, stride_ab,  # but we need proper strides.
                          stride_bk, stride_bn, stride_bb,
                          stride_ch, stride_cm, stride_cn,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute pointers for A and B
    # A: (batch, M, K)
    # B: (batch, N, K) but we need B^T: shape (batch, K, N) to align with k dimension.

    # We will iterate over K in blocks

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A[batch, offs_m, offs_k] -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch * stride_ab + offs_m[:, None] * stride_ah + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)

        # Load B[batch, offs_n, offs_k] but we need B_trans: B[batch, offs_k, offs_n]
        b_ptrs = B_ptr + batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)
        # Then dot
        acc += tl.dot(a, b)

    # Write to C[batch, offs_m, offs_n]
    c_ptrs = C_ptr + batch * stride_cb + offs_m[:, None] * stride_ch + offs_n[None, :] * stride_cn

@triton.jit
def tri_mul_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          stride_a_batch, stride_a_row, stride_a_col,
                          stride_b_batch, stride_b_row, stride_b_col,
                          stride_c_batch, stride_c_row, stride_c_col,
                          M, N, K,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        off_k = k * BLOCK_K
        offs_k = off_k + tl.arange(0, BLOCK_K)  # shape (BLOCK_K,)
        mask_k = offs_k < K

        # load A[batch, offs_m, offs_k] (M x K)
        a_ptrs = A_ptr + batch * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B[batch, offs_k, offs_n] -> shape (K x N) (transposed)
        b_ptrs = B_ptr + batch * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = C_ptr + batch * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul "outgoing" operator accelerated with Triton.

Algorithm:
1. Layer‑normalize the input tensor (dim‑wise).
2. Project the normalized tensor to a hidden dimension (left/right) and compute
   element‑wise sigmoid gates (left_gate/right_gate/out_gate) using linear layers.
3. Apply the pair‑wise mask and gates to the projected tensors.
4. For each hidden channel, compute the matrix product
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM of shape (B*H, N, N) × (B*H, N, N)^T.
   This is implemented in a Triton kernel that iterates over K‑tiles.
5. Reshape the result back to (B, N, N, H), apply a LayerNorm over H,
   multiply by the out‑gate, and finally a linear projection back to dim.
All tensor computations are performed in float16 inside the kernel for speed,
with final results cast back to float32.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the batched matrix multiplication (A @ B^T) over
# the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A[batch, m, k]  (row‑major)
        a_ptrs = A_ptr + batch_id * stride_a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B[batch, k, n]  (i.e. B^T)
        b_ptrs = B_ptr + batch_id * stride_b_batch \
                 + offs_k[:, None] * stride_b_row \
                 + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot: (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = C_ptr + batch_id * stride_c_batch \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict of weight tensors
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Extract weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------
    # Apply mask (if any) and gates
    # ------------------------------------------------------------------
    if mask is not None:
        # broadcast mask to hidden dimension
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy O(N³) GEMM
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape   # H == hidden_dim

    # Reorder to (B, H, N, N) and flatten batch*H -> matrix per hidden channel
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch * hidden
        triton.cdiv(N, BLOCK_M),               # rows
        triton.cdiv(N, BLOCK_N),               # cols
    )

    # Strides in element counts (not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)
    stride_a_col = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row = right_mat.stride(1)
    stride_b_col = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)
    stride_c_col = out_mat.stride(2)

    # ------------------------------------------------------------------
    # Launch the kernel
    # ------------------------------------------------------------------
    tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H] (float16) then to float32
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # Final LayerNorm over hidden dimension, gating, and linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate  # element‑wise gate

    # Linear back to original dim
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
