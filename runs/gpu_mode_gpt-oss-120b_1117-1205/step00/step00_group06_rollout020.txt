@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ak, stride_ak2,
                          stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          batch_stride_A, batch_stride_B, batch_stride_C,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    a_ptrs = A_ptr + batch_id * batch_stride_A + (pid_m * BLOCK_M)[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + batch_id * batch_stride_B + tl.arange(0, BLOCK_K)[:, None] * stride_bk + (pid_n * BLOCK_N)[None, :] * stride_bn
    c_ptr = C_ptr + batch_id * batch_stride_C + (pid_m * BLOCK_M) * stride_cm + (pid_n * BLOCK_N) * stride_cn

    # Actually better to compute pointer offsets more explicitly.

    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)

    # apply masks to not exceed M,N,K
    mask_m = off_m < M
    mask_n = off_n < N

    a_ptrs = A_ptr + batch_id * batch_stride_A + off_m[:, None] * stride_am + off_k[None, :] * stride_ak
    b_ptrs = B_ptr + batch_id * batch_stride_B + off_k[:, None] * stride_bk + off_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (off_k < K - k * BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(off_k[:, None] < K - k * BLOCK_K) & mask_n[None, :], other=0.0)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c = acc.to(tl.float16) if out dtype is float16 else acc.to(tl.float32) ??? We need float32 final output, but may compute in float32.

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_A, batch_stride_B, batch_stride_C,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ B where A and B are [batch, M, K] and [batch, K, N]"""
    # Program IDs
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A and B for this batch
    a_ptr = A_ptr + batch_id * batch_stride_A
    b_ptr = B_ptr + batch_id * batch_stride_B
    # c pointer
    c_ptr = C_ptr + batch_id * batch_stride_C

    # Offsets for K dimension within loop
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for current K block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B tile (note B is [batch, K, N])
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Compute
        acc += tl.dot(a, b)
        # No need to increment pointers because we recompute offsets each iteration

    # Store result
    c = acc.to(tl.float32)

    # Compute write address
    c_ptr = c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_A, batch_stride_B, batch_stride_C,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ B
    A: [batch, M, K], B: [batch, K, N], C: [batch, M, N]
    All tensors are stored in row-major order.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute threadblock start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_ptr = A_ptr + batch_id * batch_stride_A
    b_batch_ptr = B_ptr + batch_id * batch_stride_B
    c_batch_ptr = C_ptr + batch_id * batch_stride_C

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16
        )
        # Load B tile [BLOCK_K, BLOCK_N]
        b = tl.load(
            b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16
        )
        acc += tl.dot(a, b)

    # Write output tile
    c_ptr = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operator.
    It fuses the heavy O(N^3) einsum into a batched GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------
    # Load weights to device and dtype
    # -----------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    to_out_weight = weights["to_out.weight"].to(device)

    # -----------------------------
    # LayerNorm over the channel dimension
    # -----------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )
    # Cast to float16 for the heavy path
    x = x_norm.to(torch.float16)

    # -----------------------------
    # Linear projections
    # -----------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------
    # Apply mask (if provided)
    # -----------------------------
    if mask is None:
        mask_f = torch.ones_like(input_tensor[..., 0], dtype=x.dtype, device=device)
    else:
        mask_f = mask.to(x.dtype)
    mask_f = mask_f.unsqueeze(-1)   # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # -----------------------------
    # Gating
    # -----------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid().to(x.dtype)
    right_gate = F.linear(x, right_gate_weight).sigmoid().to(x.dtype)
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()   # keep float32 for later

    left = left * left_gate
    right = right * right_gate

    # -----------------------------
    # Heavy O(N^3) part: batched GEMM (left @ right^T)
    # -----------------------------
    # Transpose right on the two sequence axes so that we can do a regular matmul
    right_T = right.permute(0, 2, 1, 3).contiguous()   # [B, N, N, hidden_dim] -> (b, k, j, d)

    # Move hidden dimension to batch dimension
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_T.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]

    total_batch = B * hidden_dim
    left_mat = left_perm.view(total_batch, N, N)   # [batch*hd, N, N] dtype=half
    right_mat = right_perm.view(total_batch, N, N)

    # Allocate output (float32)
    out_mat = torch.empty((total_batch, N, N), dtype=torch.float32, device=device)

    # Strides (in elements)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    batch_stride_A = left_mat.stride(0)

    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    batch_stride_B = right_mat.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    batch_stride_C = out_mat.stride(0)

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_A, batch_stride_B, batch_stride_C,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # -----------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # -----------------------------
    # Output projection
    # -----------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # final shape [B, N, N, dim]

    return output

"""
Triton‑accelerated outgoing TriMul.

The expensive O(N³) einsum is rewritten as a batched matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
                = (left[b,:, :, d]) @ (right[b,:, :, d])ᵀ

For every hidden dimension `d` we treat the N×N slice as a separate matrix,
flatten `batch × hidden_dim` into one dimension and run a custom
batched GEMM kernel written in Triton.  All other operations (LayerNorm,
linear projections, gating, final linear) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_A, batch_stride_B, batch_stride_C,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM: C = A @ B
    A : [batch, M, K]   (float16)
    B : [batch, K, N]   (float16)
    C : [batch, M, N]   (float32)

    The kernel is launched with a 3‑D grid:
        grid = (batch, ceil(M/BLOCK_M), ceil(N/BLOCK_N))
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # start offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = A_ptr + batch_id * batch_stride_A
    b_batch = B_ptr + batch_id * batch_stride_B
    c_batch = C_ptr + batch_id * batch_stride_C

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # B tile [BLOCK_K, BLOCK_N]
        b = tl.load(
            b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        acc += tl.dot(a, b)   # accumulation in float32

    # write back
    c_ptr = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑based forward pass for the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------------- weights ---------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    to_out_weight = weights["to_out.weight"].to(device)

    # -------------------------- layer norm over dim ----------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    x = x_norm.to(torch.float16)                       # work in half for speed

    # -------------------------- linear projections -----------------------
    left = F.linear(x, left_proj_weight)              # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -------------------------- mask ------------------------------------
    if mask is None:
        mask_f = torch.ones_like(input_tensor[..., 0], dtype=x.dtype, device=device)
    else:
        mask_f = mask.to(x.dtype)
    mask_f = mask_f.unsqueeze(-1)                     # [B,N,N,1]

    left = left * mask_f
    right = right * mask_f

    # -------------------------- gating ----------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid().to(x.dtype)
    right_gate = F.linear(x, right_gate_weight).sigmoid().to(x.dtype)
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()   # keep float32

    left = left * left_gate
    right = right * right_gate

    # -------------------------- heavy O(N³) part -----------------------
    # Transpose the second sequence axis of `right`
    right_t = right.permute(0, 2, 1, 3).contiguous()   # [B,N,N,hidden_dim] (k,j)

    # Move hidden_dim into the batch dimension
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_t.permute(0, 3, 1, 2).contiguous()  # same layout

    total_batch = B * hidden_dim
    left_mat = left_perm.view(total_batch, N, N)        # half
    right_mat = right_perm.view(total_batch, N, N)      # half
    out_mat = torch.empty((total_batch, N, N), dtype=torch.float32, device=device)

    # strides (in elements)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    batch_stride_A = left_mat.stride(0)

    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    batch_stride_B = right_mat.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    batch_stride_C = out_mat.stride(0)

    # launch kernel
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_A,
        batch_stride_B,
        batch_stride_C,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # -------------------------- post‑processing -------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # final projection to `dim`

    return output
