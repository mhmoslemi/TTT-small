    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    j_start = pid_j * BLOCK_J

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_b_left, stride_i_left, stride_k_left, stride_d_left,
    stride_b_right, stride_j_right, stride_k_right, stride_d_right,

):

    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    b = pid_b

    # Generate offsets
    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)

    # Mask for i,j within bounds
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator: shape (BLOCK_I, BLOCK_J, BLOCK_D)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k0 = kb * BLOCK_K
        k_curr = k0 + k
        mask_k = k_curr < N

        # Load left tile: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + b*stride_b_left + i[:, None, None]*stride_i_left + k_curr[None, :, None]*stride_k_left + tl.arange(0, BLOCK_D)[None, None, :]*stride_d_left

```python
import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_b_left, stride_i_left, stride_k_left, stride_d_left,
    stride_b_right, stride_j_right, stride_k_right, stride_d_right,
    stride_b_out, stride_i_out, stride_j_out, stride_d_out,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """
    Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    using tiled GEMM style. The hidden dimension D is processed in blocks
    of size BLOCK_D (compile‑time constant). The kernel is launched with a
    3‑D grid: (ceil(N/BLOCK_I), ceil(N/BLOCK_J), B).
    """
    pid_i = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block
    pid_b = tl.program_id(2)  # batch index

    # compute start offsets
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    b = pid_b

    # i, j, k offsets
    i_idx = i_start + tl.arange(0, BLOCK_I)
    j_idx = j_start + tl.arange(0, BLOCK_J)
    k_idx = tl.arange(0, BLOCK_K)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k):
        k_start = kb * BLOCK_K
        cur_k = k_start + k_idx
        mask_k = cur_k < N

        # ------- load left tile (i, k, d) -------
        left_ptrs = (
            left_ptr
            + b * stride_b_left

            s_b_r,
            s_j_r,
            s_k_r,
            s_d_r,
            s_b_o,
            s_i_o,
            s_j_o,
            s_d_o,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # Convert back to float32 for the remaining operations
    out = out_contract_fp16.to(torch.float32)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],

    )

"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy i‑j‑k contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel using a tiled GEMM style.
All LayerNorms, Linear projections and gating are carried out with PyTorch
for simplicity. The kernel processes the hidden dimension in blocks of
BLOCK_D (32), which cleanly divides the hidden dimension used in the tests
(128). The kernel is launched with a 3‑D grid:
    (ceil(N/BLOCK_I), ceil(N/BLOCK_J), batch)
and masks out-of‑bounds elements.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B,
    N,
    stride_b_left,
    stride_i_left,
    stride_k_left,
    stride_d_left,
    stride_b_right,
    stride_j_right,
    stride_k_right,
    stride_d_right,
    stride_b_out,
    stride_i_out,
    stride_j_out,
    stride_d_out,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    The hidden dimension D is processed in compile‑time blocks of size BLOCK_D.
    """
    pid_i = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block
    pid_b = tl.program_id(2)  # batch index

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    b = pid_b

    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k):
        k0 = kb * BLOCK_K
        cur_k = k0 + k
        mask_k = cur_k < N

        # ----- left tile (i, k, d) -----
        left_ptrs = (
            left_ptr
            + b * stride_b_left
            + i[:, None, None] * stride_i_left
            + cur_k[None, :, None] * stride_k_left
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_d_left
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (BLOCK_I, BLOCK_K, BLOCK_D)

        # ----- right tile transposed (k, j, d) -----
        right_ptrs = (
            right_ptr
            + b * stride_b_right
            + cur_k[:, None, None] * stride_k_right
            + j[None, :, None] * stride_j_right
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_d_right
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None],
            other=0.0,
        )  # (BLOCK_K, BLOCK_J, BLOCK_D)

        # ----- dot for each d -----
        for d in range(BLOCK_D):
            a = left[:, :, d]          # (BLOCK_I, BLOCK_K)
            b_mat = right[:, :, d]     # (BLOCK_K, BLOCK_J)
            acc[:, :, d] += tl.dot(a, b_mat)  # (BLOCK_I, BLOCK_J)

    # ----- store result -----
    out_ptrs = (
        out_ptr
        + b * stride_b_out
        + i[:, None, None] * stride_i_out
        + j[None, :, None] * stride_j_out
        + tl.arange(0, BLOCK_D)[None, None, :] * stride_d_out
    )
    mask_ij = mask_i[:, None, None] & mask_j[None, :, None]
    tl.store(out_ptrs, acc, mask=mask_ij)
