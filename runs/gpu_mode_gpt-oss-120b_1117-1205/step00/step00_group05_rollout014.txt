    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

):

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    bs, N, _, dim_check = input_tensor.shape
    assert dim_check == dim

    # Cast to float16 for speed maybe GPU ; but ensure final output float32.
    # Compute layernorm manually using weights. Implementation: Xnorm = (x - mean)/sqrt(var+eps)*weight + bias.
    # But easier: use torch.nn.functional.layer_norm? Could use torch.nn.functional.layer_norm with weight and bias arguments.
    # Use F.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5).
    # Input shape 4D, we apply across last dim.

    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights['norm.weight'],
        bias=weights['norm.bias'],
        eps=1e-5
    )
    # compute left and right projections:
    # left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])
    # shape [B,N,N,hidden_dim]

    # mask: if nomask? config may have 'nomask' bool, if True then set mask all ones; else use mask.
    if config.get('nomask', False):
        mask_ = torch.ones_like(mask, dtype=left.dtype, device=device)
    else:
        mask_ = mask
    mask_exp = mask_.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # compute core multiplication via custom Triton kernel
    # We want left, right as float16 for faster compute maybe.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output tensor (float16)
    out_fp16 = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=device)

    # compute strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left_fp16.stride()
    # left_fp16 stride order (B, N, N, H)
    # We'll treat i like first N, k second N.
    right_stride_b, right_stride_j, right_stride_k, right_stride_h = right_fp16.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out_fp16.stride()
    # All strides in elements (not bytes)

    # define BLOCK sizes:
    BLOCK_M = 64  # rows
    BLOCK_N = 64  # cols
    BLOCK_K = 64  # reduction.

    # compute grid
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        bs * hidden_dim
    )
    # Launch kernel
    tr_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        bs,
        N,
        hidden_dim,

"""
TriMul (outgoing) forward pass with a Triton kernel for the heavy
pairwise multiplication.

The algorithm follows the reference PyTorch implementation:
1. LayerNorm over the channel dimension.
2. Linear projections (left/right) and per‑channel gates.
3. Optional masking.
4. Core operation:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication over the k‑dimension for each
   (batch, hidden) slice.  It is implemented by a custom Triton kernel that
   loops over k in blocks and accumulates in float32.
5. LayerNorm over the hidden dimension, element‑wise gate, final linear
   projection back to the original channel size.

All tensors stay on the GPU; the heavy O(N³) reduction is done in
Triton, while the remaining pointwise ops use PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = Σ_k left[...,i,k,h] * right[...,j,k,h]"""
    pid_m = tl.program_id(0)               # block row  (i)
    pid_n = tl.program_id(1)               # block col  (j)
    pid_hb = tl.program_id(2)              # combined batch * hidden index

    # decode batch and hidden indices
    b = pid_hb // H
    h = pid_hb % H

    # range of i / j / k for this tile
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # ------------------------------
        # Load left[b, i, k, h]  -> (M, K)
        left_ptrs = left_ptr \
            + b * stride_l_b + h * stride_l_h \
            + i_off[:, None] * stride_l_i \
            + k_off[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)                # FP16

        # ------------------------------
        # Load right[b, j, k, h] transposed to (K, N)
        # we need right indexed as (k, j) so that dot(left, right) works
        right_ptrs = right_ptr \
            + b * stride_r_b + h * stride_r_h \
            + k_off[:, None] * stride_r_k \
            + j_off[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)               # FP16

        # ------------------------------
        # Accumulate dot product (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left, right)          # acc is FP32

    # store the result (convert back to FP16)
    out_ptrs = out_ptr \
        + b * stride_o_b + h * stride_o_h \
        + i_off[:, None] * stride_o_i \
        + j_off[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel
    for the O(N³) pairwise multiplication.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor [B, N, N, dim]
            - mask:  torch.Tensor [B, N, N] (may be all‑ones if no mask)
            - weights: dict of torch.Tensors containing model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, dim_check = input_tensor.shape
    assert dim_check == dim, "dimension mismatch"

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (over the last dimension)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections
    left = F.linear(x, weights["left_proj.weight"])   # -> [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (if provided)
    if config.get("nomask", False) or mask is None:
        mask_tensor = torch.ones((bs, N, N), dtype=left.dtype, device=device)
    else:
        mask_tensor = mask
    mask_exp = mask_tensor.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core multiplication via Triton
    # Cast to FP16 for throughput
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty((bs, N, N, hidden_dim),
                           dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    s_lb, s_li, s_lk, s_lh = left_fp16.stride()
    s_rb, s_rj, s_rk, s_rh = right_fp16.stride()
    s_ob, s_oi, s_oj, s_oh = out_fp16.stride()

    # Triton block sizes (tuned for typical N <= 1024, hidden_dim = 128)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑blocks
        triton.cdiv(N, BLOCK_N),          # j‑blocks
        bs * hidden_dim,                  # batch * hidden slices
    )
    tr_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        bs, N, hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Ensure kernel has finished before using the result
    torch.cuda.synchronize()

    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Post‑processing (layer norm, gate, final linear)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # gating (broadcast over hidden dim)

    # final projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
