def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; 
    eps = 1e-5 (maybe from config; else default)
    device = input_tensor.device

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    ) # shape B,N,N,dim

    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"]) # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # apply mask
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # gating
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # compute contraction using Triton
    # flatten B*hidden_dim batch dimension
    B = input_tensor.shape[0]; N = input_tensor.shape[1];
    # reshape left and right: (B, N, N, hidden_dim) -> (B*hidden_dim, N, N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B*hidden_dim, N, N) # shape (BH, N, K)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B*hidden_dim, N, N) # shape (BH, N, K)
    # compute out_reshape = left @ right^T using triton
    out_reshape = triton_matmul_transpose(left_reshape, right_reshape) # shape (BH, N, N)
    # reshape back
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous() # shape (B, N, N, hidden_dim)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate

    # final linear
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out_final

def triton_matmul_transpose(A, B):
    # A shape (batch, M, K)
    # B shape (batch, N, K)
    # output shape (batch, M, N)
    # both contiguous
    # We'll get dimensions:
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2
    # allocate output
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # compute strides (in elements)
    stride_a_batch = A.stride(0)
    stride_a_m = A.stride(1)
    stride_a_k = A.stride(2)
    stride_b_batch = B.stride(0)
    stride_b_n = B.stride(1)
    stride_b_k = B.stride(2)
    stride_c_batch = C.stride(0)
    stride_c_m = C.stride(1)
    stride_c_n = C.stride(2)

    # but we don't need batch stride inside kernel if we treat each batch id separately.
    # We'll compute pointer offset for each batch = batch_id * batch_stride. The stride can be derived.
    # Convert to integer offsets (in elements).
    # At kernel call we must pass these strides as integers.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_trans_kernel[grid](
        A, B, C,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    return C

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # maybe float32

    # Use torch.nn.functional.layer_norm for last dimension = config["dim"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(config["dim"],),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # (2) linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # shape (B,N,N,hidden_dim)
    # (3) apply mask (if needed)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(x_norm)
        left = left * mask_
        right = right * mask_

    # (4) compute gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # (5) apply gates
    left = left * left_gate
    right = right * right_gate

    # (6) cast to fp16 for contraction
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, hidden_dim = left_f16.shape

    # Permute and reshape for batched matmul
    left_batch = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batch = right_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_batch = batched_matmul_transpose(left_batch, right_batch)  # shape (B*hidden_dim, N, N), dtype float16

    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B_T block (K x N) = B[j,k] transposed
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul_transpose(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ B^T for batched matrices where A, B have shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N). Operates in fp16 (inputs must be contiguous fp16 tensors).
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    batch2, N, K2 = B.shape
    assert batch == batch2 and K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    stride_a_batch = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_b_batch = B.stride(0)
    stride_bn = B.stride(1)
    stride_bk = B.stride(2)

    stride_c_batch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_trans_kernel[grid](
        A, B, C,
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bn,
        stride_bk,
        stride_c_batch,
        stride_cm,
        stride_cn,
        M, N, K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------

        right = right * mask_

    # ------------------------------------------------------------------

    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()

    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Contract (einsum) using Triton batched GEMM: out = left @ rightᵀ
    #    Work in FP16 to reduce memory traffic.
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # reshape to (batch*hidden_dim, N, N)
    left_batch = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batch = right_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Triton kernel
    out_batch = batched_matmul_transpose(left_batch, right_batch)  # (B*hidden_dim, N, N) fp16

    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------

"""
TriMul forward (outgoing) implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Apply two Linear projections (left/right) without bias.
3. Optionally mask the projected tensors.
4. Compute three sigmoid gates (left, right, output) with Linear layers.
5. Multiply left/right by their respective gates.
6. Fuse the N‑dimensional contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom batched GEMM kernel: out = left @ rightᵀ.
   The contraction is performed in fp16 for speed/memory,
   while accumulation is in fp32.
7. Second Layer‑norm over the hidden dimension, apply the output gate,
   and a final Linear projection back to the original dimension.
8. Return a float32 tensor of shape [B, N, N, dim].

The heavy N³ operation (step 6) is accelerated by a Triton kernel that
processes each (batch × hidden_dim) matrix pair as a separate GEMM.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _batched_matmul_trans_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_am, stride_ak,
    stride_b_batch, stride_bn, stride_bk,
    stride_c_batch, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C = A @ Bᵀ   where A: (batch, M, K), B: (batch, N, K)"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block (M × K)
        a_offs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Bᵀ block (K × N) = transpose of B (N × K)
        b_offs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (M×K)·(K×N) → (M×N)

    # store result
    c_offs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_offs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_transpose(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for batches.
    A – (batch, M, K), B – (batch, N, K), both fp16 & contiguous.
    Returns C – (batch, M, N) in fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    batch2, N, K2 = B.shape
    assert batch == batch2 and K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides in units of elements
    stride_a_batch = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_b_batch = B.stride(0)
    stride_bn = B.stride(1)
    stride_bk = B.stride(2)

    stride_c_batch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))

    _batched_matmul_trans_kernel[grid](
        A, B, C,
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bn, stride_bk,
        stride_c_batch, stride_cm, stride_cn,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] float32
            - mask:        [B, N, N] (bool or float)
            - weights: dict with all Linear/LayerNorm parameters
            - config:  dict with "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask may be all‑ones)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Contraction via Triton (fp16 for speed / memory)
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, _ = left_f16.shape   # B, N, N, hidden_dim

    # reshape to (B*hidden_dim, N, N) -> each batch corresponds to one hidden channel
    left_batch = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batch = right_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Triton GEMM: out = left_batch @ right_batchᵀ
    out_batch = _batched_matmul_transpose(left_batch, right_batch)

    # reshape back to (B, N, N, hidden_dim)
    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # clean up large fp32 tensors
    del left, right, left_f16, right_f16, left_batch, right_batch
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 7. Second LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(out.dtype)

    # ------------------------------------------------------------------
    # 8. Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    # Return float32 as required
    return out.to(torch.float32)
