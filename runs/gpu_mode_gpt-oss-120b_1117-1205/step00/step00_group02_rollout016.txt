@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0) # batch index
    pid_m = tl.program_id(1) # i block
    pid_n = tl.program_id(2) # j block

    # load offsets for block shapes
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # define k loop block offsets
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute start pointers for batch
    a_batch_ptr = a_ptr + pid * batch_stride_a
    b_batch_ptr = b_ptr + pid * batch_stride_b
    c_batch_ptr = c_ptr + pid * batch_stride_c

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load A tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)

        # load B tile (note B is K x N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        acc += tl.dot(a, b)

        # advance pointers (not needed because we compute offsets each iteration)

    # store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Ensure dtype float32 (maybe rename)
    x = input_tensor
    bsz, seq_len, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    mask_uns = mask.unsqueeze(-1)
    left = left * mask_uns
    right = right * mask_uns

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # Normalization
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask
    if mask is not None:
        mask_uns = mask.unsqueeze(-1)
        left = left * mask_uns
        right = right * mask_uns

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Prepare tensors for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # B, D, N, N (i,k)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # B, D, k, j

    batch_hd = B * hidden_dim
    A = left_perm.view(batch_hd, N, N).to(torch.float16)
    B_mat = right_T.view(batch_hd, N, N).to(torch.float16)
    C = torch.empty((batch_hd, N, N), dtype=torch.float32, device=device)

    # strides
    stride_am = A.stride(1)
    stride_ak = A.stride(2)
    batch_stride_a = A.stride(0)

    stride_bk = B_mat.stride(1)
    stride_bn = B_mat.stride(2)
    batch_stride_b = B_mat.stride(0)

    stride_cm = C.stride(1)
    stride_cn = C.stride(2)
    batch_stride_c = C.stride(0)

    M = N
    K = N
    N_dim = N

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch_hd,
            triton.cdiv(M, BLOCK_M),
            triton.cdiv(N_dim, BLOCK_N))

    batched_matmul_kernel[grid](
        A, B_mat, C,
        batch_stride_a, stride_am, stride_ak,
        batch_stride_b, stride_bk, stride_bn,
        batch_stride_c, stride_cm, stride_cn,
        M, N_dim, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape output
    out_perm = C.view(B, hidden_dim, N, N)  # B, D, N, N
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # B, N, N, D

                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"])

    out = out * out_gate

    # final linear
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections -> left/right (shape [B,N,N,HD]) and apply mask.
3. Compute three element‑wise sigmoid gates and apply them.
4. Compute the core contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   By reshaping to a batched matrix‑multiply:
       A = left.permute(0,3,1,2).reshape(B*HD, N, N)   (M×K)
       B = right.permute(0,3,1,2).permute(0,1,3,2).reshape(B*HD, N, N)   (K×N)
   A and B are multiplied with a custom Triton kernel that performs
   a batched GEMM (M,N,K) → (M,N) for each batch element.
5. Reshape the result back to [B,N,N,HD], apply a second layer‑norm,
   multiply by the output gate and a final linear projection to dimension *dim*.

The kernel is written for float16 inputs (to reduce traffic) with
float32 accumulation, and works for arbitrary batch size, sequence length
and hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B  with  A[M,K], B[K,N] for each batch."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the M and N dimensions.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch.
    a_batch = a_ptr + pid_batch * batch_stride_a
    b_batch = b_ptr + pid_batch * batch_stride_b
    c_batch = c_ptr + pid_batch * batch_stride_c

    # Accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K).
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)
        a = a.to(tl.float32)

        # Load B tile (K x N).
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = b.to(tl.float32)

        # Accumulate.
        acc += tl.dot(a, b)

    # Write out C tile (M x N).
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N]  (0/1)
            - weights: dict of parameters (see implementation)
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input.
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections.
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Apply mask (if provided) – mask is broadcast to channel dim.
    # -----------------------------------------------------------------
    if mask is not None:
        mask_uns = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left = left * mask_uns
        right = right * mask_uns

    # -----------------------------------------------------------------
    # 4️⃣ Gating.
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Core contraction via a batched GEMM Triton kernel.
    #    Reshape tensors to (B*HD, N, N) and compute
    #    C = A @ B  where B is already transposed (K,N).
    # -----------------------------------------------------------------
    # Permute so that hidden_dim becomes the second axis.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,HD,N,N]  (i,k)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,HD,N,N]  (j,k)

    # Transpose the last two axes of right to get (k,j) order required for matmul.
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # [B,HD,N,N]  (k,j)

    # Flatten batch and hidden dimension.
    batch_hd = B * hidden_dim
    A = left_perm.view(batch_hd, N, N).to(torch.float16)
    B_mat = right_T.view(batch_hd, N, N).to(torch.float16)
    C = torch.empty((batch_hd, N, N), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes).
    stride_am = A.stride(1)
    stride_ak = A.stride(2)
    batch_stride_a = A.stride(0)

    stride_bk = B_mat.stride(1)
    stride_bn = B_mat.stride(2)
    batch_stride_b = B_mat.stride(0)

    stride_cm = C.stride(1)
    stride_cn = C.stride(2)
    batch_stride_c = C.stride(0)

    M = N
    K = N
    N_dim = N

    # Tile sizes – chosen for good occupancy on B200.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_hd,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_dim, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A, B_mat, C,
        batch_stride_a, stride_am, stride_ak,
        batch_stride_b, stride_bk, stride_bn,
        batch_stride_c, stride_cm, stride_cn,
        M, N_dim, K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,HD].
    out_perm = C.view(B, hidden_dim, N, N)          # [B,HD,N,N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B,N,N,HD]

    # -----------------------------------------------------------------
    # 6️⃣ Second layer‑norm, output gate and final linear projection.
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
