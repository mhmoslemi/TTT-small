@triton.jit
def trimul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_batch_gate = tl.program_id(1)  # = batch*H index

    # compute batch and hidden index
    b = pid_batch_gate // H
    d = pid_batch_gate % H

    # compute block indices for i and j
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    i_block = pid // num_blocks_n
    j_block = pid % num_blocks_n

    # compute offsets
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # masks for i and j out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k range for this tile
        k_tile_offset = k * BLOCK_K + k_offset
        # mask for k out-of-bounds
        mask_k = k_tile_offset < N

        # compute pointers
        # left shape: (i,k) with hidden_dim d fixed
        left_ptrs = left_ptr + b*stride_left_b + i_offset[:, None] * stride_left_i + k_tile_offset[None, :] * stride_left_k + d * stride_left_d
        # right shape: (k,j)
        right_ptrs = right_ptr + b*stride_right_b + j_offset[:, None] * stride_right_j + k_tile_offset[None, :] * stride_right_k + d * stride_right_d

        # load A and B with proper masks
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b_ = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b_)
        # after dot, loop continues

    # store output
    out_ptrs = out_ptr + b*stride_out_b + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j + d * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Build weight parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure float32
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.to(torch.float32)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # no bias
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if mask.shape != (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]):
        # trust mask shape but ensure broadcast
        pass
    mask_exp = mask.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Contraction using Triton
    B,N,N2,_ = left.shape
    assert N == N2
    # Define block sizes
    BLOCK_M = 64  # maybe choose 64 for i dimension
    BLOCK_N = 64
    BLOCK_K = 64  # may adjust based on N

    # compute strides
    left_stride = left.stride()  # (b,i,k,d)
    right_stride = right.stride()
    out_shape = (B,N,N,hidden_dim)
    out = torch.empty(out_shape, dtype=torch.float32, device=left.device)

    out_stride = out.stride()

    # Launch kernel with grid
    grid_x = triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid_y = B * hidden_dim  # second axis that processes batch*hidden

    trimul_contract_kernel[grid_x, grid_y](
        left, right, out,
        B, N, hidden_dim,
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        out_stride[0], out_stride[1], out_stride[2], out_stride[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Now to_out_norm
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # final linear
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output

"""
TriMul forward implementation with a fused Triton kernel for the O(N³) contraction.

Algorithm:
1. Layer‑norm the input tensor (shape B×N×N×C).
2. Apply left/right linear projections (C → H) and mask (if present).
3. Compute element‑wise gates (sigmoid of linear projections) and apply them.
4. Perform the core contraction:
       out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d]
   which is a batched mat‑mul per hidden channel (A @ Bᵀ). This is
   realised by a Triton kernel that tiles the i, j and k dimensions.
5. Apply a second LayerNorm over the hidden dimension, multiply by the output
   gate, and project back to the original dimension C.
All tensors remain on the GPU; only the contraction is off‑loaded to Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------------- #
# Triton kernel: contraction over the k dimension (A @ Bᵀ) per batch & hidden d
# ----------------------------------------------------------------------------- #
@triton.jit
def _trimul_contract_kernel(
    left_ptr,          #   const float*  left  (B,N,N,H)
    right_ptr,         #   const float*  right (B,N,N,H)
    out_ptr,           #   float*        out   (B,N,N,H)
    B: tl.int32,       # batch size
    N: tl.int32,       # sequence length
    H: tl.int32,       # hidden dimension
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid = tl.program_id(0)          # combined (i_block, j_block)
    pid_bd = tl.program_id(1)       # batch * hidden index

    # --------------------------------------------------------------
    # decode batch index and hidden channel
    # --------------------------------------------------------------
    b = pid_bd // H                 # batch index
    d = pid_bd % H                  # hidden channel

    # --------------------------------------------------------------
    # decode i‑block and j‑block
    # --------------------------------------------------------------
    n_blocks_j = tl.cdiv(N, BLOCK_N)
    i_block = pid // n_blocks_j
    j_block = pid % n_blocks_j

    # offsets inside the block
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # out‑of‑bounds masks for i and j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over k tiles
    # --------------------------------------------------------------
    n_tiles_k = tl.cdiv(N, BLOCK_K)
    for tk in range(0, n_tiles_k):
        # current k indices (may exceed N, will be masked)
        cur_k = tk * BLOCK_K + k_offset
        mask_k = cur_k < N

        # ------------------------------------------------------------------
        # left: shape (BLOCK_M, BLOCK_K)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_offset[:, None] * stride_left_i
            + cur_k[None, :] * stride_left_k
            + d * stride_left_d
        )
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ------------------------------------------------------------------
        # right: shape (BLOCK_K, BLOCK_N)  (need transposed layout)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j_offset[:, None] * stride_right_j
            + cur_k[None, :] * stride_right_k
            + d * stride_right_d
        )
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # ------------------------------------------------------------------
        # matrix multiply of (M×K)·(K×N) → (M×N)
        # ------------------------------------------------------------------
        acc += tl.dot(a, b_tile)

    # ----------------------------------------------------------------------
    # write back result
    # ----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_offset[:, None] * stride_out_i
        + j_offset[None, :] * stride_out_j
        + d * stride_out_d
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------------- #
# Host‑side entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul module with a Triton‑accelerated O(N³) contraction.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dict containing all linear‑layer weights and LayerNorm params
            - config: dict with keys "dim" (C) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input (C → C)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (C → H) – no bias
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask if required (mask may be all‑ones when nomask=True)
    # ------------------------------------------------------------------
    if mask.ndim == 3:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp
    else:
        # unexpected shape – fall back to no masking
        pass

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Ensure contiguity for Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # Prepare output buffer for the contraction
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=device)

    # ------------------------------------------------------------------
    # Compute strides (required by the kernel)
    # ------------------------------------------------------------------
    s_left = left.stride()     # (b,i,k,d)
    s_right = right.stride()   # (b,j,k,d)
    s_out = out.stride()       # (b,i,j,d)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_x = triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid_y = B * hidden_dim

    _trimul_contract_kernel[grid_x, grid_y](
        left, right, out,
        B, N, hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ------------------------------------------------------------------
    # Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply output gate and final linear projection (H → C)
    # ------------------------------------------------------------------
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
