def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N,
    stride_abl, stride_aim, stride_aik,
    stride_bbl, stride_bik, stride_bjn,
    stride_cbl, stride_cim, stride_cjn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)

    # compute block offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for validity
    mask_m = off_m < N
    mask_n = off_n < N

    # start pointers for this block
    a_ptrs = a_ptr + pid_b * stride_abl + off_m[:, None] * stride_aim + tl.arange(0, BLOCK_K)[None, :] * stride_aik
    b_ptrs = b_ptr + pid_b * stride_bbl + tl.arange(0, BLOCK_K)[:, None] * stride_bik + off_n[None, :] * stride_bjn

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k offset range
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load A and B tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

        # pointer increment
        a_ptrs += BLOCK_K * stride_aik
        b_ptrs += BLOCK_K * stride_bik

    # write back result
    c_ptrs = c_ptr + pid_b * stride_cbl + off_m[:, None] * stride_cim + off_n[None, :] * stride_cjn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def layer_norm(x, weight, bias, eps=1e-5):
    # x shape [..., C]
    # compute mean and var across last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape  # shape [bs, seq_len, seq_len, dim]
    hidden_dim = config["hidden_dim"]  # H

    # LayerNorm on input dim (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections (no bias) for left and right:
    # left_proj weight shape (hidden_dim, dim). Actually nn.Linear(dim, hidden_dim) weight shape (hidden_dim, dim). In torch semantics, linear(x) = x @ weight.T (assuming bias optional). We'll apply torch.nn.functional.linear(x, weight) where weight shape (out_features,in_features).
    left_proj_weight = weights["left_proj.weight"] # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    # Project: left = torch.nn.functional.linear(x, left_proj_weight)
    # Note: x shape (..., dim). linear expects shape (N, dim). So we can flatten preceding dims and then reshape.
    # Use torch.nn.functional.linear with weight shape (hidden_dim, dim). It returns shape (..., hidden_dim)
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask (if mask is provided). mask shape [B,N,N]. Expand to [B,N,N,1]
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # same dtype as left (float16 later)
        left = left * mask_exp
        right = right * mask_exp

    # Compute gates: each gating linear with weight and sigmoid
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()

@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch dimension (B*H)
    pid_m = tl.program_id(1)  # i block
    pid_n = tl.program_id(0)  # j block

    # Compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers for this tile
        a_offset = pid_b * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        b_offset = pid_b * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col

        # Load tiles
        a = tl.load(a_ptr + a_offset, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptr + b_offset, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_offset = pid_b * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptr + c_offset, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

TriMul forward pass with a Triton kernel for the core tensor contraction.

The core operation computes, for each batch b and hidden channel d:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which can be expressed as a batched matrix multiplication
    out = left @ right^T
where left and right are reshaped to (B*H, N, N).  The Triton kernel
implements a tiled GEMM with FP16 inputs and FP32 accumulation.
All other operations (LayerNorms, linear projections, gating) are
performed with regular PyTorch kernels.

    N,                                 # sequence length
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B  where
       A : (batch_total, N, N)  (row‑major, K dimension is inner)
       B : (batch_total, N, N)  (row‑major, K dimension is first axis)
       The output C has shape (batch_total, N, N).
    """
    pid_b = tl.program_id(2)          # fused batch * hidden_dim dimension
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(0)          # block col (j)

    # ---- block offsets in the output matrix ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over the reduction dimension k ----
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers to the required tiles of A and B
        a_offset = (
            pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + k_offset[None, :] * stride_a_col
        )
        b_offset = (
            pid_b * stride_b_batch
            + k_offset[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        # load tiles, cast to FP32 for accumulation
        a = tl.load(a_ptr + a_offset,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptr + b_offset,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # GEMM accumulation
        acc += tl.dot(a, b)

    # ---- write back the result in FP16 ----
    c_offset = (
        pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(c_ptr + c_offset,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module with the central contraction
    performed by a Triton GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # configuration
    # ------------------------------------------------------------------
    dim = config["dim"]                     # input feature dimension
    hidden_dim = config["hidden_dim"]       # projection dimension
    eps = 1e-5                               # LayerNorm epsilon

    # ------------------------------------------------------------------
    # LayerNorm on the input tensor (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional masking
    # ------------------------------------------------------------------
    nomask = config.get("nomask", False)
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)    # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating scalars (sigmoid on linear maps of the *original* x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton GEMM (FP16)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # left: (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: need to swap the two sequence axes before reshaping
    # right: (B,N,N,H) -> (B,H,K,J) where K=j axis, J=k axis swapped
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    # Flatten the batch + channel dimension so that we have
    # A : (batch_total, N, N)   (row‑major, K inner)
    # B : (batch_total, N, N)   (row‑major, K outer)
    A = left_perm.view(batch_total, N, N)
    B = right_perm.view(batch_total, N, N)

    # Output buffer
    C = torch.empty_like(A)   # FP16, same shape as A

    # ------------------------------------------------------------------
    # Compute strides (in elements) for Triton
    # ------------------------------------------------------------------
    stride_a_batch, stride_a_row, stride_a_col = A.stride()
    stride_b_batch, stride_b_row, stride_b_col = B.stride()
    stride_c_batch, stride_c_row, stride_c_col = C.stride()

    # ------------------------------------------------------------------
    # Tiling parameters – chosen to give good occupancy across the test sizes
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_N),   # blocks along j (output columns)
        triton.cdiv(N, BLOCK_M),   # blocks along i (output rows)
        batch_total,               # one program per (batch, hidden) pair
    )

    # Launch kernel
    tri_mul_gemm_kernel[grid](
        A,
        B,
        C,
        N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to (B,N,N,hidden) and continue with LayerNorm & linear
    # ------------------------------------------------------------------
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]
    out = C.to(torch.float32)

    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the output gate (channel‑wise)
    out = out * out_gate

    # Final linear projection back to the original dimension
    out = F.linear(out, weights["to_out.weight"])
    return out
