# Triton kernel for the core contraction:
#   out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
# The kernel processes a tile of (i, j) for a single (b, h) pair.
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,      # pointers
    B, N, H,                           # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)          # i‑block
    pid_j = tl.program_id(1)          # j‑block
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # decode batch and hidden channel from pid_bh
    b = pid_bh // H
    h = pid_bh % H

    # start indices for this tile
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # global indices inside the tile
    i_idx = i_start + tl.arange(0, BLOCK_I)
    j_idx = j_start + tl.arange(0, BLOCK_J)

    # masks for out‑of‑bounds i/j
    mask_i = i_idx < N
    mask_j = j_idx < N

    # offsets for the fixed batch and hidden dimension
    off_bh_left  = b * stride_left_b  + h * stride_left_h
    off_bh_right = b * stride_right_b + h * stride_right_h
    off_bh_out   = b * stride_out_b   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k_start = kk * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

    # -----------------------------------------------------------------
    if config.get("nomask", True):
        mask_val = torch.ones_like(mask, dtype=left.dtype, device=left.device)
    else:
        mask_val = mask.to(left.dtype)
    mask_val = mask_val.unsqueeze(-1)          # [B,N,N,1]

    # Apply mask and gates (order matches reference implementation)
    left = left * mask_val * left_gate
    right = right * mask_val * right_gate

    # -----------------------------------------------------------------
    # 5) Core contraction via Triton kernel (half‑precision for bandwidth)
    # -----------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)   # allocated result

    B, N, _, H = left_fp16.shape
    stride_left = left_fp16.stride()
    stride_right = right_fp16.stride()
    stride_out = out_fp16.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B * H,
    )
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    # Back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
