def tri_mul_matmul_kernel(

    B, N, H,

    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H) # for vectorization along H

    # Set masks
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # Compute output accumulator of shape (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # Normalize
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask unsqueeze
    mask_unsq = mask.unsqueeze(-1)
    left = left * mask_unsq
    right = right * mask_unsq
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for kernel (reduce memory bandwidth)
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_f16.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_f16.device)

    # Strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_f16.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_f16.stride()
    # For right we need stride for j dimension (same as i)
    stride_right_j = stride_right_i  # correct alias

    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Block sizes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 16  # block for hidden dim of size as multiple of 16

    # Adjust Block_H if hidden_dim is smaller
    if hidden_dim < BLOCK_H:
        BLOCK_H = hidden_dim
    # compute grid
    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J), B)

    # launch kernel
    tri_mul_kernel[grid](
        left_f16, right_f16, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H
    )

    # Continue with remaining ops
    # layer norm on hidden dim:
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    # multiply by out_gate
    out = out * out_gate
    # final linear projection
    output = torch.nn.functional.linear(out, weights["to_out.weight"])
    return output

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # offsets
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)
    h = tl.arange(0, BLOCK_H)

    # masks for i,j,h
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # initialize acc to zero (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k dimension
    cur_k = 0
    while cur_k < N:
        # compute actual k indices
        kk = cur_k + k
        mask_k = kk < N

        # compute combined mask for loading left and right
        # left shape: (i, kk, h)
        mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]

        # pointers for left
        left_ptrs = left_ptr + pid_b * stride_left_b \
                    + i[:, None, None] * stride_left_i \
                    + kk[None, :, None] * stride_left_k \
                    + h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask, other=0.0, dtype=tl.float16).to(tl.float32)

        # pointers for right
        right_ptrs = right_ptr + pid_b * stride_right_b \
                    + j[:, None, None] * stride_right_j \
                    + kk[None, :, None] * stride_right_k \
                    + h[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask, other=0.0, dtype=tl.float16).to(tl.float32)

        # accumulate outer product for each k in block
        # loop over kk_index within block (0..BLOCK_K-1)
        for kk_idx in range(0, BLOCK_K):
            l = left[:, kk_idx, :]          # (BLOCK_I, BLOCK_H)
            r = right[:, kk_idx, :]         # (BLOCK_J, BLOCK_H)
            # broadcast multiply
            prod = l[:, None, :] * r[None, :, :]   # shape (BLOCK_I, BLOCK_J, BLOCK_H)
            acc += prod
        cur_k += BLOCK_K

    # store result
    out_ptrs = out_ptr + pid_b * stride_out_b \
               + i[:, None, None] * stride_out_i \
               + j[None, :, None] * stride_out_j \
               + h[None, None, :] * stride_out_h
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)
```

"""
Optimized TriMul (outgoing) forward pass.

The heavy component is the tensor contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication for each hidden dimension d.
We implement this contraction with a Triton kernel that tiles the i, j,
k and hidden (d) dimensions. All other operations (LayerNorm, Linear
projections, gating) are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------------------- #
# Triton kernel performing the N^3 contraction for a block of hidden dims.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    num_h_blocks: tl.constexpr,                 # # of hidden blocks
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # --------------------------------------------------------------------- #
    # program IDs
    # --------------------------------------------------------------------- #
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_z = tl.program_id(2)          # combined batch & hidden‑block ID

    pid_b = pid_z // num_h_blocks
    pid_hb = pid_z % num_h_blocks

    # --------------------------------------------------------------------- #
    # tile offsets
    # --------------------------------------------------------------------- #
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k_offset = tl.arange(0, BLOCK_K)
    h_offset = pid_hb * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_h = h_offset < H

    # accumulator for the tile, float32 for precision
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # iterate over the K dimension (the summed axis)
    # --------------------------------------------------------------------- #
    cur_k = 0
    while cur_k < N:
        k = cur_k + k_offset
        mask_k = k < N

        # ----------------- load left tile -----------------
        left_ptrs = left_ptr \
            + pid_b * stride_left_b \
            + i_offset[:, None, None] * stride_left_i \
            + k[None, :, None] * stride_left_k \
            + h_offset[None, None, :] * stride_left_h
        mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        left = tl.load(left_ptrs, mask=mask_left, other=0.0, dtype=tl.float16).to(tl.float32)

        # ----------------- load right tile ----------------
        right_ptrs = right_ptr \
            + pid_b * stride_right_b \
            + j_offset[:, None, None] * stride_right_j \
            + k[None, :, None] * stride_right_k \
            + h_offset[None, None, :] * stride_right_h
        mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        right = tl.load(right_ptrs, mask=mask_right, other=0.0, dtype=tl.float16).to(tl.float32)

        # ----------------- accumulate outer products over K -----------------
        # left: (BLOCK_I, BLOCK_K, BLOCK_H)
        # right: (BLOCK_J, BLOCK_K, BLOCK_H)
        for kk in range(0, BLOCK_K):
            # (BLOCK_I, BLOCK_H) * (BLOCK_J, BLOCK_H) -> (BLOCK_I, BLOCK_J, BLOCK_H)
            l = left[:, kk, :]                # (I, H)
            r = right[:, kk, :]               # (J, H)
            prod = l[:, None, :] * r[None, :, :]   # broadcast to (I, J, H)
            acc += prod
        cur_k += BLOCK_K

    # --------------------------------------------------------------------- #
    # store the result
    # --------------------------------------------------------------------- #
    out_ptrs = out_ptr \
        + pid_b * stride_out_b \
        + i_offset[:, None, None] * stride_out_i \
        + j_offset[None, :, None] * stride_out_j \
        + h_offset[None, None, :] * stride_out_h
    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul)
    using a custom Triton kernel for the heavy N³ contraction.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on input -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----- Linear projections -------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Apply mask ---------------------------------------------------------
    mask_unsq = mask.unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ----- Gating -------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to fp16 for the Triton kernel -------------------------------
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_f16.shape

    # ----- Allocate output tensor (float32) -----------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_f16.device)

    # ----- Strides ------------------------------------------------------------
    s_lb, s_li, s_lk, s_lh = left_f16.stride()
    s_rb, s_rj, s_rk, s_rh = right_f16.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # ----- Tiling parameters --------------------------------------------------
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 32  # hidden‑dim block size

    # Adjust for tiny hidden dimensions
    if hidden_dim < BLOCK_H:
        BLOCK_H = hidden_dim

    num_h_blocks = triton.cdiv(hidden_dim, BLOCK_H)

    # ----- Grid dimensions ----------------------------------------------------
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid_z = B * num_h_blocks
    grid = (grid_i, grid_j, grid_z)

    # ----- Launch kernel -------------------------------------------------------
    tri_mul_kernel[grid](
        left_f16,
        right_f16,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_rj,          # j‑stride is the same as i‑stride for shape (B,N,N,H)
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        num_h_blocks,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ----- LayerNorm on hidden dim -------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ----- Apply output gate ---------------------------------------------------
    out = out * out_gate

    # ----- Final linear projection --------------------------------------------
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
