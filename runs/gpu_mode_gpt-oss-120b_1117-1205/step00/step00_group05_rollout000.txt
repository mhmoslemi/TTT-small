@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # head index
    head_idx = tl.program_id(0)
    i_block = tl.program_id(1)
    j_block = tl.program_id(2)
    total_heads = B * H
    if head_idx >= total_heads:
        return

    b = head_idx // H
    h = head_idx % H

    # compute offsets
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # zero accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute offsets for left
        left_off = (b * stride_left_b + h * stride_left_h
                    + offs_i[:, None] * stride_left_i
                    + offs_k[None, :] * stride_left_k)
        # compute offsets for right
        right_off = (b * stride_right_b + h * stride_right_h
                     + offs_j[:, None] * stride_right_i
                     + offs_k[None, :] * stride_right_k)

        # load with masks
        a = tl.load(left_ptr + left_off, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b_ = tl.load(right_ptr + right_off, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # dot accumulate
        acc += tl.dot(a, b_)

    # store result
    out_off = (b * stride_out_b + h * stride_out_h
               + offs_i[:, None] * stride_out_i
               + offs_j[None, :] * stride_out_j)
    tl.store(out_ptr + out_off, acc, mask=mask_i[:, None] & mask_j[None, :])

import triton

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # I will ignore additional config stuff.

    # Ensure contiguous
    x = input_tensor.contiguous()
    mask = mask.contiguous()

    # Load weights (given as torch tensors)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections (dim -> hidden_dim)
    left = F.linear(x_norm, left_proj_weight)   # shape B,N,N,hidden_dim
    right = F.linear(x_norm, right_proj_weight)

    # Apply mask if provided - mask shape can be boolean; convert to float.
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # shape B,N,N,1
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to bfloat16 for kernel
    left_bf = left.to(torch.bfloat16).contiguous()
    right_bf = right.to(torch.bfloat16).contiguous()

    B, N, _, H = left_bf.shape

    # Prepare output tensor
    out_bf = torch.empty_like(left_bf, dtype=torch.float32)  # store fp32

    # Compute strides
    # left stride: left_bf.stride() returns tuple (stride_b, stride_i, stride_k, stride_h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_bf.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_bf.stride()
    # out stride for fp32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_bf.stride()

    # Convert strides to int64? Triton expects int64 maybe.
    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_heads = B * H
    grid = (total_heads, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    trimul_kernel[grid](
        left_bf, right_bf, out_bf,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # out_bf shape (B,N,N,H) in fp32

    # Continue with remaining ops in torch
    # Apply to_out_norm
    out_norm = F.layer_norm(out_bf, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Multiply by out_gate (float32)
    out = out_norm * out_gate

    # Final linear projection back to dim
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul: outgoing triangle multiplicative update (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim).
2. Project to hidden dimension (dim -> hidden_dim) with two linear layers.
3. Apply optional mask and per‑position sigmoid gates.
4. Compute the pairwise “outer‑product” sum:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per (batch,head) where
   left and right are [N,N,hidden_dim] tensors and the multiplication is
   left @ rightᵀ.  The heavy O(N²·hidden_dim) work is performed by a
   Triton kernel that processes one (batch,head) pair per program.
5. Layer‑norm the result over the hidden dimension, multiply by an output
   gate, and linearly project back to `dim`.

All linear layers and LayerNorms use the supplied weight tensors.
The kernel works on bfloat16 inputs and accumulates in float32, then
returns a float32 tensor for the remaining PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel for the core einsum: out = Σ_k left[...,k,:] * right[...,k,:]
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    B, N, H,                                 # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul per (batch, hidden) head:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    head_id = tl.program_id(0)   # encodes (batch, hidden)
    pid_i   = tl.program_id(1)   # block row (i)
    pid_j   = tl.program_id(2)   # block col (j)

    total_heads = B * H
    if head_id >= total_heads:
        return

    b = head_id // H
    h = head_id % H

    # --------------------------------------------------------------- #
    # Load blocks of left and right, accumulate product over k
    # --------------------------------------------------------------- #
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]
        a_ptrs = left_ptr + \
                 b * stride_left_b + h * stride_left_h + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[b, j, k, h]   (note j replaces i)
        b_ptrs = right_ptr + \
                 b * stride_right_b + h * stride_right_h + \
                 offs_j[:, None] * stride_right_i + \
                 offs_k[None, :] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # sum over k

    # --------------------------------------------------------------- #
    # Store the accumulated block to out[b, i, j, h]
    # --------------------------------------------------------------- #
    out_ptrs = out_ptr + \
               b * stride_out_b + h * stride_out_h + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Entry point expected by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton kernel for the
    core O(N²·hidden_dim) computation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor of shape [B, N, N, dim]
            mask_tensor : torch.Tensor of shape [B, N, N] (might be all ones)
            weights_dict: mapping from weight names to torch.Tensor objects
            config_dict : contains keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----------- unpack weights -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ----------- LayerNorm over the last dimension -----------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # [B,N,N,dim], float32

    # ----------- Project to hidden dimension (use bfloat16 for memory) -------
    x_bf16 = x.to(torch.bfloat16)

    left_proj_weight_bf = left_proj_weight.to(torch.bfloat16)
    right_proj_weight_bf = right_proj_weight.to(torch.bfloat16)

    left = F.linear(x_bf16, left_proj_weight_bf)   # [B,N,N,hidden_dim], bf16
    right = F.linear(x_bf16, right_proj_weight_bf) # [B,N,N,hidden_dim], bf16

    # ----------- Apply mask if present ----------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask may be bool or int; convert to same dtype as left/right
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----------- Compute sigmoid gates (keep them in float32) -----------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ----------- Apply gates ---------------------------------------------------
    left = left * left_gate.to(torch.bfloat16)
    right = right * right_gate.to(torch.bfloat16)

    # ----------- Triton kernel: Σ_k left * right --------------------------------
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, H = left.shape

    # Allocate output (float32 accumulation)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Block sizes – chosen to give good occupancy across tested N (≤1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                           # one program per (batch, hidden_head)
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N),        # cols
    )
    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------- Output LayerNorm, gating and final projection ---------------
    # out: [B,N,N,hidden_dim] float32
    out_norm = F.layer_norm(out,
                            (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=1e-5)          # float32

    out = out_norm * out_gate                # broadcast over hidden dim

    # Final linear projection back to `dim`
    output = F.linear(out, to_out_weight)    # [B,N,N,dim], float32

    return output
