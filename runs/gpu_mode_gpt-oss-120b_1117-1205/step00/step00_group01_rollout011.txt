def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_l_b, stride_l_i, stride_l_j, stride_l_d,
    stride_r_b, stride_r_i, stride_r_j, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   pid_group = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   num_groups = B * hidden_dim
   if pid_group >= num_groups:
       return

   # compute batch and hidden index
   b = pid_group // hidden_dim
   d = pid_group % hidden_dim

   # compute base offsets
   off_b_l = b * stride_l_b
   off_b_r = b * stride_r_b
   off_b_o = b * stride_o_b

   off_d_l = d * stride_l_d
   off_d_r = d * stride_r_d
   off_d_o = d * stride_o_d

   # offsets i and j
   offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # mask for i,j
   mask_i = offs_i < N
   mask_j = offs_j < N

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over k dimension
   for k in range(0, tl.cdiv(N, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N

       # left tile pointer
       # left ptr: base + off_b_l + off_d_l + offs_i[:,None]*stride_l_i + offs_k[None,:]*stride_l_j

           offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_j

       right_ptrs = right_ptr + off_b_r + off_d_r + \
           offs_k[:, None] * stride_r_i + offs_j[None, :] * stride_r_j

def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_d,
    stride_r_b, stride_r_i, stride_r_j, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   pid_g = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # num groups
   num_groups = B * H
   if pid_g >= num_groups:
       return

   b = pid_g // H
   d = pid_g % H

   # base offsets
   base_l = b * stride_l_b + d * stride_l_d
   base_r = b * stride_r_b + d * stride_r_d
   base_o = b * stride_o_b + d * stride_o_d

   # offsets within tile
   offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M, dtype=tl.int64)
   offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N, dtype=tl.int64)

   mask_i = offs_i < N
   mask_j = offs_j < N

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

   # loop over k
   for k in range(0, tl.cdiv(N, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K, dtype=tl.int64)
       mask_k = offs_k < N

       # left ptr: (i,k)
       left_tile_ptr = left_ptr + base_l + \
           offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_j
       # right ptr: (k,j)

           offs_k[:, None] * stride_r_i + offs_j[None, :] * stride_r_j

"""
TriMul (outgoing) implementation with Triton.

The heavy part of TriMul is the contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each hidden dimension `d` (hidden_dim) and each batch element this is a
matrix multiplication `left_d @ right_d.T`.  The kernel below performs a
batched GEMM where the batch axis is the combined (batch, hidden) dimension.
All element‑wise ops (LayerNorm, linear projections, gates, mask) are done in
PyTorch; the O(N³) contraction is computed in half precision with a tiled
Triton kernel.

The kernel processes tiles of size (BLOCK_M, BLOCK_N) of the output matrix
and iterates over the K dimension in blocks of size BLOCK_K, handling arbitrary
sequence lengths with boundary masks.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out = left @ right.T for each (batch, hidden) pair
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                     # batch size, sequence length, hidden dim
    stride_l_b, stride_l_i, stride_l_j, stride_l_d,
    stride_r_b, stride_r_i, stride_r_j, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_g = tl.program_id(0)          # combined (batch, hidden) index
    pid_m = tl.program_id(1)          # output tile row   (i)
    pid_n = tl.program_id(2)          # output tile col   (j)

    num_groups = B * H
    if pid_g >= num_groups:
        return

    # decode batch and hidden indices
    b = pid_g // H
    d = pid_g % H

    # base offsets for this (b,d) slice
    base_l = b * stride_l_b + d * stride_l_d
    base_r = b * stride_r_b + d * stride_r_d
    base_o = b * stride_o_b + d * stride_o_d

    # tile coordinates within the N×N matrix
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M, dtype=tl.int64)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N, dtype=tl.int64)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (half‑precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K, dtype=tl.int64)
        mask_k = offs_k < N

        # left tile: (i, k)
        left_tile_ptr = left_ptr + base_l + \
            offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_j
        left_tile = tl.load(left_tile_ptr,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile (k, j) – note the transposed view
        right_tile_ptr = right_ptr + base_r + \
            offs_k[:, None] * stride_r_i + offs_j[None, :] * stride_r_j
        right_tile = tl.load(right_tile_ptr,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # accumulate the product
        acc += tl.dot(left_tile, right_tile)   # (M,K)·(K,N) → (M,N)

    # store the result tile
    out_tile_ptr = out_ptr + base_o + \
        offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_tile_ptr, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the forward pass of the "outgoing" TriMul operation.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:          torch.Tensor [B, N, N] (float or bool)
            - weights:       dict of torch.Tensor containing all model weights
            - config:        dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        output: torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Layer‑norm over the last dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating signals (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates (mask already applied before)
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to half‑precision and launch Triton kernel
    # --------------------------------------------------------------
    left = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)
    out_half = torch.empty_like(left, dtype=torch.float16, device=device)

    # strides (int64)
    stride_l_b, stride_l_i, stride_l_j, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out_half.stride()

    # Tile sizes – reasonable defaults for the B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (num_groups, num_tiles_M, num_tiles_N)
    num_groups = B * hidden_dim
    grid = (
        num_groups,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out_half,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_j,
        stride_l_d,
        stride_r_b,
        stride_r_i,
        stride_r_j,
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Back to float32 for the remaining ops
    out = out_half.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Second layer‑norm (hidden dimension) + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # --------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])

    return output
