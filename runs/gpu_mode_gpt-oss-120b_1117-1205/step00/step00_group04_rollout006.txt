def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype  # assume float32

    # LayerNorm on input (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply mask if provided
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
    else:
        mask = torch.ones_like(x[..., :1])  # (B,N,N,1)
    left = left * mask * left_gate
    right = right * mask * right_gate

    # Cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape

    total_batches = B * H
    M = N
    K = N
    N_out = N

    a = left_perm.view(total_batches, M, K).contiguous()
    b = right_T.view(total_batches, K, N_out).contiguous()
    c = torch.empty((total_batches, M, N_out), dtype=torch.float16, device=device)

    # Block sizes (tune)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total_batches, triton.cdiv(M, BLOCK_M), triton.cdiv(N_out, BLOCK_N))
    batched_matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        stride_a_batch=a.stride(0), stride_a_m=a.stride(1), stride_a_k=a.stride(2),
        stride_b_batch=b.stride(0), stride_b_k=b.stride(1), stride_b_n=b.stride(2),
        stride_c_batch=c.stride(0), stride_c_m=c.stride(1), stride_c_n=c.stride(2),
        M=M, N=N_out, K=K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden_dim)

    # Continue with to_out_norm, out_gate, to_out
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # LayerNorm over hidden_dim axis
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate (broadcast)
    out = out * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    # Ensure output dtype float32
    out = out.to(torch.float32)

    return out

"""
TriMul forward implementation with a custom Triton kernel.

The kernel computes the batched matrix multiplication required for the
einsum('... i k d, ... j k d -> ... i j d') operation:
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

We treat each (batch, hidden) pair as an independent matrix multiply of
shape (N, N) @ (N, N). The kernel tiles the M, N and K dimensions with
BLOCK_M, BLOCK_N and BLOCK_K work‑groups and accumulates in float32 while
the inputs are stored in float16 for bandwidth efficiency.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM C = A @ B, where B is already transposed."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K x N) – B is already the transposed operand
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # promote to FP32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # write back in FP16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul module, using a Triton kernel for the
    O(N³·D) multiplication. All other linear / norm / gating steps are
    performed with PyTorch operations.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor:  torch.Tensor [B, N, N] (bool/int)
            - weights_dict: mapping from weight name to torch.Tensor on the same device
            - config_dict:  must contain 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating values (+ optional mask)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # mask handling (mask may be None when config["nomask"] is True)
    if mask is None or config.get("nomask", False):
        mask_exp = torch.ones_like(x[..., :1], device=device, dtype=x.dtype)
    else:
        mask_exp = mask.to(dtype=x.dtype).unsqueeze(-1)  # [B,N,N,1]

    left = left * mask_exp * left_gate
    right = right * mask_exp * right_gate

    # ------------------------------------------------------------------
    # 4. Cast to half‑precision for the heavy GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (B, hidden_dim, N, N) and transpose RHS for the matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    right_t = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N] (K x N)

    B, H, N, _ = left_perm.shape
    total_batches = B * H
    M = N
    K = N
    N_out = N

    # flatten the leading two dimensions for the kernel
    a = left_perm.view(total_batches, M, K).contiguous()
    b = right_t.view(total_batches, K, N_out).contiguous()
    c = torch.empty((total_batches, M, N_out), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 5. Triton batched GEMM launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_out, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        stride_a_batch=a.stride(0),
        stride_a_m=a.stride(1),
        stride_a_k=a.stride(2),
        stride_b_batch=b.stride(0),
        stride_b_k=b.stride(1),
        stride_b_n=b.stride(2),
        stride_c_batch=c.stride(0),
        stride_c_m=c.stride(1),
        stride_c_n=c.stride(2),
        M=M,
        N=N_out,
        K=K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to (B, N, N, hidden_dim) in float32
    # ------------------------------------------------------------------
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output layer‑norm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # final projection back to dim

    # ensure final dtype is float32
    return out.to(torch.float32)
