TriMul (outgoing) forward pass with a Triton kernel for the O(N³) contraction.

The kernel computes, for each batch b, pair (i, j) and hidden channel d:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All other operations (LayerNorm, Linear projections, gates, final Linear) are
performed in PyTorch.  The kernel works with arbitrary batch size, sequence
length and hidden dimension, handling non‑multiple block sizes via masking.
"""

def tri_mul_kernel(

    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr
):

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k in tiles
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_outer * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # offsets for left[b, i, k, h]
        left_offsets = (
            pid_b * stride_lb
            + offs_i[:, None, None] * stride_li
            + offs_k[None, :, None] * stride_lk
            + offs_h[None, None, :] * stride_lh
        )
        left_tile = tl.load(
            left_ptr + left_offsets,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # offsets for right[b, j, k, h]
        right_offsets = (
            pid_b * stride_rb
            + offs_j[:, None, None] * stride_ri
            + offs_k[None, :, None] * stride_rk
            + offs_h[None, None, :] * stride_rh
        )
        right_tile = tl.load(
            right_ptr + right_offsets,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # compute contribution for each hidden sub‑block
        for hi in range(0, BLOCK_H):
            # (BLOCK_I, BLOCK_K) dot (BLOCK_J, BLOCK_K)^T -> (BLOCK_I, BLOCK_J)
            prod = tl.dot(left_tile[:, :, hi], right_tile[:, :, hi], trans_b=True)
            acc[:, :, hi] += prod

    # store result
    out_offsets = (
        pid_b * stride_ob
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + offs_h[None, None, :] * stride_oh
    )
    tl.store(
        out_ptr + out_offsets,
        acc.to(tl.float32),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ------------------------ custom_kernel entry ------------------------ #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data: (input_tensor, mask, weights, config)
            input_tensor – [B, N, N, dim] float32
            mask         – [B, N, N] (broadcastable to the last dim)
            weights      – dict with all linear / norm parameters
            config       – {'dim': int, 'hidden_dim': int}
    Returns:
        output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, N2, D = input_tensor.shape
    assert N == N2 and D == dim, "unexpected input shape"

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    mask_exp = mask.unsqueeze(-1).type_as(x)  # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    # ---------- Gates ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguity for stride extraction
    left = left.contiguous()
    right = right.contiguous()

    # ---------- Core contraction with Triton ----------
    out = torch.empty_like(left)  # same dtype (float32) and shape

    # tensor strides (in elements)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes (tuned for typical dimensions)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B,
    )
    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑processing ----------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out_norm * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
