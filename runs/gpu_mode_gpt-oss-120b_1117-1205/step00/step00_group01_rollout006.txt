```
def custom_kernel(data):
    """
    TriMul outgoing forward pass.

    Implements the heavy O(N^3) einsum using a custom Triton kernel that
    performs a batched matrix multiplication for each hidden dimension.
    All other operations (layernorm, linear projections, gating, final
    projection) are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present
    if not nomask:
        # mask shape (B,N,N) -> (B,N,N,1)
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # broadcast
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Move to half precision for the O(N^3) part
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape

    # Prepare A (B,H,N,N) -> (B*H,N,N)
    A = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Prepare B_T (B*H,N,N) where inner dimensions are transposed
    B_T = right_h.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Output tensor
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=input_tensor.device)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid definition
    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_blocks = (B * H) * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    # Launch kernel
    batched_matmul_kernel[grid](
        A, B_T, C,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # Reshape back to (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1)

    # Final LayerNorm, gating, linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias)
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

"""
TriMul outgoing implementation.

The heavy O(N³) part (the einsum over the third axis) is rewritten as a
batched matrix multiplication C = A @ Bᵀ where for each hidden dimension d

    A[b, d, i, k] = left  (i, k, d)
    Bᵀ[b, d, k, j] = right (j, k, d)

Thus C[b, d, i, j] = Σₖ left[b, i, k, d] * right[b, j, k, d]
The batched GEMM is executed by a custom Triton kernel that works on fp16
inputs with fp32 accumulation. All other operations (LayerNorms, linear
projections, sigmoids, masking) are performed with PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[batch, m, n] = A[batch, m, k] @ B[batch, k, n]  (row‑major, fp16→fp32)."""
    pid = tl.program_id(0)

    # number of blocks in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = (
            A_ptr
            + batch_idx * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + batch_idx * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # tl.dot does (M,K)·(K,N) → (M,N)
        acc = acc + tl.dot(a, b).to(tl.float32)

    # write back
    c_ptrs = (
        C_ptr
        + batch_idx * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seqlen, seqlen, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm on the input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # (B, N, N, dim)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B, N, N, hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)  # (B, N, N, hidden)

    # ---------- Optional masking ----------
    if not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) broadcast over hidden
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for the batched GEMM ----------
    # Use fp16 for the heavy multiplication
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape  # B, seq_len, seq_len, hidden_dim

    # A: (B, H, N, N) -> (B*H, N, N)   rows = i, cols = k
    A = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # B_T: (B, H, N, N) where inner dimensions are transposed (k, j)
    B_T = right_h.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Output placeholder
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=input_tensor.device)

    # Strides (row‑major)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # ------------ Triton launch parameters ------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_blocks = (B * H) * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    # Launch the kernel
    batched_matmul_kernel[grid](
        A,
        B_T,
        C,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ---------- Reshape back to (B, N, N, hidden) ----------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1)  # (B, N, N, hidden)

    # ---------- Final LayerNorm, gating, and projection ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)  # (B, N, N, dim)

    return output
