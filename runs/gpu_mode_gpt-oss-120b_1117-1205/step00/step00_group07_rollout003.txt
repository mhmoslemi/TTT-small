@triton.jit
def gate_mask_kernel(left_ptr, left_gate_ptr, mask_ptr, out_ptr,
                     B, N, D,
                     stride_left_b, stride_left_i, stride_left_j, stride_left_d,
                     stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d,
                     stride_mask_b, stride_mask_i, stride_mask_j,
                     BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N * D
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    # compute b, i, j, d from offs
    # compute linear to multi-index
    # Compute
    # off = offs // D gives idx_ijk (0..B*N*N -1)
    # d = offs % D
    idx_ijk = offs // D
    d = offs % D

    # Now compute b, i, j
    # idx_ijk = b * N * N + i * N + j
    b = idx_ijk // (N * N)
    rem = idx_ijk % (N * N)
    i = rem // N
    j = rem % N

    # Compute address offsets:
    left_off = b * stride_left_b + i * stride_left_i + j * stride_left_j + d * stride_left_d
    left_gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + d * stride_gate_d
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    left = tl.load(left_ptr + left_off, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + left_gate_off, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=0.0) # shape broadcast
    out = left * left_gate * mask_val
    tl.store(out_ptr + left_off, out, mask=mask)

    B, N, D,

"""
TriMul (outgoing) forward pass.

The heavy O(N³) contraction `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`
is rewritten as a batched matrix multiplication:
    out = (left_reshaped) @ (right_reshaped)ᵀ
where each hidden channel is treated as an independent batch.
We fuse the elementwise mask and gating operations for `left` and `right`
into a small Triton kernel, then use PyTorch's highly‑optimized batched GEMM
(`torch.bmm`) for the contraction. All LayerNorms and linear projections are
performed with PyTorch functional APIs. The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: apply gating and optional mask (mask broadcasted on hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    a_ptr,                # pointer to left or right projection (float32)
    a_gate_ptr,           # pointer to the corresponding gate (float32)
    mask_ptr,             # pointer to mask (float32, shape [B,N,N])
    out_ptr,              # pointer to output tensor (float32)
    B, N, D,              # sizes: batch, sequence length, hidden dim
    stride_a_b, stride_a_i, stride_a_j, stride_a_d,   # strides of a
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d,  # strides of gate
    stride_mask_b, stride_mask_i, stride_mask_j,    # strides of mask
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N * D
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_boundary = offs < total

    # Decode linear index into (b, i, j, d)
    d = offs % D
    idx_ijk = offs // D
    b = idx_ijk // (N * N)
    rem = idx_ijk % (N * N)
    i = rem // N
    j = rem % N

    # Compute memory offsets
    a_off = b * stride_a_b + i * stride_a_i + j * stride_a_j + d * stride_a_d
    a_gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + d * stride_gate_d
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    a_val = tl.load(a_ptr + a_off, mask=mask_boundary, other=0.0)
    a_gate_val = tl.load(a_gate_ptr + a_gate_off, mask=mask_boundary, other=0.0)
    mask_val = tl.load(mask_ptr + mask_off, mask=mask_boundary, other=0.0)

    out_val = a_val * a_gate_val * mask_val
    tl.store(out_ptr + a_off, out_val, mask=mask_boundary)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: Tensor[B, N, N, dim] (already on CUDA)
            - mask: Tensor[B, N, N] (binary mask, may be all ones)
            - weights: dict with all linear and layer‑norm parameters
            - config: dict with "dim" and "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    # Ensure float32 computation
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.to(input_tensor.dtype) if mask is not None else torch.ones(
        input_tensor.shape[:3], dtype=input_tensor.dtype, device=device
    )
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gate linear layers + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # Fuse mask and gating using Triton
    # ------------------------------------------------------------------
    # Make tensors contiguous for predictable strides
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    # Allocate output buffers for gated tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Triton launch parameters
    BLOCK = 1024
    total_elements = bs * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK),)

    # Launch for left side
    gate_mask_kernel[grid](
        left_proj,
        left_gate,
        mask,
        left,
        bs,
        N,
        hidden_dim,
        *left_proj.stride(),
        *left_gate.stride(),
        *mask.stride(),
        BLOCK=BLOCK,
    )
    # Launch for right side
    gate_mask_kernel[grid](
        right_proj,
        right_gate,
        mask,
        right,
        bs,
        N,
        hidden_dim,
        *right_proj.stride(),
        *right_gate.stride(),
        *mask.stride(),
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # Batched contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # Implemented as batched matrix multiplication per hidden channel
    # ------------------------------------------------------------------
    # Rearrange to (B*hidden_dim, N, N) where each row corresponds to one channel
    left_reshaped = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

    # Multiply left with transpose of right across the K dimension
    out_mat = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # Restore shape (B, N, N, hidden_dim)
    out = out_mat.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # Second LayerNorm (hidden_dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
