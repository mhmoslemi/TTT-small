def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Compute linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask
    if mask is not None:
        # mask shape [bs, N, N]; unsqueeze last dimension
        mask_exp = mask.unsqueeze(-1).type_as(left)  # cast to float
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # allocate output half
    out_half = torch.empty_like(left_half)  # shape [B,N,N,hidden_dim] float16

    # Triton kernel launch
    import triton
    import triton.language as tl

    # define kernel at top-level (outside custom_kernel) for compilation
    # We'll define it separate before custom_kernel.

    # compute strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left_half.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right_half.stride()
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h = out_half.stride()
    # out half shape is same as left/right

    # Ensure kernel compiled
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), bs * hidden_dim)

    tri_mul_matmul_kernel[grid](
        left_half, right_half, out_half,
        bs, N, hidden_dim,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_k, stride_c_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Now out_half shape [B,N,N,hidden_dim] (float16). Convert to float32
    out = out_half.to(torch.float32)

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Multiply out_gate (float32)
    out = out * out_gate

    # Final projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul "outgoing" implementation with a Triton kernel for the O(N³) contraction.
The kernel computes, for every batch `b` and hidden channel `h`:

    out[b, i, j, h] = sum_{k} left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are the linearly projected and gated tensors of shape
[bs, N, N, hidden_dim].  All other operations (LayerNorm, Linear, gates)
are performed in PyTorch, while the heavy contraction is off‑loaded to Triton.
The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    batch, N, hidden_dim,               # sizes
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul contraction:
        C[b,h,i,j] = A[b,h,i,k] @ B[b,h,j,k]^T
    Here `a_ptr` points to tensor with shape (batch, N, N, hidden_dim),
    `b_ptr` the same shape, and `c_ptr` the destination.
    The kernel iterates over tiles of size BLOCK_M×BLOCK_N and accumulates
    over the K dimension.
    """
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_hb = tl.program_id(2)         # combined batch*hidden index

    # decode batch and hidden index
    b_idx = pid_hb // hidden_dim
    h_idx = pid_hb % hidden_dim

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load A tile (i,k) ----
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # (BLOCK_M, BLOCK_K)

        # ---- load B tile transposed: we need B_T[k,j] = B[j,k] ----
        # original B layout is (batch, N, N, hidden) -> (i,k)
        # B_T pointer uses k as first dimension, j as second.
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)          # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                               # (BLOCK_M, BLOCK_N)

    # ---- store result ----
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" update using a Triton kernel for
    the expensive O(N³) contraction.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --- LayerNorm on the input ---
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --- Linear projections ---
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --- Apply mask (optional) ---
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)      # shape [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --- Gates ---
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --- Apply gates ---
    left = left * left_gate
    right = right * right_gate

    # Cast to half for the Triton kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h = torch.empty_like(left_h)  # will hold [bs, N, N, hidden_dim] in half

    # --- Strides for kernel (in elements) ---
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left_h.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right_h.stride()
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h = out_h.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),          # i dimension
        triton.cdiv(N, BLOCK_N),          # j dimension
        bs * hidden_dim,                  # combined batch*hidden
    )

    tri_mul_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        bs,
        N,
        hidden_dim,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_k,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion
    torch.cuda.synchronize()

    # Convert back to float32 for the remaining layers
    out = out_h.to(torch.float32)

    # --- Second LayerNorm (hidden dimension) ---
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --- Apply output gate ---
    out = out * out_gate

    # --- Final linear projection back to original dimension ---
    out = F.linear(out, weights["to_out.weight"])

    return out
