def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Move to correct dtype maybe float32.
    x = input_tensor  # [B,N,N,dim]
    # Ensure contiguous.
    x = x.contiguous()
    bs, N, _, dim = x.shape  # we assume shape is symmetric so second dim equals third.
    hidden_dim = config["hidden_dim"]

    # Determine if mask is provided; maybe just treat mask as torch.ones_like if missing.
    if mask is None or mask.numel() == 0:
        mask = torch.ones((bs, N, N), dtype=x.dtype, device=x.device)

    mask = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]

    # LayerNorm: x_norm = (x - mean)/sqrt(var+eps) * weight + bias
    eps = 1e-5
    # Use torch.nn.functional.layer_norm which accepts weight,bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Compute linear projections:
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask to left/right
    left = left * mask
    right = right * mask

    # Gates left,right,out
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gate to left/right
    left = left * left_gate
    right = right * right_gate

    # Core multiplication using Triton

@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile col
    pid_z = tl.program_id(2)  # batch hidden index

    bh = pid_z
    b = bh // H
    h = bh % H

    # compute start offsets for i and j
    i = pid_i * BLOCK_M
    j = pid_j * BLOCK_N

    # offsets
    offs_i = i + tl.arange(0, BLOCK_M)
    offs_j = j + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile pointer
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            h * stride_left_h + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        # right tile pointer for transposed (k, j)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            h * stride_right_h + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_i

        # combine masks for loading: for left, rows i valid and k valid; for right, rows k valid and columns j valid
        mask_left = mask_i[:, None] & mask_k[None, :]
        mask_right = mask_k[:, None] & mask_j[None, :]

        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)

        # Convert to float32 if needed (assuming stored as float32)
        # Use dot
        acc += tl.dot(left_tile, right_tile)

    # store
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        h * stride_out_h + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j

    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)

"""
TriMul "outgoing" implementation with a custom Triton kernel.
The heavy O(N³) interaction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is executed as a batched matrix multiplication (N×N)·(N×N)ᵀ for each
hidden channel `d`.  All other operations (LayerNorm, linear projections,
gates, final LayerNorm and linear) are performed with PyTorch.  The Triton
kernel processes one (batch, hidden) pair per program and tiles the
output in blocks of (BLOCK_M, BLOCK_N) while reducing over the `k` dimension
with a tile size BLOCK_K.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------------------- #
@triton.jit
def trmul_kernel(
    left_ptr,          # *float32   left  [B, N, N, H]
    right_ptr,         # *float32   right [B, N, N, H]
    out_ptr,           # *float32   out   [B, N, N, H]
    B, N, H,           # tensor dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile row index
    pid_j = tl.program_id(1)          # tile column index
    pid_z = tl.program_id(2)          # index over (batch * hidden)

    # Decode batch and hidden indices
    bh = pid_z
    b = bh // H
    h = bh % H

    # Starting coordinates of the tile
    i = pid_i * BLOCK_M
    j = pid_j * BLOCK_N

    # Range of i and j indices for this tile
    offs_i = i + tl.arange(0, BLOCK_M)
    offs_j = j + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the block (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for the left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        mask_left = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0, dtype=tl.float32)

        # Pointers for the right tile (transposed on k): shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + h * stride_right_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        mask_right = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0, dtype=tl.float32)

        # Matrix multiply the two tiles and accumulate
        acc += tl.dot(left_tile, right_tile)

    # Store the computed block to the output tensor
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


# --------------------------------------------------------------------------- #
# Entry point used by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:        torch.Tensor [B, N, N] (float32 or bool)
            - weights: dict of torch.Tensor containing all model weights
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # 1. Initial LayerNorm on the input
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Apply mask (mask may be all‑ones when nomask=True)
    # ------------------------------------------------------------------- #
    if mask.dtype != dtype:
        mask = mask.to(dtype)
    mask = mask.unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------- #
    # 4. Gates
    # ------------------------------------------------------------------- #
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core TriMul (batched GEMM per hidden channel) via Triton
    # ------------------------------------------------------------------- #
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides for kernel arguments
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Block sizes (tuned for typical N; the kernel works for any size)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * hidden_dim,            # (batch, hidden) dimension
    )

    trmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
