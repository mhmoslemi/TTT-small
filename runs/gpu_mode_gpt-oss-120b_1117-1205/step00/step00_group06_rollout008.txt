def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm: x_norm = (input - mean)/sqrt(var+eps)*norm_weight + norm_bias
    # We can do torch.nn.functional.layer_norm.
    # The normalized shape is dim, weight and bias given by weights.
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,),
                                       weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Allocate output
    out_half = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Kernel constants
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Compute strides (in elements)
    left_stride_b, left_stride_i, left_stride_k, left_stride_d = left_half.stride()
    right_stride_b, right_stride_i, right_stride_k, right_stride_d = right_half.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_d = out_half.stride()

    # Compute grid sizes
    grid_i = triton.cdiv(N, BLOCK_M)

        out_stride_b, out_stride_i, out_stride_j, out_stride_d,

    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_b_left, stride_i_left, stride_k_left, stride_d_left,
    stride_b_right, stride_i_right, stride_k_right, stride_d_right,
    stride_b_out, stride_i_out, stride_j_out, stride_d_out,

    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)
    b = pid_bd // D
    d = pid_bd % D

    # compute offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    # Offsets within block
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

        offs_k = k_base + tl.arange(0, BLOCK_K)

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_b_left, stride_i_left, stride_k_left, stride_d_left,
    stride_b_right, stride_i_right, stride_k_right, stride_d_right,
    stride_b_out, stride_i_out, stride_j_out, stride_d_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    b = pid_bd // D
    d = pid_bd % D

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # Offsets inside block
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # Initialize accumulator with FP32 precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Masks for bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # Load left block (i,k) => (BLOCK_M, BLOCK_K)
        ptrs_left = left_ptr + b * stride_b_left + d * stride_d_left + \
                    offs_i[:, None] * stride_i_left + offs_k[None, :] * stride_k_left
        left = tl.load(ptrs_left, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block (k,j) => (BLOCK_K, BLOCK_N)
        ptrs_right = right_ptr + b * stride_b_right + d * stride_d_right + \
                     offs_k[:, None] * stride_k_right + offs_j[None, :] * stride_i_right
        right = tl.load(ptrs_right, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(left, right)

    # Store result in FP16
    ptrs_out = out_ptr + b * stride_b_out + d * stride_d_out + \
               offs_i[:, None] * stride_i_out + offs_j[None, :] * stride_j_out
    mask_ij = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(ptrs_out, acc.to(tl.float16), mask=mask_ij)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    Performs LayerNorm, linear projections, gating, the cubic
    `einsum('... i k d, ... j k d -> ... i j d')` using a Triton
    kernel, followed by a second LayerNorm, gating and final linear.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm on the last dimension (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if provided (mask may be bool or float)
    if mask is not None:
        if mask.dtype != x.dtype:
            mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for the cubic kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    out_half = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Kernel launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Strides (in elements)
    lsb, lsi, lsk, lsd = left_half.stride()
    rsb, rsi, rsk, rsd = right_half.stride()
    osb, osi, osj, osd = out_half.stride()

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_bd = B * hidden_dim

    tri_mul_gemm_kernel[(grid_i, grid_j, grid_bd)](
        left_half,
        right_half,
        out_half,
        B, N, hidden_dim,
        lsb, lsi, lsk, lsd,
        rsb, rsi, rsk, rsd,
        osb, osi, osj, osd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 for the rest of the pipeline
    out = out_half.to(torch.float32)

    # Second LayerNorm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dimension
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    out_ptr,           # *fp16
    B, N, D,           # batch, seq_len, hidden_dim
    stride_b_left, stride_i_left, stride_k_left, stride_d_left,
    stride_b_right, stride_i_right, stride_k_right, stride_d_right,
    stride_b_out, stride_i_out, stride_j_out, stride_d_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    for a single (b,d) pair using a tiled GEMM.
    """
    pid_i = tl.program_id(0)          # i‑block
    pid_j = tl.program_id(1)          # j‑block
    pid_bd = tl.program_id(2)         # combined batch‑and‑hidden index

    b = pid_bd // D
    d = pid_bd % D

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # left: (i,k)  → (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_b_left
            + d * stride_d_left
            + offs_i[:, None] * stride_i_left
            + offs_k[None, :] * stride_k_left
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right: (k,j) → (BLOCK_K, BLOCK_N)  (note transposed view)
        right_ptrs = (
            right_ptr
            + b * stride_b_right
            + d * stride_d_right
            + offs_k[:, None] * stride_k_right
            + offs_j[None, :] * stride_i_right
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # store result (convert back to fp16)
    out_ptrs = (
        out_ptr
        + b * stride_b_out
        + d * stride_d_out
        + offs_i[:, None] * stride_i_out
        + offs_j[None, :] * stride_j_out
    )
    mask_ij = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_ij)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.
    Steps:
      1. LayerNorm on the input.
      2. Linear projections + gating (all in PyTorch, float32).
      3. Masking.
      4. Cubic contraction `einsum('... i k d, ... j k d -> ... i j d')`
         performed as a batch of GEMMs inside a Triton kernel.
      5. Second LayerNorm, output gate, and final linear projection.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply mask (if any)
    # -----------------------------------------------------------------
    if mask is not None:
        if mask.dtype != x.dtype:
            mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Cubic contraction via Triton (half‑precision for speed)
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # strides (in elements)
    lsb, lsi, lsk, lsd = left_h.stride()
    rsb, rsi, rsk, rsd = right_h.stride()
    osb, osi, osj, osd = out_h.stride()

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_bd = B * hidden_dim

    tri_mul_gemm_kernel[
        (grid_i, grid_j, grid_bd)
    ](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        lsb,
        lsi,
        lsk,
        lsd,
        rsb,
        rsi,
        rsk,
        rsd,
        osb,
        osi,
        osj,
        osd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Second LayerNorm, output gate, final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
