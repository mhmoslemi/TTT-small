@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # compute program ids
    pid0 = tl.program_id(0)  # combine group and block_i
    pid1 = tl.program_id(1)  # block_j

    num_blocks_i = tl.cdiv(N, BLOCK_M)
    g = pid0 // num_blocks_i
    block_i = pid0 % num_blocks_i

    # decode group to batch and h
    b = g // H
    d = g % H

    # compute base offsets
    base_left = b * stride_left_batch + d * stride_left_h
    base_right = b * stride_right_batch + d * stride_right_h
    base_out = b * stride_out_batch + d * stride_out_h

    # offsets for block i and j
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for i and j (valid range)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate result
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load left tile
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # compute pointer offsets
        left_ptrs = left_ptr + base_left + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # load with mask
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k[None,:] < N), other=0.0)

        # load right tile (transposed)
        # here we want right_T[k,j] = right[j,k]
        # so we need offsets: j index (offs_j) in stride_i, k index (offs_k) in stride_k
        right_ptrs = right_ptr + base_right + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k
        # Note: shape right_tile should be (BLOCK_N, BLOCK_K), need to transpose to (BLOCK_K,BLOCK_N)
        # We can load as (BLOCK_N, BLOCK_K) then transpose using .T

        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & (offs_k[None,:] < N), other=0.0)
        # transpose:
        right_tile_T = tl.trans(right_tile) # shape (BLOCK_K, BLOCK_N)

        # compute dot
        acc += tl.dot(left_tile, right_tile_T)

    # store results to out
    out_ptrs = out_ptr + base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16) if (out_ptr.dtype == tl.float16) else acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    num_blocks_i = tl.cdiv(N, BLOCK_M)
    g = pid0 // num_blocks_i
    block_i = pid0 % num_blocks_i

    b = g // H
    d = g % H

    # base offsets
    base_left = b * stride_left_batch + d * stride_left_h
    base_right = b * stride_right_batch + d * stride_right_h
    base_out = b * stride_out_batch + d * stride_out_h

    # i and j offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile [M,K]
        left_ptrs = left_ptr + base_left + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float32)

        # right tile [N,K] (original right shape), then transpose to [K,N]
        right_ptrs = right_ptr + base_right + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k
        b_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0, dtype=tl.float32)
        b_tile_T = tl.trans(b_tile)  # shape (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b_tile_T)

    # store result
    out_ptrs = out_ptr + base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # LayerNorm over last dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask
    mask_unsq = mask.unsqueeze(-1).type_as(x)
    left = left * mask_unsq
    right = right * mask_unsq

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute core multiplication via triton kernel
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim], dtype float32

    # compute strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()
    # Convert to python ints
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = stride_left
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h = stride_right
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = stride_out

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    # number of groups = B * hidden_dim
    num_groups = B * hidden_dim

    grid = (num_groups * num_blocks_i, num_blocks_j)
    tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
        stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # After kernel, out shape same as left/right
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate
    out_norm = out_norm * out_gate

    # to_out linear
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out_norm, to_out_weight)

    return output
```

"""
TriMul (outgoing) implementation with a custom Triton kernel.

The core operation is:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which can be expressed as a batched matrix multiplication:
    out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T

We compute the layer‑norms, linear projections and gating in PyTorch,
apply the binary mask, then perform the massive N³ contraction with a
Triton kernel that tiles the i, j and k dimensions (BLOCK_M, BLOCK_N,
BLOCK_K). The kernel processes each (batch, hidden_dim) pair as an
independent GEMM and writes the result back into a tensor of shape
[batch, seq_len, seq_len, hidden_dim]. The rest of the pipeline (norm,
gate, final linear) stays in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    out_ptr,                # *float32
    # strides for left
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # strides for right
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    # strides for out
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    B, N, H,                # batch, sequence length, hidden dimension
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)           # encodes (group, block_i)
    pid1 = tl.program_id(1)           # block_j

    # number of i‑blocks per group
    num_i_blocks = tl.cdiv(N, BLOCK_M)

    # decode group index and i‑block
    g = pid0 // num_i_blocks          # group = batch * hidden_dim
    block_i = pid0 % num_i_blocks

    # decode batch and hidden dimension from group
    b = g // H
    d = g % H

    # base offsets for the current (b, d)
    base_left  = b * stride_left_batch  + d * stride_left_h
    base_right = b * stride_right_batch + d * stride_right_h
    base_out   = b * stride_out_batch   + d * stride_out_h

    # i and j coordinates for this program
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid1     * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction (k) dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        left_ptrs = left_ptr + base_left \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float32)

        # ----- load right tile (N x K) and transpose to (K x N) -----
        right_ptrs = right_ptr + base_right \
            + offs_j[:, None] * stride_right_i \
            + offs_k[None, :] * stride_right_k
        b_tile = tl.load(right_ptrs,
                         mask=mask_j[:, None] & mask_k[None, :],
                         other=0.0, dtype=tl.float32)
        b_tile_T = tl.trans(b_tile)      # (K, N)

        # ----- inner product -----
        acc += tl.dot(a, b_tile_T)

    # ----- write back -----
    out_ptrs = out_ptr + base_out \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] torch.float32 tensor on CUDA
            - mask:        [B, N, N]   torch.bool or torch.float32
            - weights: dict of model weights (see code)
            - config:  dict with keys "dim" (input dim) and "hidden_dim"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm over the last dimension -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----- Linear projections (no bias) -----
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Apply binary mask (if any) -----
    if mask is None:
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    mask_exp = mask.unsqueeze(-1).type_as(x)            # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ----- Gating values (computed from normalized x) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----- Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ----- Output tensor allocation -----
    out = torch.empty_like(left)   # shape [B, N, N, hidden_dim]

    # ----- Strides (in elements) -----
    sL = left.stride()
    sR = right.stride()
    sO = out.stride()
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = sL
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h = sR
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = sO

    # ----- Triton launch parameters -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N
    groups = B * hidden_dim                      # each (batch, hidden) pair is a GEMM

    grid = (groups * num_i_blocks, num_j_blocks)

    _tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
        stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing (LayerNorm, gate, final linear) -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                # broadcast over the hidden dim
    output = F.linear(out, weights["to_out.weight"])

    return output
