@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Determine program ids
    pid_z = tl.program_id(0) # group (b*H + h)
    pid_m = tl.program_id(1) # block row
    pid_n = tl.program_id(2) # block col

    # Compute offsets
    # base offset for group (b and d)
    # get b and h from pid_z
    b = pid_z // H
    h = pid_z % H

    # For each iteration compute offsets for A, B, C
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-range for rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute base pointers
    # left: offset = b*stride_left_b + h*stride_left_h + i*stride_left_i + k*stride_left_k
    a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_m[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k

    # but we need to add offset for rows within block
    # We'll loop over k

    # Actually compute block pointer arrays for A (left) for each k offset. For each k tile we will load A block of shape (BLOCK_M, BLOCK_K)
    # Similarly for B (right_T): offset = b*stride_right_b + h*stride_right_h + k*stride_right_k + j*stride_right_j; but right_T has rows k, cols j.
    # Since right is stored as shape (B,N,N,H) where j index is second N, k index third N.
    # For B_T (k,j) we need to read right's element at (b, j, k, h). Offsets: b*stride_right_b + j*stride_right_j + k*stride_right_k + h*stride_right_h.
    # So for j offset (offs_n) and k offset (k tile index), compute pointer similarly.

    # We will allocate accumulator in float32.

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = offs_k < N

        # Load A block 
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (right_T)
        # B_T row = k, col = j. So offset = b * stride_right_b + h * stride_right_h + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_j)
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)   # a: (BLOCK_M, BLOCK_K) ; b: (BLOCK_K, BLOCK_N) ; Dot returns (BLOCK_M, BLOCK_N) 
    # end for

    # Write output
    c_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_z = tl.program_id(2)  # group index (b*H + h)

    # decode b and h
    b = pid_z // H
    h = pid_z % H

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A = left[b,i,k,h]
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B = right_T[k,j,h] => right[b,j,k,h]
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Ensure a,b are float32 for dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Store output
    c_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype # assume float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    mask_unsqueeze = mask.unsqueeze(-1)
    left = left * mask_unsqueeze
    right = right * mask_unsqueeze

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # core trmul using Triton
    # allocate output tensor
    out = torch.empty_like(left) # shape (B,N,N,hidden_dim) dtype float32

    # strides
    B, N, _, _ = left.shape
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # get stride ints
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Define block sizes (tune)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * hidden_dim )

    tr_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # apply output gate
    out = out * out_gate

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel
@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_z = tl.program_id(2)  # group index (batch*hidden)

    # Decode batch and hidden idx from pid_z
    b = pid_z // H
    h = pid_z % H

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a block of left: left[b, i, k, h]
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load a block of right transposed: right_T[k, j, h] == right[b, j, k, h]
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate the dot product (both a,b are FP32)
        acc += tl.dot(a, b)

    # Store the result to out[b, i, j, h]
    c_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Steps:
    1. Layer‑norm the input.
    2. Linear projections (left/right) and linear *sigmoid* gates.
    3. Apply mask and gates (element‑wise).
    4. Compute the core O(N³) update:
         out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       via a Triton batched GEMM where the reduction is over k.
    5. Multiply by the output gate.
    6. Layer‑norm over the hidden dimension and final linear projection back to `dim`.

    All heavy O(N³) work is done inside `tr_mul_kernel`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1) Input layer‑norm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Mask and gates
    # -----------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(left.dtype)
    left = left * mask_exp
    right = right * mask_exp

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4) Core O(N³) multiplication via Triton
    # -----------------------------------------------------------------
    # Ensure contiguous layout for strided loads
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output tensor (float32)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Strides (converted to Python ints)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)   # dimension j
    stride_right_k = right.stride(2)   # dimension k
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Choose block sizes (tuned for typical B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tr_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 5) Apply output gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 6) Final layer‑norm + linear projection back to `dim`
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel that computes, for each (batch, hidden) pair,
#   out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
# This is exactly the einsum used in the reference TriMul.
# --------------------------------------------------------------
@triton.jit
def tr_mul_kernel(
    left_ptr,          # [B, N, N, H]  (float32)
    right_ptr,         # [B, N, N, H]  (float32)
    out_ptr,           # [B, N, N, H]  (float32)
    B, N, H,           # runtime ints
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids:
    #   0 -> block row (i)
    #   1 -> block col (j)
    #   2 -> combined batch*hidden index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_z = tl.program_id(2)          # batch*hidden

    # decode batch and hidden dimension
    b = pid_z // H
    h = pid_z % H

    # offsets inside the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction axis k
    # -----------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h \
                 + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right_T[k, j, h] == right[b, j, k, h]
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h \
                 + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product for the tile
        acc += tl.dot(a, b)   # a,b are FP32, result FP32

    # store result: out[b, i, j, h]
    c_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
             + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    1) Layer‑norm the input.
    2) Linear projections (left/right) and their sigmoid gates.
    3) Apply mask and per‑position gates.
    4) Core O(N³) update performed by `tr_mul_kernel`.
    5) Multiply by the output gate.
    6) Hidden‑dim layer‑norm + final linear projection back to `dim`.
    """

    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1) Input layer norm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Mask and gates (all in FP32)
    # -----------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(left.dtype)          # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    left = left.contiguous()

    out = torch.empty_like(left, dtype=torch.float32, device=device)
