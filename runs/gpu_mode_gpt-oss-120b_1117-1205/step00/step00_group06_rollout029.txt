@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program_id axes
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Compute batch and channel indices:
    b = pid_bh // H
    h = pid_bh % H

    # Base offset for this batch+channel
    offset_base_left = b * stride_lb + h * stride_ld
    offset_base_right = b * stride_rb + h * stride_rd
    offset_base_out = b * stride_ob + h * stride_od

    # tile coords
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range within tile
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N
        
        # compute pointers
        left_ptrs = left_ptr + offset_base_left + i_range[:, None] * stride_li + k_range[None, :] * stride_lk
        right_ptrs = right_ptr + offset_base_right + j_range[:, None] * stride_rj + k_range[None, :] * stride_rk

        # Load tiles
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        # Compute dot: left_tile (M, K) dot right_tile^T (K, N) = (M,N)
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # Store result
    out_ptrs = out_ptr + offset_base_out + i_range[:, None] * stride_oi + j_range[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul forward pass with Triton acceleration.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication using a custom Triton kernel.
All other operations (LayerNorm, Linear projections, gating) are performed in PyTorch.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication for the TriMul contraction.

    For each (batch, hidden) pair we compute a tile of the output matrix
    (i‑j plane) by summing over the k dimension.
    """
    pid_i = tl.program_id(0)          # tile row index
    pid_j = tl.program_id(1)          # tile column index
    pid_bh = tl.program_id(2)         # combined batch × hidden index

    # Decode batch and hidden‑channel from combined id
    b = pid_bh // H
    h = pid_bh % H

    # Base offsets for this (batch, hidden) slice
    base_left = b * stride_lb + h * stride_ld
    base_right = b * stride_rb + h * stride_rd
    base_out = b * stride_ob + h * stride_od

    # Tile coordinates
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Offsets for the tiles
        left_offs = base_left + i[:, None] * stride_li + k[None, :] * stride_lk
        right_offs = base_right + j[:, None] * stride_rj + k[None, :] * stride_rk

        # Load tiles with masking for out‑of‑bounds indices
        left_tile = tl.load(
            left_ptr + left_offs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        right_tile = tl.load(
            right_ptr + right_offs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Accumulate dot product: (M×K)·(N×K)ᵀ → (M×N)
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # Write the computed tile to the output tensor
    out_offs = base_out + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(
        out_ptr + out_offs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )
