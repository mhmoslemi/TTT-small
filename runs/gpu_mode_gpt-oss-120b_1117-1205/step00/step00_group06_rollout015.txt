def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # config dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)  # maybe present
    # Normalize
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # Masks
    if not nomask:
        # mask shape [B,N,N]
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f
    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    # Cast to FP16 for kernel compute
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    out_gate = out_gate.to(torch.float16)

    B, N, _, _ = input_tensor.shape

    # Permute for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden_dim, N, N] (transpose k & j)

    batch_dim = B * hidden_dim
    left_flat = left_perm.view(batch_dim, N, N)
    right_flat = right_perm.view(batch_dim, N, N)

    # Output tensor
    out_flat = torch.empty_like(left_flat)  # fp16

    # Define block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_dim)

    # Launch kernel
    matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back
    out_perm = out_flat.view(B, hidden_dim, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim] in fp16

    # Convert to float32 for layernorm and final linear
    out = out.to(torch.float32)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out = out * out_gate.to(torch.float32)

    # Final linear
    output = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return output

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched half‑precision GEMM in Triton.
All other operations (LayerNorm, linear projections, gating, final LayerNorm
and linear) are performed with PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication (A @ B) where
#   A : (batch, M, K)   – left tensor, (i, k)
#   B : (batch, K, N)   – right tensor transposed, (k, j)
#   C : (batch, M, N)   – output, (i, j)
# --------------------------------------------------------------------------- #
@triton.jit
def _bmm_kernel(
    A, B, C,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block col
    pid_b = tl.program_id(2)      # batch (B * hidden_dim)

    # Offsets of the block within the output matrix C
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32 for higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers to tiles of A and B
        a_ptr = A + pid_b * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptr = B + pid_b * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # Load tiles with boundary checks
        a = tl.load(a_ptr,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptr,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # Accumulate the product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write result back in FP16
    c_ptr = C + pid_b * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptr, acc.to(tl.float16), mask=mask)


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """Forward pass of the outgoing TriMul module using Triton for the O(N³) step."""
    input_tensor, mask, weights, config = data

    # -----------------------------------------------------------------------
    # Unpack configuration and weights
    # -----------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)   # True → no masking needed
    eps = 1e-5

    # LayerNorm weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear projection weights (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # Gate weights
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Final linear weight
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, dim], dtype stays as input dtype (float32)

    # -----------------------------------------------------------------------
    # 2. Linear projections (float32)
    # -----------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)    # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------------
    # 3. Optional masking (float32)
    # -----------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------------
    # 4. Gating (float32)
    # -----------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()   # will be used after the matmul

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5. Cast to FP16 for the expensive contraction
    # -----------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    out_gate = out_gate.to(torch.float16)

    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------------
    # 6. Prepare tensors for batched GEMM:
    #    left  -> (batch, M, K)  where M=i, K=k
    #    right -> (batch, K, N)  where K=k, N=j  (transpose of original)
    # -----------------------------------------------------------------------
    # left: [B, N, N, H] -> permute to [B, H, N, N] (i,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: transpose the last two dims -> (j,k) -> (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    batch_dim = B * hidden_dim
    left_flat = left_perm.view(batch_dim, N, N)    # (batch, M, K)
    right_flat = right_perm.view(batch_dim, N, N)  # (batch, K, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)         # (batch, M, N), FP16

    # -----------------------------------------------------------------------
    # 7. Triton GEMM launch
    # -----------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_dim,                 # batch (B * hidden_dim)
    )

    _bmm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim]
    # -----------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)               # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()           # (B, N, N, H)
    out = out.to(torch.float32)                               # to FP32 for following ops

    # -----------------------------------------------------------------------
    # 9. Output LayerNorm, gate, and final linear projection
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)   # element‑wise gate
    output = F.linear(out, to_out_weight)    # final projection to dim

    return output
