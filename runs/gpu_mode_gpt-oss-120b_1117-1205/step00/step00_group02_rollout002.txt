@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bn, stride_bk,
    stride_cm, stride_cn,
    batch_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for i block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # K loop offsets
    offs_k = tl.arange(0, BLOCK_K)

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to start of batch
    a_batch_ptr = a_ptr + pid_batch * batch_stride
    b_batch_ptr = b_ptr + pid_batch * batch_stride
    c_batch_ptr = c_ptr + pid_batch * batch_stride

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Offsets for K with block
        cur_offs_k = offs_k + k * BLOCK_K
        mask_k = cur_offs_k < K

        # Load block of A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + cur_offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load block of B transposed: we need shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + cur_offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a Triton kernel for the heavy
pairwise reduction.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension (dim).
2. Linear projections → left, right   (dim → hidden_dim)
3. Apply the optional mask (broadcast on the last dim).
4. Compute three gating logits, apply sigmoid, and multiply into left/right.
5. Compute the tensor contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication C = A @ Bᵀ where each
   hidden channel is treated as an independent batch element.
   The contraction is performed by a custom Triton kernel that
   tiles over the K dimension.
6. Layer‑norm the result over the hidden dimension, multiply by the
   output gate, and apply the final linear projection back to `dim`.

The Triton kernel is a fused batched GEMM that avoids an explicit
Python loop over the sequence length. All other operations are performed
with regular PyTorch for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for C[b,i,j] = Σₖ A[b,i,k] * B[b,j,k]
# A and B have shape (batch, M, K) and (batch, N, K) respectively.
# The kernel computes C (batch, M, N) using tiling.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bn, stride_bk,
    stride_cm, stride_cn,
    batch_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch index (B * hidden)
    pid_m = tl.program_id(1)              # tile along M (i)
    pid_n = tl.program_id(2)              # tile along N (j)

    # Offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + pid_batch * batch_stride
    b_base = b_ptr + pid_batch * batch_stride
    c_base = c_ptr + pid_batch * batch_stride

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_idx in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = offs_k + k_idx * BLOCK_K
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed: we need (BLOCK_K, BLOCK_N)
        # B is stored as (j, k); we load it as (k, j)
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write the result
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (1 where valid, 0 otherwise)
            - weights: dict of weight tensors required by the module
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype torch.float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # --------------------- 1. LayerNorm over dim ---------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------- 2. Linear projections --------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------- 3. Apply mask (if any) ------------------
    # mask is broadcasted on the hidden dimension
    mask_expanded = mask.unsqueeze(-1).to(dtype)
    left = left * mask_expanded
    right = right * mask_expanded

    # --------------------- 4. Gating --------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------- 5. Triton contraction --------------------
    # Rearrange to (B, hidden, N, N) and fuse B*hidden as batch dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape
    batch_elems = B * H

    left_flat = left.view(batch_elems, N, N)
    right_flat = right.view(batch_elems, N, N)

    out_flat = torch.empty_like(left_flat)

    # Tiling parameters (tuned for typical sequence lengths)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_elems,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides (in number of elements)
    a_stride0, a_stride1, a_stride2 = left_flat.stride()   # (N*N, N, 1)
    b_stride0, b_stride1, b_stride2 = right_flat.stride()
    c_stride0, c_stride1, c_stride2 = out_flat.stride()

    batch_stride = a_stride0  # same for a, b, c because they are contiguous along batch

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride1,
        stride_ak=a_stride2,
        stride_bn=b_stride1,
        stride_bk=b_stride2,
        stride_cm=c_stride1,
        stride_cn=c_stride2,
        batch_stride=batch_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------- 6. Output layer norm --------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    # Ensure float32 output as required
    return out.to(torch.float32)
