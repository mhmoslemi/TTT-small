def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)
    # projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])
    # mask handling
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = left * mask_exp
        right = right * mask_exp
    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate

    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # allocate output
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Triton kernel launch parameters
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 4   # small to reduce registers
    BLOCK_D = 32

    # compile strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), bs)

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        D=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_d,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

"""
TriMul (outgoing) forward pass using a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections to obtain `left` and `right` tensors of shape
   [B, N, N, D] (D = hidden_dim).
3. (Optional) apply a binary mask.
4. Compute three sigmoid gates and apply them to `left` and `right`.
5. Core computation:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication per hidden channel.
   Implemented in Triton: we tile the (i, j) output plane (BLOCK_M×BLOCK_N)
   and accumulate over k in blocks (BLOCK_K).  The hidden dimension is
   processed in groups of BLOCK_D channels.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and a final linear projection back to the original
   feature dimension.
All operations are kept in float32 (mixed‑precision possible inside the
kernel).  The Triton kernel only performs the O(N³·D) reduction; the
surrounding linear / layer‑norm work is done with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out = Σ_k left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(2)       # batch dimension
    pid_i = tl.program_id(0)       # output row tile
    pid_j = tl.program_id(1)       # output col tile

    # ------------------------------------------------------------------
    # Offsets for the i and j tiles
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        d_start = d_off * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # accumulator for the current tile and channel slice
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Reduction over the k dimension in blocks
        # ------------------------------------------------------------------
        for k_off in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # -------- left slice: shape (M, K, D) -----------------------
            left_ptrs = left_ptr + pid_b * stride_left_b \
                + offs_i[:, None, None] * stride_left_i \
                + offs_k[None, :, None] * stride_left_k \
                + offs_d[None, None, :] * stride_left_d
            left = tl.load(left_ptrs,
                           mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                           other=0.0)                     # (M, K, D)

            # -------- right slice (transposed): shape (K, N, D) -------
            right_ptrs = right_ptr + pid_b * stride_right_b \
                + offs_k[:, None, None] * stride_right_k \
                + offs_j[None, :, None] * stride_right_j \
                + offs_d[None, None, :] * stride_right_d
            right = tl.load(right_ptrs,
                            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                            other=0.0)                    # (K, N, D)

            # -------- accumulate outer product over the K block -------
            for kk in range(0, BLOCK_K):
                # left_vec: (M, D), right_vec: (N, D)
                left_vec = left[:, kk, :]            # (M, D)
                right_vec = right[kk, :, :]          # (N, D)
                # Broadcast to (M, N, D) and accumulate
                acc += left_vec[:, None, :] * right_vec[None, :, :]

        # ------------------------------------------------------------------
        # Store the computed tile for this channel slice
        # ------------------------------------------------------------------
        out_ptrs = out_ptr + pid_b * stride_out_b \
            + offs_i[:, None, None] * stride_out_i \
            + offs_j[None, :, None] * stride_out_j \
            + offs_d[None, None, :] * stride_out_d
        mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptrs, acc, mask=mask)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict of torch.Tensor containing model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # ensure mask dtype matches
        m = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul – Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tile sizes – chosen to fit registers and shared memory on B200
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 4
    BLOCK_D = 32

    # Strides (element‑wise) for the three tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),  # i‑tiles
        triton.cdiv(N, BLOCK_N),  # j‑tiles
        B,                        # batch dimension
    )

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,   # second N dimension (j)
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_d,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: second LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_gated = out_norm * out_gate
    output = F.linear(out_gated, weights["to_out.weight"])

    return output
