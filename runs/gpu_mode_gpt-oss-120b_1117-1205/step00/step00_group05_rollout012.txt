@triton.jit
def multiply_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = batch // H
    h = batch % H

    # Compute start offsets
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Offsets for k
        k_start = k * BLOCK_K

        # Define offsets vectors
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_j = j_start + tl.arange(0, BLOCK_N)
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Mask for dimensions
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # Combined mask for left tile: (BLOCK_M,BLOCK_K)
        mask_left = mask_i[:, None] & mask_k[None, :]

        # Compute pointers
        left_ptrs = left_ptr + b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h * stride_left_h

        # Load left tile
        left = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

        # Right tile pointer: shape (BLOCK_K, BLOCK_N)
        # Use k as rows, j as columns
        right_ptrs = right_ptr + b * stride_right_b + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i + h * stride_right_h

        mask_right = mask_k[:, None] & mask_j[None, :]

        right = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        # Dot product: left (M,K) dot right (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # Write back result
    out_ptrs = out_ptr + b * stride_out_b + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j + h * stride_out_h

    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype == tl.float16 else acc, mask=mask_out)
```

"""
triton_tri_mul.py

Implementation of the "outgoing" Triangle Multiplicative Update (TriMul) from
AlphaFold3.  The heavy N³ reduction is performed by a custom Triton kernel that
computes, for each hidden channel h, the batched matrix multiplication:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

The rest of the module (LayerNorms, Linear projections, gating) is kept in
PyTorch for simplicity.  The kernel is tiled over the i‑ and j‑dimensions and
loops over k in blocks, using float32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    out_ptr,                # *float32
    B, N, H,                # integers: batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size for i
    BLOCK_N: tl.constexpr,  # tile size for j
    BLOCK_K: tl.constexpr,  # tile size for reduction k
):
    # program IDs
    batch_idx = tl.program_id(0)   # 0 .. B*H-1
    pid_i = tl.program_id(1)       # tile over rows (i)
    pid_j = tl.program_id(2)       # tile over cols (j)

    # decode batch and hidden dimension
    b = batch_idx // H
    h = batch_idx % H

    # starting coordinates of the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # masks for boundary handling (i and j)
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(0, num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # mask for k dimension
        mask_k = offs_k < N

        # ---------- load left tile (M x K) ----------
        # left shape: (B, N_i, N_k, H)  -> indices (b, i, k, h)
        mask_left = mask_i[:, None] & mask_k[None, :]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

        # ---------- load right tile (K x N) ----------
        # right shape: (B, N_j, N_k, H)  -> indices (b, j, k, h)
        # we need it as (K, N) for the dot product
        mask_right = mask_k[:, None] & mask_j[None, :]
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
            + h * stride_right_h
        )
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        # ---------- accumulate ----------
        # left_tile : (M, K), right_tile : (K, N)
        acc += tl.dot(left_tile, right_tile)

    # ---------- store output ----------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=mask_ij)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul module with the core einsum offloaded to Triton.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (float or bool)
            - weights: dict of model weights (torch.Tensor)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights (all are float32 tensors already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)    # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul (batched matmul) via Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # N is seq_len
    # ensure contiguous tensors for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tiling parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions: (batch * hidden_dim, n_tiles_i, n_tiles_j)
    num_batch = B * hidden_dim
    grid = (
        num_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch kernel
    tri_mul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # --------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
