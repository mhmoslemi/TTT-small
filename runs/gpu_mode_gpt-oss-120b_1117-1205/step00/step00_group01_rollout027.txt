@triton.jit
def matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch index (flattened batch*hidden)
    pid_m = tl.program_id(1)  # row block index
    pid_n = tl.program_id(2)  # col block index

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create masks for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulation buffer
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base offset for this batch
    a_batch_offset = pid_b * stride_ab
    b_batch_offset = pid_b * stride_bb
    c_batch_offset = pid_b * stride_cb

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for k dimension
        mask_k = offs_k < K

        # load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A + a_batch_offset \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B: shape (BLOCK_K, BLOCK_N)
        # B is stored as (K,N), we want B_T (N,K) but we have swapped strides, so treat as usual.
        b_ptrs = B + b_batch_offset \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

    # write out result
    c_ptrs = C + c_batch_offset \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

    device = input_tensor.device
    dtype = input_tensor.dtype

    # config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm

    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    left = left.to(torch.float16)

    right_sh = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_sh.view(batch_h, N, N)

    stride_cb = out_flat.stride(0)

"""
TriMul "outgoing" implementation with a fused Triton GEMM.

The module performs:
1. LayerNorm over the last dimension.
2. Linear projections (left/right) and gating (left/right/out) on the normalized tensor.
3. Applies the binary mask to the projected tensors.
4. Computes the core O(N³) operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batch of matrix‑matrix multiplications
   (left[:,:,d] @ right[:,:,d]ᵀ) for each hidden channel.
   A custom Triton kernel performs the batched GEMM in FP16 with FP32 accumulation.
5. Applies a second LayerNorm, multiplies by the output gate and projects back to the
   original dimension.

Only the heavy matrix multiplication is off‑loaded to Triton; the rest is done with
efficient PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)          # batch * hidden
    pid_m = tl.program_id(1)          # row block index
    pid_n = tl.program_id(2)          # col block index

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for edges
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base offsets for this batch element
    a_batch_off = pid_b * stride_ab
    b_batch_off = pid_b * stride_bb
    c_batch_off = pid_b * stride_cb

    # loop over tiles of K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load A tile (M x K) ----
        a_ptrs = A + a_batch_off \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ---- Load B tile (K x N) ----
        # B is passed as the transpose of the original right tensor,
        # therefore we use swapped strides (see host code).
        b_ptrs = B + b_batch_off \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # ---- Multiply‑Accumulate ----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # ---- Store result ----
    c_ptrs = C + c_batch_off \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul "outgoing" update.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (binary)
            - weights: dict of model parameters (see code)
            - config: dict containing "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # (B, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy multiplication
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Triton batched GEMM: out = left @ rightᵀ per hidden channel
    # ------------------------------------------------------------------
    # Convert to (B, hidden, N, N) and flatten batch*hidden
    left_ = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_.shape
    batch_h = B * H

    left_flat = left_.view(batch_h, N, N)
    right_flat = right_.view(batch_h, N, N)

    out_flat = torch.empty((batch_h, N, N), dtype=torch.float32, device=device)

    # strides for A (left)
    stride_ab = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    # strides for B (right) – we need Bᵀ, so swap the last two strides
    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)   # original row stride (N)
    stride_bn = right_flat.stride(2)   # original column stride (1)

    # strides for C (output)
    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_h, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am, stride_ak, stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 7. Second LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate

    output = F.linear(out, weights["to_out.weight"])

    return output
