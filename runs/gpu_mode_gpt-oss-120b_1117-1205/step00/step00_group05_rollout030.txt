def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr = 32,
):
    pid = tl.program_id(0)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # masks for i,j within bounds (should be always true)
    # accumulate
    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

    # iterate over k blocks
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left: shape (BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + b*stride_left_b + i*stride_left_i + k_offset[:, None]*stride_left_k + tl.arange(0, BLOCK_D)[None, :]*stride_left_h
        left = tl.load(left_ptrs, mask=mask_k[:, None], other=0.0)  # shape (BLOCK_K, BLOCK_D)

        # load right: shape (BLOCK_K, BLOCK_D) for given j
        right_ptrs = right_ptr + b*stride_right_b + j*stride_right_j + k_offset[:, None]*stride_right_k + tl.arange(0, BLOCK_D)[None, :]*stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None], other=0.0)

        # multiply and sum over k (axis 0)
        # compute elementwise product then reduce sum along k dimension
        # We can use tl.sum for each d dimension: product shape (BLOCK_K, BLOCK_D)
        prod = left * right  # shape (BLOCK_K, BLOCK_D)
        acc += tl.sum(prod, axis=0)  # sum over k axis

    # store output
    out_ptrs = out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j + tl.arange(0, BLOCK_D)*stride_out_h
    tl.store(out_ptrs, acc, mask=True)  # no mask needed for d dimension length H.

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    b = pid_b

    # Offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Masks for i and j bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over d channel blocks
    for d_start in range(0, H, BLOCK_D):
        # D offsets and mask
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # accumulator for this d block
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # Loop over k blocks
        for k_start in range(0, N, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left tile (i,k,d)
            left_ptrs = left_ptr + \
                b * stride_l_b + \
                offs_i[:, None, None] * stride_l_i + \
                offs_k[None, :, None] * stride_l_k + \
                offs_d[None, None, :] * stride_l_h
            left = tl.load(left_ptrs,
                           mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                           other=0.0)

            # Load right tile but with k first (k,j,d) for easier dot
            right_ptrs = right_ptr + \
                b * stride_r_b + \
                offs_k[:, None, None] * stride_r_k + \
                offs_j[None, :, None] * stride_r_j + \
                offs_d[None, None, :] * stride_r_h
            right = tl.load(right_ptrs,
                            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                            other=0.0)

            # For each d within block, perform dot
            for d in range(0, BLOCK_D):
                # Check if d is valid
                d_valid = mask_d[d]
                # If not valid, skip (continue)
                # We can use mask to zero out contributions
                l = left[:, :, d]   # (BLOCK_M, BLOCK_K)
                r = right[:, :, d]   # (BLOCK_K, BLOCK_N)
                # compute dot
                prod = tl.dot(l, r)  # (BLOCK_M, BLOCK_N)
                acc[:, :, d] += prod
        # Store result for valid d
        out_ptrs = out_ptr + \
            b * stride_o_b + \
            offs_i[:, None, None] * stride_o_i + \
            offs_j[None, :, None] * stride_o_j + \
            offs_d[None, None, :] * stride_o_h
        tl.store(out_ptrs,
                 acc,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

"""
TriMul (outgoing) implementation with a Triton kernel for the core
pairwise reduction over the sequence dimension.

The kernel computes:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left and right are already projected, masked and gated.
All other operations (LayerNorms, Linear projections, sigmoids, final
projection) are performed with PyTorch for clarity.
"""
import torch
import triton
import triton.language as tl
import math
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *const float*
    right_ptr,         # *const float*
    out_ptr,           # *float*
    B, N, H,           # batch, seq_len, hidden_dim (runtime)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute the TriMul reduction for a tile (i,j) of size BLOCK_M x BLOCK_N."""
    pid_i = tl.program_id(0)      # tile along i
    pid_j = tl.program_id(1)      # tile along j
    pid_b = tl.program_id(2)      # batch index

    # ------------------------------------------------------------------
    # Offsets and masks for the i and j dimensions
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    b = pid_b

    offs_i = i_start + tl.arange(0, BLOCK_M, dtype=tl.int64)
    offs_j = j_start + tl.arange(0, BLOCK_N, dtype=tl.int64)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # clamp out‑of‑bounds indices to 0 (avoids illegal memory accesses)
    i_idx = tl.where(mask_i, offs_i, 0)
    j_idx = tl.where(mask_j, offs_j, 0)

    # ------------------------------------------------------------------
    # Loop over hidden dimension in blocks
    # ------------------------------------------------------------------
    d = 0
    while d < H:
        cur_D = tl.minimum(BLOCK_D, H - d)            # may be smaller at the end
        d_offsets = d + tl.arange(0, BLOCK_D, dtype=tl.int64)
        mask_d = d_offsets < H

        # accumulator for this D‑block   (BLOCK_M, BLOCK_N, cur_D)
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over the reduction dimension k
        # ------------------------------------------------------------------
        k = 0
        while k < N:
            cur_K = tl.minimum(BLOCK_K, N - k)
            offs_k = k + tl.arange(0, BLOCK_K, dtype=tl.int64)
            mask_k = offs_k < N
            k_idx = tl.where(mask_k, offs_k, 0)

            # ------------------------------------------------------------------
            # For each channel d inside the block compute the dot product
            # ------------------------------------------------------------------
            d_sub = 0
            while d_sub < cur_D:
                # absolute d index
                d_abs = d + d_sub
                # masks for this channel
                d_mask = d_abs < H

                # ----- load left[i,k,d] -----
                left_ptrs = (
                    left_ptr
                    + b * stride_l_b
                    + i_idx[:, None] * stride_l_i
                    + k_idx[None, :] * stride_l_k
                    + d_abs * stride_l_h
                )
                left = tl.load(
                    left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :] & d_mask,
                    other=0.0,
                )                     # (BLOCK_M, BLOCK_K)

                # ----- load right[j,k,d] (k as first dim for dot) -----
                right_ptrs = (
                    right_ptr
                    + b * stride_r_b
                    + k_idx[:, None] * stride_r_k
                    + j_idx[None, :] * stride_r_j
                    + d_abs * stride_r_h
                )
                right = tl.load(
                    right_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :] & d_mask,
                    other=0.0,
                )                     # (BLOCK_K, BLOCK_N)

                # dot product and accumulate
                prod = tl.dot(left, right)          # (BLOCK_M, BLOCK_N)
                acc[:, :, d_sub] += prod

                d_sub += 1
            k += BLOCK_K

        # ------------------------------------------------------------------
        # Store the results for this D‑block
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + b * stride_o_b
            + i_idx[:, None, None] * stride_o_i
            + j_idx[None, :, None] * stride_o_j
            + (d + tl.arange(0, BLOCK_D, dtype=tl.int64))[None, None, :] * stride_o_h
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None]
            & mask_j[None, :, None]
            & mask_d[None, None, :],
        )

        d += BLOCK_D


# ----------------------------------------------------------------------
# Host side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Uses a Triton kernel for the cubic reduction and PyTorch for the rest.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Config
    # ------------------------------------------------------------------
    bs = config["bs"]
    seq_len = config["seqlen"]
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    mask_unsq = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the O(N³) reduction
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B, N, N, hidden_dim]

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    # Note: for `right` the second dimension corresponds to `j` (same as `i` index)
    # because the tensor shape is identical to `left`. We reuse the same stride
    # ordering; only the role of the second dimension changes.
    stride_r_j = stride_r_i

    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – chosen small to keep register usage modest.
    BLOCK_M = 8
    BLOCK_N = 8
    BLOCK_K = 32
    BLOCK_D = 8  # channel block; will be looped over in the kernel

    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        bs,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Post‑processing: output norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
