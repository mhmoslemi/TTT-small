    batch_stride_left, stride_left_i, stride_left_k,
    batch_stride_right, stride_right_i, stride_right_k,

    pid_n = tl.program_id(1)

    # Compute offsets

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operation.
    The heavy O(N^3) contraction is performed by a custom batched GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    B, N, _, D = input_tensor.shape
    assert D == dim

    # LayerNorm over last dimension
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (float32)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for the contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to [B, hidden_dim, N, N] and collapse batch*hidden_dim
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Allocate output
    out_reshaped = torch.empty_like(left_reshaped)  # fp16

    # Triton kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    batched_gemm_kernel[grid](
        left_reshaped, right_reshaped, out_reshaped,
        N,
        stride_az=left_reshaped.stride(0), stride_ai=left_reshaped.stride(1), stride_ak=left_reshaped.stride(2),
        stride_bz=right_reshaped.stride(0), stride_bi=right_reshaped.stride(1), stride_bj=right_reshaped.stride(2),
        stride_cz=out_reshaped.stride(0), stride_ci=out_reshaped.stride(1), stride_cj=out_reshaped.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # Output layer norm on hidden dimension
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) Triton implementation.

The expensive O(N³) contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed by a custom batched GEMM kernel. All other operations
(LayerNorm, Linear projections, gating) are executed in PyTorch.
The kernel works on fp16 data for the contraction and accumulates
in fp32 for numerical stability.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the triangle multiplicative update.
# Input tensors have shape (batch, N, N) where batch = B * hidden_dim.
# The kernel computes C = A @ Bᵀ for each batch element.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bi, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row of output (i)
    pid_n = tl.program_id(1)          # block column of output (j)
    pid_b = tl.program_id(2)          # batch * hidden_dim index

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    a_batch = a_ptr + pid_b * stride_az   # base pointer for A of this batch
    b_batch = b_ptr + pid_b * stride_bz   # base pointer for B of this batch
    c_batch = c_ptr + pid_b * stride_cz   # base pointer for C of this batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_off = offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_batch + a_off,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: we need the transpose of the original (i, j) matrix.
        # Original B layout is (i, j). To obtain (k, j) we load with
        # offsets = k * stride_bj + j * stride_bi.
        b_off = offs_k[:, None] * stride_bj + offs_n[None, :] * stride_bi
        b = tl.load(b_batch + b_off,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # --------------------------------------------------------------
    # Write the result back (store as fp16)
    # --------------------------------------------------------------
    c_off = offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_batch + c_off, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

        stride_ak=left_reshaped.stride(2),
        stride_bz=right_reshaped.stride(0),
        stride_bi=right_reshaped.stride(1),
        stride_bj=right_reshaped.stride(2),

        BLOCK_N=BLOCK_N,

    )

    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output layer norm over hidden_dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
