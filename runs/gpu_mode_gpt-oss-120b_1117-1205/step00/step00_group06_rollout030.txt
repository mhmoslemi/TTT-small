@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # predicates
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to batch start
    a_batch_ptr = a_ptr + batch_id * a_batch_stride
    b_batch_ptr = b_ptr + batch_id * b_batch_stride
    c_batch_ptr = c_ptr + batch_id * c_batch_stride

    # Pointers to block tiles
    a_ptrs = a_batch_ptr + offs_m[:, None] * a_stride_m + tl.arange(0, BLOCK_K)[None, :] * a_stride_k
    b_ptrs = b_batch_ptr + tl.arange(0, BLOCK_K)[:, None] * b_stride_k + offs_n[None, :] * b_stride_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Bounds for current k tile
        k_base = k * BLOCK_K
        k_valid = (k_base + tl.arange(0, BLOCK_K)) < K

        a = tl.load(a_ptrs, mask = mask_m[:, None] & k_valid[None, :], other=0.0)
        b = tl.load(b_ptrs, mask = k_valid[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)  # float32

        # Move pointer to next k tile
        a_ptrs += BLOCK_K * a_stride_k
        b_ptrs += BLOCK_K * b_stride_k

    c_ptrs = c_batch_ptr + offs_m[:, None] * c_stride_m + offs_n[None, :] * c_stride_n
    tl.store(c_ptrs, acc, mask = mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    from torch.nn import functional as F

    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    device = input_tensor.device

    # Load weights and ensure correct dtype and device
    # convert all weight tensors to same dtype as input (float32)
    for key in weights:
        weights[key] = weights[key].to(device)

    # LayerNorm on dim
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Linear projections
    left = F.linear(x, weights['left_proj.weight'])
    right = F.linear(x, weights['right_proj.weight'])

    # Apply mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # Core contraction via Triton batched matmul
    B, N, _, H = left.shape
    # Should equal hidden_dim. Check H==hidden_dim
    # Permute
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    batch_mat = B * H
    left_mat = left_perm.view(batch_mat, N, N)
    right_mat = right_perm.view(batch_mat, N, N)
    out_mat = torch.empty_like(left_mat)

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid
    grid = (batch_mat, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,H]

    # To output norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])

    # Multiply gate
    out = out * out_gate

    # Final linear
    output = F.linear(out, weights['to_out.weight'])

    return output

import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * a_batch_stride
    b_batch = b_ptr + batch * b_batch_stride
    c_batch = c_ptr + batch * c_batch_stride

    # compute tile pointers initial offset
    a_tile_ptr = a_batch + offs_m[:, None] * a_stride_m + tl.arange(0, BLOCK_K)[None, :] * a_stride_k
    b_tile_ptr = b_batch + tl.arange(0, BLOCK_K)[:, None] * b_stride_k + offs_n[None, :] * b_stride_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_valid = (k * BLOCK_K + tl.arange(0, BLOCK_K)) < K
        a = tl.load(a_tile_ptr, mask = mask_m[:, None] & k_valid[None, :], other=0.0)
        b = tl.load(b_tile_ptr, mask = k_valid[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)  # automatically cast to float32

        # advance pointers to next k tile
        a_tile_ptr += BLOCK_K * a_stride_k
        b_tile_ptr += BLOCK_K * b_stride_k

    # write output
    c_tile_ptr = c_batch + offs_m[:, None] * c_stride_m + offs_n[None, :] * c_stride_n
    tl.store(c_tile_ptr, acc, mask = mask_m[:, None] & mask_n[None, :])

"""Implementation of the outgoing TriMul operator with a Triton batched
matrix‑multiplication kernel.  The heavy O(N³·C) contraction
`einsum('...ikd,...jkd->...ijd')` is rewritten as a batch of
matrix multiplications `A @ Bᵀ` for each channel `d`.  The kernel
operates on float32 tensors; the rest of the module (LayerNorm,
Linear, gating) is performed with PyTorch."""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[batch, i, j] = sum_k A[batch, i, k] * B[batch, k, j]"""
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch = a_ptr + batch * a_batch_stride
    b_batch = b_ptr + batch * b_batch_stride
    c_batch = c_ptr + batch * c_batch_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    k = 0
    while k < K:
        cur_k = tl.minimum(BLOCK_K, K - k)
        # offsets for the k‑dimension within this tile
        k_range = tl.arange(0, BLOCK_K)
        k_valid = k_range < cur_k

        # Load tiles with appropriate masks
        a = tl.load(
            a_batch + offs_m[:, None] * a_stride_m
                     + (k + k_range)[None, :] * a_stride_k,
            mask=mask_m[:, None] & k_valid[None, :],
            other=0.0
        )
        b = tl.load(
            b_batch + (k + k_range)[:, None] * b_stride_k
                     + offs_n[None, :] * b_stride_n,
            mask=k_valid[:, None] & mask_n[None, :],
            other=0.0
        )
        # Accumulate
        acc += tl.dot(a, b)

        k += BLOCK_K

    # Write the result
    c_ptr_tile = c_batch + offs_m[:, None] * c_stride_m + offs_n[None, :] * c_stride_n
    tl.store(c_ptr_tile, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # Layer‑norm over the feature dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )

    # -----------------------------------------------------------------
    # Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # -----------------------------------------------------------------
    # Apply the (optional) mask
    # -----------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Core O(N³·C) contraction via Triton batched GEMM
    # -----------------------------------------------------------------
    B, N_seq, _, H = left.shape
    # reshape to (B*H, N, N) where each slice is a matrix multiplication
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = B,H,K,J

    batch_mat = B * H
    left_mat = left_perm.view(batch_mat, N_seq, N_seq)
    right_mat = right_perm.view(batch_mat, N_seq, N_seq)
    out_mat = torch.empty_like(left_mat)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_mat,
        triton.cdiv(N_seq, BLOCK_M),
        triton.cdiv(N_seq, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        N_seq, N_seq, N_seq,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # Output post‑processing
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"].to(device))

    return output

"""TriMul forward pass with a Triton kernel for the O(N³·C) contraction.

The heavy part of the TriMul module is the Einstein sum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be written as a batch of matrix multiplications:
    out[b,i,j,d] = (left[b,:, :, d] @ right[b,:, :, d].T)[i,j].

The kernel `batched_matmul_kernel` computes C = A @ Bᵀ for many
independent batches (batch = bs * hidden_dim).  The surrounding
operations (LayerNorm, Linear projections, gating) are performed in
PyTorch.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, k, j]

    A: (batch, M, K)
    B: (batch, K, N)
    C: (batch, M, N)
    """
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * a_batch_stride
    b_batch = b_ptr + batch * b_batch_stride
    c_batch = c_ptr + batch * c_batch_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    k = 0
    while k < K:
        cur_k = tl.minimum(BLOCK_K, K - k)          # size of the current K‑tile
        k_range = tl.arange(0, BLOCK_K)
        k_valid = k_range < cur_k

        # Load a tile of A (M × cur_k)
        a = tl.load(
            a_batch
            + offs_m[:, None] * a_stride_m
            + (k + k_range)[None, :] * a_stride_k,
            mask=mask_m[:, None] & k_valid[None, :],
            other=0.0,
        )
        # Load a tile of B (cur_k × N)
        b = tl.load(
            b_batch
            + (k + k_range)[:, None] * b_stride_k
            + offs_n[None, :] * b_stride_n,
            mask=k_valid[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # accumulate in FP32
        k += BLOCK_K

    # Store the tile of C
    c_ptr_tile = c_batch + offs_m[:, None] * c_stride_m + offs_n[None, :] * c_stride_n
    tl.store(c_ptr_tile, acc, mask=mask_m[:, None] & mask_n[None, :])

    # Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # -----------------------------------------------------------------

    out = F.layer_norm(
