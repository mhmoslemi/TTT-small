def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype # assume float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # retrieve weights as torch tensors
    # Note: Linear weight shape: (out_features, in_features)
    # ensure correct dtype
    # For linear layers w/out bias

    def linear(x, weight):
        return torch.nn.functional.linear(x, weight)  # weight shape [out_features, in_features], x last dim = in_features

    # Normalize input x with LayerNorm:
    norm_weight = weights['norm.weight']  # shape [dim]
    norm_bias = weights['norm.bias']  # shape [dim]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # compute projections
    left = linear(x_norm, weights['left_proj.weight'])
    right = linear(x_norm, weights['right_proj.weight'])
    # left,right shape [B,N,N,hidden_dim]

    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.sigmoid(linear(x_norm, weights['left_gate.weight']))
    right_gate = torch.sigmoid(linear(x_norm, weights['right_gate.weight']))
    out_gate = torch.sigmoid(linear(x_norm, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # cast to half for contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # prepare shapes
    B, N, _, H = left.shape
    batch_matmul = B * H

    # reshape for kernel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch_matmul, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch_matmul, N, N)
    # need right_T (transpose K,N)
    right_T = right_mat.transpose(1, 2).contiguous()  # shape [batch_matmul, N, N]

    # allocate output
    out_mat = torch.empty_like(left_mat)  # same dtype (half) shape [batch_matmul, N, N]

    # define block sizes based on N:
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # we can tune; using 64 maybe good.

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_matmul)

    batched_matmul_kernel[grid](
        left_mat, right_T, out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_T.stride(0), right_T.stride(1), right_T.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape out_mat back to [B,N,N,H]
    out_tmp = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # convert to float32
    out = out_tmp.to(torch.float32)

    # to_out_norm
    out_norm_weight = weights['to_out_norm.weight']
    out_norm_bias = weights['to_out_norm.bias']
    # compute mean,var over dim=-1 (hidden_dim)
    mean = out.mean(dim=-1, keepdim=True)
    var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean) / torch.sqrt(var + eps)
    out_norm = out_norm * out_norm_weight + out_norm_bias

    # final linear to_out: weight shape [dim, hidden_dim]
    out_final = torch.nn.functional.linear(out_norm, weights['to_out.weight'])
    return out_final

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # Retrieve weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Move all weights to same device if needed
    for w in [norm_weight, norm_bias, left_proj_weight, right_proj_weight,
              left_gate_weight, right_gate_weight, out_gate_weight,
              to_out_norm_weight, to_out_norm_bias, to_out_weight]:
        if w.device != device:
            w = w.to(device)

    # LayerNorm on input
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for contraction
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Prepare for Triton batched matmul (left @ right_T)
    B, N, _, H = left_half.shape
    batch_dim = B * H

    left_mat = left_half.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)
    # Right matrix needs transpose on the last two dims for matmul
    right_mat = right_half.permute(0, 3, 2, 1).contiguous().view(batch_dim, N, N)  # (B, H, N, N) but transposed

    out_mat = torch.empty_like(left_mat)

    # Triton kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_dim)
    batched_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,H]
    # Cast back to float32
    out = out.to(torch.float32)

    # to_out_norm (LayerNorm)
    mean = out.mean(dim=-1, keepdim=True)
    var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean) / torch.sqrt(var + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # final linear
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output

"""
TriMul forward pass (outgoing version) with a custom Triton kernel.
The heavy contraction 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication 
    out_d[b] = left_d[b] @ right_d[b].T
for each hidden dimension d. The rest of the module
(LayerNorm, linear projections, gating) is performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B where
#   A : [batch, M, K]
#   B : [batch, K, N]   (K dimension contiguous for better performance)
#   C : [batch, M, N]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch index

    # --------------------------------------------------------------
    # Offsets for the output tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # --------------------------------------------------------------
        # Load a tile of A (shape BLOCK_M x BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (a_ptr + pid_b * stride_a_batch +
                  offs_m[:, None] * stride_a_m +
                  offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Load a tile of B (shape BLOCK_K x BLOCK_N)
        # --------------------------------------------------------------
        b_ptrs = (b_ptr + pid_b * stride_b_batch +
                  offs_k[:, None] * stride_b_k +
                  offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Compute partial dot product
        # --------------------------------------------------------------
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result tile (convert back to bfloat16)
    # --------------------------------------------------------------
    c_ptrs = (c_ptr + pid_b * stride_c_batch +
              offs_m[:, None] * stride_c_m +
              offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.bfloat16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the TriMul module (outgoing version).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float)
            - weights: dict of torch.Tensors containing all model parameters
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Unpack weights (all are float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # Input LayerNorm (over the last dimension)
    # --------------------------------------------------------------
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias   # broadcast over leading dims

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # Gating (sigmoid of linear)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Cast to bfloat16 for the heavy contraction
    # --------------------------------------------------------------
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # --------------------------------------------------------------
    # Prepare tensors for the batched matmul:
    #   left_bf  -> [B, H, N, N]
    #   right_bf -> transposed on the last two dims -> [B, H, N, N] where
    #               the middle dimension is the reduction axis K.
    # --------------------------------------------------------------
    B, N, _, H = left_bf.shape
    batch_dim = B * H

    # [B, H, N, N] where H is now the batch dimension we will fuse
    left_mat = left_bf.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)
    # Transpose the last two axes of right before flattening
    right_mat = right_bf.permute(0, 3, 2, 1).contiguous().view(batch_dim, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_dim                  # batch*hidden
    )
    batched_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and convert to float32
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden_dim)
    # ------------------------------------------------------------------
    mean = out.mean(dim=-1, keepdim=True)
    var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean) / torch.sqrt(var + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # Apply out gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to dim
    # ------------------------------------------------------------------
    output = F.linear(out_norm, to_out_weight)

    return output
