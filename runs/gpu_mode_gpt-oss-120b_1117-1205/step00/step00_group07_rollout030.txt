@triton.jit
def tri_mul_kernel(A_ptr, B_ptr, C_ptr,
                   M, N, K,
                   stride_am, stride_ak, stride_ab,
                   stride_bk, stride_bn, stride_bb,
                   stride_cm, stride_cn, stride_cb,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + pid_b * stride_ab + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = B_ptr + pid_b * stride_bb + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        acc += tl.dot(a, b)

    c_ptrs = C_ptr + pid_b * stride_cb + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated outgoing TriMul (AlphaFold3) forward pass.
    Computes out_{i,j} = sum_k left_{i,k} * right_{j,k}
    using a fused batch matmul kernel on GPU.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5
    dtype_fp16 = torch.float16

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------
    # 1. LayerNorm on input (dim)
    # --------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=norm_weight,
                                       bias=norm_bias,
                                       eps=eps)

    # Cast to half for heavy compute
    x_fp16 = x.to(dtype_fp16)

    # --------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # --------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_fp16, left_proj_weight.to(dtype_fp16))
    right = torch.nn.functional.linear(
        x_fp16, right_proj_weight.to(dtype_fp16))

    # --------------------------------------------------------------------
    # 3. Apply optional mask
    # --------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(dtype_fp16)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------------
    # 4. Gating (computed in float32 for stability)
    # --------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate.to(dtype_fp16)
    right = right * right_gate.to(dtype_fp16)

    # --------------------------------------------------------------------
    # 5. Batched matrix multiplication via Triton (outgoing version)
    #    out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}
    # --------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    M = B * hidden_dim

    # reshape for kernel: (M, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(M, N, N)
    right_mat = right_perm.view(M, N, N)

    out_mat = torch.empty_like(left_mat, dtype=dtype_fp16)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (M,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M, N, N,
        left_mat.stride(1), left_mat.stride(2), left_mat.stride(0),
        right_mat.stride(1), right_mat.stride(2), right_mat.stride(0),
        out_mat.stride(1), out_mat.stride(2), out_mat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------
    # 6. Final normalization, gating, and projection
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps)

    out_scaled = out_norm * out_gate

    output = torch.nn.functional.linear(out_scaled, to_out_weight)

    return output

"""
Optimized Triton implementation of the outgoing TriMul operator (AlphaFold3).

The core computes, for each batch b and hidden dimension d,
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

which is a batched matrix multiplication of shape (B*hidden_dim, N, N)
with its second argument transposed. The heavy O(N³) work is fused into a
single Triton kernel; the surrounding LayerNorms, linear projections,
and gating are performed in PyTorch (float32 → float16 for the kernel).
The final output is returned as float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication: C = A @ Bᵀ  (A,B shape (M,N,K))."""
    pid_b = tl.program_id(0)                 # batch*head dimension
    pid_m = tl.program_id(1)                 # output row block
    pid_n = tl.program_id(2)                 # output column block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A has shape (M,N,K) – rows = i, cols = k
        a_ptrs = A_ptr + pid_b * stride_ab \
                 + (offs_m[:, None] * stride_am) \
                 + (offs_k[None, :] * stride_ak)

        # B is interpreted as (M,K,N) = original (M,N,K)ᵀ
        b_ptrs = B_ptr + pid_b * stride_bb \
                 + (offs_k[:, None] * stride_bk) \
                 + (offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    c_ptrs = C_ptr + pid_b * stride_cb \
             + (offs_m[:, None] * stride_cm) \
             + (offs_n[None, :] * stride_cn)

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    x_fp16 = x.to(dtype_fp16)

    left = F.linear(x_fp16, left_proj_weight.to(dtype_fp16))
    right = F.linear(x_fp16, right_proj_weight.to(dtype_fp16))

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(dtype_fp16)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (computed in float32, then cast)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate.to(dtype_fp16)
    right = right * right_gate.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 5. Batched matmul with Triton: out = left @ rightᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    M = B * hidden_dim                                # total batch*head dimension

    # reshape to (M, N, N) where M = B*hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(M, N, N)
    right_mat = right_perm.view(M, N, N)

    out_mat = torch.empty_like(left_mat, dtype=dtype_fp16)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (M,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M, N, N,
        left_mat.stride(1), left_mat.stride(2), left_mat.stride(0),   # A strides
        right_mat.stride(1), right_mat.stride(2), right_mat.stride(0), # B strides (transposed view)
        out_mat.stride(1), out_mat.stride(2), out_mat.stride(0),      # C strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Restore shape (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(out_fp32,
                            (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)

    out_scaled = out_norm * out_gate                     # out_gate is float32

    output = F.linear(out_scaled, to_out_weight)         # final projection to dim

    return output
```
