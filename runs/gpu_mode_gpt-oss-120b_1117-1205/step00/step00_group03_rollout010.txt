def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Main contraction using Triton
    # reshape groups
    left_g = left.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_g = right.permute(0, 3, 1, 2).contiguous()
    G = B * hidden_dim
    left_g = left_g.view(G, N, N)
    right_g = right_g.view(G, N, N)

    # allocate output
    out_g = torch.empty_like(left_g)

    # compute matmul per group: out_g = left_g @ right_g.T
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), G)
    batch_matmul_kernel[grid](
        left_g,
        right_g,
        out_g,
        N, N, N,
        left_g.stride(0), left_g.stride(1), left_g.stride(2),
        right_g.stride(0), right_g.stride(1), right_g.stride(2),
        out_g.stride(0), out_g.stride(1), out_g.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out_g.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    eps_out = 1e-5
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps_out)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # apply out_gate
    out_norm = out_norm * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    final_out = torch.nn.functional.linear(out_norm, to_out_weight)

    return final_out

"""
TriMul forward implementation with Triton acceleration.

The core computation is:
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

We treat each (batch, channel) pair as a separate matrix multiplication:
    left_slice  : (N, N)
    right_slice : (N, N)  (we need its transpose)
    out_slice   : left_slice @ right_slice.T

The heavy N³ contraction is performed by a batched GEMM kernel written in Triton.
All other operations (LayerNorm, linear projections, gating, final LayerNorm and
linear) are done in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ag, stride_am, stride_ak,          # A: (group, M, K)
    stride_bg, stride_bk, stride_bn,          # B_T: (group, K, N)  (B is stored as (group, N, K))
    stride_cg, stride_cm, stride_cn,          # C: (group, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute C = A @ B_T  for a batch of matrices.

    A has shape (G, M, K)   (row‑major)
    B_T is interpreted from B of shape (G, N, K) as B_T[k, n] = B[n, k]
    C has shape (G, M, N)
    """

    pid_m = tl.program_id(0)   # tile over M dimension
    pid_n = tl.program_id(1)   # tile over N dimension
    pid_g = tl.program_id(2)   # group index (batch * hidden_dim)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current tile for this group
    a_base = a_ptr + pid_g * stride_ag
    b_base = b_ptr + pid_g * stride_bg
    c_base = c_ptr + pid_g * stride_cg

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        # Absolute k offsets for this tile
        cur_k = k_tile * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B_T tile (K x N) from original B (stored as N x K)
        # B_T[k, n] = B[n, k]
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module with a Triton‑accelerated contraction.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask       : torch.Tensor [B, N, N] (may be ignored if config["nomask"] is True)
            - weights    : dict of model parameters
            - config     : dict with keys "dim", "hidden_dim", "nomask" (optional)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # LayerNorm on the input (dim = last dimension)
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # compute mean/var over last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # ------------------------------------------------------------------ #
    # Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------ #
    # Optional mask
    # ------------------------------------------------------------------ #
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # Gating
    # ------------------------------------------------------------------ #
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # Main contraction: out = left @ right^T per (batch, channel)
    # ------------------------------------------------------------------ #
    # reshape to (G, N, N) where G = B * hidden_dim
    left_g = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_g = right.permute(0, 3, 1, 2).contiguous()
    G = B * hidden_dim
    left_g = left_g.view(G, N, N)
    right_g = right_g.view(G, N, N)

    out_g = torch.empty_like(left_g)

    # strides (in elements) for the flattened tensors
    stride_ag = left_g.stride(0)   # G stride
    stride_am = left_g.stride(1)   # row stride (M)
    stride_ak = left_g.stride(2)   # col stride (K)

    stride_bg = right_g.stride(0)  # G stride (same as left)
    # For B_T we need: stride over K dimension = original column stride (1)
    # and stride over N dimension = original row stride (N)
    stride_bk = right_g.stride(2)  # original column stride (1)
    stride_bn = right_g.stride(1)  # original row stride (N)

    stride_cg = out_g.stride(0)
    stride_cm = out_g.stride(1)
    stride_cn = out_g.stride(2)

    # Triton launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        G,
    )
    batch_matmul_kernel[grid](
        left_g,
        right_g,
        out_g,
        N,          # M
        N,          # N
        N,          # K
        stride_ag,
        stride_am,
        stride_ak,
        stride_bg,
        stride_bk,
        stride_bn,
        stride_cg,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_g.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # Output LayerNorm
    # ------------------------------------------------------------------ #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Apply out gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    final_out = F.linear(out_norm, to_out_weight)   # shape [B, N, N, dim]

    return final_out
