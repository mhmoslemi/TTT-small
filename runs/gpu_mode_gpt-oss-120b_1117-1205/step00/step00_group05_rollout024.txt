"""
TriMul implementation with a custom Triton GEMM kernel.

The core operation of TriMul is:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

We treat the hidden dimension `h` as an independent batch dimension and
compute a batched matrix multiplication:
    out_h = left_h @ right_h.T

The GEMM is performed by a Triton kernel (`batched_gemm_kernel`) that
processes tiles of size (BLOCK_M, BLOCK_N) with a reduction over K.
All other operations (LayerNorm, Linear projections, gating, final
LayerNorm and Linear) are carried out with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (left @ right.T) for tensors of shape
# (B, M, K) @ (B, K, N) -> (B, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers to A, B, C
    M, N, K,                            # matrix dimensions
    stride_am, stride_ai, stride_ak,    # strides for A (batch, M, K)
    stride_bm, stride_bk, stride_bn,    # strides for B (batch, K, N)
    stride_cm, stride_ci, stride_cn,    # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile row
    pid_n = tl.program_id(1)            # tile column
    pid_b = tl.program_id(2)            # batch (hidden) index

    # Offsets for the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute batch offsets
    a_batch_ptr = a_ptr + pid_b * stride_am
    b_batch_ptr = b_ptr + pid_b * stride_bm
    c_batch_ptr = c_ptr + pid_b * stride_cm

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Pointer offsets for A
        a_k = k_offset + tl.arange(0, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai) + (a_k[None, :] * stride_ak)
        # Pointer offsets for B
        b_k = k_offset + tl.arange(0, BLOCK_K)
        b_ptrs = b_batch_ptr + (b_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # Load tiles (apply boundary masks)
        a_mask = (offs_m[:, None] < M) & (a_k[None, :] < K)
        b_mask = (b_k[:, None] < K) & (offs_n[None, :] < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Rank‑K update
        acc += tl.dot(a, b)

    # Write back the result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module with a Triton-accelerated
    hidden‑dimension batched GEMM.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict of model parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # batch and sequence dimensions from the tensor itself
    B, N, _, _ = input_tensor.shape
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all stored as float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = input_tensor.to(dtype)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)    # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Apply mask if present
    # ------------------------------------------------------------------
    if use_mask and mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(x)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM:
    #   left:  [B, N, N, H] -> (B*H, N, N)
    #   right: [B, N, N, H] -> (B*H, N, N) and transpose K/N
    # ------------------------------------------------------------------
    # Bring hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Merge batch and hidden into a single batch dimension
    batch_h = B * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)          # (B*H, M=N, K=N)
    # Right needs to be transposed on the last two axes before GEMM
    right_mat = right_perm.view(batch_h, N, N).transpose(1, 2).contiguous()  # (B*H, K=N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat, dtype=dtype, device=device)

    # ------------------------------------------------------------------
    # Launch Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # columns
        batch_h                    # batch (hidden) dimension
    )

    batched_gemm_kernel[grid](
        left_mat,                 # a_ptr
        right_mat,                # b_ptr
        out_mat,                  # c_ptr
        N, N, N,                  # M, N, K (all equal to seq_len)
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape result back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and linear projection to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
