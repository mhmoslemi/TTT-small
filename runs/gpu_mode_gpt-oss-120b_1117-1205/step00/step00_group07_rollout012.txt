"""
Highly optimized Triton‑accelerated implementation of the outgoing
Triangle Multiplicative Update (TriMul) forward pass.

The kernel fuses the per‑element mask and gating multiplications:
    left = left_proj * mask * left_gate
    right = right_proj * mask * right_gate
All other linear, sigmoid and LayerNorm operations are performed with
torch.nn.functional for numerical stability.
The dominant O(N³) contraction is expressed as a batched GEMM:
    out = left @ rightᵀ  (performed as torch.bmm on a reshaped tensor)
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,           # *mut f32
    right_ptr,          # *mut f32
    mask_ptr,           # *const f32
    left_gate_ptr,      # *const f32
    right_gate_ptr,     # *const f32
    B, N, H,            # problem sizes (int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    BLOCK: tl.constexpr  # hidden dimension (=H)
):
    pid = tl.program_id(0)                     # index for (b, i, k)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,k) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    k = rem % N

    # base offsets (in element units)
    off_left = b * stride_left_b + i * stride_left_i + k * stride_left_k
    off_right = b * stride_right_b + i * stride_right_i + k * stride_right_k
    off_mask = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    off_lgate = b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k
    off_rgate = b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k

    # load mask (scalar, broadcasted)
    mask = tl.load(mask_ptr + off_mask)

    # thread‑wise offsets in the hidden dimension
    h_off = tl.arange(0, BLOCK)                 # [0, 1, ..., H-1]

    # load vectors (each thread loads ONE element of the H‑dim vector)
    left = tl.load(left_ptr + off_left + h_off * stride_left_h)
    right = tl.load(right_ptr + off_right + h_off * stride_right_h)
    left_gate = tl.load(left_gate_ptr + off_lgate + h_off * stride_lgate_h)
    right_gate = tl.load(right_gate_ptr + off_rgate + h_off * stride_rgate_h)

    # fused mask * gate * projection
    left_new = left * mask * left_gate
    right_new = right * mask * right_gate

    # write back in‑place
    tl.store(left_ptr + off_left + h_off * stride_left_h, left_new)
    tl.store(right_ptr + off_right + h_off * stride_right_h, right_new)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module with a Triton‑fused mask‑gate kernel.

    Args:
        data: (input, mask, weights, config)
            - input: Tensor[bs, seq_len, seq_len, dim]
            - mask:  Tensor[bs, seq_len, seq_len] (float, 0/1)
            - weights: dict of model weights
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        Tensor[bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (final dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Fuse mask & gate into left/right using Triton
    # ------------------------------------------------------------------
    if not nomask:
        # make sure everything is contiguous
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask = mask.contiguous()

        B, N, _, _ = left.shape

        # strides in element units
        s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
        s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
        s_mask_b, s_mask_i, s_mask_k = mask.stride()
        s_lgate_b, s_lgate_i, s_lgate_k, s_lgate_h = left_gate.stride()
        s_rgate_b, s_rgate_i, s_rgate_k, s_rgate_h = right_gate.stride()

        total = B * N * N
        grid = (total,)

        _mask_gate_fuse_kernel[grid](
            left, right, mask,
            left_gate, right_gate,
            B, N, hidden_dim,
            s_left_b, s_left_i, s_left_k, s_left_h,
            s_right_b, s_right_i, s_right_k, s_right_h,
            s_mask_b, s_mask_i, s_mask_k,
            s_lgate_b, s_lgate_i, s_lgate_k, s_lgate_h,
            s_rgate_b, s_rgate_i, s_rgate_k, s_rgate_h,
            BLOCK=hidden_dim,
        )
    else:
        # no mask – just apply the gates (pure PyTorch, cheap)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction: out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    # This is a batched matmul: out = left @ rightᵀ over the k dimension.
    # ------------------------------------------------------------------
    # Rearrange to (B*H, N, N) → (B*H, N, N) @ (B*H, N, N)ᵀ
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)            # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # rightᵀ on the last two dims
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # restore shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return output.to(torch.float32)
