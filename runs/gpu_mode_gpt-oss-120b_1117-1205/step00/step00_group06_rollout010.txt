@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid // H
    h = pid % H

    # compute offsets for i and j
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # base pointers for left and right for this batch and hidden dim h
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left block A[i,k]
        a_ptrs = left_base + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right block B_T[k,j] (transposed)
        b_ptrs = right_base + k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)  # shape (BLOCK_K, BLOCK_N)

        # compute partial product
        acc += tl.dot(a, b)  # result (BLOCK_M, BLOCK_N)
    # store result
    out_ptrs = out_base + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_k
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    # pid0: group id over (batch, hidden)
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden index
    b = pid // H
    h = pid % H

    # offsets in i and j dimensions
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # base pointer for this (b,h) pair
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # load A block [i, k]
        a_ptrs = left_base + i_off[:, None] * stride_l_i + k_off[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B block transposed [k, j]
        b_ptrs = right_base + k_off[:, None] * stride_r_k + j_off[None, :] * stride_r_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate dot product over K
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_base + i_off[:, None] * stride_o_i + j_off[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) implementation with Triton.

The core O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed by a custom Triton kernel. All other operations
(LayerNorms, linear projections, sigmoid gates) are done with PyTorch.
The kernel processes each (batch, hidden) pair independently and
tiles the i‑ and j‑dimensions with 2‑D blocks. It loads A[i,k] and the
transposed B[k,j] in blocks, accumulates a dot‑product over k and
stores the result.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    # program ids
    pid = tl.program_id(0)          # combined (batch, hidden)
    pid_i = tl.program_id(1)        # tile over i dimension
    pid_j = tl.program_id(2)        # tile over j dimension

    # decode batch index and hidden dimension index
    b = pid // H
    h = pid % H

    # offsets for i and j within this tile
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # A block: left[i, k]
        a_ptrs = left_base + i_off[:, None] * stride_l_i + k_off[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # B block transposed: right[k, j]  (originally right[j, k])
        b_ptrs = right_base + k_off[:, None] * stride_r_k + j_off[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # accumulate partial product over k
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # write the tile back to global memory
    out_ptrs = out_base + i_off[:, None] * stride_o_i + j_off[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module with a Triton kernel for the
    O(N³) contraction. Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # LayerNorm on the input (over the last dimension)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # Linear projections
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Optional masking
    # -----------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; broadcast over hidden_dim
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # make tensors contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # Triton kernel: out = Σ_k left[...,i,k,h] * right[...,j,k,h]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # shape [B, N, N, hidden_dim]

    # strides (in elements, not bytes)
    l_stride_b = left.stride(0)
    l_stride_i = left.stride(1)
    l_stride_k = left.stride(2)
    l_stride_h = left.stride(3)

    r_stride_b = right.stride(0)
    r_stride_i = right.stride(1)
    r_stride_k = right.stride(2)
    r_stride_h = right.stride(3)

    o_stride_b = out.stride(0)
    o_stride_i = out.stride(1)
    o_stride_j = out.stride(2)
    o_stride_h = out.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        l_stride_b,
        l_stride_i,
        l_stride_k,
        l_stride_h,
        r_stride_b,
        r_stride_i,
        r_stride_k,
        r_stride_h,
        o_stride_b,
        o_stride_i,
        o_stride_j,
        o_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # -----------------------------------------------------------------
    # Post‑processing: LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # final projection back to dim

    return output
