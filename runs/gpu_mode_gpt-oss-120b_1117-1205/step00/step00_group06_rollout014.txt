@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # row block index
    pid_j = tl.program_id(1)  # col block index
    pid_bc = tl.program_id(2) # batch*channel index

    # decode batch and channel
    # b = pid_bc // H
    # d = pid_bc % H
    b = pid_bc // H
    d = pid_bc % H

    # compute start offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask to handle boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for left and right
    # left base: left_ptr + b*stride_left_b + d*stride_left_d
    # left uses i and k dimensions. We'll compute pointer offsets per block.
    # We'll also need pointer for right base: right_ptr + b*stride_right_b + d*stride_right_d
    
    # flatten pointers:
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for k_idx in range(0, num_blocks_k):
        offs_k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A block: left[ b, i, k, d ] shape (BLOCK_M, BLOCK_K)
        # offsets as (i_offset, k_offset): left_base + i*stride_left_i + k*stride_left_k 
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # load with mask

    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

"""
tri_mul_kernel
==============

This file implements the "outgoing" TriMul operation from AlphaFold3 using a
custom Triton kernel for the heavy O(N³) contraction:

    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

The kernel treats the operation as a batched matrix multiplication for each
(batch, channel) pair, i.e. a GEMM between `left[b, d]` of shape (N, N) and the
transpose of `right[b, d]`.  A 3‑D grid (i‑block, j‑block, batch*channel) tiles
the output matrix and iterates over the reduction dimension `k` in blocks.
The rest of the TriMul forward pass (layer‑norms, linear projections, gating)
is performed with PyTorch for simplicity.

The kernel works with half‑precision inputs (float16) for memory bandwidth
reduction while accumulating in float32 to preserve numerical stability.
The final output is cast back to float32 before the final LayerNorm and linear
projection.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    left_ptr,          # const half *
    right_ptr,         # const half *
    out_ptr,           # half *
    # problem size
    B, N, H,           # batch, seq_len, hidden_dim
    # strides for left (shape [B, N, N, H])
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right (shape [B, N, N, H])
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    # strides for out (shape [B, N, N, H])
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul.

    Computes for a fixed (batch, hidden) pair:
        out[i, j] = Σₖ left[i, k] * right[j, k]
    """
    pid_i = tl.program_id(0)      # block row index (i)
    pid_j = tl.program_id(1)      # block column index (j)
    pid_bc = tl.program_id(2)     # combined batch * channel index

    # decode batch and hidden index
    b = pid_bc // H
    d = pid_bc % H

    # linear offsets for batch and channel (common for all accesses)
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base = out_ptr + b * stride_out_b + d * stride_out_d

    # start indices for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right block (transposed): shape (BLOCK_K, BLOCK_N)
        # we need right[j, k] → use k as row, j as column
        b_ptrs = right_base \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write results back (convert to half)
    c_ptrs = out_base \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (broadcastable)
            - weights: dict of model weight tensors
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (dim‑wise)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to half for the heavy contraction
    # ------------------------------------------------------------------
    left_half = left.to(torch.float16).contiguous()
    right_half = right.to(torch.float16).contiguous()
    # free fp32 intermediates early
    del left, right, x
    torch.cuda.empty_cache()

    B, N, _, _ = left_half.shape

    # ------------------------------------------------------------------
    # 6. Allocate output tensor (half)
    # ------------------------------------------------------------------
    out_half = torch.empty_like(left_half, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 7. Prepare strides (in element units, not bytes)
    # ------------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_half.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_half.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out_half.stride()

    # ------------------------------------------------------------------
    # 8. Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_gemm_kernel[grid](
        left_half,
        right_half,
        out_half,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_d,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9. Cast back to float32 for the remaining pipeline
    # ------------------------------------------------------------------
    out = out_half.to(torch.float32)

    # ------------------------------------------------------------------
    # 10. Post‑contraction LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
