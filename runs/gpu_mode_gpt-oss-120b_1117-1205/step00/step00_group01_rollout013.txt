import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ B  (A:(i,k), B:(k,j)).
    A, B, C are 3‑D tensors of shape [batch, M, K], [batch, K, N], [batch, M, N].
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this tile
    a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_offset = k * BLOCK_K
        cur_k = offs_k + k_offset

        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        mask_b = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store the result tile
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    Performs:
      * LayerNorm on the input.
      * Linear projections + gating.
      * Core tensor contraction sum_k left[...,k,d] * right[...,k,d] via a custom batched‑matmul kernel.
      * Output LayerNorm, gating, and final linear projection.
    Returns a float32 tensor of shape [batch, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of separate linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matmul for each hidden dimension.
    # ------------------------------------------------------------------
    # Move hidden dimension into the batch axis.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    # Flatten B and H → total_batch
    left_mat = left_perm.view(bs * hidden_dim, N, N)    # (B*H, i, k)
    right_mat = right_perm.view(bs * hidden_dim, N, N)  # (B*H, k, j)

    out_mat = torch.empty((bs * hidden_dim, N, N), dtype=dtype, device=device)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batch = bs * hidden_dim
    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides (in elements)
    a_batch_stride = left_mat.stride(0)
    b_batch_stride = right_mat.stride(0)
    c_batch_stride = out_mat.stride(0)

    stride_am = left_mat.stride(1)   # i stride
    stride_ak = left_mat.stride(2)   # k stride
    stride_bk = right_mat.stride(1)  # k stride (row)
    stride_bn = right_mat.stride(2)  # j stride (col)
    stride_cm = out_mat.stride(1)    # i stride
    stride_cn = out_mat.stride(2)    # j stride

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        a_batch_stride, b_batch_stride, c_batch_stride,
        stride_am, stride_ak, stride_bk, stride_bn,
        stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output.to(torch.float32)
