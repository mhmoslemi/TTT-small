@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    m_mask = offs_m < M
    n_mask = offs_n < N

    # pointer offset to this batch (batch_idx)
    a_batch_offset = batch_idx * stride_a_batch
    b_batch_offset = batch_idx * stride_b_batch
    c_batch_offset = batch_idx * stride_c_batch

    # a block pointer
    a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
    # b block pointer (B_T)
    b_ptrs = b_ptr + b_batch_offset + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute actual k offset
        a = tl.load(a_ptrs, mask=m_mask[:, None] & (offs_k < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=n_mask[None, :] & (offs_k < K - k*BLOCK_K), other=0.0)

        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_k

"""
TriMul forward (outgoing) implementation with a fused batched GEMM kernel in Triton.

The heavy O(N³) operation is:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batch of independent matrix multiplications:
    out[b,d] = left[b,d] @ right[b,d].T   (shape N×N)

We compute the linear projections, LayerNorm and gating in PyTorch, then launch a
custom Triton kernel that performs the batched GEMM on the (hidden_dim) channel
dimension. The final LayerNorm, gating and output projection are also done in
PyTorch. The kernel works for arbitrary batch size (≤2) and hidden dimension,
handles masks, and supports non‑multiple‑of‑block sizes via masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,                # left  [B*H, N, N]  (row-major)
    b_ptr,                # right [B*H, N, N]  (row-major, will be accessed transposed)
    c_ptr,                # out   [B*H, N, N]
    batch,                # total number of (batch, channel) pairs = B*H
    M, N, K,              # matrix dimensions (all equal to seq_len)
    stride_a_batch,       # stride to next (batch,channel) matrix in A (N*N)
    stride_a_row,         # stride between rows in A (N)
    stride_a_col,         # stride between columns in A (1)
    stride_b_batch,       # stride to next (batch,channel) matrix in B (N*N)
    stride_b_k,           # stride for k dimension in transposed view of B (1)
    stride_b_n,           # stride for n dimension in transposed view of B (N)
    stride_c_batch,       # stride to next (batch,channel) matrix in C (N*N)
    stride_c_row,         # stride between rows in C (N)
    stride_c_col,         # stride between columns in C (1)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # start indices of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # predicates for boundary handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch, channel) pair
    a_base = a_ptr + batch_idx * stride_a_batch
    b_base = b_ptr + batch_idx * stride_b_batch
    c_base = c_ptr + batch_idx * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a = tl.load(
            a_base + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block (K x N) using transposed view
        b = tl.load(
            b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module using a Triton kernel for the
    core pairwise multiplication (einsum). The rest of the computation (LayerNorm,
    linear projections and gating) is performed with PyTorch.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict of tensors with keys:
                "norm.weight", "norm.bias", "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias", "to_out.weight"
            - config: dict containing at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------------------
    # LayerNorm on the input (dim‑dimensional)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections to hidden_dim (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # Prepare for batched GEMM: reshape to [B, H, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape

    # Output tensor for the fused matmul
    out_perm = torch.empty_like(left_perm)  # [B, H, N, N]

    # -----------------------------------------------------------------------------
    # Triton kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                     # batch * hidden dimension
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # columns
    )

    # Strides (all are Python ints)
    stride_a_batch = left_perm.stride(1)   # N*N
    stride_a_row = left_perm.stride(2)     # N
    stride_a_col = left_perm.stride(3)     # 1

    stride_b_batch = right_perm.stride(1)  # N*N
    # For the transposed view of B we exchange the last two strides
    stride_b_k = right_perm.stride(3)      # 1 (original column)
    stride_b_n = right_perm.stride(2)      # N (original row)

    stride_c_batch = out_perm.stride(1)    # N*N
    stride_c_row = out_perm.stride(2)      # N
    stride_c_col = out_perm.stride(3)      # 1

    batched_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B * H,
        N,
        N,
        N,
        int(stride_a_batch),
        int(stride_a_row),
        int(stride_a_col),
        int(stride_b_batch),
        int(stride_b_k),
        int(stride_b_n),
        int(stride_c_batch),
        int(stride_c_row),
        int(stride_c_col),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before proceeding
    torch.cuda.synchronize()

    # -------------------------------------------------------------------------
    # Restore original layout: [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # LayerNorm on hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
