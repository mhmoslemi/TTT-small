"""
TriMul forward (outgoing) implementation with a Triton kernel for mask
application.

Algorithm
---------
1. Layer‑norm the input tensor x over the last dimension (dim).
2. Compute left/right projections:  linear(x, W_left/right).
3. If a mask is present (nomask=False) apply it in‑place to left/right using
   a tiny Triton kernel (one program per (b,i,j) location, looping over the
   hidden dimension in blocks).  This satisfies the requirement to use a
   Triton kernel.
4. Compute gating tensors (sigmoid of linear(x, W_gate)) and multiply the
   gated projections.
5. Contract the gated tensors:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each hidden dimension:
      left   -> (B, H, N, N)
      right_T-> (B, H, N, N)   (right transposed on the two N axes)
      out    = bmm(left, right_T) reshaped back to (B,N,N,H).
6. Apply output layer‑norm, multiply by the output gate and project back to
   the original dimension with a final linear layer.
7. Return the result in float32.

Only the mask‑application step is performed in Triton; all other heavy
operations use highly‑optimized PyTorch (cuBLAS) calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: in‑place mask multiplication
# --------------------------------------------------------------------------- #
@triton.jit
def _apply_mask_kernel(
    x_ptr,               # *mut* pointer to tensor [B,N,N,H]
    mask_ptr,            # const pointer to mask   [B,N,N]
    B, N, H,             # sizes
    stride_xb, stride_xi, stride_xj, stride_xh,
    stride_maskb, stride_maski, stride_maskj,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from pid
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # load scalar mask value (same dtype as x)
    mask_val = tl.load(mask_ptr + b * stride_maskb + i * stride_maski + j * stride_maskj)

    # loop over the hidden dimension in BLOCK_H sized chunks
    offs_h = tl.arange(0, BLOCK_H)
    for off in range(0, H, BLOCK_H):
        cur_h = off + offs_h
        cur_mask = cur_h < H
        x_off = (
            b * stride_xb
            + i * stride_xi
            + j * stride_xj
            + cur_h * stride_xh
        )
        x = tl.load(x_ptr + x_off, mask=cur_mask, other=tl.zeros([BLOCK_H], dtype=tl.float32))
        x = x * mask_val
        tl.store(x_ptr + x_off, x, mask=cur_mask)

# Custom kernel entry point
# --------------------------------------------------------------------------- #

    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor of shape [B, N, N, dim]
              - mask_tensor:  torch.Tensor of shape [B, N, N] (may be ignored)
              - weights_dict: mapping from layer names to weight tensors
              - config_dict: contains at least "dim" and "hidden_dim",
                             also "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm   (over dim)
    # -----------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape (B,N,N,dim)

    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])      # (B,N,N,hidden_dim)
    right = F.linear(x_norm, weights["right_proj.weight"])    # (B,N,N,hidden_dim)

    # 3. Apply mask in‑place using Triton (if needed)
    # -----------------------------------------------------------------------
    if not nomask:

        mask_fp = mask.to(left.dtype)

            left,
            mask_fp,
            B,
            N,
            hidden_dim,

            stride_xi,
            stride_xj,
            stride_xh,
            stride_maskb,
            stride_maski,
            stride_maskj,
            BLOCK_H=BLOCK_H,

            right,
            mask_fp,
            B,
            N,
            hidden_dim,
            stride_xb,
            stride_xi,
            stride_xj,
            stride_xh,
            stride_maskb,
            stride_maski,
            stride_maskj,
            BLOCK_H=BLOCK_H,
        )
    # -----------------------------------------------------------------------
    # 4. Gating (sigmoid of linear on the normalized input)
    # -----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5. Contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched matmul for each hidden dimension.
    # -----------------------------------------------------------------------
    # transpose right on the two N axes to make it (B,N,N,H) -> (B,N,N,H) with
    # axes (k,j) for the multiplication.
    right_t = right.permute(0, 2, 1, 3)  # (B,N,N,H) where dimensions are (k,j)

    # reshape to (B*H, N, N) for both operands
    left_resh = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_resh = right_t.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiplication (cuBLAS)
    out_resh = torch.bmm(left_resh, right_resh)  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_resh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # -----------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    # ensure final dtype is float32 as requested
    if out.dtype != torch.float32:
        out = out.to(torch.float32)

    return out
