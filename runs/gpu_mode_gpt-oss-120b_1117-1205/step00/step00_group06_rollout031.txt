def custom_kernel(data):
    # Unpack
    input_tensor, mask, weights, config = data

    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Ensure weights are float32 and on same device
    # Might already be.

    # Cast input to float32
    x = input_tensor.to(torch.float32)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x,
                                            normalized_shape=(dim,),
                                            weight=norm_weight,
                                            bias=norm_bias,
                                            eps=eps)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Normalize mask to float and apply
    if mask is None:
        mask_expanded = torch.ones_like(left[..., :1], dtype=torch.float32, device=device)
    else:
        mask_expanded = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask_expanded
    right = right * mask_expanded

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute contraction via Triton kernel
    B, N, _, H = left.shape  # left shape [B,N,N,hidden_dim]

    # Permute to bring hidden dim as second axis for flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Flatten batch * hidden
    batch_H = B * H
    left_flat = left_perm.view(batch_H, N, N)
    right_flat = right_perm.view(batch_H, N, N)

    # Transpose right for multiplication
    right_T = right_flat.transpose(1, 2)  # shape (batch_H, N, N)

    # Cast to float16 for kernel computation
    left_fp16 = left_flat.to(torch.float16)

```python
"""Implementation of the outgoing TriMul operator using Triton for the core
contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.  The kernel
performs a batched matrix multiplication of shape (B*H, N, N) × (B*H, N, N)^T
with tiling in the M, N and K dimensions.  The surrounding LayerNorm, linear
projections and gating are kept in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_out_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_Ab, stride_am, stride_ak,
    stride_Bb, stride_bk, stride_bn,
    stride_Cb, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute batched GEMM C = A @ B where
        A: [batch, M, K] (fp16)
        B: [batch, K, N] (fp16)
        C: [batch, M, N] (fp16) – stored as fp16, accumulated in fp32.
    Tiling parameters BLOCK_M/N/K are compile‑time constants.
    """
    pid_z = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Pointers to the start of the current tile in A and B
    a_ptrs = A_ptr + pid_z * stride_Ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_z * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        k_range = tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        a_mask = mask_m[:, None] & mask_k[None, :]
        b_mask = mask_k[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)   # fp32 accumulation

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back the result (convert to fp16)
    c_ptrs = C_ptr + pid_z * stride_Cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (float/bool) – broadcasted on last dim.
            - weights: dict of torch.Tensors containing the model weights.
            - config: dict with keys "dim" and "hidden_dim".
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Cast everything to float32 for the host‑side operations
    x = input_tensor.to(torch.float32)

    # ----- LayerNorm on the input -----
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ----- Linear projections (no bias) -----
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left = F.linear(x_norm, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight) # [B, N, N, hidden_dim]

    # ----- Mask (if provided) -----
    if mask is None:
        mask_expanded = torch.ones_like(left[..., :1], dtype=torch.float32, device=device)
    else:
        mask_expanded = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask_expanded
    right = right * mask_expanded

    # ----- Gating -----
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)

    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Core contraction via Triton -----
    B, N, _, H = left.shape  # H == hidden_dim
    # Move hidden dimension next to batch for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch_H = B * H
    left_flat = left_perm.view(batch_H, N, N)          # [B*H, N, N]
    right_flat = right_perm.view(batch_H, N, N)        # [B*H, N, N]

    # Transpose right for the GEMM (K,N) layout
    right_T = right_flat.transpose(1, 2)               # [B*H, N, N] (K,N)

    # Cast to fp16 for the kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_T.to(torch.float16)

    # Allocate output buffer
    out_fp16 = torch.empty((batch_H, N, N), dtype=torch.float16, device=device)

    # Strides (in elements, not bytes)
    stride_a_batch = left_fp16.stride(0)
    stride_am = left_fp16.stride(1)
    stride_ak = left_fp16.stride(2)

    stride_b_batch = right_fp16.stride(0)
    stride_bk = right_fp16.stride(1)
    stride_bn = right_fp16.stride(2)

    stride_c_batch = out_fp16.stride(0)
    stride_cm = out_fp16.stride(1)
    stride_cn = out_fp16.stride(2)

    # Tiling parameters – chosen to fit typical GPU shared‑memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_H,
    )
    batched_gemm_out_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 and restore the original layout
    out = out_fp16.to(torch.float32)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ----- Output LayerNorm, gating and final linear -----
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate

    to_out_weight = weights["to_out.weight"].to(torch.float32)
    output = F.linear(out_norm, to_out_weight)  # [B, N, N, dim]

    return output
```

"""Outgoing TriMul implementation.
The heavy contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched matrix multiplication
    (B*H, N, N) × (B*H, N, N)^T
and computed with a tiled Triton kernel.  All other
operations (LayerNorm, linear projections, gating) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_Ab, stride_am, stride_ak,
    stride_Bb, stride_bk, stride_bn,
    stride_Cb, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ B where
        A: [batch, M, K] (fp16)
        B: [batch, K, N] (fp16)
        C: [batch, M, N] (fp16, stored as fp16, accumulated in fp32)
    """
    pid_z = tl.program_id(2)   # batch (B*H)
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        cur_k = k_start + k_range
        mask_k = cur_k < K

        # Load A tile
        a = tl.load(
            A_ptr
            + pid_z * stride_Ab
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Load B tile (B is already transposed, shape [batch, K, N])
        b = tl.load(
            B_ptr
            + pid_z * stride_Bb
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Write out the tile (convert to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_Cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict of model weights
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input
    # -----------------------------------------------------------------
    x = input_tensor.to(torch.float32)
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float32)
    right_proj_w = weights["right_proj.weight"].to(torch.float32)
    left = F.linear(x_norm, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_w)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Apply mask (if provided)
    # -----------------------------------------------------------------
    if mask is None:
        mask_exp = torch.ones_like(left[..., :1], dtype=torch.float32, device=device)
    else:
        mask_exp = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask_exp
    right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float32)
    right_gate_w = weights["right_gate.weight"].to(torch.float32)
    out_gate_w = weights["out_gate.weight"].to(torch.float32)

    left_gate = F.linear(x_norm, left_gate_w).sigmoid()
    right_gate = F.linear(x_norm, right_gate_w).sigmoid()
    out_gate = F.linear(x_norm, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Move hidden dimension next to batch and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    batch_H = B * H
    left_flat = left_perm.view(batch_H, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(batch_H, N, N) # [B*H, N, N]

    # Transpose right for GEMM (K,N)
    right_T = right_flat.transpose(1, 2)  # [B*H, N, N] with layout (K,N)

    # Cast to fp16 for the kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_T.to(torch.float16)

    # Output buffer
    out_fp16 = torch.empty((batch_H, N, N), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_a_batch = left_fp16.stride(0)
    stride_am = left_fp16.stride(1)
    stride_ak = left_fp16.stride(2)

    stride_b_batch = right_fp16.stride(0)
    stride_bk = right_fp16.stride(1)
    stride_bn = right_fp16.stride(2)

    stride_c_batch = out_fp16.stride(0)
    stride_cm = out_fp16.stride(1)
    stride_cn = out_fp16.stride(2)

    # Tiling parameters (tuned for typical B200 memory constraints)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_H,                   # batch dimension (B*H)
    )
    batched_gemm_out_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 and restore original layout: [B, N, N, H]
    out = out_fp16.to(torch.float32)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Final LayerNorm, gating and output projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out_norm = out_norm * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float32)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim]

    return output
