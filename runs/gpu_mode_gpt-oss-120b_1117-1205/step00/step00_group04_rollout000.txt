def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights['left_proj.weight']  # shape (hidden_dim, dim)
    right_proj_weight = weights['right_proj.weight']
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    if mask is not None:
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Compute einsum using Triton
    # Reshape for batched GEMM
    B, N, _, _ = left.shape
    # left shape after gating: [B,N,N,hidden_dim]
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    total_batch = B * hidden_dim
    # reshape to (total_batch, N, N)
    left_mat = left_perm.view(total_batch, N, N)
    right_mat = right_perm.view(total_batch, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)  # same dtype and device

    # Launch Triton kernel
    # Define block size constants
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # may adjust for occupancy
    # compute grid dimensions
    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    grid = (total_batch, num_blocks_m, num_blocks_n)

    # get strides
    stride_l_batch = left_mat.stride(0)  # [batch, row, col]
    stride_l_m = left_mat.stride(1)
    stride_l_k = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_k = 1  # since need transposed: we can compute from right_mat.stride(2) which is 1 but we want k stride = right_mat.stride(2) = 1
    stride_r_n = right_mat.stride(1)  # row stride (j)

    # Need to ensure right strides: right_mat.stride(0) = batch stride, stride(1) = row stride (original j), stride(2) = col stride (original k)
    # For transposed, k stride = right_mat.stride(2) = 1, n stride = right_mat.stride(1) = N.

    stride_o_batch = out_mat.stride(0)
    stride_o_m = out_mat.stride(1)
    stride_o_n = out_mat.stride(2)

    triton_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        total_batch,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # optional others,
    )

    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # shape [B,N,N,hidden_dim]

    # to_out_norm
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # multiply with out_gate
    out = out * out_gate

    # final linear
    to_out_weight = weights['to_out.weight']  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)

    # return
    return out

@triton.jit
def triton_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute mask for rows and cols within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for k block
        k_offset = k * BLOCK_K

        # Pointers for left (M,K) tile
        left_ptrs = left_ptr + batch_id * stride_l_batch \
                    + (offs_m[:, None] * stride_l_m) \
                    + ((k_offset + offs_k)[None, :] * stride_l_k)

        # Load left tile with mask for rows M
        a = tl.load(left_ptrs, mask=mask_m[:, None], other=0.0)

        # Pointers for right (K,N) tile; we treat right with k as row and n as col
        right_ptrs = right_ptr + batch_id * stride_r_batch \
                     + ((k_offset + offs_k)[:, None] * stride_r_k) \
                     + (offs_n[None, :] * stride_r_n)

        # Load right tile with mask for cols N
        b = tl.load(right_ptrs, mask=mask_n[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

        # Advance k offset implicitly

    # Write output
    out_ptrs = out_ptr + batch_id * stride_o_batch \
               + (offs_m[:, None] * stride_o_m) \
               + (offs_n[None, :] * stride_o_n)

    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

Note: we need to ensure that dot works for float32. For a shape (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N), tl.dot returns matrix multiply.

Now call kernel.

"""
TriMul custom kernel (outgoing version).

The core computation is:
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

where left/right are linear projections (with gating and optional mask) of the
input tensor.  This is equivalent to a batched matrix multiplication
    out[b, d] = left[b, d] @ right[b, d].T
with left/right shaped (N, N) per batch‑and‑channel.  We fuse the N³
multiplication across the hidden dimension using a Triton kernel that treats
(b, d) as a combined batch dimension.

The rest of the TriMul pipeline (LayerNorms, linear projections, gating,
final linear) is performed in PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (left @ right.T) over the k dimension.
# ----------------------------------------------------------------------
@triton.jit
def _gemm_batch_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_len, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ right.T for many independent (M×K)×(K×N) products."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers into left (M×K) tile
        left_ptrs = left_ptr + batch_id * stride_l_batch \
                    + (offs_m[:, None] * stride_l_m) \
                    + (offs_k[None, :] * stride_l_k)
        a = tl.load(left_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        # Pointers into right (K×N) tile (right is accessed in transposed order)
        right_ptrs = right_ptr + batch_id * stride_r_batch \
                     + (offs_k[:, None] * stride_r_k) \
                     + (offs_n[None, :] * stride_r_n)
        b = tl.load(right_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store the result
    out_ptrs = out_ptr + batch_id * stride_o_batch \
               + (offs_m[:, None] * stride_o_m) \
               + (offs_n[None, :] * stride_o_n)
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a fused Triton GEMM.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask_tensor : torch.Tensor [B, N, N] (may be all‑ones)
            weights_dict: mapping from weight names to torch.Tensors
            config_dict : must contain "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input layer‑norm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not config.get("nomask", False):
        # mask shape: [B,N,N] -> broadcast to last dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core einsum via batched GEMM (Triton)
    #    out = sum_k left[...,k,:] * right[...,k,:]  -->  left @ right.T  per channel
    # --------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dimension into the batch axis for the GEMM kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    total_batch = B * hidden_dim
    left_mat = left_perm.view(total_batch, N, N)
    right_mat = right_perm.view(total_batch, N, N)

    out_mat = torch.empty_like(left_mat)  # [total_batch, N, N]

    # Tile sizes (tuned for B200; fits in shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions: (batch, rows, cols)
    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides for the left matrix (M dimension = rows = i)
    stride_l_batch = left_mat.stride(0)
    stride_l_m = left_mat.stride(1)
    stride_l_k = left_mat.stride(2)

    # Strides for the right matrix (we need K as rows, N as cols)
    stride_r_batch = right_mat.stride(0)
    stride_r_k = right_mat.stride(2)   # original K dimension (col stride = 1)
    stride_r_n = right_mat.stride(1)   # original N dimension (row stride = N)

    # Strides for the output matrix
    stride_o_batch = out_mat.stride(0)
    stride_o_m = out_mat.stride(1)
    stride_o_n = out_mat.stride(2)

    # Launch the kernel
    _gemm_batch_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        total_batch,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm and final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
