import torch
import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(val_ptr, mask_ptr, gate_ptr, out_ptr,
                           total_elems, hidden_dim, BLOCK: tl.constexpr):
    """Apply elementwise: out = val * gate * mask

    val_ptr: pointer to flattened tensor of shape (B,N,N,hidden)
    mask_ptr: pointer to flattened mask of shape (B,N,N)
    gate_ptr: pointer to flattened gate of same shape as val_ptr
    out_ptr: output pointer (can be same as val_ptr for in-place)
    total_elems: total number of elements in val_ptr (int)
    hidden_dim: size of hidden dimension (int)
    BLOCK: compile‑time block size
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load data
    val = tl.load(val_ptr + offs, mask=mask, other=tl.float32(0.0))
    gate = tl.load(gate_ptr + offs, mask=mask, other=tl.float32(0.0))

    # map flat index to mask index (each mask element repeats hidden_dim times)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float32(0.0))

    out = val * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated implementation of the outgoing TriMul operation.

    Arguments:
        data: (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square in the sequence dimensions"
    hidden_dim = config["hidden_dim"]

    # ----- Load weights ----------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm on input ---------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B,N,N,dim) float32

    # ----- Linear projections ------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----- Gates ------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()  # (B,N,N,hidden)

    # ----- Prepare data for the Triton kernel -------------------------------
    # Convert mask to float32 for multiplication
    mask_f = mask.to(torch.float32)  # (B,N,N)
    # Flatten tensors to 1‑D contiguous buffers
    left_flat = left.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_f.contiguous().view(-1)  # (B*N*N)

    total_elems = left_flat.numel()          # B * N * N * hidden
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    # ----- Apply mask and gates in a fused Triton kernel --------------------
    apply_mask_gate_kernel[grid](
        left_flat,
        mask_flat,
        left_gate_flat,
        left_flat,          # in‑place write
        total_elems,
        hidden_dim,
        BLOCK=BLOCK,
    )
    apply_mask_gate_kernel[grid](
        right_flat,
        mask_flat,
        right_gate_flat,
        right_flat,         # in‑place write
        total_elems,
        hidden_dim,
        BLOCK=BLOCK,
    )

    # ----- Reshape back ----------------------------------------------------
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # ----- Batched GEMM (left @ right^T) ------------------------------------
    # Shape to (B*hidden, N, N) for both tensors
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Matrix multiply: each (b, h) pair: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape to original 4‑D form
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ----- Output layer norm ------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ----- Apply output gate ------------------------------------------------
    out = out * out_gate  # both float32

    # ----- Final linear projection -----------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)

"""TriMul (outgoing) forward pass with a Triton kernel.

The kernel fuses the mask and per‑channel gating:
    out = value * gate * mask
for tensors with shape (B, N, N, H).  The heavy O(N³) reduction is performed
as a batched GEMM (left @ rightᵀ) using torch.bmm, while the mask‑gate
multiplication is accelerated with Triton.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def apply_mask_gate_kernel(
    val_ptr,        # *float32, flattened tensor (B,N,N,H)
    mask_ptr,       # *float32, flattened mask (B,N,N)
    gate_ptr,       # *float32, flattened gate (B,N,N,H)
    out_ptr,        # *float32, output (can alias val_ptr)
    total_elems,    # int, number of elements in the value tensor
    hidden_dim,     # int, size of the last dimension H
    BLOCK: tl.constexpr,
):
    """Apply element‑wise mask and gate:
    out = val * gate * mask   (mask is broadcast over the hidden dimension).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elems

    # Load tensors
    v = tl.load(val_ptr + offs, mask=active, other=tl.float32(0.0))
    g = tl.load(gate_ptr + offs, mask=active, other=tl.float32(0.0))

    # Each mask element repeats `hidden_dim` times in the flattened value tensor
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float32(0.0))

    res = v * g * m
    tl.store(out_ptr + offs, res, mask=active)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights: dict containing all model parameters
            - config: dict with at least `"dim"` and `"hidden_dim"`

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "pairwise input must be square"
    hidden_dim = config["hidden_dim"]

    # -------------------- Load weights -----------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # -------------------- Input LayerNorm -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim), float32

    # -------------------- Linear projections ----------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -------------------- Gates (sigmoid) -------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()   # (B,N,N,hidden)

    # -------------------- Prepare for Triton ----------------------------
    # Cast mask to float32 and flatten
    mask_f = mask.to(torch.float32).contiguous()
    mask_flat = mask_f.view(-1)                           # (B*N*N,)

    # Flatten tensors (they are already contiguous after linear)
    left_flat = left.contiguous().view(-1)                # (B*N*N*H,)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)

    total_elements = left_flat.numel()                    # B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    # -------------------- Fuse mask & gate with Triton ------------------
    apply_mask_gate_kernel[grid](
        left_flat,
        mask_flat,
        left_gate_flat,
        left_flat,                # in‑place update
        total_elements,
        hidden_dim,
        BLOCK=BLOCK,
    )
    apply_mask_gate_kernel[grid](
        right_flat,
        mask_flat,
        right_gate_flat,
        right_flat,               # in‑place update
        total_elements,
        hidden_dim,
        BLOCK=BLOCK,
    )

    # -------------------- Reshape back ---------------------------------
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # -------------------- Batched GEMM: left @ rightᵀ -------------------
    # Shape to (B*H, N, N) for both operands
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Matrix multiply per hidden channel
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Return to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------- Output LayerNorm ------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -------------------- Apply output gate -----------------------------
    out = out * out_gate

    # -------------------- Final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
