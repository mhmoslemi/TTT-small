import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_lbm, stride_lbi, stride_lbk,
    stride_rbm, stride_rbj, stride_rbk,
    stride_obm, stride_obi, stride_obj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)   # batch index (M)
    pid_i = tl.program_id(1)   # block row
    pid_j = tl.program_id(2)   # block col

    # Compute start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets within block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch
    left_base = left_ptr + pid_m * stride_lbm
    right_base = right_ptr + pid_m * stride_rbm
    out_base = out_ptr + pid_m * stride_obm

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (i,k)
        a_ptrs = left_base + (offs_i[:, None] * stride_lbi) + (offs_k[None, :] * stride_lbk)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B block (k,j) i.e., right transposed
        b_ptrs = right_base + (offs_k[:, None] * stride_rbk) + (offs_j[None, :] * stride_rbj)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = out_base + (offs_i[:, None] * stride_obi) + (offs_j[None, :] * stride_obj)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

Check pointer arithmetic: stride_lbk is stride for k dimension (innermost). Since the layout is (M,N,N), stride_lbk=1. stride_lbi = stride for i dimension = N. stride_lbm = stride for M dimension = N*N.

Right pointer: we use stride_rbk for k dimension, stride_rbj for j dimension. Offsets: we first offset by k (row) then j (col). So we compute pointer = base + k_offset * stride_rbk + j_offset * stride_rbj. Works.

        right = right * mask_exp

        pass

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))

    right = right * right_gate

    to_out_weight = weights["to_out.weight"]

"""
TriMul kernel (outgoing version) with a Triton GEMM implementation.

The algorithm follows the reference PyTorch implementation:
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections (left, right) and three gating projections.
3. Apply the optional mask and gates.
4. Compute the tensor contraction
       out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
   with a custom Triton kernel that performs a batched GEMM over the (i,k)
   and (j,k) dimensions.  The hidden dimension is folded into the batch
   dimension, turning the operation into M independent N×N matrix multiplies
   (M = batch * hidden_dim).
5. Layer‑norm the result, apply the output gate and a final linear projection.

Only the heavy contraction (step 4) is in Triton; all other ops use
torch.nn.functional for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,             # *float32   [M, N, N]
    right_ptr,            # *float32   [M, N, N]
    out_ptr,              # *float32   [M, N, N]
    N: tl.constexpr,      # size of the sequence dimension
    # strides for the left tensor
    stride_lbm, stride_lbi, stride_lbk,
    # strides for the right tensor
    stride_rbm, stride_rbj, stride_rbk,
    # strides for the output tensor
    stride_obm, stride_obi, stride_obj,
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[m, i, j] = Σₖ left[m, i, k] * right[m, j, k]
    for all m = 0 … M‑1 where M = batch * hidden_dim.
    The tensors are stored in row‑major order (C‑contiguous).
    """
    pid_m = tl.program_id(0)          # batch * hidden_dim dimension
    pid_i = tl.program_id(1)          # block row
    pid_j = tl.program_id(2)          # block col

    # starting indices of this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the current m (batch*hidden) slice
    left_base = left_ptr + pid_m * stride_lbm
    right_base = right_ptr + pid_m * stride_rbm
    out_base = out_ptr + pid_m * stride_obm

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction axis K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (BLOCK_M, BLOCK_K)  -> indices (i, k)
        a_ptrs = left_base + (offs_i[:, None] * stride_lbi) + (offs_k[None, :] * stride_lbk)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block: need right transposed, i.e. (k, j)
        # we load as (BLOCK_K, BLOCK_N) using offsets (k, j)
        b_ptrs = right_base + (offs_k[:, None] * stride_rbk) + (offs_j[None, :] * stride_rbj)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # multiply‑accumulate
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → acc: (M,N)

    # write the result
    c_ptrs = out_base + (offs_i[:, None] * stride_obi) + (offs_j[None, :] * stride_obj)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] (may be unused)
            - weights      : dict of torch.Tensors containing all model weights
            - config       : dict with keys dim, hidden_dim, nomask (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack arguments
    x, mask, weights, config = data
    eps = 1e-5

    # ensure FP32 computation
    x = x.to(torch.float32)
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ---------- 1. LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # ---------- 2. Linear projections ----------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # ---------- 3. Optional mask ----------
    if not config.get("nomask", False):
        # mask is broadcast over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- 4. Gates ----------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Tensor contraction (heavy part) ----------
    # reshape to (M, N, N) where M = B * hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides (int) for the three tensors
    stride_lbm = left_flat.stride(0)
    stride_lbi = left_flat.stride(1)
    stride_lbk = left_flat.stride(2)

    stride_rbm = right_flat.stride(0)
    stride_rbj = right_flat.stride(1)
    stride_rbk = right_flat.stride(2)

    stride_obm = out_flat.stride(0)
    stride_obi = out_flat.stride(1)
    stride_obj = out_flat.stride(2)

    # block size – 64 is a good default for the B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,                                 # program_id(0) – batch*hidden
        triton.cdiv(N, BLOCK_M),          # program_id(1) – block row
        triton.cdiv(N, BLOCK_N),          # program_id(2) – block col
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_lbm, stride_lbi, stride_lbk,
        stride_rbm, stride_rbj, stride_rbk,
        stride_obm, stride_obi, stride_obj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 6. Output LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps
    )

    # ---------- 7. Output gate ----------
    out = out * out_gate

    # ---------- 8. Final linear projection ----------
    to_out_w = weights["to_out.weight"]
    output = F.linear(out, to_out_w)   # [B, N, N, dim]

    return output
