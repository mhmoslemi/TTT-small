@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_hidden = tl.program_id(2)  # 0 .. B*H-1
    b = batch_hidden // H
    d = batch_hidden % H

    # calculate pointer offsets for this batch-d combination
    left_base = left_ptr + b * stride_left_b + d * stride_left_h
    right_base = right_ptr + b * stride_right_b + d * stride_right_h
    out_base = out_ptr + b * stride_out_b + d * stride_out_h

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        # left tile: shape (BLOCK_M, BLOCK_K) (i, k)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N) (k, j)
        right_ptrs = right_base + offs_k[:, None] * stride_right_i + offs_j[None, :] * stride_right_k
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate dot
        acc += tl.dot(left_tile, right_tile)  # left (M,K) dot right (K,N)
    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16?) Actually need float32 final, so store as float32. Use tl.float32.

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)
    b = pid_batch // H
    d = pid_batch % H

    # Compute offsets to base per batch and hidden dim
    # using local pointers for memory
    left_base = left_ptr + b * stride_l_b + d * stride_l_h
    right_base = right_ptr + b * stride_r_b + d * stride_r_h
    out_base = out_ptr + b * stride_o_b + d * stride_o_h

    # Offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M,K)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile transposed: (K,N) where K=blk K, N=blk N
        right_ptrs = right_base + offs_k[:, None] * stride_r_i + offs_j[None, :] * stride_r_k

        acc += tl.dot(left, right)
    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate

"""
Highly optimized Triton implementation of the outgoing TriMul operation.

Algorithm
---------
1. Apply LayerNorm over the channel dimension of the input tensor.
2. Compute left/right projections and corresponding gating values with
   bias‑free linear layers followed by sigmoid.
3. Apply the optional mask and element‑wise gating.
4. Fuse the O(N³) einsum:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   into a batched matrix‑multiply using a Triton kernel that iterates over
   the k‑dimension in tiles (BLOCK_K).  Each hidden channel (d) is treated
   as an independent batch, so the kernel works on tensors of shape
   [B, H, N, N] where H = hidden_dim.
5. Cast the result back to float32, apply a second LayerNorm,
   multiply by the output gate, and finish with a bias‑free linear projection
   back to the original channel dimension.

The Triton kernel performs only the heavy triple‑sum; all other
operations (LayerNorm, linear projections, gating) are done with
native PyTorch for simplicity and numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *bfloat16
    right_ptr,         # *bfloat16
    out_ptr,           # *bfloat16
    B, N, H,           # scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile row (i)
    pid_j = tl.program_id(1)          # tile col (j)
    pid_batch = tl.program_id(2)      # combined batch * hidden channel

    # Decode batch index b and hidden channel d
    b = pid_batch // H
    d = pid_batch % H

    # Base pointers for this (b,d) slice
    left_base  = left_ptr  + b * stride_l_b + d * stride_l_h
    right_base = right_ptr + b * stride_r_b + d * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + d * stride_o_h

    # Offsets for the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)  (i, k)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)  (k, j)
        right_ptrs = right_base + offs_k[:, None] * stride_r_i + offs_j[None, :] * stride_r_k
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Accumulate the matrix product (FP16 inputs → FP32 accumulation)
        acc += tl.dot(left_tile, right_tile)

    # Write out the result (cast back to bfloat16)
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc.to(tl.bfloat16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def triton_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) TriMul reduction using Triton.
    Expected dtype: torch.bfloat16
    Shapes: [B, N, N, H]
    Returns tensor of same shape and dtype.
    """
    B, N, N2, H = left.shape
    assert N == N2, "Input must be square in the two middle dimensions"
    out = torch.empty_like(left)

    # Strides (in number of elements)
    ls = left.stride()
    rs = right.stride()
    os = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * H)

    tri_mul_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        ls[0], ls[1], ls[2], ls[3],
        rs[0], rs[1], rs[2], rs[3],
        os[0], os[1], os[2], os[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton‑accelerated core.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N]  (may be all‑ones)
            - weights: dict of model parameters
            - config: dict containing "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Unpack weights (all are float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)     # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional masking (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel for the triple‑sum (einsum)
    # ------------------------------------------------------------------
    # Cast to bfloat16 for the kernel (matches reference precision)
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Compute the O(N³) reduction
    out_bf = triton_trimul(left_bf, right_bf)   # [B, N, N, hidden_dim] bfloat16

    # Back to float32 for the remaining steps
    out = out_bf.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7) Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear mapping back to original dimension
    # ------------------------------------------------------------------
    output = F.linear(out, to_out_weight)   # weight shape (dim, hidden_dim)

    return output
