```
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, row_stride_a, col_stride_a,
    batch_stride_b, row_stride_b, col_stride_b,
    batch_stride_c, row_stride_c, col_stride_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch_idx = pid // num_blocks_m
    block_i = pid % num_blocks_m
    block_j = pid_n

    # Compute offsets for i and j
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute K-loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A
        a_ptrs = a_ptr + batch_idx * batch_stride_a + offs_i[:, None] * row_stride_a + offs_k[None, :] * col_stride_a
        b_ptrs = b_ptr + batch_idx * batch_stride_b + offs_k[:, None] * row_stride_b + offs_j[None, :] * col_stride_b

        a = tl.load(a_ptrs, mask=(offs_i < M) & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K) & (offs_j < N), other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_idx * batch_stride_c + offs_i[:, None] * row_stride_c + offs_j[None, :] * col_stride_c
    tl.store(c_ptrs, acc, mask=(offs_i < M) & (offs_j < N))

```
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_aa, stride_am, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cc, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch_idx = pid // num_blocks_m
    block_i = pid % num_blocks_m
    block_j = pid_n

    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # pointers
        a_ptrs = a_ptr + batch_idx * stride_aa + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_idx * stride_bb + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn

        # masks
        mask_a = (offs_i[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_j[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # store
    c_ptrs = c_ptr + batch_idx * stride_cc + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    mask_c = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)

"""
TriMul outgoing kernel (AlphaFold3).

The forward pass consists of:
1. Layer‑norm over the last dimension.
2. Linear projections (left/right) and gating (left/right/out) applied to the normalized
   tensor (with optional mask).
3. Core O(N³·D) operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication of shape
       (B·D, N, N) @ (B·D, N, N)ᵀ .
   This is performed by a custom Triton kernel that tiles the three dimensions.
4. Second layer‑norm, out‑gate multiplication and final linear projection back to `dim`.

Only the heavy matrix‑multiplication is in Triton; the rest uses PyTorch
for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_aa, stride_am, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cc, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul C = A @ B where A,B,C have shape (batch, M, K) and (batch, K, N)."""
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch_idx = pid // num_blocks_m                     # which (B·D) we are processing
    block_i = pid % num_blocks_m                        # tile row of A / C
    block_j = pid_n                                     # tile col of B / C

    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers with proper strides
        a_ptrs = a_ptr + batch_idx * stride_aa + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_idx * stride_bb + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn

        mask_a = (offs_i[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_j[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_ptr + batch_idx * stride_cc + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    mask_c = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (ignored if config["nomask"] is True)
            - weights: dict of weight tensors required by the module
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device
    # work in float32 for stability
    x = input_tensor.to(torch.float32)

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (input norm)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)

    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        mask_f = mask.to(torch.float32).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·D) multiplication via Triton
    # ------------------------------------------------------------------
    # reshape to (B*D, N, N) and prepare the transposed right operand
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, D, N, N]
    right_t = right_perm.transpose(-2, -1).contiguous()  # (K,J) -> (K,J) for matmul

    B, D, N, _ = left_perm.shape
    batch_hidden = B * D

    a = left_perm.reshape(batch_hidden, N, N)
    b = right_t.reshape(batch_hidden, N, N)
    c = torch.empty_like(a)  # output placeholder

    # strides in element counts
    stride_aa = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    stride_bb = b.stride(0)
    stride_bk = b.stride(1)   # corresponds to K dimension after transpose
    stride_bn = b.stride(2)   # N dimension

    stride_cc = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    # tiling parameters (tuned for typical N≤1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)

    grid = (batch_hidden * num_blocks_m, num_blocks_n)

    batch_matmul_kernel[grid](
        a, b, c,
        stride_aa, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cc, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = c.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Output LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(torch.float32)
    output = F.linear(out, to_out_weight)  # final projection to dim

    return output
