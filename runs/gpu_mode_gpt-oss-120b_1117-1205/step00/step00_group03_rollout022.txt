"""
custom_kernel: Triton‑accelerated forward pass of the outgoing TriMul operator.

The module consists of:
1. LayerNorm on the input features.
2. Two linear projections (left/right) with optional masking.
3. Gating (sigmoid of linear layers) applied to the projections.
4. A fused contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batch of GEMMs (one per hidden channel) using a
   Triton kernel that tiles the i/j dimensions and loops over k.
5. LayerNorm on the hidden result, multiplication by the output gate,
   and a final linear projection back to the original feature dimension.

Only the heavy O(N³) contraction is performed in Triton; all other
operations use the highly‑optimized PyTorch kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_contract_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,               # A: rows (i), cols (k)
    stride_bk, stride_bn,               # B: rows (k), cols (j) -> B is accessed transposed
    stride_cm, stride_cn,               # C: rows (i), cols (j)
    stride_batch_a, stride_batch_b, stride_batch_c,
    stride_da, stride_db, stride_dc,    # stride on hidden dimension (normally 1)
    HIDDEN_DIM: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM computing the TriMul contraction for one hidden channel."""
    # --- batch / hidden index -------------------------------------------------
    pid = tl.program_id(2)                     # combined batch * hidden index
    batch_idx = pid // HIDDEN_DIM
    hid = pid % HIDDEN_DIM                      # hidden channel inside the batch

    # --- i / j tile ids -------------------------------------------------------
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # base pointers for this (batch, hidden) pair
    a_base = A_ptr + batch_idx * stride_batch_a + hid * stride_da
    b_base = B_ptr + batch_idx * stride_batch_b + hid * stride_db
    c_base = C_ptr + batch_idx * stride_batch_c + hid * stride_dc

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile
        a_ptrs = a_base + (offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_base + (offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn)

        # load with masking, cast to FP32 for the dot
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # matrix‑multiply the tile (M×K) · (K×N) → (M×N)
        acc += tl.dot(a, b)

    # write back the tile
    c_ptrs = c_base + (offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    contraction step.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask_tensor : [B, N, N]       (bool / int / float)
            - weights_dict: mapping of weight names to tensors
            - config_dict : contains "dim", "hidden_dim" and optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------------------------------
    # unpack arguments
    # -------------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------------------
    # extract model weights
    # -------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -------------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -------------------------------------------------------------------------
    # Cast to half for the cheap linear/gate ops
    # -------------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # -------------------------------------------------------------------------
    # 2) Linear projections (half precision)
    # -------------------------------------------------------------------------
    left = F.linear(x_h, left_proj_weight.to(torch.float16))   # [B,N,N,H]
    right = F.linear(x_h, right_proj_weight.to(torch.float16))

    # -------------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # -------------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # -------------------------------------------------------------------------
    # 4) Gating (half precision)
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight.to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight.to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5) Contraction via Triton (produces float32)
    # -------------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # block sizes (tuned for B200 – can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides for the tensors (element‑wise strides, not bytes)
    stride_a_batch = left.stride(0)
    stride_a_i = left.stride(1)
    stride_a_k = left.stride(2)
    stride_a_d = left.stride(3)

    stride_b_batch = right.stride(0)
    stride_b_k = right.stride(2)   # k dimension (after transpose)
    stride_b_j = right.stride(1)   # j dimension
    stride_b_d = right.stride(3)

    stride_c_batch = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)
    stride_c_d = out.stride(3)

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    _tri_mul_contract_kernel[grid](
        left,
        right,
        out,
        M=N,
        N=N,
        K=N,
        stride_am=stride_a_i,
        stride_ak=stride_a_k,
        stride_bk=stride_b_k,
        stride_bn=stride_b_j,
        stride_cm=stride_c_i,
        stride_cn=stride_c_j,
        stride_batch_a=stride_a_batch,
        stride_batch_b=stride_b_batch,
        stride_batch_c=stride_c_batch,
        stride_da=stride_a_d,
        stride_db=stride_b_d,
        stride_dc=stride_c_d,
        HIDDEN_DIM=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------------------
    # 6) Output LayerNorm (float32)
    # -------------------------------------------------------------------------
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias)

    # Apply the final output gate (convert to float32 for the multiplication)
    out_gate_f = out_gate.to(out_norm.dtype)
    out_norm = out_norm * out_gate_f

    # -------------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # -------------------------------------------------------------------------
    # to_out.weight has shape (dim, hidden_dim)
    final = F.linear(out_norm, to_out_weight)

    return final
