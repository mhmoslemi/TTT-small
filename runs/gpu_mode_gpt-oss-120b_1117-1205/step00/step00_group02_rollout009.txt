import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(A, B, C,
                        batch, M, N, K,
                        stride_Ab, stride_Am, stride_Ak,
                        stride_Bb, stride_Bk, stride_Bn,
                        stride_Cb, stride_Cm, stride_Cn,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles of A and B
        a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        a = tl.load(a_ptrs, mask = mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask = mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask = mask_m[:, None] & mask_n[:, None])

def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward pass.
    This implementation computes the O(N^3 * hidden) contraction using a
    batched GEMM kernel in Triton.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm over dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = F.linear(x, right_proj_weight)

    # Apply mask if present (mask shape [B,N,N])
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Prepare tensors for batched GEMM: reshape to (B*H, N, N)
    B, N, _, _ = left.shape  # B,N,N,hidden
    # Permute to bring hidden dimension next to batch for easy view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, i/k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)
    # Merge batch and hidden dims
    batch = B * hidden_dim
    A = left_perm.view(batch, N, N)
    B_mat = right_perm.view(batch, N, N)

    # Allocate output
    C = torch.empty_like(A)

    # Get strides
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bk, stride_Bn = B_mat.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    # Block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_M), batch)
    batched_gemm_kernel[grid](
        A, B_mat, C,
        batch, N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to (B,N,N,hidden)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

"""
Triton-accelerated Triangle Multiplicative Update (TriMul) forward pass.

The core O(N^3) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM kernel in Triton, where the hidden
dimension is merged into the batch dimension.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(A, B, C,
                        batch, M, N, K,
                        stride_Ab, stride_Am, stride_Ak,
                        stride_Bb, stride_Bk, stride_Bn,
                        stride_Cb, stride_Cm, stride_Cn,
                        BLOCK_M: tl.constexpr,
                        BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        a = tl.load(a_ptrs, mask = mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask = mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask = mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator using a Triton GEMM kernel.
    
    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (binary or float)
            - weights: dict of model weight tensors
            - config: dict with keys "dim" and "hidden_dim"
    
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ---------- LayerNorm over dim ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (B, N, N, hidden)
    right = F.linear(x, right_proj_weight)

    # ---------- Apply mask ----------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_
        right = right * mask_

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- Batched GEMM (TriMul contraction) ----------
    B, N, _, _ = left.shape  # (batch, seq_len, seq_len, hidden)

    # Move hidden dimension next to batch for easy view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K)
    # Right must be transposed on the last two dims (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    batch = B * hidden_dim
    A = left_perm.view(batch, N, N)   # (batch, M=N, K=N)
    B_mat = right_perm.view(batch, N, N)  # (batch, K=N, N=N)

    C = torch.empty_like(A)

    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bk, stride_Bn = B_mat.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    # Block sizes – chosen to fit into shared memory and make good use of Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(N, BLOCK_M),
        batch,
    )
    batched_gemm_kernel[grid](
        A, B_mat, C,
        batch, N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout (B, N, N, hidden)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output normalisation ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ---------- Apply out gate ----------
    out = out * out_gate

    # ---------- Final linear ----------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The O(N³·hidden) contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM kernel in Triton, where the hidden
dimension is folded into the batch dimension. All other operations (LayerNorm,
Linear, gating) remain in PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    A, B, C,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)               # batch (merged B*hidden)
    pid_m = tl.program_id(1)               # tile along M (i)
    pid_n = tl.program_id(0)               # tile along N (j)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator using Triton.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (binary or float)
            - weights: dict of weight tensors
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ---------- LayerNorm over the channel dimension ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (B, N, N, hidden)
    right = F.linear(x, right_proj_weight)

    # ---------- Apply mask ----------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_
        right = right * mask_

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- Batched GEMM (TriMul contraction) ----------
    B, N, _, _ = left.shape  # batch size, sequence length, hidden dim already known
    # Prepare tensors so that hidden dimension is merged into the batch dimension
    # left: (B, N, N, H) -> (B, H, N, K) where K = N
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    # right needs a transpose on the last two axes (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    batch = B * hidden_dim
    A = left_perm.view(batch, N, N)     # (batch, M=N, K=N)
    B_mat = right_perm.view(batch, N, N)  # (batch, K=N, N=N)

    C = torch.empty_like(A)

    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bk, stride_Bn = B_mat.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    # Tile sizes – tuned for the A100‑class hardware (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_N),   # along j
        triton.cdiv(N, BLOCK_M),   # along i
        batch,                     # merged batch*hidden
    )
    batched_gemm_kernel[grid](
        A, B_mat, C,
        batch, N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear ----------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out
