@triton.jit
def triton_trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_c = tl.program_id(0)  # batch*hidden index
    pid_m = tl.program_id(1)  # i tile
    pid_n = tl.program_id(2)  # j tile

    total_channels = B * H
    b = pid_c // H
    h = pid_c % H

    # offsets: i and j
    i_base = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_base < N
    mask_j = j_base < N

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        # k indices
        k_base = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # left: shape (BLOCK_M, BLOCK_K)
        left_offset = b * stride_left_b + i_base[:, None] * stride_left_i + k_base[None, :] * stride_left_k + h * stride_left_h
        left = tl.load(left_ptr + left_offset, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right: we need shape (BLOCK_K, BLOCK_N)
        # load as transposed: index (k, j)
        right_offset = b * stride_right_b + k_base[:, None] * stride_right_k + j_base[None, :] * stride_right_i + h * stride_right_h
        right = tl.load(right_ptr + right_offset, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product accumulate
        acc += tl.dot(left, right)  # left (M,K) dot right (K,N) -> (M,N)

    # write result
    out_offset = b * stride_out_b + i_base[:, None] * stride_out_i + j_base[None, :] * stride_out_j + h * stride_out_h
    # store only valid entries
    tl.store(out_ptr + out_offset, acc, mask=mask_i[:, None] & mask_j[None, :])

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    
    # LayerNorm on input
    # Use torch.nn.functional.layer_norm; passing weight,bias.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    
    # Linear projections and gates
    # linear: weight shape hidden_dim x dim
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()

"""
TriMul (outgoing) forward pass optimized with Triton.

The heavy O(N³) part:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where left/right = linear projections followed by element‑wise gates and mask.
We compute this contraction as a batch of independent GEMM operations
(left_i,k × right_j,kᵀ) for each hidden dimension d and each batch.
A Triton kernel tiles the i, j dimensions (BLOCK_M × BLOCK_N) and iterates
over k in tiles (BLOCK_K), accumulating in FP32.  All other operations
(LayerNorm, Linear, sigmoid, final LayerNorm/Linear) are performed with
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = Σₖ left[...,i,k,:] * right[...,j,k,:] for a single
    hidden dimension h and batch b.
    Grid dimensions:
        program_id(0) -> (b * H + h)   (total channels)
        program_id(1) -> i tile
        program_id(2) -> j tile
    """
    pid_c = tl.program_id(0)   # combined batch * hidden index
    pid_m = tl.program_id(1)   # i tile index
    pid_n = tl.program_id(2)   # j tile index

    total_channels = B * H
    b = pid_c // H
    h = pid_c % H

    # tile offsets for the i and j axes
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_off = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left tile   : (M, K)  with indices (i, k)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_off[:, None] * stride_left_i
            + k_off[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile  : (K, N)  with indices (k, j)  (transposed view)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + k_off[:, None] * stride_right_k
            + j_off[None, :] * stride_right_i
            + h * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # GEMM accumulation: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:        torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict of torch.Tensor model parameters
            - config:  dict containing at least "dim", "hidden_dim",
                       optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on the input -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional mask -----
    if not config.get("nomask", False):
        # mask: [B,N,N] -> [B,N,N,1] for broadcasting
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # ----- Gates -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Triton contraction (triangle multiplicative update) -----
    # Ensure contiguous tensors for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides expressed in number of elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tiling parameters (tuned for B200, can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_channels = B * hidden_dim
    grid = (
        total_channels,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing: to_out_norm, out_gate, final linear -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
