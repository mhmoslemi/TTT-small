def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # LayerNorm of input over dim (last dimension)
    # Use torch.nn.functional.layer_norm
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    
    # Linear projections (no bias)
    # left_proj = linear(x_norm, left_proj_weight)
    # right_proj = linear(x_norm, right_proj_weight)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask
    mask_ = mask.unsqueeze(-1).to(left.dtype)  # broadcast
    left = left * mask_
    right = right * mask_

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguity for kernel loads
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape  # shape B, N, N, hidden_dim
    # Allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=device)

    # Compute strides
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lh = left.stride(3)

    stride_rb = right.stride(0)
    stride_rj = right.stride(1)  # actually j dimension (original j)
    stride_rk = right.stride(2)  # k dimension
    stride_rh = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    # Define block sizes adaptively (maybe 128) that fit in shared memory.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid dimensions
    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    # combined batch and hidden dimension:
    grid_z = B * hidden_dim
    grid = (num_i, num_j, grid_z)

    # Launch kernel
    trimul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Now apply to_out_norm (layernorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=1e-5)
    # Gating
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

```
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # compute batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # base offsets for batch
    off_b = b * stride_lb

    # compute i,j offsets
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds i/j
    mask_i = i_base < N
    mask_j = j_base < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # Compute pointers: left_ptr + b*stride_lb + i_base*stride_li + k_base*stride_lk + h*stride_lh
        left_ptrs = left_ptr + off_b \
            + i_base[:, None] * stride_li \
            + k_base[None, :] * stride_lk \
            + h * stride_lh

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile: shape (BLOCK_N, BLOCK_K)
        # Right pointer uses j dimension for rows (j_base) and k dimension for columns.
        right_ptrs = right_ptr + off_b \
            + j_base[:, None] * stride_rj \
            + k_base[None, :] * stride_rk \
            + h * stride_rh

        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)

    # final store
    # compute out pointers
    out_ptrs = out_ptr + b * stride_ob \
        + i_base[:, None] * stride_oi \
        + j_base[None, :] * stride_oj \
        + h * stride_oh

    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_ij)
```

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # decode batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # base offsets for this batch
    off_b_l = b * stride_lb
    off_b_r = b * stride_rb
    off_b_o = b * stride_ob

    # tile offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for i and j boundaries
    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(0, num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + off_b_l \
            + i[:, None] * stride_li \
            + k[None, :] * stride_lk \
            + h * stride_lh
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: shape (BLOCK_K, BLOCK_N)  (k as rows, j as cols)
        right_ptrs = right_ptr + off_b_r \
            + k[:, None] * stride_rk \
            + j[None, :] * stride_rj \
            + h * stride_rh
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate matrix multiplication
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # store result
    out_ptrs = out_ptr + off_b_o \
        + i[:, None] * stride_oi \
        + j[None, :] * stride_oj \
        + h * stride_oh
    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_ij)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    Performs LayerNorm, linear projections with gating, a batched
    matrix multiplication over the sequence dimension via a custom
    Triton kernel, followed by output LayerNorm, gating and final
    projection back to the model dimension.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- Load weights (ensure same device/dtype) -----
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # ----- LayerNorm over model dimension -----
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ----- Linear projections (no bias) -----
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----- Apply mask (if any) -----
    mask_exp = mask.unsqueeze(-1).to(left.dtype)   # shape [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ----- Gating -----
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ----- Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous memory for kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape  # N is seq_len, hidden_dim = H

    # Prepare output tensor for the kernel (intermediate before final projection)
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=device)

    # Strides for tensors
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # ---- Triton kernel launch parameters ----
    # Block sizes (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    grid_z = B * hidden_dim

    grid = (num_i, num_j, grid_z)

    # Launch
    trimul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ----- Output processing -----
    # LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to model dimension
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,

):

    out[b,i,j,h] = sum_{k=0}^{N-1} left[b,i,k,h] * right[b,j,k,h]
    where `right` is accessed with its second and third axes swapped
    (i.e. we compute left @ right^T for each hidden channel)."""

    mask_j = j < N

    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- Load model weights onto the correct device -----
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)

    left_proj_weight  = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight  = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight   = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device)
    to_out_weight      = weights["to_out.weight"].to(device)

    # ----------------------------------------------------------------------
    # 1. LayerNorm over the model dimension
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----------------------------------------------------------------------
    # 3. Apply mask (if any) and gating
    # ----------------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape  # shape = (B, N, N, hidden_dim)

    # ----------------------------------------------------------------------
    # 4. Core Triple‑Multiplicative Update using Triton
    # ----------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=device)

    # Strides (batch, i, k, hidden)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile sizes – chosen to give good occupancy on NVIDIA GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    grid_z = B * hidden_dim  # each (batch, hidden) gets its own slice

    grid = (num_i, num_j, grid_z)

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # 5. Output LayerNorm, gate and final linear projection
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # back to `dim`

    return out
