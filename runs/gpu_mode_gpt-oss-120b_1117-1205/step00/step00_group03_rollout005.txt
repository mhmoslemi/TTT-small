def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config['dim'], config['hidden_dim']
    device = input_tensor.device
    dtype = input_tensor.dtype # expected float32
    
    # LayerNorm on input
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    eps=1e-5
    # Compute mean,var across dim (last dim)
    # input shape [B,N,N,dim]
    x = input_tensor
    # flatten last dim dimension for layernorm: compute mean over dim
    # Use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5
    )
    # Linear projections
    left_proj_weight = weights['left_proj.weight']  # shape (hidden_dim, dim)
    right_proj_weight = weights['right_proj.weight']
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)  # shape B,N,N,1
        left = left * mask
        right = right * mask
    # Gate weights
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    right = right * right_gate

TriMul forward pass with Triton acceleration.

The heavy O(N³) reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
using a custom Triton kernel. All remaining operations
(LayerNorm, Linear projections, gating, final LayerNorm and linear)
are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ   (left/right: [B, N, N, H])
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    N, K,                                    # sequence dimensions (both = seq_len)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    H: tl.constexpr,                         # hidden dimension (compile‑time constant)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1)      # col block (j)
    pid_z = tl.program_id(2)      # combined batch × hidden index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    bh = pid_z                     # flattened batch*hidden index

    # recover batch and hidden index from the flattened index
    b = bh // H
    h = bh - b * H

    # base pointers for this (b, h)
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    # offsets inside the block
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K

        # ---- load A block (i, k) ----
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k_cur[None, :], other=0.0)
        a = a.to(tl.float32)

        # ---- load B block transposed (k, j) ----
        b_ptrs = b_base + cur_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_j
        b = tl.load(b_ptrs, mask=mask_k_cur[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # ---- store result ----
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, dim]
            - mask:  torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of model parameters (tensors)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on input (over the last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]    # [hidden_dim, dim]

    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(input_tensor.shape[:3], dtype=x.dtype, device=device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)               # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core Triton kernel: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, H), dtype=dtype, device=device)

    stride_left = left.stride()   # (batch, i, k, h)
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows (i)
        triton.cdiv(N, BLOCK_N),          # cols (j)
        B * hidden_dim                    # combined batch × hidden
    )

    trimul_matmul_kernel[grid](
        left,
        right,
        out,
        N,                                   # N
        N,                                   # K (same as N)
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Output LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate
    to_out_weight = weights["to_out.weight"]      # [dim, hidden_dim]
    output = F.linear(out, to_out_weight)        # [B, N, N, dim]

    return output
