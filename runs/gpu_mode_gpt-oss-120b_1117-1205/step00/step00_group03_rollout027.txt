@triton.jit
def tri_mul_kernel(A, B, C, N, batch_h,
                   stride_ah, stride_ai, stride_ak,
                   stride_bh, stride_bi, stride_bk,
                   stride_ch, stride_ci, stride_ck,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1) # i tile
    pid_n = tl.program_id(2) # j tile

    # Compute offsets for i and j within matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute base pointers for current batch
    a_ptr_base = A + batch_id * stride_ah
    b_ptr_base = B + batch_id * stride_bh
    c_ptr_base = C + batch_id * stride_ch

    # Pointers for a sub block: shape (BLOCK_M, BLOCK_K)
    # Offsets for k will be iterated in loop
    # Allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers:
        a_ptrs = a_ptr_base + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr_base + (offs_k[:, None] * stride_bi) + (offs_n[None, :] * stride_bk)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute block multiplication:
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N) returns (BLOCK_M, BLOCK_N)

    # Store result
    c_ptrs = c_ptr_base + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_ck)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul "outgoing" forward pass with a Triton kernel for the O(N^3) tensor contraction.
    The kernel computes, for each batch and hidden dimension, a matrix multiplication
    A @ B^T where A and B are the left/right projected tensors. The rest of the
    operations (LayerNorm, linear projections, gating, final LayerNorm and linear)
    are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm on last dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # apply mask if present
    if not config.get("nomask", False):
        # mask shape [bs, N, N] -> broadcast to hidden dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp
    # gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    # permute to (bs, hidden_dim, N, N) and flatten batch & hidden dim
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = bs * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)
    out_flat = torch.empty_like(left_flat)  # float16
    
    # define kernel strides
    stride_a_batch = left_flat.stride(0)
    stride_a_i = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)
    stride_b_batch = right_flat.stride(0)
    stride_b_i = right_flat.stride(1)   # originally j dimension
    stride_b_k = right_flat.stride(2)
    stride_c_batch = out_flat.stride(0)
    stride_c_i = out_flat.stride(1)
    stride_c_k = out_flat.stride(2)

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (batch_h,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        # strides
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape to original layout
    out_perm = out_flat.view(bs, hidden_dim, N, N)
    out = out_perm.permute(0, 2, 3, 1).to(torch.float32)  # [bs, N, N, hidden_dim]
    # final layernorm + gating + linear
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

@triton.jit
def tri_mul_kernel(
    A, B, C,
    N,
    # strides
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bi, stride_bk,
    stride_ch, stride_ci, stride_ck,
    # compile-time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B^T for tensors of shape (batch, N, N)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)  # i tile
    pid_n = tl.program_id(2)  # j tile

    # compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the current batch
    a_batch_ptr = A + batch_id * stride_ah
    b_batch_ptr = B + batch_id * stride_bh
    c_batch_ptr = C + batch_id * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for A block (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        # pointers for B^T block (K x N). B is stored as (row=j, col=k); we need (k, j)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bi) + (offs_n[None, :] * stride_bk)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # block matrix multiplication
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # store result in FP16
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_ck)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched matrix multiplication C = A @ Bᵀ, where
A and B are the left/right projected tensors reshaped to
(batch_h, N, N) with batch_h = batch * hidden_dim.
The kernel computes the multiplication in FP16 while accumulating
in FP32, then the result is cast back to FP32 for the remaining
LayerNorm / gating / linear layers which are kept in PyTorch.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def tri_mul_kernel(
    A, B, C,
    N,
    # strides for A
    stride_ah, stride_ai, stride_ak,
    # strides for B (original layout, we access it as Bᵀ)
    stride_bh, stride_bi, stride_bk,
    # strides for C
    stride_ch, stride_ci, stride_ck,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ for tensors (batch_h, N, N)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # tile along i dimension
    pid_n = tl.program_id(2)   # tile along j dimension

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = A + batch_id * stride_ah
    b_batch_ptr = B + batch_id * stride_bh
    c_batch_ptr = C + batch_id * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: (M, K)   layout i * stride_ai + k * stride_ak
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ block: (K, N)
        # B is stored as (row=j, col=k). To read Bᵀ we treat k as the row index.
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bi) + (offs_n[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matmul update
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write result in FP16
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_ck)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul module using a Triton kernel for the
    N³ contraction. The rest of the arithmetic (LayerNorms, Linear
    projections, gating, final projection) is performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to FP16 for the kernel ----------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ---------- Rearrange to (batch_h, N, N) ----------
    # (bs, N, N, hidden) -> (bs, hidden, N, N) -> flatten (batch_h, N, N)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = bs * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)

    # ---------- Output buffer ----------
    out_flat = torch.empty_like(left_flat)   # FP16

    # ---------- Strides ----------
    stride_a_batch = left_flat.stride(0)
    stride_a_i = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_i = right_flat.stride(1)   # original "j" dimension
    stride_b_k = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_i = out_flat.stride(1)
    stride_c_k = out_flat.stride(2)

    # ---------- Triton launch ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_c_batch,
        stride_c_i,
        stride_c_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Restore original layout ----------
    out_perm = out_flat.view(bs, hidden_dim, N, N)                     # (bs, hidden, N, N)
    out = out_perm.permute(0, 2, 3, 1).to(torch.float32)              # (bs, N, N, hidden)

    # ---------- Final LayerNorm, gating and projection ----------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
