import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_x = tl.program_id(0) # i block
    pid_y = tl.program_id(1) # j block
    pid_z = tl.program_id(2) # batch*channel

    b = pid_z // D
    d = pid_z % D

    # base offsets
    i_base = pid_x * BLOCK_M
    j_base = pid_y * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M,K)
        left_ptrs = left_ptr + b * stride_left_b + d * stride_left_d + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # load right tile transposed (K,N) where K = offs_k, N = offs_j
        right_ptrs = right_ptr + b * stride_right_b + d * stride_right_d + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right_T = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right_T = right_T.to(tl.float32)

        # dot product (M,K) * (K,N) -> (M,N)
        acc += tl.dot(left, right_T)
    # store results
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) implementation with a Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last feature dimension.
2. Linear projections (left/right) and three gating linear layers.
3. Apply the optional mask and the left/right gates.
4. Core O(N³) update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication per channel (d) and batch (b)
   that we implement in Triton.  The kernel iterates over tiles of the
   (i,j) output plane (BLOCK_M × BLOCK_N) and accumulates partial
   products over the k dimension (BLOCK_K).  The inner dimension is
   summed in float32 for numerical stability, while inputs are stored
   as float16.
5. Layer‑norm over the hidden dimension, apply the output gate and a
   final linear projection back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # [B, N, N, D]  (float16)
    right_ptr,         # [B, N, N, D]  (float16)
    out_ptr,           # [B, N, N, D]  (float32)
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Perform out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]"""
    pid_i = tl.program_id(0)  # i‑tile
    pid_j = tl.program_id(1)  # j‑tile
    pid_z = tl.program_id(2)  # combined batch and hidden channel

    # decode batch and channel from pid_z
    b = pid_z // D
    d = pid_z % D

    # tile start offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile (M × K) -----
        left_ptrs = left_ptr \
            + b * stride_l_b + d * stride_l_d \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- right tile transposed (K × N) -----
        # we load as (k, j) which is the transposed view needed for dot
        right_ptrs = right_ptr \
            + b * stride_r_b + d * stride_r_d \
            + offs_k[:, None] * stride_r_k \
            + offs_j[None, :] * stride_r_j
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0).to(tl.float32)

        # matrix multiplication of (M×K) @ (K×N) → (M×N)
        acc += tl.dot(left, right_T)

    # write output
    out_ptrs = out_ptr \
        + b * stride_o_b + d * stride_o_d \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # shapes: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(left.dtype)  # broadcast over hidden dim
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Triton kernel for the O(N³) contraction
    # ------------------------------------------------------------------
    # Cast to fp16 for more bandwidth, accumulation stays in fp32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, D = left_fp16.shape  # D == hidden_dim
    out = torch.empty((B, N, N, D), device=device, dtype=torch.float32)

    # strides (in element units, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_d = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Tile sizes – chosen to fit into shared memory while keeping good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * D,
    )
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, D,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_j, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output post‑processing
    # ------------------------------------------------------------------
    # LayerNorm over hidden dimension
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # Apply output gate and final projection
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
