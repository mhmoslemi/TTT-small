custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32 ; use x possibly float32.
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # fetch eps
    eps = 1e-5

    # LayerNorm on input
    # compute mean var across -1 (dim)
    # using dtype=float32
    x = input_tensor
    
    # compute mean across dim
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True) # unbiased=False
    x_norm = (x - mean) / torch.sqrt(var + eps)
    # scale and bias
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]
    # note: weights["norm.weight"] and bias shape [dim]; broadcast over preceding dims.

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask
    mask_unsq = mask.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply left/right gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel to save memory
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to [B, H, N, N] and contiguous
    left_perm = left_fp16.permute(0,3,1,2).contiguous()
    right_perm = right_fp16.permute(0,3,1,2).contiguous()

    # Prepare output tensor
    out_fp32 = torch.empty_like(left_perm, dtype=torch.float32, device=device)

    # Kernel loop over groups
    B_val, H_val, N_val, _ = left_perm.shape
    # set block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # call kernel for each (b,h)
    # get strides for each slice
    # Flatten loops
    for b_idx in range(B_val):
        for h_idx in range(H_val):
            a_slice = left_perm[b_idx, h_idx]
            b_slice = right_perm[b_idx, h_idx]

    var_out = ((out - mean_out) ** 2).mean(dim=-1, keepdim=True)

@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_i, stride_a_k,
    stride_b_i, stride_b_k,
    stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute base indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds i/j
    i_mask = i < N
    j_mask = j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles

def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_i, stride_a_k,
    stride_b_i, stride_b_k,
    stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_j = tl.program_id(1)

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        b = tl.load(b_ptr + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i,

"""
TriMul forward (outgoing) implementation with a custom Triton kernel.
The kernel computes the core O(N³) operation:
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
which is equivalent to a batched matrix multiplication
    left[b, d] (N×N) @ right[b, d].T (N×N)  →  out[b, d] (N×N).

All other operations (LayerNorms, linear projections, sigmoid gates) are done in
PyTorch for simplicity. The heavy reduction over the sequence dimension is
performed by a fast tiled matmul kernel written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out = left @ right.T   (both square matrices N×N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    a_ptr,                     # left matrix pointer   (float16)
    b_ptr,                     # right matrix pointer  (float16)
    c_ptr,                     # output matrix pointer (float32)
    N,                         # sequence length
    stride_a_i, stride_a_k,    # strides for left matrix (row i, col k)
    stride_b_i, stride_b_k,    # strides for right matrix (row j, col k)
    stride_c_i, stride_c_j,    # strides for output matrix (row i, col j)
    BLOCK_M: tl.constexpr,     # tile size in i dimension
    BLOCK_N: tl.constexpr,     # tile size in j dimension
    BLOCK_K: tl.constexpr,     # tile size in reduction (k) dimension
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Block start indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K) as float32 (original data stored as fp16)
        a = tl.load(
            a_ptr + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float32,
        )
        # Load B tile transposed (K x N) -> we need B[j,k] but provide as (k,j)
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float32,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Write back the result
    tl.store(
        c_ptr + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )
# ----------------------------------------------------------------------


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor, shape [B, N, N] (float 0/1)
            - weights: dict of weight tensors (see module definition)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on input ----------
    # mean / var over the last dimension (dim)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    # ---------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for Triton kernel ----------
    # Cast to fp16 for memory bandwidth, kernel loads as fp32
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Move hidden dimension to the second axis -> [B, H, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Output placeholder (float32)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Tiling parameters (tuned for A200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Loop over (batch, hidden) groups and launch a small matmul kernel per group
    for b_idx in range(B):
        for h_idx in range(hidden_dim):
            a_slice = left[b_idx, h_idx]   # (N, N), strides (N, 1)
            b_slice = right[b_idx, h_idx]
            c_slice = out[b_idx, h_idx]

            # Grid dimensions
            grid = (
                triton.cdiv(N, BLOCK_M),  # i dimension
                triton.cdiv(N, BLOCK_N),  # j dimension
            )

            _trimul_matmul_kernel[
                grid
            ](
                a_slice,
                b_slice,
                c_slice,
                N,
                a_slice.stride(0),
                a_slice.stride(1),
                b_slice.stride(0),
                b_slice.stride(1),
                c_slice.stride(0),
                c_slice.stride(1),
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
            )

    # ---------- Restore original layout ----------
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # ---------- Final LayerNorm ----------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ---------- Apply output gate ----------
    out = out_norm * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
