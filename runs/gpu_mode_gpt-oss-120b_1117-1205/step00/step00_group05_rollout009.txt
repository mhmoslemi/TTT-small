@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, right_ptr, mask_ptr, left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    n_elements, # total elements
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # compute
    left_out = left * mask_val * left_gate
    right_out = right * mask_val * right_gate
    # store
    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # weights: all as torch tensors, presumably float32
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # layernorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )   # float32
    # compute gates (float32) before cast
    left_gate_f32 = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate_f32 = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate_f32 = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    # cast gates to bf16
    left_gate = left_gate_f32.to(torch.bfloat16)
    right_gate = right_gate_f32.to(torch.bfloat16)
    out_gate = out_gate_f32.to(torch.bfloat16)
    # cast x_norm to bf16 for projections
    x_norm_bf16 = x_norm.to(torch.bfloat16)
    # projections (bf16)
    left_proj = torch.nn.functional.linear(x_norm_bf16, left_proj_weight.to(torch.bfloat16))
    right_proj = torch.nn.functional.linear(x_norm_bf16, right_proj_weight.to(torch.bfloat16))
    # mask to bf16
    mask_bf16 = mask.unsqueeze(-1).to(torch.bfloat16)
    # allocate fused tensors
    left_fused = torch.empty_like(left_proj, dtype=torch.bfloat16, device=device)
    right_fused = torch.empty_like(right_proj, dtype=torch.bfloat16, device=device)
    # launch triton kernel
    total_elements = left_proj.numel()
    BLOCK = 1024  # tune maybe 16384
    grid = (triton.cdiv(total_elements, BLOCK),)
    fuse_gate_mask_kernel[grid](
        left_proj, right_proj, mask_bf16, left_gate, right_gate,
        left_fused, right_fused,
        total_elements,
        BLOCK=BLOCK
    )
    # compute out using einsum
    # left_fused and right_fused are bf16
    out = torch.einsum('b i k d, b j k d -> b i j d', left_fused, right_fused)
    # apply to_out_norm (layernorm) on hidden_dim (out shape [B,N,N,hidden]), using bf16? Use float32 for stability.

"""
TriMul custom forward implementation.

The algorithm follows the reference PyTorch TriMul module:
1. Layer‑norm the input tensor.
2. Compute left/right projections (linear) and three gating vectors (linear + sigmoid) in FP32.
3. Fuse element‑wise mask and gating with a Triton kernel (still in FP32).
4. Cast the fused tensors to BFloat16 and perform the core
   4‑dimensional einsum:
       out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   (equivalent to a batched matrix multiplication per hidden channel).
5. Cast back to FP32, apply a second LayerNorm, multiply by the output gate
   and a final linear projection to the original channel dimension.

Only the element‑wise fusion step is executed in Triton; the heavyweight
einsum uses Torch’s highly‑optimized BF16 kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fuse_mul_fp32_kernel(
    left_ptr,          # FP32 left projection
    right_ptr,         # FP32 right projection
    mask_ptr,          # FP32 mask (already broadcasted to hidden_dim)
    left_gate_ptr,     # FP32 left gate
    right_gate_ptr,    # FP32 right gate
    left_out_ptr,      # FP32 fused left output
    right_out_ptr,     # FP32 fused right output
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    Element‑wise: out_left = left * mask * left_gate
                 out_right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements

    zero = tl.float32(0.0)

    left = tl.load(left_ptr + offs, mask=mask, other=zero)
    right = tl.load(right_ptr + offs, mask=mask, other=zero)
    m = tl.load(mask_ptr + offs, mask=mask, other=zero)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=zero)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=zero)

    left_out = left * m * left_gate
    right_out = right * m * right_gate

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    TriMul forward pass with a small Triton kernel for the mask/gate fusion.
    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------------
    # 1. LayerNorm on the input (FP32)
    # ----------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # ----------------------------------------------------------------------
    # 2. Linear projections and gates (FP32)
    # ----------------------------------------------------------------------
    left_proj = F.linear(
        x_norm, weights["left_proj.weight"]
    )  # [B, N, N, hidden_dim]
    right_proj = F.linear(
        x_norm, weights["right_proj.weight"]
    )  # [B, N, N, hidden_dim]

    left_gate = F.linear(
        x_norm, weights["left_gate.weight"]
    ).sigmoid()  # [B, N, N, hidden_dim]
    right_gate = F.linear(
        x_norm, weights["right_gate.weight"]
    ).sigmoid()  # [B, N, N, hidden_dim]
    out_gate = F.linear(
        x_norm, weights["out_gate.weight"]
    ).sigmoid()  # [B, N, N, hidden_dim]

    # ----------------------------------------------------------------------
    # 3. Fuse mask and gates with a Triton kernel (still FP32)
    # ----------------------------------------------------------------------
    # mask: [B, N, N] -> broadcast to hidden_dim
    mask_exp = mask.unsqueeze(-1).float().expand(-1, -1, -1, hidden_dim).contiguous()

    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    total_elements = left_proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    fuse_mul_fp32_kernel[grid](
        left_proj,
        right_proj,
        mask_exp,
        left_gate,
        right_gate,
        left_fused,
        right_fused,
        total_elements,
        BLOCK=BLOCK,
    )

    # Release intermediate tensors we no longer need
    del left_proj, right_proj, left_gate, right_gate, mask_exp

    # ----------------------------------------------------------------------
    # 4. Core TriMul reduction using BF16 einsum
    # ----------------------------------------------------------------------
    left_bf16 = left_fused.to(torch.bfloat16)
    right_bf16 = right_fused.to(torch.bfloat16)

    # einsum over the k dimension:  out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    out = torch.einsum("b i k d, b j k d -> b i j d", left_bf16, right_bf16)

    # Back to FP32 for the remaining LayerNorm and linear layers
    out = out.to(torch.float32)

    # ----------------------------------------------------------------------
    # 5. Post‑processing (LayerNorm, output gate, final linear)
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )  # [B, N, N, hidden_dim] FP32

    out = out * out_gate  # broadcast over last dim

    final = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim] FP32

    return final
