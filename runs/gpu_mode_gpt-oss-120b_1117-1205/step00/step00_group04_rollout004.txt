def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']; hidden_dim = config['hidden_dim']
    # load weight parameters as contiguous tensors (float32)
    norm_weight = weights['norm.weight'].to(device)
    norm_bias = weights['norm.bias'].to(device)
    left_proj_weight = weights['left_proj.weight'].to(device)
    right_proj_weight = weights['right_proj.weight'].to(device)
    left_gate_weight = weights['left_gate.weight'].to(device)
    right_gate_weight = weights['right_gate.weight'].to(device)
    out_gate_weight = weights['out_gate.weight'].to(device)
    to_out_norm_weight = weights['to_out_norm.weight'].to(device)
    to_out_norm_bias = weights['to_out_norm.bias'].to(device)
    to_out_weight = weights['to_out.weight'].to(device)

    # Input normalization
    # Note: torch.nn.functional.layer_norm expects normalized_shape (dim)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Linear projections
    # torch.nn.functional.linear expects input shape (..., in_features) and weight shape (out_features, in_features)
    # Our input shape is [B,N,N,dim]; to apply linear across last dimension:
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [..., hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Mask (may be None)
    if mask is None:
        mask = torch.ones_like(input_tensor[...,0], dtype=x.dtype, device=device)  # shape [B,N,N]
    # Convert mask to float
    mask = mask.to(x.dtype)   # ensure same dtype

    # Apply mask and gates using Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_contig = mask.contiguous()

    total_len = left.numel()
    BLOCK = 32768   # tune; but ensure total_len may exceed block size.

    grid = (triton.cdiv(total_len, BLOCK),)

    mask_gate_kernel[grid](left, right,
                           mask_contig,
                           left_gate, right_gate,
                           total_len,
                           hidden_dim=hidden_dim,
                           BLOCK=BLOCK)

    # Compute out = left @ right^T per batch and per hidden_dim using batched GEMM
    B = input_tensor.shape[0]
    N = input_tensor.shape[1]
    # Reshape left and right to [B*hidden_dim, N, N]
    left_reshape = left.view(B, N, N, hidden_dim).permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    # Actually left is shape [B,N,N,hidden_dim]; we need [B*hidden_dim, N, N] where every hidden dim is separate.
    # So left.view(B, N, N, hidden_dim) -> permute (0,3,1,2) -> shape [B, hidden_dim, N, N] -> reshape to [B*hidden_dim, N, N].
    left_mat = left.view(B, N, N, hidden_dim).permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_mat = right.view(B, N, N, hidden_dim).permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    # Compute out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    # Reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1)   # shape [B,N,N,hidden_dim]

    # Apply out_gate multiplication via Triton kernel
    out = out.contiguous()

    # Final linear to get dim output
    output = torch.nn.functional.linear(out, to_out_weight)   # shape [B,N,N,dim]

"""
TriMul forward pass (outgoing version) with Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑norm the [B,N,N,D] input.
2. Linear projections from D → H (hidden_dim) for left/right tensors.
3. Compute three gating tensors (left_gate, right_gate, out_gate) with a sigmoid.
4. Fuse mask, left_gate and right_gate in a Triton kernel:
       left = left * mask * left_gate
       right = right * mask * right_gate
5. Batched GEMM per hidden channel:
       out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as torch.bmm on reshaped tensors.
6. Layer‑norm over the hidden dimension.
7. Fuse out_gate scaling in a Triton kernel:
       out = out * out_gate
8. Final linear projection H → D.
All tensors stay in float32; the Triton kernels operate on
contiguous 1‑D views for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernels
# -------------------------------------------------------------

@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    total_len,
    hidden_dim: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """Fuse mask and per‑channel gates into left/right tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_len

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask (shape without hidden_dim) -> compute mask index
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    left = left * m * left_gate
    right = right * m * right_gate

    # Store results back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


@triton.jit
def apply_out_gate_kernel(
    out_ptr, out_gate_ptr,
    total_len,
    BLOCK: tl.constexpr,
):
    """Elementwise multiplication of out with out_gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_len

    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(out_gate_ptr + offs, mask=mask)

    out_val = out_val * gate_val

    tl.store(out_ptr + offs, out_val, mask=mask)


# -------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor, shape [B, N, N, D]
            - mask_tensor: torch.Tensor, shape [B, N, N] (bool or float)
            - weights_dict: mapping of weight names to tensors
            - config_dict: contains "dim" (D) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # -----------------------------------------------------------------
    # Configuration & weight extraction
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm on input
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)

    # Projections
    left_proj_weight = weights["left_proj.weight"].to(device, dtype)      # (H, D)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype)    # (H, D)

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(device, dtype)     # (H, D)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype)   # (H, D)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype)       # (H, D)

    # Output normalization & projection
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype) # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype)     # (H,)
    to_out_weight = weights["to_out.weight"].to(device, dtype)           # (D, H)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(dtype)  # ensure float32

    # -----------------------------------------------------------------
    # 2. Linear projections (D -> H)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,H]
    right = F.linear(x, right_proj_weight) # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3. Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # 4. Apply mask + gates via Triton
    # -----------------------------------------------------------------
    if mask is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    mask = mask.to(dtype).contiguous()

    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_len = left.numel()  # B * N * N * H
    BLOCK = 8192  # reasonable block size for elementwise kernel
    grid = (triton.cdiv(total_len, BLOCK),)

    mask_gate_kernel[grid](
        left,
        right,
        mask,
        left_gate,
        right_gate,
        total_len,
        hidden_dim=hidden_dim,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 5. Batched GEMM over the K dimension (k‑contraction)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to [B*H, N, N] for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape [B,N,N,H]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )

    # -----------------------------------------------------------------
    # 7. Apply out_gate (Triton)
    # -----------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    total_len_out = out.numel()
    grid_out = (triton.cdiv(total_len_out, BLOCK),)

    apply_out_gate_kernel[grid_out](
        out,
        out_gate,
        total_len_out,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 8. Final linear projection H -> D
    # -----------------------------------------------------------------
    output = F.linear(out, to_out_weight)  # shape [B, N, N, D]
    return output
