    input_tensor, mask, weights, config = data

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; B = input_tensor.shape[0]; N = input_tensor.shape[1]; D = input_tensor.shape[3] (should be dim). Probably D = dim.

    # LayerNorm: using weights
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5) # shape B,N,N,dim

    # Linear projections
    left_proj = F.linear(x_norm, weights["left_proj.weight"]) # shape B,N,N,hidden_dim
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # Mask
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(left_proj.dtype)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp
    else:
        # if mask not to be used, still assume mask maybe not needed
        pass

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output tensor
    out = torch.empty_like(left)  # shape B,N,N,hidden_dim

    # compute contraction with Triton kernel
    # Define block sizes:
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32
    # Choose BLOCK_H based on hidden_dim
    BLOCK_H = 32 if hidden_dim >= 32 else hidden_dim

    # Prepare strides
    stride_bl, stride_bi, stride_bk, stride_bh = left.stride()
    stride_br, stride_bj, stride_brk, stride_brh = right.stride()
    stride_ol, stride_oi, stride_oj, stride_oh = out.stride()

    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J), B)
    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_bl, stride_bi, stride_bk, stride_bh,
        stride_br, stride_bj, stride_brk, stride_brh,
        stride_ol, stride_oi, stride_oj, stride_oh,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H
    )

    # After kernel, out shape B,N,N,H.

    # to_out_norm layernorm over hidden dim
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # Apply out_gate
    out = out * out_gate

    # Final linear transformation to dim
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) implementation using Triton for the O(N³·H) tensor contraction.
The kernel computes, for each batch, i, j and a tile of the hidden dimension H:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

The heavy contraction is performed in a fused Triton kernel with tiling over
(i, j, k) and the hidden dimension. LayerNorms, linear projections and gates are
done with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,            # *const float*
    right_ptr,           # *const float*
    out_ptr,             # *float*
    B, N, H,             # runtime sizes
    offset_h,            # int: starting index in hidden dimension for this tile
    stride_bl, stride_bi, stride_bk, stride_bh,   # left strides
    stride_br, stride_bj, stride_brk, stride_brh, # right strides
    stride_ol, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute the outgoing TriMul contraction."""
    pid_b = tl.program_id(2)          # batch dimension
    pid_i = tl.program_id(0)          # i‑block
    pid_j = tl.program_id(1)          # j‑block

    # Offsets for i, j and hidden dimension within the tile
    i_offsets = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offsets = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h_offsets = tl.arange(0, BLOCK_H)

    i_mask = i_offsets < N
    j_mask = j_offsets < N
    h_mask = h_offsets < H

    # Accumulator for this (i, j) tile and hidden‑tile
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Base pointers offset by the hidden‑tile start
    left_base  = left_ptr  + offset_h * stride_bh
    right_base = right_ptr + offset_h * stride_brh
    out_base   = out_ptr   + offset_h * stride_oh

    # Loop over k in tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        k_offsets = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # Load left tile: (I, K_tile, H)
        left_ptrs = (
            left_base
            + i_offsets[:, None, None] * stride_bi
            + k_offsets[None, :, None] * stride_bk
            + h_offsets[None, None, :] * stride_bh
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )

        # Load right tile: (J, K_tile, H)
        right_ptrs = (
            right_base
            + j_offsets[:, None, None] * stride_bj
            + k_offsets[None, :, None] * stride_brk
            + h_offsets[None, None, :] * stride_brh
        )
        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )

        # Compute partial contraction over this k‑tile:
        # left: (I, K, H) -> (I, 1, K, H)
        # right: (J, K, H) -> (1, J, K, H)
        # product -> (I, J, K, H); sum over K -> (I, J, H)
        partial = tl.sum(left[:, None, :, :] * right[None, :, :, :], axis=2)
        acc += partial

    # Store result (apply masks for boundary handling)
    out_ptrs = (
        out_base
        + i_offsets[:, None, None] * stride_oi
        + j_offsets[None, :, None] * stride_oj
        + h_offsets[None, None, :] * stride_oh
    )
    store_mask = i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using Triton for the core contraction.
    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, D] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float)
            - weights: dict of torch.Tensors with keys:
                norm.weight, norm.bias,
                left_proj.weight, right_proj.weight,
                left_gate.weight, right_gate.weight, out_gate.weight,
                to_out_norm.weight, to_out_norm.bias,
                to_out.weight
            - config: dict with keys:
                dim (int), hidden_dim (int), nomask (bool, optional)
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (per‑pair vector)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask application
    # -----------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # broadcast to H
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core O(N³·H) contraction via Triton
    # -----------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # shape [B, N, N, H]

    # Tile size choices (tuned for typical B200 memory/compute)
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32
    BLOCK_H = 32                       # max hidden‑tile size

    # Strides for the three tensors (in elements, not bytes)
    stride_bl, stride_bi, stride_bk, stride_bh = left.stride()
    stride_br, stride_bj, stride_brk, stride_brh = right.stride()
    stride_ol, stride_oi, stride_oj, stride_oh = out.stride()

    # Number of hidden tiles
    num_h_tiles = triton.cdiv(hidden_dim, BLOCK_H)

    for t in range(num_h_tiles):
        h_start = t * BLOCK_H
        tile_h = min(BLOCK_H, hidden_dim - h_start)

        # Launch kernel for this hidden tile
        grid = (
            triton.cdiv(N, BLOCK_I),   # i dimension
            triton.cdiv(N, BLOCK_J),   # j dimension
            B,                         # batch
        )
        trimul_outgoing_kernel[grid](
            left,
            right,
            out,
            B,
            N,
            tile_h,
            h_start,
            stride_bl, stride_bi, stride_bk, stride_bh,
            stride_br, stride_bj, stride_brk, stride_brh,
            stride_ol, stride_oi, stride_oj, stride_oh,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear map
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
