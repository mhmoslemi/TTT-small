    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

- Program IDs: pid_i = tl.program_id(0), pid_j = tl.program_id(1), pid_bd = tl.program_id(2)

- Compute b = pid_bd // D, d = pid_bd % D.

- Compute i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
- Compute j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

- mask_j = j_offsets < N

@triton.jit
def trimul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    # Compute batch index and channel (hidden dim) index
    b = pid_bd // D
    d = pid_bd % D

    # Offsets for i and j within tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i_offsets = i_base + tl.arange(0, BLOCK_M)
    j_offsets = j_base + tl.arange(0, BLOCK_N)

    # Masks for boundaries
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # Compute base offsets for pointers
    # b offset
    offset_b_left = b * stride_left_batch
    offset_b_right = b * stride_right_batch
    offset_b_out = b * stride_out_batch
    # d offset
    offset_d_left = d * stride_left_d
    offset_d_right = d * stride_right_d
    offset_d_out = d * stride_out_d

"""
TriMul forward kernel (outgoing version) optimized with Triton.

Algorithm
---------
1. Layer‑norm the input Tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias) produce left and right tensors of shape
   (B,N,N,H) where H = hidden_dim.
3. Apply optional mask, then element‑wise gates (sigmoid of linear layers).
4. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication left @ rightᵀ for each hidden
   channel. Implemented as a Triton kernel using tiling on the (i,j) plane
   and a k‑loop.
5. Layer‑norm over hidden dimension, multiply by the output gate,
   and a final linear projection back to C dimensions.

The heavy O(N³) contraction is performed entirely on the GPU in the
Triton kernel; all other operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel for the contracted sum over sequence dimension.
# ------------------------------------------------------------
@triton.jit
def trimul_contract_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    B, N, H,           # int scalars: batch, seq_len, hidden_dim
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)            # combined batch * hidden index

    # Decode batch and hidden channel from pid_bd
    b = pid_bd // H
    h = pid_bd % H

    # Base offsets for the current batch / channel
    offset_b_l = b * stride_lb
    offset_b_r = b * stride_rb
    offset_b_o = b * stride_ob
    offset_h_l = h * stride_ld
    offset_h_r = h * stride_rd
    offset_h_o = h * stride_od

    # Tile coordinates
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    i_offsets = i0 + tl.arange(0, BLOCK_M)
    j_offsets = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # Accumulator for the (i,j) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k_idx in range(num_k):
        k0 = k_idx * BLOCK_K
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + offset_b_l + offset_h_l \
                    + i_offsets[:, None] * stride_li \
                    + k_offsets[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[b, j, k, h] (access as (k, j)) -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + offset_b_r + offset_h_r \
                     + k_offsets[:, None] * stride_rk \
                     + j_offsets[None, :] * stride_rj
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Matrix multiplication over the K dimension
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # Store the result tile
    out_ptrs = out_ptr + offset_b_o + offset_h_o \
               + i_offsets[:, None] * stride_oi \
               + j_offsets[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ------------------------------------------------------------
# Host entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator using a Triton kernel for
    the O(N³) contraction.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask_tensor : torch.Tensor of shape [B, N, N] (float or bool)
            - weights_dict: contains all model weight tensors
            - config_dict : must contain keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load weights and move to the correct device / dtype
    # ------------------------------------------------------------------
    norm_w   = weights["norm.weight"].to(device)
    norm_b   = weights["norm.bias"].to(device)
    left_proj_w  = weights["left_proj.weight"].to(device)
    right_proj_w = weights["right_proj.weight"].to(device)
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    to_out_w      = weights["to_out.weight"].to(device)

    # ------------------------------------------------------------------
    # Normalization of the input (LayerNorm over channel dimension C)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(C,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections and gating (performed in float32)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)      # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # mask handling (mask may be all‑ones when nomask=True)
    mask_f = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # gates
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Block sizes – chosen to fit registers on an A200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides (in number of elements)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),      # dim 0: i‑tiles
        triton.cdiv(N, BLOCK_N),      # dim 1: j‑tiles
        B * hidden_dim                # dim 2: batch × hidden channel
    )

    trimul_contract_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim layer‑norm, output gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_w)    # final projection back to C dimensions

    return out
