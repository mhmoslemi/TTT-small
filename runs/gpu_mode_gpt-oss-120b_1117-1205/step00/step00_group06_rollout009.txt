    right_proj_weight = weights["right_proj.weight"]

def trigemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and head
    b = batch_id // H
    h = batch_id % H

    # Offsets for i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for validity
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base pointer offsets for this batch/head
    left_batch_offset = b * stride_left_b + h * stride_left_h
    right_batch_offset = b * stride_right_b + h * stride_right_h
    out_batch_offset = b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for left matrix A (M,K)

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The forward pass consists of:
1. Layer‑norm the input pair representation.
2. Linear projections for the left/right streams and their gates.
3. Optional mask application.
4. Element‑wise gating of the left/right streams.
5. Batched matrix multiplication across the sequence dimension:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   (equivalent to a per‑head GEMM L @ Rᵀ).
   This O(N³) part is executed by a custom Triton kernel that tiles the
   i‑j output plane and iterates over k in blocks.
6. Layer‑norm + gating of the intermediate result.
7. Final linear projection back to the original channel dimension.

All heavy arithmetic (step 5) runs in FP16 for bandwidth, with FP32
accumulation for numeric stability. The final output is returned in
float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # left strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # right strides
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # out strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for each (batch, head) pair.
    left shape : (B, N, N, H)   -> accessed as (b, i, k, h)
    right shape: (B, N, N, H)   -> accessed as (b, j, k, h)
    out shape  : (B, N, N, H)   -> (b, i, j, h)
    """
    batch_id = tl.program_id(0)   # combined batch * head index
    pid_i    = tl.program_id(1)   # tile row (i)
    pid_j    = tl.program_id(2)   # tile column (j)

    # decode batch and head
    b = batch_id // H
    h = batch_id % H

    # offsets for i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,h)
    left_batch  = b * stride_left_b + h * stride_left_h
    right_batch = b * stride_right_b + h * stride_right_h
    out_batch   = b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (i,k) -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + left_batch + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # right block (transposed): we need right[k,j] -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + right_batch + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # write result as FP16 (final conversion)
    out_fp16 = acc.to(tl.float16)
    c_ptrs = out_ptr + out_batch + \
             offs_i[:, None] * stride_out_i + \
             offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, out_fp16,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul outgoing forward pass using a fused Triton kernel for the
    heavy tensor‑product operation.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on input (shape B,N,N,dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (left/right) and gating linear layers
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # B,N,N,hidden
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: (B,N,N) -> (B,N,N,1)
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM via Triton (left @ rightᵀ) -> (B,N,N,hidden)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # strides (in element units)
    s_lb, s_li, s_lk, s_lh = left_fp16.stride()
    s_rb, s_rj, s_rk, s_rh = right_fp16.stride()
    s_ob, s_oi, s_oj, s_oh = out_fp16.stride()

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        input_tensor.shape[0] * hidden,          # batch * head
        triton.cdiv(input_tensor.shape[1], BLOCK_M),  # i tiles
        triton.cdiv(input_tensor.shape[1], BLOCK_N),  # j tiles
    )

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        input_tensor.shape[0],               # B
        input_tensor.shape[1],               # N
        hidden,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 for the remaining host‑side ops
    out = out_fp16.to(dtype)

    # ------------------------------------------------------------------
    # 6. Output layer‑norm and gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
