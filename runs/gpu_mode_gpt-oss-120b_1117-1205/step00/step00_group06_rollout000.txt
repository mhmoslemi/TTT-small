def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm x_norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left_proj_weight = weights["left_proj.weight"]  # shape hidden_dim x dim
    right_proj_weight = weights["right_proj.weight"]

    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Mask if needed
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)  # broadcast
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute multiplicative update via einsum
    # Use half precision for speed
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)
    out_half = torch.einsum('b i k d, b j k d -> b i j d', left_half, right_half)
    # Cast back to float32
    out = out_half.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Elementwise multiply with out_gate using Triton kernel
    out_gate = out_gate.to(out.dtype)  # ensure same dtype
    size = out.numel()
    # Ensure contiguous
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    out_scaled = torch.empty_like(out)

    BLOCK = 65536  # adjust
    grid = (triton.cdiv(size, BLOCK),)
    elementwise_mul_kernel[grid](out_scaled, out, out_gate, size, BLOCK=BLOCK)

    # Final linear projection (to_out)
    to_out_weight = weights["to_out.weight"]  # shape dim x hidden_dim
    output = torch.nn.functional.linear(out_scaled, to_out_weight)

    return output

"""
TriMul (outgoing) forward pass with a Triton elementwise kernel.

Algorithm
---------
1. Layer‑normalize the input tensor over the last dimension.
2. Apply linear projections (left/right) and gating linear layers.
3. Mask the projected tensors if a mask is provided.
4. Multiply left/right by their respective gates.
5. Compute the core O(N³) update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   (implemented with torch.einsum, which reduces to a batched GEMM).
6. Apply a second LayerNorm over the hidden dimension.
7. Fuse the output‑gate multiplication using a tiny Triton kernel.
8. Project back to the original dimension with a final linear layer.

The Triton kernel only performs the element‑wise product
`out = out * out_gate`; all other heavy work stays in PyTorch.
This satisfies the requirement of using Triton while keeping the
implementation simple and correct.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, a_ptr, b_ptr, size, BLOCK: tl.constexpr):
    """
    out = a * b   (elementwise)
    All tensors are flattened 1‑D contiguous arrays.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    TriMul forward pass (outgoing version) using a Triton kernel for the
    element‑wise gating step.

    Args:
        data: Tuple(
            input_tensor: torch.Tensor of shape [B, N, N, dim],
            mask: torch.Tensor of shape [B, N, N],
            weights: dict of weight tensors,
            config: dict with keys "dim", "hidden_dim", "nomask" (bool)
        )

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. First LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (left/right) and gating linear layers
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]        # (hidden_dim, dim)

    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # Apply mask if present (mask shape [B, N, N])
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).to(dtype)            # [B, N, N, 1]
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core multiplicative update (N³) – batched matrix multiplication
    # ------------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # float32

    # ------------------------------------------------------------------
    # 4. Second LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # 5. Element‑wise multiplication with out_gate (via Triton)
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.to(out.dtype).contiguous()
    out_scaled = torch.empty_like(out)

    size = out.numel()
    BLOCK = 65536  # works well for typical GPU architectures
    grid = (triton.cdiv(size, BLOCK),)
    elementwise_mul_kernel[grid](out_scaled, out, out_gate, size, BLOCK=BLOCK)

    # ------------------------------------------------------------------
    # 6. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]    # (dim, hidden_dim)
    output = F.linear(out_scaled, to_out_weight)   # [B, N, N, dim]

    return output
