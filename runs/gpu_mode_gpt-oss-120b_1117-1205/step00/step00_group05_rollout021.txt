    stride_right_j, stride_right_k, stride_right_h,

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,   # *const half*
    right_ptr,  # *const half*
    out_ptr,    # *float*
    N, H,
    stride_i, stride_k, stride_h,
    stride_i_r, stride_k_r, stride_h_r,
    out_stride_i, out_stride_j, out_stride_h,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    i = tl.program_id(0)
    j = tl.program_id(1)

    # offsets for hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # accumulator (float32)
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # loop over k in blocks
    for cur in range(0, N, BLOCK_K):
        cur_k = cur + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # compute pointer offsets
        left_ptrs = left_ptr + i * stride_i + cur_k[:, None] * stride_k + offs_h[None, :] * stride_h
        right_ptrs = right_ptr + j * stride_i_r + cur_k[:, None] * stride_k_r + offs_h[None, :] * stride_h_r

        # combined mask
        mask = mask_k[:, None] & mask_h[None, :]

        left_vals = tl.load(left_ptrs, mask=mask, other=0.0).to(tl.float32)
        right_vals = tl.load(right_ptrs, mask=mask, other=0.0).to(tl.float32)

        acc += tl.sum(left_vals * right_vals, axis=0)

    # store result (float32)
    out_ptrs = out_ptr + i * out_stride_i + j * out_stride_j + offs_h * out_stride_h
    tl.store(out_ptrs, acc, mask=mask_h)
```

def gate_mask_kernel(

"""
TriMul (outgoing) implementation with a Triton kernel for the gating+mask
operation. The heavy O(N³) reduction is expressed as a batched GEMM which
leverages cuBLAS; the element‑wise multiplication with gates and mask is
performed by a small Triton kernel. All operations return float32 tensors
compatible with the reference module.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: apply per‑position gate and mask to a projected tensor.
# The kernel processes a single (i, k) coordinate (for left) or (j, k)
# coordinate (for right) and a block of the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,           # *float16*  projected values (shape [N, N, H])
    gate_ptr,           # *float16*  gate values (same shape)
    mask_ptr,           # *float16*  mask values (shape [N, N])
    out_ptr,            # *float16*  output tensor (same shape as proj)
    N, H,               # problem size
    proj_stride_i, proj_stride_k, proj_stride_h,
    gate_stride_i, gate_stride_k, gate_stride_h,
    mask_stride_i, mask_stride_k,
    out_stride_i, out_stride_k, out_stride_h,
    BLOCK_H: tl.constexpr,
):
    i = tl.program_id(0)          # first spatial index (i or j)
    k = tl.program_id(1)          # second spatial index (k)

    # bounds (programs outside N are no‑ops)
    if i >= N or k >= N:
        return

    # hidden dimension offsets
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # pointer arithmetic (all strides are in elements)
    proj_off = i * proj_stride_i + k * proj_stride_k
    gate_off = i * gate_stride_i + k * gate_stride_k
    mask_off = i * mask_stride_i + k * mask_stride_k
    out_off  = i * out_stride_i  + k * out_stride_k

    # load scalar mask and broadcast it over the hidden dimension
    mask_val = tl.load(mask_ptr + mask_off, mask=1, other=0.0).to(tl.float32)

    # load vectors (float16 → float32 for arithmetic)
    proj_vec = tl.load(
        proj_ptr + proj_off + offs_h * proj_stride_h,
        mask=mask_h,
        other=0.0,
    ).to(tl.float32)
    gate_vec = tl.load(
        gate_ptr + gate_off + offs_h * gate_stride_h,
        mask=mask_h,
        other=0.0,
    ).to(tl.float32)

    out_vec = proj_vec * gate_vec * mask_val

    # store back as float16
    tl.store(
        out_ptr + out_off + offs_h * out_stride_h,
        out_vec.to(tl.float16),
        mask=mask_h,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool or 0/1)
            - weights: dict with all layer weights (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=dim,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # work in half precision for the projection/gate phases
    x16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(
        x16, weights["left_proj.weight"].to(torch.float16), bias=None
    )
    right_proj = torch.nn.functional.linear(
        x16, weights["right_proj.weight"].to(torch.float16), bias=None
    )  # both [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x16, weights["left_gate.weight"].to(torch.float16), bias=None
        )
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x16, weights["right_gate.weight"].to(torch.float16), bias=None
        )
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x16, weights["out_gate.weight"].to(torch.float16), bias=None
        )
    )  # all [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Apply mask and gates via Triton kernel
    # ------------------------------------------------------------------
    # Ensure mask is float16 (0/1) and has a trailing hidden dim for broadcasting
    mask_fp16 = mask.to(torch.float16)

    # Allocate gated tensors
    left_gated = torch.empty_like(left_proj)   # float16
    right_gated = torch.empty_like(right_proj)  # float16

    # Kernel launch parameters
    BLOCK_H = 128  # hidden_dim ≤ 128 in all test configs
    grid = (N, N)  # one program per (i, k) pair

    for b in range(B):
        # ----- left side -----
        lp = left_proj[b]      # [N, N, H]
        lg = left_gate[b]
        mk = mask_fp16[b]      # [N, N]
        lo = left_gated[b]

        # strides (in elements) after selecting the batch dimension
        s_i, s_k, s_h = lp.stride()
        g_i, g_k, g_h = lg.stride()
        m_i, m_k = mk.stride()
        o_i, o_k, o_h = lo.stride()

        gate_mask_kernel[grid](
            lp.data_ptr(),
            lg.data_ptr(),
            mk.data_ptr(),
            lo.data_ptr(),
            N,
            hidden_dim,
            s_i,
            s_k,
            s_h,
            g_i,
            g_k,
            g_h,
            m_i,
            m_k,
            o_i,
            o_k,
            o_h,
            BLOCK_H=BLOCK_H,
        )

        # ----- right side -----
        rp = right_proj[b]
        rg = right_gate[b]
        ro = right_gated[b]

        s_i, s_k, s_h = rp.stride()
        g_i, g_k, g_h = rg.stride()
        o_i, o_k, o_h = ro.stride()

        gate_mask_kernel[grid](
            rp.data_ptr(),
            rg.data_ptr(),
            mk.data_ptr(),          # same mask for right side
            ro.data_ptr(),
            N,
            hidden_dim,
            s_i,
            s_k,
            s_h,
            g_i,
            g_k,
            g_h,
            m_i,
            m_k,
            o_i,
            o_k,
            o_h,
            BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 5. Batched matrix multiplication to perform the O(N³) reduction
    #    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and use torch.bmm with a transpose on the second operand
    L = left_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # (BH, N, N)
    R = right_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # (BH, N, N)

    out_fp16 = torch.bmm(L, R.transpose(1, 2))  # (BH, N, N)
    # reshape back to [B, N, N, hidden_dim] and promote to float32
    out = out_fp16.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=hidden_dim,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(out.dtype)

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
