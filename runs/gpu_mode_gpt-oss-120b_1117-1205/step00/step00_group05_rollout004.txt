import triton
import triton.language as tl
import torch
import math
import torch.nn.functional as F

# Triton kernel
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid = tl.program_id(0)   # batch * hidden
    pid_m = tl.program_id(1) # i tile
    pid_n = tl.program_id(2) # j tile

    total_bd = B * D
    b = pid // D
    d = pid % D

    # i and j offsets
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # Load left: shape (BLOCK_M, BLOCK_K), dtype fp16
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + i_off[:, None] * stride_left_i \
            + k_off[None, :] * stride_left_k \
            + d * stride_left_d
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right transposed: shape (BLOCK_K, BLOCK_N) (k, j)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + k_off[:, None] * stride_right_k \
            + j_off[None, :] * stride_right_j \
            + d * stride_right_d
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute block dot product and accumulate
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + i_off[:, None] * stride_out_i \
        + j_off[None, :] * stride_out_j \
        + d * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).
    Computes out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}
    where left/right are linear projections of the input with gating and mask.
    The core contraction is performed by a Triton kernel for high performance.
    """
    import torch
    import torch.nn.functional as F
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input (last dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections (Linear without bias)
    # shape after proj: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)   # (dim,out) weight shape (hidden_dim, dim)
    right = F.linear(x, right_proj_weight)

    # Apply mask if provided (mask shape [B,N,N])
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for faster memory bandwidth in kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output tensor
    out = torch.empty_like(left_fp16, dtype=torch.float32)  # [B,N,N,hidden_dim]

    # Strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_bd = B * hidden_dim
    grid = (total_bd, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # LayerNorm on hidden dimension
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

"""
Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

The forward pass consists of:
1. LayerNorm on the input.
2. Linear projections to a hidden dimension (left/right).
3. Optional mask application.
4. Gating via element‑wise sigmoid of additional linear projections.
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   performed by a tiled Triton kernel (outer‑product‑style GEMM over the
   sequence dimension).
6. LayerNorm on the hidden dimension, gating, and final linear projection
   back to the original feature dimension.

Only the O(N³) contraction (step 5) is off‑loaded to Triton; the remaining
operations are kept in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
    using a block‑wise tiled GEMM for each (b, d) pair.
    """
    pid = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)        # i tile index
    pid_n = tl.program_id(2)        # j tile index

    total_bd = B * D
    b = pid // D                    # batch index
    d = pid % D                     # hidden‑channel index

    # Offsets for the i and j dimensions
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in tiles
    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left[b, i, k, d]   -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + i_off[:, None] * stride_left_i \
            + k_off[None, :] * stride_left_k \
            + d * stride_left_d
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # right[b, j, k, d] (need transposed view k × j) -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + k_off[:, None] * stride_right_k \
            + j_off[None, :] * stride_right_j \
            + d * stride_right_d
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # Accumulate dot product over k
        acc += tl.dot(left, right)   # left: (M,K), right: (K,N) → (M,N)

    # Store the result
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + i_off[:, None] * stride_out_i \
        + j_off[None, :] * stride_out_j \
        + d * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1) LayerNorm over the feature dimension
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight,
                     bias=norm_bias, eps=1e-5)

    # 2) Linear projections (no bias)
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # 3) Apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # 4) Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # 5) Core TriMul contraction via Triton (use FP16 for bandwidth)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = torch.empty_like(left_fp16, dtype=torch.float32)  # [B,N,N,hidden_dim]

    # Strides in element units
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_d = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Tuning parameters (adjustable if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_bd = B * hidden_dim
    grid = (
        total_bd,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_d,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 6) Output LayerNorm over hidden dimension
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)

    # 7) Output gating and final linear projection to original dim
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
