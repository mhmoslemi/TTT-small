import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_lbh, stride_li, stride_lk,
    stride_rbh, stride_rj, stride_rk,
    stride_obh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1)  # col block index
    pid_h = tl.program_id(2)  # batch * hidden index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

"""
TriMul forward pass (outgoing version) with a Triton kernel.

The heavy O(N³) reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed as a batched matrix multiplication left @ rightᵀ.
We flatten the batch and hidden dimensions (B·hidden) and launch a
blocked GEMM kernel that computes each (i, j) tile for every slice.
All other operations (LayerNorms, linear projections, gating) stay in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # fp16, shape (BH, N, N)
    right_ptr,         # fp16, shape (BH, N, N)
    out_ptr,           # fp32, shape (BH, N, N)
    N: tl.constexpr,
    stride_lbh, stride_li, stride_lk,
    stride_rbh, stride_rj, stride_rk,
    stride_obh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_h = tl.program_id(2)   # batch * hidden index

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block:   A[i,k]   shape (M, K)
        a_ptrs = left_ptr + pid_h * stride_lbh \
                 + offs_m[:, None] * stride_li \
                 + offs_k[None, :] * stride_lk

                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

                 + offs_k[:, None] * stride_rk \
                 + offs_n[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # dot‑product over K
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr + pid_h * stride_obh \
               + offs_m[:, None] * stride_oi \
               + offs_n[None, :] * stride_oj
    tl.store(out_ptrs,
             acc,
             mask=(mask_m[:, None] & mask_n[None, :]))

        stride_rbh,
        stride_rj,
        stride_rk,
        stride_obh,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
