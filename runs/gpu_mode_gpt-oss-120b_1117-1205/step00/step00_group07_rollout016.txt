def tri_mul_contract_kernel(left_ptr, right_ptr, out_ptr,
                            B, N, H,
                            stride_bl, stride_il, stride_kl, stride_hl,
                            stride_br, stride_ir, stride_kr, stride_hr,
                            stride_bo, stride_io, stride_jo, stride_ho,
                            BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    hb = pid_hb
    b = hb // H
    h = hb % H

    # base pointers for this batch and head
    left_base = left_ptr + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        # Offsets for left (i,k)
        i = i0 + tl.arange(0, BLOCK_M)
        k_off = k0 + tl.arange(0, BLOCK_K)
        # Build offsets for left: stride_il * i + stride_kl * k_off
        left_offsets = i[:, None] * stride_il + k_off[None, :] * stride_kl
        # Load left tile
        left_tile = tl.load(left_base + left_offsets,
                            mask=(i[:, None] < N) & (k_off[None, :] < N), other=0.0)

        # Offsets for right: need (k,j) shape as (BLOCK_K, BLOCK_N)
        j = j0 + tl.arange(0, BLOCK_N)
        # right offsets: k_off[:, None] * stride_kr + j[None, :] * stride_ir
        right_offsets = k_off[:, None] * stride_kr + j[None, :] * stride_ir
        # Load right tile (BATCH) as shape (BLOCK_K, BLOCK_N)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # layernorm
    norm = torch.nn.LayerNorm(dim, eps=1e-5, elementwise_affine=True).to(device)
    norm.weight = torch.nn.Parameter(weights['norm.weight'].to(device))
    norm.bias = torch.nn.Parameter(weights['norm.bias'].to(device))
    x = norm(input_tensor)

    # linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'].to(device))
    right = torch.nn.functional.linear(x, weights['right_proj.weight'].to(device))

    # apply mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight'].to(device)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight'].to(device)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight'].to(device)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # allocate output
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Get strides
    stride_bl = left.stride(0)
    stride_il = left.stride(1)
    stride_kl = left.stride(2)
    stride_hl = left.stride(3)

    stride_br = right.stride(0)
    stride_ir = right.stride(1)
    stride_kr = right.stride(2)
    stride_hr = right.stride(3)

    stride_bo = out.stride(0)
    stride_io = out.stride(1)
    stride_jo = out.stride(2)
    stride_ho = out.stride(3)

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    tri_mul_contract_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Now out is result of contraction, shape [B,N,N,hidden_dim] float32
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate (shape [B,N,N, H])
    out_norm = out_norm * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # no bias

    return output

"""
Highly optimized TriMul "outgoing" implementation.

The expensive O(N³·H) contraction is performed with a custom Triton kernel.
All other operations (LayerNorms, Linear projections, gates, masking) are
kept in PyTorch for simplicity and to reuse existing fused kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)          # tile over rows (i)
    pid_j = tl.program_id(1)          # tile over cols (j)
    pid_hb = tl.program_id(2)         # combined batch * hidden dimension

    # start indices for this tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # decode batch and head from combined pid
    hb = pid_hb
    b = hb // H
    h = hb % H

    # base pointers for the selected batch and head
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_bo + h * stride_ho

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K

        # ---- load left tile (i, k)
        i = i0 + tl.arange(0, BLOCK_M)
        k_off = k0 + tl.arange(0, BLOCK_K)
        left_offset = i[:, None] * stride_il + k_off[None, :] * stride_kl
        left_mask = (i[:, None] < N) & (k_off[None, :] < N)
        left_tile = tl.load(left_base + left_offset,
                            mask=left_mask, other=0.0)

        # ---- load right tile (j, k) as (k, j) so we can dot with left_tile
        j = j0 + tl.arange(0, BLOCK_N)
        right_offset = k_off[:, None] * stride_kr + j[None, :] * stride_ir
        right_mask = (k_off[:, None] < N) & (j[None, :] < N)
        right_tile = tl.load(right_base + right_offset,
                             mask=right_mask, other=0.0)

        # ---- update accumulator
        acc += tl.dot(left_tile, right_tile)   # (M,K) @ (K,N) -> (M,N)

    # ---- write the result
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    out_offset = i[:, None] * stride_io + j[None, :] * stride_jo
    out_mask = (i[:, None] < N) & (j[None, :] < N)
    tl.store(out_base + out_offset, acc, mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]
            mask         : torch.Tensor [B, N, N] (bool/int/float)
            weights      : dict of model weight tensors
            config       : dict with keys "dim" and "hidden_dim"
    Returns:
        output: torch.Tensor [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on input -----------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----- Linear projections (no bias) --------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ----- Optional mask ------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)                # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gates --------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(device)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(device)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(device)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Contraction via Triton --------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides for left/right/out (contiguous layout)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # Tile sizes (tuned for V100/A100 style GPUs; works on B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,   # combined batch * head dimension
    )

    tri_mul_contract_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ----- Second LayerNorm + out gating --------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
        eps=1e-5,
    )
    out_norm = out_norm * out_gate

    # ----- Final projection back to dim ---------------------------------
    output = F.linear(out_norm, weights["to_out.weight"].to(device))  # no bias

    return output
