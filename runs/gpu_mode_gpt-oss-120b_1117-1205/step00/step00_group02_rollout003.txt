```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   stride_b, stride_i, stride_k, stride_h,
                   B, N, H,
                   BLOCK_K: tl.constexpr,
                   BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    total_ij = N * N
    b = pid // total_ij
    ij = pid - b * total_ij
    i = ij // N
    j = ij - i * N

    h = 0
    while h < H:
        offs_h = h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator for this hidden chunk
        acc = tl.zeros([BLOCK_H], dtype=tl.float32)

        k = 0
        while k < N:
            cur_k = k + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # offsets for left[b,i,k,:] and right[b,j,k,:]
            left_offset = b * stride_b + i * stride_i + cur_k[:, None] * stride_k + offs_h[None, :] * stride_h
            right_offset = b * stride_b + j * stride_i + cur_k[:, None] * stride_k + offs_h[None, :] * stride_h

            left_vals = tl.load(left_ptr + left_offset,
                                mask=mask_k[:, None] & mask_h[None, :],
                                other=0.0,
                                dtype=tl.float16).to(tl.float32)
            right_vals = tl.load(right_ptr + right_offset,
                                 mask=mask_k[:, None] & mask_h[None, :],
                                 other=0.0,
                                 dtype=tl.float16).to(tl.float32)

            # element‑wise multiply and reduce over k
            acc += tl.sum(left_vals * right_vals, axis=0)

            k += BLOCK_K

        # store the accumulated hidden chunk
        out_offset = b * stride_b + i * stride_i + j * stride_k + offs_h * stride_h
        tl.store(out_ptr + out_offset, acc, mask=mask_h)

        h += BLOCK_H
```

```python
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module (outgoing mode).
    The kernel computes the N‑by‑N reduction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    All other linear / layer‑norm / gating operations are performed with
    PyTorch, while the cubic contraction is fused into a single Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm on last dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Apply mask if present
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # Cast to half for the contraction kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)
    # Allocate output
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)
    # Strides (int64)
    stride_b = left_f16.stride(0)
    stride_i = left_f16.stride(1)
    stride_k = left_f16.stride(2)
    stride_h = left_f16.stride(3)
    # Grid: one program per (b,i,j)
    total = bs * N * N
    grid = (total,)
    # Tunable block sizes
    BLOCK_K = 64
    BLOCK_H = 64
    # Launch Triton kernel
    tri_mul_kernel[grid](
        left_f16.data_ptr(),
        right_f16.data_ptr(),
        out.data_ptr(),
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        bs,
        N,
        hidden_dim,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    # to_out_norm (layernorm over hidden dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Gated output
    out = out * out_gate
    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy O(N³·C) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented in a single Triton kernel. All other operations
(layer‑norms, linear projections, gates, and the final linear) are
performed with PyTorch for simplicity and to keep the kernel small.
The kernel tiles the reduction dimension `k` and the channel dimension `d`
to stay inside registers and use only a few loads per iteration.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_b, stride_i, stride_k, stride_h,
    B, N, H,
    BLOCK_K: tl.constexpr,   # reduction tile over k
    BLOCK_H: tl.constexpr,   # channel tile over hidden dimension
):
    pid = tl.program_id(0)
    total_ij = N * N
    b = pid // total_ij
    ij = pid - b * total_ij
    i = ij // N
    j = ij - i * N

    # iterate over hidden dimension in chunks of BLOCK_H
    h_off = 0
    while h_off < H:
        # offsets for the hidden chunk
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator for this hidden chunk
        acc = tl.zeros([BLOCK_H], dtype=tl.float32)

        # reduction over the sequence dimension k
        k_off = 0
        while k_off < N:
            cur_k = k_off + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # offsets into left[b,i,k,:] and right[b,j,k,:]
            left_offset = (
                b * stride_b
                + i * stride_i
                + cur_k[:, None] * stride_k
                + offs_h[None, :] * stride_h
            )
            right_offset = (
                b * stride_b
                + j * stride_i
                + cur_k[:, None] * stride_k
                + offs_h[None, :] * stride_h
            )

            # load tiles (half precision -> promote to fp32)
            left_tile = tl.load(
                left_ptr + left_offset,
                mask=mask_k[:, None] & mask_h[None, :],
                other=0.0,
                dtype=tl.float16,
            ).to(tl.float32)
            right_tile = tl.load(
                right_ptr + right_offset,
                mask=mask_k[:, None] & mask_h[None, :],
                other=0.0,
                dtype=tl.float16,
            ).to(tl.float32)

            # element‑wise product and sum over k
            acc += tl.sum(left_tile * right_tile, axis=0)

            k_off += BLOCK_K

        # store the completed hidden chunk
        out_offset = (
            b * stride_b
            + i * stride_i
            + j * stride_k
            + offs_h * stride_h
        )
        tl.store(out_ptr + out_offset, acc, mask=mask_h)

        h_off += BLOCK_H


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: Tensor [B, N, N, dim]
            - mask : Tensor [B, N, N] (or all‑ones when not used)
            - weights: dict with the model weights
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (often 128)

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])        # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])      # [B,N,N,H]

    # ---------- Apply optional mask ----------
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1).to(x.dtype)            # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton contraction ----------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Contiguous tensors give predictable strides
    stride_b = left_f16.stride(0)
    stride_i = left_f16.stride(1)
    stride_k = left_f16.stride(2)
    stride_h = left_f16.stride(3)   # should be 1

    total_programs = B * N * N
    grid = (total_programs,)

    # Tunable tile sizes (experimentally good for the target hardware)
    BLOCK_K = 64   # tile along reduction dimension k
    BLOCK_H = 64   # tile along hidden dimension

    tri_mul_kernel[grid](
        left_f16.data_ptr(),
        right_f16.data_ptr(),
        out.data_ptr(),
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        B,
        N,
        H,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ---------- Output layer‑norm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply out‑gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
