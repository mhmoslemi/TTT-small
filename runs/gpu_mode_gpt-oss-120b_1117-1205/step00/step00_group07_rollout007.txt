@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    total_blocks_per_pair = tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N)
    pair_index = pid // total_blocks_per_pair
    block_index = pid % total_blocks_per_pair
    block_i = block_index // tl.cdiv(N, BLOCK_N)
    block_j = block_index % tl.cdiv(N, BLOCK_N)
    
    b = pair_index // H
    h = pair_index % H
    
    # Compute offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Pointers base
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute dynamic offsets
        k_offset = k * BLOCK_K
        # Pointers for tiles
        left_tile_ptr = left_base + offs_i[:, None] * stride_left_i + (k_offset + offs_k)[None, :] * stride_left_k
        right_tile_ptr = right_base + (k_offset + offs_k)[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        # Load with mask
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = (k_offset + offs_k) < N
        # left mask: both i & k
        left_mask = mask_i[:, None] & mask_k[None, :]
        # right mask: k & j
        right_mask = mask_k[:, None] & mask_j[None, :]

        left_tile = tl.load(left_tile_ptr, mask=left_mask, other=0.0)
        right_tile = tl.load(right_tile_ptr, mask=right_mask, other=0.0)

        # Compute dot
        acc += tl.dot(left_tile, right_tile)  # left MxK, right KxN
    # Store result
    out_tile_ptr = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_tile_ptr, acc, mask=mask_out)
```

"""
TriMul forward pass with a fused Triton kernel for the O(N³) contraction.
The kernel implements the matrix multiplication:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
where left and right are the projected and gated tensors of shape
[B, N, N, H] (H == hidden_dim). All other operations (LayerNorm, gating,
masking, final projection) are performed in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM per hidden dimension (B * H matrices)
# --------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # tensor sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # number of blocks in each dimension for a single (b, h) matrix
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_pair = num_blocks_m * num_blocks_n

    # map program id to (b, h, block_i, block_j)
    pair_id = pid // blocks_per_pair          # 0 .. B*H-1
    block_id = pid % blocks_per_pair
    block_i = block_id // num_blocks_n
    block_j = block_id % num_blocks_n

    b = pair_id // H
    h = pair_id % H

    # offsets within the tensors
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this (b, h) pair
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offset_k = k_iter * BLOCK_K

        # linear indices for loads
        left_off = offs_i[:, None] * stride_left_i + (offset_k + offs_k)[None, :] * stride_left_k
        right_off = (offset_k + offs_k)[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        # masks to avoid out‑of‑bounds loads
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = (offset_k + offs_k) < N
        left_mask = mask_i[:, None] & mask_k[None, :]
        right_mask = mask_k[:, None] & mask_j[None, :]

        # load tiles
        left_tile = tl.load(left_base + left_off, mask=left_mask, other=0.0)
        right_tile = tl.load(right_base + right_off, mask=right_mask, other=0.0)

        # accumulate; left: MxK, right: KxN
        acc += tl.dot(left_tile, right_tile)

    # write the result tile
    out_off = offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_base + out_off, acc, mask=out_mask)


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Performs the forward pass of the TriMul module using a Triton kernel
    for the expensive contraction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (binary, may be ignored)
            - weights: dict of torch.Tensors containing all linear and LN weights
            - config: dict with keys 'dim', 'hidden_dim', 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]        # (hidden_dim, dim)
    left_gate_weight = weights["left_gate.weight"]          # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]        # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]            # (hidden_dim, dim)
    to_out_norm_weight = weights["to_out_norm.weight"]      # (hidden_dim,)
    to_out_norm_bias = weights["to_out_norm.bias"]          # (hidden_dim,)
    to_out_weight = weights["to_out.weight"]                # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Masking (if required)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] broadcast over hidden_dim
        mask_unsq = mask.to(dtype).unsqueeze(-1)
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: O(N³) contraction (left @ rightᵀ per hidden dim)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # ensure contiguity for stride correctness
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    stride_left = left.stride()   # (B, i, k, h)
    stride_right = right.stride()
    stride_out = out.stride()

    # Block sizes (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid dimensions
    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_pairs = B * H
    blocks_per_pair = num_blocks_m * num_blocks_n
    grid = (total_pairs * blocks_per_pair,)

    # Launch kernel
    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: normalize, gate, final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=1e-5)

    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_weight)  # final projection to dim

    return output
