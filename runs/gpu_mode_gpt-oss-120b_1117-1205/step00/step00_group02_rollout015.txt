def triton_tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                         B, N, D,
                         stride_left_b, stride_left_i, stride_left_k, stride_left_d,
                         stride_right_b, stride_right_i, stride_right_k, stride_right_d,
                         stride_out_b, stride_out_i, stride_out_j, stride_out_d,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_g = tl.program_id(2)  # group (batch, channel)

    # Compute batch and channel indices
    b = pid_g // D
    d = pid_g % D

    # Offsets for rows/cols
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute masks for i,j dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers base offsets
    # Cast pointers to 1D pointer
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base = out_ptr + b * stride_out_b + d * stride_out_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (note transposed)
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot product
        acc += tl.dot(left, right)

    # Store result

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Load weights:
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm normalization over dim
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    
    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if present and not nomask
    if not config.get("nomask", False):
        mask_unsqueeze = mask.unsqueeze(-1).to(left.dtype)  # shape [B,N,N,1]
        left = left * mask_unsqueeze
        right = right * mask_unsqueeze

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # Triton kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # get strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        bs * hidden_dim,
    )
    triton_tri_mul_kernel[grid](
        left,
        right,
        out,
        B=bs,
        N=N,
        D=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_d,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Post processing
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output

@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """ Computes out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    where left and right are 4-D tensors of shape (B,N,N,D).
    The kernel tiles the i and j dimensions (BLOCK_M x BLOCK_N) and
    reduces over the k dimension in blocks of size BLOCK_K.
    """

    pid_i = tl.program_id(0)   # tile along i dimension
    pid_j = tl.program_id(1)   # tile along j dimension
    pid_g = tl.program_id(2)   # batch * channel index

    # derive batch and channel
    b = pid_g // D
    d = pid_g % D

    # offsets for the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers that already contain batch and channel offsets
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, d]
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

"""
TriMul (outgoing) forward pass with Triton acceleration.

The most expensive part of the module is the tensor contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched per‑channel matrix multiplication
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T .
We implement this reduction with a custom Triton kernel that tiles the
(i, j) plane (BLOCK_M × BLOCK_N) and reduces over k in blocks of
size BLOCK_K. All other operations (LayerNorms, linear projections,
gating and the final linear) are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].

    The kernel processes one (i‑block, j‑block, batch‑channel) tile.
    """
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_g = tl.program_id(2)          # batch * channel index

    # batch and hidden‑channel dimension
    b = pid_g // D
    d = pid_g % D

    # indices inside the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers already contain batch and channel offsets
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, d]
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[b, j, k, d] -> we need (k, j) ordering
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N)

    # write the tile
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

    """
    Triton‑accelerated forward pass for the TriMul (outgoing) operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)

            - mask: torch.Tensor of shape [B, N, N] (binary 0/1). Ignored if config["nomask"] is True.
            - weights: dict with the following keys (torch.Tensor):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optionally "nomask".
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load model parameters
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # (B,N,N,hidden_dim)

    # Tile sizes (tunable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_ri, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),          # i dimension
        triton.cdiv(N, BLOCK_N),          # j dimension
        B * hidden_dim,                   # batch * hidden channel
    )

    triton_tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_ri,
        s_rk,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing (LayerNorm, gating, final linear)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, to_out_weight)   # (B,N,N,dim)

    return output
