def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # extract dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # optional mask handling: if mask is None: create all-ones mask? But mask is provided.

    # normalization
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # left projection
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for matmul
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute for batched matmul: change to (B, H, N, N) and (B, H, N, N) transposed
    left_perm = left_fp16.permute(0,3,1,2).contiguous()  # (B, hidden_dim, seq_len, seq_len)
    right_perm = right_fp16.permute(0,3,2,1).contiguous()  # (B, hidden_dim, seq_len, seq_len)

    B, H, M, K = left_perm.shape
    # M = K = seq_len
    Z = B * H
    A = left_perm.view(Z, M, K)
    Bmat = right_perm.view(Z, K, M)  # because N = M

    # Create output placeholder
    C = torch.empty((Z, M, M), dtype=torch.float16, device=A.device)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (Z, (M + BLOCK_M - 1)//BLOCK_M, (M + BLOCK_N - 1)//BLOCK_N)

        C.stride(0), C.stride(1), C.stride(2),

        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape output
    out_fp16 = C.view(B, H, M, M).permute(0,2,3,1).contiguous()  # shape [B, N, N, H]

    out_fp32 = out_fp16.to(torch.float32)

    out_gate = out_gate  # shape [B,N,N,H], dtype float32 (from earlier)
    out = out_norm * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          stride_az, stride_ai, stride_ak,
                          stride_bz, stride_bk, stride_bj,
                          stride_cz, stride_ci, stride_cj,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_z = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile
        a_ptrs = A_ptr + pid_z * stride_az + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile
        b_ptrs = B_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = C_ptr + pid_z * stride_cz + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[:, None])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    The main O(N^3) contraction is performed with a custom batched matrix
    multiplication kernel where each hidden dimension is treated as an
    independent batch. The rest of the model (layernorms, linear projections,
    and gating) uses standard PyTorch ops.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm over the feature dimension
    x = F.layer_norm(input_tensor, (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply optional mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half precision for the heavy contraction
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to (B, hidden, N, N) and transpose right for matmul
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where 2nd N is summed

    B_sz, H_sz, N, _ = left_perm.shape
    Z = B_sz * H_sz               # flattened batch * hidden

    A = left_perm.view(Z, N, N)   # (Z, M, K)
    B_mat = right_perm.view(Z, N, N)  # (Z, K, N)

    # Output buffer
    C = torch.empty((Z, N, N), dtype=torch.float16, device=device)

    # Block sizes (tuned for typical GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (Z,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        A, B_mat, C,
        A.stride(0), A.stride(1), A.stride(2),
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = C.view(B_sz, H_sz, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # Back to fp32 for the remaining layers
    out = out.to(torch.float32)

    # Final LayerNorm and gating
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate

    # Output linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bj,
    stride_cz, stride_ci, stride_cj,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication:
       C[z, i, j] = Σ_k A[z, i, k] * B[z, k, j]
       where A shape = (Z, M, K), B shape = (Z, K, N), C shape = (Z, M, N).
    """
    pid_z = tl.program_id(0)          # flattened batch * hidden dimension
    pid_i = tl.program_id(1)          # block index over M
    pid_j = tl.program_id(2)          # block index over N

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for kb in range(num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = A_ptr + pid_z * stride_az + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile
        b_ptrs = B_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = C_ptr + pid_z * stride_cz + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.
    The O(N³) contraction is performed as a batched matrix multiplication
    where each hidden channel is treated as an independent batch.
    All other operations (LayerNorm, Linear projections, gating) use
    standard PyTorch kernels.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input layer‑norm over the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcasted on the feature dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # shape [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Heavy O(N³) contraction using a custom Triton kernel
    # -----------------------------------------------------------------
    # Cast to FP16 for better throughput
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Re‑order to (B, hidden, N, N); right needs a transpose on the two N axes
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  (k, i) order

    B_sz, H_sz, N, _ = left_perm.shape
    Z = B_sz * H_sz                     # flattened batch × hidden

    # View as batched matrices: A[z, i, k] and B[z, k, j]
    A = left_perm.view(Z, N, N)   # (Z, M, K)
    B_mat = right_perm.view(Z, N, N)  # (Z, K, N)

    # Output buffer
    C = torch.empty((Z, N, N), dtype=torch.float16, device=device)

    # Block sizes – these work well on the B200 GPU for the target sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        Z,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        A.stride(0), A.stride(1), A.stride(2),
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = C.view(B_sz, H_sz, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 6) Post‑contraction LayerNorm, gate and final projection
    # -----------------------------------------------------------------
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out
