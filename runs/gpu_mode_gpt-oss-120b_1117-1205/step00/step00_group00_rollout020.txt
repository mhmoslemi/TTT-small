@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k_base = tl.arange(0, BLOCK_K)  # will be added later in loop

    i_mask = i_base < N
    j_mask = j_base < N

    # Initialize accumulator in float32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    num_k_blocks = (N + BLOCK_K - 1) // BLOCK_K
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        k_offset = k_start + k_base
        k_mask = k_offset < N

        # for each d chunk
        for d_start in range(0, H, BLOCK_D):
            d_offset = d_start + tl.arange(0, BLOCK_D)
            d_mask = d_offset < H

            # compute pointer offsets
            left_ptrs = left_ptr + pid_b * stride_bl \
                + i_base[:, None, None] * stride_il \
                + k_offset[None, :, None] * stride_kl \
                + d_offset[None, None, :] * stride_hl
            right_ptrs = right_ptr + pid_b * stride_br \
                + j_base[:, None, None] * stride_jr \
                + k_offset[None, :, None] * stride_kr \
                + d_offset[None, None, :] * stride_hr

            # load tiles
            left_tile = tl.load(left_ptrs, mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)
            right_tile = tl.load(right_ptrs, mask = j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

            # compute matmul for each d in chunk
            # Using loop over d_idx inside block
            # left_tile shape (BLOCK_I, BLOCK_K, BLOCK_D)
            # right_tile shape (BLOCK_J, BLOCK_K, BLOCK_D)
            # For each d_idx:
            for d_idx in range(0, BLOCK_D):
                # create 2D slices
                left_i_k = left_tile[:, :, d_idx]   # shape (BLOCK_I, BLOCK_K)
                right_j_k = right_tile[:, :, d_idx] # shape (BLOCK_J, BLOCK_K)
                # tl.dot expects a (M,K) and (K,N). We'll compute left_i_k @ right_j_k.T
                # So need trans_b=True.
                prod = tl.dot(left_i_k, right_j_k, trans_b=True)
                # prod shape (BLOCK_I, BLOCK_J)
                acc[:, :, d_idx] += prod

    # store out
    out_ptrs = out_ptr + pid_b * stride_bo \
        + i_base[:, None, None] * stride_io \
        + j_base[None, :, None] * stride_jo \
        + tl.arange(0, BLOCK_D)[None, None, :] * stride_ho

    # mask for output
    mask_out = i_mask[:, None, None] & j_mask[None, :, None] & (tl.arange(0, BLOCK_D) < H)[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # offsets
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    i_mask = i < N
    j_mask = j < N

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # number of k-blocks
    num_k = tl.cdiv(N, BLOCK_K)

    for kb in range(num_k):
        k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # Loop over hidden dim block
        for d_start in range(0, H, BLOCK_D):
            d = d_start + tl.arange(0, BLOCK_D)
            d_mask = d < H

            # compute pointer offsets:
            left_ptrs = left_ptr \
                + pid_b * stride_bl \
                + i[:, None, None] * stride_il \
                + k[None, :, None] * stride_kl \
                + d[None, None, :] * stride_hl

            right_ptrs = right_ptr \
                + pid_b * stride_br \
                + j[:, None, None] * stride_jr \
                + k[None, :, None] * stride_kr \
                + d[None, None, :] * stride_hr

            # load tiles
            left_tile = tl.load(left_ptrs, mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)
            right_tile = tl.load(right_ptrs, mask = j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

            # compute per channel dot
            for d_idx in range(0, BLOCK_D):
                # shape (BLOCK_I, BLOCK_K), (BLOCK_J, BLOCK_K)
                left_mat = left_tile[:, :, d_idx]
                right_mat = right_tile[:, :, d_idx]
                # (i,k) * (j,k)^T
                acc[:, :, d_idx] += tl.dot(left_mat, right_mat, trans_b=True)

    # store out
    out_ptrs = out_ptr \
        + pid_b * stride_bo \
        + i[:, None, None] * stride_io \
        + j[None, :, None] * stride_jo \
        + tl.arange(0, BLOCK_D)[None, None, :] * stride_ho

    mask_o = i_mask[:, None, None] & j_mask[None, :, None] & (tl.arange(0, BLOCK_D) < H)[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_o)

def custom_kernel(data):
    """
    Triton implementation of the forward pass of the TriMul operator (outgoing version).
    The kernel fuses the O(N³) multiplication:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    The rest of the module (LayerNorm, Linear projections, gating) is performed in
    PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ---------- LayerNorm on input ----------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # shape [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        # mask shape [B,N,N] -> broadcast over hidden dim
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # float32
        left = left * mask_
        right = right * mask_
    # else no mask (treated as all‑ones)

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the Triton kernel ----------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape   # H == hidden_dim

    # allocate output in half precision
    out_h = torch.empty_like(left_h)   # same shape as left/right

    # strides (int64)
    stride_bl, stride_il, stride_kl, stride_hl = left_h.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right_h.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_h.stride()

    # ---------- Triton launch parameters ----------
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32                       # tile size for hidden dimension
    num_hblocks = triton.cdiv(H, BLOCK_D)

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B * num_hblocks,
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_jr,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        NUM_HBLOCK=num_hblocks,
    )

    # Cast back to FP32 for the remaining layers
    out = out_h.to(torch.float32)

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(out,
                       (H,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # ---------- Apply the output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    output = F.linear(out, weights["to_out.weight"])

    return output

    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: O(N³) multiplicative update
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape  # H == hidden_dim

    out_h = torch.empty_like(left_h)   # fp16 output placeholder

    # strides (int64)
    stride_bl, stride_il, stride_kl, stride_hl = left_h.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right_h.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_h.stride()

    # kernel tile sizes
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32                     # hidden‑dim tile
    num_hblocks = triton.cdiv(H, BLOCK_D)

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B * num_hblocks,
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_jr,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        NUM_HBLOCK=num_hblocks,
    )

    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
