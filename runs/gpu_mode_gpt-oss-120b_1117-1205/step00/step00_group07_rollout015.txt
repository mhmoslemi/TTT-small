import torch
import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(axis=0)
    pid_m = tl.program_id(axis=1)
    pid_n = tl.program_id(axis=2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A (M x K)
        a_ptrs = A_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = tl.cast(a, tl.float32)

        # Pointers for B (K x N) -> we treat B as transposed
        b_ptrs = B_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = tl.cast(b, tl.float32)

        # Compute dot product and accumulate
        acc += tl.dot(a, b)

    # Write out
    c_ptrs = C_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated implementation of the TriMul (outgoing) operation.
    Performs LayerNorm, linear projections, gating, an O(N^3) contraction via
    a fused batched GEMM kernel, and final LayerNorm + output projection.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # ----- LayerNorm on input (last dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Ensure tensor is float32
    x = x.to(torch.float32)

    # ----- Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Apply mask if provided (mask shape [B,N,N])
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # ----- Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----- Cast to half precision for the heavy contraction
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ----- Rearrange to (batch*hidden, N, N) contiguous layout
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()

    )

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel for the heavy O(N³) contraction:
# out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
# The kernel treats each (b, d) pair as an independent batch of
# (N, N) @ (N, N).T matrix multiplication.
# --------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    A_ptr,                # left  matrix pointer  (batch, M, K)
    B_ptr,                # right matrix pointer  (batch, K, N)  (transposed view)
    C_ptr,                # output pointer         (batch, M, N)
    M, N, K,              # matrix dimensions (all = seq_len)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(axis=0)
    pid_m = tl.program_id(axis=1)
    pid_n = tl.program_id(axis=2)

    # ---- block offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- iterate over K dimension --------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (M x K) slice
        a_ptrs = (A_ptr + pid_batch * stride_a_batch
                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # FP32 load (tensor stored as FP32)

        # B: (K x N) slice (original right matrix is transposed on‑the‑fly)
        b_ptrs = (B_ptr + pid_batch * stride_b_batch
                  + offs_k[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # multiply‑add
        acc += tl.dot(a, b)               # both a and b are FP32 → acc stays FP32

    # ---- write the result ------------------------------------------------
    c_ptrs = (C_ptr + pid_batch * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul implementation.
    Performs layer‑norm, linear projections, gating, the O(N³) contraction
    via a fused batched GEMM kernel, a second layer‑norm, gating and the final
    linear projection back to the original feature dimension.
    """
    # ----- unpack inputs -------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_sz, seq_len, _, _ = input_tensor.shape

    # ----- first LayerNorm ------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    ).to(torch.float32)

    # ----- linear projections (no bias) ----------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- optional mask -------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # (B, N, N, 1)
        left = left * mask_
        right = right * mask_

    # ----- gating ---------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # Heavy O(N³) contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------------
    # Rearrange to (batch*hidden, N, N) contiguous layout for the kernel.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = batch_sz * hidden_dim
    N = seq_len

    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float32, device=device)

    # strides for the contiguous (BH, N, N) tensors
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # launch kernel
    trimul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_m,       # B is accessed as (K,N) → k‑stride, n‑stride
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(batch_sz, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ----- second LayerNorm over hidden dimension -----------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # ----- apply out gate ------------------------------------------------
    out = out * out_gate

    # ----- final linear projection to original dim ----------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
