@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, seq_len, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   pid0 = tl.program_id(0)  # group id (batch * hidden)
   pid1 = tl.program_id(1)  # block row index i
   pid2 = tl.program_id(2)  # block col index j

   # compute batch and hidden index
   b = pid0 // hidden_dim
   d = pid0 % hidden_dim

   # compute start offset
   i_offset = pid1 * BLOCK_M
   j_offset = pid2 * BLOCK_N

   # accumulate
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over k dimension
   for k in range(0, tl.cdiv(seq_len, BLOCK_K)):
       # compute offsets
       k_offset = k * BLOCK_K

       # load left tile
       i = i_offset + tl.arange(0, BLOCK_M)  # shape (BLOCK_M,)
       k_idx = k_offset + tl.arange(0, BLOCK_K)  # shape (BLOCK_K,)
       # mask for bounds
       mask_i = i < seq_len
       mask_k = k_idx < seq_len
       # pointer
       left_ptrs = left_ptr + b*stride_left_b + i[:,None] * stride_left_i + k_idx[None,:] * stride_left_k + d * stride_left_d
       a = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)

       # load right tile (transpose)
       j = j_offset + tl.arange(0, BLOCK_N)
       k_idx = k_offset + tl.arange(0, BLOCK_K)  # reuse k_idx
       mask_j = j < seq_len
       # pointer: right stored as (b, j, k, d) but we load as (k, j)
       right_ptrs = right_ptr + b*stride_right_b + j[None,:] * stride_right_j + k_idx[:,None] * stride_right_k + d * stride_right_d
       b_tile = tl.load(right_ptrs, mask=mask_j[None,:] & mask_k[:,None], other=0.0)

       acc += tl.dot(a, b_tile)

   # store output
   out_i = i_offset + tl.arange(0, BLOCK_M)
   out_j = j_offset + tl.arange(0, BLOCK_N)
   mask_i = out_i < seq_len
   mask_j = out_j < seq_len
   out_ptrs = out_ptr + b*stride_out_b + out_i[:,None] * stride_out_i + out_j[None,:] * stride_out_j + d * stride_out_d
   tl.store(out_ptrs, acc, mask=mask_i[:,None] & mask_j[None,:])

    # Get dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    norm_bias = weights["norm.bias"]

    )  # shape (B,N,N,dim)

    # left_proj_weight: shape (hidden_dim, dim)

    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # (B,N,N,hidden_dim)

    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # allocate output

```python
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, seq_len, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    using a fused batched matmul per hidden channel.
    """
    pid0 = tl.program_id(0)  # group id: batch * hidden_dim
    pid1 = tl.program_id(1)  # i block
    pid2 = tl.program_id(2)  # j block

    # Decode batch and hidden dimension
    b = pid0 // hidden_dim
    d = pid0 % hidden_dim

    # Starting offsets of the tile
    i_offset = pid1 * BLOCK_M
    j_offset = pid2 * BLOCK_N

    # Create range indices
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # Compute absolute indices
    i = i_offset + i_range
    j = j_offset + j_range

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_iter in range(0, tl.cdiv(seq_len, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        k = k_offset + k_range

        # Bounds masks
        mask_i = i < seq_len
        mask_j = j < seq_len
        mask_k = k < seq_len

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_left_b + \
            i[:, None] * stride_left_i + \
            k[None, :] * stride_left_k + \
            d * stride_left_d
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile as (BLOCK_K, BLOCK_N) (k as rows, j as columns)
        right_ptrs = right_ptr + b * stride_right_b + \
            j[None, :] * stride_right_j + \
            k[:, None] * stride_right_k + \
            d * stride_right_d
        right_tile = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0)

        # Accumulate dot product
        acc += tl.dot(left_tile, right_tile)

    # Store results
    out_ptrs = out_ptr + b * stride_out_b + \
        i[:, None] * stride_out_i + \
        j[None, :] * stride_out_j + \
        d * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, seq_len, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    via tiled batched matrix multiplication. `left` and `right` have shape
    (B, N, N, hidden_dim). `out` has the same batch and hidden dimensions
    and (N, N) spatial dimensions. The kernel processes one (batch, hidden)
    pair per program (axis 0) and tiles the (i,j) output matrix.
    """
    pid0 = tl.program_id(0)        # combined batch * hidden dim
    pid1 = tl.program_id(1)        # i tile
    pid2 = tl.program_id(2)        # j tile

    # Decode batch index and hidden channel
    b = pid0 // hidden_dim
    d = pid0 % hidden_dim

    # Starting coordinates of the tile
    i_offset = pid1 * BLOCK_M
    j_offset = pid2 * BLOCK_N

    # Index ranges inside the tile
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i = i_offset + i_range
    j = j_offset + j_range

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    num_k = tl.cdiv(seq_len, BLOCK_K)
    for kk in range(0, num_k):
        k = kk * BLOCK_K + k_range

        mask_i = i < seq_len
        mask_j = j < seq_len
        mask_k = k < seq_len

        # Load a tile of `left` – shape (M, K)
        left_ptrs = left_ptr + b * stride_left_b + \
            i[:, None] * stride_left_i + \
            k[None, :] * stride_left_k + \
            d * stride_left_d
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # Load a tile of `right` transposed – shape (K, N)
        # right is stored as (b, j, k, d); we want (k, j) layout.
        right_ptrs = right_ptr + b * stride_right_b + \
            j[None, :] * stride_right_j + \
            k[:, None] * stride_right_k + \
            d * stride_right_d
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[None, :] & mask_k[:, None],
                             other=0.0)

        # Perform the matrix‑multiply‑accumulate for this K‑slice.
        acc += tl.dot(left_tile, right_tile)

    # Write the result back
    mask_i = i < seq_len
    mask_j = j < seq_len
    out_ptrs = out_ptr + b * stride_out_b + \
        i[:, None] * stride_out_i + \
        j[None, :] * stride_out_j + \
        d * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul (outgoing) module.
    Operates on a 4‑D tensor [B, N, N, C] and returns a tensor of the same
    shape after the full sequence of linear projections, gating, the triangular
    multiplicative update (computed in a Triton kernel), layer‑norms and the
    final linear projection.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch, seq_len, _, _ = input_tensor.shape

    # ---- LayerNorm on the input (dim) ----
    x = F.layer_norm(
        input_tensor.to(dtype),
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---- Linear projections ----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---- Apply mask if present ----
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # shape (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ---- Gating ----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Make tensors contiguous for efficient Triton access
    left = left.contiguous()
    right = right.contiguous()

    # ---- Triangular multiplicative update (einsum) via Triton ----
    out_hidden = torch.empty((batch, seq_len, seq_len, hidden_dim), device=device, dtype=dtype)

    # Block sizes – chosen to give good occupancy across the tested dimensions.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

        batch * hidden_dim,
        triton.cdiv(seq_len, BLOCK_M),

        stride_out[1],
        stride_out[2],
        stride_out[3],

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out_norm * out_gate  # gating after norm

    # Final linear projection back to `dim`

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, seq_len, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute
        out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
    using a tiled batched matrix‑multiply.  Each program (axis 0) works on one
    (batch, hidden‑channel) pair, while axes 1 and 2 tile the (i, j) output
    matrix.
    """
    pid0 = tl.program_id(0)          # batch * hidden_dim
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    # decode batch index and hidden channel
    b = pid0 // hidden_dim
    d = pid0 % hidden_dim

    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i = i0 + i_range
    j = j0 + j_range

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(seq_len, BLOCK_K)
    for kk in range(0, num_k):
        k = kk * BLOCK_K + k_range

        mask_i = i < seq_len
        mask_j = j < seq_len
        mask_k = k < seq_len

        # left: (M, K)  → shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_left_b + \
                    i[:, None] * stride_left_i + \
                    k[None, :] * stride_left_k + \
                    d * stride_left_d
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right (stored as (b, j, k, d)) needed as (K, N)
        right_ptrs = right_ptr + b * stride_right_b + \
                     j[None, :] * stride_right_j + \
                     k[:, None] * stride_right_k + \
                     d * stride_right_d
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[None, :] & mask_k[:, None],
                             other=0.0)

        acc += tl.dot(left_tile, right_tile)

    # store the result
    mask_i = i < seq_len
    mask_j = j < seq_len
    out_ptrs = out_ptr + b * stride_out_b + \
               i[:, None] * stride_out_i + \
               j[None, :] * stride_out_j + \
               d * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] torch.Tensor
            - mask:          [B, N, N] torch.Tensor (may be all‑ones)
            - weights:       Dict of model parameters
            - config:        Dict with keys ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------------
    # 1) Input LayerNorm
    # -------------------------------------------------------
    x = F.layer_norm(
        input_tensor.to(dtype),
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------------
    # 2) Linear projections (no bias)
    # -------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------
    # 3) Apply mask (if provided)
    # -------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)          # (B, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # -------------------------------------------------------
    # 4) Gating
    # -------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------
    # 5) Triangular multiplicative update (einsum) via Triton
    # -------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out_hidden = torch.empty((B, N, N, hidden_dim), device=device, dtype=dtype)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    sL = left.stride()
    sR = right.stride()
    sO = out_hidden.stride()

    trimul_einsum_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        # left strides
        sL[0], sL[1], sL[2], sL[3],
        # right strides
        sR[0], sR[1], sR[2], sR[3],
        # output strides
        sO[0], sO[1], sO[2], sO[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------
    # 6) Post‑update processing (norm, gate, final linear)
    # -------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out_norm * out_gate                       # gating after norm
    final = F.linear(out, weights["to_out.weight"]) # (hidden_dim -> dim)

    return final
