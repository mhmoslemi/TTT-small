def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, N2, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Normalize input
    # use torch.nn.functional.layer_norm across last dim
    x_norm = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Move to half for speed
    x_norm_h = x_norm.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_norm_h, weights['left_proj.weight'].to(torch.float16))
    right = torch.nn.functional.linear(x_norm_h, weights['right_proj.weight'].to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, weights['left_gate.weight'].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, weights['right_gate.weight'].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, weights['out_gate.weight'].to(torch.float16)))

    # Apply mask and gates
    if config.get('nomask', True):
        left = left * left_gate
        right = right * right_gate
    else:
        # mask shape [B,N,N]; broadcast to hidden_dim
        mask_h = mask.to(torch.float16)
        # Apply mask and gating using Triton kernel
        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        total_elements = left.numel()  # B*N*N*hidden_dim
        BLOCK = 1024  # tune
        grid = (triton.cdiv(total_elements, BLOCK),)

        # Flatten tensors
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)

        mask_gate_kernel[grid](
            left_flat, mask_flat, left_gate_flat, left_out.view(-1),
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
        )
        mask_gate_kernel[grid](
            right_flat, mask_flat, right_gate_flat, right_out.view(-1),
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
        )
        left = left_out
        right = right_out

    # Reshape for matmul
    # Permute dims for batched GEMM
    left_mat = left.permute(0,3,1,2).reshape(bs * hidden_dim, N, N)  # shape [B*H, N, N]

"""
TriMul forward (outgoing) with a fused mask‑gate Triton kernel.

The heavy O(N³) reduction is performed with a batched GEMM (torch.bmm).
A Triton kernel fuses the element‑wise mask and gate multiplication:
    out = (proj * mask) * gate
All other operations (LayerNorm, Linear, Sigmoid) are done with
torch.nn.functional for numerical fidelity. The final result is
returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: fused mask * gate * source
# --------------------------------------------------------------------------- #
@triton.jit
def mask_gate_kernel(
    src_ptr,          # source tensor (flattened)            [total_elems]
    mask_ptr,         # mask tensor (flattened)              [total_rows]
    gate_ptr,         # gate tensor (flattened)              [total_elems]
    dst_ptr,          # output tensor (flattened)            [total_elems]
    HIDDEN_DIM: tl.constexpr,  # hidden dimension (channels)
    total_elems: tl.int64,     # total number of elements in src/gate/dst
    BLOCK: tl.constexpr,       # work per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # load source and gate
    src = tl.load(src_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # compute row index (ignoring channel) to index mask
    row = offs // HIDDEN_DIM
    mask = tl.load(mask_ptr + row, mask=valid, other=1.0)  # default 1.0 (no mask)

    out = src * gate * mask
    tl.store(dst_ptr + offs, out, mask=valid)


# --------------------------------------------------------------------------- #
# Triton‑accelerated TriMul forward
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Arguments:
        data = (input_tensor, mask_tensor, weight_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # --------------------------------------------------------------------- #
    # Basic shapes / config
    # --------------------------------------------------------------------- #
    B, N, _, D = input_tensor.shape  # D == dim
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # 1️⃣ LayerNorm on the input (float32)
    # --------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  float32

    # Cast to float16 for the heavy path
    x_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2️⃣ Linear projections (float16)
    # --------------------------------------------------------------------- #
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))    # [B,N,N,hidden_dim]
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------------- #
    # 3️⃣ Gates (float16) – sigmoid after linear
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))

    # --------------------------------------------------------------------- #
    # 4️⃣ Apply mask + gate.
    #    When `nomask` is True we skip the mask‑kernel and use a simple multiply.
    # --------------------------------------------------------------------- #
    if not nomask:
        # Ensure contiguous layout for the flattened view
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask.to(torch.float16).contiguous()

        # Prepare output buffers
        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        # Flatten everything to 1‑D tensors (contiguous)
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        lg_flat = left_gate.view(-1)
        rg_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)                # shape B*N*N
        total_elems = left_flat.numel()            # B*N*N*hidden_dim
        BLOCK = 1024

        grid = (triton.cdiv(total_elems, BLOCK),)

        # Fuse mask, gate, and source for the left branch
        mask_gate_kernel[grid](
            left_flat,
            mask_flat,
            lg_flat,
            left_out.view(-1),
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
        )
        # Fuse mask, gate, and source for the right branch
        mask_gate_kernel[grid](
            right_flat,
            mask_flat,
            rg_flat,
            right_out.view(-1),
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
        )
        left = left_out
        right = right_out
    else:
        # No mask – simple elementwise multiplication with the gates
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5️⃣ Batched matrix multiplication over the sequence dimension.
    #     Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #     This is a batched GEMM: (i,k) @ (j,k)^T for each (b,d).
    # --------------------------------------------------------------------- #
    # Rearrange to [B*hidden_dim, N, N] for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (i,k)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (j,k)

    # GEMM (float16) – highly optimized in torch
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))            # [B*H, N, N]

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------------- #
    # 6️⃣ LayerNorm over the hidden dimension (float16)
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(out.dtype),
        bias=weights["to_out_norm.bias"].to(out.dtype),
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # 7️⃣ Apply the output gate and final linear projection
    # --------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(out.dtype))  # -> [B,N,N,dim]

    # Cast back to float32 as required by the interface
    return out.to(torch.float32)
