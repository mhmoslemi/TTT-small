def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # layer norm x
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # left and right projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask handling
    if config.get("nomask", False):
        # ignore mask; set mask = 1
        mask_f = torch.ones_like(mask, dtype=left.dtype, device=device)
    else:
        mask_f = mask
    mask_f = mask_f.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

"""
TriMul "outgoing" implementation with a Triton kernel for the
computationally heavy pairwise multiplication (einsum) step:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

The surrounding LayerNorms, Linear projections and gating are done with
torch.nn.functional for simplicity. The Triton kernel computes the
N³·hidden_dim reduction efficiently using tiled outer‑product accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: tiled outer‑product sum over the k‑dimension
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch
    pid_m = tl.program_id(1)          # i tile
    pid_n = tl.program_id(2)          # j tile
    pid_d = tl.program_id(3)          # hidden‑dim tile

    # -------------------------------------------------------------------------
    # offsets for the current tile
    # -------------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_d = offs_d < H

    # base pointers for this batch
    left_base  = left_ptr  + pid_b * stride_lb
    right_base = right_ptr + pid_b * stride_rb
    out_base   = out_ptr   + pid_b * stride_ob

    # accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # -------------------------------------------------------------------------
    # loop over k in blocks of size BLOCK_K
    # -------------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left tile: shape (BM, BK, BD)
        left_ptrs = (left_base +
                     (offs_m[:, None, None] * stride_li) +
                     (k_off[None, :, None] * stride_lk) +
                     (offs_d[None, None, :] * stride_ld))
        left_tile = tl.load(
            left_ptrs,
            mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0)

        # right tile: shape (BN, BK, BD)
        right_ptrs = (right_base +
                      (offs_n[:, None, None] * stride_rj) +
                      (k_off[None, :, None] * stride_rk) +
                      (offs_d[None, None, :] * stride_rd))
        right_tile = tl.load(
            right_ptrs,
            mask=mask_n[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0)

        # accumulate outer‑product for each element of the k‑block
        for kk in range(0, BLOCK_K):
            # slices of shape (BM, BD) and (BN, BD)
            left_kk  = left_tile[:, kk, :]          # (BM, BD)
            right_kk = right_tile[:, kk, :]         # (BN, BD)

            # broadcast to (BM, BN, BD) and accumulate
            acc += left_kk[:, None, :] * right_kk[None, :, :]

    # -------------------------------------------------------------------------
    # write out the tile
    # -------------------------------------------------------------------------
    out_ptrs = (out_base +
                (offs_m[:, None, None] * stride_oi) +
                (offs_n[None, :, None] * stride_oj) +
                (offs_d[None, None, :] * stride_od))
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_d[None, None, :])


# ----------------------------------------------------------------------------- #
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul module.
    Parts:
      * LayerNorm over the input channel dimension
      * Linear projections + optional masking
      * Gated linear layers
      * Heavy N³·hidden_dim reduction performed by a Triton kernel
      * Final LayerNorm, gating and output projection
    Args:
        data: (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # -------------------------------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x = x.to(dtype)  # ensure float32 for the rest of the pipeline

    # -------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------------
    # 3. Optional mask handling
    # -------------------------------------------------------------------------
    if config.get("nomask", False):
        mask_f = torch.ones_like(mask, dtype=x.dtype, device=device)
    else:
        mask_f = mask.to(dtype=x.dtype)
    mask_f = mask_f.unsqueeze(-1)               # [B, N, N, 1]

    left = left * mask_f
    right = right * mask_f

    # -------------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections of the normalized input)
    # -------------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Heavy tri‑multiplicative reduction via Triton
    # -------------------------------------------------------------------------
    out = torch.empty((bs, seq_len, seq_len, hidden_dim),
                      dtype=dtype, device=device)

    # strides for the three tensors (batch, i, k/j, d)
    stride_l = left.stride()    # (B, i, k, d)
    stride_r = right.stride()
    stride_o = out.stride()

    # block size configuration (tuned for typical A200 memory hierarchy)
    BLOCK_M = 64   # tiles along first sequence dimension (i)
    BLOCK_N = 64   # tiles along second sequence dimension (j)
    BLOCK_K = 8    # reduction block size over k
    BLOCK_D = 32   # tile over the hidden dimension

    grid = (
        bs,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        triton.cdiv(hidden_dim, BLOCK_D),
    )
    trimul_kernel[grid](
        left,
        right,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # -------------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # -------------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
```
