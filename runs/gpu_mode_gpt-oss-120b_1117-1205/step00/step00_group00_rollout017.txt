"""
TriMul forward (outgoing) with fused batched matrix multiplication in Triton.

The tensor x ∈ [B, N, N, C] is first layer‑normalized, then projected to a
hidden dimension H with two linear layers (left / right).  After masking the
projections, element‑wise gates (sigmoid of linear layers) are applied.
The core “TriMul” operation computes for each hidden channel d:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is a batched matrix multiplication L @ Rᵀ for every hidden slice.
The heavy reduction over k is performed by a Triton kernel that tiles the
(i, j) output matrix and accumulates over k.  The result is layer‑normed,
gated again, and finally projected back to the original channel dimension.
All other ops stay in PyTorch; the kernel is used for the O(N³) reduction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    N: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul: out = left @ rightᵀ for each (b, d) slice.

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : pointers to contiguous tensors
        shape = [B, N, N, H] (last dim = hidden)
    B, H, N : compile‑time integers
    stride_* : strides (in elements) for the corresponding dimensions
    """
    pid_batch = tl.program_id(0)      # 0 … B*H-1  (combined batch and hidden)
    pid_i = tl.program_id(1)          # tile row
    pid_j = tl.program_id(2)          # tile column

    # derive batch index and hidden slice
    b = pid_batch // H
    d = pid_batch % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, d)
    base_left = left_ptr + b * stride_left_b + d * stride_left_d
    base_right = right_ptr + b * stride_right_b + d * stride_right_d
    base_out = out_ptr + b * stride_out_b + d * stride_out_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles (compile‑time)
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K)
        left_ptrs = base_left + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block: (K, N)   (remember we need rightᵀ → use k as first dim)
        right_ptrs = base_right + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)

    # write the tile
    out_ptrs = base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    expensive einsum reduction.

    Args
    ----
    data : tuple (input, mask, weights, config)
        - input : torch.Tensor, shape [bs, L, L, dim]
        - mask  : torch.Tensor, shape [bs, L, L] (may be all‑ones)
        - weights: dict of tensors for all linear / layer‑norm parameters
        - config : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [bs, L, L, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input
    # -----------------------------------------------------------------
    x = input_tensor.to(torch.float32)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, L, L, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers) and apply
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Batched matrix multiplication via Triton
    # -----------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape  # N == seq_len
    out = torch.empty_like(left, dtype=torch.float32)

    # block sizes (tuned for the B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (in elements)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_d = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)   # second N dimension
    stride_right_k = right.stride(2)
    stride_right_d = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_d = out.stride(3)

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_einsum_kernel[grid](
        left, right, out,
        B, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        N=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Output layer norm, final gating and linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Cast back to the original dtype (if it wasn't float32)
    out = out.to(input_tensor.dtype)
    return out
