@triton.jit
def gemm_kernel(A_ptr, B_ptr, C_ptr,
                M, N, K,
                stride_am, stride_ak,
                stride_bk, stride_bn,
                stride_cm, stride_cn,
                batch_idx,
                BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0) # row block id
    pid_n = tl.program_id(1) # col block id
    pid_b = tl.program_id(2) # batch index

    # compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # bound check
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # load A and B
        a = tl.load(A_ptr + (pid_b * stride_am * M) + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak),
                    mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)  # or if apple dtype?
        b = tl.load(B_ptr + (pid_b * stride_bk * K) + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn),
                    mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        # dot product using tl.dot
        acc += tl.dot(a, b) # a shape (BLOCK_M, BLOCK_K), b shape (BLOCK_K, BLOCK_N). dot returns (BLOCK_M, BLOCK_N)
        # increment pointers? Actually using offsets each iteration

    # store results
    c_ptrs = C_ptr + (pid_b * stride_cm * M) + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16) if C dtype is half else acc, mask=mask_m[:, None] & mask_n[None, :])

        right_half,
        out_flat,
        M, N, K,

        stride_b0, stride_b1, stride_b2,
        stride_c0, stride_c1, stride_c2,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    
    # reshape output back

        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    
    # gating
    out = out * out_gate
    
    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)
    
    return out

"""
TriMul (outgoing) forward pass with a Triton kernel for the expensive
contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

All surrounding operations (LayerNorm, Linear projections, gating) are
performed with PyTorch for simplicity.  The contraction is expressed as a
batched matrix multiplication per hidden dimension (left @ right^T) and
implemented by a custom blocked GEMM kernel in Triton.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton GEMM kernel: C = A @ B   where A : (batch, M, K), B : (batch, K, N)
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_m = tl.program_id(0)   # block-row
    pid_n = tl.program_id(1)   # block-col
    pid_b = tl.program_id(2)   # batch (combined batch*hidden)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_b * stride_Ab
            + (offs_m[:, None] * stride_Am)
            + (offs_k[None, :] * stride_Ak)
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_b * stride_Bb
            + (offs_k[:, None] * stride_Bk)
            + (offs_n[None, :] * stride_Bn)
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate block multiplication
        acc += tl.dot(a, b)

    # Write back the result tile
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + (offs_m[:, None] * stride_Cm)
        + (offs_n[None, :] * stride_Cn)
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------------- #
# Public entry point expected by the evaluator
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            * input_tensor : torch.Tensor [B, N, N, dim]
            * mask         : torch.Tensor [B, N, N] (float or bool)
            * weights      : dict of torch.Tensor containing model parameters
            * config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # Load weight tensors (already on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # --------------------------------------------------------------- #
    # 1. LayerNorm on the input (over dim)
    # --------------------------------------------------------------- #
    x = input_tensor.to(dtype)
    x = F.layer_norm(x, normalized_shape=(dim,),
                     weight=norm_weight, bias=norm_bias, eps=1e-5)

    # --------------------------------------------------------------- #
    # 2. Linear projections and gating (still in fp32)
    # --------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    if use_mask:
        # mask is broadcasted over the hidden dimension
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 3. Core contraction using Triton (batched GEMM per hidden dim)
    # --------------------------------------------------------------- #
    B, N, _, _ = left.shape                     # batch size, sequence length
    # Rearrange tensors so that hidden_dim becomes part of the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) -> transposed K,N

    # Flatten the first two dimensions for the GEMM kernel
    left_flat = left_perm.view(-1, N, N)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(-1, N, N) # (B*H, K=N, N)

    # Output buffer (fp32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Strides (in elements)
    stride_Ab, stride_Am, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_flat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M
        triton.cdiv(N, BLOCK_N),   # blocks along N
        left_flat.shape[0],        # combined batch*hidden
    )

    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------- #
    # 4. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------- #
    out = F.layer_norm(out, normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
