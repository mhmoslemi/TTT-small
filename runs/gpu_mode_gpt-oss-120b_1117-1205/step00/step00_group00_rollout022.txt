    hidden_dim = config["hidden_dim"]

    # compute linear projections & gates
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    right = torch.nn.functional.linear(x, right_proj_weight)

    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()

import triton
import triton.language as tl

@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_rows, hidden_dim,
    stride_row, stride_col,
    mask_stride,
    BLOCK: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_indices = pid * BLOCK + tl.arange(0, BLOCK)
    row_mask = row_indices < total_rows

    # load scalar mask per row (broadcast later)
    mask_vals = tl.load(mask_ptr + row_indices * mask_stride, mask=row_mask, other=0.0)  # shape (BLOCK,)
    mask_vals = mask_vals[:, None]  # broadcast

    # compute base offset for rows
    row_base = row_indices * stride_row

    # Loop over hidden dim
    for offset_h in range(0, hidden_dim, BLOCK_H):
        col_indices = tl.arange(0, BLOCK_H) + offset_h
        col_mask = col_indices < hidden_dim
        # offsets matrix
        offsets = row_base[:, None] + col_indices[None, :]  # shape (BLOCK, BLOCK_H)

        # load values
        left = tl.load(left_ptr + offsets, mask=row_mask[:, None] & col_mask[None, :], other=0.0)
        left_gate = tl.load(left_gate_ptr + offsets, mask=row_mask[:, None] & col_mask[None, :], other=0.0)
        right = tl.load(right_ptr + offsets, mask=row_mask[:, None] & col_mask[None, :], other=0.0)
        right_gate = tl.load(right_gate_ptr + offsets, mask=row_mask[:, None] & col_mask[None, :], other=0.0)

        # compute gated values
        left_out = left * mask_vals * left_gate
        right_out = right * mask_vals * right_gate

        # cast to bf16
        left_out_bf16 = tl.cast(left_out, tl.bfloat16)
        right_out_bf16 = tl.cast(right_out, tl.bfloat16)

        # store
        tl.store(left_out_ptr + offsets, left_out_bf16, mask=row_mask[:, None] & col_mask[None, :])
        tl.store(right_out_ptr + offsets, right_out_bf16, mask=row_mask[:, None] & col_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure tensors on device and correct dtype
    # Usually weights are float32 already.

    # LayerNorm
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # float32
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Flatten for kernel
    total_rows = bs * N * N
    left_flat = left.reshape(total_rows, hidden_dim).contiguous()
    right_flat = right.reshape(total_rows, hidden_dim).contiguous()
    left_gate_flat = left_gate.reshape(total_rows, hidden_dim).contiguous()
    right_gate_flat = right_gate.reshape(total_rows, hidden_dim).contiguous()
    mask_flat = mask.reshape(total_rows).contiguous().to(x.dtype)

    # Allocate output tensors for fused masked and gated values (bf16)
    left_out_flat = torch.empty_like(left_flat, dtype=torch.bfloat16)
    right_out_flat = torch.empty_like(right_flat, dtype=torch.bfloat16)

    # Launch Triton kernel
    BLOCK = 128  # rows per block
    BLOCK_H = 32  # hidden per iteration
    grid = (triton.cdiv(total_rows, BLOCK),)
    mask_gate_fuse_kernel[grid](
        left_flat, left_gate_flat,
        right_flat, right_gate_flat,
        mask_flat,
        left_out_flat, right_out_flat,
        total_rows, hidden_dim,
        left_flat.stride(0), left_flat.stride(1),
        mask_flat.stride(0),
        BLOCK=BLOCK, BLOCK_H=BLOCK_H
    )

    # Release original tensors to free memory (optional)
    del left, right, left_gate, right_gate

    # Reshape back
    left_out = left_out_flat.view(bs, N, N, hidden_dim)
    right_out = right_out_flat.view(bs, N, N, hidden_dim)

    # Matmul contraction using batched bmm
    # Convert to float16 for speed
    left_mat = left_out.permute(0, 3, 1, 2).contiguous().reshape(bs * hidden_dim, N, N).to(torch.float16)
    right_mat = right_out.permute(0, 3, 1, 2).contiguous().reshape(bs * hidden_dim, N, N).to(torch.float16)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N), dtype float16

    # Reshape back to (B, N, N, H) and convert to float32
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # Apply to_out_norm (LayerNorm on hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to original dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

import triton
import triton.language as tl

@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_rows, hidden_dim,
    stride_row, stride_col,
    mask_stride,
    BLOCK: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_idx = pid * BLOCK + tl.arange(0, BLOCK)
    row_mask = row_idx < total_rows

    # Load mask values (scalar per row)
    mask_val = tl.load(mask_ptr + row_idx * mask_stride, mask=row_mask, other=0.0)

    # Base offset per row
    row_base = row_idx * stride_row

    for offset_h in range(0, hidden_dim, BLOCK_H):
        col_idx = tl.arange(0, BLOCK_H) + offset_h
        col_mask = col_idx < hidden_dim

        # Compute 2D offsets
        offs = row_base[:, None] + col_idx[None, :]

        # Load values with masking
        left = tl.load(left_ptr + offs, mask=row_mask[:, None] & col_mask[None, :], other=0.0)

        # Apply mask and gates
        left_out = left * mask_val * left_gate
        right_out = right * mask_val * right_gate

        # Cast to bfloat16 and store
        tl.store(left_out_ptr + offs, tl.cast(left_out, tl.bfloat16), mask=row_mask[:, None] & col_mask[None, :])
        tl.store(right_out_ptr + offs, tl.cast(right_out, tl.bfloat16), mask=row_mask[:, None] & col_mask[None, :])

"""
TriMul forward (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections produce `left` and `right` (shape B,N,N,H).
3. Compute three gate tensors (sigmoid) – `left_gate`, `right_gate`, `out_gate`.
4. **Triton kernel** fuses:
      left  = left  * mask * left_gate
      right = right * mask * right_gate
   The kernel works on flattened (B·N·N, H) tensors, applies the mask,
   multiplies by the gates and stores the result as bfloat16.
5. Contract the two gated tensors:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of independent matrix multiplies
   L_d @ R_dᵀ for each hidden dimension `d`.  We reshape to
   (B·H, N, N) and use `torch.bmm` (CUDA cuBLAS) on float16.
6. Layer‑norm on the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.

The heavy pairwise contraction is performed by a fast cuBLAS batched GEMM,
while the mask‑gate fusion is realised in Triton, satisfying the
requirement of having a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_fuse_kernel(
    left_ptr,          # float32  [total_rows, hidden_dim]
    left_gate_ptr,     # float32  [total_rows, hidden_dim]
    right_ptr,         # float32  [total_rows, hidden_dim]
    right_gate_ptr,    # float32  [total_rows, hidden_dim]
    mask_ptr,          # float32  [total_rows]
    left_out_ptr,      # bfloat16 [total_rows, hidden_dim]
    right_out_ptr,     # bfloat16 [total_rows, hidden_dim]
    total_rows: tl.int32,
    hidden_dim: tl.int32,
    stride_row: tl.int32,      # == hidden_dim for the flattened tensors
    stride_col: tl.int32,      # == 1  (kept for completeness)
    mask_stride: tl.int32,     # == 1
    BLOCK: tl.constexpr,       # rows per program
    BLOCK_H: tl.constexpr,     # hidden‑dim tile size
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1) Row indices this program works on
    # ------------------------------------------------------------------
    row_idx = pid * BLOCK + tl.arange(0, BLOCK)
    row_mask = row_idx < total_rows

    # Load mask (scalar per row) and broadcast to hidden dimension
    mask_val = tl.load(mask_ptr + row_idx * mask_stride,
                       mask=row_mask, other=0.0)          # (BLOCK,)
    mask_val = mask_val[:, None]                          # (BLOCK,1)

    # Base offset for the row in the flattened (row, hidden) tensor
    row_base = row_idx * stride_row                       # (BLOCK,)

    # ------------------------------------------------------------------
    # 2) Loop over hidden dimension in tiles
    # ------------------------------------------------------------------
    for off_h in range(0, hidden_dim, BLOCK_H):
        col_idx = tl.arange(0, BLOCK_H) + off_h
        col_mask = col_idx < hidden_dim

        # 2‑D offsets: (row, col)
        offs = row_base[:, None] + col_idx[None, :]  # (BLOCK, BLOCK_H)

        # Load the four input tensors
        left      = tl.load(left_ptr      + offs,
                            mask=row_mask[:, None] & col_mask[None, :],
                            other=0.0)
        left_gate = tl.load(left_gate_ptr + offs,
                            mask=row_mask[:, None] & col_mask[None, :],
                            other=0.0)
        right     = tl.load(right_ptr     + offs,
                            mask=row_mask[:, None] & col_mask[None, :],
                            other=0.0)
        right_gate= tl.load(right_gate_ptr+ offs,
                            mask=row_mask[:, None] & col_mask[None, :],
                            other=0.0)

        # Apply mask and gates
        left_out  = left  * mask_val * left_gate
        right_out = right * mask_val * right_gate

        # Store as bfloat16
        tl.store(left_out_ptr  + offs,
                 tl.cast(left_out,  tl.bfloat16),
                 mask=row_mask[:, None] & col_mask[None, :])
        tl.store(right_out_ptr + offs,
                 tl.cast(right_out, tl.bfloat16),
                 mask=row_mask[:, None] & col_mask[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated TriMul forward pass (outgoing).

    Args:
        data: Tuple (input, mask, weights, config)
            - input: Tensor[B, N, N, dim] (float32)
            - mask : Tensor[B, N, N]   (0/1, float‑compatible)
            - weights: dict of model weights (float32)
            - config: dict with "dim" and "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32) – the module output.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 0) Load weights
    # ------------------------------------------------------------------
    norm_w       = weights["norm.weight"]
    norm_b       = weights["norm.bias"]
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H) float32
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Gates
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4) Fuse mask & gates via Triton (produces bfloat16 tensors)
    # ------------------------------------------------------------------
    total_rows = B * N * N

    # flatten to (total_rows, H) – keep contiguous layout
    left_flat       = left.reshape(total_rows, H).contiguous()
    right_flat      = right.reshape(total_rows, H).contiguous()
    left_gate_flat  = left_gate.reshape(total_rows, H).contiguous()
    right_gate_flat = right_gate.reshape(total_rows, H).contiguous()
    mask_flat       = mask.reshape(total_rows).contiguous().to(x.dtype)

    # allocate output buffers (bfloat16)
    left_out_flat  = torch.empty_like(left_flat, dtype=torch.bfloat16, device=device)
    right_out_flat = torch.empty_like(right_flat, dtype=torch.bfloat16, device=device)

    # launch Triton kernel
    BLOCK    = 128   # rows per program
    BLOCK_H  = 32    # hidden‑dim tile size
    grid = (triton.cdiv(total_rows, BLOCK),)
    mask_gate_fuse_kernel[grid](
        left_flat, left_gate_flat,
        right_flat, right_gate_flat,
        mask_flat,
        left_out_flat, right_out_flat,
        total_rows, H,
        left_flat.stride(0), left_flat.stride(1),   # stride_row, stride_col (col‑stride == 1)
        mask_flat.stride(0),
        BLOCK=BLOCK, BLOCK_H=BLOCK_H
    )

    # free the original large tensors early
    del left, right, left_gate, right_gate

    # ------------------------------------------------------------------
    # 5) Batched matrix multiplication: Σ_k left * right  (per hidden dim)
    # ------------------------------------------------------------------
    left_out  = left_out_flat.view(B, N, N, H)               # (B,N,N,H) bfloat16
    right_out = right_out_flat.view(B, N, N, H)

    # reshape to (B*H, N, N) and cast to float16 for cuBLAS
    left_mat  = left_out.permute(0, 3, 1, 2).contiguous().reshape(B * H, N, N).to(torch.float16)
    right_mat = right_out.permute(0, 3, 1, 2).contiguous().reshape(B * H, N, N).to(torch.float16)

    # Batched GEMM: L @ Rᵀ  (each GEMM produces (N,N) for a hidden channel)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))           # (B*H, N, N), float16

    # Restore shape (B,N,N,H) and float32 dtype
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm + out gate
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)          # (B,N,N,dim)

    return out
