@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    dtype: tl.constexpr,
):
    # M = N = K = seq_len
    batch_idx = tl.program_id(2)  # Batch*hidden_dim dimension
    block_i = tl.program_id(0)
    block_j = tl.program_id(1)

    # Compute the start indices of this block
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute masks for out-of-bounds
    mask_i = i < M
    mask_j = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # Actually compute in float32

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k start offsets
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < K

        # Load A block: [block_i, off_k]
        a_ptrs = left_ptr + batch_idx * stride_left_batch + i[:, None] * stride_left_m + off_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B_T block: B_T is right matrix with rows = k, cols = j
        b_ptrs = right_ptr + batch_idx * stride_right_batch + off_k[:, None] * stride_right_k + j[None, :] * stride_right_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Dot product dimension K: a (M,K), b (K,N) -> (M,N)
        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_ptr + batch_idx * stride_out_batch + i[:, None] * stride_out_m + j[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(dtype), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Cast input to fp16
    x = input_tensor.to(dtype)

    # LayerNorm manually using torch.nn.functional.layer_norm
    # Provide weight and bias in fp16
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear project left/right
    left_proj_weight = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)

    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask: Ensure mask dtype matches
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)

    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Reorder to (B, hidden_dim, N, N)
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    # Transpose right to get (B, hidden_dim, N, N) with rows=k, cols=j
    right_T = right.permute(0, 1, 3, 2).contiguous()  # (B, hidden_dim, N, N)

    # Flatten batch and hidden_dim
    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right_T.view(B * hidden_dim, N, N)

    # Prepare output buffer
    out_flat = torch.empty_like(left_flat)  # same dtype

    # Get strides in elements
    stride_left_batch, stride_left_m, stride_left_k = left_flat.stride()
    stride_right_batch, stride_right_k, stride_right_n = right_flat.stride()
    stride_out_batch, stride_out_m, stride_out_n = out_flat.stride()

    # Kernel launch
    import triton
    import triton.language as tl

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_left_batch,
        stride_left_m,
        stride_left_k,
        stride_right_batch,
        stride_right_k,
        stride_right_n,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        dtype=tl.float16,
    )

    # Reshape out_flat back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate (already computed)
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"].to(dtype)
    output = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    # Cast to float32 as required by spec
    return output.to(torch.float32)

    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    # Coordinates of the block
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks for out-of-bound rows/cols
    mask_i = i < M
    mask_j = j < N
    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A block (M x K)
        a_ptrs = left_ptr + pid_batch * stride_l_batch + i[:, None] * stride_l_m + k_offset[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (K x N) – note that right matrix is already transposed (rows=k, cols=j)
        b_ptrs = right_ptr + pid_batch * stride_r_batch + k_offset[:, None] * stride_r_k + j[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result, casting back to FP16
    out_ptrs = out_ptr + pid_batch * stride_o_batch + i[:, None] * stride_o_m + j[None, :] * stride_o_n

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy part of TriMul is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    (left_reshape) @ (right_reshape).T   per hidden channel.
We implement this contraction with a fused Triton GEMM that operates on
half‑precision (fp16) tensors while accumulating in fp32 for numerical
stability.  All other operations (LayerNorm, projections, gates) are
performed with PyTorch, keeping the implementation simple and correct.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,                     # all equal to seq_len
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ right.T for each batch*channel slice."""
    pid_batch = tl.program_id(2)          # which (batch * hidden_dim) we are processing
    pid_i = tl.program_id(0)              # block row index
    pid_j = tl.program_id(1)              # block column index

    # Block coordinates
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < M
    mask_j = j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < K

        # Load a tile of left  [i, k]
        a_ptrs = left_ptr + pid_batch * stride_l_batch \
                 + i[:, None] * stride_l_m + k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load a tile of right_T  [k, j]
        b_ptrs = right_ptr + pid_batch * stride_r_batch \
                 + k[:, None] * stride_r_k + j[None, :] * stride_r_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Matrix multiply the tiles
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Store the result (fp32)
    out_ptrs = out_ptr + pid_batch * stride_o_batch \
               + i[:, None] * stride_o_m + j[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (float, 0/1)
            - weights: dict of model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        output: torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Use fp16 for the large tensors; accumulate in fp32 inside the Triton kernel.
    hp = torch.float16

    # --------------------------------------------------------------------- #
    # 1) Initial LayerNorm (dim -> dim)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"].to(hp)
    norm_bias = weights["norm.bias"].to(hp)
    x = input_tensor.to(hp)
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # 2) Linear projections to hidden_dim
    # --------------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"].to(hp)
    right_proj_w = weights["right_proj.weight"].to(hp)

    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------------- #
    # 3) Apply mask (if provided) – broadcast over hidden_dim
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask_fp = mask.to(hp).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------------- #
    # 4) Gating (sigmoid of linear transforms)
    # --------------------------------------------------------------------- #
    left_gate_w = weights["left_gate.weight"].to(hp)
    right_gate_w = weights["right_gate.weight"].to(hp)
    out_gate_w = weights["out_gate.weight"].to(hp)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5) Core contraction using Triton (batched GEMM per hidden channel)
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape

    # reshape to (B, hidden_dim, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # right must be transposed on the last two dimensions: (k, j)
    right_T = right.permute(0, 1, 3, 2).contiguous()

    # flatten batch and hidden_dim -> batch_h = B * hidden_dim
    left_flat = left.view(B * hidden_dim, N, N)          # fp16
    right_flat = right_T.view(B * hidden_dim, N, N)      # fp16
    out_flat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # strides in element counts
    stride_l_batch, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    # Triton block size – works well for the target sequence lengths
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # restore original layout: (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # --------------------------------------------------------------------- #
    # 6) Output LayerNorm (hidden_dim) + gating + final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate  # out_gate (fp16) is broadcasted to fp32 automatically

    to_out_w = weights["to_out.weight"].to(torch.float32)
    output = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim], fp32

    return output
