    device = input_tensor.device
    # get dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # weights are tensors, assume correct dtype (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    eps = 1e-5

    # LayerNorm on last dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gating linear
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    right_gate_fp16 = right_gate.to(torch.float16)
    out_gate_fp16 = out_gate  # keep fp32 for later, maybe cast later

    if mask is None:

    else:
        mask_tensor = mask.to(torch.float16)
    mask_expanded = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1], stride zero for hidden dim

    right_out = torch.empty_like(right_fp16)

"""
TriMul (outgoing) implementation with a Triton kernel.

The heavy 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is equivalent to a batch of matrix multiplications
    out_d[b] = left_d[b] @ right_d[b].T
for every hidden channel d.
We fuse the mask application and the left/right gating (mask * gate) in a
single Triton kernel, cast the tensors to fp16 for the O(N³) GEMM, and then
use a fast PyTorch batched matmul.  The remaining LayerNorms and linear
projections are performed in FP32 for numerical stability.

The kernel `mask_gate_kernel` computes:
    left_out  = left  * mask * left_gate
    right_out = right * mask * right_gate
using a 1‑D launch that processes `BLOCK` elements per program.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: fuse mask multiplication and gating for left/right tensors.
# ----------------------------------------------------------------------------- #
@triton.jit
def mask_gate_kernel(
    left_ptr,          # fp16 * left tensor (B,N,N,H)
    left_gate_ptr,     # fp16 * left gate (B,N,N,H)
    mask_ptr,          # fp16 * mask broadcast (B,N,N,1) – stride 0 on H
    right_ptr,         # fp16 * right tensor
    right_gate_ptr,    # fp16 * right gate
    left_out_ptr,      # fp16 * output left
    right_out_ptr,     # fp16 * output right
    numel,             # total number of elements (B*N*N*H)
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # Load values (default other=0.0 is fine for masked out elements)
    l   = tl.load(left_ptr      + offs, mask=mask)
    lg  = tl.load(left_gate_ptr + offs, mask=mask)
    m   = tl.load(mask_ptr      + offs, mask=mask)
    r   = tl.load(right_ptr     + offs, mask=mask)
    rg  = tl.load(right_gate_ptr+ offs, mask=mask)

    l_out = l * m * lg
    r_out = r * m * rg

    tl.store(left_out_ptr  + offs, l_out, mask=mask)
    tl.store(right_out_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------------- #
# Public entry point used by the evaluation harness.
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of torch.Tensors with the model parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype=torch.float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # Extract configuration and weights.
    # --------------------------------------------------------------------- #
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    norm_weight           = weights["norm.weight"]
    norm_bias             = weights["norm.bias"]
    left_proj_weight      = weights["left_proj.weight"]
    right_proj_weight     = weights["right_proj.weight"]
    left_gate_weight      = weights["left_gate.weight"]
    right_gate_weight     = weights["right_gate.weight"]
    out_gate_weight       = weights["out_gate.weight"]
    to_out_norm_weight    = weights["to_out_norm.weight"]
    to_out_norm_bias      = weights["to_out_norm.bias"]
    to_out_weight         = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1. LayerNorm on the input (still in FP32 for stability).
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # 2. Cast to FP16 and perform linear projections & gating in FP16.
    # --------------------------------------------------------------------- #
    x_fp16 = x.to(torch.float16)

    # Linear weights to half precision
    left_proj_w_fp16   = left_proj_weight.to(torch.float16)
    right_proj_w_fp16  = right_proj_weight.to(torch.float16)
    left_gate_w_fp16   = left_gate_weight.to(torch.float16)
    right_gate_w_fp16  = right_gate_weight.to(torch.float16)
    out_gate_w_fp16    = out_gate_weight.to(torch.float16)

    # Projections
    left  = torch.nn.functional.linear(x_fp16, left_proj_w_fp16)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, right_proj_w_fp16)

    # Gates (sigmoid)
    left_gate  = torch.nn.functional.linear(x_fp16, left_gate_w_fp16).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w_fp16).sigmoid()
    out_gate   = torch.nn.functional.linear(x_fp16, out_gate_w_fp16).sigmoid()  # FP16

    # --------------------------------------------------------------------- #
    # 3. Fuse mask, left and right gating with a Triton kernel.
    # --------------------------------------------------------------------- #
    if mask is None:
        mask_tensor = torch.ones(input_tensor.shape[:3], device=device, dtype=torch.float16)
    else:
        mask_tensor = mask.to(torch.float16)

    mask_expanded = mask_tensor.unsqueeze(-1)                # [B,N,N,1]; stride on H = 0

    # Allocate output buffers
    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Launch Triton kernel
    BLOCK = 16384   # reasonable block size (multiple of 32, fits in registers)
    numel = left.numel()
    grid = (triton.cdiv(numel, BLOCK),)
    mask_gate_kernel[grid](
        left, left_gate, mask_expanded,
        right, right_gate,
        left_out, right_out,
        numel,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------------- #
    # 4. Compute the O(N³) reduction as a batched matrix multiplication.
    #    left_out/right_out are [B,N,N,hidden] -> we transpose to
    #    [B,hidden,N,N] for a simple matmul.
    # --------------------------------------------------------------------- #
    left_t  = left_out.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right_out.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Batched matmul: each hidden channel multiplies left @ rightᵀ
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))  # [B, H, N, N]

    # Restore original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1).to(torch.float32)      # back to FP32 for norm

    # --------------------------------------------------------------------- #
    # 5. Post‑projection LayerNorm, gating, and final linear.
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # out_gate is still FP16 – convert to FP32 before multiplication
    out_gate_fp32 = out_gate.to(torch.float32)
    out = out * out_gate_fp32

    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
