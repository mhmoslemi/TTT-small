@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # compute strides for mask and H addition:
    # compute offset for linear index
    # offset_left = compute based on strides
    # Compute b,i,k,h similar but we can compute directly using the strides:
    # offset = b*stride_left_b + i*stride_left_i + k*stride_left_k + h*stride_left_h.

    # compute i,k,h using integer divisions:
    # total_per_batch = N * N * H
    # b = offs // total_per_batch
    # rem1 = offs - b * total_per_batch
    # i = rem1 // (N * H)
    # rem2 = rem1 - i * (N * H)
    # k = rem2 // H
    # h = rem2 - k * H

    total_per_batch = N * N * H
    b = offs // total_per_batch
    rem1 = offs - b * total_per_batch
    i = rem1 // (N * H)
    rem2 = rem1 - i * (N * H)
    k = rem2 // H
    h = rem2 - k * H

    # compute offsets
    left_offset = b * stride_left_b + i * stride_left_i + k * stride_left_k + h * stride_left_h
    right_offset = b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    left_gate_offset = b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k + h * stride_lgate_h
    right_gate_offset = b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + h * stride_rgate_h

    mask_offset = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # Load values
    left_val = tl.load(left_ptr + left_offset, mask=mask)
    right_val = tl.load(right_ptr + right_offset, mask=mask)
    left_gate_val = tl.load(left_gate_ptr + left_gate_offset, mask=mask)
    right_gate_val = tl.load(right_gate_ptr + right_gate_offset, mask=mask)
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask)
    # Multiply
    left_out = left_val * mask_val * left_gate_val
    right_out = right_val * mask_val * right_gate_val

    # Store results back
    tl.store(left_ptr + left_offset, left_out, mask=mask)
    tl.store(right_ptr + right_offset, right_out, mask=mask)

"""
TriMul (outgoing) forward pass with a fused mask‑gate Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections for left/right and gate values.
3. Fuse mask, left‑gate, and right‑gate via a custom Triton kernel
   (in‑place update of the left/right tensors).
4. Compute the rank‑3 contraction
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a batched GEMM (torch.bmm) on the (batch*hidden) dimension.
5. Apply a second layer‑norm, multiply by the output‑gate and a final linear
   projection back to the original dimension.

Only the elementwise mask‑gate fusion is performed in Triton; the heavy
matrix multiplication uses PyTorch’s highly‑optimized cuBLAS kernel.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H,
    # strides for left/right tensors (batch, i, k, hidden)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for mask (batch, i, k)
    stride_mask_b, stride_mask_i, stride_mask_k,
    # strides for gate tensors (same as left/right)
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    m = offs < total

    # Decompose linear index into (b,i,k,h)
    total_per_batch = N * N * H
    b = offs // total_per_batch
    rem1 = offs - b * total_per_batch
    i = rem1 // (N * H)
    rem2 = rem1 - i * (N * H)
    k = rem2 // H
    h = rem2 - k * H

    # Compute flat offsets for each tensor
    left_offset = b * stride_left_b + i * stride_left_i + k * stride_left_k + h * stride_left_h
    right_offset = b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    lgate_offset = b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k + h * stride_lgate_h
    rgate_offset = b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + h * stride_rgate_h
    mask_offset = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # Load values (masked loads return 0.0 for inactive threads)
    left_val = tl.load(left_ptr + left_offset, mask=m, other=0.0)
    right_val = tl.load(right_ptr + right_offset, mask=m, other=0.0)
    lgate_val = tl.load(left_gate_ptr + lgate_offset, mask=m, other=0.0)
    rgate_val = tl.load(right_gate_ptr + rgate_offset, mask=m, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offset, mask=m, other=0.0)

    # Apply mask and gates
    left_out = left_val * mask_val * lgate_val
    right_out = right_val * mask_val * rgate_val

    # Store results back in‑place
    tl.store(left_ptr + left_offset, left_out, mask=m)
    tl.store(right_ptr + right_offset, right_out, mask=m)


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) implementation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [bs, N, N, dim]
            - mask_tensor: torch.Tensor [bs, N, N] (float, 0/1)
            - weights_dict: mapping str → torch.Tensor containing all model weights
            - config_dict: contains at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [bs, N, N, dim]
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Configuration & shortcuts
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask & gates with a Triton kernel (in‑place)
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for predictable strides
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.to(dtype=torch.float32).contiguous()

    B, N, _, _ = left.shape  # left.shape = (B, N, N, hidden_dim)
    total_elements = B * N * N * hidden_dim
    BLOCK = 1024  # must be a multiple of 32
    grid = (triton.cdiv(total_elements, BLOCK),)

    # Strides (in elements) for each tensor
    sl_b, sl_i, sl_k, sl_h = left.stride()
    sr_b, sr_i, sr_k, sr_h = right.stride()
    sm_b, sm_i, sm_k = mask.stride()
    sg_l_b, sg_l_i, sg_l_k, sg_l_h = left_gate.stride()
    sg_r_b, sg_r_i, sg_r_k, sg_r_h = right_gate.stride()

    _mask_gate_kernel[grid](
        left, right, mask,
        left_gate, right_gate,
        B, N, hidden_dim,
        sl_b, sl_i, sl_k, sl_h,
        sr_b, sr_i, sr_k, sr_h,
        sm_b, sm_i, sm_k,
        sg_l_b, sg_l_i, sg_l_k, sg_l_h,
        sg_r_b, sg_r_i, sg_r_k, sg_r_h,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Rank‑3 contraction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Reshape to (B*hidden_dim, N, N) for a batch of GEMMs.
    # left: (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # right needs to be transposed on the reduction axis (k)
    right_flat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Second layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
