    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    pid_j = tl.program_id(1)
    # decode pid into batch, hidden, i_tile
    grid_i = tl.cdiv(N, BLOCK_M)
    bh = pid // grid_i  # batch * hidden index
    block_i = pid % grid_i
    b = bh // hidden
    h = bh % hidden
    # compute offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    i_valid = offs_i < N
    j_valid = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = offs_k < N
        # compute pointers
        a_ptrs = left_ptr + b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h * stride_left_d
        b_ptrs = right_ptr + b * stride_right_b + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k + h * stride_right_d

        mask_a = i_valid[:, None] & k_valid[None, :]
        mask_b = j_valid[:, None] & k_valid[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float16)

    # store

    mask_c = i_valid[:, None] & j_valid[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)

"""
TriMul forward implementation with a Triton kernel for the core
pairwise multiplication (einsum "... i k d, ... j k d -> ... i j d").
The kernel computes, for each batch and hidden channel, the matrix
product  L[:, :, d] @ R[:, :, d].T  where L and R have shape (N, N).
All surrounding operations (LayerNorm, Linear projections, gating,
post‑norm and final projection) are performed with PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (B,N,N,H)
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out (B,N,N,H)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled GEMM per hidden channel.
    """
    pid = tl.program_id(0)      # combined batch*hidden*tile_i
    pid_j = tl.program_id(1)    # tile_j

    # number of i‑tiles
    num_tiles_i = tl.cdiv(N, BLOCK_M)

    # decode batch, hidden, i‑tile from pid
    bh = pid // num_tiles_i          # (batch * hidden) index
    i_tile = pid % num_tiles_i
    b = bh // H                      # batch index
    h = bh % H                       # hidden channel index

    # i, j offsets for this tile
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = offs_i < N
    j_valid = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_tiles_k):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = offs_k < N

        # pointers to blocks of A = left and B = right^T
        # A: shape (BLOCK_M, BLOCK_K)   -> left[b, i, k, h]
        a_ptrs = (left_ptr
                  + b * stride_left_b
                  + offs_i[:, None] * stride_left_i
                  + offs_k[None, :] * stride_left_k
                  + h * stride_left_h)

        # B: shape (BLOCK_N, BLOCK_K)   -> right[b, j, k, h]
        b_ptrs = (right_ptr
                  + b * stride_right_b
                  + offs_j[:, None] * stride_right_i
                  + offs_k[None, :] * stride_right_k
                  + h * stride_right_h)

        mask_a = i_valid[:, None] & k_valid[None, :]
        mask_b = j_valid[:, None] & k_valid[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # tl.dot broadcasts the leading dimensions and reduces over the last (K)
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write the block to the output tensor
    c_ptrs = (out_ptr
              + b * stride_out_b
              + offs_i[:, None] * stride_out_i
              + offs_j[None, :] * stride_out_j
              + h * stride_out_h)

    mask_c = i_valid[:, None] & j_valid[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    TriMul forward (outgoing) using a Triton kernel for the O(N^3) core.
    Args:
        data: (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm on input ----------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ---------- Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the kernel ----------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ---------- Allocate output ----------
    out = torch.empty((bs, N, N, hidden), dtype=torch.float32, device=device)

    # ---------- Strides (in elements) ----------
    s_left = left_h.stride()    # (B, N, N, H)
    s_right = right_h.stride()
    s_out = out.stride()

    # ---------- Triton launch parameters ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    total_bh = bs * hidden
    grid = (total_bh * grid_i, grid_j)

    # ---------- Launch kernel ----------
    _tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        bs,
        N,
        hidden,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑processing ----------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate  # gating after norm

    final = torch.nn.functional.linear(
        out_norm, weights["to_out.weight"]
    )  # [B, N, N, dim]

    return final
