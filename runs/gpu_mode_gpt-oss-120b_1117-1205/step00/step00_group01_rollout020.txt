@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_i, stride_r_k,
    stride_m_b, stride_m_i, stride_m_j,
    stride_o_b, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    # first program id combines batch and hidden dimension
    pid = tl.program_id(0)
    b = pid // H
    d = pid % H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    i_offset = pid_i * BLOCK_I
    j_offset = pid_j * BLOCK_J

    # create range vectors
    offs_i = i_offset + tl.arange(0, BLOCK_I)
    offs_j = j_offset + tl.arange(0, BLOCK_J)
    # compute mask for i and j within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (i,k)
        left_ptrs = left_ptr + b * stride_l_b + d * 1 + \
            offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        # mask for left * mask
        mask_left_ptrs = mask_ptr + b * stride_m_b + offs_i[:, None] * stride_m_i + offs_k[None, :] * stride_m_j
        mask_left = tl.load(mask_left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left * mask_left.to(tl.float16)

        # Load right tile (j,k) but need shape (BLOCK_K, BLOCK_J)
        # compute pointers with k as first dimension
        right_ptrs = right_ptr + b * stride_r_b + d * 1 + \
            offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)
        mask_right_ptrs = mask_ptr + b * stride_m_b + offs_j[:, None] * stride_m_i + offs_k[None, :] * stride_m_j # careful: mask shape b,i,j? Actually mask stores [b,i,j], for right we need mask[b,j,k] -> dims (i,j) same as left and right but swapped; So mask_right_ptrs = mask_ptr + b * stride_m_b + offs_j[:, None] * stride_m_i + offs_k[None, :] * stride_m_j
        mask_right = tl.load(mask_right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        right = right * mask_right.to(tl.float16)

        # Compute dot product
        acc += tl.dot(left, right)   # shape (BLOCK_I, BLOCK_J)
    # after loop, store
    out_ptrs = out_ptr + b * stride_o_b + d * 1 + \
        offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, N2, dim = input_tensor.shape
    assert N == N2
    hidden_dim = config['hidden_dim']
    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # linear projections
    left_proj = torch.nn.functional.linear(x, weights['left_proj.weight'])  # shape [bs,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, weights['right_proj.weight'])
    # mask
    mask_unsq = mask.unsqueeze(-1).to(x.dtype)

    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    left = left.to(torch.float16)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_i, stride_r_k,
    stride_o_b, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    b = pid // H
    d = pid % H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over K dimension (the shared dimension)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + b * stride_l_b + d + \
            offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # Load right tile: shape (BLOCK_K, BLOCK_J)
        right_ptrs = right_ptr + b * stride_r_b + d + \
            offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # Multiply‑accumulate
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + b * stride_o_b + d + \
        offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module (outgoing version).
    The expensive triple‑sum `einsum('... i k d, ... j k d -> ... i j d')`
    is implemented as a tiled batched GEMM in a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square in the last two dimensions"
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on the channel dimension -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device)
    )

    # ----- Linear projections (no bias) -----
    left = F.linear(
        x,
        weights["left_proj.weight"].to(device)
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(device)
    )

    # ----- Apply mask (if any) -----
    mask_ = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_
    right = right * mask_

    # ----- Gating logits -----
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(device)
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(device)
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(device)
    ).sigmoid()

    # ----- Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # ----- Cast to half‑precision for the heavy kernel -----
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ----- Allocate output buffer for the triple‑sum -----
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ----- Kernel launch configuration -----
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # Strides (in elements) for the tensors
    stride_l_b, stride_l_i, stride_l_k, _ = left.stride()
    stride_r_b, stride_r_i, stride_r_k, _ = right.stride()
    stride_o_b, stride_o_i, stride_o_j, _ = out.stride()

    # Grid: (batch * hidden_dim, N/BI, N/BJ)
    grid = (bs * hidden_dim,
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J))

    trimul_kernel[grid](
        left,
        right,
        out,
        bs,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing (LayerNorm, gate, final projection) -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device)
    )

    )

"""
TriMul (outgoing version) with Triton acceleration.

The heavy part of the forward pass is the triple sum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel `d` this is a matrix multiplication
    left_d (N×N) @ right_dᵀ (N×N) → out_d (N×N).

The kernel tiles the (i, j) output plane and iterates over the shared
k‑dimension, using fp16 for the multiplication while accumulating in fp32.
All other operations (LayerNorm, linear projections, gating) are performed
with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_i, stride_r_k,
    stride_o_b, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the triple‑sum, one hidden channel per program."""
    pid = tl.program_id(0)
    b = pid // H                # batch index
    d = pid % H                 # hidden‑channel index

    pid_i = tl.program_id(1)    # tile over i
    pid_j = tl.program_id(2)    # tile over j

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # range vectors for the current tile
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # iterate over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + b * stride_l_b + d + \
            offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # right tile (transposed): shape (BLOCK_K, BLOCK_J)
        right_ptrs = right_ptr + b * stride_r_b + d + \
            offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # accumulate the product
        acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_J)

    # write the computed block
    out_ptrs = out_ptr + b * stride_o_b + d + \
        offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

            input   – Tensor [B, N, N, dim]
            mask    – Tensor [B, N, N] (binary or all‑ones)
            weights – dict with all linear/LayerNorm parameters
            config  – dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "pairwise tensor must be square"
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"].to(device))
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ------------------------------------------------------------------
    # 3) Mask (broadcast on the hidden dimension)
    mask_ = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_
    right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating logits
    left_gate = F.linear(x, weights["left_gate.weight"].to(device)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(device)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(device)).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to half‑precision for the kernel
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 7) Allocate output for the triple‑sum
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 8) Triton kernel launch
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, _ = left.stride()
    stride_r_b, stride_r_i, stride_r_k, _ = right.stride()
    stride_o_b, stride_o_i, stride_o_j, _ = out.stride()

    grid = (
        B * hidden_dim,                 # one program per (batch, hidden) pair
        triton.cdiv(N, BLOCK_I),       # i‑tiles
        triton.cdiv(N, BLOCK_J),       # j‑tiles
    )
    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9) Post‑processing: LayerNorm → out‑gate → final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"].to(device))

    return output
