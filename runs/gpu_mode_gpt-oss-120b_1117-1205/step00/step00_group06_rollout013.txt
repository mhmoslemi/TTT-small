@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # block row index (i)
    pid_n = tl.program_id(1)   # block col index (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # compute actual batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # Compute offsets for first element of block
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Compute base offsets for each tile
    # left base: pointer for this batch/hidden; add i*stride_i + k*stride_k later
    left_batch_offset = b * stride_left_batch + h * stride_left_h

    # right base offset
    right_batch_offset = b * stride_right_batch + h * stride_right_h

    # out base offset
    out_batch_offset = b * stride_out_batch + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute start K index
        k_start = k * BLOCK_K

        # compute offsets for left block: shape (BLOCK_M, BLOCK_K)
        # left[i, k]
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        # mask for valid i/k
        mask_i = offs_i < N
        mask_k = offs_k < N

        # compute left pointer addresses
        left_ptrs = left_ptr + left_batch_offset \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k

        # load left tile
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # compute offsets for right block (transposed)
        # right[k, j] (need to compute from original arrays)
        offs_j = j_start + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N

        # right tile shape (BLOCK_K, BLOCK_N): k as rows, j as cols.
        right_ptrs = right_ptr + right_batch_offset \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j

        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # compute dot product accumulation
        # acc += tl.dot(left, right)  # but tl.dot expects (M,K) and (K,N) as lengths? Actually tl.dot does reduce on last dimension: expects two tensors which broadcast to (M, K) and (K, N)? Need to check: In triton, tl.dot(a,b) does dot product of the last dimension of a with last dimension of b, broadcasting other dims? The docs: tl.dot(A, B) computes sum over the inner dimension k: A[i][k] * B[k][j] -> shape (i,j). Works if shapes are (M,K) and (K,N). So we can do: acc += tl.dot(left, right). Yes.

        acc += tl.dot(left, right)

    # after loop, store to out with mask
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    out_ptrs = out_ptr + out_batch_offset \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input, mask, weights, config = data

    # dims
    B, N, _, C = input.shape
    hidden = config["hidden_dim"]
    dim = config["dim"]  # dim equals C ?

    # Perform layernorm on input: using torch.nn.functional.layer_norm
    eps = 1e-5  # default in nn
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # shape (C,) weight, bias

    # input normalized: shape (B,N,N,C)
    x = torch.nn.functional.layer_norm(input, (C,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections: linear with weight shape (hidden, C) for left_proj, right_proj
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden, C)
    right_proj_weight = weights["right_proj.weight"]
    # Linear w/o bias: y = x @ W^T (since Linear weight shape (out_features, in_features) and forward: y = x @ weight.T + bias). So we can compute left = torch.nn.functional.linear(x, left_proj_weight) (since bias = None). Similarly right.

    left = torch.nn.functional.linear(x, left_proj_weight)  # shape: (B,N,N, hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    if mask is not None:
        # mask shape: (B,N,N)
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gating weights: left_gate_weight shape (hidden, C)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # compute gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Now compute core multiplication using Triton kernel
    # allocate output
    out = torch.empty((B, N, N, hidden), dtype=left.dtype, device=left.device)

    # compute strides
    stride_left_batch = left.stride(0)

    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # i block index
    pid_n = tl.program_id(1)   # j block index
    pid_z = tl.program_id(2)   # combined batch*hidden index

    # recover batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # start indices of this tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # compute base offsets for this (b, h)
    left_base = b * stride_left_batch + h * stride_left_h
    right_base = b * stride_right_batch + h * stride_right_h
    out_base = b * stride_out_batch + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension K (= N)
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_start = kk * BLOCK_K

        # indices for A tile
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_k = k_start + tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_k = offs_k < N

        a_ptrs = left_ptr + left_base \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # indices for B_T tile (original right with permuted dims)
        offs_j = j_start + tl.arange(0, BLOCK_N)

        mask_j = offs_j < N

        b_ptrs = right_ptr + right_base \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_j

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # write back results
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    c_ptrs = out_ptr + out_base \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j

    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul implementation.
    Computes a fused layernorm + linear projections + gating, then performs
    the cubic pairwise multiplication using a Triton GEMM kernel.
    Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm over the channel dimension ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # (B, N, N, hidden)
    right = F.linear(x, right_proj_weight)

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core cubic update (Triton) ----------
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=dtype, device=device)

    # Compute strides (in elements)
    stride_left_batch = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_batch = right.stride(0)
    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Tile sizes (tuned for B200 memory hierarchy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(seq_len, BLOCK_M),
            triton.cdiv(seq_len, BLOCK_N),
            bs * hidden_dim)

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        bs, seq_len, hidden_dim,
        stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
        stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ---------- Post‑processing ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)

    out = out * out_gate  # element‑wise gating

    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return output

"""Outgoing TriMul implementation.
Computes LayerNorm + linear projections + gating in PyTorch and
uses a Triton GEMM‑style kernel for the cubic pairwise multiplication:
out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
The result is post‑processed with a second LayerNorm, out‑gate and
a final linear layer to recover the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched GEMM with transposed B (right) side.
# -------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                                 # batch, seq_len, hidden
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_z = tl.program_id(2)          # combined batch*hidden index

    # Recover original batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # Tile start positions
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Base pointers for this (b, h)
    left_base = b * stride_left_batch + h * stride_left_h
    right_base = b * stride_right_batch + h * stride_right_h
    out_base = b * stride_out_batch + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (the reduction axis)
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_start = kk * BLOCK_K

        # ---- Load A (left) tile ----
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_i = offs_i < N
        mask_k = offs_k < N
        a_ptrs = left_ptr + left_base \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- Load B_T (right) tile – transposed via strides ----
        offs_j = j_start + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N
        b_ptrs = right_ptr + right_base \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ---- Accumulate dot product ----
        acc += tl.dot(a, b)

    # ---- Store the result tile ----
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    c_ptrs = out_ptr + out_base \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Entry point required by the evaluator.
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, dim]
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict of parameter tensors
            config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------------------------------------------------------
    # 1. Input LayerNorm
    # ---------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=1e-5)

    # ---------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3. Apply optional mask
    # ---------------------------------------------------------
    if mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------------------------------------------------------
    # 4. Gating
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5. Core cubic update via Triton
    # ---------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=dtype, device=device)

    # Compute strides (in elements)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_j, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * hidden)

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden,
        sL_b, sL_i, sL_k, sL_h,
        sR_b, sR_j, sR_k, sR_h,
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------
    # 6. Post‑processing (LayerNorm, out‑gate, final linear)
    # ---------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
