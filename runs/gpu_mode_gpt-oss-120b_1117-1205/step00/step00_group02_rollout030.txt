@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr
):
    # group dimension is flattened batch*head; each group is processed by different blocks along program ID dimensions: pid0 for group, pid1 for block row, pid2 for block col.
    
    # Compute program ids
    pid_group = tl.program_id(0)   # group index (G)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for within M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A and B tiles
        a_ptrs = a_ptr + pid_group * stride_a_batch + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        b_ptrs = b_ptr + pid_group * stride_b_batch + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute matmul
        acc += tl.dot(a, b)

        # pointer arithmetic to next K block.
        # Since we compute a_ptrs and b_ptrs using offsets, no need to increment pointer after loop.

    # Write back result
    c_ptrs = c_ptr + pid_group * stride_c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated TriMul^{outgoing} forward pass.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layer norm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])
    # Apply mask if not nomask
    if not config.get("nomask", False):
        mask_unsq = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_unsq
        right = right * mask_unsq
    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange for batched matmul (A: [B,H,N,N]), (B: [B,H,N,N] with swapped axes)
    # A: permute to (B, H, N, N)
    A = left_h.permute(0, 3, 1, 2).contiguous()
    B = right_h.permute(0, 3, 2, 1).contiguous()
    G = B.shape[0] * B.shape[1]  # B*H

    # reshape to (G, N, N)
    A_mat = A.view(-1, N, N)
    B_mat = B.view(-1, N, N)
    # allocate output
    out_mat = torch.empty_like(A_mat, dtype=torch.float32)  # shape (G,N,N)
    # Launch kernel
    M = N
    K = N
    # Strides
    stride_a_batch, stride_am, stride_ak = A_mat.stride()
    stride_b_batch, stride_bk, stride_bn = B_mat.stride()
    stride_c_batch, stride_cm, stride_cn = out_mat.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (A_mat.shape[0], triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        A_mat,
        B_mat,
        out_mat,
        M,
        N,
        K,
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # to_out_norm

                       weight=weights["to_out_norm.weight"],

    out = F.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) forward pass with a Triton-accelerated batched matrix multiplication.

The core operation is
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

After the projections, gating and optional masking, this reduces to a batch of
matrix multiplications: for each (b, d) pair compute
    out_{b,d} = left_{b,d} @ right_{b,d}.T
where left_{b,d} and right_{b,d} are N×N matrices.  The kernel fuses the
inner‑product loop over k and provides an FP16 compute path while accumulating
in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                                 # matrix sizes (M×K)·(K×N)→(M×N)
    stride_a_batch, stride_am, stride_ak,    # strides for A (batch, row, col)
    stride_b_batch, stride_bk, stride_bn,    # strides for B (batch, row, col)
    stride_c_batch, stride_cm, stride_cn,    # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: [G,M,K], B: [G,K,N], C: [G,M,N])"""

    # program IDs
    pid_batch = tl.program_id(0)   # which (batch * hidden) group
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for edge handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this group
    a_base = a_ptr + pid_batch * stride_a_batch
    b_base = b_ptr + pid_batch * stride_b_batch
    c_base = c_ptr + pid_batch * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current tile
        a_tile_ptr = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_tile_ptr = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # load tiles (FP16) with masking
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # fused dot-product; result accumulated in FP32
        acc += tl.dot(a, b)

    # write back the result tile
    c_tile_ptr = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim] (torch.float32)
            - mask_tensor  : [B, N, N] (torch.float32 or bool)
            - weights_dict : mapping of weight names to torch tensors
            - config_dict  : contains at least "dim", "hidden_dim", "nomask"

    Returns:
        output_tensor : [B, N, N, dim] (torch.float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM:   A @ Bᵀ
    #    A : left  -> (B, hidden, N, N)
    #    B : right -> (B, hidden, N, N)   where the two inner N dimensions are swapped
    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy computation
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # bring hidden_dim to the second axis and make tensors contiguous
    A = left_fp16.permute(0, 3, 1, 2).contiguous()          # [B, hidden, N, N]
    B = right_fp16.permute(0, 3, 2, 1).contiguous()        # [B, hidden, N, N] (k,j swapped)

    # flatten the (batch, hidden) dimensions → G groups
    G = B.shape[0] * B.shape[1]  # = B * hidden_dim
    A_mat = A.view(G, N, N)      # [G, M, K] with M=K=N
    B_mat = B.view(G, N, N)      # [G, K, N]

    # output buffer (FP32 accumulation)
    out_mat = torch.empty_like(A_mat, dtype=torch.float32)

    # stride information (in elements, not bytes)
    stride_a_batch, stride_am, stride_ak = A_mat.stride()
    stride_b_batch, stride_bk, stride_bn = B_mat.stride()
    stride_c_batch, stride_cm, stride_cn = out_mat.stride()

    # ------------------------------------------------------------------
    # 6. Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        G,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A_mat,
        B_mat,
        out_mat,
        N,          # M
        N,          # N
        N,          # K
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output‑side layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
