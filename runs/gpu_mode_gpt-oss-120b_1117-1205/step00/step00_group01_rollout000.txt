def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projection left and right
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # shape: out_dim x in_dim
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    mask_unsq = mask.unsqueeze(-1)  # shape B,N,N,1
    left = left * mask_unsq
    right = right * mask_unsq

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output tensor
    out_fp16 = torch.empty((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], hidden_dim), dtype=torch.float16, device=device)

    # Launch kernel
    B, N, _, _ = left_fp16.shape  # B,N,N,hidden_dim

    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out_fp16.stride()

    BLOCK_M = 64  # i dimension tile
    BLOCK_N = 64  # j dimension tile
    BLOCK_K = 32  # reduction tile

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    # compute batch index and hidden dimension
    b = pid_bd // D
    d = pid_bd % D

    # Compute base offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Mask for boundary
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # pointers for left (i,k)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            d * stride_left_d + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k

        # pointers for right (j,k) transposed to (k,j)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            d * stride_right_d + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_j

        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product (M,K)*(K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile)

    # Store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        d * stride_out_d + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j

    mask_store = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)

"""
TriMul forward kernel (outgoing version) with Triton acceleration.

The algorithm follows the reference PyTorch implementation:
1. Layer‑norm the input on its last dimension.
2. Linear projections (left / right) and gated projections (left_gate, right_gate,
   out_gate) using the provided weights.
3. Mask the projected tensors, apply the gates, and cast to fp16.
4. Compute the core O(N³) operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication L @ Rᵀ for each hidden dimension d.
   This is implemented with a tiled Triton kernel.
5. Cast the result back to fp32, apply a second LayerNorm and the out_gate,
   then a final linear projection to recover the original dimension.

All heavy work (the triple sum) is performed in a Triton kernel.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    out_ptr,                # fp16 *
    B, N, D,                # int
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr   # tile size in reduction dimension k
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    # decode batch and hidden dimension
    b = pid_bd // D
    d = pid_bd % D

    # offsets for i and j within the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M x K)
        left_ptrs = left_ptr + \
            b * stride_left_b + d * stride_left_d + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k

        # right tile (K x N) -> we load as (K, N) directly
        right_ptrs = right_ptr + \
            b * stride_right_b + d * stride_right_d + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_j

        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile).to(tl.float32)

    # store the result (cast back to fp16)
    out_ptrs = out_ptr + \
        b * stride_out_b + d * stride_out_d + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j

    mask_store = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    core O(N³) multiplication.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict of torch.Tensors containing all linear and layer‑norm
                       parameters.
            - config: dict with keys "dim" and "hidden_dim".

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on input -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # ----- Linear projections --------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Apply mask (if any) ------------------------------------------------
    mask_unsq = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ----- Gates --------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to fp16 for the triton kernel ---------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # ----- Strides ------------------------------------------------------------
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out_fp16.stride()

    # ----- Triton launch ------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    _tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Convert back to fp32 ------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ----- Second LayerNorm, output gate and final linear ----------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
