@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_ama, stride_ak, stride_bka, stride_bnb, stride_cm, stride_cn,
    # Actually we need stride for batch: stride_a_batch, stride_b_batch, stride_c_batch
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Determine mask for valid indices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers offset for each batch
    a_ptrs = A + batch_id * stride_ab + offs_m[:, None] * stride_ama + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B + batch_id * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bka + offs_n[None, :] * stride_bnb
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # loads
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bka

    # Write back
    c_ptrs = C + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute base pointers
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for K dimension
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load a tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)

        # load b tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)

        # dot product accumulation
        acc += tl.dot(a, b)

        # note pointer increment can be done by adjusting offset each iteration (above uses offs_k each iteration)
    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Build weights:
    # norm: weight shape [dim], bias shape [dim]
    # left_proj weight shape [hidden_dim, dim] as per Linear weight shape (out_features, in_features)
    # Similarly right_proj.
    # left_gate, right_gate, out_gate: weight shape [hidden_dim, dim]
    # to_out_norm weight shape [hidden_dim], bias shape [hidden_dim]
    # to_out weight shape [dim, hidden_dim]

    # We'll convert them to appropriate dtype.
    dtype_fp16 = torch.float16
    dtype_fp32 = torch.float32

    # LayerNorm: torch.nn.functional.layer_norm
    # Using weight and bias:
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Compute normalized x
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast x_norm to fp16 for linear proj
    x_norm_fp16 = x_norm.to(dtype_fp16)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(dtype_fp16)
    right_proj_weight = weights["right_proj.weight"].to(dtype_fp16)

    left = torch.nn.functional.linear(x_norm_fp16, left_proj_weight)
    right = torch.nn.functional.linear(x_norm_fp16, right_proj_weight)

    # Apply mask if not nomask
    if not config.get("nomask", True):
        # Expand mask
        mask_exp = mask.unsqueeze(-1).to(dtype_fp16) # shape (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # Gates: compute using x_norm (fp16)
    left_gate_weight = weights["left_gate.weight"].to(dtype_fp16)
    right_gate_weight = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_weight = weights["out_gate.weight"].to(dtype_fp16)

    left_gate = torch.nn.functional.linear(x_norm_fp16, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_fp16, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_fp16, out_gate_weight).sigmoid()

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Now compute contraction using Triton

    # Permute for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, hidden_dim, N, N) -> (M= N rows, K= N cols)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transpose axes 2 and 3: (B, hidden_dim, N, N)

    # Flatten batch and hidden dims into B_total
    B_total = left_perm.shape[0] * left_perm.shape[1]   # B*hidden
    left_flat = left_perm.view(B_total, N, N)
    right_flat = right_perm.view(B_total, N, N)

    # Use Triton kernel for batched matmul: A @ B (where B already transposed)
    out_flat = triton_matmul(left_flat, right_flat)

    # Reshape back to (B, hidden, N, N)
    out = out_flat.view(left_perm.shape[0], left_perm.shape[1], N, N)
    # Permute back to (B, N, N, hidden)
    out = out.permute(0, 2, 3, 1).contiguous()

    # LayerNorm on hidden dim
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate after cast to fp32
    out_gate_float = out_gate.to(dtype_fp32)
    out = out * out_gate_float

    # Final linear projection to original dim
    to_out_weight = weights["to_out.weight"].to(dtype_fp32)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

    assert A.dtype == B.dtype == torch.float16

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The algorithm follows the PyTorch reference:
1. LayerNorm over the channel dimension.
2. Linear projections (left/right) and gating (left/right/out) without bias.
3. Optional element‑wise mask.
4. Apply gates to the projected tensors.
5. Compute the core contraction:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which is a batched matrix multiplication of shape (B·D, N, N) @ (B·D, N, N)^T.
   This is performed by a custom Triton kernel (half‑precision inputs,
   float‑accumulation, half‑output).
6. LayerNorm over the hidden dimension, apply out‑gate, and final linear
   projection back to the original channel dimension.

Only the heavy N³ contraction is offloaded to Triton; the rest stays in
PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A: (batch, M, K), B: (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where
    A: (batch, M, K), B: (batch, K, N), all float16 and contiguous.
    Returns C of shape (batch, M, N) float16.
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    B_total, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((B_total, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B_total,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input layer‑norm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4. Gates (half‑precision)
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    => for each (b,d) perform (N×N) @ (N×N)^T
    # --------------------------------------------------------------

    # Permute to bring hidden dim to second axis, then make B transposed on
    # the last two spatial dimensions.
    left_perm = left.permute(0, 3, 1, 2).contiguous()       # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Both are half‑precision tensors ready for the kernel
    left_flat = left_flat.to(torch.float16)
    right_flat = right_flat.to(torch.float16)

    out_flat = triton_matmul(left_flat, right_flat)   # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H) and convert to float32 for the remaining steps
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output layer‑norm, gate, and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
