@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, hidden_dim, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # fetch program ids
    bh_id = tl.program_id(0)  # combined index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch and hidden index
    b = bh_id // hidden_dim
    h = bh_id % hidden_dim

    # Compute start positions
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Compute pointer base for this slice
    left_tile_ptr = left_ptr + b * stride_left_b + h * stride_left_h
    right_tile_ptr = right_ptr + b * stride_right_b + h * stride_right_h
    out_tile_ptr = out_ptr + b * stride_out_b + h * stride_out_h

    # Compute offsets for rows i and j
    # shape (BLOCK_M, BLOCK_K) for left, (BLOCK_K, BLOCK_N) for right.

    # Offsets for i coordinates
    i_offsets = i_start + tl.arange(0, BLOCK_M)
    j_offsets = j_start + tl.arange(0, BLOCK_N)
    k_offsets = tl.arange(0, BLOCK_K)

    # Make sure to handle mask for edges
    i_mask = i_offsets < N
    j_mask = j_offsets < N
    # Outer mask for out store: shape (BLOCK_M, BLOCK_N): combine via i_mask[:, None] & j_mask[None, :]

    # We will use broadcast for computing pointers inside loop.

    # Use row-major contiguous layout: left stride i (next row) and stride k (next column).
    # Right pointer indexing: we need K as rows, N as columns.

    # We compute for each k block loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    for k0 in range(0, tl.cdiv(N, BLOCK_K)):
        # idx for K tile start
        k_start = k0 * BLOCK_K

        # calculate actual K offsets (size BLOCK_K)
        k_idx = k_start + k_offsets

        # mask for K (maybe out-of-range)
        k_mask = k_idx < N

        # load left tile
        # pointer offsets: i_offset * stride_left_i + k_idx * stride_left_k
        # compute combined offsets using broadcast
        left_ptrs = left_tile_ptr + (i_offsets[:, None] * stride_left_i) + (k_idx[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # load right tile (transposed)
        # want shape (BLOCK_K, BLOCK_N) with k as rows, j as columns
        right_ptrs = right_tile_ptr + (k_idx[:, None] * stride_right_k) + (j_offsets[None, :] * stride_right_j)
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # multiply-accumulate
        acc += tl.dot(left, right, out_dtype=tl.float32)  # left shape (M,K) right (K,N)

        # increment loop
    # end loop

    # Store result
    out_ptrs = out_tile_ptr + (i_offsets[:, None] * stride_out_i) + (j_offsets[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

    mean = x.mean(dim=-1, keepdim=True)

@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch and h
    b = bh_id // H
    h = bh_id % H

    # compute start indices
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # compute pointers for this slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # compute valid mask for i,j positions
    i_valid = i0 + tl.arange(0, BLOCK_M) < N
    j_valid = j0 + tl.arange(0, BLOCK_N) < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k0 in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k0 * BLOCK_K

        # compute offsets for k within tile
        k_valid = k_start + tl.arange(0, BLOCK_K) < N

        # compute offsets for left: i rows, k columns
        # left_tile shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (i0 + tl.arange(0, BLOCK_M))[:, None] * stride_left_i + (k_start + tl.arange(0, BLOCK_K))[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)

        # compute offsets for right: k rows, j columns (transposed)
        # right_tile shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + (k_start + tl.arange(0, BLOCK_K))[:, None] * stride_right_k + (j0 + tl.arange(0, BLOCK_N))[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)

        # dot product: left (M,K) * right (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_base + (i0 + tl.arange(0, BLOCK_M))[:, None] * stride_out_i + (j0 + tl.arange(0, BLOCK_N))[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_valid[:, None] & j_valid[None, :])

"""
TriMul forward pass with a custom Triton kernel.

The heavy part of the TriMul operation is the tensor contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We fuse this contraction into a Triton GEMM kernel that computes, for each
(batch, hidden) slice, the matrix product left[b,d] @ right[b,d].T.
All other operations (LayerNorms, linear projections, gating, masking)
are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,                # *float32
    right_ptr,               # *float32
    out_ptr,                 # *float32
    B, H, N,                 # int32
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    bh_id = tl.program_id(0)          # combined batch * hidden dimension
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile column

    # decode batch and hidden indices
    b = bh_id // H
    h = bh_id % H

    # start indices of the tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # offsets for the outer dimensions
    i_offsets = i0 + tl.arange(0, BLOCK_M)
    j_offsets = j0 + tl.arange(0, BLOCK_N)

    # boundary masks for rows / columns
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i_offsets[:, None] * stride_left_i + k_offsets[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # right block (transposed): shape (BLOCK_K, BLOCK_N)
        # we need right[k, j] where k is the row index and j the column index
        right_ptrs = right_base + k_offsets[:, None] * stride_right_k + j_offsets[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(left, right)

    # write the tile back
    out_ptrs = out_base + i_offsets[:, None] * stride_out_i + j_offsets[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor [B, N, N, dim]
            - mask:  torch.Tensor [B, N, N]
            - weights: dict of model parameters
            - config: dict with keys "dim" and "hidden_dim" (and optional "nomask")

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(dtype=dtype)
        left = left * mask
        right = right * mask
    # If nomask is True we skip the multiplication (mask assumed all ones)

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid after a bias‑free linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the Triton kernel:
    #    shape (B, hidden_dim, N, N) with contiguous memory layout
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out_perm = torch.empty_like(left_perm)

    # Block sizes (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Launch grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    BH = B * hidden_dim
    grid = (BH, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    # Launch the Triton kernel
    tri_mul_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        hidden_dim,
        N,
        left_perm.stride(0),
        left_perm.stride(1),
        left_perm.stride(2),
        left_perm.stride(3),
        right_perm.stride(0),
        right_perm.stride(1),
        right_perm.stride(2),
        right_perm.stride(3),
        out_perm.stride(0),
        out_perm.stride(1),
        out_perm.stride(2),
        out_perm.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Recover the original layout: (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8. Apply the final output gate and linear projection to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
