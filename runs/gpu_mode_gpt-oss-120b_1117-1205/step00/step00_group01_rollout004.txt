@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # mask for i,j within N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer offsets for left tile
        left_ptrs = left_ptr + bh * stride_left_bh + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # load left tile
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # compute pointer offsets for right tile (transposed)
        right_ptrs = right_ptr + bh * stride_right_bh + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # matrix multiplication
        acc += tl.dot(left, right)  # left shape (M,K), right shape (K,N)

    # store result
    out_ptrs = out_ptr + bh * stride_out_bh + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # Apply LayerNorm to input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Compute projections
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # Apply mask
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).type_as(left)  # broadcast
        left = left * mask_unsq
        right = right * mask_unsq
    # Compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    # Cast to half for speed
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    # Reshape for kernel
    B, N, _, _ = left.shape
    # left reshape to (B*H,N,N)
    left_resh = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_resh = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # allocate output
    out_resh = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)
    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * hidden_dim, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_kernel[grid](
        left_resh, right_resh, out_resh,
        N,
        left_resh.stride(0), left_resh.stride(1), left_resh.stride(2),
        right_resh.stride(0), right_resh.stride(1), right_resh.stride(2),
        out_resh.stride(0), out_resh.stride(1), out_resh.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_resh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim]
    # Apply to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    # Apply out_gate (broadcast)
    out = out * out_gate
    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # no bias
    return out

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # coordinates
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left_ptrs = left_ptr + bh * stride_left_bh + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (K x N)
        right_ptrs = right_ptr + bh * stride_right_bh + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)  # left: (M,K), right: (K,N) -> (M,N)

    # Write out
    out_ptrs = out_ptr + bh * stride_out_bh + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

The heavy O(N³) part:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

is computed by a tiled Triton kernel (fp16 inputs, fp32 accumulation).
All other operations (LayerNorm, linear projections, gates) are performed
with PyTorch. The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the core “sum over k” operation
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                # fp16,   shape (B*H, N, N)
    right_ptr,               # fp16,   shape (B*H, N, N)
    out_ptr,                 # fp32,   shape (B*H, N, N)
    N: tl.int32,             # sequence length
    stride_left_batch,       # stride for batch*head dimension in left
    stride_left_i,           # stride for i dimension in left
    stride_left_k,           # stride for k dimension in left
    stride_right_batch,      # stride for batch*head dimension in right
    stride_right_j,          # stride for j dimension in right (used as column index)
    stride_right_k,          # stride for k dimension in right
    stride_out_batch,        # stride for batch*head dimension in out
    stride_out_i,            # stride for i dimension in out
    stride_out_j,            # stride for j dimension in out
    BLOCK_M: tl.constexpr,   # tile size for i
    BLOCK_N: tl.constexpr,   # tile size for j
    BLOCK_K: tl.constexpr,   # tile size for reduction over k
):
    bh = tl.program_id(0)               # batch*head index
    pid_i = tl.program_id(1)            # tile index over i
    pid_j = tl.program_id(2)            # tile index over j

    # ------------------------------------------------------------------
    # Compute offsets for the i and j dimensions
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + bh * stride_left_batch
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + bh * stride_right_batch
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # matrix multiply-accumulate
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + bh * stride_out_batch
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask          : torch.Tensor [B, N, N] (float or bool)
        - weights       : dict of parameter tensors
        - config        : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack and prepare
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model parameters (all are float32 in the provided dict)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (normalizes over the last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # [B, N, N, H]
    right = F.linear(x, right_proj_weight)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # Apply mask if present (mask is broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_broadcast = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # Compute gates (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Cast to half precision for the heavy O(N³) kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Reshape tensors to (B*H, N, N) to treat each (batch, head) as an
    # independent matrix‑multiply task.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Allocate output buffer (FP32 accumulator)
    out_t = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        left_t.stride(0), left_t.stride(1), left_t.stride(2),
        right_t.stride(0), right_t.stride(1), right_t.stride(2),
        out_t.stride(0), out_t.stride(1), out_t.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_t.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # Final LayerNorm over hidden dimension, gating, and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # out_gate was computed in fp32 – keep it in fp32 for the final multiplication
    out = out * out_gate

    # Project back to original dimension
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
