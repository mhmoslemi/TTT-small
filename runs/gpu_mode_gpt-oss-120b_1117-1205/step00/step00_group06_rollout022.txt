"""
Optimized Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

Algorithm:
1. Layer‑norm the input.
2. Linear projections (left/right) and gating (left/right/out) are performed in PyTorch.
3. Apply mask (if any) and element‑wise gating.
4. Reshape the gated tensors to (B·H, N, N) where H = hidden_dim.
   - left  → (b, i, k, d)  → permuted to (b, d, i, k)
   - right → (b, j, k, d)  → permuted to (b, d, k, j)
5. Perform a batched matrix multiplication with a custom Triton kernel:
   out[b·d, i, j] = Σ_k left[b·d, i, k] * right[b·d, k, j]
6. Reshape the result back to (B, N, N, H) and cast to float32.
7. Apply the output layer‑norm, the out‑gate, and the final linear projection.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,                       # batch = B*H, square matrices of size N
    stride_am, stride_ak, stride_ab,      # a: (batch, M, K)
    stride_bk, stride_bn, stride_bb,      # b: (batch, K, N)
    stride_cm, stride_cn, stride_cb,      # c: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B, where each batch element is a square matrix."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current block
    a_ptrs = a_ptr + pid_batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    k_offset = 0
    while k_offset < K:
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

        # Advance pointers to the next k‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
        k_offset += BLOCK_K

    # Write the result (store as FP16 to save bandwidth)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    O(N³·H) contraction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------- 1. LayerNorm ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # -------- 2. Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------- 3. Optional mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)  # broadcast over hidden dim
        left = left * mask_exp
        right = right * mask_exp

    # -------- 4. Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------- 5. Prepare tensors for batched GEMM ----------
    # Convert to FP16 before the heavy contraction to reduce memory traffic
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute so that the hidden dimension becomes the batch dimension for GEMM
    # left: (B, N, N, H) -> (B, H, N, N)  where axes are (i, k)
    left_t = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    # right: (B, N, N, H) -> (B, H, N, N)  with axes (k, j)
    right_t = right_h.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_t.shape
    total_batch = B * H

    # Flatten the leading two dimensions to feed the batched kernel
    a = left_t.view(total_batch, N, N)
    b = right_t.view(total_batch, N, N)

    # Output container (in FP16)
    out = torch.empty_like(a, dtype=torch.float16)

    # Strides in element units
    stride_am, stride_ak, stride_ab = a.stride()
    stride_bk, stride_bn, stride_bb = b.stride()
    stride_cm, stride_cn, stride_cb = out.stride()

    # ---- Triton launch parameters ----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a,
        b,
        out,
        total_batch,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H) and cast to FP32 for the remaining ops
    out_f = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -------- 6. Output layer‑norm and gating ----------
    out_norm = torch.nn.functional.layer_norm(
        out_f,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate

    # -------- 7. Final linear projection ----------
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final
