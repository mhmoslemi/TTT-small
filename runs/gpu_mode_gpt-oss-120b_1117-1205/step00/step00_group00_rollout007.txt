@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_size, seq_len, hidden_dim,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch index and hidden index
    b = pid_batch // hidden_dim
    h = pid_batch % hidden_dim

    # bounds
    N = seq_len
    # offsets for block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-range
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute pointer base for this (b,h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k range
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile [M, K]
        left_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile [K, N] (transposed view)
        # right indexing: right[b, j, k, h]; we need tile with k rows and j columns
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # compute dot
        acc += tl.dot(left_tile, right_tile)

    # store result
    out_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype  # likely float32.

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Projections and gates
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # apply mask if present (mask always)
    # mask broadcast
    mask_exp = mask.unsqueeze(-1).type_as(x)  # ensure same dtype
    left = left * mask_exp
    right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    batch_size, seq_len, _, _ = left_h.shape

    # allocate output
    out_h = torch.empty((batch_size, seq_len, seq_len, hidden_dim), dtype=torch.float16, device=device)

    # get strides
    l_s = left_h.stride()
    r_s = right_h.stride()
    o_s = out_h.stride()

    # define BLOCK sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid
    grid = (batch_size * hidden_dim, triton.cdiv(seq_len, BLOCK_M), triton.cdiv(seq_len, BLOCK_N))

    # launch kernel
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        batch_size,
        seq_len,
        hidden_dim,
        l_s[0], l_s[1], l_s[2], l_s[3],
        r_s[0], r_s[1], r_s[2], r_s[3],
        o_s[0], o_s[1], o_s[2], o_s[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # convert back to float32
    out = out_h.to(torch.float32)

    # to_out_norm (layernorm over hidden_dim)
    out = torch.nn.functional.layer_norm(

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden index
    b = pid_batch // H
    h = pid_batch % H

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K)
        left_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        left_tile = left_tile.to(tl.float32)

        # Load right tile (K, N) transposed.
        # Right indexing: right[b, j, k, h] -> we need (k, j)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        right_tile = right_tile.to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    # Store result
    out_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul forward pass with a Triton kernel for the O(N³) contraction.
The kernel computes, for each batch and hidden channel h,
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
which is equivalent to the einsum
    out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)

All other operations (layernorms, linear projections, gates) are performed
with PyTorch for simplicity.  The contraction is executed in float16
inside the kernel and accumulated in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,      # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,      # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,      # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: out = left @ right^T for each (b, h) pair."""
    pid_batch = tl.program_id(0)   # encodes (batch, hidden) flattened
    pid_m = tl.program_id(1)       # block row (i)
    pid_n = tl.program_id(2)       # block col (j)

    # decode batch and hidden index
    b = pid_batch // H
    h = pid_batch - b * H          # pid_batch % H

    # coordinates within the output matrix
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # right tile transposed: we need (k, j) layout
        # right[b, j, k, h] -> we stride over j then k, but we want k as leading dim
        right_ptrs = right_base \
            + offs_k[:, None] * stride_r_k \
            + offs_j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)   # (M,K) @ (K,N) -> (M,N)

    # store result in fp16 (output tensor is half)
    out_ptrs = out_base \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Executes the TriMul forward pass.
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, D]
            mask: torch.Tensor of shape [B, N, N] (float or bool)
            weights: dict with all layer parameters (float32 tensors)
            config: dict containing "dim" (D) and "hidden_dim" (H)
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # 2. Linear projections
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Apply mask (if any)
    # -------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)
    left = left * mask_exp
    right = right * mask_exp

    # -------------------------------------------------
    # 4. Gating
    # -------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Contraction via Triton kernel (half precision)
    # -------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, _ = left_h.shape

    out_h = torch.empty((B, N, N, hidden_dim),
                        dtype=torch.float16, device=device)

    # strides (in element units)
    s_l = left_h.stride()
    s_r = right_h.stride()
    s_o = out_h.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # (batch * hidden) dimension
        triton.cdiv(N, BLOCK_M),            # rows
        triton.cdiv(N, BLOCK_N),            # cols
    )

    tri_mul_contract_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, hidden_dim,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to float32 for the remainder of the pipeline
    out = out_h.to(torch.float32)

    # -------------------------------------------------
    # 6. Output LayerNorm, gating, final linear
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
