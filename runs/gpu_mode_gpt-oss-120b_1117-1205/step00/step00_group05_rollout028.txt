def triton_tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_m = tl.program_id(0)  # block row index
    pid_n = tl.program_id(1)  # block col index

Now compute start offsets:

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Offsets for left and right:
    # left index: (b, h, i, k)
    # right index: (b, h, k, j)

    # Compute offsets vectors:
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for boundary
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointer for left and right for each (b,h)
    # offset for left_base = left_ptr + b*stride_l_b + h*stride_l_h
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    # offset for right_base = right_ptr + b*stride_r_b + h*stride_r_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        k_start = kk * BLOCK_K
        # compute current k offsets
        cur_k = k_start + offs_k   # shape (BLOCK_K,)

        # mask for k out-of-range not needed for accumulation (set to zero loads).
        mask_k = cur_k < N

        # pointer addresses:
        # left_ptrs = left_base + offs_i[:, None] * stride_l_i + cur_k[None,:] * stride_l_k
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
        # right_ptrs = right_base + cur_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right_ptrs = right_base + cur_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

        # Load tiles with mask
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left_tile, right_tile)

    # Write back with output stride
    out_base = out_ptr + b * stride_o_b + h * stride_o_h
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    # store only valid region
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul forward pass (outgoing) with a Triton kernel.

The heavy part of the TriMul module is the tensor contraction

    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).  
For every hidden channel `d` this is a matrix multiplication
L_d @ R_dᵀ (both N×N).  The kernel below performs a batched GEMM over
the (B·H) independent matrices using a standard tiling strategy
(BLOCK_M×BLOCK_N tiles, loop over K with BLOCK_K).  The surrounding
operations (layer‑norms, linear projections, gating and the final
linear) are kept in PyTorch.

The kernel works for any batch size (1 or 2), any sequence length
up to 1024 and any hidden dimension (the tests use H=128).  It
produces a float‑32 output tensor of shape [B, N, N, dim].
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,          # const float *
    right_ptr,         # const float *
    out_ptr,           # float *
    N,                 # seq_len (int)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    hidden_dim,        # H (int)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for many independent (B·H) matrices.
    left  : [B, H, N, N]  (index order b,h,i,k)
    right : [B, H, N, N]  (index order b,h,k,j)   ← already transposed
    out   : [B, H, N, N]  (index order b,h,i,j)
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_bh = tl.program_id(2)         # batch * hidden index

    # recover batch and hidden ids
    b = pid_bh // hidden_dim
    h = pid_bh % hidden_dim

    # start offsets for this tile
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundaries
    mask_i = i_offset < N
    mask_j = j_offset < N

    # base pointer that already points to the correct batch & hidden slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along K dimension
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_tiles_k):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers to the current K‑slice
        left_ptrs  = left_base  + (i_offset[:, None] * stride_l_i
                                   + k_offset[None, :] * stride_l_k)
        right_ptrs = right_base + (k_offset[:, None] * stride_r_k
                                   + j_offset[None, :] * stride_r_j)

        # guarded loads
        left_tile  = tl.load(left_ptrs,
                             mask=mask_i[:, None] & mask_k[None, :],
                             other=0.0)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        acc += tl.dot(left_tile, right_tile)   # (M,K)@(K,N) -> (M,N)

    # write result
    out_ptrs = out_base + (i_offset[:, None] * stride_o_i
                           + j_offset[None, :] * stride_o_j)
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Execute the TriMul forward pass using a Triton kernel for the O(N³) contraction.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict of torch.Tensor with the model parameters
            - config: dict with keys ["dim", "hidden_dim", "nomask"]
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- 1. LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---------- 2. Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,H]

    # ---------- 3. Optional mask ----------
    if not nomask:
        m = mask.unsqueeze(-1).to(x.dtype)               # [B,N,N,1]
        left = left * m
        right = right * m

    # ---------- 4. Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Triton kernel for the contraction ----------
    # Rearrange to [B, H, N, N] and transpose right so that we compute L @ Rᵀ
    left_perm = left.permute(0, 3, 1, 2)                # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1)              # [B, H, N, N] (already transposed)

    # Allocate output buffer in the permuted layout
    out_perm = torch.empty_like(left_perm, device=device)

    # Strides (in elements)
    s_l = left_perm.stride()
    s_r = right_perm.stride()
    s_o = out_perm.stride()

    # Tiling parameters – chosen to fit typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
        B * hidden_dim                 # batch * hidden dimension
    )

    tri_mul_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to the original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ---------- 6. Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # ---------- 7. Apply output gate ----------
    out = out * out_gate

    # ---------- 8. Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
