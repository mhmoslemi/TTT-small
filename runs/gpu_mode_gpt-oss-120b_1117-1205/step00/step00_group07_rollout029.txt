"""
TriMul forward pass (outgoing version) with a Triton kernel for the heavy
pairwise reduction.

Algorithm
---------
1. Layer‑norm the input tensors (last dimension).
2. Linear projections produce `left`, `right` (shape [B,N,N,H]) and three
   gates (`left_gate`, `right_gate`, `out_gate`).
3. Apply mask (if any) and gating to `left` and `right`.
4. Compute the core update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This O(N³·H) reduction is performed by a custom Triton kernel that tiles
   the i‑, j‑ and hidden‑dimension and loops over k in blocks.
5. Layer‑norm the result (`to_out_norm`) and apply the output gate.
6. Final linear maps back to the original channel dimension.

All operations are in float32; the kernel uses float32 accumulators.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------------------- #
# Triton kernel for the O(N³·H) reduction
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers (int64)
    N, H,                                 # problem size
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)          # block index in the i dimension
    pid_j = tl.program_id(1)          # block index in the j dimension

    # ------------------------------------------------------------------- #
    # Offsets and masks for the output tile
    # ------------------------------------------------------------------- #
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    d_offset = tl.arange(0, BLOCK_D)

    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < H

    # accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Loop over k in blocks
    # ------------------------------------------------------------------- #
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        k_offset = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # ----- load left[i, k, d] -------------------------------------------------
        left_ptrs = left_ptr + (
            i_offset[:, None, None] * stride_left_i
            + k_offset[None, :, None] * stride_left_k
            + d_offset[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=(i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]),
            other=0.0,
        )  # shape (I, K, D)

        # ----- load right[j, k, d] ------------------------------------------------
        right_ptrs = right_ptr + (
            j_offset[:, None, None] * stride_right_i
            + k_offset[None, :, None] * stride_right_k
            + d_offset[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=(j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]),
            other=0.0,
        )  # shape (J, K, D)

        # ------------------------------------------------------------------- #
        # Reduction over the K dimension: accumulate left[:,k,:] * right[:,k,:]
        # ------------------------------------------------------------------- #
        for kk in range(0, BLOCK_K):
            # slices have shape (I,D) and (J,D)
            l_slice = left[:, kk, :]            # (I, D)
            r_slice = right[:, kk, :]           # (J, D)
            # Broadcast to (I,J,D) and accumulate
            acc += l_slice[:, None, :] * r_slice[None, :, :]

    # ------------------------------------------------------------------- #
    # Store the result
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + (
        i_offset[:, None, None] * stride_out_i
        + j_offset[None, :, None] * stride_out_j
        + d_offset[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=(i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]),
    )


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module using a fused Triton kernel.

    Args:
        data: (input, mask, weights, config)
            - input: Tensor [B, N, N, dim]
            - mask:  Tensor [B, N, N] (float, 0.0/1.0)
            - weights: dict of weight tensors
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # Helper to move weights onto the correct device/dtype
    # ------------------------------------------------------------------- #
    def to_dev(t):
        if t is None:
            return None
        if t.device != device:
            t = t.to(device)
        if t.dtype != dtype:
            t = t.to(dtype)
        return t

    norm_weight = to_dev(weights["norm.weight"])
    norm_bias = to_dev(weights["norm.bias"])
    left_proj_weight = to_dev(weights["left_proj.weight"])
    right_proj_weight = to_dev(weights["right_proj.weight"])
    left_gate_weight = to_dev(weights["left_gate.weight"])
    right_gate_weight = to_dev(weights["right_gate.weight"])
    out_gate_weight = to_dev(weights["out_gate.weight"])
    to_out_norm_weight = to_dev(weights["to_out_norm.weight"])
    to_out_norm_bias = to_dev(weights["to_out_norm.bias"])
    to_out_weight = to_dev(weights["to_out.weight"])

    # ------------------------------------------------------------------- #
    # 1) Layer‑norm over the channel dimension
    # ------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )

    # ------------------------------------------------------------------- #
    # 2) Linear projections and gating
    # ------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # apply mask (if any)
    mask_f = mask.unsqueeze(-1).to(dtype)  # broadcast to [...,1]
    left = left * mask_f
    right = right * mask_f

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 3) Core TriMul reduction via Triton
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # ensure contiguity for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Block sizes – tuned for register usage on a B200 GPU
    BLOCK_I = 32
    BLOCK_J = 8
    BLOCK_K = 8
    BLOCK_D = 16

    for b in range(B):
        # slice the batch dimension
        left_b = left[b]      # [N,N,H]
        right_b = right[b]    # [N,N,H]
        out_b = out[b]        # [N,N,H]

        # strides (in elements, not bytes)
        s_i = left_b.stride(0)   # stride over i
        s_k = left_b.stride(1)   # stride over k
        s_h = left_b.stride(2)   # stride over hidden dim (usually 1)

        s_i_out = out_b.stride(0)
        s_j_out = out_b.stride(1)
        s_h_out = out_b.stride(2)

        grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))
        tri_mul_kernel[grid](
            left_b.data_ptr(),
            right_b.data_ptr(),
            out_b.data_ptr(),
            N,
            hidden_dim,
            stride_left_i=s_i,
            stride_left_k=s_k,
            stride_left_h=s_h,
            stride_right_i=s_i,          # same layout for right (first dim = j)
            stride_right_k=s_k,
            stride_right_h=s_h,
            stride_out_i=s_i_out,
            stride_out_j=s_j_out,
            stride_out_h=s_h_out,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # ------------------------------------------------------------------- #
    # 4) Output normalization, gating and final projection
    # ------------------------------------------------------------------- #
    out_norm = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5
    )
    out_norm = out_norm * out_gate
    final = torch.nn.functional.linear(out_norm, to_out_weight)  # back to [B,N,N,dim]

    return final
```
