@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

   pid_m = tl.program_id(0) # row block id
   pid_n = tl.program_id(1) # col block id
   pid_z = tl.program_id(2) # batch group index

   # Offsets for rows and cols
   offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # mask for out-of-bounds
   mask_i = offs_i < N
   mask_j = offs_j < N

   # initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over K dimension blocks
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = offs_k < K

       # compute pointers
       a_ptrs = a_ptr + pid_z * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
       b_ptrs = b_ptr + pid_z * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj

       # load tiles
       a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
       b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

       # compute dot product for block (M,N)
       acc += tl.dot(a, b)

   # store result
   c_ptrs = c_ptr + pid_z * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
   tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    right_proj_weight = weights["right_proj.weight"].to(device)

"""
Triton implementation of the “outgoing” TriMul operator.

The heavy part of TriMul is the O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
We treat each (batch, hidden) pair as a separate GEMM:
    left : (i,k) , right : (j,k) → out : (i,j) = left @ rightᵀ .
A batched matrix‑multiply kernel written in Triton performs this
contraction while the surrounding linear layers, LayerNorms and gating
are done in PyTorch. The kernel uses tiling (BLOCK_M×BLOCK_N tiles)
and accumulates in float32 for numerical stability.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C[m,i,j] = Σ_k A[m,i,k] * B[m,j,k] (B is accessed transposed)."""
    pid_i = tl.program_id(0)      # tile row
    pid_j = tl.program_id(1)      # tile col
    pid_z = tl.program_id(2)      # batch * hidden dimension

    # Offsets for the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A and Bᵀ
        a_ptrs = a_ptr + pid_z * stride_am \
                 + offs_i[:, None] * stride_ai \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_z * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bj

        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)   # FP16
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)   # FP16

        # Matrix‑multiply tile
        acc += tl.dot(a, b)   # (BLOCK_M,BLOCK_K)·(BLOCK_K,BLOCK_N) → FP32

    # Write back
    c_ptrs = c_ptr + pid_z * stride_cm \
             + offs_i[:, None] * stride_ci \
             + offs_j[None, :] * stride_cj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(mask_i[:, None] & mask_j[None, :]))


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator with a Triton‑accelerated
    O(N³) contraction.

    Args:
        data: tuple (input, mask, weights, config)
              - input : torch.Tensor of shape [B, N, N, dim]
              - mask  : torch.Tensor of shape [B, N, N] (float or bool)
              - weights: dict of weight tensors
              - config : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # Load weights (all stored in FP32)
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )

    # 2) Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # 3) Apply mask if present (mask dtype may be bool/int)
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 4) Gating
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate          # [B,N,N,hidden]
    right = right * right_gate

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM kernel
    #    reshape to (M, N, N) with M = B * hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B,hidden,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)   # (M, i, k)
    right_flat = right_perm.view(M, N, N) # (M, j, k)

    # Use FP16 inside the kernel for bandwidth, accumulate in FP32
    left_half = left_flat.to(torch.float16)
    right_half = right_flat.to(torch.float16)
    out_half = torch.empty_like(left_half)   # (M, N, N), FP16

    # Strides (in elements)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # k dimension (the inner N)
    stride_bj = right_flat.stride(1)   # j dimension (the outer N)

    stride_cm = out_half.stride(0)
    stride_ci = out_half.stride(1)     # i dimension
    stride_cj = out_half.stride(2)     # j dimension

    # 6) Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        M,                         # batch*hidden
    )
    batch_gemm_kernel[grid](
        left_half,
        right_half,
        out_half,
        M, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to original layout (float32)
    out_fp32 = out_half.to(torch.float32)
    out_perm = out_fp32.view(B, hidden_dim, N, N)      # (B, hidden, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B, N, N, hidden)

    # 8) Output gating, LayerNorm over hidden dim, final projection
    out = out * out_gate

    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = torch.nn.functional.linear(out, to_out_weight)   # final [B,N,N,dim]

    return out
