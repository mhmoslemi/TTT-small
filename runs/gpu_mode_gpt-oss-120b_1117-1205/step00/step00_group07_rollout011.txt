def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype # maybe float32

    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # linear weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5
    # LayerNorm over dim
    # Input shape [B,N,N,dim]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B,N,N,dim]

    # Linear projections to hidden_dim
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # out dim=hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Mask handling
    if mask is not None:
        mask = mask[..., None]  # broadcast to channel dim
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous()

    # Allocate output
    out_perm = torch.empty_like(left_perm, dtype=torch.bfloat16)

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # decode batch and head
    batch = pid_z // H
    head = pid_z % H

    # offsets for output block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

            + batch * stride_l_b \
            + head   * stride_l_h \
            + offs_i[:, None] * stride_l_i \

        + head   * stride_o_h \

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute the TriMul contraction for a batch*head slice.
    Input tensors are stored as [B, H, N, N] where the last dimension is the sum dimension K.
    For each (b, h) we compute:
        out[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k]
    """
    pid_i = tl.program_id(0)   # block row index
    pid_j = tl.program_id(1)   # block column index
    pid_z = tl.program_id(2)   # combined batch*head index

    # decode batch and head
    batch = pid_z // H
    head  = pid_z % H

    # offsets within N
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K)
        left_ptrs = left_ptr \
            + batch * stride_l_b \
            + head   * stride_l_h \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block transposed: we want (K, N) = right[j, k] -> (k, j)
        right_ptrs = right_ptr \
            + batch * stride_r_b \
            + head   * stride_r_h \
            + offs_k[:, None] * stride_r_k \
            + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # cast to fp32 for accumulation
        left_f  = left.to(tl.float32)
        right_f = right.to(tl.float32)

        acc += tl.dot(left_f, right_f)

    # store result (bf16)
    out_ptrs = out_ptr \
        + batch * stride_o_b \
        + head   * stride_o_h \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.bfloat16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing TriangleMultiplicativeUpdate (TriMul) operator.
    Steps:
      1. Layer‑norm the input.
      2. Linear projections + optional mask.
      3. Gating (sigmoid of linear maps) and element‑wise multiplication.
      4. Compute the O(N³) contraction via a fused GEMM kernel (TriMul) in bfloat16.
      5. Layer‑norm, output gate and final linear projection back to the original dimension.
    Returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------
    # 3. Optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------
    if not config.get("nomask", False):
        # mask broadcast to channel dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------
    # 4. Gates (sigmoid of linear maps)
    # ------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 5. Cast to bf16 and launch Triton GEMM kernel for the contraction
    # ------------------------------------------------------------
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # reshape to [B, H, N, N] for the kernel
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape

    # allocate output tensor in bf16
    out_perm = torch.empty_like(left_perm, dtype=torch.bfloat16)

    # strides (in elements)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_perm.stride()

    # block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tri_mul_gemm_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        H,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_i,
        stride_r_k,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout [B, N, N, H] and cast to fp32
    out = out_perm.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------
    # 6. Output layer‑norm, gate, final linear projection
    # ------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate  # element‑wise gating

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
