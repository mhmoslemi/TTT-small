@triton.jit
def apply_mask_gate_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr, mask_ptr,
    B, N, D,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < B*N*N*D
    # load left, left_gate, right, right_gate
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # broadcast mask
    mask_val = tl.load(mask_ptr + (offs // D), mask=mask, other=0.0)
    # compute
    l_out = l * lg * mask_val
    r_out = r * rg * mask_val
    tl.store(left_ptr + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)
```

"""
TriMul forward kernel (outgoing version) with partial Triton fusion.

The implementation follows the reference PyTorch version:
1. LayerNorm over the feature dimension.
2. Linear projections to a hidden dimension (left/right).
3. Compute sigmoid gates (left_gate, right_gate, out_gate).
4. Fuse mask, left/right gates using a custom Triton kernel
   (in‑place update of the projected tensors).
5. Compute the O(N³) contraction with a batched GEMM (torch.bmm).
6. Apply a second LayerNorm, multiply by out_gate and project back
   to the original feature dimension.

Only the mask‑and‑gate fusion is performed in Triton; the heavy
matrix multiplication uses PyTorch’s highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,     # *mut f32
    left_gate_ptr,  # *const f32
    right_ptr,    # *mut f32
    right_gate_ptr, # *const f32
    mask_ptr,     # *const f32  (broadcasted over hidden_dim)
    B: tl.constexpr,
    N: tl.constexpr,
    D: tl.constexpr,      # hidden_dim
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place: left = left * left_gate * mask,
                right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * D
    mask = offs < total

    # Load values (out‑of‑bounds entries return 0.0)
    l  = tl.load(left_ptr + offs,          mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs,    mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,         mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs,   mask=mask, other=0.0)

    # mask_ptr points to [B, N, N] -> flatten -> offset = offs // D
    mask_offs = offs // D
    m = tl.load(mask_ptr + mask_offs,      mask=mask, other=0.0)

    # Apply mask and gates
    l_out = l * lg * m
    r_out = r * rg * m

    # Store results back in‑place
    tl.store(left_ptr + offs,  l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] (float or bool)
            - weights_dict: mapping of weight names to torch.Tensors
            - config_dict: contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input features (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Fuse mask, left/right gates via Triton (in‑place update)
    # ------------------------------------------------------------------
    # Ensure contiguous layout and matching dtype
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask_tensor.to(left.dtype).contiguous()

    total_elems = left.numel()
    BLOCK = 1024  # reasonable block size
    grid = (triton.cdiv(total_elems, BLOCK),)

    _apply_mask_gate_kernel[grid](
        left_ptr=left,
        left_gate_ptr=left_gate,
        right_ptr=right,
        right_gate_ptr=right_gate,
        mask_ptr=mask,
        B=left.shape[0],
        N=left.shape[1],
        D=left.shape[3],
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched matrix multiplication for speed.
    # ------------------------------------------------------------------
    B, N, _, D = left.shape  # D == hidden_dim
    # reshape to (B*D, N, N) and perform bmm
    left_bmm = left.permute(0, 3, 1, 2).reshape(B * D, N, N)
    right_bmm = right.permute(0, 3, 1, 2).reshape(B * D, N, N)

    # right transpose on the last two dims
    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))  # (B*D, N, N)

    # reshape back to [B, N, N, D]
    out = out_bmm.view(B, D, N, N).permute(0, 2, 3, 1)  # (B, N, N, D)

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over hidden_dim, apply out_gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Ensure output is float32 as required
    return out.to(torch.float32)
