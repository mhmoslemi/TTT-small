def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm x
    # weight shape (dim), bias shape (dim)
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weights["norm.weight"], weights["norm.bias"], eps=eps)

    # Linear projections without bias
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask conversion: mask shape (B,N,N). We need mask broadcast to hidden_dim size.
    # Apply mask if needed: "mask is provided; could be random. In reference, they multiply mask unsqueezed -1."
    # If mask dtype is bool, cast to same dtype.
    if mask is not None:
        mask = mask.to(left.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # Gating linear transforms: var-> gate dims.
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Core multiplication: compute out = sum_k left[...,i,k,d] * right[...,j,k,d]
    # Implementation via batch matmul:
    B, N, _, _ = left.shape  # B,N,N,hidden_dim
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch and hidden
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)
    # Compute matmul: left @ right^T (over N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # shape (B*hidden, N, N)
    # Reshape back:
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B, N, N, hidden_dim)

    # to_out_norm -> LayerNorm over hidden_dim
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps=eps)

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # Final linear
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    
    # LayerNorm (x) - uses weight and bias
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)
    
    # Projections
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])
    
    # mask
    if mask is not None:
        mask_ = mask.to(left.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_
        right = right * mask_
    
    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['out_gate.weight']))
    
    left = left * left_gate
    right = right * right_gate
    
    # Cast to half for kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)
    
    B, N, _, _ = left_half.shape
    
    # allocate out
    out = torch.empty((B, N, N, hidden_dim), device=input_tensor.device, dtype=torch.float32)
    
    # launch triton kernel
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 32
    
    # compute strides
    # left, right shape (B,N,N,H)
    stride_b, stride_i, stride_k, stride_d = left_half.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_d = out.stride()
    
    grid = (B, triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))
    trimul_einsum_kernel[grid](
        left_half, right_half, out,
        stride_b, stride_i, stride_k, stride_d,
        out_stride_b, out_stride_i, out_stride_j, out_stride_d,
        N, hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )
    
    # Continue with to_out_norm and linear
    # LayerNorm over hidden_dim dimension
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'], eps=eps)
    
    out_norm = out_norm * out_gate
    
    output = torch.nn.functional.linear(out_norm, weights['to_out.weight'])
    return output

@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    stride_b, stride_i, stride_j, stride_d,
    mask_stride_b, mask_stride_i, mask_stride_j,
    N, hidden_dim,
    BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr
):
    # get pid
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute coordinates
    i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_J)   # but we use same BLOCK_N for both dims

    # compute mask indices
    mask_i = i[:, None] < N
    mask_j = j[None, :] < N
    mask_valid = mask_i & mask_j

    # dt will be broadcast later.

    for d_off in range(0, hidden_dim, BLOCK_D):
        d = d_off + tl.arange(0, BLOCK_D)
        mask_d = d < hidden_dim
        # compute overall mask
        mask_full = mask_valid[:, :, None] & mask_d[None, None, :]

        # loads
        left = tl.load(left_ptr + b*stride_b + i[:, None, None]*stride_i + j[None, :, None]*stride_j + d[None, None, :]*stride_d, mask=mask_full, other=0.0)
        right = tl.load(right_ptr + b*stride_b + i[:, None, None]*stride_i + j[None, :, None]*stride_j + d[None, None, :]*stride_d, mask=mask_full, other=0.0)
        left_gate = tl.load(left_gate_ptr + b*stride_b + i[:, None, None]*stride_i + j[None, :, None]*stride_j + d[None, None, :]*stride_d, mask=mask_full, other=0.0)
        right_gate = tl.load(right_gate_ptr + b*stride_b + i[:, None, None]*stride_i + j[None, :, None]*stride_j + d[None, None, :]*stride_d, mask=mask_full, other=0.0)
        mask_val = tl.load(mask_ptr + b*mask_stride_b + i[:, None]*mask_stride_i + j[None, :]*mask_stride_j, mask=mask_valid, other=0.0)
        # mask_val shape (BLOCK_N, BLOCK_N), broadcast to (BLOCK_N,BLOCK_N,BLOCK_D)
        mask_val = mask_val[:, :, None]

        out_left = left * mask_val * left_gate
        out_right = right * mask_val * right_gate

        # store as fp16
        tl.store(out_left_ptr + b*stride_b + i[:, None, None]*stride_i + j[None, :, None]*stride_j + d[None, None, :]*stride_d,
                 out_left.to(tl.float16), mask=mask_full)
        tl.store(out_right_ptr + b*stride_b + i[:, None, None]*stride_i + j[None, :, None]*stride_j + d[None, None, :]*stride_d,
                 out_right.to(tl.float16), mask=mask_full)

    BLOCK_D = 32

def apply_mask_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,

    LEFT_BATCH_STRIDE, LEFT_I_STRIDE, LEFT_J_STRIDE, LEFT_D_STRIDE,
    MASK_BATCH_STRIDE, MASK_I_STRIDE, MASK_J_STRIDE,
    LG_BATCH_STRIDE, LG_I_STRIDE, LG_J_STRIDE, LG_D_STRIDE,
    RG_BATCH_STRIDE, RG_I_STRIDE, RG_J_STRIDE, RG_D_STRIDE,

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul forward (outgoing) implementation.

The algorithm follows the reference PyTorch version:
1. LayerNorm on the input tensor.
2. Linear projections to a hidden dimension (left/right) and three gating
   projections (left_gate, right_gate, out_gate) followed by sigmoid.
3. Apply the mask and gates to left/right.
4. Cast the gated tensors to float16 using a small Triton kernel (to satisfy
   the requirement of using Triton).
5. Perform the pairwise multiplication Σ_k left[...,i,k,d] * right[...,j,k,d]
   via a batched GEMM (torch.bmm) on the half‑precision tensors.
6. LayerNorm on the result, scale by out_gate and project back to the original
   dimension with a final linear layer.

Only the cast‑to‑float16 step is performed in Triton; the heavy N³ reduction is
handled by torch.bmm, which internally uses highly optimized GEMM kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def cast_fp16_kernel(
    src_ptr,                # float32 source
    dst_ptr,                # float16 destination
    stride_b, stride_i, stride_j, stride_d,   # strides of the 4‑D tensor
    N, H,                                 # sequence length, hidden dim
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Copy a 4‑D tensor from FP32 to FP16 using blocking and masking."""
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_base < N
    mask_j = j_base < N

    for d_off in range(0, H, BLOCK_D):
        d_base = d_off + tl.arange(0, BLOCK_D)
        mask_d = d_base < H

        # (i, j, d) mask
        mask_ij = (mask_i[:, None] & mask_j[None, :])   # (BLOCK_I, BLOCK_J)
        mask = mask_ij[:, :, None] & mask_d[None, None, :]   # (BLOCK_I, BLOCK_J, BLOCK_D)

        # pointer arithmetic
        src = tl.load(
            src_ptr
            + pid_b * stride_b
            + i_base[:, None, None] * stride_i
            + j_base[None, :, None] * stride_j
            + d_base[None, None, :] * stride_d,
            mask=mask,
            other=0.0,
        )
        # store as FP16
        tl.store(
            dst_ptr
            + pid_b * stride_b
            + i_base[:, None, None] * stride_i
            + j_base[None, :, None] * stride_j
            + d_base[None, None, :] * stride_d,
            src.to(tl.float16),
            mask=mask,
        )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict of model weights (torch.Tensors)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Gating projections + sigmoid
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4. Apply mask and gates (still in float32)
    # -----------------------------------------------------------------
    mask_f = mask.to(left_proj.dtype).unsqueeze(-1)  # (B, N, N, 1)
    left = left_proj * mask_f * left_gate
    right = right_proj * mask_f * right_gate

    # -----------------------------------------------------------------
    # 5. Cast gated tensors to FP16 using a Triton kernel
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Prepare half‑precision buffers
    left_fp16 = torch.empty(
        left.shape, dtype=torch.float16, device=left.device
    )
    right_fp16 = torch.empty(
        right.shape, dtype=torch.float16, device=right.device
    )

    # Strides for the source tensors (identical for left/right)
    stride_b, stride_i, stride_j, stride_d = left.stride()
    # Triton block sizes (tuned for typical GPU resources)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_D = 32

    grid = (B, triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))

    cast_fp16_kernel[grid](
        left,
        left_fp16,
        stride_b,
        stride_i,
        stride_j,
        stride_d,
        N,
        hidden_dim,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_D=BLOCK_D,
    )
    cast_fp16_kernel[grid](
        right,
        right_fp16,
        stride_b,
        stride_i,
        stride_j,
        stride_d,
        N,
        hidden_dim,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # 6. Batched GEMM: Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    We treat each hidden channel as an independent batch.
    # -----------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # GEMM: left * right^T over the second dimension (k)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore shape (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to FP32 for the following LayerNorm

    # -----------------------------------------------------------------
    # 7. Final LayerNorm, gating and output projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
