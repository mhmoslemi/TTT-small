"""
TriMul "outgoing" implementation with a Triton kernel for the O(N³) contraction.
The kernel computes, for each batch and hidden slice h,
    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to a batched matrix multiplication left @ rightᵀ.
All other operations (LayerNorm, Linear projections, gating, final
LayerNorm and Linear) are performed in PyTorch. The result matches the
reference PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,                # left tensor pointer
    b_ptr,                # right tensor pointer
    c_ptr,                # output pointer
    stride_a_batch,       # stride for batch dimension of a
    stride_a_i,           # stride for i (row) dimension of a
    stride_a_k,           # stride for k (reduction) dimension of a
    stride_a_h,           # stride for hidden dimension of a
    stride_b_batch,       # stride for batch dimension of b
    stride_b_k,           # stride for k dimension of b (original k)
    stride_b_j,           # stride for j (col) dimension of b (original j)
    stride_b_h,           # stride for hidden dimension of b
    stride_c_batch,       # stride for batch dimension of c
    stride_c_i,           # stride for i dimension of c
    stride_c_j,           # stride for j dimension of c
    stride_c_h,           # stride for hidden dimension of c
    N,                    # sequence length (compile‑time constant)
    H,                    # hidden dimension (compile‑time constant)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM computing left @ rightᵀ for each hidden slice."""
    pid_m = tl.program_id(0)        # block row index (i)
    pid_n = tl.program_id(1)        # block col index (j)
    pid_batch = tl.program_id(2)    # combined batch * hidden index

    # decode combined batch/hidden index
    b = pid_batch // H
    h = pid_batch % H

    # base pointers for this (b, h) slice
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B block transposed: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # store the result
    tl.store(
        c_base
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm on input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype=left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core contraction with Triton ----------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides (batch, i, k, h) for left/right, (batch, i, j, h) for out
    s_a_batch, s_a_i, s_a_k, s_a_h = left.stride()
    s_b_batch, s_b_j, s_b_k, s_b_h = right.stride()
    s_c_batch, s_c_i, s_c_j, s_c_h = out.stride()

    # set block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        s_a_batch,
        s_a_i,
        s_a_k,
        s_a_h,
        s_b_batch,
        s_b_k,
        s_b_j,
        s_b_h,
        s_c_batch,
        s_c_i,
        s_c_j,
        s_c_h,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑processing ----------
    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # apply output gate
    out = out * out_gate
    # final linear projection back to dim
    output = F.linear(out, weights["to_out.weight"])

    return output
