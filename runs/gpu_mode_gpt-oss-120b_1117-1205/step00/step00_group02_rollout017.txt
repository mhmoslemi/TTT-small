"""
TriMul (outgoing) forward pass implementation.

The module operates on a 4‑D tensor `x` of shape [B, N, N, C] (C = dim) and
produces an output of the same shape.  The computation is:

1. Layer‑norm over the last dimension.
2. Two linear projections (left / right) followed by element‑wise masking.
3. Gated versions of the projections (sigmoid gates).
4. Core contraction:
   out[b,i,j,d] = Σ_{k=0}^{N-1} left[b,i,k,d] * right[b,j,k,d]
   which is a batched matmul for each hidden dimension d.
   This O(N³·H) kernel is written in Triton and tiled over (i, j) and the
   reduction dimension k.
5. Layer‑norm on the hidden dimension, multiplication by an output gate,
   and a final linear projection back to the original dimension.

All heavy work (step 4) is performed by a Triton kernel; the surrounding
operations are expressed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# --------------------------------------------------------------------------- #
# Triton kernel: batched matmul for the TriMul contraction.
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_bmm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for every batch b and hidden index h:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape [B, N, N, H] in row‑major layout.
    The kernel is tiled over i (BLOCK_M) and j (BLOCK_N) with a K‑reduction
    over BLOCK_K.
    """
    pid_m = tl.program_id(0)               # i‑tile index
    pid_n = tl.program_id(1)               # j‑tile index
    pid_z = tl.program_id(2)               # combined (batch, hidden) index

    # Decode batch and hidden indices from the flattened pid_z
    b = pid_z // H
    h = pid_z % H

    # Offsets for the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator for the output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k_off = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + \
                 b * stride_left_b + h * stride_left_h + \
                 offs_m[:, None] * stride_left_i + \
                 k_off[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load a tile of right, transposed so that k is the leading dimension:
        # shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + \
                 b * stride_right_b + h * stride_right_h + \
                 k_off[:, None] * stride_right_k + \
                 offs_n[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate the dot product over the K dimension
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write the result tile back to memory
    c_ptrs = out_ptr + \
             b * stride_out_b + h * stride_out_h + \
             offs_m[:, None] * stride_out_i + \
             offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Entry point required by the evaluator.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    O(N³) contraction.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (bool/int or float)
            - weights: dict with the model parameters
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------- #
    # Load model weights
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1. Layer‑norm over the last dimension
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias) and optional mask
    # ------------------------------------------------------------------- #
    left = F.linear(x_norm, left_proj_weight)      # [B, N, N, H]
    right = F.linear(x_norm, right_proj_weight)    # [B, N, N, H]

    # Apply mask if the configuration indicates it (mask may be all‑ones)
    nomask = config.get("nomask", True)
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 3. Gating
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 4. Core contraction using Triton (batched matmul over the hidden dim)
    # ------------------------------------------------------------------- #
    B, N, N2, H = left.shape
    assert N == N2 and H == hidden_dim, "Shape mismatch after projection."

    # Ensure contiguous memory for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # will hold the contracted result

    # Strides (in elements, not bytes)
    stride_left = left.stride()    # (B, N, N, H)
    stride_right = right.stride()
    stride_out = out.stride()

    # Tiling parameters
    BLOCK_M = 32   # tiles rows (i)
    BLOCK_N = 32   # tiles cols (j)
    BLOCK_K = 32   # reduction tile size (k)

    # Grid: (i‑tiles, j‑tiles, batch×hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    # Launch the Triton kernel
    trimul_bmm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 5. Final LayerNorm, output gate and projection back to dim
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # back to original dim

    # Ensure float32 output
    return output.to(torch.float32)
