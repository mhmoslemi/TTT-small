custom_kernel(data):
    input_tensor, mask, weights, config = data
    Initialize variables.

    # Norm
    x = F.layer_norm(input_tensor, (config["dim"],), weight=norm_weight, bias=norm_bias)

    # Projections
    left = F.linear(x, left_proj_weight)
    right = F.linear(x, right_proj_weight)

    # Mask multiplication (if needed)
    if not config.get("nomask", False):
        # Use Triton kernel for each of left and right
        left = apply_mask(left, mask, hidden_dim)
        right = apply_mask(right, mask, hidden_dim)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Contraction using batch matrix multiplication:
    B, N, _, H = left.shape
    left_flat = left.permute(0,3,1,2).reshape(B*H, N, N)
    right_flat = right.permute(0,3,1,2).reshape(B*H, N, N)
    out_flat = torch.matmul(left_flat, right_flat.transpose(1,2))
    out = out_flat.reshape(B, H, N, N).permute(0,2,3,1)  # shape (B,N,N,H)

    # Output layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul forward implementation with a Triton kernel for masked elementwise multiplication.

Algorithm
---------
1. Layer‑normalize the input tensor (shape [B,N,N,D]).
2. Project the normalized tensor to a hidden dimension H using two linear layers
   (left_proj, right_proj) → tensors L,R of shape [B,N,N,H].
3. If a mask is present (config["nomask"] == False), multiply L and R by the mask
   broadcast over the hidden dimension.  This elementwise product is performed by
   a small Triton kernel (mask_mul_kernel) that works on the flattened tensors
   and avoids Python loops.
4. Compute three gates (left, right, out) from the normalized input using linear
   layers followed by a sigmoid.
5. Apply the left/right gates elementwise to L and R.
6. Perform the core contraction:
       out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
   which is equivalent to a batched matrix multiplication
   `L_reshape @ R_reshapeᵀ` where the batch dimension is B*H.
7. Layer‑normalize the result over the hidden dimension, apply the out‑gate,
   and finally project back to the original dimension with a linear layer.
8. Return the output tensor of shape [B,N,N,D] (float32).

The heavy compute (the N³ contraction) is handled by PyTorch's highly‑optimized
batched GEMM (`torch.matmul`), while the mask multiplication—often a memory‑bound
bottleneck—is offloaded to Triton for a modest speedup and to satisfy the
requirement of using a Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------- #
# Triton kernel: elementwise multiplication with a broadcasted mask
# -------------------------------------------------------------------------- #
@triton.jit
def mask_mul_kernel(
    x_ptr,            # *float32, input tensor (flattened)
    mask_ptr,         # *float32, mask tensor (flattened, shape B*N*N)
    out_ptr,          # *float32, output tensor (flattened)
    size,             # i64, number of elements in x (and out)
    hidden_dim,       # i64, size of the hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    active = offs < size

    # Load input element
    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    # Compute the corresponding mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)

    # Store result
    tl.store(out_ptr + offs, x * m, mask=active)


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    Multiply `tensor` of shape [B,N,N,H] by `mask` of shape [B,N,N] broadcast
    over the hidden dimension using the Triton kernel defined above.
    """
    # Ensure contiguous tensors and matching dtype
    tensor = tensor.contiguous()
    mask = mask.to(tensor.dtype).contiguous()

    out = torch.empty_like(tensor)

    # Flatten all tensors (C‑contiguous order)
    x_flat = tensor.view(-1)
    mask_flat = mask.view(-1)
    out_flat = out.view(-1)

    total_elems = x_flat.numel()
    BLOCK_SIZE = 1024  # reasonable block size for elementwise work
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    mask_mul_kernel[grid](
        x_flat,
        mask_flat,
        out_flat,
        total_elems,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out


# -------------------------------------------------------------------------- #
# Main custom kernel entry point
# -------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Executes the TriMul forward pass.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, D]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored if config["nomask"])
            - weights: dict of torch.Tensors with the model parameters
            - config: dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,D]

    # ----------------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ----------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,H]
    right = F.linear(x, right_proj_weight)  # [B,N,N,H]

    # ----------------------------------------------------------------------
    # 3. Optional mask multiplication (handled by Triton)
    # ----------------------------------------------------------------------
    if not nomask:
        # broadcast mask to hidden dimension via Triton kernel
        left = _apply_mask_triton(left, mask, hidden_dim)
        right = _apply_mask_triton(right, mask, hidden_dim)

    # ----------------------------------------------------------------------
    # 4. Gating
    # ----------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM using torch.matmul.
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape  # left and right are [B,N,N,H]

    # Reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # (B*H, N, N)

    # Matrix multiplication with transpose on the second operand
    out_flat = torch.matmul(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # ----------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final projection
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # [B,N,N,D]

    # Ensure output dtype is float32 as required
    out = out.to(torch.float32)
    return out
