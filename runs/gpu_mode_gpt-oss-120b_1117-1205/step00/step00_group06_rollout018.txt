def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_am, stride_ak, stride_ah,
    stride_bm, stride_bk, stride_bh,
    stride_cm, stride_cn, stride_ch,
    N: tl.constexpr,
    BATCH_H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid_z = tl.program_id(2)  # batch_h
    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Compute mask for out-of-range
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask k within bounds
        mask_k = offs_k < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype   # likely torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)
    # Apply mask if present
    if not nomask:
        mask_expanded = mask.to(left.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_expanded
        right = right * mask_expanded

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Move to half for kernel (optional)
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Permute to (B, hidden_dim, N, N) and flatten batch*head dimension
    left_perm = left_half.permute(0, 3, 1, 2).contiguous()
    right_perm = right_half.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    G = B * H
    left_mat = left_perm.reshape(G, N, N)
    right_mat = right_perm.reshape(G, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)

    # Triton kernel launch config
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (G,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # Strides
    stride_A_z, stride_A_m, stride_A_k = left_mat.stride()
    stride_B_z, stride_B_j, stride_B_k = right_mat.stride()
    stride_C_z, stride_C_i, stride_C_j = out_mat.stride()

    tri_mul_matmul_kernel[grid](

"""
TriMul (outgoing) implementation with a Triton batched GEMM kernel.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reformulated as a batched matrix multiplication per (b,d):
    L = left.permute(0,3,1,2).reshape(G,N,N)   # G = B*hidden_dim
    R = right.permute(0,3,1,2).reshape(G,N,N)
    out = L @ Rᵀ   (matrix product over the shared k dimension)

The Triton kernel computes the batched GEMM with tiling over the i, j and k
axes (BLOCK_M, BLOCK_N, BLOCK_K).  The rest of the TriMul pipeline
(LayerNorm, linear projections, gating, final LayerNorm & linear) is
performed with PyTorch for simplicity and numeric stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_A_z, stride_A_m, stride_A_k,   # A: (group, i, k)
    stride_B_z, stride_B_j, stride_B_k,   # B: (group, j, k) – we read its transpose
    stride_C_z, stride_C_i, stride_C_j,   # C: (group, i, j)
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)   # batch * hidden dimension
    pid_m = tl.program_id(1)   # i tile
    pid_n = tl.program_id(2)   # j tile

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_A_z
            + offs_m[:, None] * stride_A_m
            + offs_k[None, :] * stride_A_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ block: shape (BLOCK_K, BLOCK_N)
        # Original B layout: (group, j, k).  Bᵀ[k, j] = B[j, k].
        b_ptrs = (
            B_ptr
            + pid_z * stride_B_z
            + offs_k[:, None] * stride_B_k
            + offs_n[None, :] * stride_B_j
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write the result tile
    c_ptrs = (
        C_ptr
        + pid_z * stride_C_z
        + offs_m[:, None] * stride_C_i
        + offs_n[None, :] * stride_C_j
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    TriMul forward pass using a Triton kernel for the O(N³) contraction.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]    (float32)
            - mask:         [B, N, N]         (bool or 0/1)
            - weights: dict with all required weight tensors
            - config: dict containing "dim", "hidden_dim", "nomask" (bool)
    Returns:
        output: [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Extract config and weights
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on the last dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projection weights (no bias)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]

    # Gating weights
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Output LayerNorm and final projection
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 2. Core TriMul computation in torch (LayerNorm, projections, gates)
    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim]

    # Linear projections
    left = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # Apply optional mask (broadcast on hidden dim)
    if not nomask:
        mask_exp = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Triton batched GEMM for the contraction
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) and flatten the first two dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    G = B * H  # number of independent matmuls

    # Shape (G, N, N)
    left_mat = left_perm.view(G, N, N)
    right_mat = right_perm.view(G, N, N)

    # Output placeholder
    out_mat = torch.empty_like(left_mat)  # float32

    # Kernel tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        G,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides for A, B, C (all contiguous: (z,i,k) layout)
    stride_A_z, stride_A_m, stride_A_k = left_mat.stride()
    stride_B_z, stride_B_j, stride_B_k = right_mat.stride()
    stride_C_z, stride_C_i, stride_C_j = out_mat.stride()

    tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_A_z,
        stride_A_m,
        stride_A_k,
        stride_B_z,
        stride_B_j,
        stride_B_k,
        stride_C_z,
        stride_C_i,
        stride_C_j,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, H, N, N) then to (B, N, N, H)
    out_perm = out_mat.view(B, H, N, N)                     # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]
    # --------------------------------------------------------------
    # 4. Post‑processing: LayerNorm → gating → final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    return out
