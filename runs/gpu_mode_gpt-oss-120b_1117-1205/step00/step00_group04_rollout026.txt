@triton.jit
def trigemm_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # combined batch and hidden

    # Decode batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # Compute offsets for this block
    # compute starting row (i) and column (j) for output
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create mask for valid i/j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Compute base pointers for left and right and out for this batch and hidden
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Final offsets for i and j (row/col)
    # left block: rows i_offset, cols k offset during loop
    # right block: rows k_offset, cols j_offset (due to transpose)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left block
        # left_ptrs: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (i_offset[:, None] * stride_left_i) + (k_offset[None, :] * stride_left_k)
        # load with mask
        left_val = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # cast to fp32
        left_val = left_val.to(tl.float32)

        # Load right block transposed, shape (BLOCK_K, BLOCK_N) (k rows, j columns)
        right_ptrs = right_base + (k_offset[:, None] * stride_right_k) + (j_offset[None, :] * stride_right_j)
        right_val = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right_val = right_val.to(tl.float32)

        # Accumulate
        acc += tl.dot(left_val, right_val)  # (BLOCK_M,BLOCK_K) x (BLOCK_K,BLOCK_N)

    # Store result
    out_ptrs = out_base + (i_offset[:, None] * stride_out_i) + (j_offset[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def tri_mul_contract_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    b = pid_z // H
    h = pid_z % H

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kp in range(num_k):
        k = kp * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        left_ptrs = left_base + (i[:, None] * stride_l_i) + (k[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0), dtype=tl.float16)
        left = left.to(tl.float32)

        right_ptrs = right_base + (k[:, None] * stride_r_k) + (j[None, :] * stride_r_j)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0), dtype=tl.float16)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

    out_ptrs = out_base + (i[:, None] * stride_o_i) + (j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left, right: B,N,N,H float32
    B, N, _, H = left.shape
    device = left.device
    # Convert to float16 for bandwidth
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    tri_mul_contract_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

"""
Triton implementation of the core contraction of AlphaFold3's TriMul module.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections: `left` and `right` (shape [B,N,N,H]).
3. Optional masking of the projected tensors.
4. Compute element‑wise gates (sigmoid) and apply them to `left` and `right`.
5. **Triton kernel** – for each batch `b` and hidden channel `h`,
   compute the matrix product
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to `left @ rightᵀ` summed over the third dimension.
   The kernel tiles the `i` and `j` dimensions (BLOCK_M × BLOCK_N) and
   iterates over `k` in BLOCK_K steps, accumulating in float32 while
   loading the inputs in float16 for bandwidth efficiency.
6. Layer‑norm the result over the hidden dimension, multiply by the
   output gate, and apply the final linear projection back to `dim`.
7. Return a float32 tensor of shape [B, N, N, dim].

Only the O(N³) contraction is performed in Triton; the surrounding
operations (layer‑norms, linear layers, gating) are implemented with
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)      # block row
    pid_j = tl.program_id(1)      # block column
    pid_z = tl.program_id(2)      # combined batch * hidden

    # decode batch index and hidden index
    b = pid_z // H
    h = pid_z % H

    # offsets for the output tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left block: (i, k)
        left_ptrs = left_base + (i[:, None] * stride_l_i) + (k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=tl.float16(0.0),
                       dtype=tl.float16)
        left = left.to(tl.float32)

        # right block (transposed view): (k, j)
        right_ptrs = right_base + (k[:, None] * stride_r_k) + (j[None, :] * stride_r_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=tl.float16(0.0),
                        dtype=tl.float16)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)   # (BLOCK_M,BLOCK_K) x (BLOCK_K,BLOCK_N)

    # store result (float32)
    out_ptrs = out_base + (i[:, None] * stride_o_i) + (j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_contract(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the batched "left @ rightᵀ" contraction using the Triton kernel.
    Both `left` and `right` have shape [B, N, N, H] and dtype float32.
    Returns a float32 tensor of the same shape.
    """
    B, N, _, H = left.shape
    device = left.device

    # Half precision for the inner multiplication (bandwidth bound)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (in element counts, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    tri_mul_contract_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul operator with the heavy
    N³ contraction off‑loaded to a Triton kernel.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict of model weights
            - config: dict containing 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype torch.float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    no_mask = config.get("nomask", False)

    # ----- LayerNorm on the input ------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----- Linear projections ------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Optional masking --------------------------------------------------------
    if not no_mask:
        # mask shape [B, N, N] -> [B, N, N, 1] and cast to match left/right dtype
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # ----- Gating -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Core contraction (TriMul) ------------------------------------------------
    contracted = _tri_mul_contract(left, right)  # [B, N, N, hidden_dim] float32

    # ----- Output LayerNorm, gating, final projection -------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    contracted = F.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    contracted = contracted * out_gate

    to_out_weight = weights["to_out.weight"]
    output = F.linear(contracted, to_out_weight)  # [B, N, N, dim]

    return output
