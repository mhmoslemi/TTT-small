def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operation.
    The core triple multiplication (einsum over k) is performed by a
    batched GEMM kernel written in Triton. The surrounding linear
    projections, gating, and LayerNorms are computed with PyTorch.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm on input
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights['norm.weight'],
                     bias=weights['norm.bias'],
                     eps=eps)

    # Linear projections (no bias)
    left_proj = F.linear(x, weights['left_proj.weight'])   # shape [B, N, N, hidden_dim]
    right_proj = F.linear(x, weights['right_proj.weight'])

    # Apply mask if present
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask
    else:
        mask = None

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))   # shape [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to half for GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    batch = B * hidden_dim

    # Rearrange tensors to [batch, N, N] where batch = B*hidden_dim
    # Layout: (B, hidden_dim, N, N)
    left_t = left_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_t = right_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)

    # Output tensor
    out_h = torch.empty((batch, N, N), dtype=torch.float16, device=input_tensor.device)

    # Strides for GEMM
    stride_a_batch, stride_a_i, stride_a_k = left_t.stride()

    stride_c_batch, stride_c_i, stride_c_j = out_h.stride()

    # Block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,

        right_t,

        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

    out = out_h.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # Final linear projection to dim

    return out

"""
Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

The heavy O(N³) contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication:
    left (M×K) @ rightᵀ (K×N)  →  (M×N) for every hidden channel.
We fuse this contraction in a custom Triton kernel while the surrounding
LayerNorms, Linear projections and gating are performed with PyTorch.
The kernel processes tiles of size (BLOCK_M, BLOCK_N) and loops over K in
chunks of BLOCK_K, accumulating in FP32 for accuracy and finally storing
the result in FP16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM where the RHS is logically transposed.
# ----------------------------------------------------------------------
@triton.jit
def _gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_j, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices.

    A: (batch, M, K)      – row‑major
    B: (batch, N, K)      – row‑major (will be accessed as K×N)
    C: (batch, M, N)
    """
    pid_b = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column

    # Offsets of the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Base pointers for this batch element
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_i[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile as (BLOCK_K, BLOCK_N) – note the swapped axes (logical transpose)
        b_ptrs = b_base + (offs_k[:, None] * stride_b_k) + (offs_j[None, :] * stride_b_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Dot product for this k‑tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write the result back (store as FP16)
    c_ptrs = c_base + (offs_i[:, None] * stride_c_i) + (offs_j[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton‑backed
    batched GEMM for the O(N³) contraction.

    Args:
        data: Tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim] (float32)
            mask   – torch.Tensor [B, N, N] (bool/int/float)
            weights – dict with all layer weights
            config  – dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack and prepare
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input (over the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if provided). mask is broadcast on the channel dim.
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # Combine projections with gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate      # [B, N, N, hidden_dim]
    right = right_proj * right_gate   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    batch = B * hidden_dim

    # Permute to (B, hidden_dim, N, N) then flatten the first two dims
    left_t = left_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)   # A: (batch, M=N, K=N)
    right_t = right_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N) # B: (batch, N, K=N)

    # ------------------------------------------------------------------
    # Allocate output buffer (FP16, will be converted later)
    # ------------------------------------------------------------------
    out_h = torch.empty((batch, N, N), dtype=torch.float16, device=input_tensor.device)

    # ------------------------------------------------------------------
    # Strides for the Triton kernel (taken from the flattened tensors)
    # ------------------------------------------------------------------
    stride_a_batch, stride_a_i, stride_a_k = left_t.stride()
    stride_b_batch, stride_b_j, stride_b_k = right_t.stride()
    stride_c_batch, stride_c_i, stride_c_j = out_h.stride()

    # ------------------------------------------------------------------
    # Launch configuration (tuned for typical B200 GPU)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch,                     # independent batch dimension
    )
    _gemm_outgoing_kernel[grid](
        left_t,
        right_t,
        out_h,
        batch, N, N, N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_j, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and continue in FP32
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # Output LayerNorm over the hidden dimension, apply out‑gate and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
