@triton.jit
def tri_mul_kernel(A_ptr, B_ptr, C_ptr,
                   M, N, K,
                   stride_am, stride_an, stride_ak,
                   stride_bm, stride_bj, stride_bk,
                   stride_cm, stride_cn, stride_ck,
                   BLOCK_M: tl.constexpr,
                   BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(2)  # batch_hidden index
    pid_i = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block

    # Compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Load initial accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute pointer to start of A and B for this batch_hidden
    a_batch_ptr = A_ptr + pid_m * stride_am
    b_batch_ptr = B_ptr + pid_m * stride_bm

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k offset range
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask inside bounds for k
        a_mask = (i_offset < N)[:, None] & (k_offset < K)[None, :]
        b_mask = (k_offset < K)[:, None] & (j_offset < N)[None, :]

        # Load A block (BLOCK_M x BLOCK_K)
        a_ptrs = a_batch_ptr + i_offset[:, None] * stride_an + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # Load B block (BLOCK_K x BLOCK_N) transposed: B original indices (j,k) but we need (k,j)
        b_ptrs = b_batch_ptr + k_offset[:, None] * stride_bk + j_offset[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Update accumulator: matrix multiplication (M x K) * (K x N)
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # Write output
    c_batch_ptr = C_ptr + pid_m * stride_cm
    c_ptrs = c_batch_ptr + i_offset[:, None] * stride_cn + j_offset[None, :] * stride_ck
    c_mask = (i_offset < N)[:, None] & (j_offset < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_cn, stride_ck,

):

    pid_i = tl.program_id(0)

"""
TriMul forward kernel (outgoing version) for AlphaFold3.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication:
    left_d[b] @ right_d[b].T  for each hidden channel d.
A Triton kernel computes this multiplication in FP16 for speed,
while all surrounding operations (LayerNorm, projections, gates) are
performed with PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # M = batch*hidden, N = seq_len, K = seq_len
    stride_am, stride_an, stride_ak,    # A strides (M,N,K)
    stride_bm, stride_bj, stride_bk,    # B strides (M,N,K) – we treat B as transposed
    stride_cm, stride_cn, stride_ck,    # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(2)  # index over (batch * hidden)
    pid_i = tl.program_id(0)  # block row (i)
    pid_j = tl.program_id(1)  # block col (j)

    # Coordinate offsets for this block
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current (batch, hidden) slice
    a_batch = A_ptr + pid_m * stride_am
    b_batch = B_ptr + pid_m * stride_bm
    c_batch = C_ptr + pid_m * stride_cm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (i_off < N)[:, None] & (k_off < K)[None, :]
        b_mask = (k_off < K)[:, None] & (j_off < N)[None, :]

        # Load A block (i, k)
        a_ptrs = a_batch + i_off[:, None] * stride_an + k_off[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # Load B block transposed: we need B(k, j) but B is stored as (j, k)
        b_ptrs = b_batch + k_off[:, None] * stride_bk + j_off[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Store result (convert back to FP16)
    c_ptrs = c_batch + i_off[:, None] * stride_cn + j_off[None, :] * stride_ck
    c_mask = (i_off < N)[:, None] & (j_off < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Perform the TriMul forward pass. `data` is a tuple:
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns the processed tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, n, _, _ = input_tensor.shape  # input: [B, N, N, dim]

    # ---- LayerNorm on the input ------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---- Linear projections ----------------------------------------------------
    # x: [B,N,N,dim] -> left/right: [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---- Apply mask (if provided) --------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---- Gates -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---- Prepare tensors for Triton kernel (batched GEMM) ----------------------
    # Move hidden dimension to the batch axis and make them contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to FP16 for the kernel
    left_h = left_perm.to(torch.float16)
    right_h = right_perm.to(torch.float16)

    # Collapse batch and hidden into a single leading dimension
    mh = bs * hidden_dim
    left_h = left_h.view(mh, n, n)
    right_h = right_h.view(mh, n, n)

    # Output buffer
    out_h = torch.empty_like(left_h, dtype=torch.float16, device=device)

    # Strides (int values)
    stride_am, stride_an, stride_ak = left_h.stride()
    stride_bm, stride_bj, stride_bk = right_h.stride()
    stride_cm, stride_cn, stride_ck = out_h.stride()

    # ---- Triton launch --------------------------------------------------------
    BLOCK = 64
    BLOCK_M = min(BLOCK, n)
    BLOCK_N = min(BLOCK, n)
    BLOCK_K = min(BLOCK, n)

    grid = (
        triton.cdiv(n, BLOCK_M),   # i blocks
        triton.cdiv(n, BLOCK_N),   # j blocks
        mh,                         # batch*hidden
    )
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        mh,
        n,
        n,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---- Reshape back to original layout ---------------------------------------
    out = out_h.view(bs, hidden_dim, n, n).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]
    out = out.to(torch.float32)

    # ---- Final LayerNorm, gate, and output projection ---------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
