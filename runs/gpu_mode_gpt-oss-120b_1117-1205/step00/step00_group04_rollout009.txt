@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, H,
    stride_in_bs, stride_in_is, stride_in_js, stride_in_hs,
    stride_out_bs, stride_out_is, stride_out_js, stride_out_hs,
    eps: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # compute indices
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_h = tl.arange(0, BLOCK_H)
    # mask for valid hidden dims
    mask_h = offs_h < H

    # compute base offsets
    inp_off = b * stride_in_bs + i * stride_in_is + j * stride_in_js
    out_off = b * stride_out_bs + i * stride_out_is + j * stride_out_js

    # load data
    x = tl.load(inp_ptr + inp_off + offs_h * stride_in_hs, mask=mask_h, other=0.0)
    # compute mean
    x_masked = tl.where(mask_h, x, 0.0)
    sum_x = tl.sum(x_masked)
    mean = sum_x / H

    # diff
    diff = x - mean
    sq = diff * diff
    sq_masked = tl.where(mask_h, sq, 0.0)
    var = tl.sum(sq_masked) / H

    # normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # apply weight and bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # store
    tl.store(out_ptr + out_off + offs_h * stride_out_hs, out, mask=mask_h)

"""
TriMul forward kernel (outgoing version).

Algorithm
---------
1. Layer‑norm the input tensor `x` over its last dimension (dim).
2. Linear projections → left/right, and three gate vectors (left_gate,
   right_gate, out_gate) using the supplied weight matrices.
3. Apply the pairwise mask (if present) and the gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
4. Compute the pairwise contraction over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with an `einsum` (equivalent to a batched matrix
   multiplication). The tensors are cast to bfloat16 before the
   contraction for memory/throughput, then cast back to float32.
5. Apply a LayerNorm over the hidden dimension `hidden_dim`.  This
   reduction is performed by a small Triton kernel that computes
   per‑(b,i,j) mean/variance and applies the learned gain/bias.
6. Multiply the normalized tensor by `out_gate`.
7. Final linear projection back to `dim` using `to_out.weight`.

The Triton kernel only handles the hidden‑dim LayerNorm (step 5); all
other heavy work relies on highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,               # *float32  [B,N,N,H]
    out_ptr,               # *float32  [B,N,N,H]
    weight_ptr,            # *float32  [H]
    bias_ptr,              # *float32  [H]
    B,                     # tl.int32
    N,                     # tl.int32
    H,                     # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # tl.int32
    eps: tl.constexpr,                                   # const float
    BLOCK_H: tl.constexpr                               # compile‑time block size (≥ max hidden_dim)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # 3‑D coordinates (b,i,j) of this program instance
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for the start of the vector (h‑axis)
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # compute flat offsets into input / output tensors
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load the H‑dimensional slice
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    )
    # ------------------------------------------------------------------
    # Compute mean
    x_masked = tl.where(mask_h, x, 0.0)
    sum_x = tl.sum(x_masked)
    mean = sum_x / H

    # ------------------------------------------------------------------
    # Compute variance
    diff = x - mean
    sq = diff * diff
    sq_masked = tl.where(mask_h, sq, 0.0)
    var = tl.sum(sq_masked) / H

    # ------------------------------------------------------------------
    # Normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ------------------------------------------------------------------
    # Apply learned gain/bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h,   mask=mask_h, other=0.0)
    out = norm * w + b_

    # ------------------------------------------------------------------
    # Store result
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be unused)
            - weights: dict of pre‑loaded weight tensors
            - config: dict containing "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Ensure contiguous for later reshapes
    x = x.contiguous()

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gating (all in float32)
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) and gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)       # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction over k (TriMul core)
    # Cast to bfloat16 for reduced bandwidth as in the reference.
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    out = torch.einsum('... i k h, ... j k h -> ... i j h', left_bf, right_bf)  # [B,N,N,hidden_dim] bfloat16
    out = out.to(dtype).contiguous()

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)
    # Strides are in element units; pass them directly.
    s_b, s_i, s_j, s_h = out.stride()
    # Triton grid: one program per (b,i,j)
    total_programs = B * N * N
    grid = (total_programs,)
    BLOCK_H = 128  # hidden_dim ≤ 128 for all test config

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)   # [B*N*N, hidden_dim]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
