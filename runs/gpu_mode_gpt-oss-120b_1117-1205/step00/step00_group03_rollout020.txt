@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    total_bh = B * tl.cdiv(H, BLOCK_H)

    # compute batch index and hidden slice base
    batch_hidden = pid_bh
    batch_idx = batch_hidden // tl.cdiv(H, BLOCK_H)
    h_block = (batch_hidden % tl.cdiv(H, BLOCK_H)) * BLOCK_H
    # compute indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)
    h = h_block + tl.arange(0, BLOCK_H)

    # mask for indices
    i_mask = i < N
    j_mask = j < N
    k_mask = k < N
    h_mask = h < H

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K + k
        k_mask_ = k_base < N

        # left loads: shape (BLOCK_M, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + batch_idx * stride_left_b \
                    + i[:,None,None] * stride_left_i \
                    + k_base[None,:,None] * stride_left_k \
                    + h[None,None,:] * stride_left_h
        left = tl.load(left_ptrs, mask=i_mask[:,None,None] & k_mask_[:,None,None] & h_mask[None,None,:], other=0.0)

        # right loads with k and j swapped to get transposed:
        # shape (BLOCK_K, BLOCK_N, BLOCK_H)
        right_ptrs = right_ptr + batch_idx * stride_right_b \
                     + k_base[:,None,None] * stride_right_i \
                     + j[None,:,None] * stride_right_k \
                     + h[None,None,:] * stride_right_h
        right = tl.load(right_ptrs, mask=k_mask_[:,None,None] & j_mask[None,:,None] & h_mask[None,None,:], other=0.0)

        # compute dot products for each hidden
        # left: (BLOCK_M, BLOCK_K, BLOCK_H) -> slice per h
        # right: (BLOCK_K, BLOCK_N, BLOCK_H)
        # accumulate
        for hh in range(0, BLOCK_H):
            # We can vectorize using tl.dot on slices
            acc[:, :, hh] += tl.dot(left[:, :, hh], right[:, :, hh])
    # after reduction, store
    out_ptrs = out_ptr + batch_idx * stride_out_b \
               + i[:,None,None] * stride_out_i \
               + j[None,:,None] * stride_out_j \
               + h[None,None,:] * stride_out_h
    tl.store(out_ptrs, acc, mask=i_mask[:,None,None] & j_mask[None,:,None] & h_mask[None,None,:])
```

    a_offs = pid_b * stride_a_batch + offs_m[:, None] * stride_a_i + tl.arange(0, BLOCK_K)[None, :] * stride_a_k

    mask_n = offs_n < N
    mask_k = offs_k < N

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # ensure float32
    x = x.to(torch.float32)

    # Flatten for linear projections
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # shape [B*N*N, dim]

    # Projections
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # Apply mask if needed
    if not nomask:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim))

    left = left * left_gate
    right = right * right_gate

    # Prepare for contraction using Triton
    # Permute to [B, hidden_dim, N, N] and view as flat batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch and hidden into single dimension
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)
    out_flat = torch.empty_like(left_flat)

    # Define block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides
    stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_i, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_i, stride_c_j = out_flat.stride()

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), left_flat.shape[0])

        stride_a_batch, stride_a_i, stride_a_k,

        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    out = torch.nn.functional.linear(out.view(-1, hidden_dim), weights["to_out.weight"]).view(B, N, N, dim)

"""
TriMul forward pass – Triton‐accelerated contraction.

The algorithm
---------------
1. Layer‑norm the 4‑D input over the last channel dimension.
2. Point‑wise linear projections for the left/right streams and their gates.
3. Apply optional mask and sigmoid gates.
4. Contract the two streams:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is equivalent to a batched matrix multiplication
   C = A @ Bᵀ for each hidden slice h.
   We flatten (batch, hidden) into one batch dimension and run a
   tiled GEMM kernel written in Triton.
5. Layer‑norm over hidden_dim, multiply by the out‑gate and a final
   point‑wise linear projection back to the model dimension.

Only the O(N³) contraction is off‑loaded to Triton; all other
operations use PyTorch (which is already highly optimized for the
small per‑position work).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    a_ptr,                     # left  [B*H, N, N]
    b_ptr,                     # right [B*H, N, N] (will be read transposed)
    c_ptr,                     # out   [B*H, N, N]
    N,                         # sequence length
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i‑block
    pid_n = tl.program_id(1)   # j‑block
    pid_b = tl.program_id(2)   # combined (batch, hidden) block

    # ------------------------------------------------------------------
    # 1) Offsets for current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j indices

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator for C_{i,j}
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2) Loop over reduction axis k
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_base = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + k_base[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block (transposed): we need Bᵀ[k, j] = B[j, k]
        # original B layout: [row=j, col=k] -> stride_i for row, stride_k for col
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + k_base[:, None] * stride_b_i   # <-- k as row in Bᵀ
            + offs_n[None, :] * stride_b_k   # <-- j as column in Bᵀ
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate matmul
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 3) Write back C
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask        : torch.Tensor [B, N, N] (may be ignored)
            - weights     : dict of weight tensors
            - config      : dict with keys 'dim', 'hidden_dim', 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float32)          # ensure float32 for the rest

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1) Point‑wise linear projections (left/right) + gates
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    # left / right projections
    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # optional mask (mask is broadcast on the hidden dimension)
    if not nomask:
        m = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * m
        right = right * m

    # gates (sigmoid after linear)
    left_gate = torch.sigmoid(
        F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim)
    )
    right_gate = torch.sigmoid(
        F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim)
    )
    out_gate = torch.sigmoid(
        F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim)
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2) Contraction via Triton (batched GEMM: left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) → flatten (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(-1, N, N)                # [B*H, N, N]
    right_flat = right_perm.view(-1, N, N)
    out_flat = torch.empty_like(left_flat)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides for the flattened tensors (they are contiguous)
    stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_i, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_i, stride_c_j = out_flat.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),               # i‑blocks
        triton.cdiv(N, BLOCK_N),               # j‑blocks
        left_flat.shape[0],                    # combined (batch * hidden) blocks
    )
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 3) Output layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final linear back to model dimension
    out = F.linear(out.view(-1, hidden_dim), weights["to_out.weight"]).view(B, N, N, dim)

    return out
```
