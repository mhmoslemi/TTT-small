- LayerNorm on the last dimension (dim). Actually it's layernorm applied to each (b,i,j,:) vector of size dim. That is typical: self.norm = LayerNorm(dim).
- left_proj = Linear(dim, hidden_dim, bias=False)
- right_proj similarly.
- left_gate = Linear(dim, hidden_dim, bias=False)
- right_gate similarly.
- out_gate = Linear(dim, hidden_dim, bias=False) (for output gating)
- to_out_norm = LayerNorm(hidden_dim)
- to_out = Linear(hidden_dim, dim, bias=False)

Then forward:

- x = self.norm(x)  # [B,N,N,dim] normalized across dim
- left = self.left_proj(x)  # [B,N,N,hidden_dim]
- right = self.right_proj(x)
- mask = mask.unsqueeze(-1)  # [B,N,N,1]
- left = left * mask
- right = right * mask
- left_gate = self.left_gate(x).sigmoid()
- right_gate = self.right_gate(x).sigmoid()
- out_gate = self.out_gate(x).sigmoid()
- left = left * left_gate
- right = right * right_gate
- out = einsum('... i k d, ... j k d -> ... i j d', left, right)
- out = self.to_out_norm(out)  # layernorm on hidden_dim
- out = out * out_gate
- return self.to_out(out)  # final linear to dim

So the core heavy operation is the einsum that sums over k dimension: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]. This is like a matrix multiplication: For each batch and each head/d. For each channel d, left[b,i,:,d] shape (N). right[b,j,:,d] shape (N). Compute outer product of left and right across i and j using sum over k: Actually it's like out = left @ right^T over the k dimension for each d: left shape (B,N,N,hidden), right shape (B,N,N,hidden) - need to treat second dimension i and j with k being over sequence dimension. That's a multiplication of left (i x k) with right (j x k) but not transposed because we multiply left_{i,k} * right_{j,k} directly. @triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_al, stride_ahi, stride_ak, stride_ah,
    stride_br, stride_bj, stride_bk, stride_bh,
    stride_co, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Get program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)  # head index (batch*hidden)
    # Determine tile start offsets
    offset_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offset_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Determine batch and hidden dims from head index
    b = pid_h // hidden
    h = pid_h % hidden
    # Masks
    mask_i = offset_i < N
    mask_j = offset_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offsets for k block
        offset_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # load left tile: shape (BLOCK_M, BLOCK_K)
        # left index: [b, offset_i, offset_k, h]
        left_ptrs = left_ptr \
            + b * stride_al + offset_i[:, None] * stride_ahi \
            + offset_k[None, :] * stride_ak \
            + h * stride_ah
        # Load with mask for i/k out-of-bound
        left = tl.load(left_ptrs, mask=(mask_i[:, None] & (offset_k[None, :] < N)), other=0.0)

        # load right tile: shape (BLOCK_N, BLOCK_K) but we need right transposed across k and j?
        # right is shape (B, j, k, hidden). We need to multiply left[i,k] * right[j,k]. So we need right tile with dimensions (BLOCK_N, BLOCK_K) layout j as rows, k as columns. Right pointer: [b, offset_j, offset_k, h].
        right_ptrs = right_ptr \
            + b * stride_br + offset_j[:, None] * stride_bj \
            + offset_k[None, :] * stride_bk \
            + h * stride_bh
        right = tl.load(right_ptrs, mask=((offset_j[:, None] < N) & (offset_k[None, :] < N)), other=0.0)
        # Compute dot product across k dimension
        acc += tl.dot(left, right, trans_b=False) # dot expects shapes (M, K) and (K, N)? Actually tl.dot expects (M, K) and (K, N) for matrix multiply.
        # But here right shape is (BLOCK_N, BLOCK_K) with j as rows and k columns, we need right transposed to shape (BLOCK_K, BLOCK_N). So we could use tl.dot(left, right, trans_b=True) or use tl.dot(left, right.T). However tl.dot has axes: left is (M, K), right is (K, N) (if trans_b=False). So we need right transposed shape (K,N) i.e., shape (BLOCK_K, BLOCK_N). But we have right tile shape (BLOCK_N, BLOCK_K). We can set trans_b=True to swap dimensions, performing left (M,K) dot right.T (K,N). So use tl.dot(left, right, trans_b=True).
        # Then proceed to next k.

        left_ptrs += BLOCK_K * stride_ak
        right_ptrs += BLOCK_K * stride_bk

    # Store output
    out_ptrs = out_ptr \
        + b * stride_co + offset_i[:, None] * stride_ci \
        + offset_j[None, :] * stride_cj \
        + h * stride_ch
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_al, stride_ahi, stride_ak, stride_ah,
    stride_br, stride_bj, stride_bk, stride_bh,
    stride_co, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)  # head index

    # Determine batch and head channel
    b = pid_h // H
    h = pid_h % H

    # Note: if B maybe >1, only some heads may exceed B; but heads total B*H.

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = offs_k < N

        # Load left tile
        left_ptrs = left_ptr \
            + b * stride_al \
            + offs_i[:, None] * stride_ahi \
            + offs_k[None, :] * stride_ak \
            + h * stride_ah

        # Load right tile
        right_ptrs = right_ptr \
            + b * stride_br \
            + offs_j[:, None] * stride_bj \
            + offs_k[None, :] * stride_bk \
            + h * stride_bh

        # Actually we recompute each iteration so no increment needed.

        + b * stride_co \
        + offs_i[:, None] * stride_ci \
        + offs_j[None, :] * stride_cj \
        + h * stride_ch

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # LayerNorm on dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Mask handling
    if mask is None:
        mask_unsq = torch.ones_like(left, dtype=left.dtype, device=device)
    else:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)  # shape (B,N,N,1)

    # Apply mask and gates
    left = left * mask_unsq * left_gate
    right = right * mask_unsq * right_gate

    # Cast to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)  # same shape, dtype fp16

    # Strides
    stride_al = left_fp16.stride(0)
    stride_ahi = left_fp16.stride(1)
    stride_ak = left_fp16.stride(2)
    stride_ah = left_fp16.stride(3)

    stride_br = right_fp16.stride(0)
    stride_bj = right_fp16.stride(1)
    stride_bk = right_fp16.stride(2)
    stride_bh = right_fp16.stride(3)

    stride_co = out_fp16.stride(0)
    stride_ci = out_fp16.stride(1)
    stride_cj = out_fp16.stride(2)
    stride_ch = out_fp16.stride(3)

    # Define block sizes (tunable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_al, stride_ahi, stride_ak, stride_ah,
        stride_br, stride_bj, stride_bk, stride_bh,
        stride_co, stride_ci, stride_cj, stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32
    out = out_fp16.to(torch.float32)

    # Apply to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul forward pass with a custom Triton kernel.

The heavy O(N³) operation
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is mapped to a tiled batched matrix‑multiply kernel. All other
operations (LayerNorm, linear projections, gating, final projection)
are performed with PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr,               # *float16
    right_ptr,              # *float16
    out_ptr,                # *float16
    B, N, H,                # batch, seq_len, hidden_dim
    stride_al, stride_ahi, stride_ak, stride_ah,   # left strides
    stride_br, stride_bj, stride_bk, stride_bh,   # right strides
    stride_co, stride_ci, stride_cj, stride_ch,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # block index along i
    pid_j = tl.program_id(1)          # block index along j
    pid_h = tl.program_id(2)          # head index = batch * hidden_dim + d

    # decode batch and hidden index
    b = pid_h // H
    d = pid_h % H

    # offsets within the i/j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        left_ptrs = (
            left_ptr
            + b * stride_al
            + offs_i[:, None] * stride_ahi
            + offs_k[None, :] * stride_ak
            + d * stride_ah
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ----- load right tile (N x K) -----
        right_ptrs = (
            right_ptr
            + b * stride_br
            + offs_j[:, None] * stride_bj
            + offs_k[None, :] * stride_bk
            + d * stride_bh
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # dot: (M,K) @ (K,N)  (right is transposed inside tl.dot)
        acc += tl.dot(left, right, trans_b=True)

    # ----- write out tile (M x N) -----
    out_ptrs = (
        out_ptr
        + b * stride_co
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
        + d * stride_ch
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module using a Triton kernel for the
    cubic reduction.  `data` is a tuple (input, mask, weights, config).

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the last dimension ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B,N,N,dim]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Gating values ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ---------- Apply mask (if any) and gates ----------
    if mask is None:
        mask_unsq = torch.ones_like(left, dtype=left.dtype, device=device)
    else:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]

    # fused mask+gate multiplication
    left = left * mask_unsq * left_gate
    right = right * mask_unsq * right_gate

    # ---------- Cast to fp16 for the Triton kernel ----------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)  # shape [B,N,N,hidden], fp16

    # ---------- Strides (in elements) ----------
    stride_al, stride_ahi, stride_ak, stride_ah = (
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
    )
    stride_br, stride_bj, stride_bk, stride_bh = (
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
    )
    stride_co, stride_ci, stride_cj, stride_ch = (
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
    )

    # ---------- Triton launch parameters ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        bs * hidden_dim,
    )

    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        bs,
        seq_len,
        hidden_dim,
        stride_al,
        stride_ahi,
        stride_ak,
        stride_ah,
        stride_br,
        stride_bj,
        stride_bk,
        stride_bh,
        stride_co,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Continue in FP32 ----------
    out = out_fp16.to(torch.float32)

    # ---------- Final LayerNorm on hidden dimension ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Output gating ----------
    out = out * out_gate

    # ---------- Final linear projection back to `dim` ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
