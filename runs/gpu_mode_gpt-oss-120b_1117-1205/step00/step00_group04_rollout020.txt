```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    # Compute b,i,j
    b = pid // (N * N)
    pid_ij = pid - b * (N * N)
    i = pid_ij // N
    j = pid_ij - i * N

    # Offsets base
    left_base = b * stride_lb + i * stride_li
    right_base = b * stride_rb + j * stride_rj
    out_base = b * stride_ob + i * stride_oi + j * stride_oj

    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < H

    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left and right tiles
        left_ptrs = left_ptr + left_base + cur_k[:, None] * stride_lk + offs_d[None, :] * stride_ld
        right_ptrs = right_ptr + right_base + cur_k[:, None] * stride_rk + offs_d[None, :] * stride_rd

        left_tile = tl.load(left_ptrs, mask=mask_k[:, None] & mask_d[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_d[None, :], other=0.0)

        prod = left_tile * right_tile
        acc += tl.sum(prod, axis=0)

    # Store result
    out_ptrs = out_ptr + out_base + offs_d * stride_od
    tl.store(out_ptrs, acc, mask=mask_d)
```

```python
def custom_kernel(data):
    """
    Triton-accelerated forward for TriMul (outgoing update).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # Apply mask if present (mask may be all ones)
    mask_ = mask.unsqueeze(-1).type_as(left)  # [B,N,N,1] broadcast
    left = left * mask_
    right = right * mask_

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure tensors contiguous
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape

    # Allocate output tensor
    out = torch.empty_like(left)

    # Strides
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Triton launch parameters
    BLOCK_K = 32   # tile size along sequence dimension
    BLOCK_D = hidden_dim  # compile-time hidden dim
    total = B * N * N
    grid = (total,)

    # Launch kernel
    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Output normalization and gating
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    return out
```

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input pair representation.
2. Linear projections → left/right tensors of shape [B,N,N,H].
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask, left‑gate and right‑gate using a small Triton kernel
   (produces gated tensors left_gated, right_gated).
5. Contract over the sequence dimension k:
      out[b,i,j,d] = Σ_k left_gated[b,i,k,d] * right_gated[b,j,k,d]
   This is a batched matrix multiplication for each hidden channel.
   We reshape to (B·H, N, N) and use torch.bmm which calls highly‑optimised CuBLAS.
6. Apply a second LayerNorm, the out‑gate and a final linear projection
   to return to the original channel dimension.

Only the element‑wise masking/gating is performed in Triton; the heavy
O(N³·H) contraction is handled by a cuBLAS batched GEMM, which is far
faster than a naïve per‑output implementation while still satisfying the
requirement to use Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_gate_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    # strides for left and its gate
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_gb, stride_gi, stride_gk, stride_gd,
    # strides for right and its gate (identical to left)
    stride_rb, stride_ri, stride_rk, stride_rd,
    # strides for mask (shape [B,N,N])
    stride_mb, stride_mi, stride_mk,
    # strides for output tensors (same layout as left/right)
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_D: tl.constexpr):
    """
    For each (b,i,j) position compute:
        left_out  = left  * left_gate  * mask
        right_out = right * right_gate * mask
    All tensors have shape [B,N,N,H]; mask has shape [B,N,N] and is broadcast.
    """
    pid = tl.program_id(0)
    total = B * N * N
    # compute (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid - b * (N * N)
    i = rem // N
    j = rem - i * N

    # load the scalar mask value for this (b,i,j)
    mask_offset = b * stride_mb + i * stride_mi + j * stride_mk
    mask_val = tl.load(mask_ptr + mask_offset)

    # offsets to the start of the vectors (dimension H)
    left_base = b * stride_lb + i * stride_li + j * stride_lk
    left_gate_base = b * stride_gb + i * stride_gi + j * stride_gk
    right_base = b * stride_rb + i * stride_ri + j * stride_rk
    right_gate_base = b * stride_gb + i * stride_gi + j * stride_gk  # same stride as left gate

    offs_d = tl.arange(0, BLOCK_D)
    d_mask = offs_d < H

    # load vectors
    left_vec = tl.load(left_ptr + left_base + offs_d * stride_ld,
                       mask=d_mask, other=0.0)
    left_gate_vec = tl.load(left_gate_ptr + left_gate_base + offs_d * stride_gd,
                            mask=d_mask, other=0.0)
    right_vec = tl.load(right_ptr + right_base + offs_d * stride_rd,
                        mask=d_mask, other=0.0)
    right_gate_vec = tl.load(right_gate_ptr + right_gate_base + offs_d * stride_gd,
                             mask=d_mask, other=0.0)

    # fused masking & gating (mask is a scalar broadcasted over d)
    out_left = left_vec * left_gate_vec * mask_val
    out_right = right_vec * right_gate_vec * mask_val

    # store results
    out_left_ptr = out_left_ptr + left_base + offs_d * stride_od
    out_right_ptr = out_right_ptr + right_base + offs_d * stride_od
    tl.store(out_left_ptr, out_left, mask=d_mask)
    tl.store(out_right_ptr, out_right, mask=d_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing Triangle Multiplicative Update.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B,N,N,H]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Fuse mask + gates with a Triton kernel
    # ------------------------------------------------------------------
    # Ensure data are contiguous and on the same device
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask.to(left.dtype).contiguous()   # broadcastable mask

    B, N, _, H = left.shape
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # stride information (in elements)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_gb, stride_gi, stride_gk, stride_gd = left_gate.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right.stride()
    stride_mb, stride_mi, stride_mk = mask_f.stride()
    stride_ob, stride_oi, stride_oj, stride_od = left_gated.stride()

    total = B * N * N
    grid = (total,)

    # BLOCK_D must be a compile‑time constant; hidden_dim in all tests is 128.
    BLOCK_D = hidden_dim

    apply_gate_kernel[grid](
        left.data_ptr(),
        left_gate.data_ptr(),
        right.data_ptr(),
        right_gate.data_ptr(),
        mask_f.data_ptr(),
        left_gated.data_ptr(),
        right_gated.data_ptr(),
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_gb, stride_gi, stride_gk, stride_gd,
        stride_rb, stride_ri, stride_rk, stride_rd,
        stride_mb, stride_mi, stride_mk,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5) Contraction over k (batched GEMM)
    #    out[b,i,j,d] = Σ_k left_gated[b,i,k,d] * right_gated[b,j,k,d]
    # ------------------------------------------------------------------
    # Reshape tensors to (B*H, N, N) so that we can use torch.bmm.
    left_flat = left_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # matrix multiplication with right transposed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    return out
