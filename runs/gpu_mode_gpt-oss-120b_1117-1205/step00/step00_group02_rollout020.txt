def tri_mul_forward_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # d offset loop
    for d0 in range(0, H, BLOCK_D):
        cur_block_D = tl.minimum(BLOCK_D, H - d0)  # compile-time not allowed but we can just handle mask
        offs_d = d0 + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # Reduce over k
        for k0 in range(0, N, BLOCK_K):
            offs_k = k0 + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # compute masks for loads
            mask_l = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            mask_r = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]

            # Load left tile
            left_ptrs = left_ptr + b * stride_b \
                + offs_i[:, None, None] * stride_i \
                + offs_k[None, :, None] * stride_k \
                + offs_d[None, None, :] * stride_d

            left_tile = tl.load(left_ptrs, mask=mask_l, other=0.0)  # shape (M,K,D)

            # Load right tile (K,N,D)
            right_ptrs = right_ptr + b * stride_b \
                + offs_k[:, None, None] * stride_k \
                + offs_j[None, :, None] * stride_i \
                + offs_d[None, None, :] * stride_d

            right_tile = tl.load(right_ptrs, mask=mask_r, other=0.0)  # shape (K,N,D)

            # Multiply-accumulate per d
            # left_tile: (M,K,D), right_tile: (K,N,D)
            for d_idx in range(0, BLOCK_D):
                # maybe condition if d_idx >= cur_block_D skip? else mask will zero.
                l = left_tile[:, :, d_idx]  # (M,K)
                r = right_tile[:, :, d_idx]  # (K,N)
                # compute dot product l @ r
                acc[:, :, d_idx] += tl.dot(l, r)

        # After reduction, store
        out_ptrs = out_ptr + b * stride_b \
            + offs_i[:, None, None] * stride_i \
            + offs_j[None, :, None] * stride_k \
            + offs_d[None, None, :] * stride_d

        mask_o = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_o)
```

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def tri_mul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute offsets for i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over channel blocks
    for d0 in range(0, H, BLOCK_D):
        # offsets for hidden dimension
        offs_d = d0 + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # accumulator for this block
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # reduction over k
        for k0 in range(0, N, BLOCK_K):
            offs_k = k0 + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # load left block: shape (M, K, D)
            left_ptrs = left_ptr + pid_b * stride_b \
                + offs_i[:, None, None] * stride_i \
                + offs_k[None, :, None] * stride_k \
                + offs_d[None, None, :]
            mask_l = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            left = tl.load(left_ptrs, mask=mask_l, other=0.0)

            # load right block: shape (K, N, D) where N is j dimension
            right_ptrs = right_ptr + pid_b * stride_b \
                + offs_k[:, None, None] * stride_k \
                + offs_j[None, :, None] * stride_i \
                + offs_d[None, None, :]
            mask_r = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
            right = tl.load(right_ptrs, mask=mask_r, other=0.0)

            # dot for each d
            for d_idx in range(0, BLOCK_D):
                # left slice (M, K), right slice (K, N)
                l = left[:, :, d_idx]
                r = right[:, :, d_idx]
                acc[:, :, d_idx] += tl.dot(l, r)

        # store results
        out_ptrs = out_ptr + pid_b * stride_b \
            + offs_i[:, None, None] * stride_i \
            + offs_j[None, :, None] * stride_k \
            + offs_d[None, None, :]

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Device
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm on input
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # mask
    mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # allocate output buffer for contraction
    out = torch.empty_like(left)

    # compute strides
    stride_b, stride_i, stride_k, stride_d = left.stride()
    # all strides in elements, convert to int
    stride_b = int(stride_b)
    stride_i = int(stride_i)
    stride_k = int(stride_k)

    # tiling parameters (tune)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 16  # must divide hidden_dim; hidden_dim=128 -> 8 blocks

    grid = (left.shape[0],  # batch
            triton.cdiv(left.shape[1], BLOCK_M),
            triton.cdiv(left.shape[2], BLOCK_N))

    tri_mul_forward_kernel[grid](
        left, right, out,
        left.shape[0], left.shape[1], hidden_dim,
        stride_b, stride_i, stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Now out_norm
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out gate
    out = out_norm * out_gate

    # final linear to dim
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
pairwise contraction:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

The kernel tiles the i/j output dimensions (BLOCK_M x BLOCK_N) and the
hidden dimension d (BLOCK_D). For each tile it reduces over the sequence
dimension k in blocks of size BLOCK_K, accumulating a matrix multiplication
per hidden channel. This replaces the expensive einsum in the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_b: tl.constexpr, stride_i: tl.constexpr, stride_k: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # i and j tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over hidden‑dimension blocks
    for d0 in range(0, H, BLOCK_D):
        offs_d = d0 + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # Reduce over the sequence dimension k
        for k0 in range(0, N, BLOCK_K):
            offs_k = k0 + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left tile  (M, K, D)
            left_ptrs = left_ptr + pid_b * stride_b \
                + offs_i[:, None, None] * stride_i \
                + offs_k[None, :, None] * stride_k \
                + offs_d[None, None, :]
            mask_l = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            left_tile = tl.load(left_ptrs, mask=mask_l, other=0.0)

            # Load right tile (K, N, D) – note k dimension first, then j
            right_ptrs = right_ptr + pid_b * stride_b \
                + offs_k[:, None, None] * stride_k \
                + offs_j[None, :, None] * stride_i \
                + offs_d[None, None, :]
            mask_r = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
            right_tile = tl.load(right_ptrs, mask=mask_r, other=0.0)

            # Accumulate dot products for each d in the block
            for d_idx in range(0, BLOCK_D):
                l = left_tile[:, :, d_idx]          # (M, K)
                r = right_tile[:, :, d_idx]         # (K, N)
                acc[:, :, d_idx] = acc[:, :, d_idx] + tl.dot(l, r)

        # Store the completed output block
        out_ptrs = out_ptr + pid_b * stride_b \
            + offs_i[:, None, None] * stride_i \
            + offs_j[None, :, None] * stride_k \
            + offs_d[None, None, :]
        mask_o = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_o)


def custom_kernel(data):
    """
    Performs the TriMul forward pass using a custom Triton kernel for the
    quadratic contraction. All other linear and gating operations are carried
    out with PyTorch for simplicity.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] torch.Tensor
            - mask: [B, N, N] torch.Tensor (0/1)
            - weights: dict of model parameters (see class definition)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Ensure tensors are on the same device & dtype
    device = input_tensor.device
    dtype = torch.float32

    # ----- LayerNorm on the input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    ).to(dtype)

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Gating -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----- Apply mask and gates -----
    mask_f = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
    left = left * mask_f * left_gate
    right = right * mask_f * right_gate

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # ----- Triton kernel: pairwise contraction -----
    out = torch.empty_like(left)

    # Strides in element units
    stride_b, stride_i, stride_k, stride_d = left.stride()
    stride_b = int(stride_b)
    stride_i = int(stride_i)   # stride over the first N dimension (i / j)
    stride_k = int(stride_k)   # stride over the second N dimension (k)

    # Tile sizes (tuned for the B200 GPU)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 16  # hidden_dim (=128) is divisible by 16

    grid = (
        left.shape[0],                     # batch dimension
        triton.cdiv(left.shape[1], BLOCK_M),  # i tiles
        triton.cdiv(left.shape[2], BLOCK_N),  # j tiles
    )

    tri_mul_forward_kernel[grid](
        left,
        right,
        out,
        left.shape[0],         # B
        left.shape[1],         # N
        hidden_dim,            # H
        stride_b,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ----- Output LayerNorm, gating and final linear -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
