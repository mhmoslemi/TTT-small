def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # extract dims
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # compute "norm" on input
    # using weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # shape (dim,) each
    # Apply layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Mask handling
    if not config.get("nomask", False):
        # mask shape B,N,N
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f
    else:
        # optional skip?
        pass

    # Create gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight)).view(B,N,N,hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight)).view(B,N,N,hidden_dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight)).view(B,N,N,hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # Allocate out
    out = torch.empty((B,N,N,hidden_dim), dtype=torch.float32, device=device)

    # Triton kernel
    # get strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()
    # launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B*hidden_dim, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tr_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Apply to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden_dim)
    out_final = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)
    # Return out_final
    return out_final

"""TriMul forward pass with Triton‑accelerated O(N³) contraction.

The heavy part of the TriMul operator is the tensor contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix multiplication (left @ rightᵀ) for each hidden
dimension `h`.  This kernel tiles the i, j and k dimensions and computes the
dot products with `tl.dot`, thereby avoiding the O(N³) Python loops.
All other operations (layer‑norms, linear projections, gating) are kept in
PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: compute out = left @ rightᵀ  for each batch and hidden dim
# ----------------------------------------------------------------------
@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left: (batch, i, k, h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right: (batch, j, k, h)
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out: (batch, i, j, h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)          # combined batch * hidden dimension
    pid_i = tl.program_id(1)          # i‑tile index
    pid_j = tl.program_id(2)          # j‑tile index

    batch = pid0 // H                 # batch index
    hid   = pid0 % H                  # hidden‑dim index

    # i, j offsets within the N×N plane
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the tile (BLOCK_M × BLOCK_N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load a tile of left[b, i, k, h] -----
        left_ptrs = left_ptr \
            + batch * stride_left_b \
            + hid   * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ----- load a tile of right[b, j, k, h] -----
        right_ptrs = right_ptr \
            + batch * stride_right_b \
            + hid   * stride_right_h \
            + offs_j[:, None] * stride_right_i \
            + offs_k[None, :] * stride_right_k
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # matrix‑multiply the (M×K) and (K×N) tiles
        acc += tl.dot(left, right)

    # ----- store the result tile -----
    out_ptrs = out_ptr \
        + batch * stride_out_b \
        + hid   * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # 1. LayerNorm on the input tensor
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask shape [B, N, N] -> [B, N, N, 1]
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Triton kernel: out = left @ rightᵀ   (shape [B, N, N, hidden_dim])
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides (in elements)
    stride_left = left.stride()   # (B, i, k, h)
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * hidden_dim,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tr_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
