def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elems,
    hidden_dim,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    # compute mask index
    mask_idx = offs // hidden_dim
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out_right = right * right_gate * mask_val
    tl.store(out_left_ptr + offs, out_left, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype (should be torch.float32). We'll keep.

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5 (maybe in weights? not provided). Set default.

    # retrieve weight tensors
    norm_weight = weights["norm.weight"] # shape [dim]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"] # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"] # shape [hidden_dim]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"] # shape [dim, hidden_dim]

    # 1. LayerNorm over last dimension of input
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # 2. Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # 3. Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # 4. Apply mask and gates using Triton kernel.
    B, N, _, _ = x.shape # _ is dim, but hidden_dim is after projection.

    # Flatten tensors
    left_flat = left.contiguous()
    right_flat = right.contiguous()
    left_gate_flat = left_gate.contiguous()
    right_gate_flat = right_gate.contiguous()
    out_gate_flat = out_gate.contiguous()  # for later use

    mask_float = mask.to(dtype=left.dtype).contiguous()

    # allocate output tensors for left/right after mask & gate
    left_processed = torch.empty_like(left_flat)
    right_processed = torch.empty_like(right_flat)

    total_elems = left_flat.numel() # = B*N*N*hidden_dim

    BLOCK_SIZE = 1024  # can tune

    grid = (triton.cdiv(total_elems, BLOCK_SIZE), )
    mask_gate_kernel[grid](
        left_flat, right_flat,
        left_gate_flat, right_gate_flat,
        mask_float,
        left_processed, right_processed,
        total_elems,
        hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # 5. Compute out = left_processed @ right_processed^T per channel and batch (batched GEMM)
    # reshape for batch GEMM
    # Move hidden_dim to batch dimension
    left_reshape = left_processed.permute(0,3,1,2).contiguous()  # [B, hidden_dim, N, N]
    right_reshape = right_processed.permute(0,3,1,2).contiguous()

    # Combine batch and hidden_dim into single batch dimension
    B_hd = B * hidden_dim
    left_mat = left_reshape.view(B_hd, N, N)  # [B*H, N, N]
    right_mat = right_reshape.view(B_hd, N, N)  # same

    # Compute matrix multiply left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # [B*H, N, N]

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # [B,N,N,hidden_dim]

    # 6. Apply to_out_norm (layernorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # 7. Multiply by out_gate (broadcast)
    # Multiply out (B,N,N,hidden_dim) * out_gate (B,N,N,hidden_dim)
    out = out * out_gate

    # 8. Final linear projection to original dim
    out = torch.nn.functional.linear(out, to_out_weight)  # weight shape [dim, hidden_dim] => output shape [B,N,N,dim]

    return out

"""
TriMul "outgoing" forward pass optimized with a Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension.
2. Linear projections (left, right) and linear gates (left_gate, right_gate,
   out_gate) using the provided weights.
3. Fuse element‑wise masking and gating with a Triton kernel:
       left_out  = left  * left_gate  * mask
       right_out = right * right_gate * mask
   The mask has shape [B, N, N] and is broadcast over the hidden dimension.
4. Compute the core O(N³) operation as a batched matrix multiplication:
       out[b, d] = left_out[b, d] @ right_out[b, d].T
   where `d` iterates over the hidden dimension.  This is performed by
   reshaping the tensors and calling `torch.bmm`, which uses highly‑optimized
   cuBLAS kernels.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate, and project back to the original feature dimension.

Only the mask‑gate fusion is performed in Triton; the remaining
operations use PyTorch for readability and to leverage existing fast
kernels.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr,            # *float32   left projection   [B, N, N, H]
    right_ptr,           # *float32   right projection  [B, N, N, H]
    left_gate_ptr,       # *float32   left gate         [B, N, N, H]
    right_gate_ptr,      # *float32   right gate        [B, N, N, H]
    mask_ptr,            # *float32   mask              [B, N, N]
    out_left_ptr,        # *float32   output left       [B, N, N, H]
    out_right_ptr,       # *float32   output right      [B, N, N, H]
    total_elems,         # i64        total number of elements = B*N*N*H
    hidden_dim,          # i64        hidden dimension H
    BLOCK: tl.constexpr  # compile‑time constant, block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # mask index = flatten index without the hidden dimension
    mask_idx = offs // hidden_dim

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out_left = left * left_gate * mask_val
    out_right = right * right_gate * mask_val

    tl.store(out_left_ptr + offs, out_left, mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dict of torch.Tensors containing all model weights
            - config: dict with keys "dim" and "hidden_dim" and optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------- Load weights -------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]   # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]   # [hidden_dim]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]             # [dim, hidden_dim]

    # ------------------- 1. LayerNorm -------------------
    # F.layer_norm expects normalized_shape as a tuple of the dimensions to normalize
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------- 2. Projections -------------------
    # Linear layers without bias
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------- 3. Gates -------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask.to(dtype).contiguous()

    # Output buffers for the fused mask‑gate operation
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # ------------------- 4. Triton kernel: mask & gate fusion -------------------
    total_elements = left.numel()  # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_f,
        out_left_ptr=left_fused,
        out_right_ptr=right_fused,
        total_elems=total_elements,
        hidden_dim=hidden_dim,
        BLOCK=BLOCK,
    )

    # ------------------- 5. Core O(N³) via batched GEMM -------------------
    # reshape to (B*H, N, N) and perform left @ right^T for each hidden slice
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous()

    B = left_perm.size(0)
    H = left_perm.size(1)
    N = left_perm.size(2)

    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N).T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- 6. Second LayerNorm -------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------- 7. Apply output gate -------------------
    out = out * out_gate

    # ------------------- 8. Final linear projection -------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
