def mask_gate_kernel(
    X_ptr, Gate_ptr, Mask_ptr, Out_ptr,
    B, N, D,
    stride_xb, stride_xi, stride_xk, stride_xd,
    stride_gb, stride_gi, stride_gk, stride_gd,
    stride_mb, stride_mi, stride_mk,
    total_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # compute multi-dim indices
    # Use integer division
    # compute b = (offs // (N * N * D)), remainder r = offs % (N*N*D)
    # compute i = (r // (N * D)), remainder r2 = r % (N*D)
    # compute k = (r2 // D)
    # compute d = r2 % D

    stride_i = N * D
    stride_b = N * N * D

    b = offs // stride_b
    r = offs % stride_b
    i = r // stride_i
    r2 = r % stride_i
    k = r2 // D
    d = r2 % D

    # compute pointers
    x_ptr = X_ptr + b * stride_xb + i * stride_xi + k * stride_xk + d * stride_xd
    g_ptr = Gate_ptr + b * stride_gb + i * stride_gi + k * stride_gk + d * stride_gd

    m_ptr = Mask_ptr + b * stride_mb + i * stride_mi + k * stride_mk

    # Load

    g = tl.load(g_ptr, mask=mask, other=0.0)
    m = tl.load(m_ptr, mask=mask, other=0.0)  # mask shape broadcast
    # Multiply

"""
TriMul (outgoing) implementation with Triton acceleration.

The heavy O(N³·D) part (the triple‑sum) is performed with a batched GEMM
(`torch.bmm`) after projecting and gating the input.  The elementwise mask
application together with left/right gates is fused in a small Triton kernel
to demonstrate custom‑kernel usage.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: apply mask and per‑channel gate (left or right side)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    X_ptr,          # projected tensor (B,N,N,D)
    G_ptr,          # gate tensor        (B,N,N,D)
    M_ptr,          # mask tensor        (B,N,N)
    Out_ptr,        # output tensor      (B,N,N,D)
    B, N, D,
    stride_xb, stride_xi, stride_xk, stride_xd,
    stride_gb, stride_gi, stride_gk, stride_gd,
    stride_mb, stride_mi, stride_mk,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, VECT_D: tl.constexpr
):
    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # block over i (first N)
    pid_k = tl.program_id(2)          # block over k (second N)

    i_base = pid_i * BLOCK_M
    k_base = pid_k * BLOCK_K

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_k = k_base + tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_k = offs_k < N
    valid_ik = mask_i[:, None] & mask_k[None, :]

    # ----- load mask (broadcasted over the hidden dimension) -----
    m_ptrs = M_ptr + pid_b * stride_mb + offs_i[:, None] * stride_mi + offs_k[None, :] * stride_mk
    m_val = tl.load(m_ptrs, mask=valid_ik, other=0.0).to(tl.float32)   # (BM, BK)

    # ----- iterate over hidden dimension in chunks of VECT_D -----
    for d_off in range(0, D, VECT_D):
        d_range = tl.arange(0, VECT_D) + d_off
        d_mask = d_range < D

        # pointers for X, G and output
        x_ptrs = (X_ptr + pid_b * stride_xb
                  + offs_i[:, None, None] * stride_xi
                  + offs_k[None, :, None] * stride_xk
                  + d_range[None, None, :] * stride_xd)
        g_ptrs = (G_ptr + pid_b * stride_gb
                  + offs_i[:, None, None] * stride_gi
                  + offs_k[None, :, None] * stride_gk
                  + d_range[None, None, :] * stride_gd)
        out_ptrs = (Out_ptr + pid_b * stride_xb
                    + offs_i[:, None, None] * stride_xi
                    + offs_k[None, :, None] * stride_xk
                    + d_range[None, None, :] * stride_xd)

        # load
        x = tl.load(x_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & d_mask[None, None, :], other=0.0)
        g = tl.load(g_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & d_mask[None, None, :], other=0.0)

        # broadcast mask over hidden dim and apply
        out = x * g * m_val[:, :, None]

        tl.store(out_ptrs, out, mask=mask_i[:, None, None] & mask_k[None, :, None] & d_mask[None, None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]  (float32 or bfloat16)
            - mask_tensor:  [B, N, N]       (float or bool)
            - weights_dict: model weights with keys:
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight',
                'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config_dict: contains 'dim' and 'hidden_dim'

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])    # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # Gates (sigmoid) – also linear over the original dim
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # (B,N,N,hidden)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # Prepare mask (float) – shape (B,N,N,1) will be broadcast in kernel
    # ------------------------------------------------------------------
    mask_float = mask.to(dtype).contiguous()
    # ensure contiguous for stride sanity
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # ------------------------------------------------------------------
    # Triton kernel: apply mask and respective gates to left & right
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    BLOCK_M = 32
    BLOCK_K = 32
    VECT_D = 8  # number of hidden channels processed per iteration

    # allocate output buffers
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # grid dimensions: (batch, i‑tiles, k‑tiles)
    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_K))

    # left side
    mask_gate_kernel[grid](
        left_ptr=left,
        G_ptr=left_gate,
        M_ptr=mask_float,
        Out_ptr=left_out,
        B=B, N=N, D=hidden_dim,
        stride_xb=left.stride(0), stride_xi=left.stride(1), stride_xk=left.stride(2), stride_xd=left.stride(3),
        stride_gb=left_gate.stride(0), stride_gi=left_gate.stride(1), stride_gk=left_gate.stride(2), stride_gd=left_gate.stride(3),
        stride_mb=mask_float.stride(0), stride_mi=mask_float.stride(1), stride_mk=mask_float.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_K=BLOCK_K, VECT_D=VECT_D,
    )

    # right side
    mask_gate_kernel[grid](
        left_ptr=right,
        G_ptr=right_gate,
        M_ptr=mask_float,
        Out_ptr=right_out,
        B=B, N=N, D=hidden_dim,
        stride_xb=right.stride(0), stride_xi=right.stride(1), stride_xk=right.stride(2), stride_xd=right.stride(3),
        stride_gb=right_gate.stride(0), stride_gi=right_gate.stride(1), stride_gk=right_gate.stride(2), stride_gd=right_gate.stride(3),
        stride_mb=mask_float.stride(0), stride_mi=mask_float.stride(1), stride_mk=mask_float.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_K=BLOCK_K, VECT_D=VECT_D,
    )

    # ------------------------------------------------------------------
    # Triple‑sum: batched GEMM over the hidden dimension
    # out[b,i,j,d] = sum_k left_out[b,i,k,d] * right_out[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B*D, N, N) for left and right
    left_reshape = left_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matmul: (B*D, N, N) @ (B*D, N, N).T -> (B*D, N, N)
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Final LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    # ------------------------------------------------------------------
    # Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out.to(torch.float32)
