@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, bc,
    stride_left_bc, stride_left_i, stride_left_k,
    stride_right_bc, stride_right_k, stride_right_j,
    stride_out_bc, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bc = tl.program_id(0)   # which matrix
    pid_m = tl.program_id(1)    # tile row index
    pid_n = tl.program_id(2)    # tile col index

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Load block of left
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + pid_bc * stride_left_bc + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & (offs_k < N), other=0.0).to(tl.float16)  # maybe load as fp16

        # Load block of right
        # Right shape (K rows, N cols). So we offset rows by offs_k and cols by offs_n.
        right_ptrs = right_ptr + pid_bc * stride_right_bc + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        # shape (BLOCK_K, BLOCK_N)
        right = tl.load(right_ptrs, mask=(offs_k < N)[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # Compute dot product
        acc += tl.dot(left, right)   # size (BLOCK_M, BLOCK_N) result, tl.dot works on two dims: expects left: (M,K) right: (K,N)
        # advance pointers if contiguous (but we use offsets each iteration)
    # After loop, store result
    out_ptrs = out_ptr + pid_bc * stride_out_bc + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]   # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm x
    # Using torch.nn.functional.layer_norm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (weight shape [hidden_dim, dim])
    # Use F.linear: y = x @ weight.T
    # x shape [B,N,N,dim]
    left = F.linear(x, left_proj_weight)   # -> [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask if provided (mask can be None?) Actually mask input always provided.
    # In config["nomask"]: there might be mask as all ones; we just apply.
    # mask shape [B,N,N]; unsqueeze to broadcast: [B,N,N,1]
    mask_expand = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_expand
    right = right * mask_expand

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel compute
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to shape (B, hidden_dim, N, N) suitable for matmul
    # left: [B,N,N,hidden_dim] -> [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right need to be transposed on last two dims: shape [B,N,N,hidden_dim] -> [B, hidden_dim, N, N] where dims are (k,j)
    # Right transpose: permute (0,3,2,1)
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    B, H, N, _ = left_perm.shape   # underscore should be N

    # Flatten batch and hidden_dim
    bc = B * H
    left_flat = left_perm.view(bc, N, N)
    right_flat = right_perm.view(bc, N, N)

    # Create output tensor (float16)
    out_flat = torch.empty_like(left_flat)   # will be half

    # Launch Triton kernel
    # Determine block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (bc, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        bc,

import triton
import triton.language as tl

@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, bc,
    stride_left_bc, stride_left_i, stride_left_k,
    stride_right_bc, stride_right_k, stride_right_j,
    stride_out_bc, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bc = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dim
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # left offset: bc * stride_bc + offs_m[:,None]*stride_i + offs_k[None,:]*stride_k
        left_ptrs = left_ptr + pid_bc * stride_left_bc + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # Load right tile: shape (BLOCK_K, BLOCK_N)
        # right shape (K rows, N cols)
        right_ptrs = right_ptr + pid_bc * stride_right_bc + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        # Compute dot product (float32 accumulation)
        acc += tl.dot(left, right)  # left shape (M,K), right (K,N) returns (M,N) in float32
    # End for

    # Store result
    out_ptrs = out_ptr + pid_bc * stride_out_bc + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """...
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Load weights onto device
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = F.linear(x, left_proj_weight)   # [B,N,N,H]
    right = F.linear(x, right_proj_weight)

    # Apply mask if provided (mask can be None)
    if mask is None:
        mask_expand = torch.ones_like(left[..., :1])
    else:
        mask_expand = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_expand
    right = right * mask_expand

    # Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to make rows/cols suitable for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N] (i rows, k cols)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B,H,N,N] (k rows, j cols)

    B, H, N, _ = left_perm.shape
    bc = B * H

    left_flat = left_perm.view(bc, N, N)
    right_flat = right_perm.view(bc, N, N)

    out_flat = torch.empty_like(left_flat)  # half

    # Define block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (bc, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        bc,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out = out.to(torch.float32)

    # to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

"""
TriMul (outgoing) implementation with Triton acceleration.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched matrix multiplication per hidden channel:
    left[b,d,i,k] @ right[b,d,k,j]  ->  out[b,d,i,j]
using a custom Triton kernel that tiles the output (i,j) and the reduction
dimension k. All other operations (LayerNorm, linear projections, gating)
are done with PyTorch. The kernel accumulates in float32 while the inputs
are stored in float16 for memory‑bandwidth efficiency.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,                    # *[bc, N, N]  (half)
    right_ptr,                   # *[bc, N, N]  (half)
    out_ptr,                     # *[bc, N, N]  (half)
    N,                           # sequence length
    bc,                          # batch * hidden_dim
    stride_left_bc, stride_left_i, stride_left_k,
    stride_right_bc, stride_right_k, stride_right_j,
    stride_out_bc, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bc = tl.program_id(0)          # which matrix (batch * hidden)
    pid_m  = tl.program_id(1)          # tile row index
    pid_n  = tl.program_id(2)          # tile col index

    # Offsets for the (i,j) tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile  [M, K]
        left_ptrs = left_ptr + pid_bc * stride_left_bc \
                    + offs_m[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right tile [K, N] (right is stored as (k, j))
        right_ptrs = right_ptr + pid_bc * stride_right_bc \
                     + offs_k[:, None] * stride_right_k \
                     + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate product (fp32)
        acc += tl.dot(left, right)

    # Write the output tile
    out_ptrs = out_ptr + pid_bc * stride_out_bc \
               + offs_m[:, None] * stride_out_i \
               + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:        torch.Tensor [B, N, N] (may be None)
            - weights:     dict of model parameters
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- Load model weights -----
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)

    left_proj_weight  = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight  = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight   = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device)

    to_out_weight = weights["to_out.weight"].to(device)

    # ----- LayerNorm on input -----
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ----- Linear projections -----
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Apply mask (if any) -----
    if mask is None:
        mask_exp = torch.ones_like(left[..., :1])
    else:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
    left  = left * mask_exp
    right = right * mask_exp

    # ----- Gating -----
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ----- Cast to half for the contraction -----
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange so that each hidden channel is a separate square matrix:
    # left : (B, hidden, N, N)   rows = i, cols = k
    # right: (B, hidden, N, N)   rows = k, cols = j (transposed)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    bc = B * H

    # Flatten batch * hidden into a single leading dimension
    left_flat  = left_perm.view(bc, N, N)
    right_flat = right_perm.view(bc, N, N)
    out_flat   = torch.empty_like(left_flat)   # half precision

    # ----- Triton launch parameters -----
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (bc, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        bc,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ----- Reshape back to original layout -----
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out = out.to(torch.float32)

    # ----- Post‑processing (LayerNorm, output gate, final linear) -----
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
