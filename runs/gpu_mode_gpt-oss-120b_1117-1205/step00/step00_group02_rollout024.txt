@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # flatten over (b, i, j)
    total = B * N * N
    # Compute b,i,j from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for hidden dimension
    offs_h = tl.arange(0, BLOCK_H)

    # mask for hidden dimension
    mask_h = offs_h < H

    # accumulator
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets for block of k
        k_start = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_start < N

        # compute left and right pointers
        # left: [b, i, k, h]
        left_offsets = (b * stride_left_b
                        + i * stride_left_i
                        + k_start[:, None] * stride_left_k
                        + offs_h[None, :] * stride_left_h)  # shape [BK, BH]
        right_offsets = (b * stride_right_b
                         + j * stride_right_j
                         + k_start[:, None] * stride_right_k
                         + offs_h[None, :] * stride_right_h)

        # load left and right
        left = tl.load(left_ptr + left_offsets, mask=mask_k[:, None] & mask_h[None, :], other=0.0)
        right = tl.load(right_ptr + right_offsets, mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        # reshape to (BK, BH)
        # we need to multiply elementwise and sum over k dimension.
        # Compute per hidden dimension dot product:
        # For each h: sum_{k} left[k, h] * right[k, h]
        # We can use tl.dot of two vectors of length BLOCK_K? But each h separately.
        # Simpler: elementwise multiply and accumulate sum over k dimension.
        # We can accumulate: acc[h] += tl.sum(left[:, h] * right[:, h])
        # Compute product:
        prod = left * right
        # Sum over k dimension axis=0
        # Use tl.sum
        acc += tl.sum(prod, axis=0)  # shape (BH,)

    # write out to out tensor at (b,i,j, h)
    out_offsets = (b * stride_out_b
                   + i * stride_out_i
                   + j * stride_out_j
                   + offs_h * stride_out_h)
    tl.store(out_ptr + out_offsets, acc, mask=mask_h)

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_ij = tl.program_id(0)
    pid_h = tl.program_id(1)

    # compute indices
    total_ij = B * N * N
    b = pid_ij // (N * N)
    rem = pid_ij % (N * N)
    i = rem // N
    j = rem % N

    # hidden offset
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # accumulator
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # block of k
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # compute offsets
        left_offsets = (b * stride_left_b
                        + i * stride_left_i
                        + cur_k[:, None] * stride_left_k
                        + offs_h[None, :] * stride_left_h)
        right_offsets = (b * stride_right_b
                         + j * stride_right_j
                         + cur_k[:, None] * stride_right_k
                         + offs_h[None, :] * stride_right_h)

        left = tl.load(left_ptr + left_offsets, mask=mask_k[:, None] & mask_h[None, :], other=0.0)
        right = tl.load(right_ptr + right_offsets, mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        # accumulate dot product
        # multiply elementwise and sum over k dimension: sum_k left_kh * right_kh
        prod = left * right
        acc += tl.sum(prod, axis=0)

    # store result
    out_offsets = (b * stride_out_b
                   + i * stride_out_i
                   + j * stride_out_j
                   + offs_h * stride_out_h)
    tl.store(out_ptr + out_offsets, acc, mask=mask_h)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, N2, dim2 = input_tensor.shape

    assert N == N2
    assert dim2 == dim

    device = input_tensor.device

    # LayerNorm for input
    # We need to apply across last dim dim.
    # Compute using F.layer_norm: with normalized_shape=dim
    # Input shape [B,N,N,dim]
    # For weight, bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # apply layer norm per last dimension
    # Nor requirement: we need to use torch.nn.functional.layer_norm.
    import torch.nn.functional as F
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projection: left/right
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # These weight shapes: [hidden_dim, dim]
    left = F.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Mask: if mask is not None (or config["nomask"]?), but mask passed always.
    # mask shape [B,N,N]; unsqueeze -1.
    # Cast mask to same dtype as left (float32)
    mask_f = mask.to(x.dtype)
    mask_f = mask_f.unsqueeze(-1)  # shape [B,N,N,1]

    # Apply mask and gating
    left = left * mask_f
    right = right * mask_f

    # Gates:
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute out via Triton kernel
    # allocate out tensor
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # compute strides
    stride_l = left.stride()  # returns (N*N*H, N*H, H, 1)
    stride_r = right.stride()
    stride_out = out.stride()

    # define BLOCK_H size up to hidden_dim, maybe 64 or 32
    # choose BLOCK_H = 64 (max hidden_dim 384)
    BLOCK_H = 64
    BLOCK_K = 32  # step size over k dimension

    # grid
    total_ij = bs * N * N
    h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (total_ij, h_blocks)

    # launch kernel
    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        bs,
        N,
        hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
    )

    # Now apply to_out_norm and to_out linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate: shape [B,N,N,1] from earlier out_gate variable (which is hidden_dim? Actually out_gate from config dims: Linear(dim->hidden_dim)
    # out_gate shape [B,N,N,hidden_dim] (same hidden_dim). Wait out_gate is gating for final output, shape same hidden_dim. The original code after to_out_norm does: out = out * out_gate ; then self.to_out(out) which maps hidden_dim->dim.
    # Therefore out_gate shape [B, N, N, hidden_dim]
    # So we need to compute out_gate earlier (we already have out_gate as gating values). Multiply out = out * out_gate.

    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]

    return output

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
core N³ reduction.

The kernel computes, for each batch b, sequence indices i, j and for a
block of hidden channels h:

    out[b, i, j, h] = sum_{k=0}^{N-1} left[b, i, k, h] * right[b, j, k, h]

All surrounding LayerNorms, Linear projections and gating are
performed with regular PyTorch ops (highly‑optimized cuBLAS/cuDNN).  The
mask (if present) is applied before the Triton kernel to keep the
kernel simple.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: block‑wise reduction over the k dimension.
# -------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,               # tensor pointers
    B, N, H,                                    # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids: (b,i,j) plane and hidden‑channel block
    pid_ij = tl.program_id(0)
    pid_h  = tl.program_id(1)

    # decode (b,i,j) from flattened pid
    total_ij = B * N * N
    b = pid_ij // (N * N)
    rem = pid_ij % (N * N)
    i = rem // N
    j = rem % N

    # hidden‑channel offsets for this block
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # accumulator for the dot‑product over k
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # loop over k in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # offsets for left[b,i,k,:] and right[b,j,k,:]
        offs_left = (b * stride_l_b
                     + i * stride_l_i
                     + cur_k[:, None] * stride_l_k
                     + offs_h[None, :] * stride_l_h)
        offs_right = (b * stride_r_b
                      + j * stride_r_j
                      + cur_k[:, None] * stride_r_k
                      + offs_h[None, :] * stride_r_h)

        # load with masking on k and h dimensions
        left = tl.load(left_ptr + offs_left,
                       mask=mask_k[:, None] & mask_h[None, :],
                       other=0.0)
        right = tl.load(right_ptr + offs_right,
                        mask=mask_k[:, None] & mask_h[None, :],
                        other=0.0)

        # elementwise product and reduction over k
        prod = left * right
        acc += tl.sum(prod, axis=0)   # sum over the k axis

    # store the result for this (b,i,j) and hidden‑block
    offs_out = (b * stride_o_b
                + i * stride_o_i
                + j * stride_o_j
                + offs_h * stride_o_h)
    tl.store(out_ptr + offs_out, acc, mask=mask_h)


# -------------------------------------------------------------
# Public entry point required by the benchmark harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input, mask, weights, config)
            input   – Tensor [B, N, N, dim]
            mask    – Tensor [B, N, N] (may be all‑ones when config["nomask"] is True)
            weights – dict of pre‑loaded weight tensors
            config  – dict containing "dim" and "hidden_dim"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, N2, dim_ = input_tensor.shape
    assert N == N2 and dim == dim_, "Shape mismatch"

    # ---------------------------------------------------------
    # 1) Input layer‑norm
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------------------------------------------------------
    # 2) Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3) Mask (if present)
    # ---------------------------------------------------------
    if config.get("nomask", False):
        mask_f = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)
    else:
        mask_f = mask.to(x.dtype).unsqueeze(-1)         # [B,N,N,1]

    left = left * mask_f
    right = right * mask_f

    # ---------------------------------------------------------
    # 4) Gating (sigmoid activations)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5) Core Triton reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ---------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (contiguous layout = (N*N*H, N*H, H, 1))
    stride_l = left.stride()   # (b,i,k,h)
    stride_r = right.stride()
    stride_o = out.stride()

    BLOCK_H = 64          # hidden‑channel block size (covers up to 384)
    BLOCK_K = 32          # k‑loop block size

    total_ij = B * N * N
    h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (total_ij, h_blocks)

    _trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------
    # 6) Output layer‑norm, gating and final linear projection
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
