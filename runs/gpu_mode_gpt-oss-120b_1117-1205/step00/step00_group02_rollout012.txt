@triton.jit
def triton_matmul_kernel(A_ptr, B_ptr, C_ptr,
                         BATCH, M, N, K,
                         stride_Ab, stride_Am, stride_Ak,
                         stride_Bb, stride_Bn, stride_Bk,
                         stride_Cb, stride_Cm, stride_Cn,
                         BLOCK_M: tl.constexpr,
                         BLOCK_N: tl.constexpr,
                         BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for bounds
    m_mask = offs_m < M
    n_mask = offs_n < N

    # Pointer offsets for current block dynamics
    # We'll loop over k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0)

        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + pid_batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        # Compute dot product and accumulate
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # Write back to C
    c_ptrs = C_ptr + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=m_mask[:, None] & n_mask[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if nomask:
        # ignore mask, use ones
        mask_f = 1.0
    else:
        mask_f = mask_tensor.to(left.dtype).unsqueeze(-1)  # shape [B,N,N,1]
    if not nomask:
        left = left * mask_f
        right = right * mask_f
    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # prepare for kernel
    # permute to [B,H,N,N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    B = left_perm.shape[0]
    H = left_perm.shape[1] # hidden_dim
    N = left_perm.shape[2]

    # flatten batch dimension
    batch_total = B * H
    left_flat = left_perm.view(batch_total, N, N).contiguous()
    right_flat = right_perm.view(batch_total, N, N).contiguous()

    # convert to half for kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # allocate output
    out_flat = torch.empty((batch_total, N, N), dtype=torch.float32, device=device)

    # launch kernel
    import triton
    import triton.language as tl

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    # compute strides
    stride_Ab = left_fp16.stride(0)
    stride_Am = left_fp16.stride(1)
    stride_Ak = left_fp16.stride(2)
    stride_Bb = right_fp16.stride(0)
    stride_Bn = right_fp16.stride(1)
    stride_Bk = right_fp16.stride(2)
    stride_Cb = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    triton_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_flat,
        BATCH=batch_total,
        M=N,
        N=N,
        K=N,
        stride_Ab=stride_Ab,
        stride_Am=stride_Am,
        stride_Ak=stride_Ak,
        stride_Bb=stride_Bb,
        stride_Bn=stride_Bn,
        stride_Bk=stride_Bk,
        stride_Cb=stride_Cb,
        stride_Cm=stride_Cm,
        stride_Cn=stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape output back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    # to_out_norm
    out = torch.nn.functional.layer_norm(out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # out gate multiplication
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

import torch
import triton
import triton.language as tl

@triton.jit
def triton_matmul_kernel(A_ptr, B_ptr, C_ptr,
                         BATCH, M, N, K,
                         stride_Ab, stride_Am, stride_Ak,
                         stride_Bb, stride_Bn, stride_Bk,
                         stride_Cb, stride_Cm, stride_Cn,
                         BLOCK_M: tl.constexpr,
                         BLOCK_N: tl.constexpr,
                         BLOCK_K: tl.constexpr):
    """Batched matrix multiplication C = A @ B^T.
    A,B shapes: (batch, M, K) and (batch, N, K). Output C: (batch, M, N).
    All tensors are assumed contiguous; A and B are half, C is float32.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    m_mask = offs_m < M
    n_mask = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < K

        # Load A tile (M x K)
        a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + k_offset[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0)

        # Load B tile (N x K) as (K x N) for dot-product
        b_ptrs = B_ptr + pid_batch * stride_Bb + k_offset[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        # Accumulate product
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # Write out
    c_ptrs = C_ptr + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=m_mask[:, None] & n_mask[None, :])

"""
TriMul forward pass with a custom Triton kernel.

The heavy part of the TriMul module is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiply:
    out[b,d] = left[b,d] @ right[b,d]^T   (M = N = seq_len, K = seq_len)

We reshape the tensors so that the batch dimension becomes (batch * hidden_dim)
and launch a Triton kernel that performs a standard tiled matmul
with FP16 inputs and FP32 accumulation. The rest of the module
(layer‑norms, linear projections, gates) is executed in PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ  (batched)

    A, B: (batch, M, K) and (batch, N, K)   (FP16)
    C: (batch, M, N)                         (FP32)
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # Load A tile (M x K)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_Ab
            + offs_m[:, None] * stride_Am
            + k_off[None, :] * stride_Ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile as (K x N) to match tl.dot contract
        b_ptrs = (
            B_ptr
            + pid_batch * stride_Bb
            + k_off[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate product (a: MxK, b: KxN)
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = (
        C_ptr
        + pid_batch * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module with a Triton kernel for the O(N³) contraction.

    Args:
        data (Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]):
            - input:   Tensor of shape [B, N, N, dim]
            - mask:    Tensor of shape [B, N, N] (bool or float)
            - weights: dict containing all linear / layer‑norm parameters
            - config:  dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for the Triton kernel ----------
    # Shape: [B, N, N, hidden_dim] -> permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B = left_perm.shape[0]
    H = left_perm.shape[1]  # hidden_dim
    N = left_perm.shape[2]

    # Merge batch and hidden dimensions -> (B*H, N, N)
    batch_total = B * H
    left_flat = left_perm.view(batch_total, N, N).contiguous()
    right_flat = right_perm.view(batch_total, N, N).contiguous()

    # Cast to FP16 for the kernel (inputs) while keeping FP32 accumulator
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_flat = torch.empty((batch_total, N, N), dtype=torch.float32, device=device)

    # Strides for the kernel (in elements, not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Bn, stride_Bk = right_fp16.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    # ----- Triton launch parameters -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_flat,
        batch_total,
        N,
        N,
        N,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bn,
        stride_Bk,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output layer norm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
