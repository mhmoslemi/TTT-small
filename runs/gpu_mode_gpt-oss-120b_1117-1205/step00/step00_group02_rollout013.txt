"""
TriMul (outgoing) implementation with a fused TensorCore kernel.

The heavy part of TriMul is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
contraction (equivalent to a batch of N×N matrix multiplications for each hidden
channel).  This kernel computes that contraction in FP16 using Triton’s
`tl.dot` (TensorCore friendly) while accumulating in FP32 for precision.
All other operations (LayerNorm, Linear projections, gating) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = Σ_k left[..., i,k,h] * right[..., j,k,h] (TensorCore).

    Grid:
        (N/BLOCK_M, N/BLOCK_N, B*H)
    The third dimension encodes a combined batch index and hidden‑channel index.
    """
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_bc = tl.program_id(2)         # batch * hidden channel

    # decode batch and hidden channel
    b = pid_bc // H
    h = pid_bc % H

    # offsets within the N dimension
    i_offsets = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offsets = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # left tile  (M, K)  : indices (i, k)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + i_offsets[:, None] * stride_l_i
            + k_offsets[None, :] * stride_l_k
            + h * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # right tile (K, N)  : transposed indices (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + k_offsets[:, None] * stride_r_k
            + j_offsets[None, :] * stride_r_i
            + h * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # TensorCore matmul: (M,K) @ (K,N) → (M,N), accumulation in FP32
        acc += tl.dot(left, right)

    # write result
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + i_offsets[:, None] * stride_o_i
        + j_offsets[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    TriMul forward pass with a fused Triton kernel for the N³ contraction.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N]  (bool or float) or None
            - weights_dict: dict mapping layer names to torch.Tensor weights
            - config_dict : dict with keys "dim", "hidden_dim" and others

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # LayerNorm on the input (dim -> dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # fp32

    # -----------------------------------------------------------------
    # Linear projections (dim -> hidden_dim) – keep in FP32 for now
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)      # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # Gating and mask
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    if mask is not None:
        # mask broadcast to hidden dimension
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to FP16 and launch Triton kernel for the contraction
    # -----------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_fp16.shape

    out_fp16 = torch.empty_like(left_fp16)  # shape [B,N,N,hidden]

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    _trimul_contract_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to FP32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # Output LayerNorm (hidden -> hidden)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # Apply output gate (still in FP32)
    out = out * out_gate

    # -----------------------------------------------------------------
    # Final Linear projection (hidden_dim -> dim)
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)  # [B,N,N,dim]

    return out
