@triton.jit
def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    stride_a_batch: tl.constexpr, stride_a_row: tl.constexpr, stride_a_col: tl.constexpr,
    stride_b_batch: tl.constexpr, stride_b_row: tl.constexpr, stride_b_col: tl.constexpr,
    stride_c_batch: tl.constexpr, stride_c_row: tl.constexpr, stride_c_col: tl.constexpr
):
    pid_b = tl.program_id(2)  # batch*hidden index
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to each batch
    a_ptrs = A_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_row + tl.arange(0, BLOCK_K)[None, :] * stride_a_col
    # But we need K loop, we load inside for loop.
    # We'll compute accumulaton.
    
    # Initialize accumulate in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for K dimension within this block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = offs_k < N

        # Load A block: shape M x K
        a_ptrs = A_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        # Use mask to load zero where OOB
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block: shape K x N (since B is transposed, its row dimension is K, col dimension is N)
        b_ptrs = B_ptr + pid_b * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # tl.dot expects shape (M, K) and (K, N), returns (M,N)
    # store result
    c_ptrs = C_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])
```

"""TriMul forward with Triton acceleration.

The implementation follows the reference PyTorch module:
* LayerNorm on the input.
* Linear projections (left/right) and gates (left/right/out) without bias.
* Mask application.
* Gated left/right tensors are contracted with an O(N³) einsum:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  This contraction is performed as a batched matrix multiplication
  L @ Rᵀ using a custom Triton kernel that works on a (B·H, N, N)
  flattened view where H = hidden_dim.  The kernel loads blocks of
  size (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N) in float16, accumulates
  in float32 and writes the result in float16.  After the kernel we
  reshape, apply a second LayerNorm, out‑gate, and a final linear
  projection back to the original dimensionality.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul L (M×K) @ Rᵀ (K×N) -> C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    stride_a_batch: tl.constexpr, stride_a_row: tl.constexpr, stride_a_col: tl.constexpr,
    stride_b_batch: tl.constexpr, stride_b_row: tl.constexpr, stride_b_col: tl.constexpr,
    stride_c_batch: tl.constexpr, stride_c_row: tl.constexpr, stride_c_col: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch*hidden index
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)

    # --- offsets for the current block ---
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer to the start of this batch
    a_batch_ptr = A_ptr + pid_b * stride_a_batch
    b_batch_ptr = B_ptr + pid_b * stride_b_batch
    c_batch_ptr = C_ptr + pid_b * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (M × K)
        a_ptrs = a_batch_ptr \
                 + offs_m[:, None] * stride_a_row \
                 + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        # Load B block (K × N) – note B is already transposed (K rows, N cols)
        b_ptrs = b_batch_ptr \
                 + offs_k[:, None] * stride_b_row \
                 + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)

    # write back
    c_ptrs = c_batch_ptr \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: mapping from weight names to tensors
            - config_dict: must contain "dim" and "hidden_dim"
    Returns:
        output: torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    mask = mask.unsqueeze(-1).to(x.dtype)            # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the Triton contraction
    #    Convert to float16, bring the hidden dimension to the second axis,
    #    make the (B·H, N, N) view, and transpose the right tensor.
    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy multiplication.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # (B, H, N, N) contiguous
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    # Transpose the last two axes of the right tensor so that we can do
    # a regular matmul L @ Rᵀ.
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()

    # Flatten the batch and hidden dimensions.
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_T.view(BH, N, N)

    # Output buffer (half‑precision)
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),     # rows
        triton.cdiv(N, BLOCK_N),     # cols
        BH,                          # batch*hidden
    )
    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        stride_a_batch=left_flat.stride(0),
        stride_a_row=left_flat.stride(1),
        stride_a_col=left_flat.stride(2),
        stride_b_batch=right_flat.stride(0),
        stride_b_row=right_flat.stride(1),
        stride_b_col=right_flat.stride(2),
        stride_c_batch=out_flat.stride(0),
        stride_c_row=out_flat.stride(1),
        stride_c_col=out_flat.stride(2),
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])
    return output
