    c_ptr,   # out slice pointer

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    stride_am, stride_ak, # for a: stride_i, stride_k
    stride_bk, stride_bn, # for b: stride_k, stride_n
    stride_cm, stride_cn  # for c: stride_i, stride_j

    pid_n = tl.program_id(1)
    pid_batch = tl.program_id(2)  # batch*hidden index

    # compute offset to start for this batch slice
    a_ptr += pid_batch * stride_abatch   # offset for a_base
    b_ptr += pid_batch * stride_abatch   # same offset for b? For c later.
    c_ptr += pid_batch * stride_cbatch

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over dim
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )
    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask
    mask_unsq = mask.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute contraction with Triton kernel
    # Ensure tensors are contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # [B,N,N,hidden_dim]

    # Launch kernel
    B, N, _, H = left.shape
    # compute strides
    stride_abatch, stride_ai, stride_ak, stride_ad = left.stride()
    stride_bbatch, stride_bj, stride_bk, stride_bd = right.stride()
    stride_cbatch, stride_ci, stride_cj, stride_cd = out.stride()

    # For left and right, we use i stride = stride_ai, k stride = stride_ak
    # For right we treat as B_T: row stride = stride_bk, col stride = stride_bj
    # For out, i stride = stride_ci, j stride = stride_cj

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    trimul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N,
        hidden_dim,
        stride_abatch,
        stride_ai,
        stride_ak,
        stride_bbatch,
        stride_bk,
        stride_bj,
        stride_cbatch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # after kernel, get out (float32)
    # out_norm
    out_norm = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )
    out = out_norm * out_gate
    # final linear projection
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

"""
Triton implementation of the “outgoing” TriMul operator from AlphaFold3.

The heavy part of TriMul is the contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is a batched matrix multiplication for each hidden channel `h`.
We implement this contraction with a tiled GEMM kernel in Triton.
All other operations (LayerNorms, Linear projections, gates) are performed
with standard PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: compute left @ right^T for each (batch, hidden) slice.
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr,                     # left  tensor pointer
    b_ptr,                     # right tensor pointer
    c_ptr,                     # output tensor pointer
    N,                         # sequence length
    H,                         # hidden dimension (number of slices per batch)
    stride_a_batch, stride_ai, stride_ak,          # strides for left: (batch, i, k)
    stride_b_batch, stride_bk, stride_bj,          # strides for right as transposed: (batch, k, j)
    stride_c_batch, stride_ci, stride_cj,          # strides for output: (batch, i, j)
    BLOCK_M: tl.constexpr,     # tile size for i dimension
    BLOCK_N: tl.constexpr,     # tile size for j dimension
    BLOCK_K: tl.constexpr,     # tile size for k dimension
):
    pid_m = tl.program_id(0)   # block index in i
    pid_n = tl.program_id(1)   # block index in j
    pid_hb = tl.program_id(2)  # combined batch*hidden index

    # --------------------------------------------------------------
    # Decode batch and hidden indices from the combined program id.
    # --------------------------------------------------------------
    batch_idx = pid_hb // H
    hid_idx   = pid_hb % H

    # Base pointers for this (batch, hidden) slice.
    a_base = a_ptr + batch_idx * stride_a_batch + hid_idx
    b_base = b_ptr + batch_idx * stride_b_batch + hid_idx
    c_base = c_ptr + batch_idx * stride_c_batch + hid_idx

    # --------------------------------------------------------------
    # Offsets for the current tile.
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the tile.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the k dimension in BLOCK_K sized chunks.
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)

        # Load B block (interpreted as transposed, shape (BLOCK_K, BLOCK_N))
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)

        # Accumulate dot product over k.
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the computed tile to the output tensor.
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=(mask_i[:, None] & mask_j[None, :]))


# ----------------------------------------------------------------------
# Entry point used by the evaluator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton‑accelerated
    contraction.

    Args:
        data: Tuple (input, mask, weights, config)
            - input:  torch.Tensor of shape [B, N, N, dim]
            - mask:   torch.Tensor of shape [B, N, N] (binary or float)
            - weights: dict mapping weight names to torch.Tensors
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # Load model parameters.
    # --------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # --------------------------------------------------------------
    # LayerNorm on the input tensor.
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # Cast to float32 for the remaining ops (ensures precision).
    x = x.to(torch.float32)

    # --------------------------------------------------------------
    # Linear projections.
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # --------------------------------------------------------------
    # Apply mask (if any).  mask is expected to contain 0/1.
    # --------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    # --------------------------------------------------------------
    # Compute gates.
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Triton kernel: batched GEMM over the k dimension.
    # --------------------------------------------------------------
    B, N_, _, H_ = left.shape
    assert N_ == N, "Inconsistent sequence length."
    assert H_ == hidden_dim, "Hidden dimension mismatch."

    # Ensure contiguous tensors for efficient pointer arithmetic.
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # will hold [B, N, N, H]

    # Strides (in elements) of the tensors.
    stride_a_batch, stride_ai, stride_ak, stride_ad = left.stride()
    stride_b_batch, stride_bj, stride_bk, stride_bd = right.stride()
    stride_c_batch, stride_ci, stride_cj, stride_cd = out.stride()

    # Tiling parameters – chosen to fit typical GPU shared‑memory limits.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),               # i dimension
        triton.cdiv(N, BLOCK_N),               # j dimension
        B * hidden_dim                         # combined batch * hidden slices
    )

    trimul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N,
        hidden_dim,
        stride_a_batch, stride_ai, stride_ak,
        stride_b_batch, stride_bk, stride_bj,
        stride_c_batch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Final LayerNorm, gate, and projection back to original dim.
    # --------------------------------------------------------------
    out_norm = F.layer_norm(out,
                            (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)

    out = out_norm * out_gate
    output = F.linear(out, to_out_weight)  # projects back to `dim`

    return output
