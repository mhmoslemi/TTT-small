def custom_kernel(data):

    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])
    # mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute Einsum via Triton
    # Permute for GEMM
    left_t = left.permute(0,3,1,2).contiguous()
    right_t = right.permute(0,3,2,1).contiguous()  # swap i and k dims
    # shapes: (B, H, N, N)
    B, H, N, _ = left_t.shape
    # flatten
    left_flat = left_t.view(B*H, N, N)
    right_flat = right_t.view(B*H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Compute strides
    a_batch_stride, a_m_stride, a_k_stride = left_flat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_flat.stride()
    c_batch_stride, c_m_stride, c_n_stride = out_flat.stride()

    # BLOCK sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_M), B*H)

    # Launch
    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # shape (B,N,N,H)

    # Apply to_out_norm (layer norm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])

    out = out * out_gate

    # Final linear to dim
    out = torch.nn.functional.linear(out, weights['to_out.weight'])

    return out

def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_Ab, stride_Am, stride_Ak,
                          stride_Bb, stride_Bk, stride_Bn,
                          stride_Cb, stride_Cm, stride_Cn,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(0)  # block col
    batch = tl.program_id(2)  # batch index

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_ptr = A_ptr + batch * stride_Ab
    b_ptr = B_ptr + batch * stride_Bb
    c_ptr = C_ptr + batch * stride_Cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to A and B sub-tiles
        a = tl.load(a_ptr + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak,

        acc += tl.dot(a, b)

    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Move weights to device
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm over last dim
    # input shape (B,N,N,dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = F.linear(x, left_proj_weight)   # shape (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    out_flat = torch.empty_like(left_flat)

    a_batch_stride, a_m_stride, a_k_stride = left_flat.stride()

    c_batch_stride, c_m_stride, c_n_stride = out_flat.stride()

    # Block sizes (tune)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

        out_flat,

        b_n_stride,

        c_m_stride,

"""
Trilinear multiplicative update (TriMul) – outgoing version.

The heavy part of TriMul is the three‑tensor contraction

    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

for every batch `b` and hidden channel `d`.  This is equivalent to a batch of
matrix multiplications per hidden channel:
`out[b, d] = left[b, d] @ right[b, d].T` where each operand has shape
(N, N).  The kernel below carries out this batched GEMM in Triton, while the
surrounding linear layers, LayerNorms and gating are performed with PyTorch.

The kernel tiles the M (i) and N (j) dimensions with `BLOCK_M`×`BLOCK_N` blocks
and reduces over K (`BLOCK_K`).  A three‑dimensional grid (j‑block, i‑block,
batch×hidden) is launched, so each program computes a sub‑matrix of one
independent GEMM.  The implementation works for any `N ≤ 1024` and
`hidden_dim ≤ 768` used in the tests.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton batched GEMM kernel (A @ B.T) where
#   A : (batch, M, K)      -> left_flat  (M = N, K = N)
#   B : (batch, K, N)      -> right_flat (already transposed)
#   C : (batch, M, N)      -> out_flat
# -------------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes (all = seq_len)
    stride_Ab, stride_Am, stride_Ak,   # A: batch, row, col strides
    stride_Bb, stride_Bk, stride_Bn,   # B: batch, row(K), col(N) strides
    stride_Cb, stride_Cm, stride_Cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(1)  # block row index (i)
    pid_n = tl.program_id(0)  # block col index (j)
    batch = tl.program_id(2)   # which (batch, hidden) pair

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_ptr = A_ptr + batch * stride_Ab
    b_ptr = B_ptr + batch * stride_Bb
    c_ptr = C_ptr + batch * stride_Cb

    # Accumulator for the block
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K) and B block (K x N)
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write the result block back to C
    tl.store(
        c_ptr + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# -------------------------------------------------------------------------
# Entry point used by the evaluator
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict of model parameters
            - config: dict containing "dim", "hidden_dim", "nomask", ...

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # Load weights onto the correct device (they may be on CPU)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. LayerNorm over the channel dimension `dim`
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if not nomask:
        # mask: (B, N, N) -> (B, N, N, 1)
        mask_fp = mask.to(dtype).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton kernel for the contraction (einsum)
    #    Compute: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    #   Permute to (B, hidden, N, N) (contiguous) and transpose the right
    left_t = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N)  (k ↔ j)

    B, H, N, _ = left_t.shape
    batch_hidden = B * H

    # Flatten the first two dimensions so each (b, h) pair is a separate GEMM
    left_flat = left_t.view(batch_hidden, N, N)
    right_flat = right_t.view(batch_hidden, N, N)

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements, not bytes) for the three tensors
    a_batch_stride, a_m_stride, a_k_stride = left_flat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_flat.stride()
    c_batch_stride, c_m_stride, c_n_stride = out_flat.stride()

    # Tunable block sizes – work well for N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # j‑blocks
        triton.cdiv(N, BLOCK_M),   # i‑blocks
        batch_hidden,              # (batch, hidden) dimension
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        a_batch_stride,
        a_m_stride,
        a_k_stride,
        b_batch_stride,
        b_k_stride,
        b_n_stride,
        c_batch_stride,
        c_m_stride,
        c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure the kernel has finished before we touch `out_flat`
    torch.cuda.synchronize()

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
