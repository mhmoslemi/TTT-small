def trimul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # actually out strides: same as left? last is hidden_dim same.
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute batch and head index.
    b = pid_bh // H
    h = pid_bh % H

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # Offsets
    i_off = i_base + tl.arange(0, BLOCK_I)
    j_off = j_base + tl.arange(0, BLOCK_J)
    # Loop over K
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)

        # Mask for valid k
        mask_k = k_off < N

        # Left tile
        left_ptrs = left_ptr + b * stride_left_b + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k + h * stride_left_h
        left_mask = (i_off[:, None] < N) & mask_k[None, :]
        left_val = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)
        # Right tile
        right_ptrs = right_ptr + b * stride_right_b + j_off[:, None] * stride_right_i + k_off[None, :] * stride_right_k + h * stride_right_h
        right_mask = (j_off[:, None] < N) & mask_k[None, :]
        right_val = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # Multiply accumulate
        acc += tl.dot(left_val, right_val, trans_b=True)

    # Store result
    out_ptrs = out_ptr + b * stride_out_b + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j + h * stride_out_h
    out_mask = (i_off[:, None] < N) & (j_off[None, :] < N)
    tl.store(out_ptrs, acc, mask=out_mask)

TriMul forward pass with a Triton kernel for the high‑cost
pairwise multiplication:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

The kernel computes a batched GEMM (left @ rightᵀ) for each
(hidden) head using tiling over the i, j and k dimensions.
All other operations (LayerNorm, Linear projections, gates) are
performed in PyTorch. The final tensor has shape [B, N, N, dim] and
dtype float32.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _trimul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each (b, h) pair."""
    pid_bh = tl.program_id(0)   # combined batch * hidden_dim index
    pid_i = tl.program_id(1)    # block row (i)
    pid_j = tl.program_id(2)    # block column (j)

    # ----- decode batch and head indices -----
    b = pid_bh // H
    h = pid_bh % H

    # ----- base offsets for the i/j tiles -----
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    i_off = i_base + tl.arange(0, BLOCK_I)
    j_off = j_base + tl.arange(0, BLOCK_J)

    # masks for i / j dimensions (independent of k)
    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator for the tile
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # ----- load left tile (i, k) -----
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_off[:, None] * stride_left_i
            + k_off[None, :] * stride_left_k
            + h * stride_left_h
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # ----- load right tile (j, k) -----
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j_off[:, None] * stride_right_i
            + k_off[None, :] * stride_right_k
            + h * stride_right_h
        )
        right_mask = mask_j[:, None] & mask_k[None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # ------- accumulate dot product over k -------
        # left: (BLOCK_I, BLOCK_K), right: (BLOCK_J, BLOCK_K)
        # trans_b=True treats right as (BLOCK_K, BLOCK_J)
        acc += tl.dot(left, right, trans_b=True)

    # ----- store result tile -----
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j
        + h * stride_out_h
    )
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul module with a Triton kernel for the
    core i‑j‑k reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (float 0/1 or bool)
            - weights: dict of torch.Tensor containing all model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on the input -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # shape [B, N, N, dim], float32

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Apply mask (if any) -----
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(x)        # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gates (sigmoid of linear projections of the normalized input) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----- Gated tensors -----
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape

    # ----- Allocate output buffer -----
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ----- Strides (in elements) -----
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # ----- Triton launch configuration -----
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    _trimul_out_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing (LayerNorm, gating, final projection) -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # final shape [B, N, N, dim]

    return output
