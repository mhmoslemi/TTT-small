    out_ptr, out_gate_ptr, to_out_norm_weight_ptr, to_out_norm_bias_ptr,
    to_out_weight_ptr, final_ptr,
    B, N, H, D,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm over dim
    # compute mean and var per element across dim
    # Use torch.mean and var
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if provided and config["nomask"] is False
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Contraction using batch matmul
    B = left.shape[0]
    N = left.shape[1]
    # reshape
    left_mat = left.permute(0,3,1,2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).reshape(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1)  # shape [B,N,N,hidden_dim]

    # Ensure contiguity for kernel
    out = out.contiguous()
    out_gate = out_gate.contiguous()

    # Output tensor
    output = torch.empty((B, N, N, dim), dtype=torch.float32, device=device)

    # Load strides
    # out strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h = out_gate.stride()
    stride_norm_w_h = weights["to_out_norm.weight"].stride(0)
    stride_norm_b_h = weights["to_out_norm.bias"].stride(0)
    stride_to_out_d = weights["to_out.weight"].stride(0)
    stride_to_out_h = weights["to_out.weight"].stride(1)
    stride_out_final_b, stride_out_final_i, stride_out_final_j, stride_out_final_d = output.stride()

"""
TriMul (outgoing) kernel implementation.

Algorithm
---------
1. LayerNorm over the input channel dimension `dim`.
2. Linear projections `left` and `right` (+ optional mask).
3. Gating via learned sigmoid projections.
4. Contraction:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   performed as a batched matrix multiplication using PyTorch's
   highly‑optimized `torch.bmm`.
5. For every (b,i,j) a Triton kernel fuses:
   * LayerNorm over the hidden dimension `hidden_dim`.
   * Multiplication with the outgoing gate.
   * Final linear projection to `dim`.
   The kernel processes one (b,i,j) element at a time, handling the
   hidden‑dim reduction and the `dim`‑dim projection in a tiled fashion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: per‑pair (batch, row i, col j) processing
# --------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    out_ptr,                 # pointer to contracted tensor [B,N,N,H]
    gate_ptr,                # pointer to out_gate tensor  [B,N,N,H]
    norm_w_ptr,              # to_out_norm.weight   [H]
    norm_b_ptr,              # to_out_norm.bias     [H]
    proj_w_ptr,              # to_out.weight        [D,H]
    out_ptr_final,           # pointer to final output [B,N,N,D]

    B, N, H, D,              # runtime dimensions
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_w_h, stride_norm_b_h,
    stride_proj_d, stride_proj_h,
    stride_out_final_b, stride_out_final_i, stride_out_final_j, stride_out_final_d,

    EPS: tl.constexpr,       # epsilon for layer‑norm
    BLOCK_H: tl.constexpr,   # compile‑time tile for hidden dim (<=128)
    BLOCK_D: tl.constexpr,   # compile‑time tile for output dim
):
    # ------------------------------------------------------
    # Program IDs: (i, j, b)
    # ------------------------------------------------------
    pid_i = tl.program_id(0)          # row index i
    pid_j = tl.program_id(1)          # col index j
    pid_b = tl.program_id(2)          # batch index

    # Guard (grid is exactly N×N×B, but keep for safety)
    if pid_i >= N or pid_j >= N or pid_b >= B:
        return

    # ------------------------------------------------------------------
    # Load the hidden‑dim vectors for the contracted tensor and its gate
    # ------------------------------------------------------------------
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base pointers for this (b,i,j) element
    out_base = out_ptr   + pid_b * stride_out_b   + pid_i * stride_out_i   + pid_j * stride_out_j
    gate_base = gate_ptr + pid_b * stride_gate_b  + pid_i * stride_gate_i  + pid_j * stride_gate_j

    out_vec  = tl.load(out_base  + offs_h * stride_out_h,
                       mask=mask_h, other=0.0)
    gate_vec = tl.load(gate_base + offs_h * stride_gate_h,
                       mask=mask_h, other=0.0)

    # ------------------------------------------------------
    # Hidden‑dim layer‑norm: (x - mean) / sqrt(var + eps)
    # ------------------------------------------------------
    sum_out = tl.sum(out_vec, axis=0)               # scalar sum
    mean = sum_out / H

    # centre only the valid entries
    centered = (out_vec - mean) * mask_h
    var = tl.sum(centered * centered, axis=0) / H
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # normalized hidden vector
    norm_vec = centered * inv_std

    # affine transform of to_out_norm
    w_norm = tl.load(norm_w_ptr + offs_h * stride_norm_w_h,
                     mask=mask_h, other=0.0)
    b_norm = tl.load(norm_b_ptr + offs_h * stride_norm_b_h,
                     mask=mask_h, other=0.0)
    norm_vec = norm_vec * w_norm + b_norm

    # apply the outgoing gate
    gated_vec = norm_vec * gate_vec

    # ------------------------------------------------------
    # Final linear projection to `dim` (output channel dimension)
    # ------------------------------------------------------
    offs_d = tl.arange(0, BLOCK_D)

    d = 0
    while d < D:
        cur_d = d + offs_d
        mask_d = cur_d < D

        # Load weight slice [BLOCK_D, BLOCK_H]
        # proj_w has layout [D, H]
        w_ptrs = proj_w_ptr + cur_d[:, None] * stride_proj_d + offs_h[None, :] * stride_proj_h
        w_tile = tl.load(w_ptrs,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0)          # (BLOCK_D, BLOCK_H)

        # Dot product: sum_h gated_vec[h] * w_tile[d, h]
        prod = w_tile * gated_vec[None, :]       # (BLOCK_D, BLOCK_H)
        acc = tl.sum(prod, axis=1)               # (BLOCK_D,)

        # Store to output
        out_final_base = out_ptr_final + pid_b * stride_out_final_b \
                                         + pid_i * stride_out_final_i \
                                         + pid_j * stride_out_final_j
        out_final_ptrs = out_final_base + cur_d * stride_out_final_d
        tl.store(out_final_ptrs, acc, mask=mask_d)

        d += BLOCK_D


# --------------------------------------------------------------
# Host entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module with a fused Triton kernel.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]
        - mask         : [B, N, N]  (may be all ones)
        - weights      : dict of model parameters
        - config       : dict with keys `dim`, `hidden_dim`, `nomask`
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------ LayerNorm on the input (dim) -----------------
    # mean & var over the last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ------------ Linear projections (no bias) -----------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------ Apply mask (if any) --------------------------
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------ Gating ---------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------ Triple multiplication (batch matmul) ---------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------ Prepare tensors for the Triton kernel --------
    out_gate = out_gate.contiguous()
    output = torch.empty((B, N, N, dim), dtype=dtype, device=device)

    # Strides (in elements, not bytes)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h = out_gate.stride()
    stride_norm_w_h = weights["to_out_norm.weight"].stride(0)
    stride_norm_b_h = weights["to_out_norm.bias"].stride(0)
    stride_proj_d = weights["to_out.weight"].stride(0)
    stride_proj_h = weights["to_out.weight"].stride(1)
    stride_out_final_b, stride_out_final_i, stride_out_final_j, stride_out_final_d = output.stride()

    # Kernel launch configuration
    BLOCK_H = 128                     # >= hidden_dim (max 128 in tests)
    BLOCK_D = 64                      # tile for output channel dimension
    grid = (N, N, B)                  # (i, j, batch)

    trimul_out_kernel[grid](
        out,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        weights["to_out.weight"],
        output,
        B, N, hidden_dim, dim,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
        stride_norm_w_h, stride_norm_b_h,
        stride_proj_d, stride_proj_h,
        stride_out_final_b, stride_out_final_i, stride_out_final_j, stride_out_final_d,
        EPS=eps,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    return output
