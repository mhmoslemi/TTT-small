@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, GROUP_SIZE_M: tl.constexpr
):
    pid = tl.program_id(0) # linear id over blocks in M and N and batch? Actually we can use three axes: pid_z for batch, pid_x for N block dimension (col), pid_y for M block dimension (row). We'll just treat batch as axis 2 (pid_z).
    # Let's set grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(M, BLOCK_M), batch)
    # Get pid_x, pid_y, pid_z
    pid_x = tl.program_id(0) # N dimension (col)
    pid_y = tl.program_id(1) # M dimension (row)
    pid_z = tl.program_id(2) # batch dim

    # Compute offsets
    offs_m = pid_y * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_x * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Offsets for the K tile
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Compute whether within bounds
        a_ptrs = a_ptr + pid_z * stride_a_batch + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        b_ptrs = b_ptr + pid_z * stride_b_batch + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # perform dot product via multiply-accumulate
        acc += tl.dot(a, b) # tl.dot expects shape (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)
        # move pointers? Not needed because we compute each tile each time.

    # Write back
    c_ptrs = c_ptr + pid_z * stride_c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; # config may contain bs, seqlen etc, but not needed.

    # get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Compute layer norm over last dim (dim)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # weight shape (hidden_dim, dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Reshape for batched matmul, with support for GPU
    # permute to bring hidden_dim to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden_dim, N, N) (swap i/j)

    B, H, N, N2 = left_perm.shape
    assert N == N2
    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # get stride values
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # Choose grid
    grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_M), batch)

    # Launch kernel
    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        batch, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3, # typical tuning param
        num_warps=4
    )
    # After kernel, reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous() # shape (B,N,N,hidden_dim)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply with out_gate
    out = out * out_gate

    # Final linear projection
    output = torch.nn.functional.linear(out, to_out_weight)  # weight shape (dim, hidden_dim)
    return output

"""
TriMul (outgoing) forward pass with a Triton batched‐matrix‑multiply kernel.

The heavy part of TriMul is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We rewrite it as a batched GEMM:
    left  : (B, N, N, D_h) → (B·D_h, N, N)
    right : (B, N, N, D_h) → (B·D_h, N, N)ᵀ
    out   = left @ rightᵀ   (batch‑wise matrix multiply)

The batch‑wise GEMM is executed by a custom Triton kernel that tiles the
output (M,N) with BLOCK_M×BLOCK_N tiles and reduces over K with BLOCK_K.
All other operations (LayerNorm, Linear, gating, final projection) are
performed with native PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
#   A : (batch, M, K)   – row‑major
#   B : (batch, K, N)   – row‑major (i.e. Bᵀ in the original formulation)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                    # tensor dimensions (int)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_x = tl.program_id(0)   # column block index (N)
    pid_y = tl.program_id(1)   # row block index (M)
    pid_z = tl.program_id(2)   # batch index

    # Offsets for the output tile
    offs_m = pid_y * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_x * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the current A and B tiles
        a_ptrs = a_ptr + pid_z * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_z * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Load tiles with boundary guards
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate the product
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = c_ptr + pid_z * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton kernel.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all ones)
            - weights: dict of model parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Bring hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden_dim, N, N) – transpose i/j

    B, H, N, _ = left_perm.shape
    batch = B * H
    left_flat = left_perm.view(batch, N, N)   # (batch, M=N, K=N)
    right_flat = right_perm.view(batch, N, N) # (batch, K=N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides (in elements) for A, B, C
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Tiling parameters – tuned for typical N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # columns
        triton.cdiv(N, BLOCK_M),   # rows
        batch,                     # batch dimension
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return output
