**Key insight**: The problem is to minimize the maximum of many "similar" functions. The functions f_k(h) for different k are related (they're all cross-correlations). This structure can be exploited.

5. Repeat

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _top_lags_obj(x, n, top_m, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    top_idx = np.argsort(c)[-top_m:]
    top_c = c[top_idx]
    m = top_c.max()
    shifted = alpha * (top_c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _lag_obj(x, n, lag):
    """Objective: overlap at a specific lag"""
    dx = 2.0 / n
    m = abs(lag)
    if m == 0:
        return float(np.sum(x * (1 - x)) * dx)
    elif lag > 0:
        return float(np.sum(x[:n-m] * (1 - x[m:])) * dx)
    else:
        return float(np.sum(x[m:] * (1 - x[:n-m])) * dx)

def slsqp_optimize(h, n, obj_func, obj_args, maxiter=2000, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_func, h.copy(), args=obj_args, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def alternating_lag_min(h, n, max_rounds, deadline, maxiter_per=500):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:

        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        h_new = slsqp_optimize(h_curr, n, _lag_obj, (n, lag), maxiter=maxiter_per)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c

def targeted_local_search(h, n, deadline, max_iter=50000):

def _smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m_max = c.max()
    shifted = alpha * (c - m_max)
    exp_vals = np.exp(shifted - shifted.max())
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += weights[k] * dx * (1 - 2*x)
        else:
            m = abs(lag)
            grad[:n-m] += weights[k] * dx * (1 - x[m:])
            grad[m:] -= weights[k] * dx * x[:n-m]

        alloc = min(80, remaining * 0.2)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart on best solution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Structured perturbations (sine, blocks)
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        x = np.linspace(0, 2, n)
        for freq in [1.0, 1.5, 2.0, 3.0, 5.0]:
            if time.time() > deadline - 25:
                break
            for phase in [0, np.pi/2, np.pi, 3*np.pi/2]:
                if time.time() > deadline - 20:
                    break
                h_sine = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x + phase)
                h_sine = project_feasible(h_sine, n)
                blend = 0.3
                h_mixed = (1 - blend) * best_h + blend * h_sine
                h_mixed = project_feasible(h_mixed, n)
                alloc = min(30, (deadline - time.time() - 15) * 0.3)
                if alloc < 15:
                    break
                h_opt, c_opt = optimize_at_n(h_mixed, n, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 5: Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_run(best_h, n, _smooth_obj, (n, 500000), maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_ls, c_ls = targeted_local_search(best_h, n, deadline=min(deadline - 1, time.time() + 30))
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
