def sca_solve(h_curr, n, top_idx, c_top, trust=None):

    best_c = compute_c5(h, n)
    dx = 2.0 / n

        if time.time() > deadline - 1:
            break
        
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-m:]
        improved = False
        
        for j in range(n):
            # Compute the coefficient of h[j] in each C_k for k in top_idx
            coeffs = np.zeros(m)
            for s, idx in enumerate(top_idx):
                lag = idx - (n - 1)
                coeff = 0.0
                if lag == 0:
                    coeff = 1.0 - 2.0 * h[j]
                elif lag > 0:
                    if j + lag < n:
                        coeff += (1.0 - h[j + lag])
                    if j - lag >= 0:
                        coeff -= h[j - lag]
                else:
                    k = -lag
                    if j - k >= 0:
                        coeff += (1.0 - h[j - k])
                    if j + k < n:
                        coeff -= h[j + k]
                coeffs[s] = coeff * dx
            
            # C_{top_idx[s]}(h) = base_s + coeffs[s] * h[j]
            bases = c_all[top_idx] - coeffs * h[j]
            
            # Minimize max_s(bases[s] + coeffs[s] * t) over t in [0,1]
            candidates = [0.0, 1.0, h[j]]
            for a in range(m):
                for b in range(a + 1, m):
                    denom = coeffs[b] - coeffs[a]
                    if abs(denom) > 1e-15:
                        t_cross = (bases[a] - bases[b]) / denom
                        if -0.5 <= t_cross <= 1.5:
                            candidates.append(t_cross)
            
            best_t = h[j]
            best_val = float(np.max(bases + coeffs * h[j]))
            for t in candidates:
                t_clip = min(max(t, 0.0), 1.0)
                val = float(np.max(bases + coeffs * t_clip))
                if val < best_val - 1e-16:
                    best_val = val
                    best_t = t_clip
            
            if abs(best_t - h[j]) > 1e-12:
                h[j] = best_t
                improved = True
        
        # Project to maintain feasibility
        h = project_feasible(h, n)
        
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
        elif not improved:
            break
    
    return h, best_c

            candidates = [0.0, 1.0, h[j]]
            for a in range(m):
                for b in range(a + 1, m):
                    denom = coeffs[b] - coeffs[a]
                    if abs(denom) > 1e-15:
                        t_cross = (bases[a] - bases[b]) / denom
                        if -1.0 <= t_cross <= 2.0:
                            candidates.append(t_cross)
            
            best_t = h[j]
            best_val = float(np.max(bases + coeffs * h[j]))
            for t in candidates:
                t_clip = min(max(t, 0.0), 1.0)
                val = float(np.max(bases + coeffs * t_clip))
                if val < best_val - 1e-16:
                    best_val = val
                    best_t = t_clip
            
            if abs(best_t - h[j]) > 1e-12:
                h[j] = best_t
                improved = True
        
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
        elif not improved:
            break
    
    return h, best_c

def _smooth_obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(shifted)))) / alpha)

def _smooth_obj_active(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(shifted)))) / alpha)

def _sum_con(x, n):
    return np.sum(x) - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_full, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_polish_active(h, n, top_idx, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_active, h, args=(n, top_idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def full_pipeline(h_start, n, deadline):
    h = project_feasible(h_start, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: SCA
    if time.time() < deadline - 30:
        sca_deadline = min(deadline - 30, time.time() + 120)
        h_sca, c_sca = sca_optimize(h, n, sca_deadline, m=5)
        if c_sca < best_c:
            best_c = c_sca
            best_h = h_sca
    
    # Phase 2: SCA with different m
    if time.time() < deadline - 30:
        for m_try in [3, 7, 10]:
            if time.time() > deadline - 30:
                break
            h_sca, c_sca = sca_optimize(best_h, n, time.time() + min(60, (deadline - time.time() - 30) * 0.4), m=m_try)
            if c_sca < best_c:
                best_c = c_sca
                best_h = h_sca
    
    # Phase 3: Coordinate descent
    if time.time() < deadline - 15:
        cd_deadline = min(deadline - 15, time.time() + 80)
        h_cd, c_cd = coordinate_descent(best_h, n, cd_deadline, m=5)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    # Phase 4: SLSQP polish
    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-5:]
        h_pol = slsqp_polish_active(best_h, n, top_idx, 100000, maxiter=2000)
        c_pol = compute_c5(h_pol, n)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol
    
    # Phase 5: Final coordinate descent
    if time.time() < deadline - 3:
        h_cd, c_cd = coordinate_descent(best_h, n, deadline - 2, m=5)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
