def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):

            tot = room.sum()
            if tot < 1e-15:

        else:

            if tot < 1e-15:

def grad_lag_positive(h, n, k, dx):
    """Gradient of f_k(h) = dx * sum_{i=0}^{n-1-k} h[i](1-h[i+k]) for k > 0"""
    g = np.zeros(n)
    # From the linear term: dx for i in [0, n-1-k]
    g[:n-k] += dx
    # From the bilinear term: -dx*h[i+k] for i in [0, n-1-k]
    g[:n-k] -= dx * h[k:]
    # From the bilinear term: -dx*h[i-k] for i in [k, n-1]
    g[k:] -= dx * h[:n-k]
    return g

def grad_lag_negative(h, n, k, dx):
    """Gradient of f_{-k}(h) = dx * sum_{i=k}^{n-1} h[i](1-h[i-k]) for k > 0"""
    g = np.zeros(n)
    # h[i] for i in [k, n-1]: coefficient is dx*(1-h[i-k])
    g[k:] += dx * (1.0 - h[:n-k])
    # -h[i-k]*h[i] for i in [k, n-1]: w.r.t. h[j] where j = i-k, i.e., j in [0, n-1-k]
    g[:n-k] -= dx * h[k:]
    return g

def grad_lag_zero(h, n, dx):
    """Gradient of f_0(h) = dx * sum h[i](1-h[i])"""
    return dx * (1.0 - 2.0 * h)

    """Compute gradient of max_k f_k(h) (subgradient)"""
    dx = 2.0 / n
    c = compute_all_lags(h, n)

                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                        if delta < 1e-9:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                else:
                    if h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                        if delta < 1e-9:
                            continue
                        h[j] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
                    elif h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                        if delta < 1e-9:
                            continue
                        h[j] += delta
                        h[i] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[j] -= delta
                            h[i] += delta
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Main optimization at parent resolution
    if h_init is not None:
        alloc = min(350, (deadline - t0) * 0.35)
        h_opt, c_opt = two_phase_optimize(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt

    # Try other resolutions
    n_candidates = []
    if h_init is not None:
        for nc in [200, 400, 500, 600]:
            if nc != n_init:
                n_candidates.append(nc)
    else:
        n_candidates = [300, 400, 500]

    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 50
        if remaining < 80:
            break
        alloc = min(100, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = two_phase_optimize(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Restart with perturbations on the best solution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 30) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = two_phase_optimize(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final intensive polish on the best solution
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        # Gradient polish
        if time.time() < deadline - 10:
            h_gd, c_gd = gradient_descent_polish(best_h, n, deadline - 8, max_iters=5000)
            if c_gd < best_c:
                best_c = c_gd
                best_h = h_gd
        # Paired local search
        if time.time() < deadline - 5:
            h_ls, c_ls = paired_local_search(best_h, n, deadline - 3)
            if c_ls < best_c:
                best_c = c_ls
                best_h = h_ls
        # Final SLSQP
        if time.time() < deadline - 3:
            h_new = slsqp_optimize(best_h, n, 10000000, maxiter=8000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    if best_h is None:
        n_fallback = n_init if h_init is not None else 300
        best_h = project_feasible(np.full(n_fallback, 0.5), n_fallback)
        best_c = compute_c5(best_h, n_fallback)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
