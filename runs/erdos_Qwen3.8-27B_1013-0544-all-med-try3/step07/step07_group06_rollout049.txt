import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def precompute_k_lags(n):
    return np.arange(-(n-1), n)

def compute_swap_delta(h, i, j, k_lags, n):
    """Delta in correlation when h[i]:0->1, h[j]:1->0."""
    delta = np.zeros(len(k_lags))
    im = i - k_lags
    m = (im >= 0) & (im < n)
    delta[m] += (1.0 - h[im[m]])
    ip = i + k_lags
    m = (ip >= 0) & (ip < n)
    delta[m] -= h[ip[m]]
    jm = j - k_lags
    m = (jm >= 0) & (jm < n)
    delta[m] -= (1.0 - h[jm[m]])
    jp = j + k_lags
    m = (jp >= 0) & (jp < n)
    delta[m] += h[jp[m]]
    return delta

def vectorized_local_search(h, n, deadline, max_passes=500, seed=42):
    dx = 2.0 / n
    k_lags = np.arange(-(n-1), n)
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.sort(np.where(h == 1)[0])
    zeros = np.sort(np.where(h == 0)[0])
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        for iz in range(n_zeros):
            if time.time() > deadline:
                break
            if iz >= len(zeros):
                break
            i = zeros[iz]
            
            # Base contribution from i (0->1)
            base = np.zeros(len(k_lags))
            im = i - k_lags
            m = (im >= 0) & (im < n)
            base[m] += (1.0 - h[im[m]])
            ip = i + k_lags
            m = (ip >= 0) & (ip < n)
            base[m] -= h[ip[m]]
            
            # J contributions for all ones
            jm_mat = ones[:, None] - k_lags[None, :]
            mask_jm = (jm_mat >= 0) & (jm_mat < n)
            jm_clipped = np.clip(jm_mat, 0, n-1)
            contrib_jm = np.where(mask_jm, 1.0 - h[jm_clipped], 0.0)
            
            jp_mat = ones[:, None] + k_lags[None, :]
            mask_jp = (jp_mat >= 0) & (jp_mat < n)
            jp_clipped = np.clip(jp_mat, 0, n-1)
            contrib_jp = np.where(mask_jp, h[jp_clipped], 0.0)
            
            delta_mat = base[None, :] - contrib_jm + contrib_jp
            new_corrs = corr[None, :] + delta_mat
            new_maxes = new_corrs.max(axis=1) * dx
            
            best_j_idx = np.argmin(new_maxes)
            if new_maxes[best_j_idx] < cur_c - 1e-14:
                j = ones[best_j_idx]
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corrs[best_j_idx]
                cur_c = new_maxes[best_j_idx]
                
                # Update sets
                ones = np.delete(ones, best_j_idx)
                ones = np.sort(np.append(ones, i))
                zeros = np.delete(zeros, iz)
                zeros = np.sort(np.append(zeros, j))
                n_ones = len(ones)
                n_zeros = len(zeros)
                
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        
        if not improved:
            break
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, seed=42, n_steps=200000):

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj_fn(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fn(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_thue_morse(n):
    """Generate Thue-Morse binary sequence of length n."""
    h = np.zeros(n)
    for i in range(n):
        v = 0
        x = i
        while x:
            v ^= 1
            x >>= 1
        h[i] = float(v)
    return project_binary(h, n)
