def compute_delta_c(h, i, j, n):
    """Compute Δc[k] for swap h[i]+=1, h[j]-=1 (unit delta)."""
    # off_k = k - (n-1) for k = 0, ..., 2n-2
    off = np.arange(2*n-1) - (n-1)  # ranges from -(n-1) to (n-1)
    
    # A[k] = h[j + off_k] - h[i + off_k] (0 if out of range)
    # B[k] = h[j - off_k] - h[i - off_k] (0 if out of range)
    
    def safe_get(arr, idx):
        out = np.zeros(len(idx))
        mask = (idx >= 0) & (idx < len(arr))
        out[mask] = arr[idx[mask]]
        return out
    
    A = safe_get(h, j + off) - safe_get(h, i + off)
    B = safe_get(h, j - off) - safe_get(h, i - off)
    
    return A + B

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 300:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_direct(h, n, alpha, maxiter=6000, ftol=1e-18):
    c_old = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_val, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new <= c_old + 1e-14:
            return h_new
    except Exception:
        pass
    return h.copy()

def _sigmoid(x):
    out = np.empty_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _logit(h, eps=1e-12):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def _sgcon(x, n):
    h = _sigmoid(x)
    return float(h.sum() - n / 2.0)

def _sgcon_jac(x, n):

def pair_search(h, n, deadline, deltas, max_passes=200):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hp, hm = h[i] + d, h[j] - d
                    if 0.0 <= hp <= 1.0 and 0.0 <= hm:
                        h[i], h[j] = hp, hm
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hp - d, hm + d
                    hp, hm = h[i] - d, h[j] + d
                    if 0.0 <= hp and hm <= 1.0:
                        h[i], h[j] = hp, hm
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hp + d, hm - d
            if not improved:
                break
    return h, best_c

def random_pair_search(h, n, deadline, rng, n_iter=500000, d_scale=0.01):
    best_c = compute_c5(h, n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.001, d_scale) * (1 if rng.rand() < 0.5 else -1)
        hp, hj = h[i] + d, h[j] - d
        if not (0.0 <= hp <= 1.0 and 0.0 <= hj <= 1.0):

        if c <= best_c + 1e-15:
            best_c = min(best_c, c)
        else:
            h[i], h[j] = hp - d, hj + d
    return h, best_c
