import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_subgrad(h, n, alpha_sg=100):
    """Compute soft subgradient of max_k c[k] with respect to h."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    w = np.exp(alpha_sg * (c - max_c))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    # offset_m = m - (n-1), ranges from -(n-1) to (n-1)
    for m in range(2 * n - 1):
        if w[m] < 1e-10:
            continue
        offset = m - (n - 1)
        if offset == 0:
            grad += w[m] * (1 - 2 * h)
        elif offset > 0:
            # (1-h[j+offset]) for j in [0, n-1-offset]
            # -h[j-offset] for j in [offset, n-1]
            grad[:n-offset] += w[m] * (1 - h[offset:])
            grad[offset:] += -w[m] * h[:n-offset]
        else:
            o = -offset
            # (1-h[j+offset]) = (1-h[j-o]) for j in [o, n-1]
            # -h[j-offset] = -h[j+o] for j in [0, n-1-o]
            grad[o:] += w[m] * (1 - h[:n-o])
            grad[:n-o] += -w[m] * h[o:]
    
    return grad, max_c

def subgradient_descent(h, n, deadline, max_iters=500):
    """Subgradient descent with adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t = 0.01  # initial step size
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        grad, max_c = compute_subgrad(h, n, alpha_sg=100)
        
        # Adaptive step size
        h_new = h - t * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            t = min(t * 1.05, 0.1)  # increase step slightly
        else:
            t *= 0.8  # decrease step
        
        if t < 1e-6:
            break
    
    return best_h, best_c

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d

    return h, best_c

                if time.time() < deadline - 10:
                    hp2 = slsqp(best_h, n_best, 100000, maxiter=500)
                    cp2 = compute_c5(hp2, n_best)
                    if cp2 < best_c:
                        best_c = cp2
                        best_h = hp2

    # Phase 7: Block search (try structured moves)
    if time.time() < deadline - 15 and n_best <= 250:
        h7, c7 = block_search(best_h, n_best, min(time.time() + 30, deadline - 5),
                              block_size=3, deltas=[0.002, 0.001, 0.0005])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final intensive polish
    if time.time() < deadline - 3:
        h8 = slsqp(best_h, n_best, 500000, maxiter=2000)
        c8 = compute_c5(h8, n_best)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        # Final pair search
        if n_best <= 200:
            best_h, best_c = pair_search(best_h, n_best, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = near_pair_search(best_h, n_best, deadline - 1,
                                               [0.001, 0.0005, 0.0002, 0.0001], radius=50)

    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    return best_h, best_c, n_best
