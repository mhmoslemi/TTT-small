        if m >= 0:
            sg[:n-m] -= h[m:]       # d/dh[i] of h[i]*(1-h[i+m]) = (1-h[i+m])... 
            # Wait, R(k) = sum h[i]*(1-h[i+m])
            # dR/dh[i] = (1-h[i+m]) for the first factor
            # dR/dh[i+m] = -h[i] for the second factor
            sg[:n-m] += (1.0 - h[m:])   # first factor contribution
            sg[m:] -= h[:n-m]           # second factor contribution
        else:
            s = -m
            # R(k) = sum_{i=s}^{n-1} h[i]*(1-h[i-s])
            sg[s:] += (1.0 - h[:n-s])   # first factor
            sg[:n-s] -= h[s:]           # second factor
        
        # Normalize
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm
        
        # Step
        h_new = h - step * dx * sg
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            h_new2 = h - step * 0.3 * dx * sg
            h_new2 = project_feasible(h_new2, n)
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c:
                best_c = c_new2
                best_h = h_new2.copy()
                h = h_new2
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas=None, radius=None):
    """Fine-grained pair perturbation search"""
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = 0 if radius is None else max(0, i - radius)
                j_hi = n if radius is None else min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j or (j > i and radius is not None and j < i - radius):
                        continue
                    # Try h[i]+d, h[j]-d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i]-d, h[j]+d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c
