    h = v.copy()

        diff = target - h.sum()
        if abs(diff) < 1e-14:

    n = len(v)

Let me also think about a **coordinate descent** approach that's more targeted:

For the most active lag m*, we want to minimize:
c_{m*} = sum_{valid i} h[i+m*](1-h[i])

The gradient at h[j] is (1-h[j-m*]) - h[j+m*] (with boundary handling).

If the gradient is positive at j, decreasing h[j] helps. If negative, increasing h[j] helps.

To maintain the sum constraint, we need to move mass: decrease h[j] by δ and increase h[k] by δ.

The best k to choose is one where the gradient is most negative (so increasing h[k] helps the most).

This gives a **greedy swap**:
1. Find j with most positive gradient (for lag m*)
2. Find k with most negative gradient
3. Move δ from j to k
4. Repeat

This is O(n) per step (to find max/min gradient), much faster than the previous local search which was O(n²).

def coord_descent_lag(h, n, m, max_iter=10000, deadline=None):
    """Coordinate descent minimizing the overlap at lag m."""
    dx = 2.0 / n
    h_curr = h.copy()
    best_c = compute_all_lags(h_curr, n).max()
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        grad = lag_gradient(h_curr, n, m)
        
        # Find best pair: decrease at j (grad>0), increase at k (grad<0)
        j = int(np.argmax(grad))
        k = int(np.argmin(grad))
        
        if grad[j] <= 0 or grad[k] >= 0:
            break  # No improvement possible

        delta_max = min(h_curr[j], 1.0 - h_curr[k])

To address this, I'll use the **smoothed max** (log-sum-exp) with a moderate alpha, which gives a smooth approximation of the max. The gradient of the smooth max is a weighted average of the individual gradients, which is better than a single subgradient.

For the log-sum-exp:
φ(h) = (1/α) * log(sum_k exp(α * c_k(h)))

∇φ(h) = sum_k (exp(α*c_k) / sum_j exp(α*c_j)) * ∇c_k

This is a convex combination of the active lag gradients, weighted by their "importance" (how close they are to the max).

For α = 5000-10000, this is a very tight approximation of the max.

- The KKT conditions say the gradient of the active lag is in the normal cone of the constraint set

For the constraint set {0≤h≤1, Σh=n/2}, the normal cone at h is:
N(h) = {α * 1 + β : β_i ≤ 0 if h_i = 1, β_i ≥ 0 if h_i = 0, β_i = 0 if 0 < h_i < 1}

import numpy as np

import cvxpy as cp

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_all_lags(h, n)))

def project_box_sum(v, target_sum):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, n, m):
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    else:
        p = -m
        grad[p:] += (1.0 - h[:n - p])
        grad[:n - p] -= h[p:]
    return grad

def smooth_max_grad(h, n, alpha, top_k=15):
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_k:]
    c_top = c_all[top_idx]
    a = alpha * c_top
    a_shifted = a - a.max()
    w = np.exp(a_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        if w[i] > 1e-12:
            m = idx - (n - 1)
            grad += w[i] * lag_gradient(h, n, m)
    return grad

def proj_gradient_descent(h, n, deadline, alpha=20000, top_k=12, max_iter=20000):
    dx = 2.0 / n
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    LR = 1.0 / (n * 0.02)
    beta = 0.9
    v = np.zeros(n)
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_max_grad(h_curr, n, alpha, top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        v = beta * v - LR * grad
        h_new = project_box_sum(h_curr + v, n / 2.0)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                LR *= 0.5
                v = np.zeros(n)
                no_improve = 0
                if LR < 1e-8:
                    break
        h_curr = h_new
    return best_h, best_c

def coord_descent_active(h, n, deadline, top_lags=3, max_sweep=500):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    for sweep in range(max_sweep):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        k_max = int(np.argmax(c_all))
        top_active = np.argsort(c_all)[-top_lags:]
        improved = False
        for lag_idx in top_active:
            if time.time() > deadline:
                break
            m = lag_idx - (n - 1)
            if m == 0:
                continue
            grad = lag_gradient(h_curr, n, m)
            for _ in range(200):
                if time.time() > deadline:
                    break
                grad = lag_gradient(h_curr, n, m)
                j = int(np.argmax(grad))
                k = int(np.argmin(grad))
                if grad[j] <= 1e-10 or grad[k] >= -1e-10:
                    break
                delta = min(h_curr[j], 1.0 - h_curr[k]) * 0.3
                delta = min(delta, 0.02)
                if delta < 1e-10:
                    break
                h_curr[j] -= delta
                h_curr[k] += delta
                h_curr = project_box_sum(h_curr, n / 2.0)
                c_test = compute_c5(h_curr, n)
                if c_test < best_c - 1e-12:
                    best_c = c_test
                    improved = True
                else:
                    h_curr[j] += delta
                    h_curr[k] -= delta
                    h_curr = project_box_sum(h_curr, n / 2.0)

        if not improved:
            break
    return h_curr, best_c

def lp_outer_approx(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.SCIP]:
        try:
            prob.solve(solver=solver, timeout=15)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_box_sum(h_new, n / 2.0)

def lp_outer_loop(h, n, deadline, max_iters=6):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        h_new = lp_outer_approx(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.01, trust * 0.7)
    return best_h, best_c

def slsqp_polish(h, n, alpha=50000, maxiter=3000, deadline=None):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        return h.copy()
