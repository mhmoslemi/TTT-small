import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin[idx] = 1.0
    return h_bin

def fast_swap_max(h, n, i, j, corr):
    """Compute new max correlation after swapping h[i]:1->0, h[j]:0->1.
    Returns the new maximum of the full correlation (unscaled)."""
    # Affected lags range
    # L = k - (n-1) in corr indexing
    # Term non-zero when: i+L valid, j+L valid, i-L valid, or j-L valid
    lo1, hi1 = -i, n - 1 - i
    lo2, hi2 = -j, n - 1 - j
    lo3, hi3 = i - (n - 1), i
    lo4, hi4 = j - (n - 1), j
    
    aff_lo = max(-(n - 1), min(lo1, lo2, lo3, lo4))
    aff_hi = min(n - 1, max(hi1, hi2, hi3, hi4))
    
    if aff_lo > aff_hi:
        return corr.max()
    
    k_lo = aff_lo + (n - 1)
    k_hi = aff_hi + (n - 1)
    
    # Max over unaffected region
    if k_lo > 0:
        c_max_unaff = corr[:k_lo].max()
    else:
        c_max_unaff = 0.0
    if k_hi < 2 * n - 2:
        c_max_unaff = max(c_max_unaff, corr[k_hi + 1:].max())
    
    # Vectorized delta computation over affected region
    L = np.arange(k_lo, k_hi + 1) - (n - 1)
    delta = np.zeros(len(L))
    
    idx1 = i + L
    m1 = (idx1 >= 0) & (idx1 < n)
    if m1.any():
        delta[m1] -= (1.0 - h[idx1[m1]])
    
    idx2 = j + L
    m2 = (idx2 >= 0) & (idx2 < n)
    if m2.any():
        delta[m2] += (1.0 - h[idx2[m2]])
    
    idx3 = i - L
    m3 = (idx3 >= 0) & (idx3 < n)
    if m3.any():
        delta[m3] += h[idx3[m3]]
    
    idx4 = j - L
    m4 = (idx4 >= 0) & (idx4 < n)
    if m4.any():
        delta[m4] -= h[idx4[m4]]
    
    new_vals = corr[k_lo:k_hi + 1] + delta
    new_max = max(c_max_unaff, new_vals.max())
    return new_max

def binary_local_search(h, n, deadline, max_rounds=5000, seed=42):
    """Fast binary local search using O(n) per candidate swap evaluation."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    best_c = corr.max() * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        
        # Strategy 1: Target the worst few lags
        top_k = np.argsort(corr)[::-1][:8]
        for tl in top_k:
            if time.time() > deadline:
                break
            L = tl - (n - 1)
            if L == 0:
                continue
            
            # Find swaps that reduce this lag
            candidates = []
            if L > 0:
                for i in ones:
                    j = i + L
                    if j < n and h[j] == 0.0:
                        candidates.append((i, j))
            else:
                for i in ones:
                    j = i + L
                    if j >= 0 and h[j] == 0.0:
                        candidates.append((i, j))
            
            if not candidates:
                continue
            
            rng.shuffle(candidates)
            for (i, j) in candidates[:300]:
                if time.time() > deadline:
                    break
                new_max = fast_swap_max(h, n, i, j, corr)
                if new_max < corr.max() - 1e-12:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = np.correlate(h, 1 - h, mode='full')
                    if corr.max() * dx < best_c:
                        best_c = corr.max() * dx
                        best_h = h.copy()
                    improved = True
        
        # Strategy 2: Random swaps (to explore)
        n_random = 100 + rnd % 100
        for _ in range(n_random):
            if time.time() > deadline:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            new_max = fast_swap_max(h, n, i, j, corr)
            if new_max < corr.max() - 1e-12:
                h[i] = 0.0
                h[j] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')

                    break

                    remaining_ones.remove(i1)
                if not remaining_ones:
                    continue
                i2 = int(rng.choice(remaining_ones))
                j2 = int(rng.choice(zeros))
                if j2 == j1:
                    continue
                
                # Apply 2-swap
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                new_max = np.correlate(h, 1 - h, mode='full').max()
                if new_max < corr.max() - 1e-12:
                    corr = np.correlate(h, 1 - h, mode='full')
                    if corr.max() * dx < best_c:
                        best_c = corr.max() * dx
                        best_h = h.copy()
                    improved = True
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
        
        if not improved:
            no_improve += 1
            # Simulated annealing: accept worse moves occasionally
            if no_improve > 50 and rnd < max_rounds - 100:
                temp = 0.05 * max(0.0, 1.0 - float(no_improve) / 300.0)
                if temp > 0:
                    for _ in range(30):
                        if time.time() > deadline:
                            break
                        i = int(rng.choice(ones))
                        j = int(rng.choice(zeros))
                        new_max = fast_swap_max(h, n, i, j, corr)
                        # Accept if not too much worse

                            h[i] = 0.0
                            h[j] = 1.0
                            corr = np.correlate(h, 1 - h, mode='full')
                    # Recompute after SA phase
                    corr = np.correlate(h, 1 - h, mode='full')
                    no_improve = 0
            elif no_improve > 200:
                break
        else:
            no_improve = 0
    
    return best_h, best_c

def gradient_descent(h, n, alpha, deadline, lr=0.5, momentum=0.93):
    """Gradient descent on smooth max."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    dx = 2.0 / n
    
    while time.time() < deadline:
        c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            am = abs(lag)
            if am >= n:
                continue
            if lag == 0:
                grad[:] += w[k] * (1.0 - 2.0 * h_curr)
            elif lag > 0:
                grad[am:n] += w[k] * (1.0 - h_curr[0:n - am])
                grad[0:n - am] += w[k] * (-h_curr[am:n])
            else:
                grad[0:n - am] += w[k] * (1.0 - h_curr[am:n])
                grad[am:n] += w[k] * (-h_curr[0:n - am])
        
        grad *= dx
        v = momentum * v - lr * grad
        h_new = project_feasible(h_curr + v, n)
        h_curr = h_new
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    return best_h, best_c

def soft_refine(h, n, deadline):
    """Fine-tune non-binary values around a near-binary solution."""
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    c_all = compute_all_lags(h_curr, n)
    
    for _ in range(500):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c_all))
        L = k_star - (n - 1)
        if L == 0:
            break
        m = abs(L)
        if m >= n:
            break
        
        improved = False
        for i in range(min(n - m, 500)):
            if time.time() > deadline:
                break
            j = i + m
            # Try small adjustments
            for frac in [0.05, 0.1, 0.2, 0.3]:
                if h_curr[i] > 0.01 and h_curr[j] < 0.99:
                    delta = frac * min(h_curr[i], 1.0 - h_curr[j])
                    if delta < 1e-6:
                        continue
                    h_curr[i] -= delta
                    h_curr[j] += delta
                    c_new = compute_c5(h_curr, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_curr.copy()
                        c_all = compute_all_lags(h_curr, n)
                        improved = True
                        break
                    else:
                        h_curr[i] += delta
                        h_curr[j] -= delta
                elif h_curr[j] > 0.01 and h_curr[i] < 0.99:
                    delta = frac * min(h_curr[j], 1.0 - h_curr[i])
                    if delta < 1e-6:
                        continue
                    h_curr[i] += delta
                    h_curr[j] -= delta
                    c_new = compute_c5(h_curr, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_curr.copy()
                        c_all = compute_all_lags(h_curr, n)
                        improved = True
                        break
                    else:
                        h_curr[i] -= delta
                        h_curr[j] += delta
        if not improved:
            break
    
    return best_h, best_c
