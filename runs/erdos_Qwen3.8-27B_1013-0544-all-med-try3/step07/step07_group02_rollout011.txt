Actually, let me reconsider the binary ILS. The key question is: what's the best local search strategy for binary h?

For a given h and worst lag k:
- corr(k) = sum_{i=0}^{n-1-|k|} h[i] * (1 - h[i+|k|]) (for positive k)
- To reduce this, we want to minimize the number of positions where h[i]=1 and h[i+k]=0

A swap of h[i] (1→0) and h[j] (0→1) where j = i+k:
- Removes one contribution from corr(k): the term h[i]*(1-h[i+k]) goes from 1 to 0
- But adds contributions to other lags:
  - For lag d = j - i = k: removes 1 (good)
  - For lag d = i - j = -k: adds 1 (bad, but symmetric)
  - For other lags: the effect depends on neighbors

The net effect on the MAXIMUM is hard to predict. We just need to compute it.

For efficiency in local search:
- Instead of computing the full correlation (O(n^2)), we can compute just the change
- When we swap h[i] from 1→0 and h[j] from 0→1, only lags involving i or j change
- Specifically, for each lag d, the correlation changes by:
  - -h_old[j-d]*(1-h_old[i-d]) + h_new[j-d]*(1-h_new[i-d]) ... this is getting complicated

Let me just use the full computation. For n=500, it's fast enough.

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def binarize(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin[idx] = 1.0
    return h_bin


def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[0:n - m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n - m]
    else:
        grad[0:n - m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n - m]
    return grad


def smooth_lse(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)


def smooth_lse_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)
    w = w / w.sum()
    return w


def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_lse_weights(c, alpha)
    grad = np.zeros(n)
    num_active = np.sum(w > 1e-10)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def adam_optimize(h, n, alpha, deadline, lr=0.3, max_iter=50000):
    """Adam optimizer with smooth max objective."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    patience = 0
    lr_current = lr

    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        g = grad_smooth_max(h_curr, n, alpha)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g * g
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        step = lr_current * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_curr - step, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        if patience > 3000:
            lr_current *= 0.5
            m_vec *= 0.5
            v_vec *= 0.5
            patience = 0
        if it % 15000 == 14999:
            lr_current = max(0.01, lr_current * 0.6)

    return best_h, best_c


def slsqp_relate(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    s = alpha * (active - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)


def slsqp_sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def slsqp_sum_jac(x, n):
    return np.ones(n)

    cons = [{'type': 'eq', 'fun': slsqp_sum_con, 'jac': slsqp_sum_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_relate, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def quick_local_search(h, n, deadline, rng):
    """Quick fractional local search using targeted perturbations."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lags = compute_all_lags(h, n)

    for rnd in range(200):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(lags))
        lag = k_star - (n - 1)
        m = abs(lag)
        if m == 0 or m >= n:
            break

        improved = False
        for trial_delta in [0.03, 0.01, 0.005, 0.002]:
            if time.time() > deadline:
                break
            for _ in range(100):
                i = rng.randint(0, n - m)
                j = i + m
                if h[i] > h[j] + 1e-6:
                    delta = min(trial_delta, (h[i] - h[j]) * 0.5)
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    h = np.clip(h, 0, 1)
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        lags = compute_all_lags(h, n)
                        improved = True
                    else:
                        h[:] = best_h
                elif h[j] > h[i] + 1e-6:
                    delta = min(trial_delta, (h[j] - h[i]) * 0.5)
                    if delta < 1e-10:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    h = np.clip(h, 0, 1)
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        lags = compute_all_lags(h, n)
                        improved = True
                    else:
                        h[:] = best_h

        if not improved:
            break

    return best_h, best_c


def multi_block_search(h_bin, n, deadline, rng):
    """Try block-level swaps on binary h."""
    h = h_bin.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for it in range(200000):
        if time.time() > deadline:
            break
        # Try a block swap: pick a contiguous block of A's and swap with a block of 0's
        block_len = rng.choice([3, 5, 7, 10, 15])
        if block_len >= n // 4:
            block_len = 3
        i_start = rng.randint(0, n - block_len)
        j_start = rng.randint(0, n - block_len)
        if i_start == j_start:
            continue
        # Check if blocks are valid (one block all 1s, other all 0s)
        block_a = h[i_start:i_start + block_len]
        block_b = h[j_start:j_start + block_len]
        if not (np.all(block_a > 0.5) and np.all(block_b < 0.5)):
            continue
        # Swapped blocks must not overlap
        if i_start + block_len > j_start and j_start + block_len > i_start:
            continue

        h[i_start:i_start + block_len] = 0.0
        h[j_start:j_start + block_len] = 1.0
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-14:
            best_c = new_c
            best_h = h.copy()
        else:
            h[i_start:i_start + block_len] = 1.0
            h[j_start:j_start + block_len] = 0.0

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    # === Phase 1: Binary ILS at n=500 (primary effort) ===
    n_main = 500
    if h_init is not None:
        if n_init == n_main:
            h_work = h_init.copy()
        else:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_main)
            h_work = np.interp(x_new, x_old, h_init)
    else:
        h_work = np.full(n_main, 0.5)

    h_bin = binarize(h_work, n_main)
    # Initial local search
    ls_deadline = t0 + 20
    h_bin, c_ls = binary_local_search(h_bin, n_main, ls_deadline, rng, max_iter=5000)

    # Main ILS
    ils_deadline = min(t0 + 480, deadline - 200)
    h_best_bin, c_best_bin = binary_ils(h_bin, n_main, ils_deadline, rng)
    best_h = h_best_bin.copy()
    best_c = c_best_bin

    # === Phase 2: Fractional Adam refinement ===
    if time.time() < deadline - 150:
        h_frac = best_h.copy()
        h_frac += rng.normal(0, 0.02, n_main)
        h_frac = project_feasible(h_frac, n_main)
        adam_deadline = min(t0 + 100, deadline - 120)
        if adam_deadline > time.time() + 20:
            h_adam, c_adam = adam_optimize(h_frac, n_main, 50, adam_deadline, lr=0.3)
            if c_adam < best_c:
                best_c = c_adam
                best_h = h_adam

    # === Phase 3: SLSQP polish on best ===
    if best_h is not None and time.time() < deadline - 80:
        n_cur = len(best_h)
        slsqp_deadline_check = time.time() + 40
        if slsqp_deadline_check < deadline:
            for top_m in [3, 5]:
                h_new = slsqp_optimize(best_h, n_cur, 100000, maxiter=3000, top_m=top_m)
                c_new = compute_c5(h_new, n_cur)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new

    # === Phase 4: Try other resolutions ===
    for n_try in [400, 450, 550, 600]:
        if time.time() > deadline - 50:
            break
        remaining = deadline - time.time() - 20
        if remaining < 40:
            break
        # Interpolate
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_try = binarize(h_interp, n_try)

        # Binary ILS at this resolution
        alloc = min(60, remaining * 0.4)
        h_new, c_new = binary_ils(h_bin_try, n_try, time.time() + alloc, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

        # Quick fractional
        if time.time() < deadline - 30:
            h_frac_try = h_interp + rng.normal(0, 0.02, n_try)
            h_frac_try = project_feasible(h_frac_try, n_try)
            h_frac_try, c_frac_try = adam_optimize(h_frac_try, n_try, 30, time.time() + 30, lr=0.2)
            if c_frac_try < best_c:
                best_c = c_frac_try
                best_h = h_frac_try

    # === Phase 5: Block search on current best ===
    if time.time() < deadline - 20:
        h_bin_cur = binarize(best_h, len(best_h))
        block_dl = min(deadline - 5, time.time() + 30)
        h_block, c_block = multi_block_search(h_bin_cur, len(best_h), block_dl, rng)
        if c_block < best_c:
            best_c = c_block
            best_h = h_block

    # === Phase 6: Final quick local search ===
    if best_h is not None and time.time() < deadline - 3:
        n_final = len(best_h)
        h_final, c_final = quick_local_search(best_h, n_final, deadline - 1, rng)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        best_h = project_feasible(best_h, n_final)
        best_c = compute_c5(best_h, n_final)

    # Fallback
    if best_h is None:
        best_h = project_feasible(np.full(n_main, 0.5), n_main)
        best_c = compute_c5(best_h, n_main)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
