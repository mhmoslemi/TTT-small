def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

Actually, you know what, let me take a slightly different approach. Instead of carefully computing the delta analytically (which is error-prone), let me use a simpler but still efficient method:

For each proposed swap (i, j), compute the new correlation by directly updating the affected terms. Since a swap only affects O(n) terms in the correlation (one for each k), the update is O(n).

Here's a cleaner approach:

```python
def swap_delta(h, n, i, j):
    """Compute the change in correlation when h[i] goes 0->1 and h[j] goes 1->0.
    Returns the delta array (length 2n-1, indexed by k+(n-1))."""
    delta = np.zeros(2*n-1)
    # i goes 0->1:
    #   +(1-h[i-k]) for valid k (0 <= i-k < n)
    #   -h[i+k] for valid k (0 <= i+k < n)
    # j goes 1->0:
    #   -(1-h[j-k]) for valid k (0 <= j-k < n)
    #   -h[j+k] for valid k (0 <= j+k < n)
    
    # i: 0->1, term +(1-h[i-k])
    klo = max(i - n + 1, -(n-1))
    khi = min(i, n-1)
    if klo <= khi:
        # h[i-k] for k from klo to khi: i goes from i-khi to i-klo
        idx = klo + (n-1)
        delta[idx:idx+(khi-klo+1)] += 1.0 - h[i-khi : i-klo+1]
    
    # i: 0->1, term -h[i+k]
    klo = max(-i, -(n-1))
    khi = min(n-1-i, n-1)
    if klo <= khi:
        idx = klo + (n-1)
        delta[idx:idx+(khi-klo+1)] -= h[i+klo : i+khi+1]
    
    # j: 1->0, term -(1-h[j-k])
    klo = max(j - n + 1, -(n-1))
    khi = min(j, n-1)
    if klo <= khi:
        idx = klo + (n-1)
        delta[idx:idx+(khi-klo+1)] -= 1.0 - h[j-khi : j-klo+1]
    
    # j: 1->0, term -h[j+k]
    klo = max(-j, -(n-1))
    khi = min(n-1-j, n-1)
    if klo <= khi:
        idx = klo + (n-1)
        delta[idx:idx+(khi-klo+1)] -= h[j+klo : j+khi+1]
    
    return delta
```
