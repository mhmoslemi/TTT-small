import time

    return np.correlate(h, 1 - h, mode='full') * dx

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            tot = room.sum()
            if tot < 1e-14:

        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_shift_grad(h, n, shift_idx):
    """Get gradient of correlation at shift_idx with respect to h."""
    dx = 2.0 / n
    k = shift_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx

        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

                   perturb_every=200, perturb_scale=0.01, 
                   patience=100, best_c=None):
    """Adam optimizer with shift batching and periodic perturbation."""
    if best_c is None:
        best_c = compute_c5(h, n)
    
    n_shifts = 2 * n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Shift ordering: start with most important shifts
    best_iter = 0
    
    for epoch in range(100000):
        if time.time() > deadline:
            break
        
        # Compute current correlations
        c_vec = compute_c5_vec(h, n)
        current_c = c_vec.max()
        
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
            best_iter = epoch
        
        if epoch - best_iter > patience and epoch > 0:
            # Reset momentum and try perturbation
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            # Small random perturbation
            noise = np.random.randn(n) * perturb_scale
            h_try = project_feasible(h + noise, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                best_iter = epoch
            perturb_scale *= 0.5
            if perturb_scale < 1e-6:
                break
        
        # Select batch of shifts (focus on active ones)
        # Always include the top shifts
        top_idx = np.argsort(-c_vec)[:batch_size]
        # Add some random shifts for exploration
        if batch_size > n_shifts // 2:
            batch = np.arange(n_shifts)
        else:
            n_top = min(batch_size * 2 // 3, n_shifts)
            n_rand = batch_size - n_top
            top_idx = np.argsort(-c_vec)[:n_top]
            rand_idx = np.random.choice(n_shifts, size=n_rand, replace=False)
            batch = np.unique(np.concatenate([top_idx, rand_idx]))[:batch_size]
        
        # Compute gradient over batch using softmax weighting
        batch_vals = c_vec[batch]
        m_val = batch_vals.max()
        sh = alpha * (batch_vals - m_val)
        sh_max = sh.max()
        w = np.exp(sh - sh_max)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, idx in zip(w, batch):
            if wi < 1e-10:
                continue
            grad += wi * get_shift_grad(h, n, idx)
        
        # Adam step
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
    
    return h, best_c

def lgbfgs_active(h, n, deadline, alphas=None, top_k=20):
    """L-BFGS-B focusing on active shifts with increasing alpha."""
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000]
    
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Get shifts within 1% of max
        threshold = max_val * (1 - 0.01)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])][:top_k]
        
        if len(active) == 0:
            continue
        
        def obj(x):
            nonlocal alpha, active, n
            c = compute_c5_vec(x, n)
            vals = c[active]
            mx = vals.max()
            sh = alpha * (vals - mx)
            smx = sh.max()
            if smx > 500:
                val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
            else:
                val = mx + np.log(np.sum(np.exp(sh))) / alpha
            val += 0.001 * (x.sum() - n/2.0)**2
            return float(val)
        
        def grad(x):
            nonlocal alpha, active, n
            c = compute_c5_vec(x, n)
            vals = c[active]
            mx = vals.max()
            sh = alpha * (vals - mx)
            smx = sh.max()
            w = np.exp(sh - smx)
            w /= w.sum()
            dx_val = 2.0 / n
            g = np.zeros(n)
            for wi, idx in zip(w, active):
                if wi < 1e-12:
                    continue
                k = idx - (n - 1)
                wm = wi * dx_val
                if k >= 0:
                    nv = n - k
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk:n])
            g += 0.002 * (x.sum() - n/2.0)
            return g
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    
    return h, best_c

def fine_pair_search(h, n, deadline, radius=20):
    """Pair-wise local search."""
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            # Focus on high-value positions
            active_pos = np.where(h > 0.05)[0]
            if len(active_pos) > 100:
                active_pos = active_pos[np.random.choice(len(active_pos), 100, replace=False)]
            
            for i in active_pos:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

                        if c < best_c - 1e-15:

                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def project_feasible(h, n):

            break

            if tot < 1e-14:

                    best_c = c_try
                    best_h = h.copy()
                    no_improve = 0
                    accepted = True
                    break
            if not accepted:
                no_improve = 0  # reset to continue
    
    return best_h, best_c

def fine_pair(h, n, deadline, radius=12):
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            n_try = min(n, 300)
            idxs = np.random.choice(n, n_try, replace=False)
            for i in idxs:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def multi_start_refine(h, n, deadline, n_starts=4, start_scale=0.015):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for s in range(n_starts):
        if time.time() > deadline - 25:
            break
        scale = start_scale * (0.6 ** s)
        h_try = project_feasible(h + np.random.randn(n) * scale, n)
        c_try = compute_c5(h_try, n)
        t_sub = min(30, deadline - time.time() - 15)
        if t_sub < 10:
            continue
        h_opt, c_opt = run_lbfgs_active(h_try, n, time.time() + t_sub,
                                          alphas=[5000, 50000, 200000], top_k=15, mu=0.01)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250
    
    best_h = project_feasible(h_init, n_orig)
    n = n_orig
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Quick L-BFGS at original resolution
    t1 = min(t0 + 50, deadline - 700)
    best_h, best_c = run_lbfgs_active(best_h, n, t1,
                                        alphas=[500, 5000, 50000], top_k=15, mu=0.1)
    
    # Phase 2: Upsample to two candidate resolutions
    candidates = []
    for n_cand in [750, 1000, 1500]:
        if n_cand <= n_orig:
            continue
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_cand)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_cand)
        c_up = compute_c5(h_up, n_cand)
        candidates.append((h_up, n_cand, c_up))
    
    # Pick best candidate
    best_cand = None
    for hc, nc, cc in candidates:
        if best_cand is None or cc < best_cand[2]:
            best_cand = (hc, nc, cc)
    
    if best_cand is not None and best_cand[2] <= best_c + 0.002:
        best_h = best_cand[0]
        n = best_cand[1]
        best_c = best_cand[2]
    
    # Phase 3: Main Adam+SA optimization
    t3 = min(t0 + 500, deadline - 350)
    if time.time() < t3:
        best_h, best_c = run_adam_sa(best_h, n, t3, alpha=2000,
                                      lr=0.05, batch_size=40, patience=150,
                                      sa_start=0.005, sa_min=1e-7)
    
    # Phase 4: L-BFGS refinement
    if time.time() < deadline - 250:
        t4 = min(time.time() + 100, deadline - 250)
        best_h, best_c = run_lbfgs_active(best_h, n, t4,
                                            alphas=[5000, 20000, 100000, 500000], top_k=20, mu=0.005)
    
    # Phase 5: Adam with higher alpha
    if time.time() < deadline - 150:
        t5 = min(time.time() + 100, deadline - 150)
        best_h, best_c = run_adam_sa(best_h, n, t5, alpha=10000,
                                      lr=0.02, batch_size=25, patience=200,
                                      sa_start=0.002, sa_min=1e-8)
    
    # Phase 6: Multi-start perturbation
    if time.time() < deadline - 90:
        t6 = min(time.time() + 60, deadline - 90)
        best_h, best_c = multi_start_refine(best_h, n, t6, n_starts=4, start_scale=0.01)
    
    # Phase 7: L-BFGS final
    if time.time() < deadline - 40:
        t7 = min(time.time() + 35, deadline - 40)
        best_h, best_c = run_lbfgs_active(best_h, n, t7,
                                            alphas=[50000, 200000, 500000], top_k=10, mu=0.001)
    
    # Phase 8: Adam micro-refinement
    if time.time() < deadline - 25:
        t8 = min(time.time() + 20, deadline - 25)
        best_h, best_c = run_adam_sa(best_h, n, t8, alpha=50000,
                                      lr=0.005, batch_size=15, patience=100,
                                      sa_start=0.0005, sa_min=1e-9)
    
    # Phase 9: Fine pair search
    if time.time() < deadline - 3:
        t9 = min(time.time() + 20, deadline - 3)
        best_h, best_c = fine_pair(best_h, n, t9, radius=10)
    
    # Final validation
    best_h = project_feasible(best_h, n)
    final_c = compute_c5(best_h, n)
    if final_c < best_c:
        best_c = final_c
    
    return best_h, best_c, n
