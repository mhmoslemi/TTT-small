**Approach: Direct max reduction with subgradient**

The idea: compute the correlation vector, find the argmax lag k*, then compute the gradient of c[k*] with respect to h, and take a step in the negative gradient direction (projected to maintain feasibility).

For c[k] = dx * Σ_i h[i](1-h[i+k]):
∂c[k]/∂h[i] = dx * (1-h[i+k]) for i in range where i+k < n
∂c[k]/∂h[i+k] = -dx * h[i] for i in range where i+k < n

So the gradient of the max (at lag k*) is:
g[i] = dx * (1-h[i+k*]) - dx * h[i-k*] (with appropriate bounds)

We want to decrease h where g[i] > 0 and increase where g[i] < 0.

But we also need to maintain the sum constraint. The projected gradient step:
h_new[i] = clip(h[i] - lr * g[i] + lr * mean(g), 0, 1)
Then project to sum = n/2.

Let me implement this:

```python
def subgradient_max(h, n, deadline, lr_schedule=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    lr = 0.01
    for step in range(100000):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)  # actual lag
        
        # Compute gradient of c[k] w.r.t. h
        g = np.zeros(n)
        if k >= 0:
            # c[k] = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dc dh[i] = dx*(1-h[i+k]) for i < n-k
            # d/dc dh[i] = -dx*h[i-k] for i >= k
            for i in range(n - k):
                g[i] += dx * (1 - h[i + k])
            for i in range(k, n):
                g[i] -= dx * h[i - k]
        else:
            k_neg = -k
            # c[-k_neg] = dx * sum_{i=k_neg}^{n-1} h[i]*(1-h[i-k_neg])
            # equivalently sum_{i=0}^{n-1-k_neg} h[i+k_neg]*(1-h[i])
            for i in range(k_neg, n):
                g[i] -= dx * (1 - h[i - k_neg])
            for i in range(n - k_neg):
                g[i] += dx * h[i + k_neg]
        
        # Projected gradient step
        mean_g = g.mean()
        h_new = h - lr * (g - mean_g)
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            # Reduce learning rate
            lr *= 0.95
            if lr < 1e-8:
                break
    
    return h, best_c
```

```python
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h, n, deadline, rng):
    h = (h > 0.5).astype(float)
    if h.sum() != n // 2:
        idx = np.argsort(-h)
        h = np.zeros(n)
        h[idx[:n // 2]] = 1.0
    best_c = compute_c5(h, n)
    passes = 0
    while time.time() < deadline:
        improved = False
        passes += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            if time.time() > deadline:
                return h, best_c
            if h[i] != 1:
                continue
            for j in zeros:
                if time.time() > deadline:
                    return h, best_c
                if h[j] != 0:
                    continue
                h[i] = 0
                h[j] = 1
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    h[i] = 1
                    h[j] = 0
        if not improved and passes > 2:
            break
    return h, best_c

def _sum_con_fun(x, n):
    return x.sum() - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(len(x))

def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        logsum = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        logsum = np.log(np.sum(np.exp(shifted))) / alpha
    return m + logsum

def slsqp_run(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(softmax_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def near_pair_search(h, n, deadline, deltas, radius=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def direct_pair_search(h, n, deadline, deltas):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def directed_optimize(h, n, deadline, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.02
    stall_count = 0
    
    for iteration in range(500000):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        
        improved = False
        candidates = []
        
        if k >= 0:
            for i in range(n - k):
                if h[i] > 0.1 and h[i + k] < 0.9:
                    candidates.append((i, i + k))
        else:
            kn = -k
            for i in range(kn, n):
                if h[i] > 0.1 and h[i - kn] < 0.9:
                    candidates.append((i, i - kn))
        
        rng.shuffle(candidates)
        
        for (a, b) in candidates[:50]:
            if time.time() > deadline:
                return h, best_c
            amt = min(step, h[a] * 0.3, (1.0 - h[b]) * 0.3)
            if amt < 1e-7:
                continue
            h[a] -= amt
            h[b] += amt
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
            else:
                h[a] += amt
                h[b] -= amt
        
        if not improved:
            stall_count += 1
            step *= 0.8
            if step < 1e-6:
                break
        else:
            stall_count = 0
            if iteration % 500 == 0 and step > 0.005:
                step *= 0.9
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > deadline - 30:
            break
        ht = slsqp_run(h, n, alpha, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    
    remaining = deadline - time.time()
    if remaining > 20:
        if n <= 140:
            h, best_c = direct_pair_search(h, n, time.time() + min(remaining * 0.4, 120),
                                           [0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            h, best_c = near_pair_search(h, n, time.time() + min(remaining * 0.4, 120),
                                         [0.01, 0.005, 0.002, 0.001, 0.0005], radius=50)
    
    if time.time() < deadline - 10:
        ht = slsqp_run(h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    
    if n <= 140 and time.time() < deadline - 5:
        h, best_c = direct_pair_search(h, n, deadline - 2,
                                       [0.0005, 0.0002, 0.0001, 0.00005])
    elif time.time() < deadline - 5:
        h, best_c = near_pair_search(h, n, deadline - 2,
                                     [0.0005, 0.0002, 0.0001], radius=40)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(200, 0.5)
        n_parent = 200
    
    best_h = h_parent.copy()
    best_c = compute_c5(best_h, n_parent)
    
    # Phase 1: Binary local search from parent (most promising for small gains)
    if time.time() < deadline - 500:
        h_bin = make_binary(h_parent, n_parent)
        c_bin = compute_c5(h_bin, n_parent)
        alloc_bin = min(350, deadline - time.time() - 400)
        if alloc_bin > 30:
            h_b, c_b = binary_local_search(h_bin, n_parent, time.time() + alloc_bin, rng)
            if c_b < best_c:
                best_c = c_b
                best_h = h_b.copy()
        
        # Try a few random binary starts
        for ri in range(3):
            if time.time() > deadline - 350:
                break
            h_r = np.zeros(n_parent)
            ones = rng.choice(n_parent, n_parent // 2, replace=False)
            h_r[ones] = 1.0
            alloc_r = min(40, (deadline - time.time() - 300) * 0.3)
            if alloc_r < 10:
                break
            h_b2, c_b2 = binary_local_search(h_r, n_parent, time.time() + alloc_r, rng)
            if c_b2 < best_c:
                best_c = c_b2
                best_h = h_b2.copy()
    
    # Phase 2: Continuous SLSQP + pair search at parent resolution
    if time.time() < deadline - 200:
        alloc2 = min(200, deadline - time.time() - 100)
        if alloc2 > 30:
            h_opt, c_opt = optimize_at_n(h_parent, n_parent, time.time() + alloc2, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
    
    # Phase 3: Try different resolutions
    for n_try in [200, 250, 300, 350]:
        if time.time() > deadline - 80:
            break
        if n_try == n_parent:
            continue
        h_interp = interpolate_h(best_h, n_parent, n_try)
        alloc3 = min(60, (deadline - time.time() - 60) * 0.3)
        if alloc3 < 20:
            break
        h_opt, c_opt = optimize_at_n(h_interp, n_try, time.time() + alloc3, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    
    # Phase 4: Directed optimization on best
    if time.time() < deadline - 60:
        alloc4 = min(120, deadline - time.time() - 40)
        if alloc4 > 20:
            h_dir, c_dir = directed_optimize(best_h, len(best_h), time.time() + alloc4, rng)
            if c_dir < best_c:
                best_c = c_dir
                best_h = h_dir.copy()
    
    # Phase 5: Final fine polish
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        # SLSQP with very high alpha
        if time.time() < deadline - 5:
            ht = slsqp_run(best_h, n_best, 1000000, maxiter=3000)
            c = compute_c5(ht, n_best)
            if c < best_c:
                best_c = c
                best_h = ht
        
        # Fine pair search
        if time.time() < deadline - 2:
            if n_best <= 140:
                best_h, best_c = direct_pair_search(best_h, n_best, deadline - 1,
                                                    [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = near_pair_search(best_h, n_best, deadline - 1,
                                                  [0.001, 0.0005, 0.0002, 0.0001], radius=40)
        
        # One more binary attempt on the best continuous
        if time.time() < deadline - 5:
            h_bin2 = make_binary(best_h, n_best)
            c_bin2 = compute_c5(h_bin2, n_best)
            if c_bin2 < best_c:
                alloc_b = min(30, deadline - time.time() - 3)
                if alloc_b > 10:
                    h_bin2, c_bin2 = binary_local_search(h_bin2, n_best, time.time() + alloc_b, rng)
                    if c_bin2 < best_c:
                        best_c = c_bin2
                        best_h = h_bin2
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
