def greedy_local_search(h, n, deadline):

    h = h.copy()

    cur_c = int(corr.max()) * dx  # For binary, corr values are integers
    m_range = np.arange(-(n - 1), n)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    while time.time() < deadline:
        improved = False
        best_gain = 0
        best_a = -1
        best_b = -1
        best_delta = None
        best_new_c = cur_c
        
        for a in zeros:
            for b in ones:
                delta = delta_single_swap(h, n, a, b)
                new_corr_max = (corr + delta).max()
                new_c = float(int(new_corr_max)) * dx
                if new_c < best_new_c - 1e-15:

                    best_a = a
                    best_b = b
                    best_delta = delta

            ones.remove(best_b)
            zeros.append(best_b)
            ones.append(best_a)
            improved = True
        
        if not improved:
            break
    
    return h, cur_c

def block_move_search(h, n, deadline, seed=42):
    """Try moving contiguous blocks of 1's."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5_binary(h, n)
    
    while time.time() < deadline:
        # Find a contiguous block of 1's of length 2-5
        block_len = rng.randint(2, 6)
        start = rng.randint(0, n - block_len + 1)
        # Check it's a block of 1's
        if not np.all(h[start:start+block_len] == 1):
            continue
        # Try to move it to a random position
        new_start = rng.randint(0, n - block_len + 1)
        if new_start == start:
            continue
        # Check destination is all 0's
        if not np.all(h[new_start:new_start+block_len] == 0):
            continue
        # Check overlap
        if new_start < start + block_len and new_start + block_len > start:
            continue
        
        h_new = h.copy()
        h_new[start:start+block_len] = 0
        h_new[new_start:new_start+block_len] = 1
        c_new = compute_c5_binary(h_new, n)
        
        if c_new < best_c:
            h = h_new
            best_h = h.copy()
            best_c = c_new
    
    return best_h, best_c

    t0 = time.time()
    total_time = budget_s - 5
    deadline = t0 + total_time

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    hb = to_binary(h0, n)
    best_h = hb.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Greedy local search to establish local optimum
    p1 = t0 + 0.10 * total_time
    best_h, best_c = greedy_ls(best_h, n, p1, max_sweeps=30)

    # Phase 2: Long annealing with kick
    p2 = t0 + 0.35 * total_time
    ha, ca = anneal_with_kick(best_h, n, p2, seed=seed, T_init=0.10, cool_factor=0.99997)
    ha, ca = greedy_ls(ha, n, min(p2 + 3, deadline), max_sweeps=10)
    if ca < best_c:
        best_c = ca
        best_h = ha.copy()

    # Phase 3: Multi-restart annealing
    p3 = t0 + 0.85 * total_time
    best_h, best_c = multi_anneal(best_h, n, p3, seed=seed, n_restarts=4)

    # Phase 4: Try other resolutions
    best_h, best_c, n = try_other_resolution(best_h, n, best_c, deadline - 10, seed)

    # Phase 5: Final greedy polish
    best_h, best_c = greedy_ls(best_h, n, deadline - 2, max_sweeps=5)

    return best_h, best_c, n
