from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def fast_corr(h, n):
    N = 2*n - 1
    return ifft(fft(h, N) * np.conj(fft(1.0-h, N))).real

def c5_value(h, n):
    return float(np.max(fast_corr(h, n) * 2.0/n))

def proj(h, n):
    t = n/2.0
    h = np.clip(h.copy(), 0, 1)
    for _ in range(100):
        d = t - h.sum()
        if abs(d) < 1e-15: break
        m = (h < 1-1e-12) if d > 0 else (h > 1e-12)
        if m.sum() == 0: break
        h[m] += d/m.sum()
        h = np.clip(h, 0, 1)
    return h

def l2_corr_obj(x, n):
    c = fast_corr(x, n) * 2.0/n
    return c.dot(c)

def l2_corr_grad(x, n):
    c = fast_corr(x, n) * 2.0/n
    # gradient of sum(c_k^2) = 2 * sum_k c_k * dc_k/dx
    # dc_k/dx[j] = dx * [(1-x[j-k]) - x[j+k]] (with boundary handling)
    # This equals 2*dx * (c * (1-2x))[j+n-1]... wait
    # Actually: sum_k c_k * dc_k/dx[j]
    # dc_k/dx[j] = dx * [1 - 2*x[j-k_lag]] * I(valid)
    # where k_lag = k - (n-1)
    # sum_k c_k * [1-2*x[j-k_lag]] * I(valid)
    # = (c * (1-2x))[j+n-1] (convolution)
    # So grad[j] = 2 * dx * (c * (1-2x))[j+n-1]
    f = 1 - 2*x
    Nc = 3*n
    C = fft(c, Nc)
    F = fft(f, Nc)
    conv = ifft(C * F).real

One more thought: the coordinate descent can be made much faster by not recomputing the full correlation each time. When I change h[i] by +d and h[j] by -d, the change in the correlation is:
Δc[k] = d * [(1-h[j-k]) - h[i+k]] - d * [(1-h[i-k]) - h[j+k]] (approximately)

This is O(1) per lag, O(n) total. But implementing this correctly with the boundary effects is tricky. Let me just use the FFT approach for now.

Actually, for the coordinate descent, I can precompute the correlation once and then update it incrementally. When h changes at positions i and j:
- h[i] → h[i] + d
- h[j] → h[j] - d

The change in c[k] = ∑_m h[m+k] * (1-h[m]):
Δc[k] from h[i]:
  - h[i] appears in h[m+k] when m = i-k: Δ = d * (1-h[i-k])
  - h[i] appears in (1-h[m]) when m = i: Δ = -h[i+k] * d
  
So Δc[k] from h[i] change: d * (1-h[i-k]) - d * h[i+k]
Δc[k] from h[j] change: -d * (1-h[j-k]) + d * h[j+k]

Total: Δc[k] = d * [(1-h[i-k]) - h[i+k] - (1-h[j-k]) + h[j+k]]

This is O(n) to compute for all k! Much faster than O(n²) or O(n log n).

But I need to be careful about the range of valid k for each change.

Let me implement this incremental update for the coordinate descent.

For the full correlation vector of length 2n-1:
c[k_full] where k_full = 0,...,2n-2, lag k = k_full - (n-1)

Δc[k_full] from changing h[i] by d:
  - h[i] in h[m+k] when m+k=i → m = i-k = i-(k_full-(n-1)) = i-k_full+n-1
    Valid: 0 ≤ m < n → 0 ≤ i-k_full+n-1 < n → k_full-n+1 ≤ i ≤ k_full-1
    i.e., k_full ∈ [i+1, i+n-1] (clipped to [0, 2n-2])
    Contribution: d * (1-h[m]) = d * (1-h[i-k_full+n-1])
  
  - h[i] in (1-h[m]) when m=i:
    Valid: 0 ≤ i < n (always true)
    Contribution: -d * h[m+k] = -d * h[i+k_full-n+1]
    Valid: 0 ≤ i+k_full-n+1 < n → n-1-i ≤ k_full ≤ 2n-2-i

So for each k_full in [0, 2n-2]:
Δc[k_full] = d * (1-h[i-k_full+n-1]) * I(0 ≤ i-k_full+n-1 < n) 
            - d * h[i+k_full-n+1] * I(0 ≤ i+k_full-n+1 < n)

This is O(1) per k_full, O(n) total for all k_full.

Similarly for h[j] change by -d:
Δc[k_full] += -d * (1-h[j-k_full+n-1]) * I(valid) + d * h[j+k_full-n+1] * I(valid)

Total: O(n) per pair move, vs O(n log n) for FFT.

For n=250, O(n) = 250 operations vs O(n log n) ≈ 2000 operations. Not a huge speedup, but the constant factor is much smaller.

Actually, the real speedup is that we can compute the new max(c) in O(n) after the update, instead of O(n log n) for FFT. So the total per move is O(n) instead of O(n log n).

But in practice, the FFT is implemented in optimized C code, so the constant factor might make them comparable. Let me just stick with the simple approach.

    return 1 << (n-1).bit_length()

def fast_corr(h, n):
    N = _next_pow2(2*n - 1)
    return ifft(fft(h, N) * np.conj(fft(1.0-h, N))).real[:2*n-1]

def c5_value(h, n):
    c = fast_corr(h, n) * (2.0/n)
    return float(c.max())

def proj(h, n):
    t = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = t - h.sum()
        if abs(d) < 1e-15: break
        mask = (h < 1.0 - 1e-12) if d > 0 else (h > 1e-12)
        cnt = mask.sum()
        if cnt == 0: break

        h = np.clip(h, 0.0, 1.0)

    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
