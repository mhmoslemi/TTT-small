def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)

def nesterov_gradient_descent(h, n, deadline, lr_init=0.005, alpha=5000):

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)  # velocity
    h_curr = h.copy()
    lr = lr_init
    momentum = 0.9
    
    for it in range(50000):
        if time.time() > deadline:
            break
        # Nesterov lookahead
        h_look = h_curr - momentum * v
        g = grad_smooth_max(h_look, n, alpha)
        
        # Update
        v = momentum * v + lr * g
        h_new = h_curr - v
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new <= best_c:
            h_curr = h_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
        else:
            # Revert and decrease learning rate
            lr *= 0.7
            v = np.zeros(n)
            if lr < 1e-14:
                break
        
        # Adapt momentum
        if it % 1000 == 0:
            momentum = min(0.99, 0.9 + it * 0.0001)
    
    return best_h, best_c

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    dx = 2.0 / n
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for m in range(n_full):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha, mu):

    m = c.max()

            momentum = min(0.98, 0.85 + 0.01 * (it // 500))
        if lr < 1e-15 or stall > 500:
            break
    return best_h, best_c

def subgradient_gd(h, n, deadline, lr_init=0.01):
    """Direct subgradient method: always step against the argmax lag."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stall = 0
    for it in range(100000):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        m_idx = int(np.argmax(c_vec))
        k = m_idx - (n - 1)
        dx = 2.0 / n
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[0:nv] += (1.0 - h[k:n]) * dx
            g[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += (1.0 - h[0:nv]) * dx
            g[0:nv] += (-h[kk:n]) * dx
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new  # still move, but don't update best
            stall += 1
            if stall % 100 == 0:
                lr *= 0.95
            if stall > 2000:
                lr *= 0.8
        if lr < 1e-14:
            break
    return best_h, best_c

def pair_search_radius(h, n, deadline, radius=20):
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        if time.time() > deadline - 1:
            break
        improved_any = False
        max_passes = 50
        for _pass in range(max_passes):
            if time.time() > deadline - 1:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline - 1:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[j] >= d and h[i] <= 1.0 - d:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
        if not improved_any:
            break
    return h, best_c

def gradient_pair_search(h, n, deadline, step_size=0.002):
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000, 20000, 100000]
    for alpha in alphas:
        while time.time() < deadline - 1:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline - 1:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-10:
                        continue
                    if h[i] + move >= 0.0 and h[i] + move <= 1.0 and \
                       h[j] - move >= 0.0 and h[j] - move <= 1.0:
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x, nn):
        return float(x.sum() - nn / 2.0)
    def cons_jac(x, nn):
        return np.ones(nn)
    try:
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}]
        res = minimize(smooth_max_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Quick L-BFGS-B at original resolution
    t1 = min(t0 + 100, deadline - 650)
    if t1 > t0 + 10:
        best_h, best_c = lbfgs_phase(best_h, best_n, t1, max_iters=200)

    # Phase 2: Interpolate to n=500 and optimize with L-BFGS-B
    n_target = 500
    if time.time() < deadline - 400:
        h500 = interpolate_to(best_h, best_n, n_target)
        t2 = min(time.time() + 280, deadline - 200)
        h500, c500 = lbfgs_phase(h500, n_target, t2, max_iters=200)
        if c500 < best_c:
            best_c = c500
            best_h = h500
            best_n = n_target

    # Phase 3: SLSQP at current best resolution
    if time.time() < deadline - 200:
        t3 = min(time.time() + 60, deadline - 150)
        h_s, c_s = slsqp_phase(best_h, best_n, t3, alpha=500000, maxiter=800)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 4: Nesterov gradient descent
    if time.time() < deadline - 150:
        t4 = min(time.time() + 100, deadline - 80)
        h_ns, c_ns = nesterov_gd(best_h, best_n, t4, lr_init=0.002, alpha=50000)
        if c_ns < best_c:
            best_c = c_ns
            best_h = h_ns

    # Phase 5: Subgradient GD (direct argmax approach)
    if time.time() < deadline - 100:
        t5 = min(time.time() + 50, deadline - 60)
        h_sg, c_sg = subgradient_gd(best_h, best_n, t5, lr_init=0.005)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    # Phase 6: L-BFGS-B again to clean up
    if time.time() < deadline - 80:
        t6 = min(time.time() + 60, deadline - 40)
        best_h, best_c = lbfgs_phase(best_h, best_n, t6, max_iters=200)

    # Phase 7: Gradient-guided pair search
    if time.time() < deadline - 60:
        t7 = min(time.time() + 50, deadline - 30)
        best_h, best_c = gradient_pair_search(best_h, best_n, t7, step_size=0.001)

    # Phase 8: Fine pair search with radius
    if time.time() < deadline - 30:
        t8 = min(time.time() + 50, deadline - 5)
        best_h, best_c = pair_search_radius(best_h, best_n, t8, radius=15)

    # Phase 9: Try n=600 if time permits
    if time.time() < deadline - 40:
        h600 = interpolate_to(best_h, best_n, 600)
        t9 = min(time.time() + 30, deadline - 10)
        h600, c600 = lbfgs_phase(h600, 600, t9, max_iters=100)
        if c600 < best_c:
            best_c = c600
            best_h = h600
            best_n = 600

    # Phase 10: Final polish
    if time.time() < deadline - 8:
        best_h, best_c = lbfgs_phase(best_h, best_n, deadline - 8, max_iters=100)
    if time.time() < deadline - 2:
        best_h, best_c = pair_search_radius(best_h, best_n, deadline - 1, radius=10)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
