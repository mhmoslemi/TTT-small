            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def block_search(h, n, deadline, block_sizes, deltas):
    best_c = compute_c5(h, n)
    for bs in block_sizes:
        for d in deltas:
            if time.time() > deadline:
                return h, best_c
            improved = False
            n_blocks = n - bs + 1
            for i in range(0, n_blocks - bs + 1):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + bs, n_blocks + 1):
                    if j - i < bs:
                        continue
                    block_i = h[i:i+bs]
                    block_j = h[j:j+bs]
                    can_add = np.all(block_i + d <= 1.0) and np.all(block_j - d >= 0.0)
                    can_sub = np.all(block_i - d >= 0.0) and np.all(block_j + d <= 1.0)
                    if can_add:
                        h[i:i+bs] += d
                        h[j:j+bs] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i:i+bs] -= d
                            h[j:j+bs] += d
                    elif can_sub:
                        h[i:i+bs] -= d
                        h[j:j+bs] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i:i+bs] += d
                            h[j:j+bs] -= d
            if not improved:
                break
    return h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    # SLSQP with increasing alpha
    alphas = [300, 1000, 5000, 20000, 100000, 500000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_optimize(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    # Subgradient phase
    if time.time() < deadline - 20:
        sg_deadline = min(deadline - 10, time.time() + max(30, (deadline - time.time()) * 0.3))
        ht, c = subgradient_optimize(h, n, max_iter=50000, deadline=sg_deadline, start_lr=0.02, momentum=0.3)
        if c < best_c:
            best_c = c
            h = ht

    # SLSQP after subgradient
    if time.time() < deadline - 10:
        for a in [100000, 500000]:
            ht = slsqp_optimize(h, n, a, maxiter=2000)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                h = ht

    # Pair search
    if time.time() < deadline - 5:
        if n <= 200:
            h, best_c = pair_search_fast(h, n, deadline - 3,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        elif n <= 350:
            h, best_c = pair_search_fast(h, n, deadline - 3,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            # Use local pair search for large n
            h, best_c = local_pair_search(h, n, deadline - 3,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=80)

    # Block search (small blocks)
    if n <= 300 and time.time() < deadline - 3:
        h, best_c = block_search(h, n, deadline - 1, [2, 3], [0.005, 0.002, 0.001])

    # Final SLSQP
    if time.time() < deadline - 2:
        ht = slsqp_optimize(h, n, 1000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def local_pair_search(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(150, 0.5)
        n_init = 150

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize at parent resolution
    alloc1 = min(250, (deadline - t0) * 0.30)
    h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try other resolutions
    x_init = np.linspace(0, 2, n_init)
    other_ns = [200, 300, 400]
    for n_try in other_ns:
        remaining = deadline - time.time()
        if remaining < 80:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(150, remaining * 0.25)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Subgradient restarts on best solution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(3, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 20) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Aggressive pair search on best
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        if n <= 200:
            best_h, best_c = pair_search_fast(best_h, n, deadline - 5,
                                             [0.005, 0.002, 0.001, 0.0005,
                                              0.0002, 0.0001, 0.00005])
        elif n <= 350:
            best_h, best_c = local_pair_search(best_h, n, deadline - 5,
                                              [0.005, 0.002, 0.001, 0.0005,
                                               0.0002, 0.0001], radius=100)
        else:
            best_h, best_c = local_pair_search(best_h, n, deadline - 5,
                                              [0.005, 0.002, 0.001, 0.0005,
                                               0.0002], radius=150)

    # Phase 5: Final subgradient + SLSQP polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        ht, c = subgradient_optimize(best_h, n, max_iter=30000,
                                     deadline=deadline - 8, start_lr=0.01, momentum=0.2)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 3:
            ht = slsqp_optimize(best_h, n, 2000000, maxiter=5000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht

    # Phase 6: Final tiny pair search
    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        if n <= 250:
            best_h, best_c = pair_search_fast(best_h, n, deadline - 0.5,
                                             [0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = local_pair_search(best_h, n, deadline - 0.5,
                                              [0.0005, 0.0002, 0.0001], radius=60)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
