    corr = np.correlate(h, 1 - h, mode='full')
    c = corr * dx
    
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exps = np.exp(shifted)
    total = exps.sum()
    weights = exps / total
    smoothed_val = m + np.log(total) / alpha
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-15:
            continue
        lag = idx - (n - 1)
        # dc(lag)/dh[j] = [0<=j-lag<n]*(1-h[j-lag]) - [0<=j+lag<n]*h[j+lag]
        # First term: j-lag in [0,n-1] => j in [lag, lag+n-1] ∩ [0, n-1]
        j1s = max(0, lag)
        j1e = min(n - 1, lag + n - 1)
        if j1s <= j1e:
            grad[j1s:j1e+1] += w * (1 - h[j1s-lag:j1e-lag+1])
        # Second term: j+lag in [0,n-1] => j in [-lag, n-1-lag] ∩ [0, n-1]
        j2s = max(0, -lag)
        j2e = min(n - 1, n - 1 - lag)
        if j2s <= j2e:
            grad[j2s:j2e+1] -= w * h[j2s+lag:j2e+lag+1]
    
    grad *= dx
    return smoothed_val, grad

def pair_search(h, n, deadline, deltas, radius=None):
    """Local pair search for improvement."""
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n  # full search
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000):
    """SLSQP optimization with smooth max."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            sh = sh - mx
        exps = np.exp(sh)
        total = exps.sum()
        weights = exps / total
        # Gradient is same as smooth_max_grad but without the smoothed value
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            w = weights[idx]
            if w < 1e-15:
                continue
            lag = idx - (n - 1)
            j1s = max(0, lag)
            j1e = min(n - 1, lag + n - 1)
            if j1s <= j1e:
                grad[j1s:j1e+1] += w * (1 - x[j1s-lag:j1e-lag+1])
            j2s = max(0, -lag)
            j2e = min(n - 1, n - 1 - lag)
            if j2s <= j2e:
                grad[j2s:j2e+1] -= w * x[j2s+lag:j2e+lag+1]
        grad *= dx
        return grad
    
    def sum_con(x):
        return x.sum() - n / 2.0
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy().astype(float)
    # Bisection on mu for exact projection
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    hp = np.clip(h - mu, 0.0, 1.0)

    diff = target - hp.sum()
    if abs(diff) > 1e-14:

            if tot > 1e-15:
                hp += (diff / tot) * room
        else:

            if tot > 1e-15:
