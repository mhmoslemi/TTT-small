```python
    c_base: current correlation values (for active lags)
    g: gradient difference g[m] = dc[m]/dh[i] - dc[m]/dh[j]
    hi, hj: current values of h[i], h[j]
    d_max: maximum allowed delta based on bounds
    Returns: optimal delta (can be negative, meaning move in opposite direction)
    """

    # new_c[m] = c_base[m] + delta * g[m]
    # We want to minimize max(new_c) subject to |delta| <= min(hj, 1-hi) for positive delta
    # or |delta| <= min(hi, 1-hj) for negative delta

    # For negative delta: -min(hi, 1-hj) <= delta <= 0 = d_neg_max
    
    d_pos_max = min(hj, 1.0 - hi)
    d_neg_max = -min(hi, 1.0 - hj)
    
    # Find breakpoints where c_base[m] + delta*g[m] = c_base[m'] + delta*g[m']

    n_active = len(c_base)
    for m1 in range(n_active):
        for m2 in range(m1+1, n_active):
            denom = g[m1] - g[m2]
            if abs(denom) < 1e-15:
                continue
            delta_bp = (c_base[m2] - c_base[m1]) / denom
            if d_neg_max <= delta_bp <= d_pos_max:

    current_max = np.max(c_base)
    for m in range(n_active):
        if abs(g[m]) > 1e-15:
            delta_bp = (current_max - c_base[m]) / g[m]
            if d_neg_max <= delta_bp <= d_pos_max:
                candidates.append(delta_bp)

    best_max = np.inf

        if mx < best_max:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if abs(mx) > 50:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        s = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if s >= 0:
            for i in range(n - s):
                grad[i] += wm * (1.0 - x[i + s])
            for i in range(s, n):
                grad[i] += wm * (-x[i - s])
        else:
            t = -s
            for i in range(t, n):
                grad[i] += wm * (1.0 - x[i - t])
            for i in range(n - t):
                grad[i] += wm * (-x[i + t])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, alpha, deadline, maxiter=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mu = 1.0
    
    try:
        res = minimize(_smooth_max_obj, h, jac=_smooth_max_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-15, 'maxfun': maxiter * 20,
                               'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    
    return h, best_c

def compute_active_grad(c, h, n, top_idx):
    """Compute dc/dh[i] for each active lag and each i.
    Returns (len(top_idx), n) matrix."""
    result = np.zeros((len(top_idx), n))
    for ai, m in enumerate(top_idx):
        s = m - (n - 1)
        if s >= 0:
            for i in range(n - s):
                result[ai, i] += (1.0 - h[i + s])
            for i in range(s, n):
                result[ai, i] -= h[i - s]
        else:
            t = -s
            for i in range(t, n):
                result[ai, i] += (1.0 - h[i - t])
            for i in range(n - t):
                result[ai, i] -= h[i + t]
    return result

def find_optimal_delta(c_active, g, hi, hj):
    """Find optimal delta for moving mass from j to i.
    c_active: current correlations for active lags
    g: gradient difference g[m] = dc[m]/dh[i] - dc[m]/dh[j]
    Returns (delta, new_max_active)"""
    d_pos_max = min(hj, 1.0 - hi)
    d_neg_max = -min(hi, 1.0 - hj)
    
    if d_pos_max < -1e-15 and d_neg_max > 1e-15:
        return 0.0, np.max(c_active)
    
    candidates = [0.0]
    if d_pos_max > 1e-15:
        candidates.append(d_pos_max)
    if d_neg_max < -1e-15:
        candidates.append(d_neg_max)
    
    na = len(c_active)
    for m1 in range(na):
        for m2 in range(m1 + 1, na):
            denom = g[m1] - g[m2]
            if abs(denom) < 1e-15:
                continue
            delta_bp = (c_active[m2] - c_active[m1]) / denom
            if d_neg_max - 1e-12 <= delta_bp <= d_pos_max + 1e-12:
                candidates.append(delta_bp)
    
    best_delta = 0.0
    best_val = np.max(c_active)
    for d in candidates:
        d = np.clip(d, d_neg_max, d_pos_max)
        if d < -1e-15 and d > d_neg_max + 1e-15:
            continue
        new_c = c_active + d * g
        val = np.max(new_c)
        if val < best_val:
            best_val = val
            best_delta = d
    
    return best_delta, best_val

            # For changing h[i] to x (and fixing sum by spreading to all others):
            # Actually, let's just try a small perturbation and project.

            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]:
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def _smax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if abs(mx) > 40:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def _smax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        s = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if s >= 0:
            grad[:n - s] += wm * (1.0 - x[s:])
            grad[s:] += wm * (-x[:n - s])
        else:
            t = -s
            grad[t:] += wm * (1.0 - x[:n - t])
            grad[:n - t] += wm * (-x[t:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if abs(mx) > 40:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)


def _slsqp_con(x, n):
    return float(x.sum() - n / 2.0)


def _slsqp_jac(x, n):
    return np.ones(n)


def lbfgs_optimize(h, n, alpha, deadline, maxiter=400):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mu = 1.0
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-16, 'maxfun': maxiter * 25,
                                'maxls': 60})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    if time.time() < deadline - 2:
        try:
            res2 = minimize(_smax_obj, h, jac=_smax_grad, args=(n, alpha, mu),
                            method='L-BFGS-B', bounds=bounds,
                            options={'maxiter': maxiter, 'ftol': 1e-18,
                                     'gtol': 1e-16, 'maxfun': maxiter * 10,
                                     'maxls': 40})
            h_new2 = project_feasible(res2.x, n)
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c:
                best_c = c_new2
                h = h_new2
        except Exception:
            pass
    return h, best_c


def slsqp_optimize(h, n, alpha, deadline, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac, 'args': (n,)}]

    except Exception:

def compute_active_grad(h, n, top_idx):
    result = np.zeros((len(top_idx), n))
    for ai, m in enumerate(top_idx):
        s = m - (n - 1)
        if s >= 0:
            result[ai, :n - s] += (1.0 - h[s:])
            result[ai, s:] += (-h[:n - s])
        else:
            t = -s
            result[ai, t:] += (1.0 - h[:n - t])
            result[ai, :n - t] += (-h[t:])
    return result


def find_optimal_delta(c_active, g, hi, hj):
    d_min = -min(hi, 1.0 - hj)
    d_max = min(1.0 - hi, hj)
    if d_max - d_min < 1e-15:
        return 0.0, np.max(c_active)

    candidates = [d_min, d_max, 0.0]
    na = len(c_active)
    for m1 in range(na):
        for m2 in range(m1 + 1, na):
            denom = g[m1] - g[m2]
            if abs(denom) < 1e-15:
                continue
            delta_bp = (c_active[m2] - c_active[m1]) / denom
            if d_min - 1e-12 <= delta_bp <= d_max + 1e-12:
                candidates.append(delta_bp)

    best_delta = 0.0
    best_val = np.max(c_active)
    for d in candidates:
        d = max(d_min, min(d_max, d))
        new_c = c_active + d * g
        val = np.max(new_c)
        if val < best_val - 1e-16:
            best_val = val
            best_delta = d
    return best_delta, best_val


def active_pair_search(h, n, deadline, n_active=25, n_try=80):
    best_c = compute_c5(h, n)
    dx = 2.0 / n

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        c = compute_c5_vec(h, n)
        current_max = c.max()

        threshold = current_max - 0.004
        top_idx = np.where(c >= threshold)[0]
        if len(top_idx) == 0:
            top_idx = np.argsort(c)[-n_active:]
        if len(top_idx) > n_active:
            top_idx = np.argsort(c)[-n_active:]

        c_active = c[top_idx]
        grad_active = compute_active_grad(h, n, top_idx)
        scores = grad_active.sum(axis=0)
        order = np.argsort(scores)

        nt = min(n_try, n // 2)
        improved = False
        n_accepted = 0

        for i_idx in range(nt):
            if time.time() > deadline:
                return h, best_c
            i = order[n - 1 - i_idx]
            for j_idx in range(nt):
                j = order[j_idx]
                if i == j:
                    continue

                g = grad_active[:, i] - grad_active[:, j]
                if np.max(np.abs(g)) < 1e-14:
                    continue

                delta, new_max_active = find_optimal_delta(c_active, g, h[i], h[j])
                if abs(delta) < 1e-11:
                    continue
                if new_max_active * dx >= best_c * (1.0 - 1e-9):
                    continue

                h[i] += delta
                h[j] -= delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                    n_accepted += 1
                    if n_accepted % 8 == 0:
                        c = compute_c5_vec(h, n)
                        threshold = c.max() - 0.004
                        top_idx = np.where(c >= threshold)[0]
                        if len(top_idx) == 0:
                            top_idx = np.argsort(c)[-n_active:]
                        if len(top_idx) > n_active:
                            top_idx = np.argsort(c)[-n_active:]
                        c_active = c[top_idx]
                        grad_active = compute_active_grad(h, n, top_idx)
                        scores = grad_active.sum(axis=0)
                        order = np.argsort(scores)
                else:
                    h[i] -= delta
                    h[j] += delta

        if not improved:
            break
        if iteration > 500:
            break

    return h, best_c


def exhaustive_pair_search(h, n, deadline, n_active=20):
    best_c = compute_c5(h, n)
    dx = 2.0 / n

    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        current_max = c.max()
        threshold = current_max - 0.003
        top_idx = np.where(c >= threshold)[0]
        if len(top_idx) == 0:
            top_idx = np.argsort(c)[-n_active:]
        if len(top_idx) > n_active:
            top_idx = np.argsort(c)[-n_active:]

        c_active = c[top_idx]
        grad_active = compute_active_grad(h, n, top_idx)

        improved = False
        n_accepted = 0

        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(i + 1, n):
                g = grad_active[:, i] - grad_active[:, j]
                if np.max(np.abs(g)) < 1e-14:
                    continue

                delta, new_max_active = find_optimal_delta(c_active, g, h[i], h[j])
                if abs(delta) < 1e-11:
                    continue
                if new_max_active * dx >= best_c * (1.0 - 1e-8):
                    continue

                h[i] += delta
                h[j] -= delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                    n_accepted += 1
                    if n_accepted % 5 == 0:
                        c = compute_c5_vec(h, n)
                        threshold = c.max() - 0.003
                        top_idx = np.where(c >= threshold)[0]
                        if len(top_idx) == 0:
                            top_idx = np.argsort(c)[-n_active:]
                        if len(top_idx) > n_active:
                            top_idx = np.argsort(c)[-n_active:]
                        c_active = c[top_idx]
                        grad_active = compute_active_grad(h, n, top_idx)
                else:
                    h[i] -= delta
                    h[j] += delta

        if not improved:
            break

    return h, best_c


def simple_pair_search(h, n, deadline, deltas=None):
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS-B with increasing alpha (budget: ~250s)
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 350:
            break
        alloc = min(40, max(10, rem - 350))
        h, c = lbfgs_optimize(h, n, alpha, time.time() + alloc, maxiter=300)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: SLSQP (budget: ~80s)
    if time.time() < deadline - 250:
        for alpha in [500000, 2000000]:
            if time.time() > deadline - 250:
                break
            alloc = min(40, max(10, deadline - time.time() - 250))
            h, c = slsqp_optimize(best_h, n, alpha, time.time() + alloc, maxiter=1500)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 3: Active pair search (budget: ~250s)
    if time.time() < deadline - 200:
        alloc = min(200, deadline - time.time() - 200)
        h, c = active_pair_search(best_h, n, time.time() + alloc, n_active=25, n_try=80)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Try n=400 (budget: ~200s)
    if time.time() < deadline - 180:
        n2 = 400
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2, n2)

        for alpha in [100000, 500000, 2000000]:
            if time.time() > deadline - 80:
                break
            alloc = min(35, max(10, deadline - time.time() - 80))
            h2, c2 = lbfgs_optimize(h2, n2, alpha, time.time() + alloc, maxiter=200)

        if time.time() < deadline - 60:
            alloc = min(50, deadline - time.time() - 60)
            h2, c2 = active_pair_search(h2, n2, time.time() + alloc, n_active=25, n_try=70)

        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            n = n2

    # Phase 5: More L-BFGS on current best (budget: ~100s)
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        for alpha in [10000000, 50000000]:
            if time.time() > deadline - 30:
                break
            alloc = min(30, max(8, deadline - time.time() - 30))
            h, c = lbfgs_optimize(best_h, n_cur, alpha, time.time() + alloc, maxiter=200)
            if c < best_c:
                best_c = c
                best_h = h

    # Phase 6: Final active pair search (budget: ~50s)
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        alloc = min(45, deadline - time.time() - 5)
        h, c = active_pair_search(best_h, n_cur, time.time() + alloc, n_active=35, n_try=100)
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 7: Simple fine pair search as final polish
    if time.time() < deadline - 3:
        n_cur = len(best_h)
        alloc = min(15, deadline - time.time() - 2)
        h, c = simple_pair_search(best_h, n_cur, time.time() + alloc,
                                  deltas=[0.0005, 0.0002, 0.0001])
        if c < best_c:
            best_c = c
            best_h = h

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
```
