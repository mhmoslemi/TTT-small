def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_val_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    wsum = w.sum()
    val = m + (sm + np.log(wsum)) / alpha
    wn = w / wsum
    grad = np.ones(n)
    for k in range(2 * n - 1):
        wk = wn[k]
        if wk < 1e-16:
            continue
        shift = k - (n - 1)
        lo = max(0, -shift)
        hi = min(n, n - shift)
        if lo < hi:
            grad[lo:hi] -= wk * h[lo + shift:hi + shift]
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - shift:hi2 - shift]
    grad *= dx
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_val_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    wsum = w.sum()
    val = m + (sm + np.log(wsum)) / alpha
    wn = w / wsum
    grad = np.ones(n)
    for k in range(2 * n - 1):
        wk = wn[k]
        if wk < 1e-16:
            continue
        shift = k - (n - 1)
        lo = max(0, -shift)
        hi = min(n, n - shift)
        if lo < hi:
            grad[lo:hi] -= wk * h[lo + shift:hi + shift]
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - shift:hi2 - shift]
    grad *= dx
    return val, grad

def lbgfs_opt(h, n, alpha, maxiter, deadline):
    def fgrad(x):
        v, g = lse_val_grad(x, n, alpha)
        return v, g
    try:
        res = minimize(fgrad, h, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxls': 40, 'maxfun': maxiter * 40, 'maxcor': 20})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, deadline, max_rounds=50):
    target = n / 2.0
    h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.2, -0.2]:
                nh = h.copy()
                nh[i] = np.clip(nh[i] + delta, 0.0, 1.0)
                nh = project_feasible(nh, n)
                if np.allclose(nh, h, atol=1e-12):
                    continue
                nc = np.correlate(nh, 1 - nh, mode='full')
                if nc.max() < cur_max - 1e-12:
                    h = nh
                    corr = nc
                    cur_max = corr.max()
                    improved = True
                    break
        if not improved:
            break
    return h

def binary_search_local(h_init, n, deadline, max_sweeps=300, n_restarts=3, seed=42):
    if n % 2 != 0:
        target = n // 2
        idx = np.argsort(h_init)[::-1]
        h = np.zeros(n)
        h[idx[:target]] = 1.0
        h[idx[target]] = h_init[idx[target]]
    else:
        target = n // 2
        idx = np.argsort(h_init)[::-1]
        h = np.zeros(n)
        h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    current_h = h.copy()

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            n_flips = rng.randint(3, 8)
            for _ in range(n_flips):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h < 0.5)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    current_h[a] = 0.0
                    current_h[b] = 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
        cur_c = corr.max() * dx
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            for ii in range(len(zeros)):
                if time.time() > deadline:
                    break
                i = zeros[ii]
                for j in ones:
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk2 = (imn >= 0) & (imn < n)
                    delta[msk2] -= h[imn[msk2]]
                    msk3 = (jm >= 0) & (jm < n)
                    delta[msk3] -= 1.0 - h[jm[msk3]]
                    msk4 = (jmn >= 0) & (jmn < n)
                    delta[msk4] += h[jmn[msk4]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    delta[k2] += 1.0
                    new_corr = corr + delta
                    nc = new_corr.max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = new_corr
                        cur_c = nc
                        ones.remove(j)
                        ones.append(i)
                        zeros.remove(i)
                        zeros.append(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()

    for alpha in [20000, 200000, 2000000, 20000000]:

        c = compute_c5(h, best_n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if time.time() < dl - 5:

                best_h = h.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
