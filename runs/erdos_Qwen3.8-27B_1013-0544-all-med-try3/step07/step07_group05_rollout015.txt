import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 0:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 0:
                h += (diff / tot) * h
    return h

def shift_gradient(h, n, corr_index):
    """Gradient of c[corr_index] (before dx multiplication) with respect to h."""
    k = corr_index - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1 - 2 * h
    elif k > 0:
        # c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        grad[0:n-k] += (1 - h[k:n])
        grad[k:n] -= h[0:n-k]
    else:
        kp = -k
        # c_{-kp} = sum_{i=kp}^{n-1} h[i-kp]*(1-h[i])
        # = sum_{j=0}^{n-1-kp} h[j]*(1-h[j+kp])  ... wait
        # Actually: c_{-kp} = sum_i h[i]*(1-h[i-kp]) for i in kp..n-1
        grad[kp:n] += (1 - h[0:n-kp])
        grad[0:n-kp] -= h[kp:n]
    return grad * dx

def subgradient_descent(h, n, deadline, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Adaptive step size
    eta = 0.5 / n
    momentum = np.zeros(n)
    beta = 0.9
    
    n_iter = 0
    while time.time() < deadline:
        # Compute correlations
        c = compute_correlations(h, n)
        c_max = c.max()
        
        if c_max < best_c - 1e-15:
            best_c = c_max
            best_h = h.copy()
        
        # Find active shifts (those within 2% of max)
        threshold = c_max - 0.02
        active = np.where(c > threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Average gradient of active shifts
        grad = np.zeros(n)
        for idx in active:
            grad += shift_gradient(h, n, idx)
        grad /= len(active)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - eta * momentum
        h_new = project_feasible(h_new, n)
        
        # Adaptive step size: if f increased, reduce eta
        c_new = compute_c5(h_new, n)
        if c_new < c_max + 1e-10:
            h = h_new
            eta = min(eta * 1.05, 2.0/n)
        else:
            eta *= 0.7
            h = h.copy()  # don't update
        
        n_iter += 1
        if n_iter % 500 == 0:
            beta = min(beta + 0.01, 0.98)
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 400:
            val = np.sum(np.exp(shifted - mx))
            return float(m + (mx + np.log(val)) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 400:
            exp_vals = np.exp(shifted - mx)
        else:
            exp_vals = np.exp(shifted)
            # rescale
            exp_vals = exp_vals / exp_vals.sum() * exp_vals.sum()
        # softmax weights
        weights = exp_vals / exp_vals.sum()
        # Gradient of weighted sum c_k
        grad = np.zeros(n)
        for k_idx in range(len(c)):
            if weights[k_idx] > 1e-10:
                grad += weights[k_idx] * shift_gradient(x, n, k_idx)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 0:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 0:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, corr_index):
    k = corr_index - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[0:n-k] += (1.0 - h[k:n])
        grad[k:n] -= h[0:n-k]
    else:
        kp = -k
        grad[kp:n] += (1.0 - h[0:n-kp])
        grad[0:n-kp] -= h[kp:n]
    return grad * dx

def subgradient_descent(h, n, deadline, rng, n_top=5):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    eta = 1.0 / n
    momentum = np.zeros(n)
    beta = 0.85
    patience = 0
    max_patience = 2000
    n_iter = 0
    while time.time() < deadline:
        c = compute_correlations(h, n)
        c_max = c.max()
        if c_max < best_c - 1e-15:
            best_c = c_max
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        threshold = c_max - 0.03
        active = np.where(c > threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > n_top:
            top_idx = np.argsort(c[active])[::-1][:n_top]
            active = active[top_idx]
        grad = np.zeros(n)
        for idx in active:
            grad += shift_gradient(h, n, idx)
        grad /= len(active)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - eta * momentum
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_max + 0.5 * eta * np.linalg.norm(grad):
            h = h_new
            eta = min(eta * 1.02, 3.0 / n)
        else:
            eta *= 0.75
        n_iter += 1
        if n_iter % 1000 == 0:
            beta = min(beta + 0.005, 0.97)
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 400:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_obj, h, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def directed_pair_search(h, n, deadline, deltas, n_target=5):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(80):
            if time.time() > deadline:
                return h, best_c
            c_arr = compute_correlations(h, n)
            target_idx = np.argsort(c_arr)[::-1][:n_target]
            improved = False
            for idx in target_idx:
                if time.time() > deadline:
                    return h, best_c
                k = idx - (n - 1)
                if k == 0:
                    continue
                kp = abs(k)
                for _ in range(3):
                    if time.time() > deadline:
                        return h, best_c
                    improved_this = False
                    if k > 0:
                        for i in range(0, n - kp):
                            if time.time() > deadline:
                                return h, best_c
                            for delta_dir in [1, -1]:
                                hi = h[i] + delta_dir * d
                                for j in range(kp, n):
                                    if j == i:
                                        continue
                                    hj = h[j] - delta_dir * d
                                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                                        h[i], h[j] = hi, hj
                                        c = compute_c5(h, n)
                                        if c < best_c - 1e-15:
                                            best_c = c
                                            improved = True
                                            improved_this = True
                                        else:
                                            h[i], h[j] = hi - delta_dir * d, hj + delta_dir * d
                    else:
                        for i in range(0, n - kp):
                            if time.time() > deadline:
                                return h, best_c
                            for delta_dir in [1, -1]:
                                hi = h[i] - delta_dir * d
                                for j in range(kp, n):
                                    if j == i:
                                        continue
                                    hj = h[j] + delta_dir * d
                                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                                        h[i], h[j] = hi, hj
                                        c = compute_c5(h, n)
                                        if c < best_c - 1e-15:
                                            best_c = c
                                            improved = True
                                            improved_this = True
                                        else:
                                            h[i], h[j] = hi + delta_dir * d, hj - delta_dir * d
                    if not improved_this:
                        break
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=60, max_passes=100):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    t_subgrad = min(180, (deadline - time.time()) * 0.4)

            if time.time() > deadline - 20:
                break
            ht = slsqp_smooth(best_h, n, a, maxiter=2000, ftol=1e-15)
            c = compute_c5(ht, n)

                best_c = c
                best_h = ht
    if time.time() < deadline - 10:
        if n <= 300:
            best_h, best_c = directed_pair_search(best_h, n, deadline - 5,
                                                  [0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 5,
                                              [0.01, 0.005, 0.002], radius=50, max_passes=50)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
