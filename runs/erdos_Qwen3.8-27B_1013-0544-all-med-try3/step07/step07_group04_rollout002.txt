def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(v, n):
    target = n / 2.0
    lo = v.min() - 1.5
    hi = v.max() + 0.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(v - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    diff = target - out.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - out
            tot = room.sum()
            if tot > 1e-14:
                out += (diff / tot) * room
        else:
            tot = out.sum()
            if tot > 1e-14:
                out *= (target / tot)
        out = np.clip(out, 0.0, 1.0)
    return out

def lag_gradient(x, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = (1.0 - 2.0 * x) * dx
    elif lag > 0:
        if n - m > 0:
            grad[:n - m] += (1.0 - x[m:]) * dx
            grad[m:] += (-x[:n - m]) * dx
    else:
        if n - m > 0:
            grad[m:] += (1.0 - x[:n - m]) * dx
            grad[:n - m] += (-x[m:]) * dx
    return grad

def smooth_obj_grad(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    logsumexp = mx + np.log(np.sum(np.exp(shifted - mx)))
    f = m + logsumexp / alpha
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return f, np.zeros(n)
    w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        lag_k = k - (n - 1)
        grad += w[k] * lag_gradient(x, n, lag_k)
    return f, grad

def sum_eq_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=3000):
    f0, g0 = smooth_obj_grad(h, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj_grad, h, args=(n, alpha), jac=True,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h.copy()

def pgbfgs_step(h, n, alpha, maxiter=5000):
    f0, g0 = smooth_obj_grad(h, n, alpha)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_grad, h, args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 50, 'maxfun': 200000})
        if res.x is not None:
            out = project_box_sum(res.x, n)
            return out
    except Exception:
        pass
    return h.copy()

def projected_grad_descent(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [20, 50, 100, 300, 800, 2000, 5000, 15000, 50000, 150000]
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        f_curr, g_curr = smooth_obj_grad(h_curr, n, alpha)
        step = 0.05
        stalled = 0
        for it in range(30000):
            if time.time() > deadline - 8:
                break
            h_new = h_curr - step * g_curr
            h_new = project_box_sum(h_new, n)
            f_new, g_new = smooth_obj_grad(h_new, n, alpha)
            if f_new < f_curr - 1e-15:
                h_curr = h_new
                f_curr = f_new
                g_curr = g_new
                step = min(step * 1.05, 0.2)
                stalled = 0
                c_now = compute_c5(h_curr, n)
                if c_now < best_c - 1e-13:
                    best_c = c_now
                    best_h = h_curr.copy()
            else:
                step *= 0.6
                stalled += 1
                if step < 1e-12 or stalled > 30:
                    break
    return best_h, best_c

def coordinate_descent(h, n, deadline, iters_per_pass=3):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    for _ in range(iters_passes_limit):
        pass
    return best_h, best_c

def iters_passes_limit():
    return 0

def local_search_pairs(h, n, deadline, max_rounds=50):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    dx = 2.0 / n
    for _round in range(max_rounds):
        if time.time() > deadline - 2:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            # Try to reduce the self-overlap: spread out values
            improved_any = False
            idx_sorted = np.argsort(h_curr)
            for _i in range(min(20, n // 2)):
                lo_i = idx_sorted[_i]
                hi_i = idx_sorted[n - 1 - _i]
                delta = 0.002
                h_try = h_curr.copy()
                h_try[lo_i] += delta
                h_try[hi_i] -= delta
                h_try = np.clip(h_try, 0, 1)
                if abs(h_try.sum() - n / 2.0) > 1e-10:
                    h_try = project_box_sum(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-14:
                    h_curr = h_try
                    best_c = c_try
                    best_h = h_curr.copy()
                    improved_any = True
            if not improved_any:
                break
        else:
            m = abs(lag)
            improved_any = False
            for i in range(n - m):
                if time.time() > deadline - 2:
                    break
                j = i + m
                for d in [0.003, 0.001, 0.0005]:
                    if h_curr[i] > h_curr[j]:
                        h_try = h_curr.copy()
                        h_try[i] -= d
                        h_try[j] += d
                    else:
                        h_try = h_curr.copy()
                        h_try[i] += d
                        h_try[j] -= d
                    h_try = np.clip(h_try, 0, 1)
                    h_try = project_box_sum(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c - 1e-14:
                        h_curr = h_try
                        best_c = c_try
                        best_h = h_curr.copy()
                        improved_any = True
                        break
            if not improved_any:
                # Try random pairs
                rng_local = np.random.RandomState(_round + 42)
                found = False
                for _rep in range(500):
                    i1 = rng_local.randint(n)
                    i2 = rng_local.randint(n)
                    if i1 == i2:
                        continue
                    if h_curr[i1] < 0.01 or h_curr[i2] > 0.99:
                        continue
                    d = 0.002
                    h_try = h_curr.copy()
                    if h_curr[i1] > h_curr[i2]:
                        h_try[i1] -= d
                        h_try[i2] += d
                    else:
                        h_try[i1] += d
                        h_try[i2] -= d
                    h_try = project_box_sum(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c - 1e-14:
                        h_curr = h_try
                        best_c = c_try
                        best_h = h_curr.copy()
                        found = True
                        break
                if not found:
                    break
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_box_sum(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Projected gradient with increasing smoothness
    if time.time() < deadline - 20:
        pg_end = time.time() + min(60, (deadline - time.time()) * 0.35)
        h_pg, c_pg = projected_grad_descent(best_h, n, pg_end)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    # Phase 2: L-BFGS-B with penalty on sum (handle via projection)
    if time.time() < deadline - 15:
        for alpha in [5000, 20000, 80000]:
            if time.time() > deadline - 15:
                break
            h_try = pgbfgs_step(best_h, n, alpha)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-13:
                best_c = c_try
                best_h = h_try
    # Phase 3: SLSQP polish
    if time.time() < deadline - 15:
        for alpha in [30000, 100000, 300000]:
            if time.time() > deadline - 15:
                break
            h_try = slsqp_step(best_h, n, alpha)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-13:
                best_c = c_try
                best_h = h_try
    # Phase 4: Local search
    if time.time() < deadline - 4:
        best_h, best_c = local_search_pairs(best_h, n, min(deadline - 1, time.time() + 30))
    best_h = project_box_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250

    best_h = None
    best_c = float('inf')

    # If we have initial values, optimize at that resolution first
    if h_init is not None:
        alloc = min(250, max(80, (deadline - t0) * 0.3))
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Try higher resolutions
    n_candidates = [400, 500, 350]
    if n_init is not None:
        n_candidates = [x for x in n_candidates if x != n_init]
    n_candidates.insert(0, 450)

    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        if best_h is not None:
            h_start = upsample(best_h, len(best_h), n_try)
        elif h_init is not None:
            h_start = upsample(h_init, n_init, n_try)
        else:
            h_start = np.full(n_try, 0.5)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Restart with noise at best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_box_sum(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.5)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Very high alpha SLSQP
        if time.time() < deadline - 5:
            h_try = slsqp_step(best_h, n, 500000, maxiter=5000)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
        # Final local search
        if time.time() < deadline - 2:
            best_h, best_c = local_search_pairs(best_h, n, deadline - 1, max_rounds=80)
        best_h = project_box_sum(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n_fallback = 300
        best_h = project_box_sum(np.full(n_fallback, 0.5), n_fallback)
        best_c = compute_c5(best_h, n_fallback)

    return best_h, best_c, len(best_h)
