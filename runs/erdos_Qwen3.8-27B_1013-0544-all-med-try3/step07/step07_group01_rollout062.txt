import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(150):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = (1.0 - h).sum()
            if room < 1e-14:
                break
            h += (diff / room) * (1.0 - h)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    g = np.zeros(n)
    for kk in range(2 * n - 1):
        if w[kk] < 1e-13:
            continue
        k = kk - (n - 1)
        wm = w[kk] * dx
        if k >= 0:
            nv = n - k
            g[0:nv] += wm * (1.0 - x[k:n])
            g[k:n] += wm * (-x[0:nv])
        else:
            kk2 = -k
            nv = n - kk2
            g[kk2:n] += wm * (1.0 - x[0:nv])
            g[0:nv] += wm * (-x[kk2:n])
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def clean_slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

def clean_slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def clean_slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_polish(h, n, deadline, max_iters=200):
    h = project_h(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0), (300, 5.0), (1000, 10.0), (5000, 50.0), (20000, 200.0), (100000, 1000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time()
        t_sec = min(40, max(5, remaining / 2.0))
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 8, 'maxls': 40})
            h_new = project_h(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_polish(h, n, deadline, alpha=200000, maxiter=1500):
    h = project_h(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': clean_slsqp_con, 'jac': clean_slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(clean_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_h(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def coord_descent(h, n, deadline, max_sweeps=5000):
    h = project_h(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            old_hi = h[i]
            lo, hi_t = 0.0, 1.0
            for _ in range(18):
                m1 = lo + (hi_t - lo) / 3.0
                m2 = hi_t - (hi_t - lo) / 3.0
                h[i] = m1
                c1 = np.max(np.correlate(h, 1 - h, mode='full'))
                h[i] = m2
                c2 = np.max(np.correlate(h, 1 - h, mode='full'))
                if c1 <= c2:
                    hi_t = m2
                else:
                    lo = m1
            h[i] = (lo + hi_t) / 2.0
            # Also check boundaries
            h[i] = 0.0
            c0 = np.max(np.correlate(h, 1 - h, mode='full'))
            h[i] = 1.0
            c1b = np.max(np.correlate(h, 1 - h, mode='full'))
            h[i] = (lo + hi_t) / 2.0
            cm = np.max(np.correlate(h, 1 - h, mode='full'))
            if c0 < cm:
                h[i] = 0.0
            elif c1b < cm:
                h[i] = 1.0
        h = project_h(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def grad_pair_search(h, n, deadline, step_fracs=None):
    h = project_h(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    if step_fracs is None:
        step_fracs = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    alpha_s = [50, 200, 1000, 5000, 20000]
    for alpha in alpha_s:
        if time.time() > deadline:
            break
        for sf in step_fracs:
            if time.time() > deadline:
                break
            improved_any = False
            for _ in range(50):
                if time.time() > deadline:
                    return h, best_c
                c_full = np.correlate(h, 1 - h, mode='full') * dx
                m = c_full.max()
                sh = alpha * (c_full - m)
                mx = sh.max()
                w = np.exp(sh - mx)
                w /= w.sum()
                g = np.zeros(n)
                for kk in range(2 * n - 1):
                    if w[kk] < 1e-12:
                        continue
                    k = kk - (n - 1)
                    wm = w[kk] * dx
                    if k >= 0:
                        nv = n - k
                        g[0:nv] += wm * (1.0 - h[k:n])
                        g[k:n] += wm * (-h[0:nv])
                    else:
                        kk2 = -k
                        nv = n - kk2
                        g[kk2:n] += wm * (1.0 - h[0:nv])
                        g[0:nv] += wm * (-h[kk2:n])
                g_sort = np.argsort(g)
                pair_found = False
                for _p in range(20):
                    i = g_sort[_p]
                    j = g_sort[-(_p + 1)]
                    if i == j:
                        continue
                    delta_max_fwd = min(1.0 - h[i], h[j])
                    delta_max_bwd = min(1.0 - h[j], h[i])
                    if delta_max_fwd < 1e-6 and delta_max_bwd < 1e-6:
                        continue
                    # Try forward: h[i]+=d, h[j]-=d
                    d = min(sf, delta_max_fwd)
                    if d <= 0:
                        continue
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved_any = True
                        pair_found = True
                    else:
                        h[i] -= d
                        h[j] += d
                    # Try backward: h[j]+=d, h[i]-=d
                    d2 = min(sf, delta_max_bwd)
                    if d2 <= 0:
                        continue
                    h[j] += d2
                    h[i] -= d2
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved_any = True
                        pair_found = True
                    else:
                        h[j] -= d2
                        h[i] += d2
                    if pair_found:
                        break
                if not pair_found:
                    break
            if not improved_any:
                break
    h = project_h(h, n)
    return h, best_c

def local_pair_search(h, n, deadline, deltas=None, radius=40):
    h = project_h(h, n)
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = False
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def perturb_restart(h, n, seed, amount=0.02):
    rng = np.random.RandomState(seed)
    h_new = h.copy()
    mask = rng.rand(n) < 0.1
    h_new[mask] += rng.normal(0, amount, mask.sum())
    h_new = project_h(h_new, n)
    return h_new

    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    # Initialize

    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
