import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    if active_idx is not None:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:

def _lag_gradient(h, n, lag):
    """Gradient of overlap at signed lag w.r.t. h, times dx."""
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] -= h[m:n]
        grad[m:n] += (1.0 - h[0:n - m])
    else:
        m = -lag
        grad[0:n - m] += (1.0 - h[m:n])
        grad[m:n] -= h[0:n - m]
    return dx * grad

def subgradient_optimize(h, n, deadline, n_iters=300, top_k=5):
    """Subgradient method on max of overlaps with line search."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for it in range(n_iters):
        if time.time() > deadline - 5:

        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_k:]

        for idx in top_idx:
            lag = idx - (n - 1)

                h_curr = h_trial
                improved = True

    return best_h, best_c

                break
            # Pick from top of gradient (high) and bottom (low)
            i_candidates = order[::-1][np.random.randint(min(10, n))]
            j_candidates = order[np.random.randint(min(10, n))]
            
            i = int(i_candidates)
            j = int(j_candidates)
            if i == j:
                continue
            
            # Determine direction: move mass from high-grad to low-grad
            if grad[i] > grad[j]:
                src, dst = i, j
            elif grad[j] > grad[i]:
                src, dst = j, i
            else:
                continue
            
            max_delta = min(0.05, h_curr[src], 1.0 - h_curr[dst])
            if max_delta < 1e-10:
                continue
            
            for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
                delta = max_delta * frac
                if delta < 1e-10:
                    break
                h_curr[src] -= delta
                h_curr[dst] += delta
                c_new = compute_c5(h_curr, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_curr.copy()
                    n_moved += 1
                    break
                else:
                    h_curr[src] += delta
                    h_curr[dst] -= delta
        
        if n_moved == 0:
            # Try random pairs based on active lag gradient
            m = abs(lag)
            if m == 0 or m >= n:
                if rnd > 30:
                    break
                continue
            found_move = False
            for _ in range(200):
                if time.time() > deadline - 2:
                    break
                i = np.random.randint(0, n - m)
                j = i + m
                if h_curr[i] > h_curr[j]:
                    delta = min(0.02, (h_curr[i] - h_curr[j]) * 0.4, h_curr[i], 1 - h_curr[j])
                elif h_curr[j] > h_curr[i]:
                    delta = min(0.02, (h_curr[j] - h_curr[i]) * 0.4, h_curr[j], 1 - h_curr[i])
                else:
                    continue
                if delta < 1e-10:
                    continue
                if h_curr[i] > h_curr[j]:
                    h_curr[i] -= delta
                    h_curr[j] += delta
                else:
                    h_curr[i] += delta
                    h_curr[j] -= delta
                c_new = compute_c5(h_curr, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_curr.copy()
                    found_move = True
                else:
                    if h_curr[i] > h_curr[j] or (h_curr[i] < h_curr[j] and False):
                        pass
                    # undo
                    h_curr[i] += delta if h_curr[i] < h_curr[j] else -delta
                    h_curr[j] -= delta if h_curr[i] > h_curr[j] else -delta
            if not found_move and rnd > 50:
                break
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def block_local_search(h, n, deadline, max_rounds=100):
    """Block-based local search: shift small blocks of h values."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    rng = np.random.RandomState(42)
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        # Pick a random block and try to shift it
        block_size = rng.randint(3, min(20, n // 5) + 1)
        start = rng.randint(0, max(1, n - block_size))
        end = start + block_size
        shift = rng.choice([-1, 1]) * rng.randint(1, block_size + 1)
        
        if shift >= 0:
            new_start = start + shift
            new_end = end + shift
        else:
            new_end = end + shift
            new_start = start + shift
        
        if new_start < 0 or new_end > n:
            continue
        
        # Create trial
        h_trial = h_curr.copy()
        block_vals = h_trial[start:end].copy()
        dest_vals = h_trial[new_start:new_end].copy()
        
        # Try to average: reduce peak in block, fill trough in dest
        h_trial[start:end] = dest_vals.copy()
        h_trial[new_start:new_end] = block_vals
        
        h_trial = project_feasible(h_trial, n)
        c_new = compute_c5(h_trial, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_trial.copy()
            h_curr = h_trial
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    """Full optimization pipeline at a given resolution."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    remaining = deadline - time.time()
    
    # Phase 1: SLSQP with moderate alpha
    if time.time() < deadline - 30:
        h_new = slsqp_opt(best_h, n, 5000, maxiter=3000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 2: Subgradient method
    if time.time() < deadline - 60:
        sg_deadline = time.time() + min(60, (deadline - time.time()) * 0.3)
        h_sg, c_sg = subgradient_optimize(best_h, n, sg_deadline, n_iters=200, top_k=7)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    
    # Phase 3: SLSQP with high alpha
    if time.time() < deadline - 40:
        h_new = slsqp_opt(best_h, n, 50000, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 4: Active set SLSQP (top 10 lags)
    if time.time() < deadline - 30:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-10:]
        h_new = slsqp_opt(best_h, n, 100000, maxiter=3000, ftol=1e-16, active_idx=top_idx)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 5: Fine local search (gradient-directed)
    if time.time() < deadline - 20:
        ls_deadline = min(deadline - 5, time.time() + min(40, (deadline - time.time()) * 0.5))
        best_h, best_c = fine_local_search(best_h, n, ls_deadline, max_rounds=300)
    
    # Phase 6: Block search
    if time.time() < deadline - 10:
        best_h, best_c = block_local_search(best_h, n, deadline - 5, max_rounds=50)
    
    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 5:
        h_new = slsqp_opt(best_h, n, 200000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 8: Final local search
    if time.time() < deadline - 2:
        best_h, best_c = fine_local_search(best_h, n, deadline - 1, max_rounds=200)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Phase A: Optimize at parent resolution
    alloc_a = min(300, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc_a, rng)
    best_h, best_c = h_opt, c_opt
    
    # Phase B: Try higher resolutions
    for n_try in [400, 500, 600]:
        if time.time() > deadline - 120:
            break
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        
        alloc_b = min(150, (deadline - time.time()) * 0.3)
        if alloc_b < 60:
            break
        h_b, c_b = optimize_at_resolution(h_start, n_try, time.time() + alloc_b, rng)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b
    
    # Phase C: Try lower resolutions (sometimes discrete effects help)
    for n_try in [200, 250]:
        if time.time() > deadline - 100:
            break
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        
        alloc_c = min(100, (deadline - time.time()) * 0.3)
        if alloc_c < 50:
            break
        h_c, c_c = optimize_at_resolution(h_start, n_try, time.time() + alloc_c, rng)
        if c_c < best_c:
            best_c = c_c
            best_h = h_c
    
    # Phase D: Perturbation restarts at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.005 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc_r = min(70, (deadline - time.time() - 30) * 0.4)
            if alloc_r < 30:
                break
            h_r, c_r = optimize_at_resolution(hp, n_best, time.time() + alloc_r, rng)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    
    # Phase E: Final intensive local search + polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # SLSQP with very high alpha
        h_new = slsqp_opt(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Final local search
        best_h, best_c = fine_local_search(best_h, n, deadline - 2, max_rounds=500)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
