                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0

                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j

                h[best_j] = 0.0
                corr = np.correlate(h, 1-h, mode='full').astype(float)
                cur = corr.max() * dx
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c
