- f_k(h*) ≤ v* for all k
- h* minimizes the maximum

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_val(c, alpha):
    M = c.max()
    s = alpha * (c - M)
    # Clip for numerical stability
    s = np.maximum(s, -50)
    e = np.exp(s)
    return M + np.log(e.sum()) / alpha

def smax_weights(c, alpha):
    M = c.max()
    s = alpha * (c - M)
    s = np.maximum(s, -50)
    e = np.exp(s)
    return e / e.sum()

def obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    v = smax_val(c, alpha)
    return v + mu * (x.sum() - n / 2.0) ** 2

def grad_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smax_weights(c, alpha)
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += -wm * x[:nv]
        else:
            j = -k
            nv = n - j
            g[j:] += wm * (1.0 - x[:nv])
            g[:nv] += -wm * x[j:]
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    return smax_val(c, alpha)

def slsqp_grad(x, n, alpha):

    w = smax_weights(c, alpha)
    dx = 2.0 / n
    g = np.zeros(n)

def cons_fun_eq(x, n):
    return x.sum() - n / 2.0

def cons_jac_eq(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=200, alpha_start=100, alpha_end=5000000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [alpha_start]
    a = alpha_start
    while a < alpha_end:
        a = min(a * 5, alpha_end)
        alphas.append(a)

            break

                if c_new < best_c - 1e-16:

                    h = h_new

    return h, best_c
