```python
def binary_swap_delta(h, i, j, n):
    """Exact correlation change for binary swap h[i]:1→0, h[j]:0→1"""
    dc = np.zeros(2*n-1)
    
    # s > 0: full[n-1+s]
    for s in range(1, n):
        d = 0.0
        # Position i (1→0)
        if i >= s:
            d -= (1 - h[i-s])
        if i < n - s:
            d += h[i+s]
        # Position j (0→1)
        if j >= s:
            d += (1 - h[j-s])
        if j < n - s:
            d -= h[j+s]
        dc[n-1+s] += d
        
        # Position i (1→0) for full[n-1-s]
        d2 = 0.0
        if i < n - s:
            d2 -= (1 - h[i+s])
        if i >= s:
            d2 += h[i-s]
        # Position j (0→1)
        if j < n - s:
            d2 += (1 - h[j+s])
        if j >= s:
            d2 -= h[j-s]
        dc[n-1-s] += d2
    
    return dc
```
