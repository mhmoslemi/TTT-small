def shift_gradient(h, n, shift_idx):
    grad = np.zeros(n)
    dx = 2.0 / n
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

        if h_pushed[i] < 0.5:
            h_pushed[i] = h_pushed[i] - strength * h_pushed[i] * (1 - h_pushed[i])
        else:
            h_pushed[i] = h_pushed[i] + strength * h_pushed[i] * (1 - h_pushed[i])
    h_pushed = np.clip(h_pushed, 0.0, 1.0)
    h_pushed = project_feasible(h_pushed, n)
    return h_pushed

def lbfgs_active(h, n, deadline, alphas=None, top_k=10, mu=0.001):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 500000]
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Select shifts within 2% of max
        threshold = max_val * (1.0 - 0.02)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(-c_vec[active])
            active = active[:top_k]
        if len(active) < 1:
            active = np.argsort(-c_vec)[:top_k]
        
        try:
            t_sub = min(30, (deadline - time.time()) * 0.4)
            res = minimize(active_set_obj, best_h, jac=active_set_grad,
                          args=(n, active, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': 20000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def full_smooth_lbfgs(h, n, deadline, alphas=None, mus=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    if alphas is None:
        alphas = [100, 500, 2000, 10000]
    if mus is None:
        mus = [5.0, 20.0, 100.0, 500.0]
    
    for i, alpha in enumerate(alphas):
        if time.time() > deadline:
            break
        mu = mus[i % len(mus)]
        
        def obj(x):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            if mx > 500:
                val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
            else:
                val = m + np.log(np.sum(np.exp(sh))) / alpha
            val += mu * (x.sum() - n / 2.0) ** 2
            return float(val)
        
        def grad(x):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            w = np.exp(sh - mx)
            w /= w.sum()
            dx = 2.0 / n
            g = np.zeros(n)
            for mm in range(2 * n - 1):
                if w[mm] < 1e-14:
                    continue
                k = mm - (n - 1)
                wm = w[mm] * dx
                if k >= 0:
                    nv = n - k
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk:n])
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def coordinate_descent_polish(h, n, deadline, n_sweeps=20):
    """Polish by trying small perturbations at individual coordinates."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            # Try small adjustments to h[i]
            base = h[i]
            for delta in [0.001, 0.0005, -0.001, -0.0005, 0.0001, -0.0001]:
                new_val = base + delta
                if new_val < 0 or new_val > 1:
                    continue
                # Compensate sum by adjusting another coordinate
                diff = -delta
                # Find best compensation: adjust the coordinate with most room
                h_try = h.copy()
                h_try[i] = new_val
                h_try[1 - i % n] += diff
                if h_try[1 - i % n] < 0 or h_try[1 - i % n] > 1:
                    # Try another compensation
                    candidate = None
                    if diff > 0:
                        room = 1.0 - h_try
                        room[i] = -1
                        room[1 - i % n] = -1
                        idx = np.argmax(room)
                        if room[idx] < 1e-6:
                            continue
                        h_try[1 - i % n] = h[1 - i % n]
                        h_try[idx] += diff
                    else:
                        cand = h_try.copy()
                        cand[i] = -1
                        cand[1 - i % n] = -1
                        idx = np.argmin(cand)
                        if cand[idx] < 1e-6:
                            continue
                        h_try[1 - i % n] = h[1 - i % n]
                        h_try[idx] += diff
                    if h_try[idx] < 0 or h_try[idx] > 1:
                        continue
                else:
                    h_try = h.copy()
                    h_try[i] = new_val
                    h_try[1 - i % n] -= delta
                
                h_try = np.clip(h_try, 0, 1)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
                    best_h = h_try.copy()
                    improved = True
                    break
    
    return best_h, best_c

def random_perturb(h, n, seed=0):
    """Create a perturbed version for multi-start."""
    rng = np.random.RandomState(seed)
    h_p = h.copy()
    # Add small random noise
    noise = rng.randn(n) * 0.02
    h_p += noise
    h_p = np.clip(h_p, 0, 1)
    h_p = project_feasible(h_p, n)
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick L-BFGS at original resolution
    if time.time() < deadline - 500:
        t1 = min(time.time() + 60, deadline - 500)
        best_h, best_c = full_smooth_lbfgs(best_h, n_orig, t1,
            alphas=[200, 1000], mus=[50.0, 500.0])
    
    # Phase 2: Try at n=500 (or upsample if needed)
    n = 500
    if n_orig != n:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n)
        h_work = np.interp(x_new, x_old, best_h)
        h_work = project_feasible(h_work, n)
    else:
        h_work = best_h.copy()
    
    c_work = compute_c5(h_work, n)
    if c_work < best_c:
        best_c = c_work
        best_h = h_work
    
    # Phase 3: Active-set L-BFGS at n=500
    if time.time() < deadline - 350:
        t3 = min(time.time() + 150, deadline - 350)
        h_work, c_work = lbfgs_active(h_work, n, t3,
            alphas=[1000, 5000, 20000, 100000, 500000],
            top_k=8, mu=0.001)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work
    
    # Phase 4: Subgradient descent
    if time.time() < deadline - 250:
        t4 = min(time.time() + 80, deadline - 250)
        h_work, c_work = subgradient_descent(h_work, n, t4,
            lr_init=0.003, decay=0.985, top_frac=0.03)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work
    
    # Phase 5: Another active-set round with tighter focus
    if time.time() < deadline - 200:
        t5 = min(time.time() + 60, deadline - 200)
        h_work, c_work = lbfgs_active(h_work, n, t5,
            alphas=[10000, 50000, 200000],
            top_k=5, mu=0.0005)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work
    
    # Phase 6: Try binary push + reoptimize
    if time.time() < deadline - 150:
        h_bin = binary_push(best_h, n, strength=0.2)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        # Optimize from the binary-pushed point
        t6 = min(time.time() + 40, deadline - 130)
        h_work2, c_work2 = lbfgs_active(h_bin, n, t6,
            alphas=[5000, 50000], top_k=6, mu=0.001)
        if c_work2 < best_c:
            best_c = c_work2
            best_h = h_work2
    
    # Phase 7: Up-try at n=1000
    if time.time() < deadline - 80:
        n_hi = 1000
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_hi)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        
        t7 = min(time.time() + 50, deadline - 30)
        h_hi, c_hi = lbfgs_active(h_hi, n_hi, t7,
            alphas=[5000, 20000, 100000], top_k=6, mu=0.001)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
    
    # Phase 8: Final subgradient polish at best resolution
    if time.time() < deadline - 20:
        n_final = len(best_h)
        t8 = deadline - 15
        best_h, best_c = subgradient_descent(best_h, n_final, t8,
            lr_init=0.001, decay=0.99, top_frac=0.02)
    
    # Phase 9: Final active-set polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_work3, c_work3 = lbfgs_active(best_h, n_final, deadline - 3,
            alphas=[100000, 500000], top_k=4, mu=0.0001)
        if c_work3 < best_c:
            best_c = c_work3
            best_h = h_work3
    
    # Final projection and verification
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
