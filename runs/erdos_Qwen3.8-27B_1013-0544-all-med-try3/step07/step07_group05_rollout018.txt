        # h[i] * (1-h[i+k]) is large when h[i]=1 and h[i+k]=0

            # Positions where h[i]=1 and h[i+k]=0 are "bad"

            for i in range(n - k):
                if h[i] >= 0.5 and h[i+k] < 0.5:
                    bad_sources.append(i)
            # Also positions where h[i]=0 and h[i-k]=1 (i.e., h[i+k]=1 and h[i]=0)
            # These contribute to the "complement" side
            
            if bad_sources:
                # Pick a random bad source to change
                i = bad_sources[rng.randint(len(bad_sources))]
                # Swap: h[i] 1→0, find a zero to set to 1
                j = zeros[rng.randint(len(zeros))]
                if j == i:

                h[i], h[j] = 0.0, 1.0
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    ones.remove(i)
                    zeros.remove(j) if j in zeros else None
                    ones.append(j)
                    zeros.append(i)
                    continue
                else:
                    h[i], h[j] = 1.0, 0.0
                    continue
        
        # Fallback: random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c_new = compute_c5(h, n)

            ones.remove(i)

        else:

                        best_h = h.copy()
                        improved = True
                    else:

            h[idx] = 0

            h[rand_pick] = 0

            h[z[:half-count]] = 1

def compute_c5(h, n):

def compute_correlation(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    grad = np.zeros(n)
    for idx in idxs:
        k = int(idx - (n - 1))
        g = np.zeros(n)
        if k == 0:
            g = (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n - k] += (1.0 - h[k:]) * dx
            g[k:] -= h[:n - k] * dx
        else:
            ka = -k
            g[:n - ka] += (1.0 - h[ka:]) * dx
            g[ka:] -= h[:n - ka] * dx

    if best_h is not None and time.time() < deadline - 12:
        n = len(best_h)
        # More swap search
        h_b = binary_round(best_h, n)
        h_b, c_b = swap_search(h_b, n, deadline - 25, rng, max_iters=500000, stall_limit=20000)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b.copy()
        # Double swap
        if time.time() < deadline - 18:
            h_d, c_d = double_swap_search(best_h, n, deadline - 18, rng, max_iters=80000)
            if c_d < best_c:
                best_c = c_d
                best_h = h_d.copy()
        # Gradient
        if time.time() < deadline - 12:
            h_g, c_g = gradient_bottleneck(best_h, n, deadline - 12, lr_init=0.02)
            if c_g < best_c:
                best_c = c_g
                best_h = h_g.copy()
        # Final continuous
        if time.time() < deadline - 4:
            h_f, c_f = continuous_fine(best_h, n, deadline - 4, rng, refine_iters=30000)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f.copy()

    if best_h is None:
        n = n_parent
        best_h = np.full(n, 0.5)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
