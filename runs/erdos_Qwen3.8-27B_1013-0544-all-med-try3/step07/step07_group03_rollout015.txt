import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_influence(h, n):
    """v[i][k] = influence of h[i] on correlation at index k"""
    # corr index n-1+m corresponds to lag m
    v = np.zeros((n, 2*n - 1))
    for i in range(n):
        # v[i][n-1+m] = 1 - h[i+m] - h[i-m] (0 if out of range)
        m_lo = max(-i, -(n-1))
        m_hi = min(n-1-i, n-1)
        m_vals = np.arange(m_lo, m_hi + 1)
        idx = m_vals + (n-1)
        v[i, idx] = 1.0 - h[i + m_vals] - h[i - m_vals]
    return v

def binary_greedy_search(h, n, deadline, max_iter=100000):
    """Greedy binary local search using vectorized evaluation"""
    # Ensure h is binary
    h = (h >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h.sum())
    n_target = n // 2
    if n_ones > n_target:
        ones_idx = np.where(h == 1)[0]
        # Remove random ones
        h[ones_idx[-(n_ones - n_target):]] = 0
    elif n_ones < n_target:
        zeros_idx = np.where(h == 0)[0]
        h[zeros_idx[:n_target - n_ones]] = 1
    
    c5_val = compute_c5(h, n)
    c = np.correlate(h, 1 - h, mode='full')  # unscaled
    c_max = c.max()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        v = compute_influence(h, n)
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        V1 = v[ones_idx, :]  # (n1, 2n-1)
        V0 = v[zeros_idx, :]  # (n0, 2n-1)
        
        # For swap (a from ones, b from zeros): new_c = c - V1[a] + V0[b]
        # new_max = max(c - V1[a] + V0[b])
        # = max over k of c[k] - V1[a][k] + V0[b][k]
        
        # Vectorized: for all pairs simultaneously
        # diff[a,b,k] = V0[b,k] - V1[a,k]
        diff = V0[None, :, :] - V1[:, None, :]  # (n1, n0, 2n-1)
        new_max = (c[None, None, :] + diff).max(axis=2)  # (n1, n0)
        
        best_pos = np.argmin(new_max)
        best_a = best_pos // len(zeros_idx)
        best_b = best_pos % len(zeros_idx)
        best_val = new_max[best_a, best_b]
        
        if best_val >= c_max - 1e-14:
            break  # No improvement possible
        
        # Apply swap
        i_swap = ones_idx[best_a]
        j_swap = zeros_idx[best_b]
        h[i_swap] = 0
        h[j_swap] = 1
        
        c = np.correlate(h, 1 - h, mode='full')
        c_max = c.max()
        c5_val = c_max * (2.0 / n)
    
    return h, c5_val

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search_fast(h, n, deadline, deltas, radius=None):
    """Local search with 2-point moves"""
    best_c = compute_c5(h, n)
    cr = 2.0 / n
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            c_curr = np.correlate(h, 1 - h, mode='full')
            c_max_val = c_curr.max()
            
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                
                if radius is not None:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                else:
                    lo, hi = 0, n
                
                for j in range(lo, hi):
                    if i == j:
                        continue
                    
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = np.max(np.correlate(h, 1 - h, mode='full')) * cr
                        if c < best_c - 1e-15:

                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = np.max(np.correlate(h, 1 - h, mode='full')) * cr
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            
            if not improved:
                break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_influence(h, n):
    v = np.zeros((n, 2 * n - 1))
    for i in range(n):
        m_lo = max(-i, -(n - 1))
        m_hi = min(n - 1 - i, n - 1)
        m_vals = np.arange(m_lo, m_hi + 1)
        idx = m_vals + (n - 1)
        v[i, idx] = 1.0 - h[i + m_vals] - h[i - m_vals]
    return v

def binary_greedy_search(h, n, deadline, max_iter=50000):
    h = (h >= 0.5).astype(float).copy()
    n_target = n // 2
    n_ones = int(h.sum())
    if n_ones > n_target:
        ones_idx = np.where(h == 1)[0]
        rng_local = np.random.RandomState(int(time.time() * 1000) % 100000)
        remove = rng_local.choice(ones_idx, n_ones - n_target, replace=False)
        h[remove] = 0
    elif n_ones < n_target:
        zeros_idx = np.where(h == 0)[0]
        rng_local = np.random.RandomState(int(time.time() * 1000) % 100000)
        add = rng_local.choice(zeros_idx, n_target - n_ones, replace=False)
        h[add] = 1
    
    c5_val = compute_c5(h, n)
    c = np.correlate(h, 1 - h, mode='full').copy()
    c_max = c.max()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        v = compute_influence(h, n)
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        V1 = v[ones_idx, :]
        V0 = v[zeros_idx, :]
        diff_arr = V0[None, :, :] - V1[:, None, :]
        new_max = (c[None, None, :] + diff_arr).max(axis=2)
        best_pos = np.argmin(new_max)
        best_a = best_pos // len(zeros_idx)
        best_b = best_pos % len(zeros_idx)
        best_val = new_max[best_a, best_b]
        if best_val >= c_max - 1e-12:
            break
        i_swap = ones_idx[best_a]
        j_swap = zeros_idx[best_b]
        h[i_swap] = 0.0
        h[j_swap] = 1.0
        c = np.correlate(h, 1 - h, mode='full')
        c_max = c.max()
        c5_val = c_max * (2.0 / n)
    return h, c5_val

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def incremental_pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    cr = 2.0 / n
    c_curr = np.correlate(h, 1 - h, mode='full').copy()
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            v = compute_influence(h, n)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is not None:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                else:
                    lo, hi = 0, n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        delta_c = d * (v[i] - v[j])
                        new_max_val = (c_curr + delta_c).max() * cr
                        if new_max_val < best_c - 1e-14:
                            h[i] += d
                            h[j] -= d
                            c_curr = c_curr + delta_c
                            best_c = new_max_val
                            improved = True
                            v = compute_influence(h, n)
                    if h[i] - d >= -1e-15 and h[j] + d <= 1.0 + 1e-15:
                        delta_c = -d * (v[i] - v[j])
                        new_max_val = (c_curr + delta_c).max() * cr
                        if new_max_val < best_c - 1e-14:
                            h[i] -= d
                            h[j] += d
                            c_curr = c_curr + delta_c
                            best_c = new_max_val
                            improved = True
                            v = compute_influence(h, n)
            if not improved:
                break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def steepest_2point(h, n, deadline, max_iter=5000):
    best_c = compute_c5(h, n)
    cr = 2.0 / n
    c_curr = np.correlate(h, 1 - h, mode='full').copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        v = compute_influence(h, n)
        best_improve = 1e-14
        best_move = None
        best_delta_c = None
        for i in range(n):
            for j in range(i + 1, n):
                if h[i] + 0.02 <= 1.0 and h[j] - 0.02 >= 0:
                    dc = 0.02 * (v[i] - v[j])
                    nm = (c_curr + dc).max() * cr
                    imp = best_c - nm
                    if imp > best_improve:
                        best_improve = imp
                        best_move = (i, j, 0.02)
                        best_delta_c = dc
                if h[i] - 0.02 >= 0 and h[j] + 0.02 <= 1.0:
                    dc = -0.02 * (v[i] - v[j])
                    nm = (c_curr + dc).max() * cr
                    imp = best_c - nm
                    if imp > best_improve:
                        best_improve = imp
                        best_move = (i, j, -0.02)
                        best_delta_c = dc
        if best_move is None:
            for i in range(n):
                for j in range(i + 1, n):
                    if h[i] + 0.005 <= 1.0 and h[j] - 0.005 >= 0:
                        dc = 0.005 * (v[i] - v[j])
                        nm = (c_curr + dc).max() * cr
                        imp = best_c - nm
                        if imp > best_improve:
                            best_improve = imp
                            best_move = (i, j, 0.005)
                            best_delta_c = dc
                    if h[i] - 0.005 >= 0 and h[j] + 0.005 <= 1.0:
                        dc = -0.005 * (v[i] - v[j])
                        nm = (c_curr + dc).max() * cr
                        imp = best_c - nm
                        if imp > best_improve:
                            best_improve = imp
                            best_move = (i, j, -0.005)
                            best_delta_c = dc
        if best_move is None:
            break
        i, j, d = best_move
        h[i] += d
        h[j] -= d
        c_curr = c_curr + best_delta_c
        best_c = (c_curr).max() * cr
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
