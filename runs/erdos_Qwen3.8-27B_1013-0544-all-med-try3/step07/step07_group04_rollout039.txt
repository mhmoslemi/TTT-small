The smooth max of c with parameter alpha is:
SM(c, alpha) = max(c) + log(sum(exp(alpha*(c - max(c))))) / alpha

This is always ≥ max(c) and approaches max(c) as alpha → ∞.

The gradient of SM w.r.t. x is:
sum_k softmax_k * dc_k/dx

where softmax_k = exp(alpha*(c_k - max(c))) / sum_j exp(alpha*(c_j - max(c)))

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):

    shifted = alpha * (c - m)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    val = m + np.log(w_sum) / alpha

def _lbfgs_full_obj(x, n, alpha, penalty):

    c = np.correlate(x, 1 - x, mode='full') * dx
    val, _ = _smooth_max_weights(c, alpha)
    pen = np.sum(x) - n / 2.0
    val += penalty * pen * pen
    return float(val)

def _lbfgs_full_grad(x, n, alpha, penalty):

    c = np.correlate(x, 1 - x, mode='full') * dx
    _, w = _smooth_max_weights(c, alpha)
    n_lags = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_lags):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        m_ = abs(lag)
        if m_ >= n:
            continue
        if m_ == 0:
            grad += wk * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[:n - m_] += wk * dx * (1.0 - x[m_:])
            grad[m_:] -= wk * dx * x[:n - m_]

def _lbfgs_active_grad(x, n, alpha, penalty, top_idx):

    grad += 2.0 * penalty * pen
    return grad

def lbfgs_optimize(h, n, alpha, penalty, maxiter=5000, deadline=None, active_idx=None):
    h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if active_idx is not None:
        res = minimize(_lbfgs_active_obj, h, args=(n, alpha, penalty, active_idx),
                       jac=_lbfgs_active_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
    else:
        res = minimize(_lbfgs_full_obj, h, args=(n, alpha, penalty),
                       jac=_lbfgs_full_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
    if deadline is not None and time.time() > deadline:
        return h
    return project_feasible(res.x, n)

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if m == 0:
            grad_vec = (1.0 - 2.0 * h_curr)
        elif lag > 0:
            grad_vec = np.zeros(n)
            grad_vec[:n - m] -= h_curr[m:]
            grad_vec[m:] += (1.0 - h_curr[:n - m])
        else:
            grad_vec = np.zeros(n)
            grad_vec[m:] += (1.0 - h_curr[:n - m])
            grad_vec[:n - m] -= h_curr[m:]
        cons.append(c[k] + dx * (grad_vec @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=50):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

def local_search(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.004, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and (time.time() < deadline):
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if delta < 1e-9:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if delta < 1e-9:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def simulated_annealing(h, n, deadline, seed=99):
    rng = np.random.RandomState(seed)
    h = h.copy()
    curr_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = curr_c
    T = 0.003
    max_iters = 3000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = max(T * 0.9995, 1e-8)
        n_pert = max(1, n // 80)
        idx = rng.choice(n, n_pert, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.uniform(-0.015, 0.015, n_pert)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        delta = c_new - curr_c
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def multi_point_local_search(h, n, deadline, seed=77):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    max_iters = 2000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        n_candidates = min(50, n - m)
        idx = rng.choice(n - m, n_candidates, replace=False)
        best_move = None
        best_c_new = best_c
        for i in idx:
            j = i + m
            if h[i] > h[j]:
                for delta in [0.003, 0.001, 0.0005]:
                    d = min(delta, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if d < 1e-9:
                        continue
                    h[i] -= d
                    h[j] += d
                    c_new = compute_c5(h, n)
                    h[i] += d
                    h[j] -= d
                    if c_new < best_c_new:
                        best_c_new = c_new
                        best_move = (i, j, -d)
            elif h[j] > h[i]:
                for delta in [0.003, 0.001, 0.0005]:
                    d = min(delta, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if d < 1e-9:
                        continue
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    h[i] -= d
                    h[j] += d
                    if c_new < best_c_new:
                        best_c_new = c_new
                        best_move = (i, j, d)
        if best_move is not None:
            i, j, d = best_move
            h[i] += d
            h[j] -= d
            best_c = best_c_new
        else:
            break
    return h, best_c

def optimize_single(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if time.time() < deadline - 40:
        oo_end = time.time() + min(60, (deadline - time.time()) * 0.3)
        h_oo, c_oo = outer_approx(best_h, n, oo_end, max_iters=30)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo.copy()

    penalty = 1e4
    for alpha in [5000, 20000, 100000]:
        if time.time() > deadline - 20:
            break
        maxiter = min(3000, max(500, int((deadline - time.time() - 10) / 0.5)))
        h_new = lbfgs_optimize(best_h, n, alpha, penalty, maxiter=maxiter)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()

    if time.time() < deadline - 15:
        c = compute_all_lags(best_h, n)
        top_m = min(8, 2 * n - 1)
        top_idx = np.argsort(c)[-top_m:]
        for alpha in [30000, 150000]:
            if time.time() > deadline - 15:
                break
            maxiter = min(2000, max(300, int((deadline - time.time() - 10) / 0.5)))
            h_new = lbfgs_optimize(best_h, n, alpha, penalty, maxiter=maxiter, active_idx=top_idx)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new.copy()

    if time.time() < deadline - 5:
        best_h, best_c = local_search(best_h, n, min(deadline - 1, time.time() + 20))

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize the parent at its resolution
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.30)
        h_opt, c_opt = optimize_single(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Phase 2: Try different resolutions
    n_candidates = []
    if n_init is not None:
        n_candidates = [n_init - 50, n_init + 50, n_init + 100, n_init + 150]
        n_candidates = [n for n in n_candidates if 80 <= n <= 700]
        if n_init not in n_candidates:
            n_candidates.insert(0, n_init)
    else:
        n_candidates = [200, 300, 400, 500]

    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 40:
            break
        alloc = min(120, remaining * 0.25)
        if alloc < 30:
            break

        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)

        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_single(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 3: Diversification - simulated annealing and restarts
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 30

        # Simulated annealing
        sa_time = min(80, remaining * 0.3)
        if sa_time > 20:
            h_sa, c_sa = simulated_annealing(best_h, n, time.time() + sa_time, seed=seed + 1)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.copy()

        # Multiple restarts from best + perturbation
        n_restarts = max(1, int((deadline - time.time() - 30) / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 25) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_single(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    # Phase 4: Final intensive polish on the best solution
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        penalty = 1e5

        # High-alpha L-BFGS-B
        for alpha in [50000, 200000, 800000]:
            if time.time() > deadline - 5:
                break
            maxiter = min(5000, max(500, int((deadline - time.time() - 3) / 0.4)))
            if maxiter < 100:
                break
            h_new = lbfgs_optimize(best_h, n, alpha, penalty, maxiter=maxiter)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()

        # Active-set L-BFGS-B on top lags
        if time.time() < deadline - 5:
            c = compute_all_lags(best_h, n)
            top_m = min(5, 2 * n - 1)
            top_idx = np.argsort(c)[-top_m:]
            h_new = lbfgs_optimize(best_h, n, 500000, penalty, maxiter=3000, active_idx=top_idx)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()

        # Multi-point local search
        if time.time() < deadline - 3:
            best_h, best_c = multi_point_local_search(best_h, n, deadline - 2, seed=seed + 42)

        # Final local search
        if time.time() < deadline - 1:
            best_h, best_c = local_search(best_h, n, deadline - 0.5)

        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
