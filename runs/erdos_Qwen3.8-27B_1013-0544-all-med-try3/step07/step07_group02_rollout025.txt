import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def active_lag_gradient(h, n, top_k=10):
    """Compute gradient of smooth max using only top-k lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Get top-k lags
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        lag = k - (n - 1)
        if lag == 0:
            grad += 1.0 - 2.0 * h
        elif lag > 0:
            grad[lag:n] += (1.0 - h[0:n-lag])
            grad[0:n-lag] += -h[lag:n]
        else:
            m = -lag
            grad[0:n-m] += (1.0 - h[m:n])
            grad[m:n] += -h[0:n-m]
    return grad * dx / len(top_idx)

def flatten_lag(h, n, m, eps):
    """Apply flattening for lag m using Neumann BC Laplacian along m-spaced paths."""
    h_new = h.copy()
    for r in range(m):
        path = np.arange(r, n, m)
        t = len(path)
        if t < 2:
            continue
        vals = h_new[path]
        change = np.zeros(t)
        change[0] = eps * (vals[1] - vals[0])
        change[-1] = eps * (vals[-2] - vals[-1])
        if t > 2:
            change[1:-1] = eps * (vals[:-2] - 2.0*vals[1:-1] + vals[2:])
        h_new[path] = np.clip(vals + change, 0.0, 1.0)
    return h_new

        h = h_new

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def flatten_lag(h, n, m, eps):
    """Apply Neumann-BC Laplacian smoothing along m-spaced paths."""
    h_new = h.copy()
    for r in range(m):
        path = np.arange(r, n, m)
        t = len(path)
        if t < 2:
            continue
        vals = h_new[path]
        change = np.zeros(t)
        change[0] = eps * (vals[1] - vals[0])
        change[-1] = eps * (vals[-2] - vals[-1])
        if t > 2:
            change[1:-1] = eps * (vals[:-2] - 2.0 * vals[1:-1] + vals[2:])
        h_new[path] = np.clip(vals + change, 0.0, 1.0)

def gradient_step(h, n, top_k=10, lr=0.5):
    """Single gradient step using top-k lags."""
    dx = 2.0 / n

        if lag == 0:

            grad[0:n-lag] += -h[lag:n]

            grad[0:n-m] += (1.0 - h[m:n])
            grad[m:n] += -h[0:n-m]
