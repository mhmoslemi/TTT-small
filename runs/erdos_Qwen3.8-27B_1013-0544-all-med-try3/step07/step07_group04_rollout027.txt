def subgradient_step(h, n, alpha):
    """Take one projected subgradient step."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Subgradient: pick the max lag (or average of near-max lags)
    max_val = c.max()
    # Soft subgradient: weight by exp(α(c - max_val))
    shifted = alpha * (c - max_val)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Compute gradient: weighted sum of lag gradients
    grad = np.zeros(n)

        lag = k - (n - 1)

    # Step

    return h_new

            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(h - lam, 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - x
            tot = room.sum()
            if tot > 1e-12:
                x += (diff / tot) * room
        else:
            tot = x.sum()
            if tot > 1e-12:
                x += (diff / tot) * x
    return np.clip(x, 0, 1)

def lag_gradient(h, n, lag):
    """Gradient of overlap at given lag, without dx factor."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # c = sum_{i=0}^{n-1-m} h[i+m] * (1-h[i])
        # d/dh[j] = (1-h[j-m]) for j >= m (h[j] appears as h[i+m] with i=j-m)
        #           - h[j+m] for j <= n-1-m (h[j] appears as h[i] with i=j, derivative of 1-h[i] is -1)
        # Wait: c = sum_{i=0}^{n-1-m} h[i+m] * (1-h[i])
        # If h[j] appears as h[i+m] (i.e., j = i+m, so i = j-m): contributes (1-h[j-m])
        #   valid when 0 <= j-m <= n-1-m, i.e., m <= j <= n-1
        # If h[j] appears as h[i] (i.e., j = i): contributes -h[j+m] (derivative of 1-h[i] w.r.t. h[j] is -1 when j=i, and h[i+m] = h[j+m])
        #   valid when 0 <= j <= n-1-m
        grad[m:] += (1.0 - h[:n-m])       # for j >= m: 1-h[j-m], h[:n-m] = h[0..n-m-1], so h[j-m] for j=m..n-1
        grad[:n-m] -= h[m:]                # for j <= n-1-m: -h[j+m], h[m:] = h[m..n-1], so h[j+m] for j=0..n-1-m
    else:
        # lag < 0, m = -lag
        # c = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])
        # d/dh[j] = (1-h[j+m]) for j <= n-1-m (h[j] appears as h[i] with i=j)
        #           -h[j-m] for j >= m (h[j] appears as h[i+m] with i=j-m)
        grad[:n-m] += (1.0 - h[m:])        # for j=0..n-1-m: 1-h[j+m]
        grad[m:] -= h[:n-m]                # for j=m..n-1: -h[j-m]
    return grad

def outer_approx_step(h_curr, n, trust, max_lags=None):
    """One outer approximation step: solve LP linearization."""
    dx = 2.0 / n
    overlaps = compute_all_overlaps(h_curr, n)
    max_ov = overlaps.max()
    
    # Select which lags to include in the LP
    if max_lags is not None:
        threshold = max_ov - 0.001  # include lags within 0.001 of max
        active_mask = overlaps >= threshold
        if active_mask.sum() < 5:
            # At least include top 5
            top_idx = np.argsort(overlaps)[-5:]
            active_mask = np.zeros(len(overlaps), dtype=bool)
            active_mask[top_idx] = True
    else:
        active_mask = np.ones(len(overlaps), dtype=bool)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(cp.abs(h_var - h_curr), np.inf) <= trust]
    
    for k in range(len(overlaps)):
        if not active_mask[k]:
            continue
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        # Linearization: overlaps[k] + dx * grad · (h - h_curr) <= t
        cons.append(overlaps[k] + dx * (grad @ h_var) - dx * (grad @ h_curr) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.PDLP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.GLOP, verbose=False)
            except Exception:
                return h_curr.copy(), None
    
    if h_var.value is None:
        return h_curr.copy(), None
    h_new = np.array(h_var.value).flatten()
    return project_feasible(h_new, n), None

def outer_approx_loop(h, n, deadline, max_iters=200):
    """Iterate outer approximation with decreasing trust region."""
    h_curr = project_feasible(h.copy(), n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.05
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        h_new, _ = outer_approx_step(h_curr, n, trust)
        if h_new is None or np.all(h_new == h_curr):
            trust *= 0.7
            continue
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step_size = np.max(np.abs(h_new - h_curr))
        h_curr = h_new
        if step_size < 1e-9:
            # Convergence: try smaller trust
            trust *= 0.5
            if trust < 1e-6:
                break
        elif step_size < 1e-7:
            trust *= 0.8
        else:
            # Good progress, maintain or slightly reduce trust
            trust = max(trust * 0.9, 0.01)
    
    return best_h, best_c

def slsqp_smooth_max(h, n, alpha, maxiter=3000):
    """Optimize smooth max using SLSQP."""
    dx = 2.0 / n
    
    def smooth_obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        max_s = shifted.max()
        if max_s > 500:
            return float(m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def l_bfgsb_augmented(h, n, alpha, rho_init=10.0, max_outer=50, max_iter_inner=500):
    """L-BFGS-B with augmented Lagrangian for the sum constraint."""
    dx = 2.0 / n
    h_curr = h.copy()
    rho = rho_init
    lam = 0.0
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for outer in range(max_outer):
        if time.time() > deadline_global - 10:
            break
        
        def obj(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            m = c.max()
            shifted = alpha * (c - m)
            max_s = shifted.max()
            lse = m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha
            s = x.sum() - n / 2.0
            return lse + rho / 2.0 * s * s + lam * s
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h_curr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter_inner, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = res.x.copy()
        except Exception:
            break
        
        s = h_new.sum() - n / 2.0
        if abs(s) < 1e-8:
            break
        lam = lam + rho * s
        rho *= 2.0
        if rho > 1e8:
            break
        h_curr = project_feasible(h_new, n)
        c_new = compute_c5(h_curr, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_curr.copy()
    
    return best_h, best_c

def local_binary_search(h, n, deadline):
    """Local search that tries to push h toward binary values."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    # First, try flipping fractional values to 0 or 1
    for _ in range(3):
        if time.time() > deadline:
            break
        changed = True
        while changed and time.time() < deadline:
            changed = False
            # Find indices with values in (0.01, 0.99)
            frac_idx = np.where((h > 0.01) & (h < 0.99))[0]
            for i in frac_idx:
                if time.time() > deadline:
                    break
                # Try setting h[i] to 0 or 1
                for target in [0.0, 1.0]:
                    delta = target - h[i]
                    if abs(delta) < 1e-6:
                        continue
                    h[i] = target
                    # Adjust another index to maintain sum
                    # Find the index closest to 0.5 to adjust
                    remaining = n / 2.0 - h.sum()
                    if remaining > 0:
                        # Need to add 'remaining' somewhere
                        candidates = np.where(h < 1.0 - 1e-6)[0]
                        if len(candidates) == 0:
                            h[i] = target - delta  # revert
                            continue
                        # Add to the index with most room
                        room = 1.0 - h[candidates]
                        j = candidates[np.argmax(room)]
                        add = min(remaining, room[j])
                        h[j] += add
                        remaining -= add
                        if remaining > 1e-6:
                            # Distribute among others
                            for j2 in candidates:
                                if j2 == j:
                                    continue
                                if remaining <= 0:
                                    break
                                room2 = 1.0 - h[j2]
                                add2 = min(remaining, room2)
                                h[j2] += add2
                                remaining -= add2
                    elif remaining < 0:
                        # Need to subtract
                        candidates = np.where(h > 1e-6)[0]
                        if len(candidates) == 0:
                            h[i] = target - delta
                            continue
                        remove = min(-remaining, h[candidates])
                        j = candidates[np.argmin(h[candidates])]
                        h[j] -= remove
                        remaining += remove
                        if remaining < -1e-6:
                            for j2 in candidates:
                                if j2 == j:
                                    continue
                                if remaining >= 0:
                                    break
                                remove2 = min(-remaining, h[j2])
                                h[j2] -= remove2
                                remaining += remove2
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        changed = True
                    else:
                        h[i] = target - delta
                        h = project_feasible(h, n)
    
    return h, best_c
