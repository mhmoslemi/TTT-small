    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def grad_corr_matrix(h, n):
    """Compute full gradient matrix of shape (2n-1, n) for all correlations."""
    dx = 2.0 / n
    nc = 2 * n - 1
    G = np.zeros((nc, n))
    for m in range(nc):
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            G[m, :nv] = (1.0 - h[k:n])
            G[m, k:n] = -h[0:nv]
        else:
            kk = -k
            nv = n - kk
            G[m, kk:n] = (1.0 - h[0:nv])
            G[m, 0:nv] = -h[kk:n]

def lp_subproblem(h0, n):
    """Solve the LP outer approximation subproblem."""
    c0 = compute_corr(h0, n)
    G = grad_corr_matrix(h0, n)  # (2n-1, n)
    const = c0 - G @ h0  # constant term for each constraint

    t_var = cp.Variable()

        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0

    for m in range(2 * n - 1):
        constraints.append(G[m, :] @ h_var + const[m] <= t_var)

    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK)
            except Exception:

            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n / 2.0
        ]
        constraints.append(G @ h_var + const <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                try:
                    prob.solve(solver=cp.GLPK)
                except Exception:
                    break
        if h_var.value is None:
            break
        h_lp = np.array(h_var.value).flatten()
        if not np.all(np.isfinite(h_lp)):
            break
        h_lp = project_feasible(h_lp, n)
        c_lp = compute_c5(h_lp, n)
        if c_lp < best_c - 1e-15:
            best_c = c_lp
            best_h = h_lp
        else:
            improved = False
            for frac in [0.3, 0.5, 0.7, 0.2, 0.8]:
                h_blend = project_feasible(frac * h_lp + (1.0 - frac) * best_h, n)
                c_blend = compute_c5(h_blend, n)
                if c_blend < best_c - 1e-15:
                    best_c = c_blend
                    best_h = h_blend
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    if radius is None:
        radius = min(40, n // 4)
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_outer = True
        while improved_outer and time.time() < deadline:
            improved_outer = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_outer = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_outer = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def gradient_pair_search(h, n, deadline, step=0.002, alphas=None):
    if alphas is None:
        alphas = [50, 200, 1000, 5000]
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        improved_outer = True
        while improved_outer and time.time() < deadline:
            improved_outer = False
            c = compute_corr(h, n)
            mx = c.max()
            s = alpha * (c - mx)
            w = np.exp(s)
            w /= w.sum()
            dx = 2.0 / n
            grad = np.zeros(n)
            nc = 2 * n - 1
            for m in range(nc):
                if w[m] < 1e-14:
                    continue
                k = m - (n - 1)
                wm_dx = w[m] * dx
                if k >= 0:
                    nv = n - k
                    grad[:nv] += wm_dx * (1.0 - h[k:n])
                    grad[k:n] += wm_dx * (-h[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += wm_dx * (1.0 - h[0:nv])
                    grad[0:nv] += wm_dx * (-h[kk:n])
            gdiff = grad[:, None] - grad[None, :]
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    move = step * (grad[j] - grad[i])
                    if abs(move) < 1e-10:
                        continue
                    hi_new = h[i] + move
                    hj_new = h[j] - move
                    if 0.0 <= hi_new <= 1.0 and 0.0 <= hj_new <= 1.0:
                        h[i] = hi_new
                        h[j] = hj_new
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_outer = True
                        else:
                            h[i] = hi_new - move
                            h[j] = hj_new + move
    return h, best_c

def gradient_descent_bt(h, n, deadline, lr_init=0.005, alpha=500.0):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = lr_init
    for _ in range(10000):
        if time.time() > deadline:
            break
        g = _grad_lse(best_h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        improved = False
        for _ in range(30):
            h_new = project_feasible(best_h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-12:
                break
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old)
    x_new = np.linspace(0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except NameError:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS at original resolution (quick warmup)
    t1_end = min(t0 + 100, deadline - 400)
    best_h, best_c = lbfgs_optimize(best_h, n_init, t1_end,
                                     alphas=[100, 500, 2000, 10000, 50000])

    # Phase 2: Upsample to n=600 and optimize
    n2 = 600
    h2 = upsample(best_h, n_init, n2)
    c2 = compute_c5(h2, n2)

    t2_end = min(t0 + 350, deadline - 250)
    h2, c2 = lbfgs_optimize(h2, n2, t2_end,
                            alphas=[50, 200, 1000, 5000, 20000, 100000])
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Active set optimization (focus on bottleneck lags)
    if time.time() < deadline - 150:
        t3_end = min(time.time() + 100, deadline - 150)
        h2, c2 = active_set_optimize(h2, n2, t3_end, n_active=15)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 4: LP outer approximation
    if time.time() < deadline - 120:
        t4_end = min(time.time() + 120, deadline - 120)
        h2, c2 = lp_outer_approx(h2, n2, t4_end, max_iters=30)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 5: Gradient pair search
    if time.time() < deadline - 80:
        t5_end = min(time.time() + 60, deadline - 80)
        h2, c2 = gradient_pair_search(h2, n2, t5_end, step=0.002)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 6: Fine pair search
    if time.time() < deadline - 40:
        t6_end = min(time.time() + 40, deadline - 40)
        h2, c2 = pair_search(h2, n2, t6_end,
                            deltas=[0.001, 0.0005, 0.0002, 0.0001],
                            radius=30)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 7: Try even higher resolution n=800
    if time.time() < deadline - 50:
        n3 = 800
        h3 = upsample(best_h, len(best_h), n3)
        t7_end = min(time.time() + 30, deadline - 15)
        h3, c3 = lbfgs_optimize(h3, n3, t7_end,
                                alphas=[500, 2000, 10000, 50000])
        if time.time() < deadline - 8:
            h3, c3 = pair_search(h3, n3, deadline - 5,
                                deltas=[0.001, 0.0005, 0.0002],
                                radius=25)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 8: Final L-BFGS polish + pair search
    nf = len(best_h)
    if time.time() < deadline - 5:
        h_f, c_f = lbfgs_optimize(best_h, nf, deadline - 3,
                                  alphas=[50000, 200000])
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
