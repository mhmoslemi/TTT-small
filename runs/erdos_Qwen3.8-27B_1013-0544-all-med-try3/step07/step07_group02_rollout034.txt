δc[k] = (contribution from p=i) + (contribution from p=j) + (contribution from q=i) + (contribution from q=j)

But we need to be careful about double-counting when p and q are both in {i,j}.

Let me handle it case by case:
- p=i, q≠i,j: δ -= (1-h[q])
- p=i, q=j: δ -= 1
- p=j, q≠i,j: δ += (1-h[q])
- p=j, q=i: δ += 1
- p=j, q=j: δ += 0 (no change)
- p≠i,j, q=i: δ += h[p] where p = i-k+(n-1)
- p≠i,j, q=j: δ -= h[p] where p = j-k+(n-1)

But we need p to be valid (0≤p<n) and p ∉ {i,j} for the last two cases.

So for a given k:
q_i = i + k - (n-1)  [second index when p=i]
q_j = j + k - (n-1)  [second index when p=j]
p_i = i - k + (n-1)  [first index when q=i]
p_j = j - k + (n-1)  [first index when q=j]

δc[k] = 0
if 0 ≤ q_i < n:
    if q_i == j: δc[k] -= 1
    else: δc[k] -= (1 - h[q_i])
if 0 ≤ q_j < n:
    if q_j == i: δc[k] += 1
    elif q_j == j: pass  # no change
    else: δc[k] += (1 - h[q_j])
if 0 ≤ p_i < n and p_i != i and p_i != j:
    δc[k] += h[p_i]
if 0 ≤ p_j < n and p_j != i and p_j != j:
    δc[k] -= h[p_j]

This is O(1) per k! But k ranges over [0, 2n-2], so it's O(n) per swap.

For n=500, that's 1000 operations per swap. If we try 10000 swaps, that's 10M operations, which is fast.

But actually, we only care about the maximum. So we can:
1. Compute c_all (O(n²) with correlate, or O(n log n) with FFT)
2. Find the max value and its index
3. For a candidate swap, compute δc[k] for all k, update c_all, and check if the new max is smaller

But updating c_all for all k is O(n) per swap. That's fine.

Actually, even better: we only need to compute δc[k] for k values where it's non-zero. From the formula above, δc[k] is non-zero only when:
- 0 ≤ q_i < n: k ∈ [n-1-i, n-1-i+n-1] = [n-1-i, 2(n-1)-i]
- 0 ≤ q_j < n: k ∈ [n-1-j, 2(n-1)-j]
- 0 ≤ p_i < n and p_i ≠ i,j: k ∈ [n-1-p_i for valid p_i] = [n-1-0, n-1-(n-1)] shifted... 

Actually p_i = i - k + (n-1), so k = n-1 + i - p_i. For p_i in [0, n-1], k ranges from n-1+i to i. So k ∈ [i, n-1+i].

Similarly p_j gives k ∈ [j, n-1+j].

So the affected k range is [min(n-1-i, n-1-j, i, j), max(2(n-1)-i, 2(n-1)-j, n-1+i, n-1+j)], which is essentially [min(i,j), max(n-1+i, n-1+j)] ⊆ [0, 2n-2].

In the worst case, this is the entire range [0, 2n-2], so O(n) per swap.

For efficiency, I'll precompute the correlation, then for each candidate swap, compute the delta only at the affected indices, update the correlation, and check the new max.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n//2]
    h_bin[idx] = 1.0
    return h_bin

def smooth_max_grad(h, n, alpha):
    """Compute gradient of smooth max of correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += (1.0 - 2.0*h) * w[k]
        elif lag > 0 and lag < n:
            grad[lag:n] += (1.0 - h[0:n-lag]) * w[k]
            grad[0:n-lag] += (-h[lag:n]) * w[k]
        elif lag < 0 and lag > -n:
            m2 = -lag
            grad[0:n-m2] += (1.0 - h[m2:n]) * w[k]
            grad[m2:n] += (-h[0:n-m2]) * w[k]
    return grad * dx

def gradient_descent(h, n, deadline, alpha=50, lr=0.5, momentum=0.9):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    
    while time.time() < deadline:
        g = smooth_max_grad(h, n, alpha)
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
        lr = max(0.01, lr * 0.999)
    
    return best_h, best_c

def binary_local_search(h, n, deadline, rng, max_rounds=200, n_tries_per_round=3000):
    h = h.astype(float).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n_tries_per_round):
            if time.time() > deadline:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            if i == j:
                continue
            
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
                improved = True
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        if not improved:
            # Try structured moves
            c_all = compute_all_lags(h, n)
            worst_k = c_all.argmax()
            lag = worst_k - (n-1)
            m = abs(lag)
            if m > 0 and m < n:
                # Focus swaps on the worst lag
                for _ in range(2000):
                    if time.time() > deadline:
                        break
                    if lag > 0:
                        i = int(rng.randint(0, n - m))
                        j = i + m
                    else:
                        j = int(rng.randint(0, n + lag))
                        i = j - lag
                    if h[i] != h[j]:
                        # Swap to reduce overlap at this lag
                        if lag > 0:
                            # We want h[i](1-h[j]) to decrease
                            # If h[i]=1, h[j]=0: make h[i]=0, h[j]=1 (but need to find another swap partner)
                            pass
                        # Just do random swap biased toward the worst lag region
                        i2 = int(rng.choice(ones))
                        j2 = int(rng.choice(zeros))
                        h[i2] = 0.0
                        h[j2] = 1.0
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                            ones = np.where(h > 0.5)[0]
                            zeros = np.where(h < 0.5)[0]
                        else:
                            h[i2] = 1.0
                            h[j2] = 0.0
            if not improved:
                break
    
    return best_h, best_c

def binary_simulated_annealing(h, n, deadline, rng, start_temp=0.005, n_steps=100000):
    h = h.astype(float).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_curr = best_c
    total_time = deadline - time.time()
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    step = 0
    while step < n_steps and time.time() < deadline - 2:
        step += 1
        temp = start_temp * 0.9995 ** step
        
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h, n)
        
        delta = c_new - c_curr
        if delta < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta / temp)):
            c_curr = c_new
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c

def continuous_polish(h, n, deadline):
    """Final continuous optimization to squeeze out a bit more."""
    from scipy.optimize import minimize
    
    def obj(x):
        alpha = 50000
        c = compute_all_lags(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    
    def obj_grad(x):
        return smooth_max_grad(x, n, 50000)
    
    def sum_con(x):
        return np.sum(x) - n / 2.0
    
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial solution

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin[idx] = 1.0
    return h_bin

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h) * w[k]
        elif 0 < lag < n:
            grad[lag:n] += (1.0 - h[0:n - lag]) * w[k]
            grad[0:n - lag] += (-h[lag:n]) * w[k]
        elif -n < lag < 0:
            m2 = -lag
            grad[0:n - m2] += (1.0 - h[m2:n]) * w[k]
            grad[m2:n] += (-h[0:n - m2]) * w[k]
    return grad * dx

def gradient_descent(h, n, deadline, alpha=80, lr=0.4, momentum=0.92):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr_cur = lr
    while time.time() < deadline:
        g = smooth_max_grad(h, n, alpha)
        v = momentum * v - lr_cur * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
    return best_h, best_c

    if n % p != 0:
        return None, float('inf')
    n_repeats = n // p
    ones_per = p // 2
    best_h = None
    best_c = float('inf')
    budget_per_start = max(5, (deadline - time.time()) / n_starts)
    for s in range(n_starts):
        start_time = time.time()
        h_p = np.zeros(p)
        idx = rng.choice(p, ones_per, replace=False)
        h_p[idx] = 1.0
        h = np.tile(h_p, n_repeats)
        c_curr = compute_c5(h, n)
        ones_p = np.where(h_p > 0.5)[0]
        zeros_p = np.where(h_p < 0.5)[0]
        for it in range(20000):
            if time.time() > start_time + budget_per_start:
                break
            i = int(rng.choice(ones_p))
            j = int(rng.choice(zeros_p))
            h_p[i] = 0.0
            h_p[j] = 1.0
            h = np.tile(h_p, n_repeats)
            c_new = compute_c5(h, n)
            if c_new < c_curr - 1e-15:
                c_curr = c_new
                ones_p = np.where(h_p > 0.5)[0]
                zeros_p = np.where(h_p < 0.5)[0]
            else:
                h_p[i] = 1.0
                h_p[j] = 0.0
        if c_curr < best_c:
            best_c = c_curr
            best_h = np.tile(h_p, n_repeats)
    return best_h, best_c

def continuous_polish(h, n, deadline):
    def obj(x):
        alpha = 30000
        c = compute_all_lags(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return m + np.log(np.sum(np.exp(shifted))) / alpha

    def obj_grad(x):
        return smooth_max_grad(x, n, 30000)

    def sum_con(x):
        return np.sum(x) - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def soft_binarize(h, n, n_soft=10):
    h_raw = h.copy()
    order = np.argsort(np.abs(h_raw - 0.5))
    soft_idx = order[:n_soft]
    h_bin = binarize(h_raw, n)
    for i in soft_idx:
        h_bin[i] = h_raw[i]
    return project_feasible(h_bin, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        h_work = project_feasible(h_init, n_init)
        best_c = compute_c5(h_work, n_init)
        best_h = h_work.copy()
    else:
        h_work = np.full(n_init, 0.5)
        h_work += rng.normal(0, 0.05, n_init)
        h_work = project_feasible(h_work, n_init)
        best_c = compute_c5(h_work, n_init)
        best_h = h_work.copy()

    n = len(best_h)

    # Phase 1: Gradient descent on continuous solution
    gd_dl = t0 + 40
    h_gd, c_gd = gradient_descent(best_h, n, gd_dl, alpha=100, lr=0.35, momentum=0.92)
    if c_gd < best_c:
        best_c = c_gd
        best_h = h_gd

    # Phase 2: Binarize and directed local search
    h_bin = binarize(best_h, n)
    c_bin = compute_c5(h_bin, n)
    dl2 = min(deadline - 180, time.time() + 150)
    h_dir, c_dir = directed_binary_search(h_bin, n, dl2, rng, max_rounds=400)
    if c_dir < best_c:
        best_c = c_dir
        best_h = h_dir

    # Phase 3: General local search on binary
    if time.time() < deadline - 120:
        dl3 = min(deadline - 100, time.time() + 120)
        h_ls, c_ls = binary_local_search(best_h, n, dl3, rng, max_rounds=800, n_tries=8000)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Phase 4: Two-swap search
    if time.time() < deadline - 60:
        dl4 = min(deadline - 40, time.time() + 40)
        h_2s, c_2s = two_swap_search(best_h, n, dl4, rng, max_rounds=150)
        if c_2s < best_c:
            best_c = c_2s
            best_h = h_2s

    # Phase 5: Periodic constructions
    if time.time() < deadline - 60:
        for p in [10, 20, 25, 50, 100]:
            if time.time() > deadline - 40:
                break
            if n % p == 0:
                dl_p = min(deadline - 30, time.time() + 20)
                h_per, c_per = periodic_search(n, p, dl_p, rng, n_starts=8)
                if h_per is not None and c_per < best_c:
                    best_c = c_per
                    best_h = h_per
                    # Polish the periodic solution
                    dl_pol = min(deadline - 20, time.time() + 30)
                    h_pol, c_pol = binary_local_search(h_per, n, dl_pol, rng, max_rounds=200, n_tries=5000)
                    if c_pol < best_c:
                        best_c = c_pol
                        best_h = h_pol

    # Phase 6: Try different resolutions
    n_candidates = [400, 450, 550, 600, 700]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 70:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        # Quick gradient
        gd_dl2 = time.time() + min(15, remaining * 0.2)
        h_gd2, c_gd2 = gradient_descent(h_interp, n_try, gd_dl2, alpha=80, lr=0.3)
        # Binarize
        h_bin2 = binarize(h_gd2, n_try)
        # Local search
        ls_dl2 = time.time() + min(50, remaining * 0.4)
        h_ls2, c_ls2 = directed_binary_search(h_bin2, n_try, ls_dl2, rng, max_rounds=300)
        if c_ls2 < best_c:
            best_c = c_ls2
            best_h = h_ls2
        # Also try the continuous version
        if c_gd2 < best_c:
            best_c = c_gd2
            best_h = h_gd2

    # Phase 7: Simulated annealing on best binary
    if time.time() < deadline - 30:
        h_bin_sa = binarize(best_h, len(best_h)) if not np.allclose(best_h, np.round(best_h)) else best_h.copy()
        sa_dl = min(deadline - 5, time.time() + 80)
        h_sa, c_sa = binary_sa(h_bin_sa, len(h_bin_sa), sa_dl, rng, start_temp=0.003, n_steps=300000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 8: Final directed local search
    if time.time() < deadline - 5:
        n_fin = len(best_h)
        h_fin_bin = binarize(best_h, n_fin) if not np.allclose(best_h, np.round(best_h)) else best_h.copy()
        final_dl = deadline - 1
        h_fin, c_fin = directed_binary_search(h_fin_bin, n_fin, final_dl, rng, max_rounds=500)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Phase 9: Try soft-binarize + continuous polish
    if time.time() < deadline - 15:
        n_sf = len(best_h)
        h_soft = soft_binarize(best_h, n_sf, n_soft=15)
        cp_dl = deadline - 2
        h_cp, c_cp = gradient_descent(h_soft, n_sf, cp_dl, alpha=200, lr=0.1, momentum=0.9)
        if c_cp < best_c:
            best_c = c_cp
            best_h = h_cp

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
