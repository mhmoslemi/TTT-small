def projected_subgradient(h, n, alpha, lr, momentum, max_iter, deadline):
    dx = 2.0/n
    h = h.copy()
    v = np.zeros(n)  # momentum
    t = 1.0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute objective and gradient
        corr = np.correlate(h, 1-h, mode='full') * dx
        sm = corr.max()
        exp_terms = np.exp(alpha * (corr - sm))
        weights = exp_terms / exp_terms.sum()
        obj = sm + np.log(exp_terms.sum()) / alpha
        
        # Gradient is weighted sum of individual gradients
        # grad_k[j] = dx * ((1-h[j+k]) - h[j-k]) for valid indices
        grad = np.zeros(n)
        for k in range(-(n-1), n):
            w = weights[n-1+k]
            if w < 1e-10:
                continue
            # Add w * grad_k
            for j in range(n):
                g = 0.0
                jk = j + k
                jmk = j - k
                if 0 <= jk < n:
                    g += 1 - h[jk]
                if 0 <= jmk < n:
                    g -= h[jmk]
                grad[j] += w * dx * g
        
        # Nesterov step
        t_prev = t
        t = (1 + np.sqrt(1 + 4*t*t)) / 2  # or t = (1+sqrt(1+4*t_prev^2))/2

        h_new = project_feasible(h_new, n)

        # x_new = argmin f(y) + (1/(2*lr))||x - y||^2
        # This is more complex. Let me just use momentum instead.
        
        h = h_new

            obj = sm + np.log(e.sum())/alpha

            c = compute_c5(h_new, n)

                best_h = h_new.copy()
            
            h = h_new
            
            # Adaptive learning rate
            if obj < target_c:
                break
        
        if best_c < target_c:
            break
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_smooth_obj_grad(h, n, alpha):
    """Compute smoothed objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    sm = corr.max()
    shifted = alpha * (corr - sm)
    # For numerical stability
    max_shifted = shifted.max()
    if max_shifted > 700:
        shifted = shifted - max_shifted
    exp_terms = np.exp(shifted)
    log_sum = np.log(exp_terms.sum())
    obj = sm + (max_shifted + log_sum) / alpha
    weights = exp_terms / exp_terms.sum()
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        w = weights[idx]
        if w < 1e-14:
            continue
        # Contribution: w * dx * ((1-h[j+k]) - h[j-k])
        # (1-h[j+k]) part: j+k in [0,n)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1+k:hi1+k])
        # (-h[j-k]) part: j-k in [0,n)
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2-k:hi2-k]
    grad *= dx
    
    return obj, grad

                 lr=0.05, momentum=0.9, max_iter=50000, perturb_every=1000,
                 perturb_scale=0.01, seed=42):

    rng = np.random.RandomState(seed)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_start
    
    # Log-scale alpha progression
    log_a_start = np.log(alpha_start)
    log_a_end = np.log(alpha_end)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Progress alpha
        frac = it / max_iter
        alpha = np.exp(log_a_start + frac * (log_a_end - log_a_start))
        
        obj, grad = compute_smooth_obj_grad(h, n, alpha)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * lr * grad
        h_new = h - v
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        
        h = h_new
        
        # Periodic perturbation

            h_pert = project_feasible(h + pert, n)
            c_pert = compute_c5(h_pert, n)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert.copy()
                h = h_pert.copy()
                v = np.zeros(n)
    
    return best_h, best_c
