import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(x0, n, alpha, top_idx=None, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, x0, args=(n, alpha, top_idx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return x0.copy()

def lag_gradient(h, n, lag):
    """Gradient of c[lag] with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
        return grad
    if lag > 0:
        grad[:n - m] -= dx * h[m:]
        grad[m:] += dx * (1.0 - h[:n - m])
    else:
        grad[:n - m] += dx * (1.0 - h[m:])
        grad[m:] -= dx * h[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + grad @ (h_var - h_curr) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def coordinate_descent(h, n, deadline, max_passes=200):
    """Fine coordinate descent: adjust pairs to reduce max correlation."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        for _ in range(max_passes):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            improved = False
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                # Try to reduce c[k_star] by adjusting h[i] and h[j]
                if lag > 0:
                    # c decreases when h[i] decreases or h[j] increases
                    # (since c ~ h[i]*(1-h[j]))
                    if h[i] > h[j]:
                        delta = min(step, (h[i] - h[j]) * 0.4, h[i] * 0.9, (1 - h[j]) * 0.9)
                    elif h[j] < h[i]:
                        delta = min(step, (h[i] - h[j]) * 0.4, h[i] * 0.9, (1 - h[j]) * 0.9)
                    else:
                        continue
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                else:
                    # lag < 0: c ~ h[j]*(1-h[i])
                    if h[j] > h[i]:
                        delta = min(step, (h[j] - h[i]) * 0.4, h[j] * 0.9, (1 - h[i]) * 0.9)
                    else:
                        continue
                    if delta < 1e-10:

                    h[i] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[j] += delta
                        h[i] -= delta
            if not improved:
                break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)


def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_con_jac(x, n):
    return np.ones(n)


def slsqp_run(x0, n, alpha, top_idx=None, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, x0, args=(n, alpha, top_idx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return x0.copy()


def lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif lag > 0:
        grad[:n - m] -= dx * h[m:]
        grad[m:] += dx * (1.0 - h[:n - m])
    else:
        grad[:n - m] += dx * (1.0 - h[m:])
        grad[m:] -= dx * h[:n - m]
    return grad


def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + grad @ (h_var - h_curr) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

    # Round: keep top n/2 values as 1, rest as 0

    # Fix sum

    if n_ones > n // 2:

        h_sorted = np.sort(h[idx_ones])[::-1]
        # Remove the smallest ones

        order = np.argsort(h[idx_ones])[:remove_count]
        h_bin[idx_ones[order]] = 0.0
    elif n_ones < n // 2:
        idx_zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[idx_zeros])[:n // 2 - n_ones]
        h_bin[idx_zeros[order]] = 1.0
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    return h_bin, c_bin

    return best_h, best_c, len(best_h)
