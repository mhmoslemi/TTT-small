import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def pairwise_optimize(h, n, i, j, max_iters=50):
    """Optimize h[i] and h[j] keeping their sum constant."""
    S = h[i] + h[j]
    t_lo = max(0.0, S - 1.0)
    t_hi = min(1.0, S)
    if t_hi - t_lo < 1e-12:
        return h.copy()
    
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    def obj_1d(t):
        h_try = h.copy()
        h_try[i] = t
        h_try[j] = S - t
        c = np.correlate(h_try, 1 - h_try, mode='full')
        return float(np.max(c) * dx)
    
    # Use bounded scalar minimization
    try:
        res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded',
                              options={'xatol': 1e-10})
        if res.success:
            h_new = h.copy()
            h_new[i] = res.x
            h_new[j] = S - res.x
            return h_new
    except Exception:
        pass
    return h.copy()

def pairwise_cd(h, n, deadline, max_rounds=20, n_pairs_per_round=500):
    """Cycling pairwise coordinate descent."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_lags = 2 * n - 1
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break
        improved = False
        # Random pairs
        pairs = []
        for _ in range(n_pairs_per_round):
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if i != j:
                pairs.append((i, j))
        
        for (i, j) in pairs:
            if time.time() > deadline - 2:
                break
            h_new = pairwise_optimize(best_h, n, i, j)
            if not np.array_equal(h_new, best_h):
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
            best_h = project_feasible(h_new, n)
        
        if not improved:
            # Try structured pairs (neighbors)
            improved2 = False
            for i in range(0, n - 1):
                if time.time() > deadline - 2:
                    break
                h_new = pairwise_optimize(best_h, n, i, i + 1)
                if not np.array_equal(h_new, best_h):
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved2 = True
                best_h = project_feasible(h_new, n)
            if not improved2:
                break
    
    return best_h, best_c

def subgradient_method(h, n, deadline, max_iters=5000):
    """Subgradient descent with projection."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_lags = 2 * n - 1
    eta0 = 0.1
    lam = 1.0
    
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        grad = lag_gradient(h, n, lag)
        
        # Project gradient to sum-zero subspace
        grad = grad - np.mean(grad)
        
        eta = eta0 / (np.sqrt(it + 1))
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-15):
    """SLSQP with log-sum-exp smooth max."""
    dx = 2.0 / n
    
    def smooth_obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(smooth_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_slsqp(h, n, top_m, alpha, maxiter=3000):
    """SLSQP optimizing only top-M lags."""
    dx = 2.0 / n
    
    def active_obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        active = c[top_m]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(active_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def interpolate_h(h_src, n_src, n_dst):
    """Interpolate h from n_src points to n_dst points."""
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    # Periodic-style interpolation (linear for simplicity)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def outer_approx_one_step(h_curr, n, trust):
    """One step of outer approximation method."""
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    n_lags = 2 * n - 1
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    
    for k in range(n_lags):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    solver_list = [cp.GLOP, cp.CBC, cp.CPLEX, cp.GLPK]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, **{'gap_rel': 1e-8, 'time_limit': 30})
            if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
                return project_feasible(h_var.value, n)
        except Exception:
            continue
    
    return h_curr.copy()

def outer_approx_loop(h, n, deadline, max_iters=30):
    """Multiple outer approximation steps with decreasing trust region."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        h_new = outer_approx_one_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.002, trust * 0.8)
    
    return best_h, best_c

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    
    # Phase 1: Optimize at original resolution
    n = n_init
    print(f"Phase 1: Optimizing at n={n}, initial C5={best_c:.6f}", flush=True)
    
    # 1a: SLSQP smooth max
    if time.time() < deadline - 200:
        h1 = slsqp_smooth(best_h, n, alpha=20000, maxiter=4000, ftol=1e-15)
        c1 = compute_c5(h1, n)
        if c1 < best_c:
            best_c = c1
            best_h = h1
        print(f"  After SLSQP smooth: C5={best_c:.6f}", flush=True)
    
    # 1b: Active set SLSQP
    if time.time() < deadline - 150:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-7:]
        h2 = active_set_slsqp(best_h, n, top_idx, alpha=50000, maxiter=4000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        print(f"  After active set: C5={best_c:.6f}", flush=True)
    
    # 1c: Outer approximation
    if time.time() < deadline - 120:
        oo_deadline = min(deadline - 60, time.time() + 60)
        h3, c3 = outer_approx_loop(best_h, n, oo_deadline, max_iters=20)
        if c3 < best_c:
            best_c = c3
            best_h = h3
        print(f"  After outer approx: C5={best_c:.6f}", flush=True)
    
    # 1d: Pairwise coordinate descent
    if time.time() < deadline - 40:
        h4, c4 = pairwise_cd(best_h, n, min(deadline - 10, time.time() + 40),
                            max_rounds=10, n_pairs_per_round=400)
        if c4 < best_c:
            best_c = c4
            best_h = h4
        print(f"  After pairwise CD: C5={best_c:.6f}", flush=True)
    
    # 1e: Greedy local search
    if time.time() < deadline - 10:
        h5, c5 = greedy_local_search(best_h, n, min(deadline - 2, time.time() + 25),
                                     base_delta=0.005)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        print(f"  After greedy LS: C5={best_c:.6f}", flush=True)
    
    # Phase 2: Try higher resolutions
    print(f"\nPhase 2: Exploring higher resolutions (best so far: {best_c:.6f})", flush=True)
    
    resolutions_to_try = [int(n * 1.5), int(n * 2), int(n * 2.5)]
    for n_new in resolutions_to_try:
        if time.time() > deadline - 80:
            break
        if n_new > 1200 or n_new < 50:
            continue
        
        h_interp = interpolate_h(best_h, n, n_new)
        c_interp = compute_c5(h_interp, n_new)
        print(f"  n={n_new}: interpolated C5={c_interp:.6f}", flush=True)
        
        # Optimize at new resolution
        alloc = min(60, (deadline - time.time() - 40) * 0.5)
        if alloc < 20:
            continue
        
        # SLSQP
        if time.time() < deadline - 50:
            h_new_res = slsqp_smooth(h_interp, n_new, alpha=30000, maxiter=3000)
            c_new_res = compute_c5(h_new_res, n_new)
            if c_new_res < c_interp:
                c_interp = c_new_res
                h_interp = h_new_res
        
        # Active set
        if time.time() < deadline - 30:
            c_arr = compute_all_lags(h_interp, n_new)
            top_idx = np.argsort(c_arr)[-5:]
            h_new_res = active_set_slsqp(h_interp, n_new, top_idx, alpha=80000, maxiter=3000)
            c_new_res = compute_c5(h_new_res, n_new)
            if c_new_res < c_interp:
                c_interp = c_new_res
                h_interp = h_new_res
        
        # Pairwise CD
        if time.time() < deadline - 15:
            h_new_res, c_new_res = pairwise_cd(h_interp, n_new, deadline - 5,
                                              max_rounds=5, n_pairs_per_round=300)
            if c_new_res < c_interp:
                c_interp = c_new_res
                h_interp = h_new_res
        
        # Greedy LS
        if time.time() < deadline - 5:
            h_new_res, c_new_res = greedy_local_search(h_interp, n_new, deadline - 2,
                                                       base_delta=0.003)
            if c_new_res < c_interp:
                c_interp = c_new_res
                h_interp = h_new_res
        
        print(f"  n={n_new}: optimized C5={c_interp:.6f}", flush=True)
        
        if c_interp < best_c:
            best_c = c_interp
            best_h = h_interp.copy()
            n = n_new
            print(f"  NEW BEST: C5={best_c:.6f} at n={n}", flush=True)
    
    # Phase 3: Random restarts at best resolution
    print(f"\nPhase 3: Random restarts (best: {best_c:.6f} at n={n})", flush=True)
    
    if time.time() < deadline - 30:
        n_restarts = max(1, int((deadline - time.time() - 20) / 35))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.008 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, len(best_h))
            hp = project_feasible(hp, n)
            
            # Quick optimization
            if time.time() < deadline - 20:
                hn = slsqp_smooth(hp, n, alpha=50000, maxiter=2000)
                cn = compute_c5(hn, n)
                if cn < best_c:
                    best_c = cn
                    best_h = hn
                    print(f"  Restart {r}: improved to {best_c:.6f}", flush=True)
    
    # Phase 4: Final polish
    print(f"\nPhase 4: Final polish (best: {best_c:.6f})", flush=True)
    
    if time.time() < deadline - 30:
        # Very high alpha SLSQP
        h_final = slsqp_smooth(best_h, n, alpha=200000, maxiter=5000, ftol=1e-16)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        
        # More pairwise CD
        if time.time() < deadline - 15:
            h_final, c_final = pairwise_cd(best_h, n, deadline - 5,
                                          max_rounds=15, n_pairs_per_round=500)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        
        # Final greedy search with very small steps
        if time.time() < deadline - 5:
            h_final, c_final = greedy_local_search(best_h, n, deadline - 1,
                                                   base_delta=0.002)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
    
    # Final projection
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    print(f"\nFinal result: C5={best_c:.6f} at n={len(best_h)}", flush=True)
    return best_h, best_c, len(best_h)
