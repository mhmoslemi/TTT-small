    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    m = c.max()
    sh = alpha * (c - m)

                     args=(n, alpha, mu, fidx, fh),
                     method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': iters, 'ftol': 1e-20,
                              'gtol': 1e-15, 'maxfun': iters * 20,
                              'maxls': 50})
        hf = fh.copy()
        hf[fidx] = project_exact(r.x, fh[fidx].sum())
        cn = compute_c5(hf, n)
        if cn < bc - 1e-16:
            return hf, cn
    except Exception:
        pass
    return bh, bc


def pair_search(h, n, deadline, deltas, radius):
    h = h.copy()
    bc = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, bc
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < bc - 1e-15:
                            bc, imp = c, True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < bc - 1e-15:
                            bc, imp = c, True
                        else:
                            h[i] += d
                            h[j] -= d
            if not imp:
                break
    return h, bc


def momentum_desc(h, n, deadline, alpha, lr0, beta):
    h = project_feasible(h, n)
    bc = compute_c5(h, n)
    bh = h.copy()
    v = np.zeros(n)
    lr = lr0
    for _ in range(200000):
        if time.time() > deadline:
            break
        g = grad_c5(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        v = beta * v - lr * g
        hn = project_feasible(h + v, n)
        cn = compute_c5(hn, n)
        if cn < bc - 1e-16:
            h, bc, bh = hn, cn, hn.copy()
            lr = min(lr * 1.05, lr0 * 3.0)
        else:
            h = h.copy()
            lr *= 0.5
            v *= 0.3
            if lr < 1e-16:
                break
    return bh, bc


def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h[:n_old])


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(300, 0.5)
        n0 = 300

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: L-BFGS at original resolution
    t1 = min(t0 + 60, deadline - 400)
    best_h, best_c = opt_lBFGS(best_h, best_n, t1,
                               [(200, 5.0, 200), (2000, 20.0, 300), (20000, 100.0, 400)])

    # SLSQP at original resolution
    if time.time() < deadline - 350:
        t2 = min(time.time() + 40, deadline - 350)
        best_h, best_c = opt_slsqp(best_h, best_n, t2, 200000, 2000)

    # Phase 2: Upsample to n=600
    n600 = 600
    h600 = upsample(best_h, best_n, n600)
    h600 = project_feasible(h600, n600)
    c600 = compute_c5(h600, n600)

    # Heavy L-BFGS at n=600
    t3 = min(t0 + 400, deadline - 300)
    h600, c600 = opt_lBFGS(h600, n600, t3,
                           [(200, 5.0, 300), (2000, 20.0, 500), (20000, 100.0, 600),
                            (200000, 500.0, 800), (2000000, 2000.0, 1000)])

    if c600 < best_c:
        best_c, best_h, best_n = c600, h600, n600

    # SLSQP at n=600
    if time.time() < deadline - 250:
        t4 = min(time.time() + 80, deadline - 250)
        h600, c600 = opt_slsqp(h600, n600, t4, 2000000, 3000)
        if c600 < best_c:
            best_c, best_h, best_n = c600, h600, n600

    # Active-set refinement at n=600
    if time.time() < deadline - 200:
        t5 = min(time.time() + 60, deadline - 200)
        for th in [0.02, 0.005]:
            if time.time() > t5:
                break
            ha, ca = opt_active(best_h, best_n, t5, th, 500000, 2000, 500)
            if ca < best_c:
                best_c, best_h = ca, ha

    # Momentum descent at n=600
    if time.time() < deadline - 180:
        t6 = min(time.time() + 40, deadline - 180)
        hm, cm = momentum_desc(best_h, best_n, t6, 50000, 0.005, 0.9)
        if cm < best_c:
            best_c, best_h = cm, hm

    # Pair search at n=600 (coarse to fine)
    if time.time() < deadline - 100:
        t7 = min(time.time() + 150, deadline - 100)
        best_h, best_c = pair_search(best_h, best_n, t7,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005], 80)

    # Phase 3: Try n=900
    n900 = 900
    if time.time() < deadline - 120:
        h900 = upsample(best_h, best_n, n900)
        h900 = project_feasible(h900, n900)
        t8 = min(time.time() + 100, deadline - 120)
        h900, c900 = opt_lBFGS(h900, n900, t8,
                               [(1000, 20.0, 300), (10000, 50.0, 400), (100000, 200.0, 500)])
        if time.time() < deadline - 60:
            h900, c900 = opt_slsqp(h900, n900, deadline - 60, 500000, 1500)
        if c900 < best_c:
            best_c, best_h, best_n = c900, h900, n900

    # Phase 4: Try n=1200 if time allows
    n1200 = 1200
    if time.time() < deadline - 80:
        h1200 = upsample(best_h, best_n, n1200)
        h1200 = project_feasible(h1200, n1200)
        t9 = min(time.time() + 60, deadline - 80)
        h1200, c1200 = opt_lBFGS(h1200, n1200, t9,
                                  [(5000, 50.0, 400), (50000, 200.0, 500)])
        if c1200 < best_c:
            best_c, best_h, best_n = c1200, h1200, n1200

    # Final pair search on best solution
    if time.time() < deadline - 5:
        tf = deadline - 3
        best_h, best_c = pair_search(best_h, best_n, tf,
                                     [0.001, 0.0005, 0.0002, 0.0001, 0.00005], 60)

    # Final L-BFGS polish
    if time.time() < deadline - 3:
        best_h, best_c = opt_lBFGS(best_h, best_n, deadline - 2,
                                   [(100000, 500.0, 200)])

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
