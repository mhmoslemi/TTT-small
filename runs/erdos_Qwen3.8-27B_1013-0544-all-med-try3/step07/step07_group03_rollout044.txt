            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.sum(np.exp(shifted)) / alpha)

def obj_smooth2(x, n, alpha, weights):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    return float(m + logsum / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def coord_descent(h, n, deadline, max_rounds=80):
    best_c = compute_c5(h, n)
    target_sum = n / 2.0
    steps = [0.005, -0.005, 0.002, -0.002, 0.001, -0.001,
             0.0005, -0.0005, 0.0002, -0.0002, 0.0001, -0.0001]
    idx = np.arange(n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        np.random.shuffle(idx)
        for i in idx:
            if time.time() > deadline:
                return h, best_c
            old_i = h[i]
            for d in steps:
                new_i = old_i + d
                if new_i < 0 or new_i > 1:
                    continue
                adj = -d
                if adj > 0:
                    candidates = np.where(h + adj <= 1.0 + 1e-12)[0]
                    if len(candidates) == 0:
                        continue
                    j = candidates[np.argmax(h[candidates])]
                else:
                    candidates = np.where(h + adj >= -1e-12)[0]
                    if len(candidates) == 0:
                        continue
                    j = candidates[np.argmin(h[candidates])]
                if j == i:
                    continue
                if adj > 0 and h[j] + adj > 1.0 + 1e-12:
                    continue
                if adj < 0 and h[j] + adj < -1e-12:
                    continue
                h[i] = new_i
                h[j] = h[j] + adj
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    old_i = new_i
                else:
                    h[i] = old_i
                    h[j] = h[j] - adj
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    alphas = [200, 500, 2000, 8000, 30000, 100000, 400000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        ht = slsqp_opt(h, n, a, maxiter=2500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            h = ht

    if n <= 260:
        if time.time() < deadline - 15:
            h, best_c = pair_search(h, n, deadline - 8,
                                    [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    else:
        if time.time() < deadline - 15:
            h, best_c = near_pair_search(h, n, deadline - 8,
                                         [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                         radius=70)

    if time.time() < deadline - 10:
        ht = slsqp_opt(h, n, 800000, maxiter=4000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    if time.time() < deadline - 8:
        h, best_c = coord_descent(h, n, deadline - 4, max_rounds=40)

    if n <= 260:
        if time.time() < deadline - 3:
            h, best_c = pair_search(h, n, deadline - 1,
                                    [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
    else:
        if time.time() < deadline - 3:
            h, best_c = near_pair_search(h, n, deadline - 1,
                                         [0.0005, 0.0002, 0.0001, 0.00005], radius=50)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize at parent resolution
    alloc1 = min(280, (deadline - t0) * 0.27)
    h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt.copy(), c_opt
    n_best = n_init

    # Phase 2: Try a couple other resolutions
    x_orig = np.linspace(0, 2, n_init)
    for n_try in [200, 300]:
        if time.time() > deadline - 160:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_orig, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.15)
        if alloc < 30:
            break
        h_opt2, c_opt2 = optimize_at_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt2 < best_c:
            best_c = c_opt2
            best_h = h_opt2.copy()
            n_best = n_try

    # Phase 3: Multi-restart with diverse perturbations on best
    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(4, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            if r % 4 == 0:
                hp = best_h + rng.normal(0, 0.015, n)
            elif r % 4 == 1:
                hp = best_h.copy()
                bs = max(5, n // 25)
                b = rng.randint(0, n - bs)
                hp[b:b+bs] += rng.normal(0, 0.025, bs)
            elif r % 4 == 2:
                freq = rng.choice([2, 3, 4, 5])
                phase = rng.uniform(0, 2 * np.pi)
                x = np.linspace(0, 2, n)
                hp = best_h + 0.02 * np.sin(freq * np.pi * x + phase)
            else:
                # Antisymmetric perturbation
                hp = best_h.copy()
                half = n // 2
                pick = rng.randint(0, half)
                hp[pick] += 0.04
                hp[pick + half] -= 0.04
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 30) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                n_best = n

    # Phase 4: Intensive fine pair search on best
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        if n <= 260:
            best_h, best_c = pair_search(best_h, n, deadline - 8,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 8,
                                              [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                              radius=80)

    # Phase 5: Final SLSQP polish + Ultra-fine search
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 3000000, maxiter=5000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 5:
            best_h, best_c = coord_descent(best_h, n, deadline - 3, max_rounds=60)
        if n <= 260 and time.time() < deadline - 2:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        elif time.time() < deadline - 2:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.0002, 0.0001, 0.00005, 0.00002], radius=50)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
