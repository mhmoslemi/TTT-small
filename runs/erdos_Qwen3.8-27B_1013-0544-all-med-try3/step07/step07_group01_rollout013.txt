    return h, best_c

def exact_pair_delta(lag_vals, lag_grads, i, j, h_i, h_j):
    slopes = lag_grads[:, i] - lag_grads[:, j]
    intercepts = lag_vals
    lo = max(-h_i, h_j - 1.0)
    hi = min(1.0 - h_i, h_j)
    if hi - lo < 1e-15:
        return 0.0
    g0 = np.max(intercepts)
    a, b = lo, hi
    ga = np.max(intercepts + a * slopes)
    gb = np.max(intercepts + b * slopes)
    gr = 0.6180339887498949
    for _ in range(60):
        if b - a < 1e-14:
            break
        m1 = a + (1.0 - gr) * (b - a)
        m2 = a + gr * (b - a)
        g1 = np.max(intercepts + m1 * slopes)
        g2 = np.max(intercepts + m2 * slopes)
        if g1 < g2:
            b = m2
        else:
            a = m1
    d = (a + b) / 2.0
    g_d = np.max(intercepts + d * slopes)
    if g_d >= g0 - 1e-14:
        return 0.0
    return d

def pair_descent_core(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    lag_vals = compute_all_lags(h, n)
    lag_grads = compute_lag_gradients(h, n)
    no_improve_sweeps = 0
    max_no_improve = 4

    while time.time() < deadline:
        improved = False

        # Strategy 1: Focus on top active lags
        top_k = min(8, 2 * n - 1)
        active_lags = np.argsort(lag_vals)[-top_k:]

        for ai in active_lags:
            if time.time() > deadline:
                break
            sg = lag_grads[ai]
            i_cands = np.argsort(sg)[:12]
            j_cands = np.argsort(sg)[-12:]

            for i in i_cands:
                if improved or time.time() > deadline:
                    break
                for j in j_cands:
                    if i == j:
                        continue
                    d = exact_pair_delta(lag_vals, lag_grads, i, j, h[i], h[j])
                    if abs(d) > 1e-14:
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            lag_vals = compute_all_lags(h, n)
                            lag_grads = compute_lag_gradients(h, n)
                            break
                        else:
                            h[i] -= d
                            h[j] += d

        if improved:
            no_improve_sweeps = 0
            continue

        # Strategy 2: Full round-robin pair search (every other pair for speed)
        sweep_improved = False
        for i in range(0, n, 2):
            if time.time() > deadline:
                break
            if not sweep_improved:
                for j in range(1, n, 2):
                    if i == j:
                        continue
                    d = exact_pair_delta(lag_vals, lag_grads, i, j, h[i], h[j])
                    if abs(d) > 1e-14:
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            sweep_improved = True
                            improved = True
                            lag_vals = compute_all_lags(h, n)
                            lag_grads = compute_lag_gradients(h, n)
                        else:
                            h[i] -= d
                            h[j] += d
            else:
                break

        if not sweep_improved:
            # Strategy 3: Try odd-odd pairs
            for i in range(1, n, 2):
                if time.time() > deadline:
                    break
                for j in range(0, n, 2):
                    if i == j:
                        continue
                    d = exact_pair_delta(lag_vals, lag_grads, i, j, h[i], h[j])
                    if abs(d) > 1e-14:
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            sweep_improved = True
                            improved = True
                            lag_vals = compute_all_lags(h, n)
                            lag_grads = compute_lag_gradients(h, n)
                        else:
                            h[i] -= d
                            h[j] += d
                if sweep_improved:
                    break

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= max_no_improve:
                break

    return h, best_c

def random_perturb(h, n, scale=0.05):
    h_new = h + scale * np.random.RandomState(np.random.randint(100000)).randn(n)
    return project_feasible(h_new, n)

def try_multiple_n(h_best, n_best, best_c, deadline):
    if time.time() > deadline - 30:
        return h_best, best_c, n_best

    n_candidates = [n_best - 50, n_best + 50, n_best + 100, n_best - 30, n_best + 30]
    n_candidates = [x for x in n_candidates if 100 <= x <= 800 and x != n_best]

    for n_try in n_candidates:
        if time.time() > deadline - 10:
            break
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_feasible(h_try, n_try)

        alloc = min(25, deadline - time.time() - 8)
        if alloc < 8:
            break
        h_t, c_t = pair_descent_core(h_try, n_try, time.time() + alloc)
        if c_t < best_c - 1e-10:
            best_c = c_t
            h_best = h_t
            n_best = n_try

    return h_best, best_c, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except NameError:
        h_init = np.full(300, 0.5)
        n_init = 300

    h = project_feasible(h_init, n_init)
    best_h = h.copy()
    n = n_init
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick L-BFGS-B to polish (up to 50s)
    t1 = min(t0 + 50, deadline - 100)
    if t1 > t0 + 5:
        h1, c1 = lbfgs_optimize(h, n, t1, max_iters=500)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
            h = h1

    # Phase 2: Main pair descent (majority of remaining time)
    t2 = deadline - 80
    h2, c2 = pair_descent_core(h, n, t2)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
        h = h2

    # Phase 3: Restart with perturbation if time allows
    if time.time() < deadline - 60:
        h_pert = random_perturb(best_h, n, scale=0.03)
        alloc = min(30, deadline - time.time() - 20)
        if alloc > 10:
            h_p, c_p = pair_descent_core(h_pert, n, time.time() + alloc)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p.copy()

    # Phase 4: Try different resolutions
    if time.time() < deadline - 20:
        best_h, best_c, n = try_multiple_n(best_h, n, best_c, deadline)

    # Phase 5: Final L-BFGS + pair descent polish
    if time.time() < deadline - 5:
        nf = len(best_h)
        t_final1 = min(time.time() + 15, deadline - 5)
        h_f, c_f = lbfgs_optimize(best_h, nf, t_final1, max_iters=200)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
        if time.time() < deadline - 2:
            h_f2, c_f2 = pair_descent_core(h_f, nf, deadline - 1)
            if c_f2 < best_c:
                best_c = c_f2
                best_h = h_f2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
