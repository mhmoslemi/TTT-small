and B[m] = (1 - h[i0+k_m]) - h[i0-k_m] (with appropriate bounds)

The objective is max_m (A[m] + h[i0] * B[m]) * dx.

- 1, if all B[m] ≤ 0 (function is decreasing)
- At a breakpoint, where A[m1] + h*B[m1] = A[m2] + h*B[m2] for some m1, m2

2. If the max is at the same m for both, no breakpoint is active, and the minimum is at one of the endpoints

    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def swap_delta(h, n, i, j):
    m_range = np.arange(-(n - 1), n)
    two_n_1 = 2 * n - 1
    delta = np.zeros(two_n_1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < two_n_1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < two_n_1:
        delta[k2] += 1.0
    return delta

def local_search_1swap(h, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = set(int(x) for x in np.where(h == 1)[0])
    zeros = set(int(x) for x in np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zero_list = sorted(zeros)
        for i in zero_list:
            if time.time() > deadline:
                break
            best_j = None
            best_val = cur_c
            for j in sorted(ones):
                d = swap_delta(h, n, i, j)
                nc = (corr + d).max() * dx
                if nc < best_val - 1e-15:
                    best_val = nc
                    best_j = j
            if best_j is not None:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                zeros.discard(i)
                ones.discard(best_j)
                ones.add(i)
                zeros.add(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def anneal_binary(h, n, deadline, seed=42, T_start=0.01, T_end=0.0001, max_iters=2000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    
    cooling = (T_end / T_start) ** (1.0 / max(max_iters, 1))
    T = T_start
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        
        d = swap_delta(h, n, i, j)
        new_corr = corr + d
        nc = float(new_corr.max()) * dx
        
        if nc <= cur_c or (T > 1e-14 and rng.random() < np.exp(-(nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cooling
    
    return best_h, best_c
