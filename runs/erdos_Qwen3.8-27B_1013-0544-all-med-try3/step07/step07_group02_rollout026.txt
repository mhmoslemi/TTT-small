def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def get_smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c_all = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c_all.max()
    shifted = alpha * (c_all - c_max)
    shifted = np.clip(shifted, -200, 200)
    w = np.exp(shifted)
    w = w / w.sum()
    log_sum = np.log(np.sum(np.exp(shifted)))
    smooth_val = c_max + log_sum / alpha
    grad = np.zeros(n)
    active = w > 1.0 / (2 * len(w))
    for k in np.where(active)[0]:
        lag = int(k) - (n - 1)
        grad += w[k] * _lag_gradient(h, n, lag)
    grad *= dx
    return smooth_val, grad, w, c_all, c_max

def subgradient_method(h, n, deadline, alpha_start=200, binarize_eps=0.0003, seed=42):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    alpha = alpha_start
    no_improve = 0
    step_mult = 1.0
    rng = np.random.RandomState(seed)
    
    while time.time() < deadline:
        smooth_val, grad, w, c_all, c_max = get_smooth_max_and_grad(h_curr, n, alpha)
        
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve % 150 == 0 and no_improve < 800:
            alpha = min(alpha * 1.3, 30000)
        
        if no_improve >= 800:
            alpha = alpha_start
            step_mult *= 0.8
            no_improve = 0
            noise = 0.005 * (rng.rand(n) - 0.5)
            h_curr = project_feasible(h_curr + noise, n)
            continue
        
        if binarize_eps > 0 and no_improve > 80:
            grad += binarize_eps * (1.0 - 2.0 * h_curr)
        
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-24:
            break
        
        gap = smooth_val - best_c + 0.0002
        if gap < 1e-7:
            gap = 1e-7
        step = gap / (g_norm_sq + 1e-20)
        step = min(step, 6.0 / n) * step_mult
        step = max(step, 1e-10)
        
        h_new = project_feasible(h_curr - step * grad, n)
        h_curr = h_new
    
    return best_h, best_c

def _slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -200, 200)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _slsqp_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -200, 200)
    w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    active = w > 1.0 / (2 * len(w))
    for k in np.where(active)[0]:
        lag = int(k) - (n - 1)
        grad += w[k] * _lag_gradient(x, n, lag)
    return grad * dx

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha=5000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]

                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
