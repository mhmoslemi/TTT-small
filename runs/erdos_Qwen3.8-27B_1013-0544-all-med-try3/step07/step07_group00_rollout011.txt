def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14 * max(1, target):
            break

            room = 1.0 - h
            tot = room.sum()

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def smooth_obj_softplus(x, n, alpha, mu):
    """Smooth max using log-sum-exp."""
    c = compute_c5_vec(x, n)
    m = c.max()

        val = m + np.log(np.sum(np.exp(sh))) / alpha

    return float(val)

            if tot < 1e-13:
                break

    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()

    else:

    return float(val)

    sm = s.max()
    w = np.exp(s - sm)

                    if i == j:
                        continue
                    if h[i] >= d and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            changed = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

    deadline = t0 + budget_s - 8

        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)

        h_parent = np.full(500, 0.5)
        n_parent = 500

    best_h = project_feasible(h_parent, n_parent)
    best_c = compute_c5(best_h, n_parent)

    # Phase 1: Optimize at parent resolution
    if time.time() < deadline - 500:
        alphas = [100, 500, 2000, 10000]
        for a in alphas:
            if time.time() > deadline - 500:
                break
            best_h, best_c = lbfgs_opt(best_h, n_parent,
                time.time() + 40, a, 2000)

    # Phase 2: Upsample and optimize at 2x resolution
    n2 = min(2 * n_parent, 2000)
    if n2 > n_parent and time.time() < deadline - 350:
        x_p = np.linspace(0, 2, n_parent)
        x_2 = np.linspace(0, 2, n2)
        h_2 = np.interp(x_2, x_p, best_h)
        h_2 = project_feasible(h_2, n2)

                break

        if time.time() < deadline - 150:
