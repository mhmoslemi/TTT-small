import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += -h[:n-m]
    else:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] += -h[m:]
    return grad

def balanced_quadratic_obj(h, n, lags):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.mean(c[lags]))

def balanced_quadratic_jac(h, n, lags):
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag_idx in lags:
        lag = lag_idx - (n - 1)
        grad += lag_gradient(h, n, lag)
    grad *= dx / len(lags)
    return grad

def smooth_max_obj(h, n, alpha, lags):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    active = c[lags]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return float(val)

def smooth_max_jac(h, n, alpha, lags):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    active = c[lags]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    exp_shifted = np.exp(shifted - mx)
    weights = exp_shifted / exp_shifted.sum()
    grad = np.zeros(n)
    for i, lag_idx in enumerate(lags):
        lag = lag_idx - (n - 1)
        grad += weights[i] * lag_gradient(h, n, lag)
    grad *= dx
    return grad

def sum_constraint(h, n):
    return float(np.sum(h) - n / 2.0)

def sum_constraint_jac(h, n):
    return np.ones(n)

def slsqp_balanced(h, n, lags, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(balanced_quadratic_obj, h, args=(n, lags), jac=balanced_quadratic_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-15})
        if res.success or res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def slsqp_smooth(h, n, alpha, lags, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha, lags), jac=smooth_max_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-15})
        if res.success or res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def subgradient_method(h, n, deadline, n_iter=5000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    avg_h = h_curr.copy()
    avg_c = best_c
    step_size_init = 0.1
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = step_size_init / (1.0 + 0.05 * it)
        d = -step * grad / gnorm
        h_new = h_curr + d
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            step_size_init = min(0.2, step_size_init * 1.1)
        else:
            step_size_init = max(0.001, step_size_init * 0.95)
        avg_h = 0.98 * avg_h + 0.02 * h_new
        c_avg = compute_c5(avg_h, n)
        if c_avg < avg_c:
            avg_c = c_avg
    if compute_c5(avg_h, n) < best_c:
        best_h = avg_h
        best_c = avg_c
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_rounds=10000):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_all_lags(h, n)
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    m = abs(lag)
    if m == 0 or m >= n:
        return h, best_c
    if lag > 0:
        pairs = [(i, i+m) for i in range(n-m)]
    else:
        pairs = [(i+m, i) for i in range(n-m)]
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    sa = step_sizes
    for d_idx, d in enumerate(sa):
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline - 1:
                break
            improved = False
            for idx, (i, j) in enumerate(pairs):
                if time.time() > deadline - 1:
                    break
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.8, (1.0 - h[j]) * 0.8)
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.8, (1.0 - h[i]) * 0.8)
                    if delta < 1e-10:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def multi_start_optimize(h_best, n, deadline, rng, n_starts=3):
    best_h = h_best.copy()
    best_c = compute_c5(best_h, n)
    for r in range(n_starts):
