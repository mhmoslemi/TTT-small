                idx1 = n - 1 + m_range[mask1]
                grads[p] += np.dot(w[idx1], 1 - x[pk[mask1]]) * dx
                pk2 = p - m_range
                mask2 = (pk2 >= 0) & (pk2 < n)
                idx2 = n - 1 + m_range[mask2]
                grads[p] -= np.dot(w[idx2], x[pk2[mask2]]) * dx
            return grads
        
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=[(0, 1)] * n,
                          options={'maxiter': 100, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def compute_swap_delta(h, i, j, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def simulated_annealing(h_init, n, t_deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    
    total_time = time.time() - (t_deadline - 400)
    available = max(t_deadline - time.time(), 1.0)
    T_start = 0.4 * dx
    T_final = 0.003 * dx
    
    delta_buf = np.zeros(2 * n - 1)
    one_mask = (h == 1.0).astype(np.int32)
    zero_mask = (h == 0.0).astype(np.int32)
    one_positions = np.where(one_mask > 0)[0]
    zero_positions = np.where(zero_mask > 0)[0]
    
    step = 0
    no_improve = 0
    last_improve_time = time.time()
    
    while time.time() < t_deadline:
        progress = min(1.0, (time.time() - (t_deadline - available)) / available)
        T = T_start * (T_final / T_start) ** progress
        
        i = zero_positions[rng.randint(len(zero_positions))]
        j = one_positions[rng.randint(len(one_positions))]
        
        delta_buf[:] = compute_swap_delta(h, i, j, n, m_range)
        new_corr = corr + delta_buf
        new_c = new_corr.max() * dx
        delta_c = new_c - cur_c
        
        if delta_c <= 0 or rng.rand() < np.exp(-delta_c / T):
            h[i] = 1.0
            h[j] = 0.0
            corr += delta_buf
            cur_c = new_c
            one_positions = np.where(h == 1.0)[0]
            zero_positions = np.where(h == 0.0)[0]
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                last_improve_time = time.time()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        step += 1
        if step % 50000 == 0 and time.time() - last_improve_time > 60:
            break
    
    return best_h, best_c

def hill_climb(h_init, n, t_deadline, seed=42, max_sweeps=100000):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    delta_buf = np.zeros(2 * n - 1)
    
    sweep = 0
    while time.time() < t_deadline and sweep < max_sweeps:
        improved = False
        zero_positions = np.where(h == 0.0)[0]
        rng.shuffle(zero_positions)
        
        for i in zero_positions:
            if time.time() > t_deadline:
                break
            one_positions = np.where(h == 1.0)[0]
            rng.shuffle(one_positions)
            
            for j in one_positions:
                delta_buf[:] = compute_swap_delta(h, i, j, n, m_range)
                new_corr = corr + delta_buf
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += delta_buf
                    cur_c = nc
                    improved = True
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    break
            if improved:
                break
        sweep += 1
        if not improved:
            break
    
    return best_h, best_c

def multi_start_sa(h_init, n, t_deadline, seed=42, n_restarts=5):
    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    
    time_per_restart = (t_deadline - time.time()) / n_restarts
    
    for r in range(n_restarts):
        t_r = time.time() + time_per_restart - 2
        if time.time() > t_r:
            break
        
        if r == 0:
            h_curr = best_h.copy()
        else:
            h_curr = best_h.copy()
            n_flips = rng.randint(3, 10)
            one_pos = np.where(h_curr == 1.0)[0]
            zero_pos = np.where(h_curr == 0.0)[0]
            for _ in range(n_flips):
                if len(one_pos) == 0 or len(zero_pos) == 0:
                    break
                oi = one_pos[rng.randint(len(one_pos))]
                zi = zero_pos[rng.randint(len(zero_pos))]
                h_curr[oi] = 0.0
                h_curr[zi] = 1.0
                one_pos = np.where(h_curr == 1.0)[0]
                zero_pos = np.where(h_curr == 0.0)[0]
        
        h_sa, c_sa = simulated_annealing(h_curr, n, t_r, seed=seed + r * 100)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
        
        t_hc = min(t_deadline, time.time() + time_per_restart * 0.3)
        if time.time() < t_hc:
            h_hc, c_hc = hill_climb(h_sa, n, t_hc, seed=seed + r * 200)
            if c_hc < best_c:
                best_c = c_hc
                best_h = h_hc.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    
    # Phase 1: Continuous optimization at original resolution
    p1_end = t0 + 0.06 * (budget_s - 5)
    if time.time() < p1_end:
        h_cont, c_cont = continuous_optimize(best_h, n0, p1_end, alphas=[500, 2000, 10000])
        if c_cont < best_c:
            best_c = c_cont
            best_h = h_cont.copy()
    
    # Phase 2: Go to n=500
    n_work = 500
    if n_work != n0:
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n_work)
        h_work = np.interp(x_new, x_old, best_h)
        h_work = project_feasible(h_work, n_work)
    else:
        h_work = best_h.copy()
    
    h_work = project_feasible(h_work, n_work)
    h_bin = to_binary(h_work, n_work)
    c_bin = compute_c5(h_bin, n_work)
    
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
        best_n = n_work
    
    # Phase 3: Continuous optimization at n=500
    p3_end = t0 + 0.15 * (budget_s - 5)
    if time.time() < p3_end:
        h_cont2, c_cont2 = continuous_optimize(h_work, n_work, p3_end, alphas=[500, 2000, 10000])
        if c_cont2 < best_c:
            best_c = c_cont2
            best_h = h_cont2.copy()
            best_n = n_work
        h_bin2 = to_binary(h_cont2, n_work)
        c_bin2 = compute_c5(h_bin2, n_work)
        if c_bin2 < best_c:
            best_c = c_bin2
            best_h = h_bin2.copy()
            best_n = n_work
    
    # Phase 4: Main SA + hill climbing at n=500
    p4_end = t0 + 0.55 * (budget_s - 5)
    h_work = best_h.copy()
    if len(h_work) != n_work:
        x_old = np.linspace(0, 2, len(h_work))
        x_new = np.linspace(0, 2, n_work)
        h_work = np.interp(x_new, x_old, h_work)
        h_work = project_feasible(h_work, n_work)
    h_bin = to_binary(h_work, n_work)
    
    h_best, c_best = multi_start_sa(h_bin, n_work, p4_end, seed=seed, n_restarts=4)
    if c_best < best_c:
        best_c = c_best
        best_h = h_best.copy()
        best_n = n_work
    
    # Phase 5: Longer hill climbing
    p5_end = t0 + 0.72 * (budget_s - 5)
    if time.time() < p5_end:
        h_work5 = to_binary(best_h, n_work) if len(best_h) == n_work else to_binary(
            np.interp(np.linspace(0, 2, n_work), np.linspace(0, 2, len(best_h)), best_h), n_work)
        h_hc, c_hc = hill_climb(h_work5, n_work, p5_end, seed=seed + 1000, max_sweeps=200000)
        if c_hc < best_c:
            best_c = c_hc
            best_h = h_hc.copy()
            best_n = n_work
    
    # Phase 6: Try other resolutions
    p6_end = t0 + 0.88 * (budget_s - 5)
    for nh in [400, 600]:
        if time.time() > p6_end - 30:
            break
        if nh == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nh)
        h_other = np.interp(x_new, x_old, best_h)
        h_other = project_feasible(h_other, nh)
        h_bin_other = to_binary(h_other, nh)
        
        t_sub = min(p6_end, time.time() + (p6_end - time.time()) / 2)
        h_o, c_o = multi_start_sa(h_bin_other, nh, t_sub, seed=seed + nh, n_restarts=2)
        if c_o < best_c:
            # Polish with hill climbing
            t_hc2 = min(p6_end, time.time() + 20)
            h_o2, c_o2 = hill_climb(h_o, nh, t_hc2, seed=seed + nh + 50)
            if c_o2 < best_c:
                best_c = c_o2
                best_h = h_o2.copy()
                best_n = nh
            elif c_o < best_c:
                best_c = c_o
                best_h = h_o.copy()
                best_n = nh
    
    # Phase 7: Final polish
    p7_end = deadline
    if time.time() < p7_end - 10:
        h_final = best_h.copy()
        if len(h_final) != best_n:
            h_final = np.interp(np.linspace(0, 2, best_n), np.linspace(0, 2, len(h_final)), h_final)
            h_final = project_feasible(h_final, best_n)
        h_final_bin = to_binary(h_final, best_n)
        
        # SA with more time
        t_final = p7_end - 5
        h_f, c_f = simulated_annealing(h_final_bin, best_n, t_final, seed=seed + 9999)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()
        
        # Quick hill climb
        if time.time() < p7_end - 2:
            h_hc, c_hc = hill_climb(best_h, best_n, p7_end - 1, seed=seed + 8888, max_sweeps=50000)
            if c_hc < best_c:
                best_c = c_hc
                best_h = h_hc.copy()
    
    # Final: ensure valid and recompute
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
