import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-15
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, step_size, top_k=5):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    # Find near-maximum shifts
    threshold = c_max - 0.01
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # Cap number of active constraints
    if len(active) > top_k:
        idx = np.argsort(c[active])[::-1][:top_k]
        active = active[idx]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)  # actual shift
        for i in range(n):
            ik = i + k
            ikk = i - k
            g = 0.0
            if 0 <= ik < n:
                g += (1.0 - h[ik]) * dx
            if 0 <= ikk < n:
                g -= h[ikk] * dx
            grad[i] += g
    return grad / len(active), c_max

def subgradient_descent(h, n, deadline, eta0=0.1, decay=0.999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = eta0
    it = 0
    while time.time() < deadline:
        grad, c_cur = subgradient_step(h, n, eta)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        eta *= decay
        it += 1
        if it % 100 == 0 and it > 1000:
            eta *= 0.95  # extra decay
    return best_h, best_c

    c = np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-12
            if mask.sum() == 0:
                break
            tot = h[mask].sum()
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h_init, n, deadline, rng, T0=0.05, T_min=0.001):
    """Local search on binary vectors with exactly n/2 ones."""
    h = (h_init > 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_ones = len(ones)
    n_half = n // 2
    if n_ones > n_half:
        # Remove excess ones
        excess = n_ones - n_half
        to_remove = rng.choice(ones, excess, replace=False)
        h[to_remove] = 0
    elif n_ones < n_half:
        # Add missing ones
        missing = n_half - n_ones
        to_add = rng.choice(zeros, missing, replace=False)
        h[to_add] = 1
    
    best_h = h.copy()
    c = get_correlation(h, n)
    best_c = c.max()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    n_iter = 0
    T = T0
    total_iters = 50000
    for it in range(total_iters):
        if time.time() > deadline:
            break
        T = T0 * (T_min / T0) ** (it / total_iters)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap
        h[i], h[j] = 0.0, 1.0
        c_new = get_correlation(h, n)
        c_new_max = c_new.max()
        
        if c_new_max < best_c:
            best_c = c_new_max
            best_h = h.copy()
            c = c_new
        elif T > 0 and rng.rand() < np.exp((best_c - c_new_max) / T):
            c = c_new
        else:
            h[i], h[j] = 1.0, 0.0
            continue
        
        # Update tracking
        ones[rng.searchsorted(ones, i)] = j  # or just recompute occasionally
        # Actually, just recompute every 1000 iterations
        if it % 1000 == 0:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        
        n_iter += 1
    
    return best_h, best_c

Now for the binary search, I want to be very efficient. Let me precompute what I can:

For a swap (i→0, j→1) where h[i]=1 and h[j]=0:
The new correlation values can be computed as:
c_new[k_idx] = c[k_idx] + delta[k_idx]

where delta depends on the change in h at positions i and j.

Specifically, the correlation at "shift" k (where k = k_idx - (n-1)) is:
c[k_idx] = sum_{valid i} h[i] * (1-h[i+k]) * dx

After swap (h[i]: 1→0, h[j]: 0→1):
- The term h[i]*(1-h[i+k]) changes from (1-h[i+k]) to 0: delta = -(1-h[i+k])*dx
- The term h[j]*(1-h[j+k]) changes from 0 to (1-h[j+k]): delta = +(1-h[j+k])*dx
- For any m where m+k = i, the term h[m]*(1-h[i]) changes from 0 to h[m]: delta = +h[m]*dx
- For any m where m+k = j, the term h[m]*(1-h[j]) changes from h[m] to 0: delta = -h[m]*dx

So for a given k:
delta[k] = dx * (-(1-h[i+k]) + (1-h[j+k]) + h[i-k] - h[j-k])
where out-of-range terms are 0.

This is O(1) per k, so O(n) total for all k. Then c_new = c + delta, and max(c_new) is O(n).

Total per swap: O(n) = O(250). Very fast!

For 15625 swaps per full sweep: 15625 * 250 = 3.9M operations. At numpy speed (~100M ops/s for simple array ops), this is about 40ms per sweep. In 300 seconds, I can do ~7500 full sweeps. More than enough.

def eval_swap_delta(i, j, h, n):
    """Compute change in correlation vector for swapping h[i]:1→0, h[j]:0→1."""
    dx = 2.0 / n
    delta = np.zeros(2 * n - 1)
    # i+k in range: k from -i to n-1-i
    # j+k in range: k from -j to n-1-j
    # i-k in range: k from i-(n-1) to i
    # j-k in range: k from j-(n-1) to j
    
    k_offset = n - 1  # index offset in full correlation
    
    # Term: -(1-h[i+k])
    k_start = max(0, -i)
    k_end = min(n - 1, n - 1 - i)
    for k in range(k_start, k_end + 1):

    # Term: +(1-h[j+k])
