    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = c - m  # shift so max is 0
    # log-sum-exp for numerical stability
    sh_max = sh.max()
    if sh_max > 500:
        return float(m + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    return float(x.sum() - n / 2.0)

                        h[j] += ds
                        c = compute_c5(h, n)

def simulated_annealing(h, n, deadline, n_steps=100000, T_start=0.01, T_end=0.0001):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    current_c = best_c
    T = T_start
    n_reps = max(1, n_steps)
    
    for step in range(n_reps):
        if time.time() > deadline:
            break
        # Anneal temperature
        frac = step / n_reps
        T = T_start * (T_end / T_start) ** frac
        
        # Random pair
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        
        # Random direction and step
        d = np.random.choice([-1, 1]) * np.random.choice([0.01, 0.005, 0.002, 0.001, 0.0005])
        if h[i] + d < -1e-10 or h[i] + d > 1 + 1e-10:
            continue
        if h[j] - d < -1e-10 or h[j] - d > 1 + 1e-10:
            continue
        
        h[i] += d
        h[j] -= d
        new_c = compute_c5(h, n)
        
        if new_c < current_c:
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # Accept with probability exp(-delta/T)
            if np.random.random() < np.exp((current_c - new_c) / T):
                current_c = new_c
            else:
                h[i] -= d
                h[j] += d
    
    return best_h, best_c

    m = c.max()
    sh = c - m
