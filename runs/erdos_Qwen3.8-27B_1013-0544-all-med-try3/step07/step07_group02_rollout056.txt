                        if delta < 1e-8:
                            continue
                        h[i] += delta
                        h[i + m] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_swap_c:
                            best_swap_c = c_new
                            best_swap = (i, i + m, -delta, lag)
                        h[i] -= delta
                        h[i + m] += delta
                else:
                    m2 = -lag
                    if h[i + m2] > h[i] + 1e-8:
                        delta = h[i + m2] - h[i]
                        if delta < 1e-8:
                            continue
                        h[i + m2] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_swap_c:
                            best_swap_c = c_new
                            best_swap = (i + m2, i, -delta, lag)
                        h[i + m2] += delta
                        h[i] -= delta
                    elif h[i] > h[i + m2] + 1e-8:
                        delta = h[i] - h[i + m2]
                        if delta < 1e-8:
                            continue
                        h[i] -= delta
                        h[i + m2] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_swap_c:
                            best_swap_c = c_new
                            best_swap = (i, i + m2, delta, lag)
                        h[i] += delta
                        h[i + m2] -= delta
        if best_swap is not None:
            i, j, delta, _ = best_swap
            h[i] -= delta
            h[j] += delta
            h = project_feasible(h, n)
            best_c = compute_c5(h, n)
            c_all = compute_all_lags(h, n)
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 3:
                break
    return h, best_c

def fine_local_search(h, n, deadline, max_rounds=50000):
    h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    small_delta = [0.02, 0.01, 0.005, 0.002, 0.001]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        for d in small_delta:
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if lag > 0:
                    if h[i] > h[j] + 1e-8:
                        delta = min(d, h[i] - h[j])
                        if delta < 1e-10:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-13:
                            best_c = c_new
                            c_all = compute_all_lags(h, n)
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-8:
                        delta = min(d, h[j] - h[i])
                        if delta < 1e-10:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-13:
                            best_c = c_new
                            c_all = compute_all_lags(h, n)
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                else:
                    m2 = -lag
                    j2 = i + m2
                    if h[j2] > h[i] + 1e-8:
                        delta = min(d, h[j2] - h[i])
                        if delta < 1e-10:
                            continue
                        h[j2] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-13:
                            best_c = c_new
                            c_all = compute_all_lags(h, n)
                            improved = True
                        else:
                            h[j2] += delta
                            h[i] -= delta
                    elif h[i] > h[j2] + 1e-8:
                        delta = min(d, h[i] - h[j2])
                        if delta < 1e-10:
                            continue
                        h[i] -= delta
                        h[j2] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-13:
                            best_c = c_new
                            c_all = compute_all_lags(h, n)
                            improved = True
                        else:
                            h[i] += delta
                            h[j2] -= delta
        if not improved:
            # Try other top lags
            top_lags = np.argsort(c_all)[-8:]
            for k2 in top_lags:
                lag2 = k2 - (n - 1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for d in [0.01, 0.005]:
                    for i in range(0, n - m2, max(1, (n - m2) // 40)):
                        if time.time() > deadline:
                            break
                        if lag2 > 0:
                            j = i + m2
                            if h[i] > h[j] + 1e-8:
                                delta = min(d, h[i] - h[j])
                                if delta < 1e-10:
                                    continue
                                h[i] -= delta
                                h[j] += delta
                                c_new = compute_c5(h, n)
                                if c_new < best_c - 1e-13:
                                    best_c = c_new
                                    c_all = compute_all_lags(h, n)
                                    improved = True
                                else:
                                    h[i] += delta
                                    h[j] -= delta
                            elif h[j] > h[i] + 1e-8:
                                delta = min(d, h[j] - h[i])
                                if delta < 1e-10:
                                    continue
                                h[i] += delta
                                h[j] -= delta
                                c_new = compute_c5(h, n)
                                if c_new < best_c - 1e-13:
                                    best_c = c_new
                                    c_all = compute_all_lags(h, n)
                                    improved = True
                                else:
                                    h[i] -= delta
                                    h[j] += delta
                        else:
                            j = i + m2
                            if h[j] > h[i] + 1e-8:
                                delta = min(d, h[j] - h[i])
                                if delta < 1e-10:
                                    continue
                                h[j] -= delta
                                h[i] += delta
                                c_new = compute_c5(h, n)
                                if c_new < best_c - 1e-13:
                                    best_c = c_new
                                    c_all = compute_all_lags(h, n)
                                    improved = True
                                else:
                                    h[j] += delta
                                    h[i] -= delta
                            elif h[i] > h[j] + 1e-8:
                                delta = min(d, h[i] - h[j])
                                if delta < 1e-10:
                                    continue
                                h[i] -= delta
                                h[j] += delta
                                c_new = compute_c5(h, n)
                                if c_new < best_c - 1e-13:
                                    best_c = c_new
                                    c_all = compute_all_lags(h, n)
                                    improved = True
                                else:
                                    h[i] += delta
                                    h[j] -= delta
            if not improved:
                break
    return h, best_c

def slsqp_polish(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        return smooth_max(c, alpha)
    def obj_grad(x):
        return grad_objective(x, n, alpha)
    def con_fun(x):
        return float(np.sum(x) - n / 2.0)
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       jac=obj_grad, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def interpolate_to(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_interp = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_interp, n_dst)

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Gradient phase
    g_time = min(100, (deadline - time.time()) * 0.4)
    if g_time > 20:
        g_dl = time.time() + g_time
        h_opt, c_opt = gradient_descent(h, n, g_dl,
                                         start_alpha=80, end_alpha=60000, max_iter=40000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        h = h_opt
    # SLSQP polish
    if time.time() < deadline - 80:
        for a in [50000, 150000]:
            if time.time() > deadline - 60:
                break
            h_new = slsqp_polish(h, n, a, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            h = h_new
    # Best-improvement local search
    if time.time() < deadline - 40:
        ls_dl = min(deadline - 30, time.time() + min(120, (deadline - time.time()) * 0.35))
        h_ls, c_ls = best_improvement_local_search(best_h, n, ls_dl, max_rounds=15000)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
    # Fine local search
    if time.time() < deadline - 10:
        fd_dl = min(deadline - 5, time.time() + min(60, (deadline - time.time()) * 0.4))
        h_fine, c_fine = fine_local_search(best_h, n, fd_dl, max_rounds=40000)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine
    return project_feasible(best_h, n), best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Optimize at initial resolution
    if time.time() < deadline - 500:
        alloc1 = min(200, deadline - time.time() - 500)
        h_opt, c_opt = optimize_at_resolution(best_h, n_init, time.time() + alloc1, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 2: Higher resolutions
    n_tries = [n_init + 50, n_init + 100, n_init + 200]
    n_tries = [x for x in n_tries if x >= n_init + 20 and x < n_init + 400]
    if n_tries:
        n_tries.append(n_init)
    for n_try in n_tries:
        if time.time() > deadline - 150:
            break
        remaining = deadline - time.time() - 50
        if remaining < 100:
            break
        alloc = min(180, remaining * 0.4)
        if alloc < 60:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 3: Multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 60:
        remaining = deadline - time.time() - 40
        n_starts = max(1, int(remaining / 90))
        for r in range(n_starts):
            if time.time() > deadline - 50:
                break
            alloc = min(80, deadline - time.time() - 40)
            if alloc < 40:
                break
            if r == 0:
                scale = 0.008
            elif r == 1:
                scale = 0.02
            else:
                scale = 0.03 + 0.01 * r
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            g_dl = time.time() + min(40, alloc * 0.45)
            h_g, c_g = gradient_descent(hp, n_best, g_dl,
                                         start_alpha=200, end_alpha=100000, max_iter=20000)
            if c_g > best_c:
                h_g = best_h
                c_g = best_c
            if time.time() < deadline - 45:
                ls_dl = min(deadline - 40, time.time() + min(30, alloc * 0.35))
                h_g, c_g = best_improvement_local_search(h_g, n_best, ls_dl, max_rounds=10000)
            if c_g < best_c:
                best_c = c_g
                best_h = h_g
    # Phase 4: Final intensive local search
    if time.time() < deadline - 15:
        fd_dl = min(deadline - 3, time.time() + min(50, deadline - time.time() - 8))
        best_h, best_c = fine_local_search(best_h, n_best, fd_dl, max_rounds=60000)
    # Final: best improvement local search
    if time.time() < deadline - 5:
        bi_dl = deadline - 2
        best_h, best_c = best_improvement_local_search(best_h, n_best, bi_dl, max_rounds=5000)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
