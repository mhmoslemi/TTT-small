```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            frac = diff / tot
            h = np.minimum(1.0, h + frac * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            frac = -diff / tot
            h = np.maximum(0.0, h * (1 - frac))
    return np.clip(h, 0.0, 1.0)

def smooth_obj_full(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad_full(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    n_active = (w > 1e-10).sum()
    # Only sum over active shifts
    dx = 2.0 / n
    grad = np.zeros(n)
    active_idx = np.where(w > 1e-10)[0]
    for mm in active_idx:
        k = int(mm) - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def top_shifts_obj(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def top_shifts_grad(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_lbfgs(h, n, deadline, alpha=None, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    if alpha is None:
        alpha = 5000.0
    try:
        res = minimize(smooth_obj_full, h, jac=smooth_grad_full,
                      args=(n, alpha), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxiter*10, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception as e:
        return h.copy(), compute_c5(h, n)

def pair_search_fine(h, n, deadline, radius=30):
    """Fine 2-point exchange search."""
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h * (1.0 + diff / tot))
    return np.clip(h, 0.0, 1.0)


def smooth_obj_full(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)


def smooth_grad_full(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    active_idx = np.where(w > 1e-12)[0]
    for mm in active_idx:
        k = int(mm) - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad


def top_shifts_obj(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)


def top_shifts_grad(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad


def run_lbfgs(h, n, deadline, alpha=5000, maxiter=5000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_full, best_h, jac=smooth_grad_full,
                      args=(n, alpha), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxiter * 10, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    except Exception:
        pass
    return best_h, best_c

        h_try = best_h.copy()
        # Push values in (0.1, 0.9) more aggressively toward 0 or 1

        # Make it smoother via simple averaging

        h_try = project_feasible(best_h + p_smooth, n)

            best_c = c_try
            best_h = h_try

            h_try = best_h.copy()
            h_try[start + shift:start + shift + block_len] = \

            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:

                best_h = h_try

        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try

def fine_pair_search(h, n, deadline, radius=25, steps=None):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.004, 0.002, 0.001, 0.0005, 0.0002]
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
```
