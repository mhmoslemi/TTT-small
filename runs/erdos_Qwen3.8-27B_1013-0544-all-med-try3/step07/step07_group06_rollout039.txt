def compute_delta_c(h, i, j, n):
    """Compute change in correlation when h[i] goes 1->0 and h[j] goes 0->1."""
    delta = np.zeros(2*n-1)
    
    # A: -(1-h[i+t]) for t = k-(n-1), valid when 0<=i+t<n => 0<=i+k-(n-1)<n
    lo_a = max(0, n-1-i)
    hi_a = min(2*n-2, n-1+i)
    idx_a = np.arange(lo_a, hi_a+1)
    h_idx = idx_a - (n-1) + i
    delta[lo_a:hi_a+1] -= (1 - h[h_idx])
    
    # B: +(1-h[j+t]) similarly
    lo_b = max(0, n-1-j)
    hi_b = min(2*n-2, n-1+j)
    idx_b = np.arange(lo_b, hi_b+1)
    h_idx_b = idx_b - (n-1) + j
    delta[lo_b:hi_b+1] += (1 - h[h_idx_b])
    
    # C: +h[i-t] for t = k-(n-1), valid when 0<=i-t<n => 0<=i-k+(n-1)<n
    # i-k+(n-1) in [0,n-1] => k in [i-(n-1), i+(n-1)]... wait
    # i - (k-(n-1)) in [0,n-1] => i-k+n-1 in [0,n-1] => n-1 <= i-k+n-1 <= n-1+n-1
    # => 0 <= i-k <= n-1 => i-(n-1) <= k <= i
    lo_c = max(0, i-(n-1))
    hi_c = min(2*n-2, i)
    idx_c = np.arange(lo_c, hi_c+1)
    h_idx_c = i - (idx_c - (n-1))  # = i - idx_c + n - 1
    delta[lo_c:hi_c+1] += h[h_idx_c]
    
    # D: -h[j-t] similarly
    lo_d = max(0, j-(n-1))
    hi_d = min(2*n-2, j)
    idx_d = np.arange(lo_d, hi_d+1)
    h_idx_d = j - (idx_d - (n-1))
    delta[lo_d:hi_d+1] -= h[h_idx_d]
    
    # E: +1 at t=j-i => k = j-i+n-1
    k_e = j - i + n - 1
    if 0 <= k_e < 2*n-1:
        delta[k_e] += 1
    
    # F: +1 at t=i-j => k = i-j+n-1
    k_f = i - j + n - 1
    if 0 <= k_f < 2*n-1:
        delta[k_f] += 1
    
    return delta
