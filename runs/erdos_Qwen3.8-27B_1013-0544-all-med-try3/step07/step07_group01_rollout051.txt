def correlate_fft(h, g, n):
    """Compute np.correlate(h, g, mode='full') using FFT"""
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    H = np.fft.rfft(h, fft_size)
    G = np.fft.rfft(g, fft_size)
    # Correlation is convolution with flipped g
    result = np.fft.irfft(H * np.conj(G), fft_size)

    # result[n-1] = sum a[i]*b[i] (the "full" center)
    return result[:size]

                        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        w = np.exp(c_shift - mx)
    else:
        w = np.exp(c_shift)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_phase(h, n, deadline, alphas=None, max_iters=300):
    if alphas is None:
        alphas = [5000, 50000, 500000, 2000000]
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        mu = 0.05
        def obj(x):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            if mx > 500:
                val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
            else:
                val = m + np.log(np.sum(np.exp(sh))) / alpha
            val += mu * (x.sum() - n / 2.0) ** 2
            return float(val)
        def grad_fn(x):
            g = grad_smooth_max(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=grad_fn, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 4})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

            g_norm = np.linalg.norm(g_proj)

                break
            g_dir = g_proj / g_norm
            improved = False
            s = lr

    best_c = compute_c5(h, n)

    T = T_init

            break

            i = rng.randint(n)
            j = rng.randint(n)

                h[j] -= d

                    n_accept += 1
                    if cur_c < best_c - 1e-15:
                        best_c = cur_c
                        best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
            elif h[i] - d >= 0.0 and h[j] + d <= 1.0:

        return np.ones(n)
