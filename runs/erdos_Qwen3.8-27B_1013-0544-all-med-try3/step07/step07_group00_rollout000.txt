        while improved and time.time() < t_end:
            improved = False
            for ii in range(min(80, len(top_indices))):
                if time.time() > t_end:
                    break
                i = top_indices[ii]
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def single_point_search(h, n, t_end, base_deltas=None, radius=100):
    best_c = compute_c5(h, n)
    if base_deltas is None:
        base_deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in base_deltas:
        improved = True
        while improved and time.time() < t_end:
            improved = False
            for i in range(n):
                if time.time() > t_end:
                    break
                if h[i] + d <= 1.0 and h.sum() + d <= n / 2.0 + 1e-10:
                    h[i] += d
                    # Need to compensate: find j to decrease
                    j_best = None
                    c_best = compute_c5(h, n)
                    if c_best < best_c - 1e-14:
                        # But sum constraint violated; skip for now
                        pass
                    h[i] -= d
                if h[i] - d >= 0.0 and h.sum() - d >= n / 2.0 - 1e-10:
                    h[i] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] += d
    return h, best_c

def slqp_refine(h, n, t_end, alpha=200000):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def cons_fun(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(len(x))
    try:
        res = minimize(obj_full, h, jac=grad_full, args=(n, alpha, 0.0),
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun,
                                     'jac': cons_jac}],
                       options={'maxiter': 2000, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Quick L-BFGS at initial resolution
    if time.time() < deadline - 600:
        t1_end = min(t0 + 40, deadline - 600)
        for alpha, mu in [(500, 1.0), (5000, 10.0), (50000, 100.0)]:
            if time.time() > t1_end:
                break
            c_vec = compute_c5_vec(best_h, n_init)
            active = get_active_shifts(c_vec, n_init, k_count=30, rel_tol=0.005)
            h_try = lbfgs_step(best_h, n_init, t1_end, alpha, mu,
                               top_idx=active, maxiter=2000)
            c_try = compute_c5(h_try, n_init)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try

    # Phase 2: Upsample to n=1000
    n_target = 1000
    x_old = np.linspace(0, 2, n_init)
    x_new = np.linspace(0, 2, n_target)
    h_fine = np.interp(x_new, x_old, best_h)
    h_fine = project_feasible(h_fine, n_target)
    c_fine = compute_c5(h_fine, n_target)
    if c_fine < best_c:
        best_h = h_fine
        best_c = c_fine
    else:
        best_h = h_fine
        best_c = c_fine

    # Phase 3: Main L-BFGS optimization at n=1000
    n = n_target
    if time.time() < deadline - 350:
        t3_end = min(time.time() + 250, deadline - 350)
        lbfgs_configs = [
            (500, 1.0, 30, 0.005),
            (5000, 10.0, 25, 0.004),
            (20000, 50.0, 20, 0.003),
            (100000, 200.0, 15, 0.002),
            (500000, 1000.0, 12, 0.001),
            (2000000, 5000.0, 10, 0.001),
        ]
        for alpha, mu, k_shifts, rel_tol in lbfgs_configs:
            if time.time() > t3_end:
                break
            c_vec = compute_c5_vec(best_h, n)
            active = get_active_shifts(c_vec, n, k_count=k_shifts, rel_tol=rel_tol)
            remaining = t3_end - time.time()
            maxiter = max(500, int(remaining * 200))
            h_try = lbfgs_step(best_h, n, t3_end, alpha, mu,
                               top_idx=active, maxiter=maxiter)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try

    # Phase 4: Full-correlation L-BFGS (all shifts weighted)
    if time.time() < deadline - 250:
        t4_end = min(time.time() + 120, deadline - 250)
        for alpha in [1000, 10000, 100000]:
            if time.time() > t4_end:
                break
            h_try = lbfgs_step(best_h, n, t4_end, alpha, 10.0,
                               top_idx=None, maxiter=3000)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try

    # Phase 5: Iterative top-shift refinement
    if time.time() < deadline - 200:
        t5_end = min(time.time() + 120, deadline - 200)
        for _round in range(5):
            if time.time() > t5_end:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            for rel_tol in [0.001, 0.003, 0.01]:
                active = get_active_shifts(c_vec, n, k_count=15, rel_tol=rel_tol)
                if len(active) < 2:
                    active = np.argsort(-c_vec)[:10]
                remaining = t5_end - time.time()
                if remaining < 5:
                    break
                for alpha in [50000, 500000, 2000000]:
                    if time.time() > t5_end:
                        break
                    maxiter = max(500, int((t5_end - time.time()) * 100))
                    h_try = lbfgs_step(best_h, n, t5_end, alpha, 50.0,
                                       top_idx=active, maxiter=maxiter)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c:
                        best_c = c_try
                        best_h = h_try
                        c_vec = compute_c5_vec(best_h, n)
                        max_val = c_vec.max()

    # Phase 6: Local pair search
    if time.time() < deadline - 100:
        t6_end = min(time.time() + 80, deadline - 100)
        for deltas in [
            [0.005, 0.002, 0.001, 0.0005],
            [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
            [0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
        ]:
            if time.time() > t6_end:
                break
            best_h, best_c = local_pair_search(best_h, n, t6_end,
                                              base_deltas=deltas, radius=60)

    # Phase 7: SLSQP polish
    if time.time() < deadline - 40:
        t7_end = min(time.time() + 40, deadline - 40)
        for alpha in [100000, 1000000]:
            if time.time() > t7_end:
                break
            h_try, c_try = slqp_refine(best_h, n, t7_end, alpha=alpha)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try

    # Phase 8: Final fine local search
    if time.time() < deadline - 10:
        t8_end = min(time.time() + 25, deadline - 10)
        best_h, best_c = local_pair_search(best_h, n, t8_end,
                                          base_deltas=[0.0001, 0.00005,
                                                      0.00002, 0.00001,
                                                      0.000005],
                                          radius=40)

    # Phase 9: Final SLSQP
    if time.time() < deadline - 5:
        h_try, c_try = slqp_refine(best_h, n, deadline - 5, alpha=500000)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try

    # Also try at n=500 (the parent resolution) in case it gives better result
    n500 = 500
    x500 = np.linspace(0, 2, n500)
    h500 = np.interp(x500, np.linspace(0, 2, n), best_h)
    h500 = project_feasible(h500, n500)
    c500 = compute_c5(h500, n500)
    if c500 < best_c:
        best_h = h500
        best_c = c500
        n = n500

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
