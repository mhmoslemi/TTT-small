import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def top_shifts_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def top_shifts_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def nesterov_pg(h, n, deadline, top_k=3, alpha=50000, lr=0.001, momentum=0.9):
    """Nesterov-accelerated projected gradient on top-shifts objective."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    h_curr = h.copy()
    
    for step in range(10000):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h_curr, n)
        # Find active shifts
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])[:top_k]]
        
        # Nesterov lookahead
        h_look = h_curr + momentum * v
        
        # Gradient at lookahead point (clipped)
        h_look = np.clip(h_look, 0, 1)
        grad = top_shifts_grad(h_look, n, active, alpha, 0.0)
        
        # Update

            best_h = h_new.copy()

        # Decay learning rate

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def top_shifts_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def top_shifts_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def nesterov_pg(h, n, deadline, top_k=3, alpha=100000, lr=0.005, beta=0.9):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    h_curr = h.copy()
    stale = 0

    for step in range(50000):
        if time.time() > deadline:
            break

        c_vec = compute_c5_vec(h_curr, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])[:top_k]]

        h_look = np.clip(h_curr + beta * v, 0, 1)
        grad = top_shifts_grad(h_look, n, active, alpha, 0.0)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        lr_adapt = lr / (1.0 + 0.1 * np.linalg.norm(h_curr))
        v = beta * v - lr_adapt * grad
        h_new = h_curr + v
        h_new = project_feasible(h_new, n)

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
            if stale > 500:
                lr *= 0.5
                stale = 0
                if lr < 1e-6:
                    break

        h_curr = h_new

    return best_h, best_c

def reoptimize_top_shifts(h, n, deadline, top_k=3, alphas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

                                   'gtol': 1e-15, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def gradient_local_search(h, n, deadline, n_candidates=30, n_steps=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    base_step = 0.005

    for iteration in range(n_steps):

        for idx in active:
            grad += shift_grad_single(best_h, n, idx)
        grad /= len(active)

        cand_idx = np.argsort(-np.abs(grad))[:n_candidates]
        found_improve = False

        for ii in range(len(cand_idx)):
            if time.time() > deadline:
                return best_h, best_c
            i = cand_idx[ii]
            for jj in range(len(cand_idx)):
                if ii == jj:
                    continue
                j = cand_idx[jj]
                if best_h[i] + best_h[j] < 1e-12:
                    continue

                for direction in [1, -1]:
                    for mult in [1.0, 0.5, 0.25, 0.1]:

                        if direction == 1:

                                best_h[i] -= step
                                best_h[j] += step
                                c = compute_c5(best_h, n)
                                if c < best_c - 1e-15:
                                    best_c = c
                                    found_improve = True
                                else:
                                    best_h[i] += step
                                    best_h[j] -= step
                        else:
                            if best_h[i] + step <= 1 and best_h[j] - step >= 0:
                                best_h[i] += step
                                best_h[j] -= step
                                c = compute_c5(best_h, n)
                                if c < best_c - 1e-15:
                                    best_c = c
                                    found_improve = True
                                else:
                                    best_h[i] -= step
                                    best_h[j] += step

                if found_improve:
                    break
            if found_improve:
                break

        if not found_improve:
            base_step *= 0.5
            if base_step < 5e-6:
                break

    return best_h, best_c

def random_perturb(h, n, magnitude=0.05):

    mask = np.random.random(n) < 0.3
    h_new[mask] += np.random.randn(mask.sum()) * magnitude
