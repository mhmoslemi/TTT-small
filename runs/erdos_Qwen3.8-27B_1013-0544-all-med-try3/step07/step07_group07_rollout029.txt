import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    m_range = np.arange(-(n - 1), n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

def pgd_solve(h, n, alphas, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        cur_c = best_c
        step = min(float(n) / (4.0 * alpha), 0.5)
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            grad = lse_grad(h, n, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            direction = -grad / gnorm
            accepted = False
            for ls in range(30):
                h_new = project(h + step * (0.5 ** ls) * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    h = h_new
                    cur_c = c_new
                    step = step * (0.5 ** ls)
                    accepted = True
                    break
            if accepted:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                break
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def delta_1for1(h, n, i, j):
    """Compute delta corr when swapping: set h[i]=1 (was 0), h[j]=0 (was 1)."""
    d = np.zeros(2 * n - 1)
    m_range = np.arange(-(n - 1), n)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    d[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    d[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    d[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    d[msk] += h[jmn[msk]]
    d[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        d[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        d[k2] += 1.0
    return d

def delta_2for2(h, n, i1, i2, j1, j2):
    """Compute delta corr for 2-for-2 swap: set h[i1]=h[i2]=1 (were 0), h[j1]=h[j2]=0 (were 1)."""
    d = np.zeros(2 * n - 1)
    m_range = np.arange(-(n - 1), n)
    for i in (i1, i2):
        im = i + m_range
        msk = (im >= 0) & (im < n)
        d[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        d[msk] -= h[imn[msk]]
    for j in (j1, j2):
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        d[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        d[msk] += h[jmn[msk]]
    d[n - 1] -= 4.0
    positions = [i1, i2, j1, j2]
    dvals = [1, 1, -1, -1]
    for a in range(4):
        for b in range(4):
            if a == b:
                continue
            m = positions[b] - positions[a]
            if -(n - 1) <= m <= n - 1:
                d[n - 1 + m] -= dvals[a] * dvals[b]
    return d

def local_search_1for1(h, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    m_range = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_list):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                d = delta_1for1(h, n, i, j)
                nc = (corr + d).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                d = delta_1for1(h, n, i, best_j)
                h[i], h[best_j] = 1.0, 0.0
                corr += d
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def local_search_2for2(h, n, deadline, max_iters=50000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        a1 = rng.randint(len(zeros_arr))
        a2 = rng.randint(len(zeros_arr))
        b1 = rng.randint(len(ones_arr))
        b2 = rng.randint(len(ones_arr))
        if a1 == a2 or b1 == b2:
            continue
        i1, i2 = int(zeros_arr[a1]), int(zeros_arr[a2])
        j1, j2 = int(ones_arr[b1]), int(ones_arr[b2])
        d = delta_2for2(h, n, i1, i2, j1, j2)
        nc = (corr + d).max() * dx
        if nc < cur_c - 1e-15:
            h[i1], h[i2], h[j1], h[j2] = 1.0, 1.0, 0.0, 0.0
            corr += d
            cur_c = nc
            zeros_arr[a1], zeros_arr[a2] = j1, j2
            ones_arr[b1], ones_arr[b2] = i1, i2
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def anneal_binary(h, n, deadline, seed=42, T0=0.03, cool=0.99995, n_iters=1000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = T0
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        if T < 1e-10:
            break
        use_2for2 = rng.random() < 0.3
        if use_2for2:
            a1 = rng.randint(len(zeros_arr))
            a2 = rng.randint(len(zeros_arr))
            b1 = rng.randint(len(ones_arr))
            b2 = rng.randint(len(ones_arr))
            if a1 == a2 or b1 == b2:
                T *= cool
                continue
            i1, i2 = int(zeros_arr[a1]), int(zeros_arr[a2])
            j1, j2 = int(ones_arr[b1]), int(ones_arr[b2])
            d = delta_2for2(h, n, i1, i2, j1, j2)
            nc = (corr + d).max() * dx
            if nc <= cur_c or rng.random() < np.exp(-(nc - cur_c) / T):
                h[i1], h[i2], h[j1], h[j2] = 1.0, 1.0, 0.0, 0.0
                corr += d
                cur_c = nc
                zeros_arr[a1], zeros_arr[a2] = j1, j2
                ones_arr[b1], ones_arr[b2] = i1, i2
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        else:
            ai = rng.randint(len(zeros_arr))
            bi = rng.randint(len(ones_arr))
            i = int(zeros_arr[ai])
            j = int(ones_arr[bi])
            d = delta_1for1(h, n, i, j)
            nc = (corr + d).max() * dx
            if nc <= cur_c or rng.random() < np.exp(-(nc - cur_c) / T):
                h[i], h[j] = 1.0, 0.0
                corr += d
                cur_c = nc
                zeros_arr[ai] = j
                ones_arr[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        T *= cool
    return best_h, best_c

def try_resolution(h_best, n_best, c_best, deadline, seed, min_time=30):
    """Try to improve by optimizing at a new resolution."""
    if time.time() > deadline - min_time:
        return h_best, c_best, n_best
    for nh in [800, 1200]:
        if time.time() > deadline - 20:
            break
        if nh <= n_best and abs(nh - n_best) < 100:
            continue
        if time.time() > deadline - min_time:
            break
        # Upsample
        xo = np.linspace(0, 2, n_best + 1)
        xx = np.linspace(0, 2, nh + 1)
        hh = np.interp(xx[:nh], xo[:n_best], h_best)
        hh = project(hh, nh)
        # PGD
        alphas = [50, 200, 1000, 5000, 20000]
        time_left = deadline - 15
        pgd_dl = time_left - 0.4 * time_left
        hh, ch = pgd_solve(hh, nh, alphas, pgd_dl, max_iter=3000)
        if ch >= c_best:
            hb = to_binary(hh, nh)
        else:
            c_best = ch
            h_best = hh.copy()
            n_best = nh
            hb = to_binary(hh, nh)
        # Binary local search
        ls_dl = time_left - 0.15 * time_left
        hb, chb = local_search_1for1(hb, nh, ls_dl, max_sweeps=300)
        if chb < c_best:
            c_best = chb
            h_best = hb.copy()
            n_best = nh
        # 2-for-2 local search
        ls2_dl = time_left - 0.05 * time_left
        hb2 = to_binary(h_best, n_best)
        hb2, ch2 = local_search_2for2(hb2, n_best, ls2_dl, max_iters=30000, rng=np.random.RandomState(seed + nh))
        if ch2 < c_best:
            c_best = ch2
            h_best = hb2.copy()
            n_best = n_best
    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total
    rng = np.random.RandomState(seed)

    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    n = n0
    h0 = project(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: PGD at initial resolution
    p1_end = t0 + 0.20 * total
    alphas1 = [10, 50, 200, 1000, 5000, 20000]
    best_h, best_c = pgd_solve(best_h, n, alphas1, p1_end, max_iter=6000)

    # Phase 2: Binary local search (1-for-1)
    p2_end = t0 + 0.35 * total
    hb = to_binary(best_h, n)
    hb, cb = local_search_1for1(hb, n, p2_end, max_sweeps=500)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: 2-for-2 local search
    p3_end = t0 + 0.42 * total
    hb2 = to_binary(best_h, n)
    hb2, cb2 = local_search_2for2(hb2, n, p3_end, max_iters=40000, rng=np.random.RandomState(seed + 1))
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()

    # Phase 4: Annealing
    p4_end = t0 + 0.55 * total
    ha = to_binary(best_h, n)
    ha, ca = anneal_binary(ha, n, p4_end, seed=seed + 2, T0=0.04, cool=0.99994, n_iters=800000)
    if ca < best_c:
        best_c = ca
        best_h = ha.copy()

    # Phase 5: PGD polish from best binary
    p5_end = t0 + 0.62 * total
    alphas5 = [100, 500, 2000, 10000, 50000]
    best_h, best_c = pgd_solve(best_h, n, alphas5, p5_end, max_iter=4000)

    # Phase 6: Try higher resolution
    best_h, best_c, n = try_resolution(best_h, n, best_c, t0 + 0.82 * total, seed, min_time=60)

    # Phase 7: Final refinement at best resolution found
    p7_end = dl - 5
    if time.time() < p7_end - 20:
        hb_final = to_binary(best_h, n)
        ls_dl = p7_end - 0.3 * (p7_end - time.time())
        hb_final, cb_final = local_search_1for1(hb_final, n, ls_dl, max_sweeps=400)
        if cb_final < best_c:
            best_c = cb_final
            best_h = hb_final.copy()
        # 2-for-2 polish
        if time.time() < p7_end - 10:
            hb_f2 = to_binary(best_h, n)
            hb_f2, cb_f2 = local_search_2for2(hb_f2, n, p7_end - 5, max_iters=30000, rng=np.random.RandomState(seed + 100))
            if cb_f2 < best_c:
                best_c = cb_f2
                best_h = hb_f2.copy()
        # Final anneal
        if time.time() < p7_end - 15:
            ha_final = to_binary(best_h, n)
            ha_final, ca_final = anneal_binary(ha_final, n, p7_end - 2, seed=seed + 200, T0=0.02, cool=0.99996, n_iters=200000)
            if ca_final < best_c:
                best_c = ca_final
                best_h = ha_final.copy()

    # Final: ensure feasibility
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
