def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

def project_binary(h, n):

def lse_grad(h, n, alpha):

import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(np.max(c)) * dx

def project_fractional(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def project_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=float)
    hb[idx[:n // 2]] = 1.0
    return hb

    c = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n-1), n):

        if pk < 1e-16:

        i_min = max(0, k)

        i_min2 = max(0, -k)
        i_max2 = min(n-1, n-1-k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2+1] -= pk * h[i_min2+k:i_max2+k+1]
    return grad

        cur_c = best_c

            h_new = project_fractional(h_curr + v, n)

    return best_h, best_c

import numpy as np
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(np.max(c)) * dx

def project_fractional(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def project_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=float)
    hb[idx[:n // 2]] = 1.0
    return hb

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

def pgd_solve(h, n, alphas, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_curr = best_h.copy()
        v = np.zeros(n)
        lr = min(float(n) / (4.0 * alpha), 0.1)
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            h_look = project_fractional(h_curr - 0.9 * v, n)
            grad = lse_grad(h_look, n, alpha)
            v = 0.9 * v - lr * grad
            h_new = project_fractional(h_curr + v, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h_curr = h_new
            lr *= 0.9995
        c_final = compute_c5(h_curr, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_curr.copy()
    return best_h, best_c

def swap_delta(h, i, j, n):
    m_range = np.arange(-(n - 1), n)
    L = 2 * n - 1
    delta = np.zeros(L)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk2 = (imn >= 0) & (imn < n)
    delta[msk2] -= h[imn[msk2]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk2 = (jmn >= 0) & (jmn < n)
    delta[msk2] += h[jmn[msk2]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < L:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < L:
        delta[k2] += 1.0
    return delta

def binary_local_search(hb, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = hb.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_list):
            if time.time() > deadline:
                break
            if h[i] != 0:
                continue
            best_j = -1
            best_nc = cur_c - 1e-15
            for j in ones_set:
                if h[j] != 1:
                    continue
                delta = swap_delta(h, i, j, n)
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_set.discard(best_j)
                ones_set.add(i)
                if i in zeros_list:
                    zeros_list.remove(i)
                zeros_list.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def anneal_binary(hb, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = hb.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = float(corr.max()) * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    T = 0.008
    cool = 0.999992
    min_T = 1e-9
    it = 0
    while time.time() < deadline:
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = swap_delta(h, i, j, n)
        nc = float((corr + delta).max()) * dx
        accept = False
        if nc <= cur_c + 1e-15:
            accept = True
        elif T > min_T and rng.random() < math.exp(-(nc - cur_c) / T):
            accept = True
        if accept:
            h[i], h[j] += 1.0, -1.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
        it += 1
    return best_h, best_c

def perturbed_restart(hb, n, rng, n_flip=5):
    h = hb.astype(float).copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_flip = min(n_flip, len(ones), len(zeros))
    for _ in range(n_flip):
        oi = rng.randint(len(ones))
        zi = rng.randint(len(zeros))
        h[ones[oi]], h[zeros[zi]] = 1.0 - h[ones[oi]], 1.0 - h[zeros[zi]]
    # Re-sort one/zero lists
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    return h
