import numpy as np

def project_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    if n <= 0: return h
    target = 0.5 * float(n)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - float(h.sum())
    if abs(diff) < 1e-11:
        return h
    if diff > 0.0:
        room = 1.0 - h
        s = float(room.sum())
        if s > 1e-15:
            h += (diff / s) * room
        else:
            h.fill(target / n)
    else:
        s = float(h.sum())
        if s > 1e-15:
            h += (diff / s) * h
        else:
            h.fill(target / n)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - float(h.sum())
    if abs(diff) < 1e-10:
        return h
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        s = float(hp.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

            s = float(h.sum())

def continuous_active_optimize(h0,n,deadline,c_margins,step_factor=5.0,max_stall=None):
    h0 = np.asarray(h0, dtype=float).copy()
    best_h = h0.copy()
    best_c = evaluate_h(best_h,n)
    h = best_h.copy()
    cur_c = best_c
    factors = [1.0, 2.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]
    if max_stall is None:
       max_stall = 250
    for cm in c_margins:
       if time.time() > deadline: break
       step = min(0.05, max(1e-5, float(cm) * step_factor))
       stall = 0
       stall_limit = max_stall
       if cm < 2e-4: stall_limit = max(max_stall, 600)
       while time.time() < deadline:
           grad, corr, maxc = active_gradient(h,n,cm)
           gnorm = float(np.linalg.norm(grad))
           if gnorm < 1e-14: break
           direction = -grad / gnorm
           accepted = False
           used = None
           for fac in factors:
               if time.time() > deadline: break
               trial = project_sum(h + step * fac * direction, n)
               c = evaluate_h(trial, n)
               if c < cur_c - 1e-15:
                   h = trial; cur_c = c; accepted = True; used = fac; break
           if accepted:
               if cur_c < best_c:
                   best_c = cur_c; best_h = h.copy()
               stall = 0
               if used is not None:
                   step = min(0.1, max(1e-6, step * float(used) * 1.1))
           else:
               stall += 1
               step *= 0.8
               if stall > stall_limit or step < 1e-7:
                   break
       h = best_h.copy()
       cur_c = best_c
    return best_h, float(best_c)

import numpy as np
import time


def project_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    if n <= 0:
        return h
    target = 0.5 * float(n)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - float(h.sum())
    if abs(diff) < 1e-11:
        return h

    if diff > 0.0:
        room = 1.0 - h
        s = float(room.sum())
        if s > 1e-15:
            h += (diff / s) * room
        else:
            h.fill(target / float(n))
    else:
        s = float(h.sum())
        if s > 1e-15:
            h += (diff / s) * h
        else:
            h.fill(target / float(n))

    np.clip(h, 0.0, 1.0, out=h)
    diff = target - float(h.sum())
    if abs(diff) < 1e-10:
        return h

    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(38):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        s = float(hp.sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            room = 1.0 - h
            s = float(room.sum())
            if s > 1e-15:
                h += (diff / s) * room
        else:
            s = float(h.sum())
            if s > 1e-15:
                h += (diff / s) * h
    np.clip(h, 0.0, 1.0, out=h)
    return h


def evaluate_h(h, n):
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / float(n)))


def add_lag_grad(grad, h, k, pk, n):
    if pk == 0.0:
        return
    i_min = max(0, k)
    i_max = min(n - 1, n - 1 + k)
    if i_min <= i_max:
        grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])

    i_min2 = max(0, -k)
    i_max2 = min(n - 1, n - 1 - k)
    if i_min2 <= i_max2:
        grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]


def active_gradient(h, n, c_margin):
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = float(corr.max())
    margin = max(1e-6, float(c_margin) * float(n) * 0.5)
    idx = np.where(corr >= maxc - margin)[0]
    if idx.size == 0:
        idx = np.array([int(corr.argmax())])
    if idx.size > 80:
        vals = corr[idx]
        order = np.argsort(vals)[-80:]
        idx = idx[order]

    scale = max(1e-6, margin * 0.25)
    expv = np.exp((corr[idx] - maxc) / scale)
    wsum = float(expv.sum())
    if wsum <= 0.0:
        w = np.ones(idx.size, dtype=float) / float(idx.size)
    else:
        w = expv / wsum

    grad = np.zeros(n, dtype=float)
    for pos, wk in zip(idx, w):
        wk = float(wk)
        if wk <= 1e-12:
            continue
        k = int(pos - (n - 1))
        add_lag_grad(grad, h, k, wk, n)
    return grad, corr, maxc


def continuous_active_optimize(h0, n, deadline, c_margins, seed=0,
                               step_factor=5.0, max_stall=250):
    h0 = np.asarray(h0, dtype=float).copy()
    best_h = h0.copy()
    best_c = evaluate_h(best_h, n)

    h = best_h.copy()
    cur_c = best_c
    rng = np.random.RandomState(seed)
    factors = [1.0, 2.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]

    for cm in c_margins:
        if time.time() > deadline:
            break
        step = min(0.05, max(1e-5, float(cm) * float(step_factor)))
        stall = 0
        stall_limit = max_stall
        if cm < 2e-4:
            stall_limit = max(max_stall, 600)

        while time.time() < deadline:
            grad, corr, maxc = active_gradient(h, n, cm)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-14:
                break
            direction = -grad / gnorm
            accepted = False
            used = None

            for fac in factors:
                if time.time() > deadline:
                    break
                trial = project_sum(h + step * fac * direction, n)
                c = evaluate_h(trial, n)
                if c < cur_c - 1e-15:
                    h = trial
                    cur_c = c
                    accepted = True
                    used = fac
                    break

            if accepted:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                stall = 0
                if used is not None:
                    step = min(0.1, max(1e-6, step * float(used) * 1.1))
            else:
                stall += 1
                step *= 0.8
                if stall > stall_limit or step < 1e-7:
                    break

        h = best_h.copy()
        cur_c = best_c

    return best_h, float(best_c)

def binary_from_values(vals, n):
    vals = np.asarray(vals, dtype=float)
    b = np.zeros(n, dtype=np.int16)
    if n <= 0:
        return b
    k = n // 2

    if n % 2 == 0:
        vb = (vals >= 0.5).astype(np.int16)
        if int(vb.sum()) == k:
            return vb

    idx = np.argsort(vals)[::-1][:k]
    b[idx] = 1
    return b


def binary_eval(h, z):
    corr = np.correlate(h, z, mode="full")
    corr = corr.astype(np.int32, copy=False)
    return corr, int(corr.max())


def split_corr_binary(corr, n):
    p = np.zeros(n, dtype=np.int32)
    q = np.zeros(n, dtype=np.int32)
    if n > 1:
        p[1:] = corr[:n - 1][::-1]
        q[1:] = corr[n:]
    return p, q


def build_hot_binary(p, q, n, limit):
    hot = []
    for s in range(1, n):
        if p[s] > limit:
            hot.append((s, 0))
        if q[s] > limit:
            hot.append((s, 1))
    return hot


def sample_offending_binary(h, s, side, n, rng):
    length = n - s
    if length <= 0:
        return None

    for _ in range(20):
        i = int(rng.randint(length))
        a = int(h[i])
        b = int(h[i + s])
        if side == 0:
            if a == 1 and b == 0:
                return i, i + s
        else:
            if a == 0 and b == 1:
                return i, i + s

    if side == 0:
        idx = np.where((h[:length] == 1) & (h[s:] == 0))[0]
    else:
        idx = np.where((h[:length] == 0) & (h[s:] == 1))[0]

    if idx.size == 0:
        return None
    i = int(idx[int(rng.randint(idx.size))])
    return i, i + s


def choose_not(lst, rng, excl):
    length = len(lst)
    if length <= 1:
        return None
    while True:
        idx = int(rng.randint(length))
        val = int(lst[idx])
        if val != excl:
            return val

        h0 = np.asarray(initial_h_values, dtype=float).copy()

    except Exception:
        n = 400
        h0 = np.full(n, 0.5, dtype=float)

    h0 = project_sum(h0, n)
    best_h = h0.copy()
    best_c = evaluate_h(best_h, n)

    t1 = t0 + 0.18 * total
    if time.time() < t1:
        hh, cc = continuous_active_optimize(
            best_h, n, t1,
            [0.01, 0.005, 0.002, 0.001, 0.0005],
            seed=int(seed) + 1,
            max_stall=180
        )
        if cc < best_c:
            best_c = cc
            best_h = hh

    t2 = t0 + 0.34 * total
    if n % 2 == 0 and time.time() < t2:
        bh = binary_from_values(best_h, n)
        ztmp = 1 - bh
        bcorr, bmax = binary_eval(bh, ztmp)
        bcand = 2.0 * float(bmax) / float(n)
        if bcand < best_c:
            best_c = bcand
            best_h = bh

        target_limit = int(np.floor(0.3808 * float(n) * 0.5))
        limit = min(target_limit, max(0, int(bmax) - 1))
        if limit >= 0 and time.time() < t2:
            bh2, bmax2, ok = binary_limit_search(bh, n, limit, t2, int(seed) + 11)
            bcand2 = 2.0 * float(bmax2) / float(n)
            if bcand2 < best_c:
                best_c = bcand2
                best_h = bh2

    t3 = t0 + 0.88 * total
    if time.time() < t3:
        hh, cc = continuous_active_optimize(
            best_h, n, t3,
            [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00003],
            seed=int(seed) + 23,
            max_stall=350
        )
        if cc < best_c:
            best_c = cc
            best_h = hh

    if time.time() < deadline - 2.0 and best_c > 0.3807:
        hh, cc = continuous_small_polish(best_h, n, deadline - 2.0, seed=int(seed) + 97)
        if cc < best_c:
            best_c = cc
            best_h = hh

    rem = deadline - time.time()
    if rem > 70.0 and n < 800:
        nh = int(2 * n)
        x = np.linspace(0.0, 2.0, n)
        xn = np.linspace(0.0, 2.0, nh)
        hh = np.interp(xn, x, best_h)
        hh = project_sum(hh, nh)
        ch = evaluate_h(hh, nh)
        if ch < best_c + 0.0005:
            up_deadline = max(time.time(), min(deadline - 2.0, time.time() + 60.0))
            hh2, ch2 = continuous_active_optimize(
                hh, nh, up_deadline,
                [0.005, 0.002, 0.001, 0.0005, 0.0002],
                seed=int(seed) + 131,
                max_stall=150
            )
            if ch2 < best_c:
                best_c = ch2
                best_h = hh2
                n = nh

    best_h = project_sum(best_h, n)
    best_c = evaluate_h(best_h, n)
    return best_h, float(best_c), int(n)
