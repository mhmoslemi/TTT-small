```python
            # Sensitivity for this shift
            # S[i] = dx * [1_{0<=i-k<n}*(1-h[i-k]) - 1_{0<=i+k<n}*h[i+k]]
            S = np.zeros(n)
            if k >= 0:
                S[0:n-k] += (1.0 - h[k:n])  # first index contribution
                S[k:n] -= h[0:n-k]           # second index contribution
            else:
                kk = -k
                S[kk:n] += (1.0 - h[0:n-kk])
                S[0:n-kk] -= h[kk:n]
            S *= dx
            
            # For pair (i,j): Δc[m_idx] = d * (S[i] - S[j])
            # We want S[i] > S[j] to decrease c[m_idx] with h[i]+=d, h[j]-=d
            # Or S[i] < S[j] to decrease with h[i]-=d, h[j]+=d
            
            # Sort by S to find best pairs
            order = np.argsort(S)
            # Best "increase" indices (low S) and "decrease" indices (high S)
            
            # Try a few promising pairs
            for _ in range(min(20, n)):
                i = order[_]  # low S, want to increase h[i]
                j = order[n-1-_]  # high S, want to decrease h[j]
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d; h[j] -= d
                    c_new = c_vec.copy()
                    # Compute change in all c[k]
                    # For each shift k': Δc[k'] = d * (S_all[k'][i] - S_all[k'][j])
                    # This is expensive... 
                    c_new_val = compute_c5(h, n)
                    if c_new_val < best_c - 1e-14:
                        best_c = c_new_val; improved = True
                        c_new = compute_c5_vec(h, n)
                    else:
                        h[i] -= d; h[j] += d
        
        if not improved:
            break
    
    return h, best_c
```

A much better approach: **coordinate descent** where we only change ONE variable at a time.
- For each i, find the best value of h[i] (given all others fixed)
- This is a 1D optimization: minimize max_k c_k subject to the integral constraint
- The optimal h[i] is at a boundary (0 or 1) or where the max is "flat"
- Even simpler: try h[i] = 0, h[i] = 1, and the current value, pick the best

Actually, for coordinate descent:
- Fix all h[j] for j≠i
- The integral constraint means h[i] = n/2 * dx - sum_{j≠i} h[j] * dx = (n/2 - sum_{j≠i} h[j]) * dx... 

No wait, the constraint is sum(h) = n/2. If we fix all except h[i], then h[i] is DETERMINED: h[i] = n/2 - sum_{j≠i} h[j]. So coordinate descent doesn't work directly!

Unless we relax the constraint and use the penalty method. Or we do "pair moves" (change two variables to maintain the sum).

OK so pair moves are necessary. Let me think about how to make them efficient.

**Key optimization**: Instead of evaluating the full C5 (O(n²)) for each pair, evaluate only the TOP few shifts. If the current max is c[k*], and we're making a small move, it's unlikely that a previously non-max shift becomes the new max. So we can:
1. Identify the top 5 shifts
2. For each candidate pair, compute the change in just these 5 shifts (O(5) per pair)
3. If the predicted max decreases, do the full evaluation to confirm

This reduces the per-pass cost from O(n⁴) to O(n² * 5) = O(n²) for the screening, plus occasional O(n²) full evaluations.

```python
def smart_pair_search(h, n, deadline, top_m=5):
    best_c = compute_c5(h, n)
    dx = 2.0 / n

    for d in deltas:
        while time.time() < deadline:
            c_vec = compute_c5_vec(h, n)
            best_c = c_vec.max()
            # Top m shifts
            top_idx = np.argsort(c_vec)[-top_m:]
            
            # Compute sensitivities for top shifts
            S_top = np.zeros((top_m, n))
            for a, m_idx in enumerate(top_idx):
                k = m_idx - (n-1)
                if k >= 0:
                    S_top[a, 0:n-k] += (1.0 - h[k:n])
                    S_top[a, k:n] -= h[0:n-k]
                else:
                    kk = -k
                    S_top[a, kk:n] += (1.0 - h[0:n-kk])
                    S_top[a, 0:n-kk] -= h[kk:n]
            S_top *= dx
            
            # For pair (i,j) with h[i]+=d, h[j]-=d:
            # Δc[m] = d * (S_top[:,i] - S_top[:,j])
            # New max ≈ max(c_vec[top_idx] + d * (S_top[:,i] - S_top[:,j]))
            # (ignoring the effect on non-top shifts)
            
            improved = False
            n_screened = 0
            n_accepted = 0
            
            # Strategy: for each i, find best j
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c

                # If S_top[:, i] is large (positive), increasing h[i] increases top c's

                # If S_top[:, i] is small (negative), decreasing h[i] increases top c's

                # We want this to be < best_c
                # So we need: max_top (S_top[:,j] - S_top[:,i]) < (best_c - max_top c) / d
                # But max_top c = best_c (since top includes the max), so threshold is 0
                # We need S_top[:,j] < S_top[:,i] for ALL top shifts

                # Simplification: just try a few j values

                # Best j for "decrease h[i]": maximize min_a S_top[a, j]

                # and j that maximizes max_a (S_top[a,i] - S_top[a,j])
```
