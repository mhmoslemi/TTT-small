def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        val = m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):
    return grad_smooth_max(h, n, alpha)

def slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(len(x))

def lbfgs_optimize(h, n, deadline, max_iters=400):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (50, 0.0), (200, 0.0), (1000, 0.0), (5000, 0.0),
        (20000, 0.0), (100000, 0.0), (500000, 0.0),
        (2000000, 0.0), (10000000, 0.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        if deadline - time.time() < 5:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 30,
                                   'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=3000):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def projected_grad(h, n, deadline, alpha=50000, max_iters=10000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.01
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        improved = False
        step = lr
        for _ in range(60):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=50):
    if deltas is None:
        deltas = [0.01, 0.005, 0.001, 0.0005, 0.0001, 0.00005, 0.00001]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] = min(1.0, h[i] + d)
                        h[j] = max(0.0, h[j] - d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            h = project_feasible(h, n)
                        else:
                            h[i] -= d
                            h[j] += d
                            h = np.clip(h, 0.0, 1.0)
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] = max(0.0, h[i] - d)
                        h[j] = min(1.0, h[j] + d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            h = project_feasible(h, n)
                        else:
                            h[i] += d
                            h[j] -= d
                            h = np.clip(h, 0.0, 1.0)
            if not improved:
                break
    return h, best_c

def three_point_search(h, n, deadline, delta=0.005):
    """Move 3 points simultaneously for better escapes."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    while time.time() < deadline:
        improved = False
        for _ in range(5000):
            if time.time() > deadline:
                return h, best_c
            i, j, k = rng.choice(n, size=3, replace=False)
            h_save = h.copy()
            # Move delta from i to j, and delta/2 from j to k
            h[i] -= delta
            h[j] += delta - delta * 0.3
            h[k] += delta * 0.3
            if h[i] < 0 or h[j] > 1 or h[k] > 1:
                h = h_save
                continue
            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
            else:
                h = h_save
        if not improved:
            break
    return h, best_c

def perturb_reoptimize(h, n, deadline, rng, n_perturb=6, scale=0.04):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    per_trial = max(5, (deadline - time.time()) / max(1, n_perturb))
    for _ in range(n_perturb):
        if time.time() > deadline:
            break
        h_p = h.copy()
        mask = rng.random(n) < 0.4
        h_p[mask] += rng.normal(0, scale, int(mask.sum()))
        h_p = project_feasible(h_p, n)
        t_end = min(time.time() + per_trial * 0.7, deadline - 2)
        h_p, c_p = lbfgs_optimize(h_p, n, t_end, max_iters=150)
        if time.time() < deadline - 3:
            h_p, c_p2 = pair_search(h_p, n, deadline - 2,
                                     deltas=[0.001, 0.0001], radius=25)
            if c_p2 < c_p:
                c_p = c_p2
            h_p = np.clip(h_p, 0, 1)
            h_p = project_feasible(h_p, n)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p.copy()
    return best_h, best_c

def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def binary_local_search(h, n, deadline):
    """Try to find a better binary (or near-binary) arrangement."""
    h = project_feasible(h, n)
    # Threshold to binary
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Fix sum
    excess = h_bin.sum() - n / 2.0
    if excess > 0:
        order = np.argsort(h[ h_bin == 1.0 ])
        for idx in order[:int(excess)]:
            h_bin[idx] = 0.0
    elif excess < 0:
        order = np.argsort(-h[ h_bin == 0.0 ])
        for idx in order[:int(-excess)]:
            h_bin[idx] = 1.0
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    
    # Local search on binary: swap a 1 and a 0
    best_c = c_bin
    best_h = h_bin.copy()
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    
    while time.time() < deadline:
        improved = False
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                ii, jj = int(i), int(j)
                best_h[ii] = 0.0
                best_h[jj] = 1.0
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-12:
                    best_c = c
                    improved = True
                else:
                    best_h[ii] = 1.0
                    best_h[jj] = 0.0
                    best_h = project_feasible(best_h, n)
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B aggressive (up to 180s)
    t1 = min(t0 + 180, deadline - 600)
    if t1 > time.time() + 10:
        best_h, best_c = lbfgs_optimize(best_h, n_init, t1, max_iters=400)

    # Phase 2: SLSQP precision (up to 80s)
    if time.time() < deadline - 450:
        t2 = min(time.time() + 80, deadline - 450)
        for a in [100000, 500000, 2000000]:
            if time.time() > t2 - 5:
                break
            t_alloc = min(30, t2 - time.time())
            if t_alloc < 5:
                break
            h2, c2 = slsqp_optimize(best_h, n_init, time.time() + t_alloc, alpha=a, maxiter=2000)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Projected gradient (up to 80s)
    if time.time() < deadline - 350:
        t3 = min(time.time() + 80, deadline - 350)
        h3, c3 = projected_grad(best_h, n_init, t3, alpha=100000, max_iters=10000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Pair search - main local optimization (up to 250s)
    if time.time() < deadline - 200:
        t4 = min(time.time() + 250, deadline - 200)
        best_h, best_c = pair_search(best_h, n_init, t4,
                                     deltas=[0.005, 0.001, 0.0005, 0.0001, 0.00002, 0.000005],
                                     radius=50)

    # Phase 5: Multi-resolution (up to 150s)
    if time.time() < deadline - 100:
        for n_try in [250, 350, 400, 500]:
            rem = deadline - time.time() - 15
            if rem < 35:
                break
            h_interp = interpolate_h(best_h, len(best_h), n_try)
            alloc = min(35, rem * 0.35)
            if alloc < 12:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc, max_iters=200)
            if time.time() < deadline - 10:
                h_t, c_t2 = pair_search(h_t, n_try, time.time() + min(15, deadline - time.time() - 5),
                                        deltas=[0.001, 0.0001, 0.00002], radius=30)
                if c_t2 < c_t:
                    c_t = c_t2
            if c_t < best_c:
                best_c = c_t
                best_h = h_t

    # Phase 6: Perturbation + re-optimize (up to 80s)
    if time.time() < deadline - 30:
        t6 = min(time.time() + 80, deadline - 30)
        nh = len(best_h)
        best_h, best_c = perturb_reoptimize(best_h, nh, t6, rng, n_perturb=5, scale=0.03)

    # Phase 7: Binary local search attempt (up to 40s)
    if time.time() < deadline - 15:
        nh = len(best_h)
        t7 = min(time.time() + 30, deadline - 10)
        h_b, c_b = binary_local_search(best_h, nh, t7)
        if c_b < best_c:
            # Unfreeze: allow small continuous perturbations around binary
            best_c = c_b
            best_h = h_b
            if time.time() < deadline - 5:
                best_h, best_c = pair_search(best_h, nh, deadline - 3,
                                             deltas=[0.01, 0.005, 0.001, 0.0001], radius=40)

    # Phase 8: Final fine pair search polish (remaining time)
    if time.time() < deadline - 2:
        nh = len(best_h)
        best_h, best_c = pair_search(best_h, nh, deadline - 1,
                                     deltas=[0.00005, 0.00001, 0.000001],
                                     radius=40)

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
