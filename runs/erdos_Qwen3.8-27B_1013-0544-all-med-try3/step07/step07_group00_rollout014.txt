def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def equalize_active(h, n, active_idx):
    """Find direction that equalizes active shift values while preserving sum.
    
    We want d such that:
    - sum(d) = 0
    - g_k · d = constant for all k in active (where g_k is gradient of c_k)
    
    This is: min ||G d - 1||² s.t. 1' d = 0
    where G is the matrix of active gradients.
    """
    m = len(active_idx)
    G = np.zeros((m, n))
    for i, idx in enumerate(active_idx):
        G[i] = shift_gradient(h, n, idx)
    
    # Solve: min ||G d - 1||² s.t. 1' d = 0
    # KKT: [2G'G + λ11', G'; G, 0] [d; λ] = [2G'; 0]
    # Or use Lagrange: G'd = (1/2m) * 1' * 1 ... 
    
    # Simpler: min ||Gd||² - 2*1'Gd s.t. 1'd = 0
    # => 2G'G d - 2G'1 + λ*1 = 0
    # => 1'd = 0
    
    A = 2 * G.T @ G
    b = 2 * G.T @ np.ones(m)
    
    # Solve [A 1; 1' 0] [d; λ] = [b; 0]
    K = np.zeros((n+1, n+1))
    K[:n, :n] = A
    K[:n, n] = 1.0
    K[n, :n] = 1.0
    rhs = np.zeros(n+1)
    rhs[:n] = b
    
    try:
        sol = np.linalg.solve(K, rhs)
        d = sol[:n]
    except np.linalg.LinAlgError:
        # Fallback: project gradient of mean
        g_mean = G.mean(axis=0)
        d = g_mean - g_mean.mean()
    
    # Clip to stay in [0,1]
    max_step = 0.01
    if np.max(np.abs(d)) > 0:
        d = d / np.max(np.abs(d)) * max_step
    d -= d.mean()  # re-center
    
    return d

def run_lbfgs(h, n, deadline, alphas, mu=1.0, maxiter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-18,
                                  'gtol': 1e-12, 'maxfun': maxiter * 8, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])

def equalize_direction(h, n, active_idx):
    """Compute direction to equalize active shift correlation values."""
    m = len(active_idx)
    G = np.zeros((m, n))
    for i, idx in enumerate(active_idx):
        G[i] = shift_gradient(h, n, idx)
    # Direction: minimize variance of Gd, s.t. sum(d)=0
    # Gradient of variance: 2*G'(Gd - mean(Gd)) projected to sum-0 space
    Gd = G @ h  # not used, just for structure
    g_mean = G.mean(axis=0)
    # The direction that most reduces the spread:
    # We want G*d to be as "flat" as possible while sum(d)=0
    # Use: d = -(G'G - G'11'G/m) * (G'1/m - G'G * 1/n) ... complex
    # Simpler: d = -g_mean + projected to reduce individual shifts
    # Use the "anti-spread" direction: move mass from high-c to low-c regions
    vals = compute_c5_vec(h, n)[active_idx]
    # Direction: for each active shift, its gradient points to increase c_k
    # We want to DECREASE the max, so move opposite to weighted gradient
    # Weight by (c_k - mean) to focus on the top ones
    weights = vals - vals.mean()
    if weights.max() - weights.min() < 1e-15:
        return np.zeros(n)

        g_i = shift_gradient(h, n, idx)
        if weights[i] > 0:
            d -= weights[i] * g_i  # decrease high-c shifts
        else:
            d += abs(weights[i]) * g_i  # increase low-c shifts
    d -= d.mean()  # preserve sum
    # Normalize
    norm = np.linalg.norm(d)
    if norm > 1e-15:
        d = d / norm
    return d

                        c_try = compute_c5(best_h, n)
                        if c_try < best_c - 1e-15:
                            best_c = c_try
                            found = True
                        else:
                            best_h[i] += step
                            best_h[j] -= step
                if found:
                    break
            if not found:
                break
    return best_h, best_c

def perturb_restart(h, n, deadline, rng, magnitude=0.05):
    """Perturb and re-optimize to escape local optima."""
    h_pert = h + rng.normal(0, magnitude, n)
    h_pert = project_feasible(h_pert, n)
    c_pert = compute_c5(h_pert, n)
    
    if c_pert > compute_c5(h, n):
        return h, compute_c5(h, n)  # revert
    
    # Quick L-BFGS on perturbed
    if time.time() < deadline - 30:
        t = min(time.time() + 30, deadline - 30)
        h_pert, c_pert = run_lbfgs(h_pert, n, t, alphas=[5000, 20000], mu=1.0, maxiter=1500)
    
    return h_pert, c_pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    # Quick polish at original resolution
    best_h_orig = project_feasible(h_init, n_orig)
    best_c_orig = compute_c5(best_h_orig, n_orig)
    
    if time.time() < deadline - 900:
        t1 = min(time.time() + 50, deadline - 900)
        best_h_orig, best_c_orig = run_lbfgs(best_h_orig, n_orig, t1,
                                               alphas=[1000, 5000, 20000], mu=1.0, maxiter=2000)
    
    # Upsample to n=1000
    n = 1000
    x_orig = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h = np.interp(x_new, x_orig, best_h_orig)
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    
    # Phase 1: L-BFGS at n=1000 (escalating alpha)
    if time.time() < deadline - 550:
        t = min(time.time() + 250, deadline - 550)
        h, c = run_lbfgs(h, n, t, alphas=[300, 1000, 5000, 20000], mu=1.0, maxiter=3000)
    
    # Phase 2: Active-set focused optimization
    if time.time() < deadline - 400:
        t = min(time.time() + 150, deadline - 400)
        h, c = active_set_opt(h, n, t, top_k=6, alphas=[20000, 100000, 300000], mu=0.1)
    
    # Phase 3: Equalization iterations
    if time.time() < deadline - 280:
        t = min(time.time() + 80, deadline - 280)
        h, c = equalize_iterate(h, n, t, max_outer=50)
    
    # Phase 4: More L-BFGS with high alpha
    if time.time() < deadline - 200:
        t = min(time.time() + 100, deadline - 200)
        h, c = run_lbfgs(h, n, t, alphas=[50000, 200000, 500000], mu=0.05, maxiter=4000)
    
    # Phase 5: Active-set again
    if time.time() < deadline - 130:
        t = min(time.time() + 60, deadline - 130)
        h, c = active_set_opt(h, n, t, top_k=5, alphas=[100000, 500000], mu=0.05)
    
    # Phase 6: Equalization
    if time.time() < deadline - 80:
        t = min(time.time() + 40, deadline - 80)
        h, c = equalize_iterate(h, n, t, max_outer=60)
    
    # Phase 7: Gradient pair descent
    if time.time() < deadline - 40:
        t = min(time.time() + 30, deadline - 40)
        h, c = gradient_pair_descent(h, n, t)
    
    # Phase 8: Final polish L-BFGS
    if time.time() < deadline - 15:
        t = min(time.time() + 10, deadline - 15)
        h, c = run_lbfgs(h, n, t, alphas=[500000], mu=0.01, maxiter=2000)
    
    # Phase 9: Try perturbation restart if time allows
    if time.time() < deadline - 40:
        t = min(time.time() + 30, deadline - 40)
        h_pert, c_pert = perturb_restart(h, n, t, rng, magnitude=0.03)
        if c_pert < c:
            h, c = h_pert, c_pert
            # Quick polish
            if time.time() < deadline - 10:
                t2 = min(time.time() + 5, deadline - 10)
                h, c = run_lbfgs(h, n, t2, alphas=[200000], mu=0.05, maxiter=1000)
    
    # Check at n=500 as well
    c1000 = c
    h1000 = h.copy()
    
    n_low = 500
    x_h = np.linspace(0, 2, n)
    x_l = np.linspace(0, 2, n_low)
    h_low = np.interp(x_l, x_h, h1000)
    h_low = project_feasible(h_low, n_low)
    c_low = compute_c5(h_low, n_low)
    
    # Also check original
    c_orig_check = compute_c5(project_feasible(best_h_orig, n_orig), n_orig)
    
    # Pick best
    if c_low < c1000 and c_low < c_orig_check:
        best_h, best_c, n_final = h_low, c_low, n_low
    elif c_orig_check < c1000:
        best_h, best_c, n_final = project_feasible(best_h_orig, n_orig), c_orig_check, n_orig
    else:
        best_h, best_c, n_final = h1000, c1000, n
    
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
