import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    ac = alpha * c
    amax = ac.max()
    if amax > 500:
        val = c.max() + (c.max() + np.log(np.sum(np.exp(ac - amax)))) / alpha
    else:
        val = np.log(np.sum(np.exp(ac))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    dx = 2.0 / n
    ac = alpha * c
    amax = ac.max()
    e = np.exp(ac - amax)
    w = e / e.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def pnorm_obj(x, n, p, mu):
    c = compute_c5_vec(x, n)
    cp = np.maximum(c, 1e-30) ** p
    sp = cp.sum()
    val = sp ** (1.0 / p)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def pnorm_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    c_safe = np.maximum(c, 1e-30)
    dx = 2.0 / n
    cp = c_safe ** p
    sp = cp.sum()
    fact = sp ** (1.0 / p - 1.0)
    wc = fact * c_safe ** (p - 1)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if wc[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = wc[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 10.0), (500, 50.0), (2000, 200.0),
        (10000, 1000.0), (50000, 5000.0),
        (200000, 20000.0), (1000000, 100000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time() - 3
        max_it = min(max_iters, max(20, int(remaining * 8)))
        if max_it < 15:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_it, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_it * 20,
                                   'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def pnorm_optimize(h, n, deadline, p=100, max_iters=200):

                                   'gtol': 1e-12, 'maxfun': max_it * 20})
            h_new = project_feasible(res.x, n)

            pass
    return best_h, best_c

def basin_hopping(h, n, deadline, rng, max_rounds=4):

    return best_h, best_c

def gradient_pair_search(h, n, deadline, rng, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in [200, 1000, 5000, 20000]:
        for step in step_sizes:
            while time.time() < deadline:
                improved = False
                g = smooth_max_grad(h, n, alpha, 0)
                g_norm = np.linalg.norm(g)
                if g_norm < 1e-15:
                    break
                g = g / g_norm
                n_try = min(n, 150)
                idx = np.random.choice(n, size=n_try, replace=False)
                for ii in range(n_try):
                    if time.time() > deadline:
                        return best_h, best_c
                    i = idx[ii]
                    for j in rng.choice(n, size=4, replace=False):
                        j = int(j)
                        if j == i:
                            continue
                        move = -step * (g[i] - g[j])
                        if abs(move) < 1e-12:
                            continue
                        hi_n = h[i] + move
                        hj_n = h[j] - move
                        if hi_n >= 0 and hi_n <= 1 and hj_n >= 0 and hj_n <= 1:
                            h[i] = hi_n
                            h[j] = hj_n
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i] -= move
                                h[j] += move
                if not improved:
                    break
    return best_h, best_c

def structured_local_search(h, n, deadline, rng):
    """Try structured moves: swap adjacent, transfer to neighbors"""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        changed = True
        while changed and time.time() < deadline:
            changed = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for delta in [1, -1, 2, -2, 3, -3]:
                    j = i + delta
                    if j < 0 or j >= n:
                        continue
                    for direction in [(1, -1), (-1, 1)]:
                        hi_n = h[i] + direction[0] * d
                        hj_n = h[j] + direction[1] * d
                        if 0 <= hi_n <= 1 and 0 <= hj_n <= 1:

                            h[j] = hj_n
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h.copy()
                                changed = True
                            else:
                                h[i] -= direction[0] * d
                                h[j] -= direction[1] * d
    return best_h, best_c

def upscale(h, n, n_new):

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h, kind='cubic')
    return project_feasible(h_new, n_new)
