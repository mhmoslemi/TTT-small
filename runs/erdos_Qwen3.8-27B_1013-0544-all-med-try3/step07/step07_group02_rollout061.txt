def _lag_grad(h, n, shift):
    """Gradient of dx * cor(h, 1-h) at the given shift."""
    dx = 2.0 / n
    m = abs(shift)
    if m >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif shift > 0:
        # f = dx * sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        g[:n-m] -= h[m:]       # -h[i+m] for i = 0..n-1-m, so g[i] -= h[i+m] for i in 0..n-1-m

def compute_c5(h, n):

def compute_all_lags(h, n):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, shift):
    dx = 2.0 / n
    m = abs(shift)
    if m >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift == 0:
        g[:] = (1.0 - 2.0 * h)
    elif shift > 0:
        g[:n-m] -= h[m:]
        g[m:] += (1.0 - h[:n-m])
    else:
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, shift):
    dx = 2.0 / n
    m = abs(shift)
    if m >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift == 0:
        g[:] = (1.0 - 2.0 * h)
    elif shift > 0:
        g[:n - m] -= h[m:]
        g[m:] += (1.0 - h[:n - m])
    else:
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g * dx

def smooth_max_grad(h, n, alpha):
    c = compute_all_lags(h, n)
    mval = c.max()
    shifted = alpha * (c - mval)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w /= w.sum()
    n_active = np.sum(w > 1e-12)
    if n_active == 0:
        return np.zeros(n)
    active_k = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    for k in active_k:
        lag = k - (n - 1)
        grad += w[k] * _lag_grad(h, n, lag)
    return grad

def grad_descent(h, n, alpha, lr, deadline, max_iter=500000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    momentum = 0.85
    stall = 0

        if time.time() > deadline:
            break
        g = smooth_max_grad(h_curr, n, alpha)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:

            best_h = h_new.copy()
            stall = 0

        h_curr = h_new

            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break
        if it % 10000 == 9999:
            lr = max(0.01, lr * 0.8)
    return best_h, best_c

def _slsqp_obj(x, n, top_k, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    active = c[top_k]
    mval = active.max()
    shifted = alpha * (active - mval)
    mx = shifted.max()
    if mx > 500:
        return float(mval + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(mval + np.log(np.sum(np.exp(shifted))) / alpha)

def _slsqp_con_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _slsqp_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, top_k, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, top_k, alpha),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _compute_move_coefficients(h, n, m):

    coeff[i] < 0 means moving mass from i to i+m reduces f_m.
    coeff[i] > 0 means moving mass from i+m to i reduces f_m."""

    coeff = np.zeros(NUm)

    if m <= n - 1 - m:
        # p = i+m is valid when 0 <= i+m <= n-1-m
        # Already handled above
        pass

            h = project_feasible(h, n)
            c_test = compute_c5(h, n)
            if c_test < best_c_delta:
                best_c_delta = c_test
                best_delta = mid
                lo = mid
            else:
                hi = mid
            h[i] += direction * mid
            h[j] -= direction * mid
            h = project_feasible(h, n)
        
        if best_delta > 0:
            h[i] -= direction * best_delta
            h[j] += direction * best_delta
            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)
            c_test = compute_c5(h, n)
            if c_test < best_c - 1e-16:
                best_c = c_test
                best_h = h.copy()
            else:
                h[i] += direction * best_delta
                h[j] -= direction * best_delta
                h = project_feasible(h, n)
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        h = project_feasible(h_init, n_init)
        n = n_init
    else:
        h = np.full(n_init, 0.5)
        n = n_init
    
    # Phase 1: Quick gradient descent to ensure near local optimum (80s)
    phase1_end = t0 + 80
    for alpha in [20, 100, 500]:
        if time.time() > phase1_end:
            break
        sub_deadline = min(phase1_end, time.time() + 25)
        h, c = grad_descent(h, n, alpha, lr=0.3, deadline=sub_deadline)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: SLSQP on top-5 (80s)
    phase2_end = t0 + 160
    if time.time() < phase2_end:
        c_all = compute_all_lags(h, n)
        top5 = np.argsort(c_all)[-5:]
        h2 = slsqp_optimize(h, n, top5, 50000, maxiter=1500)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
        h = h2
    
    # Phase 3: Aggressive local search (500s) - main workhorse
    phase3_end = t0 + 660
    if time.time() < phase3_end:
        h3, c3 = aggressive_local_search(h, n, phase3_end)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    
    # Phase 4: SLSQP polish on top-3
    if time.time() < deadline - 200:
        c_all = compute_all_lags(best_h, n)
        top3 = np.argsort(c_all)[-3:]
        h4 = slsqp_optimize(best_h, n, top3, 100000, maxiter=2000)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
    
    # Phase 5: Fine local search (150s)
    phase5_end = t0 + 900
    if time.time() < phase5_end:
        h5, c5 = fine_local_search(best_h, n, phase5_end)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
    
    # Phase 6: Try n=1000
    if time.time() < deadline - 60:
        n_new = 1000
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_new)
        
        # Quick gradient descent
        sub_dl = min(deadline - 30, time.time() + 40)
        h_new, c_new = grad_descent(h_new, n_new, 200, lr=0.2, deadline=sub_dl)
        
        # Local search
        if c_new < best_c and time.time() < deadline - 10:
            sub_dl2 = min(deadline - 5, time.time() + 50)
            h_new, c_new = aggressive_local_search(h_new, n_new, sub_dl2)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    # Phase 7: Final fine search on best solution
    if time.time() < deadline - 2:
        n = len(best_h)
        sub_dl = deadline - 1
        h_fin, c_fin = fine_local_search(best_h, n, sub_dl)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
