def compute_swap_delta(p, q, h, n):
    """Returns delta vector of length 2n-1"""
    delta = np.zeros(2*n-1)
    # Only lags that are affected
    affected_lags = set()
    for j in [p, q]:
        # j as direct index: affects lags where j is in valid range
        # max(0,-l) <= j <= min(n-1,n-1-l)
        # -j <= l <= n-1-j
        for l in range(max(-j, -(n-1)), min(n-1-j, n-1)+1):
            affected_lags.add(l)
        # j-l' = j as the "other" index: l' = j - j' for some valid j'
        # Actually, j appears as the "top" index (j+l) when l = j - j' for valid j'
        # This means for the lag l, the pair (j' , j'+l) has j'+l = j, so l = j-j'
        # j' ranges over [max(0,-l), min(n-1,n-1-l)]
        # For a given j, l can be anything from -(n-1) to n-1 such that j' = j-l is valid
        # j-l >= max(0,-l) and j-l <= min(n-1, n-1-l)
        # j >= max(0,-l)+l and j <= min(n-1,n-1-l)+l
        # For l >= 0: j >= 0+l = l, j <= n-1. So l in [max(0, j-(n-1)), min(j, n-1)]
        # For l < 0: j >= -l+l = 0, j <= n-1-l. So j+l <= n-1, j <= n-1 (always). l in [j-n+1, -1]
        # Combined: l in [max(-(n-1), j-n+1), min(n-1, j)]
        for l in range(max(-(n-1), j-(n-1)), min(n-1, j)+1):
            affected_lags.add(l)
    
    for l in affected_lags:
        m = l + n - 1
        j_min = max(0, -l)
        j_max = min(n-1, n-1-l)
        # Check each affected j
        total_delta = 0.0
        counted_j = set()
        for j_candidate in [p, p-l, q, q-l]:
            if j_candidate in counted_j:
                continue
            if j_min <= j_candidate <= j_max:
                counted_j.add(j_candidate)
                top = j_candidate + l
                if 0 <= top < n:
                    old = h[top] * (1.0 - h[j_candidate])
                    # New h
                    new_top = h[top]
                    if top == p: new_top = 1.0
                    if top == q: new_top = 0.0
                    new_bot = h[j_candidate]
                    if j_candidate == p: new_bot = 1.0
                    if j_candidate == q: new_bot = 0.0
                    new = new_top * (1.0 - new_bot)
                    total_delta += new - old
        delta[m] = total_delta
    return delta
