_LASTC = None


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(np.max(compute_all_lags(h, n)))


def _fix_sum_exact(h, n):
    target = 0.5 * n
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h = h.astype(float, copy=False)
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h


def project_feasible(y, n):
    target = 0.5 * n
    h = np.array(y, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) < 1e-13:
        return _fix_sum_exact(h, n)

    lo = -1.0
    hi = 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(h - lam, 0.0, 1.0)
    return _fix_sum_exact(h, n)


def fix_feasible_light(y, n):
    h = np.array(y, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n

    for _ in range(120):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if not np.any(free):
            break

            room = 1.0 - h
            tot = float(np.sum(room[free]))

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return np.array(h_old, dtype=float).copy()
    if n_old == 1:
        return np.full(n_new, float(h_old[0]), dtype=float)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, dtype=float)


def random_smooth_h(n, rng):

    g = np.zeros(n, dtype=float)

        g += a * np.sin(math.pi * k * x)
        g += b * np.cos(math.pi * k * x * 2.0)
    h = 0.5 + g

def select_active(c, max_count, tie_tol=1e-9):
    nlag = len(c)
    if nlag == 0:
        return np.empty(0, dtype=int)
    max_count = int(max(1, min(int(max_count), nlag)))
    m = float(np.max(c))
    thr = m - max(float(tie_tol), 10.0 * np.finfo(float).eps * max(1.0, abs(m)))
    order = np.argsort(-c)
    top = order[:max_count]
    near = np.flatnonzero(c >= thr)
    mask = np.zeros(nlag, dtype=bool)
    mask[top] = True
    mask[near] = True
    idx = np.flatnonzero(mask)
    cap = 2 * max_count
    if idx.size > cap:
        idx = order[:cap]
    return idx.astype(int, copy=False)


def _lag_grad_no_dx(h, n, lag):
    grad = np.zeros(n, dtype=float)
    m = abs(int(lag))
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += 1.0 - h[:n - m]
    else:
        grad[:n - m] += 1.0 - h[m:]
        grad[m:] -= h[:n - m]
    return grad


def _score_c(x):
    global _LASTH, _LASTC
    x = np.asarray(x, dtype=float)
    n = int(x.shape[0])
    if _LASTH is not None and _LASTH.shape == x.shape and np.array_equal(_LASTH, x):
        return _LASTC
    c = compute_all_lags(x, n)
    _LASTH = x.copy()
    _LASTC = c
    return c


def _score_obj(x):
    c = _score_c(x)
    top = _TOPG
    if top is None or len(top) == 0:
        return float(np.max(c))
    ct = c[top]
    mc = float(np.max(ct))
    s = _ALPHAG * (ct - mc)
    se = np.exp(s)
    logsum = float(np.log(float(np.sum(se))))
    val = mc + logsum / float(_ALPHAG)
    if not np.isfinite(val):
        val = mc
    return float(val)


def _score_grad(x):
    c = _score_c(x)
    top = _TOPG
    x = np.asarray(x, dtype=float)
    n = int(x.shape[0])
    grad = np.zeros(n, dtype=float)
    if top is None or len(top) == 0:
        return grad
    ct = c[top]
    mc = float(np.max(ct))
    s = _ALPHAG * (ct - mc)
    se = np.exp(s)
    w = se / float(np.sum(se))
    dx = _DXG
    for idx, wi in zip(top, w):
        if wi < 1e-14:
            continue
        lag = int(idx) - (n - 1)
        m = abs(lag)
        if m == 0:
            grad += wi * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[:n - m] -= wi * dx * x[m:]
            grad[m:] += wi * dx * (1.0 - x[:n - m])
        else:
            grad[:n - m] += wi * dx * (1.0 - x[m:])
            grad[m:] -= wi * dx * x[:n - m]
    return grad


def _sum_con(x):
    return float(np.sum(x) - _TARGETG)


def _sum_con_jac(x):
    return np.ones(_NG, dtype=float)


def _score_obj_pen(x):
    val = _score_obj(x)
    s = float(np.sum(x) - _TARGETG)
    val += 0.5 * _RHO * s * s + _MULTI * s
    return float(val)


def _score_grad_pen(x):
    g = _score_grad(x)
    s = float(np.sum(x) - _TARGETG)
    g = g + (_RHO * s + _MULTI) * np.ones_like(g)
    return g


def _slsqp_run(h0, n, alpha, maxiter, top_count, ftol):
    global _NG, _DXG, _TARGETG, _ALPHAG, _TOPG, _RHO, _MULTI, _LASTH

    h = fix_feasible_light(h0, n)
    if not np.all(np.isfinite(h)):
        h = project_feasible(h0, n)

    _NG = int(n)
    _TARGETG = 0.5 * n
    _DXG = 2.0 / n
    _ALPHAG = float(max(1.0, alpha))
    _RHO = 0.0
    _MULTI = 0.0
    _LASTH = None

    c = compute_all_lags(h, n)
    _TOPG = select_active(c, int(top_count), 1e-9)

    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": _sum_con, "jac": _sum_con_jac}]

    try:
        res = minimize(
            _score_obj,
            h,
            jac=_score_grad,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(max(20, maxiter)), "ftol": float(ftol), "disp": False},
        )
        x = np.asarray(res.x, dtype=float)
    except Exception:
        return h

    if x.shape[0] != n:
        return h
    x = fix_feasible_light(x, n)
    if not np.all(np.isfinite(x)):
        x = project_feasible(h0, n)
    return x


def _lbfgs_run(h0, n, alpha, maxiter, top_count, ftol):
    global _NG, _DXG, _TARGETG, _ALPHAG, _TOPG, _RHO, _MULTI, _LASTH

    h = fix_feasible_light(h0, n)
    if not np.all(np.isfinite(h)):
        h = project_feasible(h0, n)

    _NG = int(n)
    _TARGETG = 0.5 * n
    _DXG = 2.0 / n
    _ALPHAG = float(max(1.0, alpha))
    _RHO = 10.0
    _MULTI = 0.0
    _LASTH = None

    c = compute_all_lags(h, n)
    _TOPG = select_active(c, int(top_count), 1e-9)

    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(
            _score_obj_pen,
            h,
            jac=_score_grad_pen,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": int(max(20, maxiter)), "ftol": float(ftol), "gtol": 1e-14, "maxls": 20},
        )
        x = np.asarray(res.x, dtype=float)
    except Exception:
        return h

    if x.shape[0] != n:
        return h
    x = fix_feasible_light(x, n)
    if not np.all(np.isfinite(x)):
        x = project_feasible(h0, n)
    return x


def _outer_single_step(h, n, trust):
    if cp is None:
        return None
    solvers = []
    if hasattr(cp, "GLOP"):
        solvers.append(cp.GLOP)
    if hasattr(cp, "CBC"):
        solvers.append(cp.CBC)
    if not solvers:
        return None

    c = compute_all_lags(h, n)
    dx = 2.0 / n
    hv = cp.Variable(n)
    tv = cp.Variable()

    cons = [
        cp.sum(hv) == 0.5 * n,
        hv >= 0,
        hv <= 1,
    ]
    try:
        cons.append(cp.abs(hv - h) <= float(trust))
    except Exception:
        cons.append(hv - trust <= h)
        cons.append(h <= hv + trust)

    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = _lag_grad_no_dx(h, n, lag)
        cons.append(c[k] + dx * (g @ (hv - h)) <= tv)

    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    x = hv.value
    if x is None:
        return None
    return fix_feasible_light(x, n)


def _outer_refine(h0, n, end_time, max_iters=4):
    h = fix_feasible_light(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur = h.copy()
    trust = 0.18

    for _ in range(max_iters):
        if time.time() > end_time - 3:
            break
        h_new = _outer_single_step(cur, n, trust)
        if h_new is None:
            break
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = float(np.max(np.abs(h_new - cur)))
        cur = h_new
        if step < 1e-9:
            break
        if c_new > best_c + 0.03:
            break
        trust = max(0.015, trust * 0.72)

    return best_h, best_c


def _weighted_active_grad(h, n, alpha, max_count, tie_tol=1e-9):
    c = compute_all_lags(h, n)
    top = select_active(c, int(max_count), tie_tol)
    if len(top) == 0:
        return np.zeros(n, dtype=float)
    ct = c[top]
    mc = float(np.max(ct))
    alpha = float(max(1.0, alpha))
    s = alpha * (ct - mc)
    se = np.exp(s)
    w = se / float(np.sum(se))
    dx = 2.0 / n
    h = np.asarray(h, dtype=float)
    grad = np.zeros(n, dtype=float)
    for idx, wi in zip(top, w):
        if wi < 1e-14:
            continue
        lag = int(idx) - (n - 1)
        m = abs(lag)
        if m == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - m] -= wi * dx * h[m:]
            grad[m:] += wi * dx * (1.0 - h[:n - m])
        else:
            grad[:n - m] += wi * dx * (1.0 - h[m:])
            grad[m:] -= wi * dx * h[:n - m]
    return grad


def _feasible_direction(h, d):
    h = np.asarray(h, dtype=float)
    d = np.asarray(d, dtype=float).copy()
    if d.shape[0] == 0:
        return d
    for _ in range(10):
        d[(h <= 1e-12) & (d < 0.0)] = 0.0
        d[(h >= 1.0 - 1e-12) & (d > 0.0)] = 0.0
        d -= float(np.mean(d))
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.any(free):
        d[free] -= float(np.sum(d)) / float(np.sum(free))
    d[(h <= 1e-12) & (d < 0.0)] = 0.0
    d[(h >= 1.0 - 1e-12) & (d > 0.0)] = 0.0
    return d


def _line_search_polish(h0, n, end_time, max_iters=6):
    h = fix_feasible_light(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if n < 2 or time.time() > end_time - 8:
        return best_h, best_c

    fracs = [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.025, 0.0125, 0.005]

    for _ in range(max_iters):
        if time.time() > end_time - 3:
            break
        g = _weighted_active_grad(h, n, 5000.0, 128, 1e-9)
        if not np.all(np.isfinite(g)):
            break
        if float(np.max(np.abs(g))) < 1e-12:
            break
        d = -g
        d = _feasible_direction(h, d)
        norm = float(np.max(np.abs(d)))
        if norm < 1e-13:
            break
        d = d / norm

        lim = 0.12
        if np.any(d < 0.0):
            lim = min(lim, float(np.min(h[d < 0.0] / (-d[d < 0.0]))))
        if np.any(d > 0.0):
            lim = min(lim, float(np.min((1.0 - h[d > 0.0]) / d[d > 0.0])))
        if not np.isfinite(lim) or lim <= 1e-10:
            break

        improved = False
        best_lam = 0.0
        for f in fracs:
            lam = f * lim
            if lam <= 0.0:
                continue
            hn = fix_feasible_light(h + lam * d, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-14:
                best_c = cn
                best_h = hn
                best_lam = lam
                improved = True
        if not improved:
            break
        h = best_h
        _ = best_lam

    return best_h, best_c


def _choose_stages(quick, n):
    if quick:
        return [
            (20, 32, 60, 1e-11),
            (200, 128, 100, 1e-12),
            (2000, 512, 150, 1e-13),
        ]
    if n <= 120:
        return [
            (10, 32, 150, 1e-11),
            (50, 64, 250, 1e-11),
            (200, 128, 350, 1e-12),
            (1000, 256, 500, 1e-12),
            (5000, 512, 600, 1e-13),
            (20000, 1024, 700, 1e-14),
        ]
    if n <= 300:
        return [
            (10, 32, 80, 1e-11),
            (50, 80, 120, 1e-11),
            (200, 160, 150, 1e-12),
            (1000, 320, 200, 1e-12),
            (5000, 640, 250, 1e-13),
            (20000, 1200, 300, 1e-14),
        ]
    return [
        (10, 32, 60, 1e-11),
        (100, 96, 100, 1e-11),
        (500, 256, 150, 1e-12),
        (3000, 600, 200, 1e-13),
        (10000, 1000, 250, 1e-14),
    ]


def optimize_candidate(h0, n, deadline, rng=None, quick=False):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if n > 1:
        hs = 0.5 * (h + h[::-1])
        hs = fix_feasible_light(hs, n)
        cs = compute_c5(hs, n)
        if cs < best_c - 1e-15:
            best_h = hs
            best_c = cs

    if (not quick) and cp is not None and n <= 330 and time.time() < deadline - 30.0:
        end = min(deadline - 8.0, time.time() + 25.0)
        if end > time.time() + 8.0:
            hoo, coo = _outer_refine(best_h, n, end, max_iters=3 if n > 220 else 4)
            if coo < best_c - 1e-14:
                best_c = coo
                best_h = hoo

    stages = _choose_stages(bool(quick), int(n))
    for alpha, top_req, mi_req, ftol in stages:
        if time.time() > deadline - 12.0:
            break
        rem = deadline - time.time()
        if rem < 18.0:
            break
        mi = int(mi_req)
        if n > 500:
            mi = max(30, mi // 3)
        elif n > 350:
            mi = max(35, mi // 2)
        if rem < 120.0:
            mi = min(mi, 180)
        if rem < 60.0:
            mi = min(mi, 80)

        top_count = int(min(top_req, 2 * n - 1, 1500))
        h_new = _slsqp_run(best_h, n, alpha, mi, top_count, ftol)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
            if time.time() < deadline - 18.0:
                h_new2 = _slsqp_run(best_h, n, alpha, mi, top_count, ftol)
                c_new2 = compute_c5(h_new2, n)
                if c_new2 < best_c - 1e-14:
                    best_c = c_new2
                    best_h = h_new2

    if (not quick) and n <= 500 and time.time() < deadline - 25.0:
        hlf = _lbfgs_run(best_h, n, 3000.0, 150, min(600, 2 * n - 1), 1e-14)
        clf = compute_c5(hlf, n)
        if clf < best_c - 1e-14:
            best_c = clf
            best_h = hlf

    if n <= 450 and time.time() < deadline - 10.0:
        hl, cl = _line_search_polish(best_h, n, deadline - 2.0, max_iters=3 if quick else 6)
        if cl < best_c - 1e-14:
            best_c = cl
            best_h = hl

    if n > 1:
        hs = 0.5 * (best_h + best_h[::-1])
        hs = fix_feasible_light(hs, n)
        cs = compute_c5(hs, n)
        if cs < best_c - 1e-15:
            best_h = hs
            best_c = cs

    best_h = fix_feasible_light(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def _final_polish(h0, n, deadline):
    h = fix_feasible_light(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if n > 1:
        hs = 0.5 * (h + h[::-1])
        hs = fix_feasible_light(hs, n)
        cs = compute_c5(hs, n)
        if cs < best_c - 1e-15:
            best_h = hs
            best_c = cs

    if n <= 450 and time.time() < deadline - 20.0:
        hl, cl = _line_search_polish(best_h, n, min(deadline - 2.0, time.time() + 35.0), max_iters=8)
        if cl < best_c - 1e-14:
            best_h = hl
            best_c = cl

    final_stages = [
        (2000, 600, 220),
        (8000, 1000, 280),
        (30000, 2000, 350),
        (100000, 3000, 450),
    ]
    for alpha, top_req, mi_request in final_stages:
        if time.time() > deadline - 8.0:
            break
        rem = deadline - time.time()
        if rem < 12.0:
            break
        mi = int(min(mi_request, max(40, int(rem * 1.2))))
        top = int(min(top_req, 2 * n - 1, 3000))
        h_new = _slsqp_run(best_h, n, alpha, mi, top, 1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new

    if n <= 450 and time.time() < deadline - 5.0:
        hl, cl = _line_search_polish(best_h, n, deadline - 0.5, max_iters=6)
        if cl < best_c - 1e-14:
            best_h = hl
            best_c = cl

    if n > 1:
        hs = 0.5 * (best_h + best_h[::-1])
        hs = fix_feasible_light(hs, n)
        cs = compute_c5(hs, n)
        if cs < best_c - 1e-15:
            best_h = hs
            best_c = cs

    best_h = fix_feasible_light(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(60.0, float(budget_s))
    deadline = t0 + budget - 3.0
    if deadline < t0 + 30.0:
        deadline = t0 + max(20.0, budget - 2.0)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float("inf")

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        if parent.ndim != 1:
            parent = parent.ravel()
        if parent.size >= 4:
            hp = project_feasible(parent, parent.size)
            cpv = compute_c5(hp, parent.size)
            best_h = hp
            best_c = cpv
            if parent.size > 1:
                hs = 0.5 * (hp + hp[::-1])
                hs = fix_feasible_light(hs, parent.size)
                cs = compute_c5(hs, parent.size)
                if cs < best_c:
                    best_h = hs
                    best_c = cs
    except Exception:
        best_h = None

    if best_h is None:
        n0 = 240
        best_h = project_feasible(np.full(n0, 0.5, dtype=float), n0)
        best_c = compute_c5(best_h, n0)
        for _ in range(8):
            hr = random_smooth_h(n0, rng)
            cr = compute_c5(hr, n0)
            if cr < best_c:
                best_h = hr
                best_c = cr

    primary_time = max(30.0, min(360.0, (deadline - t0) - 180.0))
    primary_deadline = min(deadline, t0 + primary_time)
    h_opt, c_opt = optimize_candidate(best_h, len(best_h), primary_deadline, rng, quick=False)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    nb = len(best_h)
    for dn in (-7, -5, -3, -2, -1, 1, 2, 3, 5, 7):
        n_try = nb + dn
        if n_try < 24 or time.time() > deadline - 120.0:
            continue
        try:
            ht = resample_h(best_h, n_try)
            ct = compute_c5(ht, n_try)
            if ct < best_c:
                best_h, best_c = ht, ct
            if n_try > 1:
                hs = 0.5 * (ht + ht[::-1])
                hs = fix_feasible_light(hs, n_try)
                cs = compute_c5(hs, n_try)
                if cs < best_c:
                    best_h, best_c = hs, cs
        except Exception:
            pass

    alt_candidates = []
    if best_h is not None:
        nb = len(best_h)
        for mult in (0.5, 0.75, 1.25, 1.5, 2.0):
            n_try = int(round(nb * mult))
            if 24 <= n_try <= 700:
                alt_candidates.append(n_try)
        for n_try in (240, 300, 330, 360, 420, 480, 600):
            if 24 <= n_try <= 700:
                alt_candidates.append(n_try)
        alt_candidates = sorted(set(alt_candidates))
        if len(best_h) in alt_candidates:
            alt_candidates.remove(len(best_h))

    for n_try in alt_candidates[:6]:
        if time.time() > deadline - 160.0:
            break
        remaining = deadline - time.time()
        alloc = min(80.0, max(0.0, remaining - 120.0) * 0.35)
        if alloc < 30.0:
            break
        h_start = resample_h(best_h, n_try)
        h_opt, c_opt = optimize_candidate(h_start, n_try, time.time() + alloc, rng, quick=True)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    if best_h is not None:
        scales = [0.006, 0.012, 0.022, 0.035]
        for r in range(4):
            if time.time() > deadline - 120.0:
                break
            n = len(best_h)
            remaining = deadline - time.time()
            alloc = min(80.0, max(0.0, remaining - 80.0) * 0.30)
            if alloc < 28.0:
                break
            sc = scales[min(r, len(scales) - 1)]
            hp = best_h + rng.normal(0.0, sc, n)
            hp = project_feasible(hp, n)
            if r == 2:
                hp = 0.82 * hp + 0.18 * random_smooth_h(n, rng)
                hp = project_feasible(hp, n)
            h_opt, c_opt = optimize_candidate(hp, n, time.time() + alloc, rng, quick=True)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    if best_h is not None:
        h_opt, c_opt = _final_polish(best_h, len(best_h), deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    best_h = fix_feasible_light(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
