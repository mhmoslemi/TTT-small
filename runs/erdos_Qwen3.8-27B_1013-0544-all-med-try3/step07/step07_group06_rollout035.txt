    lags = np.arange(-(n-1), n)
    # A_i(t) = (1-h[i+lag])*[0≤i+lag<n] - h[i-lag]*[0≤i-lag<n]
    # B_j(t) = (h[j+lag]-1)*[0≤j+lag<n] + h[j-lag]*[0≤j-lag<n]

    im = i - lags
    jp = j + lags
    jm = j - lags
    
    A = np.zeros(2*n-1)
    mask_p = (ip >= 0) & (ip < n)
    A[mask_p] = (1 - h[ip[mask_p]])
    mask_m = (im >= 0) & (im < n)
    A[mask_m] -= h[im[mask_m]]

    # This uses the CURRENT h (before swap). But the actual delta uses h_after.
    # The difference only occurs at positions where i+lag=j or i-lag=j.
