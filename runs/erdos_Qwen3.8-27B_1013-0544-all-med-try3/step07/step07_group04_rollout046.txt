2. Solve: min_{p ∈ P} g · p

   To minimize: set p[i] = 1 where g[i] is most negative, p[i] = 0 where g[i] is most positive, and p[i] = fraction for at most one i.
3. Update: h = (1-γ)*h + γ*p, where γ is the step size (e.g., 2/(t+2))

                grad[:n-m] -= w[k] * h_curr[m:]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(800):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_greedy(c, alpha):
    mx = c.max()
    shifted = alpha * (c - mx)
    shifted -= shifted.max()
    return float(mx + np.log(np.sum(np.exp(shifted))) / alpha)

def weighted_gradient(h, n, c, weights, top_k):
    dx = 2.0 / n
    top_idx = np.argsort(weights)[-top_k:]
    w_top = weights[top_idx].copy()
    w_sum = w_top.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    w_top /= w_sum
    grad = np.zeros(n)
    for ii, k in enumerate(top_idx):
        lag = k - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        w = w_top[ii]
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
        else:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
    return grad

def frank_wolfe_optimize(h, n, deadline, max_iter=500000, alpha_start=20.0, alpha_end=2000.0):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    n_ones = n // 2
    stall_count = 0
    for t in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / (max_iter * 0.6))
        shifted = alpha * (c - c.max())
        shifted -= shifted.max()
        weights = np.exp(shifted)
        top_k = min(50, 2 * n - 1)
        grad = weighted_gradient(h_curr, n, weights, top_k)
        s = np.zeros(n)
        part_idx = np.argpartition(grad, n_ones)[:n_ones]
        s[part_idx] = 1.0
        gamma = 2.0 / (t + 2.0)
        if t > 200:
            gamma = max(0.01, gamma)
        h_new = (1.0 - gamma) * h_curr + gamma * s
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        h_curr = h_new
        if stall_count > 5000 and t > 5000:
            break
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_rounds=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_lags = np.argsort(c)[-8:]
        improved = False
        for k in top_lags:
            if time.time() > deadline:
                break
            lag = k - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            if lag > 0:
                a_idx = np.arange(n - m)
                b_idx = a_idx + m
            else:
                b_idx = np.arange(n - m)
                a_idx = b_idx + m
            for _ in range(300):
                if time.time() > deadline:
                    break
                val_a = h[a_idx]
                val_b = h[b_idx]
                diff_val = val_a - val_b
                if lag > 0:
                    # g_m = sum h[i]*(1-h[i+m]), want to reduce
                    # Transfer from a to b if h[a] > h[b] (reduces h[i]*(1-h[i+m]))
                    move_mask = (diff_val > 0.001)
                else:
                    move_mask = (diff_val < -0.001)
                    diff_val = -diff_val
                if not move_mask.any():
                    break
                candidates = np.where(move_mask)[0]
                order = np.argsort(np.abs(diff_val[candidates]))[::-1]
                for ci in order[:50]:
                    idx = candidates[ci]
                    delta = min(0.01, 0.4 * abs(diff_val[idx]))
                    if delta < 1e-7:
                        continue
                    if lag > 0:
                        h[a_idx[idx]] -= delta
                        h[b_idx[idx]] += delta
                    else:
                        h[b_idx[idx]] += delta
                        h[a_idx[idx]] -= delta
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        if lag > 0:
                            h[a_idx[idx]] += delta
                            h[b_idx[idx]] -= delta
                        else:
                            h[b_idx[idx]] -= delta
                            h[a_idx[idx]] += delta
                        h = project_feasible(h, n)
        if not improved and rnd > 5:
            break
    return h, best_c

def slsqp_fine(h, n, deadline, alpha=20000, maxiter=3000):
    dx = 2.0 / n
    target_sum = n / 2.0
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        mx = c.max()
        shifted = alpha * (c - mx)
        shifted -= shifted.max()
        return float(mx + np.log(np.sum(np.exp(shifted))) / alpha)
    def obj_jac(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        mx = c.max()
        shifted = alpha * (c - mx)
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        top_k = 30
        top_idx = np.argsort(weights)[-top_k:]
        for ii, k in enumerate(top_idx):
            lag = k - (n - 1)
            m = abs(lag)
            if m >= n:
                continue
            w = weights[ii]
            if m == 0:
                grad += w * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[:n - m] += w * (1.0 - x[m:])
                grad[m:] -= w * x[:n - m]
            else:
                grad[m:] += w * (1.0 - x[:n - m])
                grad[:n - m] -= w * x[m:]
        return grad
    def con_fun(x):
        return float(np.sum(x) - target_sum)
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

        res = minimize(obj, h, jac=obj_jac, method='SLSQP', bounds=bounds,

        return h.copy()

def binary_reproject(h, n, deadline):
    """Try binary projection and refine."""
    n_ones = n // 2
    h_bin = np.zeros(n)
    h_bin[np.partition(h, n - n_ones)[n - n_ones:]] = 1.0
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    if c_bin < compute_c5(h, n):
        return h_bin
    # Try perturbed binary
    for _ in range(5):
        if time.time() > deadline:

        perturb = np.random.RandomState(int(time.time()*1000) % 2**31)
        mask = perturb.random(n) < 0.05

        h_pert[mask] = 0.5 + perturb.random(mask.sum()) * 0.3
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < c_bin:
            h_bin = h_pert.copy()
            c_bin = c_pert
    return h_bin

        best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
