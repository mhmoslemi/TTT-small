import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_subgradient(h, n, top_k=10):
    """Compute subgradient of C5 at h. Returns gradient (averaged over top-k active lags)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = corr.max()
    # Find active lags (where corr is close to max)
    threshold = max_c - 1e-10
    active = np.where(corr >= threshold)[0]
    if len(active) > top_k:
        # Take top-k
        active = active[np.argsort(-corr[active])][:top_k]
    
    grad = np.zeros(n)
    for t in active:
        k = t - (n - 1)  # lag
        # d/dh[j] of corr[t] = (1-h[j-k])*1[0≤j-k<n] - h[j+k]*1[0≤j+k<n]
        # For all j:
        # term1: (1-h[j-k]) for 0≤j-k<n → j in [k, n-1+k] (clipped to [0,n-1])
        # term2: -h[j+k] for 0≤j+k<n → j in [-k, n-1-k] (clipped to [0,n-1])
        
        # Vectorized:
        # term1 at j: 1-h[j-k] when 0≤j-k≤n-1
        j_min1 = max(0, k)
        j_max1 = min(n-1, n-1+k)
        if j_min1 <= j_max1:
            grad[j_min1:j_max1+1] += (1.0 - h[j_min1-k:j_max1-k+1])
        
        # term2 at j: -h[j+k] when 0≤j+k≤n-1
        j_min2 = max(0, -k)
        j_max2 = min(n-1, n-1-k)
        if j_min2 <= j_max2:
            grad[j_min2:j_max2+1] -= h[j_min2+k:j_max2+k+1]
    
    # Average over active lags and scale by dx
    grad = grad / len(active) * dx
    return grad, max_c * dx

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    if alpha * (g - m).max() < -50:
        return m
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_opt(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
        if res.success or res.fun < smooth_max_obj(h, n, alpha):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def projected_subgradient(h, n, alpha_init, n_iters, deadline, rng):
    """Projected subgradient method."""
    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    x = h.copy()
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        # Step size
        alpha = alpha_init / (1 + 0.01 * it)
        
        # Compute subgradient
        corr = np.correlate(x, 1 - x, mode='full')
        max_c = corr.max()
        threshold = max_c - max(1e-8, 0.01 * max_c * max_c)
        active = np.where(corr >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Average subgradient over active
        grad = np.zeros(n)
        for t in active:
            k = t - (n - 1)
            j_min1 = max(0, k)
            j_max1 = min(n-1, n-1+k)
            if j_min1 <= j_max1:
                grad[j_min1:j_max1+1] += (1.0 - x[j_min1-k:j_max1-k+1])
            j_min2 = max(0, -k)
            j_max2 = min(n-1, n-1-k)
            if j_min2 <= j_max2:
                grad[j_min2:j_max2+1] -= x[j_min2+k:j_max2+k+1]
        
        grad *= dx / len(active)
        
        # Gradient step
        x = x - alpha * grad
        
        # Project
        x = project_feasible(x, n)
        
        c = compute_c5(x, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = x.copy()
        
        # Random perturbation every 100 iters to escape local minima
        if it > 0 and it % 200 == 0:
            x = x + 0.01 * rng.randn(n)
            x = project_feasible(x, n)
    
    return best_h, best_c

def coord_descent_cd(h, n, n_passes, deadline, rng, lam_init=0.0):
    """Coordinate descent with Lagrange for sum constraint."""
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1 - h, mode='full').copy()
    lam = lam_init
    
    best_c = np.max(corr) * dx
    best_h = h.copy()
    
    # Number of candidate x values to try per coordinate
    n_cand = 21
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        
        # Update lambda based on sum constraint
        s = h.sum()
        err = s - target
        step = 1.0 / (1 + 0.05 * p)
        lam = lam + step * err
        lam = np.clip(lam, -5.0, 5.0)
        
        # Random order for diversity
        order = rng.permutation(n)
        
        for j in order:
            old = h[j]
            
            # Compute d vector
            d = np.zeros(2*n - 1)
            d[j:j+n] += (1.0 - h[n-1::-1])
            d[n-1-j : 2*n-1-j] -= h
            
            # Find optimal x
            # f(x) = max(corr + (x-old)*d) + lam*x
            # Try n_cand equally spaced points in [0,1]
            xs = np.linspace(0.0, 1.0, n_cand)
            xs = np.append(xs, old)  # ensure old is included
            
            best_x = old
            best_v = np.max(corr) + lam * old  # at x=old
            
            for x in xs:
                v = np.max(corr + (x - old) * d) + lam * x
                if v < best_v - 1e-15:
                    best_v = v
                    best_x = x
            
            # Refine around best_x with finer grid
            if 0 < best_x < 1:
                span = 0.05
                fine_xs = np.linspace(max(0, best_x-span), min(1, best_x+span), 11)
                for x in fine_xs:
                    v = np.max(corr + (x - old) * d) + lam * x
                    if v < best_v - 1e-15:
                        best_v = v
                        best_x = x
            
            if abs(best_x - old) > 1e-12:
                h[j] = best_x
                corr += (best_x - old) * d
        
        # Project
        h = project_feasible(h, n)
        corr = np.correlate(h, 1 - h, mode='full')
        
        c = np.max(corr) * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Convergence check
        if p > 10 and c > best_c - 1e-15:
            continue
    
    return best_h, best_c

_g_n = 0

def _smo(x):

    alpha = _g_alpha
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    s = alpha * (g - m)
    sm = s.max()

        return m
    return m + (sm + np.log(np.exp(s - sm).sum())) / alpha

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):

    for _ in range(200):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)
