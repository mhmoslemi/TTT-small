import numpy as np
from scipy.optimize import minimize

except Exception:
    _HAS_FFT = False

_CB_DEADLINE = 0.0
_CB_X = None
_CB_COUNT = 0


class TimeUp(Exception):
    pass


def time_callback(xk):
    global _CB_X, _CB_COUNT
    _CB_COUNT += 1
    if _CB_COUNT % 25 == 0:
        _CB_X = np.array(xk, dtype=float, copy=True)
    if time.time() >= _CB_DEADLINE:
        _CB_X = np.array(xk, dtype=float, copy=True)
        raise TimeUp
    return None


def linear_convolve(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    if _HAS_FFT and (len(a) * len(b) > 200000):
        try:
            return np.real(fftconvolve(a, b, mode="full"))
        except Exception:
            pass
    return np.asarray(np.convolve(a, b, mode="full"), dtype=float)


def c_vec_fast(h, n):
    dx = 2.0 / float(n)
    v = (1.0 - h)[::-1]
    c = linear_convolve(h, v)
    c = np.real(c).astype(float, copy=False)
    if c.size:
        c = np.maximum(c, 0.0)
    return c * dx


def exact_c_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_c5(h, n):
    c = exact_c_vec(h, n)
    return float(np.max(c)) if c.size else 0.0


def project_feasible(h, n):
    target = 0.5 * float(n)
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    s = float(h.sum())
    diff = target - s
    if diff > 0.0:
        room = 1.0 - h
        tot = float(room.sum())
        if tot > 1.0e-15:
            h += (diff / tot) * room
    elif diff < 0.0:
        tot = s
        if tot > 1.0e-15:
            h += (diff / tot) * h

    h = np.clip(h, 0.0, 1.0)

    for _ in range(30):
        diff = target - float(h.sum())
        if abs(diff) < 1.0e-13:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, max(0.0, 1.0 - h[idx]))
            h[idx] += add
        else:
            idx = int(np.argmin(h))
            add = max(diff, min(0.0, -h[idx]))
            h[idx] += add

    diff = target - float(h.sum())
    if abs(diff) > 1.0e-12:
        if diff > 0.0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / float(np.count_nonzero(mask))

    return np.clip(h, 0.0, 1.0)


def symmetrize(h):
    h = 0.5 * (h + h[::-1])
    return project_feasible(h, len(h))


def resample(h, n):
    m = len(h)
    if m == n:
        return project_feasible(np.array(h, dtype=float, copy=True), n)
    x_old = (np.arange(m) + 0.5) * (2.0 / float(m))
    x_new = (np.arange(n) + 0.5) * (2.0 / float(n))
    y = np.interp(x_new, x_old, h)

def make_pre(n):

        "n": n,
        "idx": np.arange(1, n),

def base_gradient_weights(h, n, weights, pre):
    h = np.asarray(h, dtype=float)
    weights = np.asarray(weights, dtype=float)
    center = n - 1
    dx = 2.0 / float(n)
    L = 2 * n - 1

    g = weights[center] * (1.0 - 2.0 * h)

    if n > 1:
        P = weights[center + 1:]
        N = weights[:center]
        S = P + N

        if (pre is not None and isinstance(pre, dict) and "A" in pre
                and len(pre["A"]) == L):
            A = pre["A"]
            A.fill(0.0)
            idx = pre.get("idx")
            if idx is None or len(idx) != n - 1:
                idx = np.arange(1, n)
        else:
            A = np.zeros(L, dtype=float)
            idx = np.arange(1, n)

        A[center - idx] = S
        A[center + idx] = S

        conv = linear_convolve(A, h)
        B = conv[center:center + n]
        if len(B) < n:
            B = np.concatenate([B, np.zeros(n - len(B), dtype=float)])

        Pcum = np.zeros(n, dtype=float)
        Ncum = np.zeros(n, dtype=float)
        if n > 1:
            Pcum[1:] = np.cumsum(P)
            Ncum[1:] = np.cumsum(N)

        const = Pcum + Ncum[::-1]
        g = g + const - B

    g *= dx
    return g


def smooth_obj_fft(x, n, alpha, mu, pre=None):
    x = np.asarray(x, dtype=float)
    if x.size != n:
        return 1.0e100
    if not np.all(np.isfinite(x)):
        return 1.0e100

    c = c_vec_fast(x, n)
    if c.size == 0:
        return 1.0e100
    m = float(np.max(c))
    if not math.isfinite(m):
        return 1.0e100

    sh = alpha * (c - m)
    sh = np.minimum(sh, 0.0)
    e = np.exp(sh)
    se = float(np.sum(e))
    if not math.isfinite(se) or se <= 0.0:
        le = 0.0
    else:
        le = math.log(se)

    target = 0.5 * float(n)
    err = float(np.sum(x) - target)
    return float(m + le / alpha + mu * err * err)


def smooth_grad_fft(x, n, alpha, mu, pre=None):
    x = np.asarray(x, dtype=float)
    if x.size != n:
        return np.zeros(n, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)

    c = c_vec_fast(x, n)
    m = float(np.max(c))
    if not math.isfinite(m):
        return np.zeros(n, dtype=float)

    sh = alpha * (c - m)
    sh = np.minimum(sh, 0.0)
    e = np.exp(sh)
    se = float(np.sum(e))
    if not math.isfinite(se) or se <= 0.0:
        w = np.full(c.size, 1.0 / float(c.size), dtype=float)
    else:
        w = e / se

    g = base_gradient_weights(x, n, w, pre)
    target = 0.5 * float(n)
    err = float(np.sum(x) - target)
    g += 2.0 * mu * err
    return g

def optimize_point(h, n, phase_deadline, configs, max_run_s):
    global _CB_DEADLINE, _CB_X, _CB_COUNT

    h = project_feasible(h, n)
    best = exact_c5(h, n)
    pre = make_pre(n)
    bounds = [(0.0, 1.0)] * n

    for alpha, mu in configs:
        now = time.time()
        if now > phase_deadline - 3.0:
            break
        run_deadline = min(phase_deadline - 1.0, now + max_run_s)
        if run_deadline <= now + 1.0:

        _CB_X = None
        _CB_COUNT = 0
        xopt = None

                    "maxfun": 15000,
                    "maxls": 30,
                },
            )
            if hasattr(res, "x") and res.x is not None:

        except TimeUp:
            xopt = _CB_X
        except Exception:

            h2 = project_feasible(xopt, n)

            cs = exact_c5(hs, n)
            if cs < best - 1.0e-15:
                best = cs
                h = hs

    return h, best


def random_fourier_start(n, seed):

    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = 0.5 * np.ones(n, dtype=float)

    modes = [1, 2, 3, 4, 5, 6, 8, 10]
    for m in modes:
        amp = (0.25 / (1.0 + 0.25 * float(m))) * (0.5 + float(rng.random()))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        h += amp * np.cos(math.pi * float(m) * x + ph)

    return project_feasible(h, n)


def five_block_start(n, seed):
    if n % 5 != 0 or n < 20:
        return None
    b = n // 5
    rng = np.random.default_rng(seed)
    block_len = 0.4
    xs = (np.arange(b) + 0.5) * (block_len / float(b))

    base = 0.5 * np.ones(b, dtype=float)
    for m in [1, 2, 3, 4]:
        amp = float(rng.normal(0.0, 0.18))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        base += amp * np.cos(2.0 * math.pi * float(m) * xs / block_len + ph)

    h = np.tile(base, 5)
    return project_feasible(h, n)


def alternate_search(seed, deadline, parent_h):
    best_h = project_feasible(np.array(parent_h, dtype=float, copy=True), len(parent_h))
    best_n = len(best_h)
    best_c = exact_c5(best_h, best_n)

    n = 500
    p = resample(parent_h, n)
    s = symmetrize(p)
    candidates = [p, s]
    r1 = random_fourier_start(n, seed + 17)
    r2 = random_fourier_start(n, seed + 911)
    r3 = five_block_start(n, seed + 577)
    if r1 is not None:
        candidates.append(r1)
    if r2 is not None:
        candidates.append(r2)
    if r3 is not None:
        candidates.append(r3)

    evals = []
    for idx, hh in enumerate(candidates):
        hh = project_feasible(hh, n)
        c = exact_c5(hh, n)
        evals.append((c, idx, hh))
    evals.sort()

    runs = 0
    for _, _, hh in evals:
        if time.time() >= deadline - 5.0:
            break
        runs += 1
        if runs > 2:
            break
        end = min(deadline, time.time() + 35.0)
        h2, c2 = optimize_point(hh, n, end, configs_for_n(n), 18.0)
        if c2 < best_c:
            best_h = h2
            best_n = n
            best_c = c2

    return best_h, best_c, best_n


def active_polish(h, n, deadline, pre=None):
    h = project_feasible(h, n)
    best = exact_c5(h, n)
    if pre is None:
        pre = make_pre(n)

    tol_seq = [1.0e-3, 5.0e-4, 2.0e-4, 1.0e-4, 5.0e-5,
               2.0e-5, 1.0e-5, 5.0e-6, 2.0e-6, 1.0e-6]
    etas = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
            0.001, 0.0005, 0.0002, 0.0001, 5.0e-5, 2.0e-5, 1.0e-5]

    for tol in tol_seq:
        if time.time() >= deadline - 2.0:
            break
        while time.time() < deadline - 2.0:
            c = c_vec_fast(h, n)
            m = float(np.max(c))
            threshold = m - max(1.0e-12, tol * max(1.0, m))
            act = np.where(c >= threshold)[0]
            if act.size == 0:
                break

            w = np.zeros(c.size, dtype=float)
            w[act] = 1.0 / float(act.size)
            g = base_gradient_weights(h, n, w, pre)
            gn = float(np.max(np.abs(g)))
            if gn < 1.0e-15:
                break

            accepted = False
            for eta in etas:
                step = eta / gn
                if step > 0.2:
                    step = 0.2
                y = h - step * g
                y = project_feasible(y, n)
                cf = float(np.max(c_vec_fast(y, n)))
                if cf < best - 1.0e-12:
                    ce = exact_c5(y, n)
                    if ce < best - 1.0e-14:
                        h = y
                        best = ce
                        accepted = True
                        break
            if not accepted:
                break

    return h, best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 10.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        h0 = 0.5 * np.ones(256, dtype=float)
        n0 = 256

    h0 = project_feasible(h0, n0)
    best_h = h0
    best_n = n0
    best_c = exact_c5(best_h, best_n)

    hs = symmetrize(best_h)
    cs = exact_c5(hs, best_n)
    if cs < best_c:
        best_h = hs
        best_c = cs

    work_h = np.array(best_h, dtype=float, copy=True)
    work_n = best_n

    if best_n < 1000 and time.time() < deadline - 800.0:
        ns = max(64, min(1000, best_n))
        h_work = resample(best_h, ns)
        t_end = min(deadline - 750.0, time.time() + 45.0)
        h_work, c_work = optimize_point(h_work, ns, t_end, configs_for_n(ns)[:3], 18.0)
        work_h = h_work
        work_n = ns
        if c_work < best_c:
            best_h = h_work
            best_n = ns
            best_c = c_work

    if time.time() < deadline - 600.0:
        n1 = 1000
        h_work = resample(work_h, n1)
        h_alt = resample(best_h, n1)
        cw = exact_c5(h_work, n1)
        ca = exact_c5(h_alt, n1)
        start = h_alt if ca < cw - 0.001 else h_work
        t_end = min(deadline - 520.0, time.time() + 180.0)
        h_work, c_work = optimize_point(start, n1, t_end, configs_for_n(n1)[:5], 30.0)
        work_h = h_work
        work_n = n1
        if c_work < best_c:
            best_h = h_work
            best_n = n1
            best_c = c_work

    if time.time() < deadline - 480.0:
        t_alt = min(deadline - 420.0, time.time() + 130.0)
        h_alt, c_alt, n_alt = alternate_search(int(seed) + 101, t_alt, best_h)
        if c_alt < best_c:
            best_h = h_alt
            best_n = n_alt
            best_c = c_alt

    if time.time() < deadline - 160.0:
        n2 = 2000
        h_work = resample(work_h, n2)
        h_alt = resample(best_h, n2)
        cw = exact_c5(h_work, n2)
        ca = exact_c5(h_alt, n2)
        start = h_alt if ca < cw - 0.002 else h_work
        t_end = min(deadline - 90.0, time.time() + 600.0)
        h_work, c_work = optimize_point(start, n2, t_end, configs_for_n(n2), 60.0)
        work_h = h_work
        work_n = n2
        if c_work < best_c:
            best_h = h_work
            best_n = n2
            best_c = c_work

    if time.time() < deadline - 80.0:
        n3 = 4000
        h_work = resample(work_h, n3)
        h_alt = resample(best_h, n3)
        cw = exact_c5(h_work, n3)
        ca = exact_c5(h_alt, n3)
        start = h_alt if ca < cw - 0.002 else h_work
        t_end = min(deadline - 35.0, time.time() + 150.0)
        h_work, c_work = optimize_point(start, n3, t_end, configs_for_n(n3)[-4:], 45.0)
        work_h = h_work
        work_n = n3
        if c_work < best_c:
            best_h = h_work
            best_n = n3
            best_c = c_work

    best_h = project_feasible(best_h, best_n)
    hs = symmetrize(best_h)
    cs = exact_c5(hs, best_h.shape[0])
    if cs < best_c:
        best_h = hs
        best_c = cs

    if time.time() < deadline - 10.0:
        best_h, best_c = active_polish(best_h, best_n, deadline - 3.0)

    if time.time() < deadline - 20.0 and work_h is not None and work_n != best_n:
        cw = exact_c5(work_h, work_n)
        if cw < best_c + 2.0e-4:
            hh, cc = active_polish(work_h, work_n, deadline - 2.0)
            if cc < best_c:
                best_h = hh
                best_n = work_n
                best_c = cc

    best_h = project_feasible(best_h, best_n)
    best_c = exact_c5(best_h, best_n)

    return best_h, float(best_c), int(best_n)
