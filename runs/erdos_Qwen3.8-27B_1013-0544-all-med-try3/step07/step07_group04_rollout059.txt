    m = abs(lag)

    if m == 0:
        return 1.0 - 2.0 * h

        grad[m:] += (1.0 - h[:n - m])

        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

    """Gradient of smooth max approximation using all lags."""
    c_all = compute_all_lags(h, n)
    m = c_all.max()
    shifted = alpha * (c_all - m)
    # numerical stability
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] > 1e-8:
            lag = k - (n - 1)
            grad += weights[k] * lag_gradient(h, n, lag)
    return grad

            alpha = alpha_init / (1.0 + 0.1 * no_improve)
            if no_improve > 20:
                break
    return best_h, best_c

- For j = 0,...,ℓ-1: ∂f/∂h[j] = -h[j+ℓ] (negative, so we want to increase h[j])
- For j = ℓ,...,n-ℓ-1: ∂f/∂h[j] = 1 - h[j-ℓ] - h[j+ℓ] (can be positive or negative)
- For j = n-ℓ,...,n-1: ∂f/∂h[j] = 1 - h[j-ℓ] (positive if h[j-ℓ] < 1, so we want to decrease h[j])

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    return result

def lag_gradient(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_max_grad(h, n, alpha, top_k=15):
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_k:]
    m = c_all[top_idx].max()
    shifted = alpha * (c_all[top_idx] - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for idx, k in enumerate(top_idx):
        if w[idx] > 1e-10:
            lag = k - (n - 1)
            grad += w[idx] * lag_gradient(h, n, lag)
    return grad

def projected_gradient(h, n, deadline, max_iters=30000, base_alpha=0.003):
    h = project_box_sum(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = base_alpha
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 2000 == 0 and it > 0:
            alpha = max(base_alpha * 0.5, alpha * 0.7)
        g = smooth_max_grad(h, n, alpha=500, top_k=20)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        step = alpha * g / gnorm
        h_try = project_box_sum(h - step, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-13:
            best_c = c_try
            best_h = h_try.copy()
            h = h_try
            stall = 0
            alpha = min(alpha * 1.03, 0.05)
        else:
            h_try = project_box_sum(h - 0.5 * step, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-13:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                stall = 0
            else:
                alpha *= 0.85
                stall += 1
                if stall > 50:
                    break
    return best_h, best_c

def grad_local_search(h, n, deadline, max_rounds=5000):
    h = project_box_sum(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    delta = 0.02
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            g = 1.0 - 2.0 * h
        else:
            g = lag_gradient(h, n, lag)
        gmax = np.max(np.abs(g))
        if gmax < 1e-10:
            break
        improved = False
        for d in [delta, delta * 0.5, delta * 0.25, delta * 0.1]:
            h_try = h - d * g / gmax
            h_try = project_box_sum(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-12:
                best_c = c_try
                best_h = h_try.copy()

                improved = True
                delta = min(delta * 1.15, 0.08)
                stall = 0

            delta *= 0.85
            stall += 1
    return best_h, best_c

def pair_local_search(h, n, deadline):
    h = project_box_sum(h.copy(), n)
    best_c = compute_c5(h, n)
    deltas = [0.004, 0.002, 0.001, 0.0005, 0.0001]

                    elif g[j] > g[i] and h[i] > h[j]:

                        if delta < 1e-9:

                        h[i] -= delta

                        c_new = compute_c5(h, n)

            else:

                            improved = True

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    top_idx = np.argsort(c)[-12:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in top_idx:
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_box_sum(np.asarray(h_new, dtype=float), n)

def outer_approx(h, n, deadline, max_iters=60):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.12
    for it in range(max_iters):
        if time.time() > deadline - 2:

        h_curr = h_new
        trust = max(0.003, trust * 0.85)
        if it % 15 == 14:
            trust = min(trust + 0.02, 0.15)
    return best_h, best_c

def _smooth_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c_all = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c_all[top_idx]
    else:
        c = c_all
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha=50000, maxiter=3000, ftol=1e-15, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha, top_idx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_box_sum(res.x, n)
    except Exception:
        return h.copy()

def binary_greedy_init(n):
    h = np.zeros(n)
    count_ones = 0
    target_ones = n // 2
    for _ in range(target_ones):
        best_pos = -1
        best_max_lag = 0
        for pos in range(n):
            h[pos] = 1.0
            c = np.correlate(h, 1 - h, mode='full')
            curr_max = np.max(c)
            if best_pos == -1 or curr_max < best_max_lag:
                best_max_lag = curr_max
                best_pos = pos
            h[pos] = 0.0
        h[best_pos] = 1.0
        count_ones += 1
    return h

def optimize_at_n(h_start, n, deadline):
    h = project_box_sum(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Outer approximation
    if time.time() < deadline - 60:

        h_oo, c_oo = outer_approx(best_h, n, oo_end)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    # Phase 2: Projected gradient
    if time.time() < deadline - 40:
        pg_end = time.time() + min(80, (deadline - time.time()) * 0.35)
        h_pg, c_pg = projected_gradient(best_h, n, pg_end)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    # Phase 3: SLSQP with increasing alpha

        if time.time() > deadline - 20:
            break
        c_all = compute_all_lags(best_h, n)
        top_idx = np.argsort(c_all)[-10:]
        h_new = slsqp_optimize(best_h, n, alpha, maxiter=2000, ftol=1e-15, top_idx=top_idx)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 4: Gradient local search

    # Phase 5: Pair local search

            best_c = c_new
            best_h = h_new
    best_h = project_box_sum(best_h, n)
