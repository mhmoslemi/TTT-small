def project_feasible(h, n):

            if s < 1e-14:

def project_binary(h, n):
    target = n // 2
    h = np.round(h).astype(float)
    n_ones = int(h.sum())
    if n_ones == target:
        return h
    elif n_ones > target:
        ones_idx = np.where(h == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - target):
            h[ones_idx[order[i]]] = 0

        for i in range(target - n_ones):
            h[zeros_idx[order[i]]] = 1
    return h

def get_delta(h, n, i, j, m_range):
    """Compute correlation delta for swapping h[i]:0->1, h[j]:1->0."""
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    
    delta = np.zeros(2 * n - 1)
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    
    return delta

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return val
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:
        return h

        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n_best = n0

    # Phase 1: SLSQP on parent resolution
    p1 = t0 + 0.08 * (budget_s - 8)
    h_work = project_feasible(best_h.copy(), n0)
    h_work, c1 = slsqp_multi_alpha(h_work, n0, [500, 2000, 10000, 50000, 200000], 1000, p1)
    if c1 < best_c:
        best_c = c1
        best_h = h_work.copy()
        n_best = n0

    # Phase 1b: Quick binary search on parent
    p1b = t0 + 0.18 * (budget_s - 8)
    hb, cb = binary_local_search(best_h, n0, p1b, max_sweeps=3000, n_restarts=3, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        n_best = n0

    # Phase 2: Work at higher resolution n=500
    n_high = 500
    if time.time() < p1b:
        pass
    p2 = t0 + 0.28 * (budget_s - 8)
    xo = np.linspace(0, 2, n0)
    xh = np.linspace(0, 2, n_high)
    h_high = np.interp(xh, xo, best_h)
    h_high = project_feasible(h_high, n_high)
    h_high, c2 = slsqp_multi_alpha(h_high, n_high, [200, 1000, 5000, 20000, 100000], 800, p2)
    if c2 < best_c:
        best_c = c2
        best_h = h_high.copy()
        n_best = n_high

    # Phase 2b: Binary search at n=500
    p2b = t0 + 0.82 * (budget_s - 8)
    h_bin = project_binary(best_h, n_best)
    hb, cb = binary_local_search(h_bin, n_best, p2b, max_sweeps=20000, n_restarts=8, seed=seed+10, n_perturb_range=(5, 12))
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        n_best = n_best

    # Phase 3: Try n=1000 if we have a good enough solution
    n_vhigh = 1000
    if time.time() < dl - 60 and n_best < 1000:
        p3 = min(t0 + 0.55 * (budget_s - 8), dl - 50)
        xo2 = np.linspace(0, 2, n_best)
        xv = np.linspace(0, 2, n_vhigh)
        h_v = np.interp(xv, xo2, best_h)
        h_v = project_feasible(h_v, n_vhigh)
        h_v, c3 = slsqp_multi_alpha(h_v, n_vhigh, [500, 5000, 50000], 500, p3)
        if c3 < best_c:
            best_c = c3
            best_h = h_v.copy()
            n_best = n_vhigh
        h_bin_v = project_binary(best_h, n_best)
        hb_v, cb_v = binary_local_search(h_bin_v, n_best, p3, max_sweeps=5000, n_restarts=3, seed=seed+20, n_perturb_range=(8, 15))
        if cb_v < best_c:
            best_c = cb_v
            best_h = hb_v.copy()
            n_best = n_best

    # Phase 4: Final extended binary search at best resolution
    if time.time() < dl - 15:
        h_final_bin = project_binary(best_h, n_best)
        hb, cb = binary_local_search(h_final_bin, n_best, dl - 10, max_sweeps=50000, n_restarts=5, seed=seed+42, n_perturb_range=(3, 10))
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            n_best = n_best

    # Phase 5: SLSQP polish
    if time.time() < dl - 5:
        h_pol = project_feasible(best_h + 0.03 * rng.randn(n_best), n_best)
        h_pol, c_pol = slsqp_multi_alpha(h_pol, n_best, [100000, 500000, 2000000], 1500, dl - 3)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol.copy()
        # Round and quick binary
        if time.time() < dl - 1:
            h_rb = project_binary(best_h, n_best)
            hb5, cb5 = binary_local_search(h_rb, n_best, dl, max_sweeps=2000, n_restarts=2, seed=seed+99)
            if cb5 < best_c:
                best_c = cb5
                best_h = hb5.copy()

    # Final: ensure feasibility and recompute
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)

    return best_h, best_c, n_best
