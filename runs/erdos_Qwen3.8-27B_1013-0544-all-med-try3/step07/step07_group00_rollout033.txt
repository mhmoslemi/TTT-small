    if vals.max() > vals.mean() + 1e-10:
        obj += 0.5 * (vals - vals.mean()).max()
    obj += mu * (x.sum() - n / 2.0) ** 2
    return float(obj)

def subset_grad(x, n, subset_idx, mu):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = len(subset_idx)
    for idx in subset_idx:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - x[k:n]) * dx
            grad[k:n] += (-x[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - x[0:nv]) * dx
            grad[0:nv] += (-x[kk:n]) * dx
    grad /= m
    # Add max penalty gradient
    c = compute_c5_vec(x, n)
    vals = c[subset_idx]
    mean_v = vals.mean()
    above = vals > mean_v + 1e-10
    if above.any():
        max_pos = np.where(above)[0][np.argmax(vals[above])]
        max_idx = subset_idx[max_pos]
        k = max_idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += 0.5 * (1.0 - x[k:n]) * dx
            grad[k:n] += 0.5 * (-x[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += 0.5 * (1.0 - x[0:nv]) * dx
            grad[0:nv] += 0.5 * (-x[kk:n]) * dx
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgradient_descent(h, n, deadline, init_step=0.01, top_k=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = init_step
    consecutive_fail = 0
    
    while time.time() < deadline and step > 1e-8:
        c = compute_c5_vec(best_h, n)
        max_c = c.max()
        # Use top_k lags for the subgradient
        top_idx = np.argsort(-c)[:top_k]
        
        grad = np.zeros(n)
        dx = 2.0 / n
        for idx in top_idx:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(top_idx)
        
        new_h = best_h - step * grad
        new_h = project_feasible(new_h, n)
        new_c = compute_c5(new_h, n)
        
        if new_c < best_c - 1e-15:
            best_c = new_c
            best_h = new_h
            consecutive_fail = 0
        else:
            consecutive_fail += 1
            if consecutive_fail > 10:
                step *= 0.5
                consecutive_fail = 0
            else:
                step *= 0.85
    
    return best_h, best_c

def pair_move_local_search(h, n, deadline, base_step=0.005):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = base_step
    
    while time.time() < deadline and step > 1e-8:
        c = compute_c5_vec(best_h, n)
        max_c = c.max()
        
        # Active lags within tolerance
        tol = max(1e-6, 0.002 * max_c)
        active = np.where(c >= max_c - tol)[0]
        
        if len(active) == 0:
            step *= 0.5
            continue
        
        # Compute combined gradient over active lags
        grad = np.zeros(n)
        dx = 2.0 / n
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        
        improved = False
        # Try moves: decrease h where grad > 0, increase where grad < 0
        pos = np.where(grad > 0)[0]
        neg = np.where(grad < 0)[0]
        
        if len(pos) > 0 and len(neg) > 0:
            # Sort by magnitude
            pos_order = pos[np.argsort(-grad[pos])][:40]
            neg_order = neg[np.argsort(grad[neg])][:40]
            
            for i in pos_order:
                if time.time() > deadline:
                    return best_h, best_c
                for j in neg_order:
                    if time.time() > deadline:
                        return best_h, best_c
                    if i == j:
                        continue
                    s = step
                    if best_h[i] < s:
                        s = best_h[i] * 0.5
                    if 1.0 - best_h[j] < s:
                        s = min(s, (1.0 - best_h[j]) * 0.5)
                    if s < 1e-10:
                        continue
                    best_h[i] -= s
                    best_h[j] += s
                    new_c = compute_c5(best_h, n)
                    if new_c < best_c - 1e-15:
                        best_c = new_c
                        improved = True
                    else:
                        best_h[i] += s
                        best_h[j] -= s
        
        if not improved:
            step *= 0.5
    
    return best_h, best_c

def active_set_cycling(h, n, deadline, num_cycles=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for cycle in range(num_cycles):
        if time.time() > deadline - 10:
            break
        
        c = compute_c5_vec(best_h, n)
        max_c = c.max()
        # Select active set: lags within tolerance of max
        for frac in [0.0, 0.01, 0.05, 0.1, 0.2]:
            if time.time() > deadline - 10:
                break
            tol = frac * max_c
            active = np.where(c >= max_c - tol)[0]
            if len(active) < 2:
                active = np.argsort(-c)[:5]
            if len(active) > 50:
                active = np.argsort(-c)[:50]
            
            try:
                res = minimize(subset_obj, best_h, jac=subset_grad,
                              args=(n, active, 100.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 20000, 'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    return best_h, best_c

def perturbation_restart(h, n, deadline, num_tries=15):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(12345)
    
    for _ in range(num_tries):
        if time.time() > deadline - 15:
            break
        
        # Random perturbation with decreasing scale
        scale = 0.02 + 0.03 * rng.random()
        noise = rng.normal(0, scale, n)
        perturbed = project_feasible(best_h + noise, n)
        
        # Quick L-BFGS
        try:
            res = minimize(soft_max_obj, perturbed, jac=soft_max_grad,
                          args=(n, 500000, 10000.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 1000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 10000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def multiresolve尝试(h, n, deadline):
    """Try upsample to higher resolution and re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds_new = [(0.0, 1.0)] * (2 * n)
    
    n2 = 2 * n
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n2)
    h2 = np.interp(x_new, x_old, h)
    h2 = project_feasible(h2, n2)
    c2 = compute_c5(h2, n2)
    
    if c2 < best_c or True:  # Always try, but only keep if better
        # Quick optimization at higher res
        if time.time() < deadline - 20:
            try:
                res = minimize(soft_max_obj, h2, jac=soft_max_grad,
                              args=(n2, 200000, 5000.0), method='L-BFGS-B',
                              bounds=bounds_new,
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 30000, 'maxls': 30})
                h2_new = project_feasible(res.x, n2)
                c2_new = compute_c5(h2_new, n2)
                if c2_new < best_c:
                    return h2_new, c2_new, n2
            except Exception:
                pass
    
    return best_h, best_c, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    np.random.seed(seed)
    
    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    # Phase 1: Aggressive L-BFGS with high alpha (we're already near optimum)
    alpha_configs = [
        (5000, 100.0, 2000),
        (50000, 1000.0, 3000),
        (500000, 10000.0, 3000),
        (5000000, 100000.0, 2000),
        (50000000, 1000000.0, 2000),
    ]
    
    for alpha, mu, maxiter in alpha_configs:
        if time.time() > deadline - 400:
            break
        try:
            res = minimize(soft_max_obj, best_h, jac=soft_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxiter * 10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    # Phase 2: Active set cycling
    if time.time() < deadline - 300:
        t2 = min(time.time() + 120, deadline - 300)
        best_h, best_c = active_set_cycling(best_h, n, t2, num_cycles=30)
    
    # Phase 3: Subgradient descent
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 200)
        for top_k in [3, 5, 10, 20]:
            if time.time() > t3:
                break
            h_sg, c_sg = subgradient_descent(best_h, n, t3, init_step=0.02, top_k=top_k)
            if c_sg < best_c:
                best_c = c_sg
                best_h = h_sg
    
    # Phase 4: Pair move local search
    if time.time() < deadline - 150:
        t4 = min(time.time() + 100, deadline - 150)
        best_h, best_c = pair_move_local_search(best_h, n, t4, base_step=0.005)
    
    # Phase 5: Another high-alpha L-BFGS pass
    if time.time() < deadline - 120:
        for alpha in [100000000, 500000000]:
            if time.time() > deadline - 60:
                break
            try:
                res = minimize(soft_max_obj, best_h, jac=soft_max_grad,
                              args=(n, alpha, 500000.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': 30000, 'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 6: Perturbation and restart
    if time.time() < deadline - 50:
        t6 = min(time.time() + 80, deadline - 50)
        best_h, best_c = perturbation_restart(best_h, n, t6, num_tries=20)
    
    # Phase 7: Final subgradient + pair move polish
    if time.time() < deadline - 20:
        t7 = deadline - 10
        h_sg, c_sg = subgradient_descent(best_h, n, min(t7, time.time() + 30), init_step=0.005, top_k=5)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
        if time.time() < t7:
            best_h, best_c = pair_move_local_search(best_h, n, t7, base_step=0.002)
    
    # Phase 8: Try higher resolution
    if time.time() < deadline - 10:
        h_hr, c_hr, n_hr = multiresolve尝试(best_h, n, deadline - 3)
        if c_hr < best_c:
            best_c = c_hr
            best_h = h_hr
            n = n_hr
    
    # Final
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
