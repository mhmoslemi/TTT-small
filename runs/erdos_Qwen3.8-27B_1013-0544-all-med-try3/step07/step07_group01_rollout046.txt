            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj_smax(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_smax(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _obj_slsqp(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _con_eq(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def lbfgs_optimize(h, n, deadline, max_per_alpha=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [5, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    mus = [1.0, 5.0, 10.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0, 50000.0]
    for idx in range(len(alphas)):
        if time.time() > deadline - 5:
            break
        alpha = alphas[idx]
        mu = mus[idx]
        maxit = max_per_alpha if idx < 4 else max_per_alpha // 2
        t_alloc = min(45, deadline - time.time() - 3)
        try:
            res = minimize(_obj_smax, h, jac=_grad_smax,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxfun': maxit * 8,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=1500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_slsqp, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def line_search_single(h, n, i, direction, deadline):
    """Try to move h[i] in the given direction while adjusting another coordinate."""
    best_c = compute_c5(h, n)
    step = 0.01
    for _ in range(30):
        if time.time() > deadline:
            break
        h_new = h.copy()
        h_new[i] += direction * step
        if h_new[i] < 0 or h_new[i] > 1:
            step *= 0.5
            continue
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            step = min(step * 1.5, 0.1)
        else:
            step *= 0.5
            if step < 1e-10:
                break
    return h, best_c

def grad_guided_local(h, n, deadline, max_rounds=200):
    """Gradient-guided local search: move probability mass along negative gradient."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    alpha = 5000
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        # Find best donor (high h, high gradient) and receiver (low h, low gradient)
        score = g * (h - 0.5)
        # We want to decrease h where g*h is large, increase where g*h is small
        # Actually: delta objective ≈ sum_j delta_j * g_j
        # If we move eps from i to j: delta = eps*(g_j - g_i)
        # Want g_j < g_i (move from high grad to low grad)
        # Sort by gradient
        idx_sorted = np.argsort(g)
        improved = False
        for a in range(0, n, max(1, n // 200)):
            if time.time() > deadline:
                break
            for b_offset in [1, 2, 3, 5, 10, 20, 50]:
                b = (a + b_offset) % n
                if a == b:
                    continue
                if g[a] > g[b]:
                    donor, recv = a, b
                else:
                    donor, recv = b, a
                    if g[donor] - g[recv] < 1e-10:
                        continue
                for eps in [0.005, 0.002, 0.001, 0.0005]:
                    if h[donor] - eps < 0 or h[recv] + eps > 1:
                        continue
                    h[donor] -= eps
                    h[recv] += eps
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        h[donor] += eps
                        h[recv] -= eps
                        h = project_feasible(h, n)
                if improved:
                    break
            if improved:
                break
        if not improved:
            alpha = max(alpha * 0.8, 500)
    return h, best_c

def pair_search_fine(h, n, deadline, deltas, radius=80):
    """Fine pair search with local radius."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in deltas:
        progress = True
        while progress and time.time() < deadline:
            progress = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c
                            progress = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c
                            progress = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def block_search(h, n, deadline, block_sizes=[2, 3, 5, 8]):
    """Try adjusting blocks of consecutive values together."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for bs in block_sizes:
        if time.time() > deadline:
            break
        for start in range(0, n - bs + 1, bs // 2 + 1):
            if time.time() > deadline:
                return h, best_c
            block = h[start:start+bs]
            for shift in [0.005, -0.005, 0.002, -0.002]:
                h_new = h.copy()
                h_new[start:start+bs] = np.clip(block + shift, 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    h = h_new
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h_start = np.array(initial_h_values, dtype=float).copy()
        n_start = len(h_start)
    except Exception:
        h_start = np.full(250, 0.5)
        n_start = 250

    best_h = project_feasible(h_start, n_start)
    best_c = compute_c5(best_h, n_start)
    best_n = n_start

    # Phase 1: Quick optimization at starting resolution (up to 150s)
    t1 = min(t0 + 120, deadline - 400)
    if time.time() < t1:
        best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_per_alpha=200)
        best_n = len(best_h)

    # Phase 2: SLSQP at starting resolution
    if time.time() < deadline - 350:
        t2 = min(time.time() + 60, deadline - 350)
        for a in [50000, 200000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_optimize(best_h, best_n, t2, alpha=a, maxiter=1000)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Interpolate to higher resolution (n=600)
    n_target = 600
    if time.time() < deadline - 300:
        h600 = interp_h(best_h, best_n, n_target)
        t3 = min(time.time() + 120, deadline - 300)
        h600, c600 = lbfgs_optimize(h600, n_target, t3, max_per_alpha=150)
        if c600 < best_c:
            best_c = c600
            best_h = h600
            best_n = n_target

        # SLSQP at n=600
        if time.time() < deadline - 200:
            h600, c600 = slsqp_optimize(h600, n_target, deadline - 200, alpha=200000, maxiter=1000)
            if c600 < best_c:
                best_c = c600
                best_h = h600
                best_n = n_target

    # Phase 4: Gradient-guided local search
    if time.time() < deadline - 120:
        t4 = min(time.time() + 80, deadline - 120)
        h4, c4 = grad_guided_local(best_h, best_n, t4, max_rounds=300)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Fine pair search
    if time.time() < deadline - 60:
        t5 = min(time.time() + 70, deadline - 60)
        best_h, best_c = pair_search_fine(best_h, best_n, t5,
                                          [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                          radius=60)

    # Phase 6: Try even higher resolution (n=900)
    if time.time() < deadline - 80:
        n_high = 900
        h900 = interp_h(best_h, best_n, n_high)
        t6 = min(time.time() + 50, deadline - 80)
        h900, c900 = lbfgs_optimize(h900, n_high, t6, max_per_alpha=100)
        if c900 < best_c:
            best_c = c900
            best_h = h900
            best_n = n_high
        # Quick local search at n=900
        if time.time() < deadline - 30 and best_n == n_high:
            h900, c900 = grad_guided_local(h900, n_high, deadline - 30, max_rounds=100)
            if c900 < best_c:
                best_c = c900
                best_h = h900

    # Phase 7: Final L-BFGS + pair search on best
    if time.time() < deadline - 10:
        nf = len(best_h)
        if time.time() < deadline - 8:
            h7, c7 = lbfgs_optimize(best_h, nf, deadline - 8, max_per_alpha=50)
            if c7 < best_c:
                best_c = c7
                best_h = h7
        if time.time() < deadline - 3:
            best_h, best_c = pair_search_fine(best_h, len(best_h), deadline - 2,
                                              [0.001, 0.0005, 0.0002, 0.0001],
                                              radius=40)
    # Phase 8: Block search
    if time.time() < deadline - 2:
        best_h, best_c = block_search(best_h, len(best_h), deadline - 1)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
