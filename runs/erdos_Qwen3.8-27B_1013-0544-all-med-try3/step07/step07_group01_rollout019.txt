```python
def compute_grad_weighted(C, h, n, alpha):
    """Compute gradient of smooth max of C, weighted by position in h-space."""
    dx = 2.0 / n  # already included in C
    current_max = C.max()
    w = np.exp(alpha * (C - current_max))
    w /= w.sum()
    
    # Combine weights for ±k lags
    # m = n-1+k for k = -(n-1), ..., n-1
    # lag k corresponds to m = n-1+k
    # lag -k corresponds to m = n-1-k
    grad = np.zeros(n)
    
    for l in range(n):
        # lag = +l → m = n-1+l
        # lag = -l → m = n-1-l (for l > 0)
        if l == 0:
            comb_w = w[n-1]
        else:
            comb_w = w[n-1+l] + w[n-1-l]
        
        if comb_w < 1e-14:
            continue
        
        grad[l:] += comb_w * (1.0 - h[:n-l])
        grad[:n-l] += comb_w * (-h[l:])
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_smooth(C, h, n, alpha):
    """Vectorized gradient of smooth max."""
    current_max = C.max()
    w = np.exp(alpha * (C - current_max))
    w /= w.sum()
    
    grad = np.zeros(n)
    for l in range(n):
        if l == 0:
            cw = w[n-1]
        else:
            cw = w[n-1+l] + w[n-1-l]
        if cw < 1e-14:
            continue
        grad[l:] += cw * (1.0 - h[:n-l])
        grad[:n-l] -= cw * h[l:]
    
    # Project to sum-zero (for the constraint)
    grad -= grad.mean()
    return grad

def smooth_max_obj(h, n, alpha):
    C = compute_c5_vec(h, n)
    m = C.max()
    sh = alpha * (C - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    C = compute_c5_vec(h, n)
    return compute_grad_smooth(C, h, n, alpha)

def lbfgs_optimize(h, n, deadline, max_iters=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [(1000, 10.0), (10000, 50.0), (100000, 200.0), (1000000, 1000.0)]
    
    for alpha, mu in alphas:
        if time.time() > deadline - 5:
            break
        
        def obj(x):
            C = compute_c5_vec(x, n)
            m = C.max()
            sh = alpha * (C - m)
            mx = sh.max()
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
            val += mu * (x.sum() - n/2.0)**2
            return float(val)
        
        def grad(x):
            C = compute_c5_vec(x, n)
            g = compute_grad_smooth(C, x, n, alpha)
            g += 2.0 * mu * (x.sum() - n/2.0)
            return g
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxfun': max_iters*5, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except:
            pass
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_influence(h, n):
    """Compute INFL[p, m] = dC(m)/dh[p] * dx for all p, m."""
    num_lags = 2 * n - 1
    INFL = np.zeros((n, num_lags))
    for l in range(n):
        m_pos = n - 1 + l
        INFL[l:, m_pos] = (1.0 - h[:n - l])
        INFL[:n - l, m_pos] -= h[l:]
        if l > 0:
            m_neg = n - 1 - l
            INFL[l:, m_neg] = (1.0 - h[:n - l])
            INFL[:n - l, m_neg] -= h[l:]
    return INFL

def compute_grad_smooth(C, h, n, alpha):
    """Compute gradient of smooth max objective wrt h."""
    current_max = C.max()
    w = np.exp(alpha * (C - current_max))
    w /= w.sum()
    grad = np.zeros(n)
    for l in range(n):
        if l == 0:
            cw = w[n - 1]
        else:
            cw = w[n - 1 + l] + w[n - 1 - l]
        if cw < 1e-14:
            continue
        grad[l:] += cw * (1.0 - h[:n - l])
        grad[:n - l] -= cw * h[l:]
    grad -= grad.mean()
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    C = compute_c5_vec(x, n)
    m = C.max()
    sh = alpha * (C - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _lbfgs_grad(x, n, alpha, mu):
    C = compute_c5_vec(x, n)
    g = compute_grad_smooth(C, x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    C = compute_c5_vec(x, n)
    m = C.max()
    sh = alpha * (C - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (500, 5.0), (5000, 20.0), (50000, 100.0),
        (500000, 500.0), (2000000, 2000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_lbfgs_obj, h, jac=_lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 8,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_descent(h, n, deadline, lr_init=0.05, alpha_init=500.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    for _ in range(50000):
        if time.time() > deadline:
            break
        C = compute_c5_vec(h, n)
        g = compute_grad_smooth(C, h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        improved = False
        step_lr = lr
        for _ in range(40):
            h_new = project_feasible(h - step_lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step_lr *= 0.5
        if not improved:
            lr *= 0.5
            alpha *= 1.5
        if lr < 1e-13:
            break
    return best_h, best_c

def pair_search(h, n, deadline, deltas):

def block_search(h, n, deadline, block_sizes, delta):

        for n_try in [250, 350, 200]:
            if time.time() > t7:
                break
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_interp = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[-1]]], axis=0),
                                  period=2.0) if False else np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, t7 - time.time() - 15)
            if alloc < 20:
                break
            ht, ct = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5, max_iters=100)
            if time.time() < t7 - 10:
                ht, ct = pair_search(ht, n_try, t7 - 5, [0.005, 0.002, 0.001, 0.0005])
            if ct < best_c:
                best_c = ct
                best_h = ht.copy()
                n = n_try
                h = ht

    # Phase 8: Final intensive polish (100s)
    nf = len(best_h)
    if time.time() < deadline - 10:
        t8 = deadline - 3
        h8, c8 = pair_search(best_h, nf, min(time.time() + 50, t8 - 25),
                            [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()
        if time.time() < t8 - 5:
            h8, c8 = pair_search(best_h, nf, t8,
                                [0.0001, 0.00005, 0.00002, 0.00001])
            if c8 < best_c:
                best_c = c8
                best_h = h8.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
