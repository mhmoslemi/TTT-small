def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        # Smooth max approximation
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = np.log(np.exp(shifted - sm).sum())
        return m + (logsum) / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        exp_s = np.exp(shifted - sm)
        logsum = np.log(exp_s.sum())
        val = m + logsum / alpha
        
        # Gradient is more complex - use finite differences or skip
        # SLSQP can work without explicit gradient
        
        return val
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def vectorized_binary_improve(h_init, n, deadline, n_restarts=5, seed=42,
                              max_sweeps=5000, top_k=8):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_c = np.inf
    best_h = h_init.copy()
    L_range = np.arange(-(n - 1), n)  # length 2n-1

    h = h_init.copy()
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            nperturb = rng.randint(2, min(7, len(ones)))
            for _ in range(nperturb):
                a = rng.randint(len(ones))
                b = rng.randint(len(zeros))
                h[ones[a]], h[zeros[b]] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

        corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
        cur_max = corr.max()
        cur_c = cur_max * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            zeros = np.where(h == 0)[0].astype(np.int64)
            ones = np.where(h == 1)[0].astype(np.int64)
            nz = len(zeros)
            no = len(ones)
            if nz == 0 or no == 0:
                break

            # Build contribution matrices
            zL = zeros[:, None] + L_range[None, :]
            zLm = zeros[:, None] - L_range[None, :]
            oL = ones[:, None] + L_range[None, :]
            oLm = ones[:, None] - L_range[None, :]

            mzL = (zL >= 0) & (zL < n)
            mzLm = (zLm >= 0) & (zLm < n)
            moL = (oL >= 0) & (oL < n)
            moLm = (oLm >= 0) & (oLm < n)

            A = np.zeros((nz, 2 * n - 1), dtype=np.float64)
            B = np.zeros((no, 2 * n - 1), dtype=np.float64)

            A[mzL] = h[zL[mzL]]
            A[mzLm] += h[zLm[mzLm]] - 1.0
            B[moL] = -h[oL[moL]]
            B[moLm] += 1.0 - h[oLm[moLm]]

            # For each zero z, find best one o to swap with
            best_delta_max = cur_max + 1e-12
            best_zi = -1
            best_oi = -1
            best_new_corr = None

            for zi in range(nz):
                if time.time() > deadline:
                    break
                base = corr + A[zi]  # (2n-1,)
                M = base[None, :] + B.T  # (2n-1, no)
                new_maxes = M.max(axis=0)  # (no,)

                # Get top-k candidates
                k = min(top_k, no)
                top_idx = np.argpartition(new_maxes, k)[:k]
                top_idx = top_idx[np.argsort(new_maxes[top_idx])]

                for oi in top_idx:
                    z_pos = zeros[zi]
                    o_pos = ones[oi]
                    nc = base + B[oi]
                    # Cross corrections
                    nc[n - 1] -= 2.0
                    k1 = n - 1 + (o_pos - z_pos)
                    if 0 <= k1 < 2 * n - 1:
                        nc[k1] += 1.0
                    k2 = n - 1 + (z_pos - o_pos)
                    if 0 <= k2 < 2 * n - 1:
                        nc[k2] += 1.0
                    nm = nc.max()
                    if nm < best_delta_max:
                        best_delta_max = nm
                        best_zi = zi
                        best_oi = oi
                        best_new_corr = nc.copy()

            if best_zi < 0:
                break  # No improvement

            # Apply swap
            h[zeros[best_zi]] = 1.0
            h[ones[best_oi]] = 0.0
            corr = best_new_corr  # already computed
            cur_max = corr.max()
            cur_c = cur_max * dx

            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()

    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=20000):
    """Simulated annealing on binary assignments."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_c = np.inf
    best_h = h_init.copy()
    target = n // 2

    h = h_init.copy()
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0

    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()

    T0 = dx * 0.1
    T_end = dx * 0.001
    n_total = n_iters

    for it in range(n_total):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max(1, n_total - 1))

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        a = rng.randint(len(ones))
        b = rng.randint(len(zeros))
        oa = ones[a]
        ob = zeros[b]

        # Compute delta
        h[oa] = 0.0
        h[ob] = 1.0
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c = new_corr.max() * dx
        dC = new_c - cur_c

        if dC < 0 or rng.random() < np.exp(-dC / max(T, 1e-15)):
            cur_c = new_c
            corr = new_corr
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[oa] = 1.0
            h[ob] = 0.0

    return best_h, best_c
