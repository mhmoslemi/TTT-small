                active, _ = get_active(best_h, n, top_k=top_k)
                try:
                    res = minimize(topk_obj, best_h, jac=topk_grad,
                                  args=(n, active, alpha, mu), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': 5000, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxfun': 12000, 'maxls': 30})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                except Exception:
                    pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h_old, n_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def gradient_local(h, n, deadline):
    best_c = compute_c5(h, n)
    base_step = 0.005
    for _iter in range(500):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        max_idx = int(np.argmax(c_vec))
        dx = 2.0 / n
        k = max_idx - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
        g_top = np.argsort(-np.abs(grad))[:40]
        found = False
        for ii in range(len(g_top)):
            if found:
                break
            for jj in range(len(g_top)):
                if found:
                    break
                if ii == jj:
                    continue
                i = g_top[ii]
                j = g_top[jj]
                for sgn in [1, -1]:
                    step = base_step * sgn
                    if 0.0 <= h[i] + step <= 1.0 and 0.0 <= h[j] - step <= 1.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            found = True
                        else:
                            h[i] -= step
                            h[j] += step
        if not found:
            base_step *= 0.5
            if base_step < 1e-6:
                break
    return h, best_c

def local_2pt(h, n, deadline, radius=25):
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 4):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 4):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n0 = 200

    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: Full softplus at original resolution (60s)
    t1 = min(t0 + 60, deadline - 800)
    best_h, best_c = lcfgs_opt(best_h, n0, t1,
                                alphas=[1000, 5000, 20000], top_k=5, mu=5.0, use_full=True)

    # Phase 2: Top-k at original resolution (40s)
    t2 = min(t0 + 100, deadline - 760)
    best_h, best_c = lcfgs_opt(best_h, n0, t2,
                                alphas=[50000, 200000, 1000000], top_k=5, mu=10.0)

    # Phase 3: Upsample to n=1000, main L-BFGS optimization (400s)
    n1 = 1000
    h1 = upsample(best_h, len(best_h), n1)
    best_h = h1
    best_c = compute_c5(h1, n1)

    t3 = min(t0 + 520, deadline - 350)
    best_h, best_c = lcfgs_opt(best_h, n1, t3,
                                alphas=[5000, 20000, 100000, 500000, 2000000],
                                top_k=5, mu=10.0, nrestarts=3)

    # Phase 4: Multi-start perturbations at n=1000 (150s)
    t4 = min(t0 + 670, deadline - 250)
    for trial in range(5):
        if time.time() > t4:
            break
        amp = 0.008 * (trial + 1)
        h_try = best_h + amp * rng.randn(n1)
        h_try = project_feasible(h_try, n1)
        h_try, c_try = lcfgs_opt(h_try, n1, min(time.time() + 25, t4),
                                   alphas=[50000, 300000], top_k=5, mu=10.0, nrestarts=1)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try

    # Phase 5: Try n=2000 (130s)
    n2 = 2000
    h2 = upsample(best_h, len(best_h), n2)
    t5 = min(t0 + 800, deadline - 200)
    if time.time() < t5:
        h2, c2 = lcfgs_opt(h2, n2, t5,
                            alphas=[10000, 100000, 500000], top_k=5, mu=10.0, nrestarts=2)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 6: Refine back at n=1000 with higher alpha (80s)
    if len(best_h) != n1:
        best_h = upsample(best_h, len(best_h), n1)
        best_c = compute_c5(best_h, n1)
    t6 = min(t0 + 880, deadline - 120)
    if time.time() < t6:
        best_h, best_c = lcfgs_opt(best_h, n1, t6,
                                    alphas=[500000, 2000000, 5000000],
                                    top_k=3, mu=50.0, nrestarts=2)

    # Phase 7: Gradient-guided local search (70s)
    t7 = min(t0 + 950, deadline - 60)
    if time.time() < t7:
        best_h, best_c = gradient_local(best_h, n1, t7)

    # Phase 8: Fine 2-point search (40s)
    t8 = t0 + budget_s - 10
    if time.time() < t8:
        best_h, best_c = local_2pt(best_h, n1, t8, radius=20)

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
