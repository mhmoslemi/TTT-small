Actually, for the continuous case, let me think of a simpler approach:
- The objective is max_k f_k(h) where f_k(h) = Σ_i h[i](1-h[i+k])*dx
- We want to minimize this subject to 0≤h≤1, Σh=n/2
- This is a non-convex problem (bilinear terms)

But we can use the **max-flow** or **water-filling** analogy:
- If the max is at lag k*, we want to reduce f_{k*}(h)
- The gradient of f_{k*} w.r.t. h[i] is: (1-h[i+k*])*dx - h[i-k*]*dx (with boundary handling)
- We can do gradient descent on the max, projected to the feasible set

This is like:
h ← P_feasible(h - α * grad_f_k*(h))

where k* is the current argmax lag.

This is a subgradient method. It should converge (slowly) to a local optimum.

        # (subgradient of max is convex combination of gradients at active constraints)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_local_search_vec(h_init, n, deadline, max_sweeps=5000, seed=42, n_restarts=3):
    target = n // 2
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    best_c = np.inf
    best_h = h_init.copy()
    rng = np.random.RandomState(seed)

    for restart in range(n_restarts):
        if time.time() > deadline:
            break

        if restart == 0:
            h = np.zeros(n)
            h[np.argsort(h_init)[::-1][:target]] = 1.0
        else:
            h = best_h.copy()
            n_swaps = rng.randint(3, 10)
            ones_list = np.where(h == 1)[0]
            zeros_list = np.where(h == 0)[0]
            for _ in range(n_swaps):
                if len(ones_list) > 0 and len(zeros_list) > 0:
                    a = rng.choice(ones_list)
                    b = rng.choice(zeros_list)
                    h[a], h[b] = 0.0, 1.0
                    ones_list = np.setdiff1d(ones_list, [a])
                    zeros_list = np.setdiff1d(zeros_list, [b])

        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        cur_c = corr.max() * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break

            improved = False
            z_list = rng.permutation(list(zeros)).tolist()

            for i in z_list:
                if time.time() > deadline:
                    break
                if i not in zeros:
                    continue

                flip_i = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                flip_i[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                flip_i[msk] -= h[imn[msk]]

                j_arr = np.array(sorted(ones))
                if len(j_arr) == 0:
                    break

                jm = j_arr[:, None] + m_range[None, :]
                jmn = j_arr[:, None] - m_range[None, :]
                msk_j = (jm >= 0) & (jm < n)
                msk_jn = (jmn >= 0) & (jmn < n)

                h_jm = np.where(msk_j, h[np.clip(jm, 0, n - 1)], 0.0)
                h_jmn = np.where(msk_jn, h[np.clip(jmn, 0, n - 1)], 0.0)

                flip_j = -msk_j * (1.0 - h_jm) + msk_jn * h_jmn

                new_corr = corr[None, :] + flip_i[None, :] + flip_j
                new_corr[:, n - 1] -= 2.0

                cross1 = (n - 1) + (j_arr - i)
                cross2 = (n - 1) + (i - j_arr)
                c1v = (cross1 >= 0) & (cross1 < 2 * n - 1)
                c2v = (cross2 >= 0) & (cross2 < 2 * n - 1)
                idx1 = np.where(c1v)[0]
                idx2 = np.where(c2v)[0]
                new_corr[idx1, cross1[c1v]] += 1.0
                new_corr[idx2, cross2[c2v]] += 1.0

                new_max = new_corr.max(axis=1) * dx
                best_idx = np.argmin(new_max)
                best_val = new_max[best_idx]

                if best_val < cur_c - 1e-15:
                    j = j_arr[best_idx]
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = new_corr[best_idx].copy()
                    cur_c = best_val
                    ones.discard(j)
                    zeros.discard(i)
                    ones.add(i)
                    zeros.add(j)
                    improved = True
                    break

            if not improved:
                break

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

    return best_h, best_c

def subgradient_polish(h, n, deadline, max_iter=50000):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 0.1
    eps = 1e-6

    for it in range(max_iter):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        cur_c = max_val * dx

        # Find active lags (within eps of max)
        active = np.where(corr >= max_val - eps * max_val)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        # Average subgradient over active lags
        grad = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            i_plus = np.arange(n) + lag
            i_minus = np.arange(n) - lag
            msk_p = (i_plus >= 0) & (i_plus < n)
            msk_m = (i_minus >= 0) & (i_minus < n)
            grad[msk_p] += (1.0 - h[i_plus[msk_p]]) * dx
            grad[msk_m] -= h[i_minus[msk_m]] * dx
        grad /= len(active)

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        new_c = compute_c5(h_new, n)

        if new_c < cur_c - 1e-15:
            h = h_new
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            alpha = min(alpha * 1.02, 0.5)
        else:
            alpha = max(alpha * 0.9, 0.001)
            if alpha < 0.002 and it > 10000:
                break

    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        lse = sm + np.log(np.exp(shifted - sm).sum())
        return m + lse / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def round_and_search(h, n, deadline, seed):

    h_bin = np.zeros(n)
    h_bin[np.argsort(h)[::-1][:target]] = 1.0
    hb, cb = binary_local_search_vec(h_bin, n, deadline, max_sweeps=3000, seed=seed, n_restarts=2)
    return hb, cb
