import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    """Snap to binary preserving sum constraint."""
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def swap_delta_c(h, i, j, n):
    """Compute change in correlation vector when h[i]:1->0, h[j]:0->1.
    Returns vector of length 2n-1."""
    d = np.arange(2*n-1) - (n-1)  # lags: -(n-1), ..., 0, ..., n-1
    # Term from h[i] changing: -(1-h[i+d])
    term1 = np.zeros(2*n-1)
    valid_i = (i + d >= 0) & (i + d < n)
    term1[valid_i] = -(1 - h[i + d[valid_i]])
    # Term from h[j] changing: +(1-h[j+d])
    valid_j = (j + d >= 0) & (j + d < n)
    term1[valid_j] += (1 - h[j + d[valid_j]])
    # Term from (1-h[...]) at position i: +h[i-d]
    valid_i2 = (i - d >= 0) & (i - d < n)
    term1[valid_i2] += h[i - d[valid_i2]]
    # Term from (1-h[...]) at position j: -h[j-d]
    valid_j2 = (j - d >= 0) & (j - d < n)
    term1[valid_j2] -= h[j - d[valid_j2]]
    return term1

def sa_binary(h, n, deadline, rng, max_iters=200000):
    """Simulated annealing on binary solutions."""
    h = make_binary(h, n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    best_c = c_all.max()
    curr_c = best_c
    curr_lags = c_all.copy()
    
    T = 0.1 * best_c  # initial temperature
    T_min = 0.001 * best_c
    n_ones = n // 2
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Pick random i with h[i]=1 and j with h[j]=0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Compute change
        delta_c = swap_delta_c(h, i, j, n)
        new_lags = curr_lags + delta_c
        new_max = new_lags.max()
        
        # Acceptance criterion
        if new_max < curr_c or rng.random() < np.exp(-(new_max - curr_c) / T):
            # Accept
            h[i] = 0
            h[j] = 1
            curr_c = new_max
            curr_lags = new_lags
            if new_max < best_c - 1e-14:
                best_c = new_max
                best_h = h.copy()
        
        # Cool down
        cooling = max(T_min, T * (1 - it / max_iters))
        T = cooling
    
    return best_h, best_c

def project_feasible(h, n):

def swap_delta_c(h, i, j, n):
    d = np.arange(2 * n - 1) - (n - 1)
    delta = np.zeros(2 * n - 1)
    valid_i = (i + d >= 0) & (i + d < n)
    delta[valid_i] += -(1.0 - h[i + d[valid_i]])
    valid_j = (j + d >= 0) & (j + d < n)
    delta[valid_j] += (1.0 - h[j + d[valid_j]])
    valid_i2 = (i - d >= 0) & (i - d < n)
    delta[valid_i2] += h[i - d[valid_i2]]
    valid_j2 = (j - d >= 0) & (j - d < n)
    delta[valid_j2] -= h[j - d[valid_j2]]
    return delta

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * x)
        elif lag > 0:

            grad[0:n - lag] += w[k] * (-x[lag:n])

            grad[0:n - m_abs] += w[k] * (1.0 - x[m_abs:n])
            grad[m_abs:n] += w[k] * (-x[0:n - m_abs])
    return grad * dx

            break
        g = smooth_max_grad(h_curr, n, alpha)
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * (g * g)
        m_hat = m_v / (1 - beta1 ** it)
        v_hat = v_v / (1 - beta2 ** it)
        h_new = h_curr - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
        if it % 2000 == 0:
            alpha *= 1.5
            lr *= 0.9
        
        if it > 5000 and it % 2000 == 0:
            c_cur = compute_c5(h_curr, n)
            if c_cur > best_c * 1.05:
                h_curr = best_h.copy()
    
    return best_h, best_c

def momentum_optimize(h, n, deadline, alpha=30000, lr=1.5, momentum=0.9):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    
    for it in range(200000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h_curr, n, alpha)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
        if it % 3000 == 0:
            alpha *= 1.3
            lr *= 0.92
    
    return best_h, best_c

def fine_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lags = compute_all_lags(h, n)
    
    for rnd in range(100):
        if time.time() > deadline:
            break
        improved = False
        k_star = int(np.argmax(lags))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m_abs = abs(lag)
        
        for d_val in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time.time() > deadline:
                break
            for i in range(n - m_abs):
                if time.time() > deadline:
                    break
                j = i + m_abs
                if h[i] > h[j] + 1e-10:
                    d = min(d_val, (h[i] - h[j]) * 0.3)
                    if d < 1e-12:
                        continue
                    h[i] -= d
                    h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        lags = compute_all_lags(h, n)
                        improved = True
                        if improved and rnd > 0:
                            break
                    else:
                        h[i] += d
                        h[j] -= d
                elif h[j] > h[i] + 1e-10:
                    d = min(d_val, (h[j] - h[i]) * 0.3)
                    if d < 1e-12:
                        continue
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        lags = compute_all_lags(h, n)
                        improved = True
                        if rnd > 0:
                            break
                    else:
                        h[i] -= d
                        h[j] += d
            if improved and rnd > 0:
                break
        if not improved:
            # Try all top lags
            top_idx = np.argsort(lags)[-5:]
            for k2 in top_idx:
                if time.time() > deadline:
                    break
                lag2 = int(k2) - (n - 1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for i in range(n - m2):
                    if time.time() > deadline:
                        break
                    j2 = i + m2
                    if h[i] > h[j2] + 1e-8:
                        d = 0.003
                        if d >= h[i] or d >= 1.0 - h[j2]:
                            continue
                        h[i] -= d
                        h[j2] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            lags = compute_all_lags(h, n)
                            improved = True
                        else:
                            h[i] += d
                            h[j2] -= d
                    elif h[j2] > h[i] + 1e-8:
                        d = 0.003
                        if d >= h[j2] or d >= 1.0 - h[i]:
                            continue
                        h[i] += d
                        h[j2] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            lags = compute_all_lags(h, n)
                            improved = True
                        else:
                            h[i] -= d
                            h[j2] += d
            if not improved:
                break
    
    return best_h, best_c

def slsqp_polish(h, n, alpha=100000, maxiter=2000, top_m=None):
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    def obj_grad(x):
        return smooth_max_grad(x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    if top_m is not None:
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        
        def top_obj(x):
            cc = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
            active = cc[top_idx]
            m = active.max()
            shifted = alpha * (active - m)
            mx = shifted.max()
            if mx > 500:
                return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
            return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
        
        def top_grad(x):
            cc = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
            active = cc[top_idx]
            m = active.max()
            shifted = alpha * (active - m)
            mx = shifted.max()
            if mx > 500:
                w = np.exp(shifted - mx)
            else:
                w = np.exp(shifted)
            w = w / w.sum()
            dx = 2.0 / n
            grad = np.zeros(n)
            for idx in range(len(top_idx)):
                k = top_idx[idx]
                if w[idx] < 1e-10:
                    continue
                lag = k - (n - 1)
                if lag == 0:
                    grad += w[idx] * (1.0 - 2.0 * x)
                elif lag > 0:
                    grad[lag:n] += w[idx] * (1.0 - x[0:n - lag])
                    grad[0:n - lag] += w[idx] * (-x[lag:n])
                else:
                    m_abs = -lag
                    grad[0:n - m_abs] += w[idx] * (1.0 - x[m_abs:n])
                    grad[m_abs:n] += w[idx] * (-x[0:n - m_abs])
            return grad * dx
        
        try:
            res = minimize(top_obj, h, jac=top_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()
    else:
        try:
            res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Continuous optimization from initial (30% budget)
    if h_init is not None:
        h_start = project_feasible(h_init, n_init)
    else:
        h_start = np.full(n_init, 0.5)
    
    alloc1 = min(250, (deadline - t0) * 0.28)
    h_phase1, c_phase1 = momentum_optimize(h_start, n_init, t0 + alloc1, alpha=20000, lr=1.5, momentum=0.92)
    best_h, best_c = h_phase1, c_phase1
    
    alloc1b = min(80, (deadline - t0) * 0.1)
    if time.time() < deadline - 60:
        h_phase1b, c_phase1b = adam_optimize(best_h, n_init, time.time() + alloc1b, alpha=50000, lr=0.4)
        if c_phase1b < best_c:
            best_c = c_phase1b
            best_h = h_phase1b
    
    # Phase 2: Binary optimization (40% budget)
    n = len(best_h)
    h_bin = make_binary(best_h, n)
    c_bin = compute_c5(h_bin, n)
    
    alloc2 = min(400, (deadline - t0) * 0.38)
    if time.time() < deadline - 100:
        # Directed local search
        dl_deadline = min(time.time() + alloc2 * 0.3, deadline - 150)
        h_dl, c_dl, h_dl_curr, lags_dl = directed_binary_ls(h_bin, n, dl_deadline, max_rounds=5000)
        if c_dl < best_c:
            best_c = c_dl
            best_h = h_dl
    
    # Exponentiated gradient binary search
    if time.time() < deadline - 80:
        alloc2b = min(200, (deadline - time.time()) * 0.35)
        h_eg, c_eg = expgard_binary_search(h_bin, n, time.time() + alloc2b, rng, max_iters=400000)
        if c_eg < best_c:
            best_c = c_eg
            best_h = h_eg
    
    # SA for escape
    if time.time() < deadline - 60:
        alloc2c = min(150, (deadline - time.time()) * 0.3)
        h_sa, c_sa = sa_binary(h_bin, n, time.time() + alloc2c, rng, max_iters=300000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
    
    # Phase 3: Try different resolutions
    n_candidates = [n_init]
    for nc in [n_init - 10, n_init + 10, n_init - 20, n_init + 20, n_init + 50]:
        if nc >= 100 and nc != n_init:
            n_candidates.append(nc)
    
    for n_try in n_candidates[:4]:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 40
        if remaining < 80:
            break
        
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc_res = min(100, remaining * 0.25)
        if alloc_res < 40:
            continue
        
        # Quick momentum optimization
        h_r, c_r = momentum_optimize(h_interp, n_try, time.time() + alloc_res * 0.6, alpha=30000, lr=1.0)
        
        # Binary snap and quick LS
        h_rb = make_binary(h_r, n_try)
        if time.time() < deadline - 50:
            h_rb2, c_rb2 = directed_binary_ls(h_rb, n_try, time.time() + min(30, (deadline - time.time()) * 0.2))
            if c_rb2 < c_r:
                h_r = h_rb2
                c_r = c_rb2
        
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    
    # Phase 4: Final polish (remaining time)
    if best_h is not None and time.time() < deadline - 15:
        n_final = len(best_h)
        
        # SLSQP polish
        if time.time() < deadline - 30:
            h_sl = slsqp_polish(best_h, n_final, alpha=200000, maxiter=2000, top_m=5)
            c_sl = compute_c5(h_sl, n_final)
            if c_sl < best_c:
                best_c = c_sl
                best_h = h_sl
        
        # Fine local search
        if time.time() < deadline - 10:
            best_h, best_c = fine_local_search(best_h, n_final, deadline - 3)
        
        # Final binary snap + quick LS
        if time.time() < deadline - 5:
            h_fb = make_binary(best_h, n_final)
            h_fb2, c_fb2 = directed_binary_ls(h_fb, n_final, deadline - 2, max_rounds=500)
            if c_fb2 < best_c:
                best_c = c_fb2
                best_h = h_fb2
    
    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
