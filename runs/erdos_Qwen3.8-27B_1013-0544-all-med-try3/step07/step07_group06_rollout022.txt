        a = rng.choice(ones)
        b = rng.choice(zeros)
        
        # Compute delta for swap a->0, b->1
        m_range = np.arange(-(n-1), n)
        delta = np.zeros(2 * n - 1)
        im = b + m_range  # b becomes 1
        imn = b - m_range
        jm = a + m_range  # a becomes 0
        jmn = a - m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]

        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n - 1 + (b - a)
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0

        new_c = new_corr.max() * dx
        delta_c = new_c - cur_c

            corr = new_corr

            if cur_c < best_c:
