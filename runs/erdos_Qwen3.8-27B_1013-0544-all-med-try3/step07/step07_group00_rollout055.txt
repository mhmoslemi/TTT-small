    # with the constraint sum(h) = n/2.

        grad += 2.0 * 0.01 * (x.sum() - n/2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def meank_obj(x, n, k):
    c = compute_c5_vec(x, n)
    sorted_c = np.sort(c)[::-1]
    val = np.mean(sorted_c[:k])
    val += 0.001 * (x.sum() - n / 2.0) ** 2
    return float(val)

def meank_grad(x, n, k):
    c = compute_c5_vec(x, n)
    top_idx = np.argsort(c)[::-1][:k]
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in top_idx:
        kk = idx - (n - 1)
        wm = dx / k
        if kk >= 0:

def lbfgs_run(h, n, deadline, configs):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(full_obj, h, jac=full_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 50})

            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def upsample(h_old, n_old, n_new):

    return project_feasible(h_new, n_new)
