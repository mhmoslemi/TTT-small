def fast_binary_optimize(h_binary, n, deadline):
    """
    Optimize a binary sequence h_binary of length n (with n/2 ones)
    to minimize max overlap * 2/n.
    """
    dx = 2.0 / n
    h = h_binary.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Local search: try all swaps, keep improvements
    # For efficiency, compute the effect of a swap analytically
    
    while time.time() < deadline:
        improved = False
        for i in zeros.tolist():
            if time.time() > deadline:
                return best_h, best_max * dx
            for j in ones.tolist():
                # Compute effect of swapping i and j
                delta = compute_swap_delta(h, i, j, n)
                new_max = (corr + delta).max()
                if new_max < cur_max - 1e-12:

    # i: position changing 0→1
    # j: position changing 1→0
    # k: lag
    # Returns the change in corr[n-1+k]
    
    delta = 0.0
    
    # Term a=i (if i+k valid)
    if 0 <= i+k < n:
        old = h[i] * (1-h[i+k])
        new_h_i = 1.0
        new_h_i_k = 0.0 if (i+k == j) else h[i+k]
        new = new_h_i * (1-new_h_i_k)
        delta += new - old
    
    # Term a=j (if j+k valid)
    if 0 <= j+k < n:

        new_h_j_k = 1.0 if (j+k == i) else h[j+k]
