def compute_c5(h, n):

def compute_all_lags(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

For a move (h[i] -= δ, h[j] += δ), the change in L_k is:
ΔL_k ≈ -δ * ∂L_k/∂h[i] + δ * ∂L_k/∂h[j] = δ * (∂L_k/∂h[j] - ∂L_k/∂h[i])
