                        active_lag_vals = c_all[active_lags].copy()
                        stall = 0
                    else:
                        # Revert if too bad
                        if c_new > best_c + 0.005:
                            h = best_h.copy()
                            c_all = compute_all_lags(h, n)
                            sorted_idx = np.argsort(c_all)[::-1]
                            active_lags = sorted_idx[:top_k].copy()
                            active_lag_vals = c_all[active_lags].copy()
    
    return best_h, best_c

def binary_local_search(h, n, deadline, rng):
    """Work with near-binary solutions using combinatorial swaps."""
    h = h.copy()
    threshold = 0.5
    h_bin = (h >= threshold).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        # Remove excess: set the smallest h values among 1s to 0
        ones_idx = np.where(h_bin == 1)[0]
        excess = n_ones - n // 2
        remove = ones_idx[np.argsort(h[ones_idx])[:excess]]
        h_bin[remove] = 0
    elif n_ones < n // 2:
        zeros_idx = np.where(h_bin == 0)[0]
        deficit = n // 2 - n_ones
        add = zeros_idx[np.argsort(-h[zeros_idx])[:deficit]]
        h_bin[add] = 1
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    c_all = compute_all_lags(h_bin, n)
    
    for it in range(5000000):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        
        improved = False
        # Try swapping a 1 and 0
        for _ in range(50):
            if time.time() > deadline:
                break
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            if h_bin[i] == 0 and h_bin[j] == 1:
                h_bin[i] = 1
                h_bin[j] = 0
                c_new = compute_c5(h_bin, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_bin.copy()
                    c_all = compute_all_lags(h_bin, n)
                    improved = True
                else:
                    h_bin[i] = 0
                    h_bin[j] = 1
        if not improved and it % 1000 == 0 and it > 5000:
            # Try targeted: for the worst lag, try to fix specific positions
            if lag > 0:
                # S_lag = sum h[j+m]*(1-h[j]), want to reduce
                # Find j where h[j+m]=1 and h[j]=0, try swapping with nearby 0/1
                for j in range(n - m):
                    if h_bin[j+m] == 1 and h_bin[j] == 0:
                        # Find a position p where h_bin[p]=1 and h_bin[p+m]=0
                        for p in range(n - m):
                            if p != j and h_bin[p] == 1 and h_bin[p+m] == 0:
                                h_bin[j] = 1
                                h_bin[p] = 0
                                c_new = compute_c5(h_bin, n)
                                if c_new < best_c - 1e-14:
                                    best_c = c_new
                                    best_h = h_bin.copy()
                                    c_all = compute_all_lags(h_bin, n)
                                    improved = True
                                else:
                                    h_bin[j] = 0
                                    h_bin[p] = 1
                                break
                        if improved:
                            break
                if improved:
                    continue
    
    return best_h, best_c

def interpolate_to(h_from, n_from, n_to):
    x_from = np.linspace(0, 2, n_from)
    x_to = np.linspace(0, 2, n_to)
    h_new = np.interp(x_to, x_from, h_from)
    return project_feasible(h_new, n_to)

def optimize_at_resolution(h_start, n, deadline, rng):
    """Full optimization pipeline at a single resolution."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase A: Gradient optimization
    grad_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.3)
    h_g, c_g = gradient_optimize(h, n, grad_deadline)
    if c_g < best_c:
        best_c = c_g
        best_h = h_g
    h = h_g
    
    # Phase B: Fast local search
    ls_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.6)
    h_ls, c_ls = fast_local_search(h, n, ls_deadline, rng)
    if c_ls < best_c:
        best_c = c_ls
        best_h = h_ls
    
    # Phase C: SLSQP polish
    if time.time() < deadline - 5:
        try:
            dx = 2.0 / n
            def obj(x):
                c = np.correlate(x, 1 - x, mode='full') * dx
                m = c.max()
                alpha = 10000
                shifted = alpha * (c - m)
                sm = shifted.max()
                if sm > 500:
                    return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
                return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
            
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
            h_slsqp = project_feasible(res.x, n)
            c_slsqp = compute_c5(h_slsqp, n)
            if c_slsqp < best_c:
                best_c = c_slsqp
                best_h = h_slsqp
        except Exception:
            pass
    
    # Phase D: Final local search
    if time.time() < deadline - 2:
        h_fin, c_fin = fast_local_search(best_h, n, deadline - 1, rng)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Load initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    
    # Phase 1: Quick gradient polish at initial resolution
    p1_deadline = t0 + min(80, (deadline - t0) * 0.1)
    h_g, c_g = gradient_optimize(h_init, n_init, p1_deadline)
    if c_g < best_c:
        best_c = c_g
        best_h = h_g
    
    # Phase 2: Try higher resolutions
    n_candidates = [600, 700, 800]
    if n_init > 500:
        n_candidates = [n_init + 100, n_init + 200]
    
    for n_new in n_candidates:
        if time.time() > deadline - 150:
            break
        remaining = deadline - time.time() - 100
        if remaining < 80:
            break
        alloc = min(150, remaining * 0.4)
        if alloc < 50:
            break
        h_interp = interpolate_to(best_h, len(best_h), n_new)
        h_opt, c_opt = optimize_at_resolution(h_interp, n_new, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Also try the original resolution with more time
    if time.time() < deadline - 120:
        remaining = deadline - time.time() - 80
        alloc = min(120, remaining * 0.4)
        if alloc > 40:
            h_opt, c_opt = optimize_at_resolution(best_h, len(best_h), time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 3: Intensive local search at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 20:
        ls_time = min(deadline - time.time() - 10, 400)
        if ls_time > 30:
            h_ls, c_ls = fast_local_search(best_h, n_best, time.time() + ls_time, rng, top_k=7)
            if c_ls < best_c:
                best_c = c_ls
                best_h = h_ls
    
    # Phase 4: Try binary solution
    if time.time() < deadline - 30:
        h_bin_start = (best_h >= 0.5).astype(float)
        n_ones = int(h_bin_start.sum())
        if n_ones > n_best // 2:
            ones_idx = np.where(h_bin_start == 1)[0]
            excess = n_ones - n_best // 2
            h_bin_start[ones_idx[np.argsort(best_h[ones_idx])[:excess]]] = 0
        elif n_ones < n_best // 2:
            zeros_idx = np.where(h_bin_start == 0)[0]
            deficit = n_best // 2 - n_ones
            h_bin_start[zeros_idx[np.argsort(-best_h[zeros_idx])[:deficit]]] = 1
        c_bin = compute_c5(h_bin_start, n_best)
        if c_bin < best_c:
            # Binary is better, polish it
            bin_deadline = time.time() + min(60, deadline - time.time() - 15)
            h_bp, c_bp = fast_local_search(h_bin_start, n_best, bin_deadline, rng, top_k=5)
            if c_bp < best_c:
                best_c = c_bp
                best_h = h_bp
    
    # Phase 5: Multistart with perturbations
    if time.time() < deadline - 30:
        n_restarts = max(1, int((deadline - time.time() - 20) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.03 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(45, deadline - time.time() - 20)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 6: Final polish
    if best_h is not None and time.time() < deadline - 5:
        n_fin = len(best_h)
        h_fin, c_fin = fast_local_search(best_h, n_fin, deadline - 2, rng, top_k=5)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    
    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
