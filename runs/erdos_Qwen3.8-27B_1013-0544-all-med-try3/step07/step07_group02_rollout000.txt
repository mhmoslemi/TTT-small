import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-6:
            continue
        lag = k - (n-1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def grad_top_lags(h, n, top_count):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_count:]
    grad = np.zeros(n)
    for k in top_idx:
        lag = k - (n-1)
        grad += _lag_gradient(h, n, lag)
    return grad * dx / top_count

def nesterov_optimize(h, n, deadline, top_count=5, max_iter=5000):
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = 0.1
    gamma = 0.9
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h_curr + gamma * (h_curr - h_prev)
        y = project_feasible(y, n)
        g = grad_top_lags(y, n, top_count)
        h_new = project_feasible(h_curr - lr * g, n)
        c_new = compute_c5(h_new, n)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]

def slsqp_top(h, n, top_m, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_active_sum_obj, h, args=(n, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

        h_start = project_feasible(h_start, n_init)
        alloc = min(150, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_pipeline(h_start, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Try higher resolutions
    n_candidates = [600, 700, 800]
    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 30
        if remaining < 80:
            break
        alloc = min(150, remaining * 0.30)
        if alloc < 50:
            break
        h_start = interpolate_h(best_h, len(best_h), n_try)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_pipeline(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Also try n=400 (sometimes lower resolution works better due to different discretization)
    if time.time() < deadline - 80:
        n_try = 400
        h_start = interpolate_h(best_h, len(best_h), n_try)
        h_start = project_feasible(h_start, n_try)
        alloc = min(80, (deadline - time.time() - 30) * 0.25)
        if alloc > 40:
            h_opt, c_opt = optimize_pipeline(h_start, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 50:
        remaining = deadline - time.time() - 25
        n_restarts = max(1, int(remaining / 50))
        n_restarts = min(n_restarts, 5)
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(45, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_pipeline(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final intensive polishing on best solution
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # SLSQP with various top_m
        if time.time() < deadline - 15:
            for tm in [2, 3, 5]:
                if time.time() > deadline - 8:
                    break
                h_new = slsqp_top(best_h, n, tm, maxiter=2000, ftol=1e-16)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
        # Final SLSQP smooth
        if time.time() < deadline - 12:
            h_new = slsqp_smooth(best_h, n, 3, 50000, maxiter=2000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        # Final local search with longer budget
        if time.time() < deadline - 3:
            best_h, best_c = local_search_efficient(best_h, n, deadline - 1)
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
    
    if best_h is None:
        n_fb = 500
        best_h = project_feasible(np.full(n_fb, 0.5), n_fb)
        best_c = compute_c5(best_h, n_fb)
    
    return best_h, best_c, len(best_h)
