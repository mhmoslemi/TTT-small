from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent_ternary(h, n, max_rounds, deadline, inner_iters=20):
    """Coordinate descent with ternary search for each coordinate."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            old_val = h[i]
            lo, hi = 0.0, 1.0
            # Ternary search
            for _ in range(inner_iters):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[i] = m1
                f1 = compute_c5(h, n)
                h[i] = m2
                f2 = compute_c5(h, n)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            h[i] = (lo + hi) / 2
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
            else:
                h[i] = old_val
        if not improved:
            break
    return h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)
