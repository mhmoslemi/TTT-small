    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    target_sum = n / 2.0

            if tot < 1e-14:

def _lag_gradient(h, n, lag):

def binary_to_continuous(h_bin, n, frac=0.15, rng=None):

def simulated_annealing(h, n, deadline, rng, t0=0.1, cooling=0.9995, max_rounds=20000):

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        T *= cooling
        if T < 0.001:
            T = 0.001
        
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)

            break

        if m >= n:
            continue
        
        i = rng.randint(0, n - m)
        j = i + m
        
        # Choose move size
        d = rng.uniform(0.01, 0.08) * rng.choice([-1, 1])
        
        # Ensure feasibility
        if h[i] + d < -0.01 or h[i] + d > 1.01:
            continue
        if h[j] - d < -0.01 or h[j] - d > 1.01:
            continue
        
        h[i] += d
        h[j] -= d
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        
        delta = c_new - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d 
            h = project_feasible(h, n)
    
    return best_h, best_c

        alloc_a = min(150, (deadline - t0_global) * 0.15)
        h_a, c_a = optimize_at_resolution(h_start, n_init, t0_global + alloc_a, rng)
        best_h, best_c = h_a, c_a
    
    # Phase B: Higher resolution optimization
    # Try n=800 and n=1000
    n_high_candidates = [800, 1000]
    for n_try in n_high_candidates:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 40
        if remaining < 80:
            break
        alloc_b = min(remaining * 0.55, 350)
        if alloc_b < 50:
            break
        h_start_hi = interpolate_h(best_h, len(best_h), n_try)
        h_hi, c_hi = optimize_at_resolution(h_start_hi, n_try, time.time() + alloc_b, rng)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
    
    # Phase C: Binary initialization at high resolution
    if time.time() < deadline - 80:
        n_bin = max(800, len(best_h))
        # Create binary from best solution
        h_bin = np.zeros(n_bin)
        x_best = np.linspace(0, 2, len(best_h))
        x_bin = np.linspace(0, 2, n_bin)
        h_interp = np.interp(x_bin, x_best, best_h)
        idx = np.argsort(h_interp)[::-1][:n_bin // 2]
        h_bin[idx] = 1.0
        h_bin = project_feasible(h_bin, n_bin)
        
        # Relax binary
        h_relaxed = binary_to_continuous(h_bin, n_bin, frac=0.1, rng=rng)
        
        alloc_c = min(200, (deadline - time.time() - 30) * 0.6)
        if alloc_c > 40:
            h_c, c_c = optimize_at_resolution(h_relaxed, n_bin, time.time() + alloc_c, rng)
            if c_c < best_c:
                best_c = c_c
                best_h = h_c
    
    # Phase D: Multi-start at best resolution
    n_best = len(best_h)
    remaining = deadline - time.time() - 20
    if remaining > 40:
        n_restarts = max(1, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            # Different perturbation strategies
            if r == 0:
                hp = best_h + rng.normal(0, 0.03, n_best)
            elif r == 1:
                hp = best_h + rng.normal(0, 0.05, n_best)
            elif r == 2:
                # Try binary-like perturbation
                hp = best_h.copy()
                mask = rng.random(n_best) > 0.9
                hp[mask] += 0.2 * rng.choice([-1, 1], mask.sum())
            else:
                # Interpolate between best and random
                h_rand = project_feasible(rng.uniform(0, 1, n_best), n_best)
                hp = 0.7 * best_h + 0.3 * h_rand
            
            hp = project_feasible(hp, n_best)
            alloc_r = min(50, (deadline - time.time() - 20) * 0.35)
            if alloc_r < 20:
                break
            h_r, c_r = optimize_at_resolution(hp, n_best, time.time() + alloc_r, rng)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    
    # Phase E: Final intensive polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Very focused SLSQP
        if time.time() < deadline - 10:
            h_new = slsqp_minimax(best_h, n, deadline, top_m=3, maxiter=5000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Final gradient with very high alpha
        if time.time() < deadline - 5:
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-2:]
            h_new, c_new = gradient_descent(best_h, n, alpha=50000,
                                             deadline=min(deadline - 1, time.time() + 15),
                                             lr=0.3, momentum=0.96, top_idx=top_idx, max_iter=5000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Final local search
        if time.time() < deadline - 2:
            h_new, c_new = targeted_local_search(best_h, n, deadline - 1, rng, max_rounds=3000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
