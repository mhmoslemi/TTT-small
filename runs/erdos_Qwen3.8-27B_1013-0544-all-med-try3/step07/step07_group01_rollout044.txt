def grad_direction_search(h, n, deadline, alpha=5000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.002
    for _ in range(5000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(best_h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step = lr
        improved = False
        for _ in range(50):
            h_new = project_feasible(best_h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_c

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Quick L-BFGS polish at original resolution
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_iters=300)

    # Phase 2: SLSQP at original resolution
    if time.time() < deadline - 400:
        t2 = min(time.time() + 80, deadline - 400)
        for a in [500000, 2000000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_optimize(best_h, best_n, min(time.time() + 40, t2), alpha=a, maxiter=2000)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Try higher resolutions
    for n_try in [400, 500, 600]:
        if time.time() > deadline - 200:
            break
        h_interp = upsample_h(best_h, best_n, n_try)
        alloc = min(120, deadline - time.time() - 100)
        if alloc < 30:
            break
        h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.5, max_iters=250)
        if time.time() < deadline - 60:
            h_t, c_t = slsqp_optimize(h_t, n_try, min(time.time() + min(30, deadline - time.time() - 40), 60),
                                       alpha=2000000, maxiter=1500)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 4: Gradient direction search
    if time.time() < deadline - 100:
        t4 = min(time.time() + 60, deadline - 100)
        h4, c4 = grad_direction_search(best_h, best_n, t4, alpha=10000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Fine pair search
    if time.time() < deadline - 40:
        t5 = min(time.time() + 100, deadline - 40)
        best_h, best_c = fine_pair_search(best_h, best_n, t5)

    # Phase 6: Multi-start with Fourier perturbations
    if time.time() < deadline - 60:
        for amp in [0.05, 0.03, 0.01]:
            if time.time() > deadline - 40:
                break
            for trial in range(3):
                if time.time() > deadline - 30:
                    break
                h_p = fourier_perturb(best_h, best_n, amp, rng)
                alloc = min(25, deadline - time.time() - 15)
                if alloc < 8:
                    break
                h_pp, c_pp = lbfgs_optimize(h_p, best_n, time.time() + alloc, max_iters=100)
                if c_pp < best_c:
                    best_c = c_pp
                    best_h = h_pp

    # Phase 7: Final SLSQP + pair search
    if time.time() < deadline - 15:
        t7 = min(time.time() + 20, deadline - 10)
        h7, c7 = slsqp_optimize(best_h, best_n, t7, alpha=5000000, maxiter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    if time.time() < deadline - 3:
        best_h, best_c = fine_pair_search(best_h, best_n, deadline - 1)

    # Final: also try downsampled version if n > 300
    if best_n > 300 and time.time() < deadline - 5:
        for ns in [300, 350, 400, 450]:
            if ns >= best_n:
                continue
            if time.time() > deadline - 3:
                break
            h_ds = upsample_h(best_h, best_n, ns)
            c_ds = compute_c5(h_ds, ns)
            if c_ds < best_c:
                best_c = c_ds
                best_h = h_ds
                best_n = ns

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
