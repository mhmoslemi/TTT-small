import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_run(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj_fn(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = np.max(g)
        sh = alpha * (g - m)
        sm = np.max(sh)
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def con_fn(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def swap_delta(corr, h, i, j, n, m_range):
    """Compute correlation delta for swapping h[i]=0->1, h[j]=1->0."""
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range

    delta = np.zeros(2 * n - 1)
    msk = (im >= 0) & (im < n)
    delta[msk] += (1.0 - h[im[msk]])
    msk2 = (imn >= 0) & (imn < n)
    delta[msk2] -= h[imn[msk2]]
    msk3 = (jm >= 0) & (jm < n)
    delta[msk3] -= (1.0 - h[jm[msk3]])
    msk4 = (jmn >= 0) & (jmn < n)
    delta[msk4] += h[jmn[msk4]]

    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    k2 = n - 1 + (i - j)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

                best_c = cb
                best_h = hb.copy()
                best_n = nh

    return best_h, best_n, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Load initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: SLSQP warm-up (8% of budget)
    p1 = t0 + 0.08 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [3000, 15000, 80000, 400000]:
        if time.time() > p1:
            break
        h = slsqp_run(h, n, alpha, 800, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Primary binary search (30% of budget)
    p2 = t0 + 0.38 * (budget_s - 5)
    hb, cb = binary_search_enhanced(best_h, n, p2,
                                    max_sweeps=3000, n_restarts=4,
                                    use_2swap=True, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: SLSQP from binary solution (8%)
    p3 = t0 + 0.46 * (budget_s - 5)
    if time.time() < p3:
        h = project_feasible(best_h + 0.04 * rng.randn(n), n)
        for alpha in [30000, 200000, 1000000]:
            if time.time() > p3:
                break
            h = slsqp_run(h, n, alpha, 1200, p3)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 4: Multi-resolution exploration (18%)
    p4 = t0 + 0.64 * (budget_s - 5)
    if time.time() < p4:
        best_h, n, best_c = multi_resolution_try(best_h, n, best_c, p4, seed + 10)

    # Phase 5: Extended binary search on best (22%)
    p5 = t0 + 0.86 * (budget_s - 5)
    hb, cb = binary_search_enhanced(best_h, n, p5,
                                    max_sweeps=8000, n_restarts=4,
                                    use_2swap=True, seed=seed + 200)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 6: Final polish (8%)
    if time.time() < dl - 5:
        h = slsqp_run(best_h + 0.01 * rng.randn(n), n, 3000000, 2000, dl - 10)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

        if time.time() < dl - 3:
            hb, cb = binary_search_enhanced(best_h, n, dl,
                                            max_sweeps=500, n_restarts=3,
                                            use_2swap=True, seed=seed + 500)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
