import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
import math

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max_val(c, alpha)

def active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    return smooth_max_val(active, alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, top_idx, alpha, maxiter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(active_smooth_obj, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_grad(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[:n - m] += (1.0 - h[m:]) * dx
        grad[m:] -= h[:n - m] * dx
    else:
        grad[m:] += (1.0 - h[:n - m]) * dx
        grad[:n - m] -= h[m:] * dx
    return grad

def outer_approx_step(h_curr, n, trust, top_m=30):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    sorted_idx = np.argsort(c)[::-1]
    active_idx = sorted_idx[:top_m]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(h_var - h_curr, np.inf) <= trust]
    for k in active_idx:
        lag = k - (n - 1)
        grad = lag_grad(h_curr, n, lag)
        cons.append(c[k] + grad @ (h_var - h_curr) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.SCIP, time_limit=30)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, time_limit=30)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(np.asarray(h_new), n)

def outer_approx_loop(h, n, deadline, max_iters=20, top_m=30):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        h_new = outer_approx_step(h_curr, n, trust, top_m)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.01, trust * 0.85)
    return best_h, best_c

def gradient_local_search(h, n, deadline, max_rounds=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_all_lags(h, n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for d in step_sizes:
        if time.time() > deadline:
            break
        for rnd in range(max_rounds):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            grad = lag_grad(h, n, lag)
            # Find best single-variable move
            best_improve = 0
            best_i = -1
            best_dir = 0
            # For lag > 0: increasing h[i] for i < n-m helps if (1-h[i+lag]) > 0
            # and increasing h[i] for i >= m hurts if h[i-lag] > 0
            if lag > 0:
                for i in range(n):
                    if i < n - m:
                        # h[i] appears as (1-h[i]) * h[i+lag] -> decreasing h[i] helps
                        change = -grad[i]
                    elif i >= m:
                        change = -grad[i]
                    else:
                        continue
                    if change > best_improve:
                        best_improve = change
                        best_i = i
                        best_dir = -1
                for i in range(n):
                    if i < n - m:
                        change = -grad[i]
                    elif i >= m:
                        change = -grad[i]
                    else:
                        continue
                    if change < best_improve - 2 * best_improve if best_improve > 0 else False:
                        pass
            else:
                for i in range(n):
                    change = -grad[i]
                    if change > best_improve:
                        best_improve = change
                        best_i = i
                        best_dir = -1
            if best_i < 0 or best_improve < 1e-12:
                break
            delta = d * best_dir
            new_val = h[best_i] + delta
            if new_val < 0 or new_val > 1:
                delta = -delta
                new_val = h[best_i] + delta
                if new_val < 0 or new_val > 1:
                    break
            h[best_i] += delta
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
            else:
                h[best_i] -= delta
                h = project_feasible(h, n)
    return h, best_c

def pair_swap_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 1e-4, 5e-5]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-8:
                    delta = min(d, (h[i] - h[j]) * 0.3)
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-8:
                    delta = min(d, (h[j] - h[i]) * 0.3)
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def snap_01(h, n):
    """Snap values close to 0 or 1, then fix sum."""
    h2 = h.copy()
    h2[h2 < 0.02] = 0.0
    h2[h2 > 0.98] = 1.0
    h2 = project_feasible(h2, n)
    return h2

def optimize_n(h_start, n, deadline):

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase A: SLSQP with increasing alpha
    for alpha in [500, 5000, 50000]:

        if c_try < best_c:
            best_c = c_try
            best_h = h_try
    
    # Phase B: Active set optimization
    if time.time() < deadline - 30:
        for top_m in [5, 10, 15]:
            if time.time() > deadline - 30:

            c_lags = compute_all_lags(best_h, n)
            top_idx = np.argsort(c_lags)[-top_m:]
            for alpha in [10000, 100000]:
                if time.time() > deadline - 30:
                    break
                h_try = slsqp_active(best_h, n, top_idx, alpha, maxiter=2000, ftol=1e-13)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try
    
    # Phase C: Outer approximation
    if time.time() < deadline - 40:

            best_h = h_try
