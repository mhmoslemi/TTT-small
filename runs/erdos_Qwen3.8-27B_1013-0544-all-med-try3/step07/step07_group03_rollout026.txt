import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, alpha, max_sweeps=300, tol=1e-11, deadline=None):
    h = h.copy()
    n_corr = 2 * n - 1
    prev_obj = compute_c5(h, n)
    stagnant = 0
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        v = 1.0 - h
        corr = np.correlate(h, v, mode='full')
        
        for j in range(n):
            m_start = j
            m_end = min(j + n, n_corr)
            if m_end <= m_start:
                continue
            h_idx = np.arange(m_start, m_end) - j
            
            a = np.zeros(n_corr)
            a[m_start:m_end] = 1.0 - 2.0 * h[h_idx]
            hj = h[j]
            b = corr - a * hj
            
            lo, hi = 0.0, 1.0
            for _ in range(45):
                mid = (lo + hi) * 0.5
                z = alpha * (a * mid + b)
                z_max = z.max()
                p = np.exp(z - z_max)
                p_sum = p.sum()
                if p_sum < 1e-300:
                    break
                deriv = np.dot(a, p) / p_sum
                if deriv > 0:
                    lo = mid
                else:
                    hi = mid
            
            h[j] = (lo + hi) * 0.5
        
        h = project_feasible(h, n)
        curr_obj = compute_c5(h, n)
        
        if curr_obj < prev_obj - tol:
            prev_obj = curr_obj
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 3:
                break
    
    return h, compute_c5(h, n)

def pair_search(h, n, deadline, deltas):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:

            if not improved:

    return h, best_c
