    """Compute change in c(k) when h[a]: 1->0 and h[b]: 0->1."""
    dx = 2.0 / n
    affected_i = set()
    for idx in [a, b, a-k, b-k]:
        if 0 <= idx <= n-1-k:
            affected_i.add(idx)
    
    delta = 0.0
    for i in affected_i:
        ik = i + k
        # old value
        old_ha = h[a] if a != b else h[a]  # old_h[a] = 1
        old_hb = 0.0  # old_h[b] = 0
        new_ha = 0.0
        new_hb = 1.0
        
        def get_old(idx):
            if idx == a: return 1.0
            if idx == b: return 0.0
            return h[idx]
        
        def get_new(idx):
            if idx == a: return 0.0
            if idx == b: return 1.0
            return h[idx]
        
        old_term = get_old(i) * (1 - get_old(ik))
        new_term = get_new(i) * (1 - get_new(ik))
        delta += new_term - old_term
    
    return delta * dx

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    binary = (h > 0.5).astype(float)
    target = n / 2.0
    deficit = target - binary.sum()
    if deficit > 0.5:
        zeros = np.where(binary == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for idx in order[:int(deficit)]:
            binary[idx] = 1.0
    elif deficit < -0.5:
        ones = np.where(binary == 1)[0]
        order = ones[np.argsort(h[ones])]
        for idx in order[:int(-deficit)]:
            binary[idx] = 0.0
    # Handle the fractional part
    frac = deficit - int(deficit)
    if frac > 0.01:
        zeros = np.where(binary == 0)[0]
        if len(zeros) > 0:
            best_z = zeros[np.argmax(h[zeros])]
            binary[best_z] = frac
    elif frac < -0.01:
        ones = np.where(binary == 1)[0]
        if len(ones) > 0:
            best_o = ones[np.argmin(h[ones])]
            binary[best_o] = 1.0 + frac
    return project_feasible(binary, n)

def swap_delta_c(k, a, b, h, n):
    """Compute exact change in c(k) when h[a]: 1->0, h[b]: 0->1. h is pre-swap state."""
    affected_i = []
    for idx in [a, b, a - k, b - k]:
        if 0 <= idx <= n - 1 - k:
            if idx not in affected_i:
                affected_i.append(idx)
    
    def get_old(idx):
        if idx == a:
            return 1.0
        if idx == b:
            return 0.0
        if 0 <= idx < n:
            return h[idx]
        return 0.0
    
    def get_new(idx):
        if idx == a:
            return 0.0
        if idx == b:
            return 1.0
        if 0 <= idx < n:
            return h[idx]
        return 0.0
    
    delta = 0.0
    for i in affected_i:
        ik = i + k
        old_term = get_old(i) * (1.0 - get_old(ik))
        new_term = get_new(i) * (1.0 - get_new(ik))
        delta += new_term - old_term
    return delta

        if not active_ks:

def targeted_swap_search(h, n, deadline, max_rounds=100):

def _obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 300:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(shifted)))) / alpha)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(len(x))

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coordinate_descent(h, n, deadline, max_sweeps=20):
    """Coordinate descent: for each i, try to move h[i] toward 0 or 1."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            # Try pushing toward 0
            if h[i] > 1e-6:
                h_old = h[i]
                h[i] = max(0.0, h[i] - 0.3)
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                else:
                    h = project_feasible(h, n)
            # Try pushing toward 1
            if h[i] < 1 - 1e-6:
                h[i] = min(1.0, h[i] + 0.3)
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                else:
                    h = project_feasible(h, n)
        if not improved:

    return h, best_c

def pair_search_h(h, n, deadline, deltas, radius=None):
    """Pair search with optional radius for large n."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:

                    return h, best_c

                    hi = min(n, i + radius + 1)
                # Also check backward

        h_opt, c_opt = optimize_candidate(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 4: Multi-restart with perturbation
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.05 + 0.1 * rng.random()
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_candidate(hp, n, time.time() + alloc, rng)
            if c_opt < best_c - 1e-14:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Extended pair search on best
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if n <= 150:
            best_h, best_c = pair_search_h(best_h, n, deadline - 3,
                                           [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        elif n <= 300:
            best_h, best_c = pair_search_h(best_h, n, deadline - 3,
                                           [0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=60)
        else:
            best_h, best_c = pair_search_h(best_h, n, deadline - 3,
                                           [0.001, 0.0005, 0.0002], radius=40)
    
    # Phase 6: Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp_optimize(best_h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 2:
            if n <= 200:
                best_h, best_c = pair_search_h(best_h, n, deadline - 1,
                                               [0.0005, 0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = pair_search_h(best_h, n, deadline - 1,
                                               [0.0005, 0.0002, 0.0001], radius=30)
    
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
