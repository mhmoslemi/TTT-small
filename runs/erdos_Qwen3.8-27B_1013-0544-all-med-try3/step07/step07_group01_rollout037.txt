            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

    """Run L-BFGS-B with increasing alpha."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    configs = [
        (100, 1.0), (500, 5.0), (2000, 10.0),
        (10000, 50.0), (50000, 100.0), (200000, 500.0),
        (1000000, 2000.0), (5000000, 10000.0)
    ]
    
    for alpha, mu in configs:

            break

                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-14, 'maxfun': max_iters * 20,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def subgradient_opt(h, n, deadline):
    """Subgradient descent focusing on the worst constraint."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.005
    
    for it in range(50000):
        if time.time() > deadline:
            break
        
        c = compute_c5_vec(h, n)
        k_max = c.argmax()
        g = grad_smooth_max(h, n, 1e6)  # Very sharp -> essentially gradient of max
        
        # Projected gradient step
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new.copy()
                h = h_new
                improved = True
                break
            step *= 0.5
        
        if not improved:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return best_h, best_c

def smart_local_search(h, n, deadline):
    """Local search using gradient-guided pair moves."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    # Compute gradient to guide moves
    alpha = 5000
    g = grad_smooth_max(h, n, alpha)
    
    # Try moves from high-gradient to low-gradient positions
    order = np.argsort(g)  # ascending: most negative first
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # Try moving from positions with low g to positions with high g
            # (i.e., decrease h where g is high, increase h where g is low)
            for i in range(0, n, 2):  # stride 2 for speed
                if time.time() > deadline:
                    return h, best_c
                for j in range(n - i):
                    src = order[j]  # position to increase
                    dst = order[n - 1 - j]  # position to decrease
                    if src == dst:
                        continue
                    if h[src] + d > 1.0 or h[dst] - d < 0.0:
                        continue
                    h[src] += d
                    h[dst] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        # Update gradient
                        g = grad_smooth_max(h, n, alpha)
                        order = np.argsort(g)
                    else:
                        h[src] -= d
                        h[dst] += d
    
    return h, best_c

**Better idea: Frank-Wolfe (Conditional Gradient) method**
The feasible set is the simplex-like set {h : 0 ≤ h ≤ 1, sum(h) = n/2}. The Frank-Wolfe method does:
1. Compute gradient g of the objective at current h
2. Find s = argmin_{s in feasible set} g^T s (the "vertex" of the feasible set that minimizes the linear approximation)
3. Move: h_new = (1-γ)h + γ*s
4. Update γ (step size)

For our feasible set, the "vertices" are 0-1 vectors with exactly n/2 ones (assuming n is even). Finding the exact vertex is NP-hard, but we can approximate:
- s[i] = 1 if g[i] is among the smallest n/2 values, else s[i] = 0

This is a great approximation! And it's very cheap to compute.

def frank_wolfe_opt(h, n, deadline):
    """Frank-Wolfe (conditional gradient) method for the overlap problem."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(10000):
        if time.time() > deadline:
            break
        
        # Compute gradient of smooth max
        g = grad_smooth_max(h, n, 5000)

        # (since we want to minimize g^T s, and s is a 0-1 vector with n/2 ones)
        # Actually, we also need sum(s) = n/2, so s has exactly n/2 ones
        threshold = np.partition(g, n // 2)[n // 2]
        s = (g <= threshold).astype(float)
        # Handle ties: if more than n/2 are below threshold, randomly select
        if s.sum() > n / 2:
            idx = np.where(s == 1)[0]
            keep = rng.choice(len(idx), size=int(n/2), replace=False)
            s = np.zeros(n)
            s[idx[keep]] = 1
        elif s.sum() < n / 2:
            idx = np.where(s == 0)[0]
            need = int(n/2) - int(s.sum())
            add = rng.choice(len(idx), size=need, replace=False)
            s[idx[add]] = 1
        
        # Step size: use standard FW step size γ_k = 2/(k+2)
        gamma = 2.0 / (it + 2)

        best_step = gamma
        best_val = compute_c5((1 - gamma) * h + gamma * s, n)

                best_val = c_test

        c = compute_c5(h, n)

            best_c = c

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    mus = [0.5, 1.0, 5.0, 10.0, 50.0, 100.0, 500.0, 2000.0]
    for i in range(len(alphas)):
        if time.time() > deadline - 3:
            break
        alpha = alphas[i]
        mu = mus[i]
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 20,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fw_step(h, n, it, alpha):
    """One Frank-Wolfe step."""
    n_half = n // 2
    g = grad_smooth_max(h, n, alpha)
    idx = np.argpartition(g, n_half)[:n_half]
    s = np.zeros(n)
    s[idx] = 1.0
    gamma = min(1.0, 2.0 / (it + 2))
    h_new = (1.0 - gamma) * h + gamma * s
    return np.clip(h_new, 0.0, 1.0)
