import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        return m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def run_slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        c_new = compute_c5(res.x, n)
        c_old = compute_c5(h, n)
        if c_new < c_old:
            return project_feasible(res.x, n)
    except:
        pass
    return h.copy()

def upsample_step(h_old, n_old, n_new):
    j = np.arange(n_new)
    i = np.minimum(((j + 0.5) * n_old / n_new).astype(int), n_old - 1)
    h_new = h_old[i].copy()
    return project_feasible(h_new, n_new)

def subgradient_optimize(h, n, deadline, initial_lr=0.1, decay=0.999):
    """Subgradient method on the max term."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = initial_lr
    n_full = 2 * n - 1
    
    while time.time() < deadline:
        c = compute_corr_full(h, n)
        k_star = np.argmax(c)
        m = k_star - (n - 1)
        
        # Compute gradient
        j = np.arange(n)
        valid_first = (j + m >= 0) & (j + m < n)
        valid_second = (j - m >= 0) & (j - m < n)
        grad = np.zeros(n)
        grad[valid_first] = (1.0 - h[j[valid_first] + m])
        grad[valid_second] -= h[j[valid_second] - m]
        
        # Step
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr *= decay
        
        if lr < 1e-8:
            break
    
    return best_h, best_c

def pair_delta(h, n, i, j, c=None):
    """Find optimal delta for h[i]+=delta, h[j]-=delta."""
    if c is None:
        c = compute_corr_full(h, n)
    n_full = 2 * n - 1
    m = np.arange(n_full) - (n - 1)
    
    # g_i
    vf_i = (i + m >= 0) & (i + m < n)
    vs_i = (i - m >= 0) & (i - m < n)
    g_i = np.zeros(n_full)
    g_i[vf_i] = (1.0 - h[i + m[vf_i]])
    g_i[vs_i] -= h[i - m[vs_i]]
    
    # g_j
    vf_j = (j + m >= 0) & (j + m < n)
    vs_j = (j - m >= 0) & (j - m < n)
    g_j = np.zeros(n_full)
    g_j[vf_j] = (1.0 - h[j + m[vf_j]])
    g_j[vs_j] -= h[j - m[vs_j]]
    
    d = g_i - g_j
    # c_new(delta) = c + delta * d
    # Minimize max_k c_new(delta) subject to bounds
    
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    
    if delta_lo >= delta_hi:
        return 0.0
    
    # Binary search for minimum of convex function
    lo, hi = delta_lo, delta_hi
    for _ in range(40):
        mid = (lo + hi) * 0.5
        c_mid = c + mid * d
        k_max = np.argmax(c_mid)
        if d[k_max] >= 0:
            hi = mid
        else:
            lo = mid
    best_delta = (lo + hi) * 0.5
    
    # Verify it's an improvement
    c_best = (c + best_delta * d).max()
    if c_best < c.max() - 1e-15:
        return best_delta
    return 0.0

    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    
    if n_pairs_per_sweep is None:
        n_pairs_per_sweep = min(n * n // 2, 2000)
    
    total_pairs = n * (n - 1) // 2

        c = compute_corr_full(h, n)
        current_max = c.max()
        
        if n_pairs_per_sweep >= total_pairs:
            # Full sweep: iterate over all pairs
            indices = []
            for i in range(n):
                for j in range(i + 1, n):
                    indices.append((i, j))
            # Randomize order
            rng.shuffle(indices)
        else:
            indices = []
            for _ in range(n_pairs_per_sweep):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    indices.append((i, j))
        
        for (i, j) in indices:
            if time.time() > deadline:
                break
            delta = pair_delta(h, n, i, j, c=c)
            if delta != 0.0:

                h[i] = np.clip(h[i], 0.0, 1.0)
                h[j] = np.clip(h[j], 0.0, 1.0)

def apply_pair_optimal(h, n, i, j, c):

    n_full = 2 * n - 1
    m = np.arange(n_full) - (n - 1)

    g_j[vs_j] -= h[j - m[vs_j]]

    delta_hi = min(1.0 - h[i], h[j])

    for _ in range(40):

        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        return float(m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _con_fun(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def run_slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        c_new = compute_c5(res.x, n)
        c_old = compute_c5(h, n)
        if c_new < c_old:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def upsample_step(h_old, n_old, n_new):
    j = np.arange(n_new)
    i = np.minimum(((j + 0.5) * n_old / n_new).astype(int), n_old - 1)
    h_new = h_old[i].copy()
    return project_feasible(h_new, n_new)

def subgradient_optimize(h, n, deadline, initial_lr=0.05, decay=0.9995):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    hr = h.copy()
    lr = initial_lr
    stalled = 0
    while time.time() < deadline:
        c = compute_corr_full(hr, n)
        k_star = np.argmax(c)
        m = k_star - (n - 1)
        j = np.arange(n)
        vf = (j + m >= 0) & (j + m < n)
        vs = (j - m >= 0) & (j - m < n)
        grad = np.zeros(n)
        grad[vf] = (1.0 - hr[j[vf] + m])
        grad[vs] -= hr[j[vs] - m]
        hr_new = project_feasible(hr - lr * grad, n)
        c_new = compute_c5(hr_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = hr_new.copy()
            stalled = 0
        else:
            stalled += 1
        hr = hr_new
        lr *= decay
        if lr < 1e-10 or stalled > 500:
            lr = initial_lr * 0.5
            stalled = 0
            if lr < 1e-12:
                break
    return best_h, best_c

def _compute_g(h, n, idx, n_full, m_arr):
    vf = (idx + m_arr >= 0) & (idx + m_arr < n)
    vs = (idx - m_arr >= 0) & (idx - m_arr < n)
    g = np.zeros(n_full)
    g[vf] = (1.0 - h[idx + m_arr[vf]])
    g[vs] -= h[idx - m_arr[vs]]
    return g

def apply_pair_move(h, n, i, j, c, m_arr, n_full, n_binary_iters=40):
    g_i = _compute_g(h, n, i, n_full, m_arr)
    g_j = _compute_g(h, n, j, n_full, m_arr)
    d = g_i - g_j
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    if delta_lo >= delta_hi - 1e-15:
        return c
    lo, hi = delta_lo, delta_hi
    for _ in range(n_binary_iters):
        mid = (lo + hi) * 0.5
        c_mid = c + mid * d
        if d[np.argmax(c_mid)] >= 0:
            hi = mid
        else:
            lo = mid
    best_delta = (lo + hi) * 0.5
    c_new = c + best_delta * d
    if c_new.max() < c.max() - 1e-14:
        h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
        h[j] = np.clip(h[j] - best_delta, 0.0, 1.0)
        c = c_new
    return c

    sweep = 0
    no_improve_count = 0

    while time.time() < deadline:
        sweep += 1
        if mode == 'full':
            for i in range(n):
                for j in range(i + 1, n):
                    if time.time() > deadline:
                        break
                    c = apply_pair_move(h, n, i, j, c, m_arr, n_full)
        elif mode == 'random':
            for _ in range(n_random):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    c = apply_pair_move(h, n, i, j, c, m_arr, n_full)
        else:
            # mixed: random + local
            n_rand = n_random // 2
            n_loc = n_random // 2
            for _ in range(n_rand):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:

            for _ in range(n_loc):

def coordinate_descent(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_corr_full(h, n)
    n_full = 2 * n - 1
    m_arr = np.arange(n_full) - (n - 1)
    rng = np.random.RandomState(777)
    order = np.arange(n)

    while time.time() < deadline:
        improved = False
        rng.shuffle(order)
        for i in order:
            if time.time() > deadline:
                break
            # Find best j to pair with i
            g_i = _compute_g(h, n, i, n_full, m_arr)
            best_j = -1
            best_delta = 0.0
            best_cmax = c.max()
            # Try a subset of j's
            candidates = rng.choice(n - 1, size=min(50, n - 1), replace=False)
            candidates = np.delete(candidates, candidates > i, 0) if i < len(candidates) else candidates
            for j in range(n):
                if j == i:
                    continue
                g_j = _compute_g(h, n, j, n_full, m_arr)
                d = g_i - g_j
                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(1.0 - h[i], h[j])
                if delta_lo >= delta_hi - 1e-15:
                    continue
                lo, hi = delta_lo, delta_hi
                for _ in range(30):
                    mid = (lo + hi) * 0.5
                    c_mid = c + mid * d
                    if d[np.argmax(c_mid)] >= 0:
                        hi = mid
                    else:
                        lo = mid
                bd = (lo + hi) * 0.5
                cmax_test = (c + bd * d).max()
                if cmax_test < best_cmax - 1e-14:
                    best_cmax = cmax_test
                    best_j = j
                    best_delta = bd
            if best_j >= 0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h[best_j] = np.clip(h[best_j] - best_delta, 0.0, 1.0)
                c = c + best_delta * (_compute_g(h, n, i, n_full, m_arr) - _compute_g(h, n, best_j, n_full, m_arr))
                improved = True

        h = project_feasible(h, n)
        c = compute_corr_full(h, n)
        c_val = float(c.max())
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        if not improved:
            break

    return best_h, best_c
