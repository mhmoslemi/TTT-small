def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

# Power mean smoothing: (sum c_k^p)^(1/p) approximates max for large p
def power_mean_obj(h, n, p, mu):
    c = compute_c5_vec(h, n)
    val = np.power(c.sum() ** 0, 0)  # won't use this
    # Actually: (sum c_k^p)^(1/p)
    # For numerical stability, use: m * (sum (c_k/m)^p)^(1/p)
    m = c.max()
    if m <= 0:
        return 0.0
    ratio = c / m
    val = m * np.power(np.sum(ratio ** p), 1.0 / p)
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def power_mean_grad(h, n, p, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    if m <= 0:
        return np.zeros(n)
    ratio = c / m
    # d/dh of m * (sum (c/m)^p)^(1/p)
    # = m * (1/p) * (sum (c/m)^p)^(1/p-1) * p * (c/m)^(p-1) * (1/m) * dc/dh
    # = (sum (c/m)^p)^(1/p-1) * (c/m)^(p-1) * dc/dh
    sp = np.sum(ratio ** p)
    factor = np.power(sp, 1.0/p - 1.0)

def fine_local_search(h, n, deadline, radius=40):

    h = h.copy()

    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for step in step_sizes:
        if time.time() > deadline:

        improved = True
        while improved:
            improved = False
            c_vec = compute_c5_vec(h, n)
            # Focus on positions that affect active shifts
            max_val = c_vec.max()
            active = np.where(c_vec >= max_val - 0.01 * max_val)[0]
            
            # Get the set of positions involved in active shifts

            for idx in active:
                k = idx - (n-1)
                if k >= 0:
                    positions.update(range(0, n-k))
                    positions.update(range(k, n))
                else:
                    kk = -k
                    positions.update(range(0, n-kk))
                    positions.update(range(kk, n))
            
            pos_list = list(positions)
            if len(pos_list) > 200:
                pos_list = pos_list[:200]
            
            for i in pos_list:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):

                        else:

                        h[j] -= step; h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
