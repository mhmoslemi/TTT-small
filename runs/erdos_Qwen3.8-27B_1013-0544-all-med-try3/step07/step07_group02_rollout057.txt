import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / max(tot, 1e-14)) * h
    return np.clip(h, 0.0, 1.0)

def _sm_objective(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    sv = x.sum() - n / 2.0
    return float(val + penalty * sv * sv)

def _sm_gradient(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        e = np.exp(sh - mx)
    else:
        e = np.exp(sh)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += w[k] * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:n] += w[k] * (1.0 - x[:n - lag])
            g[:n - lag] += w[k] * (-x[lag:n])
        else:
            a = -lag
            g[:n - a] += w[k] * (1.0 - x[a:n])
            g[a:n] += w[k] * (-x[:n - a])
    g *= dx
    sv = x.sum() - n / 2.0
    g += 2.0 * penalty * sv
    return g

def lbfgs_optimize(h, n, deadline, alpha_start=50.0, penalty=1e4):
    bounds = [(0.0, 1.0)] * n
    alpha = alpha_start
    cur_h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    iters_per_round = 2000
    while time.time() < deadline - 10:
        try:
            res = minimize(_sm_objective, cur_h, jac=_sm_gradient,
                         args=(n, alpha, penalty), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': iters_per_round, 'ftol': 1e-22,
                                   'gtol': 1e-10, 'maxls': 50})
            cur_h = project_feasible(res.x, n)
            c_val = compute_c5(cur_h, n)
            if c_val < best_c:
                best_c = c_val
                best_h = cur_h.copy()
        except Exception:
            break
        alpha *= 4.0
        if alpha > 5e7:
            break
        iters_per_round = max(500, iters_per_round // 2)
    return best_h, best_c

def _update_correlation(c_all, h, n, i, j, delta):
    for k in range(-(n - 1), n):
        if abs(i + k) >= n or abs(i - k) >= n or abs(j + k) >= n or abs(j - k) >= n:
            ci = 0.0
            if 0 <= i + k < n:
                ci += (1.0 - h[i + k])
            if 0 <= i - k < n:
                ci -= h[i - k]
            cj = 0.0
            if 0 <= j + k < n:
                cj += (1.0 - h[j + k])
            if 0 <= j - k < n:
                cj -= h[j - k]

            cj = (1.0 - h[j + k]) - h[j - k]
            c_all[k + (n - 1)] += delta * (ci - cj)
