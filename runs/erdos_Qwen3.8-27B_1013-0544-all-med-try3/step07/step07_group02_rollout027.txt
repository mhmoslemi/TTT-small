import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_gradient(h, n, alpha, top_k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_k is not None and top_k < len(c):
        # Only consider top k lags
        top_idx = np.argsort(c)[-top_k:]
        c_active = c[top_idx]
        m = c_active.max()
        shifted = alpha * (c_active - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        # Build gradient from active lags only
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[i] * lag_gradient(h, n, lag)
        return grad * dx
    else:
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-10:
                continue
            lag = k - (n-1)
            grad += w[k] * lag_gradient(h, n, lag)
        return grad * dx

def adam_optimize(h_start, n, deadline, alpha=500, top_k=None, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    for it in range(100000):
        if time.time() > deadline:
            break
        t += 1
        g = smooth_max_gradient(h, n, alpha, top_k)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def momentum_backtrack(h_start, n, deadline, alpha=500, top_k=None, lr_init=1.0):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    momentum = 0.9
    lr = lr_init
    patience = 0
    
    for it in range(100000):
        if time.time() > deadline:
            break
        g = smooth_max_gradient(h, n, alpha, top_k)
        v = momentum * v - lr * g
        # Backtracking line search
        step = v.copy()
        for trial in range(30):
            h_new = project_feasible(h - step, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15 or trial == 0:
                break
            step *= 0.5
        h = project_feasible(h - step, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        if patience > 500:
            lr *= 0.7
            v *= 0.5
            patience = 0
        if it % 5000 == 4999:
            lr = max(0.01, lr * 0.9)
    
    return best_h, best_c

def local_search_refined(h, n, deadline):
    """Refined local search that targets the top lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    for rnd in range(200):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        # Get top 5 lags
        top_idx = np.argsort(c_all)[-5:]
        improved = False
        
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                # For lag 0, we want to make h more binary
                # Move values towards 0 or 1
                for i in range(n):
                    if time.time() > deadline:
                        break
                    if 0.01 < h[i] < 0.99:
                        # Push towards nearest binary value
                        target = 0.0 if h[i] < 0.5 else 1.0
                        delta = (target - h[i]) * 0.1
                        if abs(delta) < 1e-12:
                            continue
                        h[i] += delta
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= delta
                            h = project_feasible(h, n)
            else:
                m = abs(lag)
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-10:
                        # Move mass from i to j (reduce overlap at this lag)
                        for frac in [0.001, 0.005, 0.01, 0.02]:
                            delta = min(frac, (h[i] - h[j]) * 0.3)
                            if delta < 1e-12:
                                continue
                            h[i] -= delta
                            h[j] += delta
                            h = project_feasible(h, n)
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h.copy()
                                improved = True
                                break
                            else:
                                h[j] -= delta
                                h[i] += delta
                                h = project_feasible(h, n)
                    elif h[j] > h[i] + 1e-10:
                        for frac in [0.001, 0.005, 0.01, 0.02]:
                            delta = min(frac, (h[j] - h[i]) * 0.3)
                            if delta < 1e-12:
                                continue
                            h[j] -= delta
                            h[i] += delta
                            h = project_feasible(h, n)
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h.copy()
                                improved = True
                                break
                            else:
                                h[i] -= delta
                                h[j] += delta
                                h = project_feasible(h, n)
        
        if not improved:
            # Try random perturbations
            rng = np.random.RandomState(rnd)
            for _ in range(20):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                delta = rng.uniform(-0.01, 0.01)
                h[i] += delta
                h[j] -= delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h[j] += delta
                    h[i] -= delta
                    h = project_feasible(h, n)
            if not improved:
                break
    
    return best_h, best_c
