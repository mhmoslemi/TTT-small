```python
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    """Gradient of smooth max of correlation lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            # c_k = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d(c_k)/d(h[i]) for 0<=i<n-lag: (1-h[i+lag])
            # d(c_k)/d(h[i+lag]) for 0<=i<n-lag: -h[i]
            grad[0:n-lag] += w[k] * (1.0 - h[lag:n])
            grad[lag:n] += w[k] * (-h[0:n-lag])
        else:

            # c_k = sum_{i=m}^{n-1} h[i]*(1-h[i-m])

            # d(c_k)/d(h[i-m]) for m<=i<n: -h[i]
            grad[m:n] += w[k] * (1.0 - h[0:n-m])
            grad[0:n-m] += w[k] * (-h[m:n])
    return grad * dx

def objective_and_grad(h, n, alpha):
    obj = smooth_max(compute_all_lags(h, n), alpha)
    grad = smooth_max_grad(h, n, alpha)
    return obj, grad
```

```python
    # O(n) numpy operations
    grad[0:n-lag] += ...  # one numpy op
    grad[lag:n] += ...  # one numpy op
```

Each Python iteration has overhead of ~100ns + numpy ops. If only a few lags have significant weight (high alpha), the loop is fast. With alpha=10000, only 1-3 lags will have weight > 1e-10. So the effective loop is only 1-3 iterations.

With low alpha (50), many lags contribute. But 999 Python iterations * 500 elements * few ops ≈ manageable.

Let me also think about the "binary round-trip" more carefully:

1. Binarize: h_bin[i] = 1 if h[i] > 0.5 else 0
2. Check: sum(h_bin) should be n/2 (if not, adjust)
3. Local binary search: for each pixel i, try flipping it. If sum changes, also flip another pixel to maintain sum. Evaluate C₅.
4. Actually, flipping one pixel changes sum by ±1. To maintain sum=n/2, I need to flip two pixels: one from 0→1 and one from 1→0.
5. This is an O(n²) search per "pass", but I can be smart about which pairs to try.

The binary local search is:
- Find the lag k* that achieves the max
- For positions i < n-k*, if h_bin[i]=1 and h_bin[i+k*]=1, one of them should become 0
- Try flipping h_bin[i] from 1 to 0 and h_bin[j] from 0 to 1 (for some j)
- Choose j to also help reduce other high lags

This is getting complex. Let me simplify:

For the binary search, I'll do:
- Random pair swaps: pick i,j at random, swap h_bin[i] and h_bin[j] if they're different

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    n_active = np.where(w > 1e-10)[0]
    for k in n_active:
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[0:n-lag] += wk * (1.0 - h[lag:n])
            grad[lag:n] += wk * (-h[0:n-lag])
        else:
            m = -lag
            grad[m:n] += wk * (1.0 - h[0:n-m])
            grad[0:n-m] += wk * (-h[m:n])
    return grad * dx

def lbfgs_smooth_max(h, n, alphas, deadline, maxiter_each=300):
    h_work = h.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        def obj(x):
            return smooth_max_value(compute_all_lags(x, n), alpha)
        def obj_grad(x):
            return smooth_max_gradient(x, n, alpha)
        try:
            res = minimize(obj, h_work, jac=obj_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-14})
            h_candidate = project_feasible(res.x, n)
            c_candidate = compute_c5(h_candidate, n)
            if c_candidate < best_c - 1e-15:
                best_c = c_candidate
                best_h = h_candidate.copy()
            h_work = h_candidate
        except Exception:
            pass
    return best_h, best_c

def binary_c5(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    c = np.correlate(h_bin.astype(float), 1.0 - h_bin.astype(float), mode='full')
    return float(np.max(c) * dx)

def binary_annealing(h_start, n, deadline, rng, max_iter=2000000):
    h_bin = np.zeros(n)
    thresh = np.where(h_start > 0.5, 1.0, 0.0)
    # Ensure sum = n/2
    while thresh.sum() > n // 2:
        idx = np.where(thresh == 1)[0]
        if len(idx) == 0:
            break
        j = rng.choice(idx)
        thresh[j] = 0
    while thresh.sum() < n // 2:
        idx = np.where(thresh == 0)[0]
        if len(idx) == 0:
            break
        j = rng.choice(idx)
        thresh[j] = 1
    h_bin = thresh.copy()
    best_c = binary_c5(h_bin)
    best_h = h_bin.copy()
    T = 0.02
    T_min = 0.001
    cooling = 0.99999
    n_ones = n // 2
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Pick a random 1-position and a random 0-position, swap
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0
        h_bin[j] = 1
        c_new = binary_c5(h_bin)
        delta_c = c_new - best_c
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            best_check = c_new
            if best_check < best_c:
                best_c = best_check
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
        T = max(T_min, T * cooling)
    return best_h, best_c

def continuous_from_binary(h_bin, n, deadline, rng):
    """Refine a binary solution in continuous space."""
    h0 = h_bin.astype(float).copy()
    # Add small noise to break ties
    h0 = h0 + rng.uniform(-0.05, 0.05, n)
    h0 = project_feasible(h0, n)
    best_h, best_c = lbfgs_smooth_max(h0, n, [50, 200, 1000, 5000, 20000], deadline)
    return best_h, best_c

def focused_lag_search(h, n, deadline, max_rounds=200):
    """Targeted search on the most problematic lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        # Try multiple step sizes
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time.time() > deadline:
                break
            for i in range(0, n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-6:
                    delta = min(step, (h[i] - h[j]) * 0.5, h[i] * 0.8, (1 - h[j]) * 0.8)
                    if delta < 1e-8:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-6:
                    delta = min(step, (h[j] - h[i]) * 0.5, h[j] * 0.8, (1 - h[i]) * 0.8)
                    if delta < 1e-8:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
        if not improved:
            # Try next-worst lags
            c_all = compute_all_lags(h, n)
            top_lags = np.argsort(c_all)[-5:]
            for k2 in top_lags:
                if time.time() > deadline:
                    break
                lag2 = k2 - (n - 1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for i in range(0, n - m2, 2):
                    if time.time() > deadline:
                        break
                    j2 = i + m2
                    if h[i] > h[j2] + 1e-6:
                        delta = 0.003
                        if delta < h[i] and delta < 1 - h[j2]:
                            h[i] -= delta
                            h[j2] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] += delta
                                h[j2] -= delta
                    elif h[j2] > h[i] + 1e-6:
                        delta = 0.003
                        if delta < h[j2] and delta < 1 - h[i]:
                            h[i] += delta
                            h[j2] -= delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] -= delta
                                h[j2] += delta
            if not improved:
                break
    return h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000, top_m=5):
    """SLSQP on the smooth max with active constraints."""
    dx = 2.0 / n
    h0 = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    
    if top_m is not None and top_m < 2 * n - 1:
        c = compute_all_lags(h0, n)
        top_idx = np.argsort(c)[-top_m:]
        
        def act_obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full') * dx
            active = cc[top_idx]
            return smooth_max_value(active, alpha)
    else:
        top_idx = None
        
        def act_obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full') * dx
            return smooth_max_value(cc, alpha)
    
    def sum_con(x):
        return np.sum(x) - n / 2.0
    
    def sum_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(act_obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def slsqp_grad_polish(h, n, alpha, top_m=5):
    """SLSQP with gradient for better convergence."""
    dx = 2.0 / n
    h0 = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    
    if top_m is not None and top_m < 2 * n - 1:
        c = compute_all_lags(h0, n)
        top_idx = np.argsort(c)[-top_m:]
        
        def act_obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full') * dx
            active = cc[top_idx]
            return smooth_max_value(active, alpha)
        
        def act_grad(x):
            cc = np.correlate(x, 1.0 - x, mode='full') * dx
            active = cc[top_idx]
            w = smooth_max_weights(active, alpha)
            # Only compute gradient for top m lags
            lag_offsets = top_idx - (n - 1)
            grad = np.zeros(n)
            for idx_i, k in enumerate(top_idx):
                wk = w[idx_i]
                if wk < 1e-10:
                    continue
                lag = k - (n - 1)
                if lag == 0:
                    grad[:] += wk * (1.0 - 2.0 * x)
                elif lag > 0:
                    grad[0:n-lag] += wk * (1.0 - x[lag:n])
                    grad[lag:n] += wk * (-x[0:n-lag])
                else:
                    m2 = -lag
                    grad[m2:n] += wk * (1.0 - x[0:n-m2])
                    grad[0:n-m2] += wk * (-x[m2:n])
            return grad * dx
    else:
        def act_obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full') * dx
            return smooth_max_value(cc, alpha)
        
        def act_grad(x):
            return smooth_max_gradient(x, n, alpha)
    
    def sum_con(x):
        return np.sum(x) - n / 2.0
    
    def sum_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(act_obj, h0, jac=act_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 3000, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def perturb_and_optimize(h, n, deadline, rng, scale):
    """Perturb, then optimize."""
    h_pert = h + rng.standard_normal(n) * scale
    h_pert = project_feasible(h_pert, n)
    h_new, c_new = lbfgs_smooth_max(h_pert, n, [100, 500, 2000, 10000], deadline)
    return h_new, c_new
```
