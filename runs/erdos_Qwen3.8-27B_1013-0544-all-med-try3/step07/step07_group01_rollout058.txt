import numpy as np
from scipy.optimize import minimize
import time

# === Helper functions ===

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            s = h.sum()
            if s < 1e-16:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            nv = n - shift
            grad[:nv] += wk * (1.0 - h[shift:])
            grad[shift:] -= wk * h[:nv]
        else:
            s2 = -shift
            nv = n - s2
            grad[s2:] += wk * (1.0 - h[:nv])
            grad[:nv] -= wk * h[s2:]
    return grad

def obj_penalty(h, n, alpha, mu):
    return smooth_max_val(h, n, alpha) + mu * (h.sum() - n/2.0)**2

def jac_penalty(h, n, alpha, mu):
    return smooth_max_grad(h, n, alpha) + 2.0 * mu * (h.sum() - n/2.0)

# === Optimization phases ===

def lbfgs_optimize(h, n, deadline, alphas):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        mu = min(1000.0, alpha * 0.01)
        maxiter = max(50, min(500, int((deadline - time.time()) / alphas.index(alpha) / len(alphas)) | 0))
        # Actually, let me just allocate a fixed number of iterations per alpha
        try:
            res = minimize(obj_penalty, best_h, jac=jac_penalty,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def gradient_descent(h, n, deadline, alpha=500.0, lr=0.005):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_lr = lr
    cur_alpha = alpha
    
    for it in range(50000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, cur_alpha)
        g += 0.01 * (h.sum() - n/2.0) * np.ones(n)  # integral constraint
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Line search
        step = cur_lr
        improved = False
        for _ in range(30):
            h_new = h - step * g
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            cur_lr *= 0.5
            cur_alpha *= 2
            if cur_lr < 1e-14:
                break
            if cur_alpha > 1e6:
                cur_alpha = 500.0
                cur_lr = lr
    return best_h, best_c

def coordinate_descent(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    epsilons = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for eps in epsilons:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for delta in [eps, -eps]:
                    h_try = h.copy()
                    h_try[i] += delta
                    if h_try[i] < 0 or h_try[i] > 1:

                    h_try = project_feasible(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c - 1e-15:
                        h = h_try
                        best_c = c_try
                        best_h = h.copy()
                        improved = True
                        break  # move on to next i
    return best_h, best_c

def pair_local_search(h, n, deadline, max_passes=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    for pass_idx in range(max_passes):
        if time.time() > deadline:
            break
        eps = 0.01 / (pass_idx + 1)
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            # Try moving mass to/from nearby positions
            for j_offset in range(1, min(30, n)):
                for j in [i + j_offset, i - j_offset]:
                    if j < 0 or j >= n:
                        continue
                    for delta in [eps, -eps]:
                        h_try = h.copy()
                        h_try[i] += delta
                        h_try[j] -= delta
                        if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                            continue
                        h_try = project_feasible(h_try, n)
                        c_try = compute_c5(h_try, n)
                        if c_try < best_c - 1e-14:
                            h = h_try
                            best_c = c_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c

from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            s = h.sum()
            if s < 1e-16:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            nv = n - shift
            grad[:nv] += wk * (1.0 - h[shift:])
            grad[shift:] -= wk * h[:nv]
        else:
            s2 = -shift
            nv = n - s2
            grad[s2:] += wk * (1.0 - h[:nv])
            grad[:nv] -= wk * h[s2:]
    return grad
