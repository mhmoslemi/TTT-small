import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    if sm > 500:
        w = np.exp(shifted - sm)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        lag = k - (n-1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def adam_optimize(h, n, deadline, alpha_start=20.0, alpha_end=20000.0,
                  lr=1.0, betas=(0.9, 0.999), eps=1e-8, patience=3000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_v = np.zeros(n)
    m_g2 = np.zeros(n)
    t = 0
    no_improve = 0
    log_as = np.log(alpha_start)
    log_ae = np.log(alpha_end)
    total_iters = 20000
    while time.time() < deadline:
        t += 1
        frac = min(t / (total_iters * 0.5), 1.0)
        alpha = np.exp(log_as + frac * (log_ae - log_as))
        g = grad_smooth_max(h_curr, n, alpha)
        m_v = betas[0]*m_v + (1-betas[0])*g
        m_g2 = betas[1]*m_g2 + (1-betas[1])*g**2
        m_v_hat = m_v / (1 - betas[0]**t)
        m_g2_hat = m_g2 / (1 - betas[1]**t)
        step = lr * m_v_hat / (np.sqrt(m_g2_hat) + eps)
        h_new = project_feasible(h_curr - step, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                lr *= 0.85
                m_v *= 0.5
                no_improve = 0
                if lr < 0.05:
                    break
        h_curr = h_new
    return best_h, best_c

def _topk_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    sm = shifted.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n/2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_topk(h, n, top_k=10, alpha=50000, maxiter=8000):
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_k:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_topk_obj, h, args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _full_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_full(h, n, alpha=200000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_full_obj, h, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fine_local_search(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idxs = np.argsort(c_all)[-8:]
        improved = False
        for k_idx in top_idxs:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            for i in range(0, n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-8:
                    delta = min(0.005, (h[i]-h[j])*0.5)
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-8:
                    delta = min(0.005, (h[j]-h[i])*0.5)
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
        if not improved:
            for d2 in [0.002, 0.001, 0.0005]:
                if time.time() > deadline:
                    break
                c_all = compute_all_lags(h, n)
                top_idxs2 = np.argsort(c_all)[-5:]
                improved2 = False
                for k_idx in top_idxs2:
                    if time.time() > deadline:
                        break
                    lag = k_idx - (n-1)
                    if lag == 0:
                        continue
                    m2 = abs(lag)
                    for i in range(n - m2):
                        if time.time() > deadline:
                            break
                        j = i + m2
                        if h[i] >= 0.99 or h[j] <= 0.01:
                            continue
                        if h[i] > h[j]:
                            delta = min(d2, h[i]*0.3, (1-h[j])*0.3)
                        else:
                            delta = min(d2, h[j]*0.3, (1-h[i])*0.3)
                            i, j = j, i
                        if delta < 1e-10:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved2 = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                if improved2:
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c

def binary_push(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(20000):
        if time.time() > deadline:
            break
        idx = np.random.randint(0, n)
        old_val = h[idx]
        if old_val > 0.5:
            new_val = 1.0
        else:
            new_val = 0.0
        if abs(new_val - old_val) < 1e-6:
            continue
        diff = old_val - new_val
        h[idx] = new_val
        if diff > 0:
            room = 1.0 - h
            room[idx] = 0.0
            tot = room.sum()
            if tot > 1e-14:
                h += (diff / tot) * room
                h[idx] = new_val
            else:
                h[idx] = old_val
                continue
        else:
            others = np.ones(n)
            others[idx] = 0
            h_others = h[others]
            tot = h_others.sum()
            if tot > 1e-14:
                h[others] = h_others * (tot + diff) / tot
                h = np.clip(h, 0, 1)
                h[idx] = new_val
            else:
                h[idx] = old_val
                continue
        h = np.clip(h, 0, 1)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
        else:
            # revert
            h[idx] = old_val
            # reproject
            h = project_feasible(h, n)
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Adam gradient descent with increasing alpha
    alloc1 = (deadline - time.time()) * 0.35
    d1 = time.time() + min(alloc1, 200)
    h_new, c_new = adam_optimize(h, n, d1, alpha_start=20, alpha_end=15000, lr=1.2)
    if c_new < best_c:
        best_c = c_new
        best_h = h_new
    h = h_new

    # Phase 2: SLSQP on top-10 lags
    if time.time() < deadline - 30:
        for tk in [10, 15, 20]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_topk(h, n, top_k=tk, alpha=50000, maxiter=6000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Phase 3: SLSQP full
    if time.time() < deadline - 20:
        h_new = slsqp_full(h, n, alpha=100000, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: Fine local search
    if time.time() < deadline - 5:
        ls_dl = deadline - 2
        h_new, c_new = fine_local_search(best_h, n, ls_dl)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 5: Binary push
    if time.time() < deadline - 5:
        h_new, c_new = binary_push(best_h, n, deadline - 2)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interpolate(h_from, n_from, n_to):
    x_from = np.linspace(0, 2, n_from)
    x_to = np.linspace(0, 2, n_to)
    h_to = np.interp(x_to, x_from, h_from)
    return project_feasible(h_to, n_to)
