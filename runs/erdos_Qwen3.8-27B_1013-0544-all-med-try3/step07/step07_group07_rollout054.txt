    # h[i] as first factor: contributes (1-h[i+k]) to corr at lag k
    for k in range(-(n-1), n):
        if 0 <= i+k < n:
            a_m[k+n-1] += 1.0 - h[i+k]
        if 0 <= i-k < n:
            a_m[k+n-1] -= h[i-k]
    b_m = corr - h[i] * a_m
    
    # Minimize max(a_m * t + b_m) over t in [0,1]
    # Vectorized evaluation
    ts = np.linspace(0, 1, 200)
    vals = np.max(a_m[None,:] * ts[:,None] + b_m[None,:], axis=1)
    t_approx = ts[np.argmin(vals)]
    
    # Refine with golden section
    a, b = max(0, t_approx-0.05), min(1, t_approx+0.05)
    gr = (np.sqrt(5)-1)/2

        f2 = np.max(a_m*c2 + b_m)

def coord_descent(h, n, max_passes=50, deadline=None):
    """Coordinate descent: optimize each h[i] one at a time."""
    dx = 2.0 / n
    curr = h.copy()
    corr = np.correlate(curr, 1 - curr, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = curr.copy()
    
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved_total = 0.0
        for i in range(n):
            # Compute a_m for coordinate i
            a_m = np.zeros(2*n-1)
            # h[i] as first factor in corr: (1-h[i+k]) for valid i+k
            k_lo = max(0, -i)  # k such that i+k >= 0
            k_hi = min(n-1, n-1-i)  # k such that i+k <= n-1

            if k_lo <= k_hi:

            k_hi2 = min(n-1, i)  # k such that i-k <= n-1

            best_h = curr.copy()
    
    return best_h, best_c

Wait, I realize there's an issue with the coordinate descent. After changing h[i], the correlation changes for ALL lags, and the a_m for subsequent coordinates changes. But since I recompute the correlation after each change, this should be fine (just expensive).

For n=400, recomputing the correlation is O(n²) = 160K operations. With 400 coordinates per pass, that's 64M operations per pass. At ~100M ops/s, that's 0.64s per pass. With 50 passes, that's 32s. Feasible within the budget.

Actually, I can be smarter. Instead of recomputing the full correlation after each coordinate update, I can update it incrementally:
When h[i] changes from old to new:
- corr[m] changes by (new - old) * a_m[i] where a_m is the coefficient vector for coordinate i

But since I'm computing a_m for coordinate i anyway, I can just do:
corr += (t_opt - curr[i]) * a_m

This is O(n) instead of O(n²) per coordinate! Total per pass: O(n²) instead of O(n³).

Let me revise:

```python
def coord_descent(h, n, max_passes=100, deadline=None):
    dx = 2.0 / n
    curr = h.copy()
    corr = np.correlate(curr, 1 - curr, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = curr.copy()
    
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        n_changes = 0
        for i in range(n):
            # Compute a_m for coordinate i
            a_m = np.zeros(2*n-1)
            k_lo = max(0, -i)
            k_hi = min(n-1, n-1-i)
            if k_lo <= k_hi:
                a_m[k_lo+n-1:k_hi+n] += 1.0 - curr[i+k_lo:i+k_hi+1]
            k_lo2 = max(0, i-(n-1))
            k_hi2 = min(n-1, i)
            if k_lo2 <= k_hi2:
                a_m[k_lo2+n-1:k_hi2+n] -= curr[i-k_lo2:i-k_hi2+1]
            
            b_m = corr - curr[i] * a_m
            
            # 1D: min_t max(a_m*t + b_m)
            ts = np.linspace(0, 1, 80)
            vals = np.max(a_m[None,:] * ts[:,None] + b_m[None,:], axis=1)
            t_approx = ts[np.argmin(vals)]
            
            lo = max(0.0, t_approx - 0.05)
            hi = min(1.0, t_approx + 0.05)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(20):
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(a_m * c1 + b_m)
                f2 = np.max(a_m * c2 + b_m)
                if f1 < f2:
                    hi = c2
                else:
                    lo = c1
            t_opt = max(0.0, min(1.0, (lo + hi) / 2.0))
            
            delta = t_opt - curr[i]
            if abs(delta) > 1e-12:
                curr[i] = t_opt
                corr += delta * a_m
                n_changes += 1
                if (n_changes % 50 == 0) or i == n-1:
                    c = corr.max() * dx
                    if c < best_c:
                        best_c = c
                        best_h = curr.copy()
        
        if n_changes == 0:
            break
        c_now = corr.max() * dx
        if c_now < best_c:
            best_c = c_now
            best_h = curr.copy()
    
    # Verify
    corr_final = np.correlate(best_h, 1 - best_h, mode='full').astype(float)
    best_c = corr_final.max() * dx
    best_c = min(best_c, compute_c5(best_h, n))
    return best_h, best_c
```
