```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s > 1e-15:
            h += (diff / s) * room
    else:
        s = h.sum()
        if s > 1e-15:
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def binary_local_search(h, n, deadline, max_sweeps=5000):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    m_range = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for zi in range(len(zeros)):
            if time.time() > deadline:
                break
            i = zeros[zi]
            best_j = -1
            best_delta = None
            best_nc = cur_c
            for oj in range(len(ones)):
                j = ones[oj]
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0:
                j = best_j
                h[i], h[j] = 1.0, 0.0
                corr = corr + best_delta
                cur_c = corr.max() * dx
                oi = ones.index(j)
                ones.pop(oi)
                ones.append(i)
                zeros.pop(zi)
                zeros.append(j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def anneal_binary(h, n, deadline, seed=42, n_iters=5000000):

    dx = 2.0 / n
    h = h.copy()

    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c

    zeros = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    T_start = 0.05
    T_end = 0.00001
    cool = (T_end / T_start) ** (1.0 / max(n_iters, 1))
    T = T_start
    accent = 3000.0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)

        k2 = n - 1 + (i - j)

        nc = (corr + delta).max() * dx
        delta_c = nc - cur_c

    return best_h, best_c

            # 1-swap
            ai = rng.randint(len(zeros))
            bi = rng.randint(len(ones))
            i = zeros[ai]
            j = ones[bi]
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            if 0 <= k1 < 2 * n - 1:
                delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            if 0 <= k2 < 2 * n - 1:
                delta[k2] += 1.0
            nc = (corr + delta).max() * dx
            dc = nc - cur_c
            if dc <= 0 or (T > 1e-12 and rng.random() < np.exp(-accent * dc / T)):
                h[i], h[j] = 1.0, 0.0
                corr = corr + delta
                cur_c = nc
                zeros[ai] = j
                ones[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        else:
            # 2-swap: pick two zeros and two ones
            if len(zeros) < 2 or len(ones) < 2:
                T *= cool
                continue
            a1 = rng.randint(len(zeros))
            a2 = rng.randint(len(zeros))
            if a2 == a1:
                a2 = (a2 + 1) % len(zeros)
            b1 = rng.randint(len(ones))
            b2 = rng.randint(len(ones))
            if b2 == b1:
                b2 = (b2 + 1) % len(ones)
            i1 = zeros[a1]
            i2 = zeros[a2]
            j1 = ones[b1]
            j2 = ones[b2]
            # Compute combined delta
            delta = np.zeros(2 * n - 1)
            for ii in (i1, i2):
                im = ii + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = ii - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                delta[n - 1] -= 2.0
            for jj in (j1, j2):
                jm = jj + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = jj - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
            # Interaction terms for 1-swap (i1,j1)
            k = n - 1 + (j1 - i1)
            if 0 <= k < 2 * n - 1:
                delta[k] += 1.0
            k = n - 1 + (i1 - j1)
            if 0 <= k < 2 * n - 1:
                delta[k] += 1.0
            # Interaction terms for 1-swap (i2,j2)
            k = n - 1 + (j2 - i2)
            if 0 <= k < 2 * n - 1:
                delta[k] += 1.0
            k = n - 1 + (i2 - j2)
            if 0 <= k < 2 * n - 1:
                delta[k] += 1.0
            # Cross interaction: i1 vs j2, i2 vs j1
            k = n - 1 + (j2 - i1)
            if 0 <= k < 2 * n - 1:
                delta[k] += 1.0
            k = n - 1 + (i1 - j2)
            if 0 <= k < 2 * n - 1:
                delta[k] += 1.0
            k = n - 1 + (j1 - i2)
            if 0 <= k < 2 * n - 1:
                delta[k] += 1.0
            k = n - 1 + (i2 - j1)
            if 0 <= k < 2 * n - 1:
                delta[k] += 1.0
            nc = (corr + delta).max() * dx
            dc = nc - cur_c
            if dc <= 0 or (T > 1e-12 and rng.random() < np.exp(-accent * dc / T)):
                h[i1], h[j1] = 1.0, 0.0
                h[i2], h[j2] = 1.0, 0.0
                corr = corr + delta
                cur_c = nc
                zeros[a1] = j1
                zeros[a2] = j2
                ones[b1] = i1
                ones[b2] = i2
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        T *= cool
    return best_h, best_c

def iterated_local_search(h, n, deadline, seed=42, n_restarts=50):
    rng = np.random.RandomState(seed + 300)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        # Perturb: random swaps
        hp = best_h.copy()
        n_swaps = rng.randint(8, 25)
        ones = np.where(hp == 1)[0]
        zeros = np.where(hp == 0)[0]
        for _ in range(n_swaps):
            zi = rng.randint(len(zeros))
            oj = rng.randint(len(ones))
            hp[zeros[zi]], hp[ones[oj]] = 1.0, 0.0
            zval = zeros[zi]
            oval = ones[oj]
            zeros[zi] = oval
            ones[oj] = zval
        # Local search from perturbed solution
        hp, cp = binary_local_search(hp, n, min(time.time() + 15, deadline))
        if cp < best_c - 1e-12:
            best_c = cp
            best_h = hp.copy()
    return best_h, best_c

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

def pgd_solve(h, n, alphas, deadline, max_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        cur_c = best_c
        step = min(float(n) / (4.0 * alpha), 0.3)
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            grad = lse_grad(h, n, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            direction = -grad / gnorm
            accepted = False
            for ls in range(30):
                h_new = project(h + step * (0.5 ** ls) * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    h = h_new
                    cur_c = c_new
                    step = step * (0.5 ** ls)
                    accepted = True
                    break
            if accepted:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                step *= 0.5
                if step < 1e-10:
                    break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_budget = budget_s - 10
    deadline = t0 + total_budget
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    
    # Project to feasible
    h0 = project(h0, n)
    hb = to_binary(h0, n)
    best_h = hb.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Aggressive binary local search
    p1 = t0 + 0.20 * total_budget
    hb, c1 = binary_local_search(hb, n, p1, max_sweeps=3000)
    if c1 < best_c:
        best_c = c1
        best_h = hb.copy()
    
    # Phase 2: Simulated annealing (1-swap)
    p2 = t0 + 0.38 * total_budget
    ha, c2 = anneal_binary(best_h, n, p2, seed=seed, n_iters=5000000)
    if c2 < best_c:
        best_c = c2
        best_h = ha.copy()
    
    # Phase 3: Local search after annealing
    p3 = t0 + 0.48 * total_budget
    hb, c3 = binary_local_search(best_h, n, p3, max_sweeps=2000)
    if c3 < best_c:
        best_c = c3
        best_h = hb.copy()
    
    # Phase 4: 2-swap annealing
    p4 = t0 + 0.62 * total_budget
    h2, c4 = anneal_binary_2swap(best_h, n, p4, seed=seed, n_iters=1500000)
    if c4 < best_c:
        best_c = c4
        best_h = h2.copy()
    
    # Phase 5: Local search
    p5 = t0 + 0.70 * total_budget
    hb, c5 = binary_local_search(best_h, n, p5, max_sweeps=1500)
    if c5 < best_c:
        best_c = c5
        best_h = hb.copy()
    
    # Phase 6: Iterated local search
    p6 = t0 + 0.82 * total_budget
    hb, c6 = iterated_local_search(best_h, n, p6, seed=seed, n_restarts=40)
    if c6 < best_c:
        best_c = c6
        best_h = hb.copy()
    
    # Phase 7: Try higher resolution
    p7 = t0 + 0.90 * total_budget
    for nh in [500, 600, 800]:
        if time.time() > p7 - 5:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project(hh, nh)
        hh_b = to_binary(hh, nh)
        t_sub = min(time.time() + 20, p7)
        hh_b, ch = binary_local_search(hh_b, nh, t_sub, max_sweeps=500)
        if ch < best_c:
            best_c = ch
            best_h = hh_b.copy()
            n = nh
    
    # Phase 8: Final local search at current best resolution
    if time.time() < deadline - 30:
        hb, cf = binary_local_search(best_h, n, deadline - 15, max_sweeps=2000)
        if cf < best_c:
            best_c = cf
            best_h = hb.copy()
    
    # Phase 9: PGD polish (continuous refinement)
    if time.time() < deadline - 5:
        alphas = [500, 5000, 50000]
        best_h, best_c = pgd_solve(best_h, n, alphas, deadline - 2, max_iter=2000)
    
    # Final projection and evaluation
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
