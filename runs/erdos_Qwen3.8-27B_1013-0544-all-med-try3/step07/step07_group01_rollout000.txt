import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    val = smooth_max_obj(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    return float(smooth_max_obj(x, n, alpha))

def _slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 2.0), (500, 5.0), (1000, 10.0),
        (5000, 50.0), (20000, 100.0), (100000, 500.0),
        (500000, 2000.0), (2000000, 5000.0), (5000000, 10000.0)

    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=2000):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_jac, 'args': (n,)}]

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

    # Phase 8: Very fine pair search (100s)
    if time.time() < deadline - 50:
        t8 = min(time.time() + 100, deadline - 50)
        best_h, best_c = pair_search(best_h, best_n, t8,
                                     [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                     radius=40)

    # Phase 9: Shake and refine (80s)
    if time.time() < deadline - 30:
        t9 = min(time.time() + 80, deadline - 30)
        h9, c9 = shake_and_refine(best_h, best_n, t9, num_shakes=4)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Final polishing - very small pair search + L-BFGS
    if time.time() < deadline - 5:
        alloc = deadline - time.time() - 2
        if alloc > 20:
            best_h, best_c = pair_search(best_h, best_n, time.time() + max(10, alloc*0.5),
                                         [0.00005, 0.00002, 0.00001],
                                         radius=30)
        if time.time() < deadline - 3:
            best_h, best_c = lbfgs_phase(best_h, best_n, deadline - 1, max_iters=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
