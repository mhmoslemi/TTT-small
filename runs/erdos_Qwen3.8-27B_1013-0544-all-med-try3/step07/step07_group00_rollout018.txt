from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu, active_idx=None):
    if active_idx is not None and len(active_idx) < len(x)*2:
        c_all = compute_c5_vec(x, n)
        c = c_all[active_idx]
    else:
        c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu, active_idx=None):
    if active_idx is not None:
        c_all = compute_c5_vec(x, n)
        c = c_all[active_idx]
        idx = active_idx
    else:
        c = compute_c5_vec(x, n)
        idx = np.arange(len(c))
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    nn = len(x)
    grad = np.zeros(nn)
    for wi, ii in zip(w, idx):
        k = ii - (nn - 1)
        wm = wi * dx
        if k >= 0:
            nv = nn - k
            grad[0:nv] += wm * (1.0 - x[k:nn])
            grad[k:nn] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = nn - kk
            grad[kk:nn] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:nn])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

            best_h = h.copy()
            stuck_count = 0

                    best_c = c_trial

def correlate_fast(a, b):
    n = len(a)
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fsize = 1
    while fsize < size * 2:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    corr = np.fft.irfft(fa * fb)
    return corr[:size]

def smooth_obj(x, n, alpha, mu, idx=None):
    c = compute_c5_vec(x, n)
    if idx is not None:
        c = c[idx]
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 300:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu, idx=None):
    c = compute_c5_vec(x, n)
    if idx is not None:
        c_use = c[idx]
        idx_use = idx
    else:
        c_use = c
        idx_use = np.arange(len(c))
    m = c_use.max()
    sh = alpha * (c_use - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    nn = len(x)
    grad = np.zeros(nn)
    for wi, ii in zip(w, idx_use):
        if wi < 1e-14:
            continue
        k = int(ii) - (nn - 1)
        wm = wi * dx
        if k >= 0:
            nv = nn - k
            grad[0:nv] += wm * (1.0 - x[k:nn])
            grad[k:nn] += wm * (-x[0:nv])

            kk = -k
            nv = nn - kk
            grad[kk:nn] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:nn])

import time

def project_feasible(h, n):

            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def sm_obj(x, n, alpha, mu, aidx):
    c = compute_c5_vec(x, n)
    if aidx is not None:
        c = c[aidx]
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 300:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + mu * (x.sum() - n/2.0)**2)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha + mu * (x.sum() - n/2.0)**2)

def sm_grad(x, n, alpha, mu, aidx):
    c = compute_c5_vec(x, n)
    if aidx is not None:
        cu = c[aidx]
        iu = aidx
    else:
        cu = c
        iu = np.arange(len(c))
    m = cu.max()
    sh = alpha * (cu - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    nn = len(x)
    g = np.zeros(nn)
    for wi, ii in zip(w, iu):
        if wi < 1e-15:
            continue
        k = int(ii) - (nn - 1)
        wm = wi * dx
        if k >= 0:
            nv = nn - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = nn - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n/2.0)
    return g

    return x.sum() - n / 2.0

    return np.ones(n)

def mom_grad(h, n, deadline, lr0=0.003, alpha=5000, mu=0.1, mom=0.9):

    best_c = compute_c5(h, n)
    lr = lr0

    stall = 0

        mv = cv.max()
        thr = mv - max(1e-6, 0.003 * mv)
        act = np.where(cv >= thr)[0]
        if len(act) > 40:
            act = np.argsort(-cv[act])[:40]
        try:

            v = mom * v + (1.0 - mom) * g
            h_new = h - lr * v
            h_new = project_feasible(h_new, n)

def interp_h(h, n_old, n_new):
    xo = np.linspace(0.0, 2.0, n_old)
    xn = np.linspace(0.0, 2.0, n_new)
    return np.interp(xn, xo, h)

def coord_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    radius = 50
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    hv = h[i]
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] = hv
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] = hv
                            h[j] -= d
    return best_h, best_c

                break
            t3 = min(time.time() + 35, deadline - 20)
            h3, c3 = slqp_run(best_h, n, t3, k, a)
            if c3 < best_c:
                best_c, best_h = c3, h3

    # Phase 4: Second round L-BFGS with higher alpha (~120s)
    if deadline - time.time() > 150:
        t4 = min(time.time() + 120, deadline - 130)
        h4, c4 = lbfgs_opt(best_h, n, t4,
            alphas=[5000, 20000, 100000, 500000],
            mus=[500.0, 5000.0, 50000.0, 200000.0])
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Try n=1000 (~130s)
    if deadline - time.time() > 100:
        n_hi = 1000
        h_hi = interp_h(best_h, n, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        t5 = min(time.time() + 120, deadline - 70)
        h_hi, c_hi = lbfgs_opt(h_hi, n_hi, t5,
            alphas=[500, 5000, 50000],
            mus=[50.0, 500.0, 5000.0])
        if c_hi < best_c:
            best_c, best_h, n = c_hi, h_hi, n_hi

    # Phase 6: If at n=1000, also optimize back at n=500 (~60s)
    if n != 500 and deadline - time.time() > 50:
        h_bk = interp_h(best_h, n, 500)
        h_bk = project_feasible(h_bk, 500)
        c_bk = compute_c5(h_bk, 500)
        t6 = min(time.time() + 55, deadline - 35)
        h_bk, c_bk = lbfgs_opt(h_bk, 500, t6,
            alphas=[2000, 20000, 100000],
            mus=[200.0, 2000.0, 20000.0])
        if c_bk < best_c:
            best_c, best_h, n = c_bk, h_bk, 500

    # Phase 7: Momentum gradient at final resolution (~70s)
    if deadline - time.time() > 40:
        t7 = min(time.time() + 65, deadline - 25)
        h7, c7 = mom_grad(best_h, n, t7, lr0=0.001, alpha=10000, mu=0.15, mom=0.85)
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Phase 8: Coordinate / pair search (~50s)
    if deadline - time.time() > 25:
        t8 = min(time.time() + 45, deadline - 10)
        h8, c8 = coord_search(best_h, n, t8)
        if c8 < best_c:
            best_c, best_h = c8, h8

    # Phase 9: Final SLSQP polish
    if deadline - time.time() > 8:
        h9, c9 = slqp_run(best_h, n, deadline - 4, 3, 100000)
        if c9 < best_c:
            best_c, best_h = c9, h9

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
