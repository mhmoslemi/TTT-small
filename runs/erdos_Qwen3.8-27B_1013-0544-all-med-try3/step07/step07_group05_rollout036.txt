def shave_block(h, n, deadline, M=1, patience=6):
    h = h.copy()
    dx = 2.0/n
    S = n/2.0
    C = maxC(h)
    eta = 0.02
    since = 0
    topo = M
    maxeta=0.04
    mineta=1e-5
    it=0
    while time.time() < deadline:
        it+=1
        F = compute_F(h)
        C = F.max()*dx
        # active: top M
        Mnow = min(topo, F.size)
        idx = np.argpartition(F, -Mnow)[-Mnow:]  # top Mnow indices
        G = np.zeros(n)
        for kk in idx:
            G += grad_fk(h, kk-(n-1))

        # candidate
        cand = proj(h - eta*G, S)
        Cc = maxC(cand)
        if Cc < C - 1e-13:

            since = 0
            eta = min(eta*1.03, maxeta)

Alternatively, the sort-based: sort x; compute the projection by finding the free set.
Standard algorithm:
```
a = np.sort(x)           # ascending
# We need t such that sum clip(a - t,0,1) = S.
# Equivalent: find t. Use the fact that the solution's free indices are a contiguous block.
# Define prefix sums.
# Let cand: suppose free block is [p, q) (0-index), so a[p..q-1] are in (t, t+1), 
#   a[0..p-1] ≤ t  (project to 0), a[q..n-1] ≥ t+1 (project to 1).
# sum = (n - q) + sum_{i=p}^{q-1}(a_i - t) = (n-q) + (prefix[q]-prefix[p]) - (q-p)*t
# set = S => (q-p)*t = (n-q) + (prefix[q]-prefix[p]) - S
# t = [(n-q) + (prefix[q]-prefix[p]) - S]/(q-p)
# constraints: a[p-1] ≤ t, t < a[p], t > a[q-1]-1, a[q] ≥ t+1 (with boundary handling).
```
Finding the correct (p,q) requires iterating. This is O(n) with a while loop that adjusts p,q. Since h is usually "mostly interior" (values in (0,1)), p=0,q=n initially and t=(S - sum)/n = (n/2 - sum)/n small. If t < a[0] (i.e., a[0] > t) fine (no zeros), and t > a[n-1]-1 (i.e., a[n-1] < t+1) fine (no ones). If satisfied → all free, done in O(n log n) (the sort). If not, some saturated; adjust p or q. Typically 1-3 adjustments. So proj cost ~ O(n log n) + few O(n) → faster than 45-iter binary (which is 45*O(n)). For n=250, sort ~ 250*log250 ~ 2000 compares + a few O(250) → ~ 3000 ops, vs binary 45*250=11250. So sort-based ~3-4x faster. And exact. import numpy as np
import time
import sys
from scipy.optimize import minimize


def compute_F(h):
    return np.correlate(h, 1.0 - h, mode='full')


def maxF(h):
    return float(np.max(compute_F(h)))


def proj(h, S):
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    lo = -1.0
    hi = 1.0
    for _ in range(44):
        t = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - t, 0.0, 1.0))
        if s > S:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    y = np.clip(x - t, 0.0, 1.0)
    for _ in range(12):
        d = S - y.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - y
            tr = r.sum()
            if tr < 1e-14:
                break
            y = y + (d / tr) * r
        else:
            tr = y.sum()
            if tr < 1e-14:
                break
            y = y + (d / tr) * y
    return np.clip(y, 0.0, 1.0)


def grad_fk(h, k):
    n = h.size
    g = np.zeros(n)
    nlo = max(0, -k)
    nhi = min(n - 1, n - 1 - k)
    if nlo > nhi:
        return g
    a = max(0, nlo + k)
    b = min(n - 1, nhi + k)
    if a <= b:
        idx = np.arange(a, b + 1)
        g[idx] = g[idx] + (1.0 - h[idx - k])
    a2 = nlo
    b2 = min(nhi, n - 1 - k)
    if a2 <= b2:
        idx2 = np.arange(a2, b2 + 1)
        g[idx2] = g[idx2] - h[idx2 + k]
    return g


def Cof_d(h, i, j, d):
    hi = h[i] + d
    hj = h[j] - d
    if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
        return 1e9
    h2 = h.copy()
    h2[i] = hi
    h2[j] = hj
    return float(np.max(compute_F(h2)))


def maxF_h2(h, i, j, d):
    hi = h[i] + d
    hj = h[j] - d
    if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
        return 1e9
    h2 = h.copy()
    h2[i] = hi
    h2[j] = hj
    return float(np.max(compute_F(h2)))


def ls_convex(h, i, j, dlo, dhi, evals):
    if dhi <= dlo:
        return 0.0
    invgr = 0.6180339887498949
    a = dlo
    b = dhi
    x1 = a + (1.0 - invgr) * (b - a)
    x2 = a + invgr * (b - a)
    f1 = Cof_d(h, i, j, x1)
    f2 = Cof_d(h, i, j, x2)
    for _ in range(evals):
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = a + (1.0 - invgr) * (b - a)
            f1 = Cof_d(h, i, j, x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + invgr * (b - a)
            f2 = Cof_d(h, i, j, x2)
    dm = 0.5 * (a + b)
    fm = Cof_d(h, i, j, dm)
    if f1 < fm:
        dm = x1
        fm = f1
    if f2 < fm:
        dm = x2
    return dm


def shine(h, n, deadline, M=1):
    h = h.copy()
    S = n / 2.0
    C = maxF(h)
    eta = 0.02
    st = 0
    mineta = 1e-5
    maxeta = 0.05
    while time.time() < deadline:
        F = compute_F(h)
        Mn = min(M, F.size)
        if Mn >= F.size:
            idx = np.arange(F.size)
        else:
            idx = np.argpartition(F, -Mn)[-Mn:]
        G = np.zeros(n)
        for kk in idx:
            G = G + grad_fk(h, kk - (n - 1))
        G = G / Mn
        cand = proj(h - eta * G, S)
        Cc = maxF(cand)
        if Cc < C - 1e-13:
            h = cand
            C = Cc
            st = 0
            eta = min(eta * 1.06, maxeta)
        else:
            st += 1
            cand2 = proj(h - 0.5 * eta * G, S)
            Cc2 = maxF(cand2)
            if Cc2 < C - 1e-13:
                h = cand2
                C = Cc2
                st = 0
            elif st < 8:
                eta = max(eta * 0.85, mineta)
            else:
                eta = max(eta * 0.6, mineta)
                st = 0
    return h, C


def pair_block(h, n, deadline, radius=30, evals=26):
    h = h.copy()
    n2 = n - 1
    curF = maxF(h)
    while time.time() < deadline:
        K = int(np.argmax(compute_F(h))) - n2
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, curF
            hi = min(n, i + radius + 1)
            for j in range(i + 1, hi):
                dlo = max(-h[i], h[j] - 1.0)
                dhi = min(1.0 - h[i], h[j])
                if dhi <= dlo:
                    continue
                dr = ls_convex(h, i, j, dlo, dhi, evals)
                v = maxF_h2(h, i, j, dr)
                if v < curF - 1e-13:
                    h[i] += dr
                    h[j] -= dr
                    curF = v
                    improved = True
            for j in (i + K, i - K):
                if j == i or j < 0 or j >= n:
                    continue
                if abs(j - i) <= radius:
                    continue
                if time.time() > deadline:
                    return h, curF
                dlo = max(-h[i], h[j] - 1.0)
                dhi = min(1.0 - h[i], h[j])
                if dhi <= dlo:
                    continue
                dr = ls_convex(h, i, j, dlo, dhi, evals)
                v = maxF_h2(h, i, j, dr)
                if v < curF - 1e-13:
                    h[i] += dr
                    h[j] -= dr
                    curF = v
                    improved = True
        if not improved:
            break
    return h, curF


def slsq_obj(x, M):
    F = np.correlate(x, 1.0 - x, mode='full')
    order = np.sort(F)
    return float(order[-M:].sum())


def slsq_jac(x, M):
    n = x.size
    F = np.correlate(x, 1.0 - x, mode='full')
    thr = np.sort(F)[-M]
    idx = np.nonzero(F >= thr - 1e-12)[0]
    G = np.zeros(n)
    for kk in idx:
        G = G + grad_fk(x, kk - (n - 1))
    return G


def slsq_lcons(x):
    return float(x.sum() - x.size / 2.0)


def slsq_lcon_jac(x):
    return np.ones(x.size)


def slsqp_polish(h, n, deadline, M=3):
    h = proj(h.copy(), n / 2.0)
    best = h.copy()
    bestF = maxF(h)
    bounds = [(0.0, 1.0)] * n

                           options={'maxiter': 1200, 'ftol': 1e-14})
            x = proj(res.x, n / 2.0)
            f = maxF(x)
            if f < bestF - 1e-13:
                best = x.copy()
                bestF = f
            h = x.copy()
        except Exception:
            pass
    return best, bestF

            bestF = cF
            last = time.time()
        h = cur
        if time.time() >= deadline - 2.0:
            break
        tnow = time.time()
        rem = deadline - 2.0 - tnow
        if rem <= 0:
            break
        tp = min(100.0, max(5.0, rem * 0.5))
        t2 = min(time.time() + tp, deadline - 2.0)
        cur, cF = pair_block(h, n, t2, radius=rad)
        if cF < bestF - 1e-13:
            best = cur.copy()
            bestF = cF
            last = time.time()
        h = cur
        if time.time() >= deadline - 2.0:
            break
        if (time.time() - last) > 40.0:
            if time.time() < deadline - 22.0:
                sd = min(time.time() + 30.0, deadline - 2.0)
                shh, shF = slsqp_polish(h, n, sd, M=3)
                if shF < bestF - 1e-13:
                    best = shh.copy()
                    bestF = shF
                    last = time.time()
                    h = best.copy()
            h = perturb(best, n, scale, rng)
            scale = max(0.008, scale * 0.7)
            last = time.time()
    best = proj(best.copy(), n / 2.0)
    return best, maxF(best) * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    Bmax = max(20.0, float(budget_s))
    tdead = t0 + Bmax - 6.0
    rng = np.random.RandomState(seed)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None or h0.size < 10 or (not np.all(np.isfinite(h0))):
        h0 = none_start(seed)
    n0 = h0.size
    if h0.min() < 0.0 or h0.max() > 1.0 or abs(h0.sum() - n0 / 2.0) > 1e-6:
        h0 = proj(h0, n0 / 2.0)

    best = h0.copy()
    bestC = maxF(h0) * (2.0 / n0)
    bestn = n0
    print('baseline n=%d C=%.7f' % (bestn, bestC))
    sys.stdout.flush()

    # Track A: refine at the parent resolution
    deadlineA = min(tdead - 12.0, t0 + 0.36 * Bmax)
    if deadlineA > t0 + 15.0:
        hh, CC = refine(h0.copy(), n0, deadlineA, rng)
        if CC < bestC - 1e-12:
            best = hh
            bestC = CC
            bestn = n0
    print('A best n=%d C=%.7f' % (bestn, bestC))
    sys.stdout.flush()

    # Track B: finer resolution (re-fit the shape on a denser grid)
    tnow = time.time()
    if tnow < tdead - 70.0:
        nB = int(round(bestn * 1.6))
        if nB > bestn and nB <= 2000:
            x0 = np.linspace(0.0, 2.0, bestn, endpoint=False)
            xB = np.linspace(0.0, 2.0, nB, endpoint=False)
            hB = np.interp(xB, x0, best, period=2.0)
            hB = proj(hB, nB / 2.0)
            deadlineB = min(tdead - 16.0, tnow + 0.34 * Bmax)
            if deadlineB > tnow + 20.0:
                hh, CC = refine(hB, nB, deadlineB, rng)
                if CC < bestC - 1e-12:
                    best = hh
                    bestC = CC
                    bestn = nB
    print('B best n=%d C=%.7f' % (bestn, bestC))
    sys.stdout.flush()

    # Track C: still finer
    tnow = time.time()
    if bestn > 70 and bestn < 1400 and tnow < tdead - 55.0:
        nC = int(round(bestn * 1.5))
        if abs(nC - bestn) > 25 and nC <= 2000:
            x0 = np.linspace(0.0, 2.0, bestn, endpoint=False)
            xC = np.linspace(0.0, 2.0, nC, endpoint=False)
            hC = np.interp(xC, x0, best, period=2.0)
            hC = proj(hC, nC / 2.0)
            deadlineC = min(tdead - 10.0, tnow + 0.18 * Bmax)
            if deadlineC > tnow + 15.0:
                hh, CC = refine(hC, nC, deadlineC, rng)
                if CC < bestC - 1e-12:
                    best = hh
                    bestC = CC
                    bestn = nC
    print('C best n=%d C=%.7f' % (bestn, bestC))
    sys.stdout.flush()

    # Final polish at the best resolution
    tnow = time.time()
    if tnow < tdead - 3.0:
        deadlineF = min(tdead - 2.0, tnow + min(0.10 * Bmax, 130.0))
        if deadlineF > tnow + 8.0:
            hh, CC = refine(best.copy(), bestn, deadlineF, rng)
            if CC < bestC - 1e-12:
                best = hh
                bestC = CC

    best = proj(best.copy(), bestn / 2.0)
    bestC = maxF(best) * (2.0 / bestn)
    print('FINAL n=%d C=%.7f' % (bestn, bestC))
    sys.stdout.flush()
    return best, float(bestC), int(bestn)
