def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:

            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def lse_gradient(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

            break
        j = best_j
        ones_candidates = list(ones_set)
        if not ones_candidates:
            break
        rng_choice = it % len(ones_candidates)
        j_target = ones_candidates[rng_choice]
        if j_target == j:
            continue
        delta = np.zeros(2 * n - 1)
        m_range = np.arange(-(n - 1), n)
        im = j + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = j - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jmn = j_target + m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] -= 1.0 - h[jmn[msk]]
        jm2 = j_target - m_range
        msk = (jm2 >= 0) & (jm2 < n)
        delta[msk] += h[jm2[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j_target - j)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (j - j_target)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        new_corr = corr + delta
        new_max = new_corr.max()
        if new_max <= cur_c_raw:
            h[j] = 1.0
            h[j_target] = 0.0
            corr = new_corr
            cur_c_raw = new_max
            ones_set.discard(j_target)
            ones_set.add(j)
            zeros_set.discard(j)
            zeros_set.add(j_target)
            if cur_c_raw < best_c_raw:
                best_c_raw = cur_c_raw
                best_h = h.copy()
    return best_h, float(best_c_raw * dx)


def anneal_binary(hb, n, deadline, seed=42, n_iters=1000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = hb.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c_raw = corr.max()
    best_h = h.copy()
    best_c_raw = cur_c_raw
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    T = 0.05
    cool = 0.99995
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        new_max = (corr + delta).max()
        accept = new_max <= cur_c_raw
        if not accept and T > 1e-10:
            prob = np.exp(-max(0, new_max - cur_c_raw) * 100.0 / T)
            accept = rng.random() < prob
        if accept:
            h[i] = 1.0
            h[j] = 0.0
            corr += delta
            cur_c_raw = new_max
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c_raw < best_c_raw:
                best_c_raw = cur_c_raw
                best_h = h.copy()
        T *= cool
    return best_h, float(best_c_raw * dx)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0_low = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0_low)
    except Exception:
        n0 = 400
        h0_low = np.full(n0, 0.5)

    # Work at n=800 for continuous optimization
    n = 800
    if n0 == n:
        h0 = h0_low.copy()
    else:
        xo = np.linspace(0, 2, n0, endpoint=False)
        xn = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(xn, xo, h0_low)
        h0 = project(h0, n)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: PGD with multiple alphas (15% time)
    p1 = t0 + 0.15 * total
    alphas1 = [10, 50, 200, 1000, 5000]
    best_h, best_c = pgd_solve(best_h, n, alphas1, p1, max_iter=3000)

    # Phase 2: Binary local search at n=800 (25% time)
    p2 = t0 + 0.40 * total
    hb = to_binary(best_h, n)
    hb, cb = vectorized_binary_ls(hb, n, p2, max_sweeps=800)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: PGD from improved binary (15% time)
    p3 = t0 + 0.55 * total
    h_cont = best_h.copy()
    alphas3 = [500, 2000, 10000, 50000, 200000]
    h_cont, cc = pgd_solve(h_cont, n, alphas3, p3, max_iter=4000)
    if cc < best_c:
        best_c = cc
        best_h = h_cont.copy()

    # Phase 4: Binary local search again (15% time)
    p4 = t0 + 0.70 * total
    hb2 = to_binary(best_h, n)
    hb2, cb2 = vectorized_binary_ls(hb2, n, p4, max_sweeps=600)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()

    # Phase 5: Annealing for exploration (15% time)
    p5 = t0 + 0.85 * total
    ha = to_binary(best_h, n)
    ha, ca = anneal_binary(ha, n, p5, seed=seed, n_iters=2000000)
    if ca < best_c:
        best_c = ca
        best_h = ha.copy()

    # Phase 6: Final PGD polish (10% time)
    p6 = dl - 3
    if time.time() < p6 - 15:
        alphas6 = [5000, 20000, 100000, 500000]
        h_polish = best_h.copy()
        h_polish, cp = pgd_solve(h_polish, n, alphas6, p6, max_iter=3000)
        if cp < best_c:
            best_c = cp
            best_h = h_polish.copy()

    # Phase 7: Try binary local search on final answer
    if time.time() < dl - 5:
        hb_final = to_binary(best_h, n)
        hb_final, cbf = vectorized_binary_ls(hb_final, n, dl - 2, max_sweeps=200)
        if cbf < best_c:
            best_c = cbf
            best_h = hb_final.copy()

    # Final evaluation
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
