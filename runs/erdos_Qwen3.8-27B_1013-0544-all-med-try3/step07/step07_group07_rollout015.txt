import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, -5.0, 5.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-14:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

def adam_frac(h0, n, deadline, max_iters=100000):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [10, 50, 200, 1000, 5000, 20000, 50000, 100000, 200000, 500000]
    lr_base = 0.04
    for ai, alpha in enumerate(alphas):
        if time.time() > deadline:
            break
        h = best_h.copy()
        m = np.zeros(n)
        v = np.zeros(n)
        t_s = 0
        lr_cur = lr_base / (1.0 + ai * 0.2)
        iters = max(200, max_iters // len(alphas))
        for _ in range(iters):
            if time.time() > deadline:
                break
            t_s += 1
            g = lse_grad(h, n, alpha)
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * g * g
            m_hat = m / (1.0 - 0.9 ** t_s)
            v_hat = v / (1.0 - 0.999 ** t_s)
            h_new = h - lr_cur * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_new = project(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def make_delta(h, i, j, m_range, n):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def sa_binary(hb, n, deadline, seed=42, max_iters=5000000, T_start=0.02, T_end=0.0003):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = hb.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    m_range = np.arange(-(n - 1), n)
    log_T_start = np.log(max(T_start, 1e-10))
    log_T_end = np.log(max(T_end, 1e-10))
    log_cool = (log_T_end - log_T_start) / max(1, max_iters - 1)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = np.exp(log_T_start + log_cool * it)
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        delta = make_delta(h, i, j, m_range, n)
        nc = (corr + delta).max() * dx
        delta_c = nc - cur_c
        if delta_c <= 1e-14 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i], h[j] = 1.0, 0.0
            corr = corr + delta
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50000 and T > 0.001:
                log_T_start = np.log(0.01)
                log_cool = (log_T_end - log_T_start) / max(1, max_iters - it - 1)
                no_improve = 0
    return best_h, best_c
