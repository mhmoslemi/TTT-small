    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_for_lag(h, n, s):
    dx = 2.0 / n
    grad = dx * np.ones(n)
    t = abs(int(s))
    if t > 0:
        grad[:n - t] -= dx * h[t:]
        grad[t:] -= dx * h[:n - t]
    else:
        grad -= 2.0 * dx * h
    return grad

def projected_gradient(h, n, deadline, max_iters=50000, topk=3, smooth_temp=0.0005):
    C = compute_c5(h, n)
    center = n - 1
    eta = 0.5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        C = float(corr.max())
        # Find top-k active lags
        sorted_idx = np.argsort(corr)[::-1]
        k = min(topk, len(corr))
        active = sorted_idx[:k]
        active_vals = corr[active]
        # Softmax weights (higher weight to larger values)
        w = np.exp(-(active_vals - active_vals[0]) / smooth_temp)
        w = w / w.sum()
        # Weighted gradient
        g = np.zeros(n)
        for idx, p in enumerate(active):
            s = int(p) - center
            g += w[idx] * grad_for_lag(h, n, s)
        # Backtracking line search
        improved = False
        for trial in range(20):
            h_trial = project_feasible(h - eta * g, n)
            C_trial = compute_c5(h_trial, n)
            if C_trial < C - 1e-17:
                h = h_trial
                C = C_trial
                improved = True
                break
            eta *= 0.7
        if not improved:
            eta = max(eta * 0.5, 1e-6)
            stall += 1
            if stall > 30:
                break
        else:
            if stall > 0:
                eta = min(eta / 0.5, 1.0)
            stall = 0
    return h, compute_c5(h, n)

def slsqp_smooth(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            se = np.exp(sh - mx)
            sm = np.sum(se)
            vals = (mx + np.log(sm)) / alpha
        else:
            se = np.exp(sh)
            sm = np.sum(se)
            vals = np.log(sm) / alpha
        g = np.zeros(n)
        se_norm = se / sm
        for p in range(len(c)):
            s = p - (n - 1)
            t = abs(int(s))
            gl = np.zeros(n) + dx
            if t > 0:
                gl[:n - t] -= dx * x[t:]
                gl[t:] -= dx * x[:n - t]
            else:
                gl -= 2.0 * dx * x
            g += se_norm[p] * gl
        return g / alpha
    def sum_c(x):
        return float(x.sum() - n / 2.0)
    def sum_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_c, 'jac': sum_j}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search_local(h, n, deadline, deltas, radius=80):
