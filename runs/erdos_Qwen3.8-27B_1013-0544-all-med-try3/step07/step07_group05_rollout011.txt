        ht, ct = subgradient_optimize(best_h, n, time.time() + alloc)
        if ct < best_c:
            best_c = ct
            best_h = ht

    # Fine targeted search
    if time.time() < deadline - 15:
        alloc = min(80, (deadline - time.time()) * 0.4)
        best_h, best_c = targeted_search(best_h, n, time.time() + alloc,
                                          [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                          num_worst=4)

    # Triple moves
    if time.time() < deadline - 10:
        alloc = min(30, (deadline - time.time()) * 0.3)
        ht, ct = triple_move_search(best_h, n, time.time() + alloc,
                                     [0.005, 0.002, 0.001], num_worst=2)
        if ct < best_c:
            best_c = ct
            best_h = ht

    # Ultra-fine targeted search
    if time.time() < deadline - 5:
        alloc = min(50, (deadline - time.time()) * 0.8)
        best_h, best_c = targeted_search(best_h, n, time.time() + alloc,
                                          [0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002],
                                          num_worst=5, max_passes=50000)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = thue_morse_init(256)
        n_init = len(h_init)

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial at native resolution
    alloc1 = min(350, (deadline - t0) * 0.35)
    h1, c1 = optimize_single(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h1, c1

    # Phase 2: Thue-Morse initialization at n=256
    if time.time() < deadline - 100:
        n_tm = 256
        h_tm = thue_morse_init(n_tm)
        alloc2 = min(120, (deadline - time.time()) * 0.15)
        h_tm, c_tm = optimize_single(h_tm, n_tm, time.time() + alloc2, rng)
        if c_tm < best_c:
            best_c = c_tm
            best_h = h_tm

    # Phase 3: Higher resolution n=500
    if time.time() < deadline - 80:
        n_hr = 500
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_hr)
        h_hr = np.interp(x_new, x_old, best_h)
        h_hr = project_feasible(h_hr, n_hr)
        alloc3 = min(200, (deadline - time.time()) * 0.25)
        h_hr, c_hr = optimize_single(h_hr, n_hr, time.time() + alloc3, rng)
        if c_hr < best_c:
            best_c = c_hr
            best_h = h_hr

    # Phase 4: Random restarts
    if best_h is not None and time.time() < deadline - 60:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 70))
        scales = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]
        for r in range(n_restarts):
            if time.time() > deadline - 45:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_cur)
            hp = project_feasible(hp, n_cur)
            alloc_r = min(55, (deadline - time.time() - 45) * 0.5)
            if alloc_r < 20:
                break
            hp, cp = optimize_single(hp, n_cur, time.time() + alloc_r, rng)
            if cp < best_c:
                best_c = cp
                best_h = hp

    # Phase 5: Final ultra-fine polish
    if best_h is not None and time.time() < deadline - 3:
        n_fin = len(best_h)
        best_h, best_c = targeted_search(best_h, n_fin, deadline - 1,
                                          [0.00001, 0.000005, 0.000002, 0.000001],
                                          num_worst=5, max_passes=100000)
        best_h = project_feasible(best_h, n_fin)
        best_c = compute_c5(best_h, n_fin)

    if best_h is None:
        n = 250
        best_h = np.full(n, 0.5)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
