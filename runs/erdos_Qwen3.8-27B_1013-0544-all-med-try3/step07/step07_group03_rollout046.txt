def compute_c5(h, n):
    return float(np.max(compute_c5_array(h, n)))

def project_feasible(h, n):
    """Project to satisfy sum(h) = n/2 and 0 <= h <= 1"""
    h = h.copy()

            if tot < 1e-15:

def active_set_gradient(h, n, top_k=5):
    """Compute gradient of top-k overlap values"""
    c = compute_c5_array(h, n)
    dx = 2.0 / n
    # Find top k indices
