def pair_search(h,n,deadline):
  h=project; bestC=compute_c5; currentC=bestC
  deltas=[0.05,0.02,0.01,0.004,0.0015]
  for d in deltas:
    if time>deadline: break
    for pass_ in range(3):
      if time>deadline: break
      improved=False
      c=compute_lags(h,n)
      top_idx=np.argsort(c)[-5:]
      for idx in top_idx:
        if time>deadline: break

def project_feasible(v,n):
    v=np.asarray(v,dtype=float)
    if v.size != n: v=np.resize(v,n) # maybe? If wrong size? 
    v=np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    target=n/2.0
    lo=np.min(v)-1.0; hi=np.max(v)+1.0
    # ensure sum(lo) >= target, sum(hi) <= target
    for _ in range(20):
        s_lo=np.sum(np.clip(v-lo,0.0,1.0))
        if s_lo < target: lo -= 1.0
        else: break
    for _ in range(20):
        s_hi=np.sum(np.clip(v-hi,0.0,1.0))
        if s_hi > target: hi += 1.0
        else: break
    for _ in range(70):
        a=0.5*(lo+hi)
        s=np.sum(np.clip(v-a,0.0,1.0))
        if s > target: lo=a
        else: hi=a
    a=0.5*(lo+hi)
    out=np.clip(v-a,0.0,1.0)
    for _ in range(10):
       diff=target-out.sum()
       if abs(diff) <= 1e-14: break
       free=(out>1e-12)&(out<1.0-1e-12)
       if free.any(): out[free] += diff / float(free.sum())
       else: out += diff / n
       out=np.clip(out,0.0,1.0)
    return out

def sigmoid_arr(x):
    x = np.asarray(x, dtype=float)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out


def sigmoid(x):
    if _expit is not None:
        return np.asarray(_expit(x), dtype=float)
    return sigmoid_arr(x)


def project_feasible(v, n):
    v = np.asarray(v, dtype=float).reshape(-1)
    if v.size != n:
        v = np.resize(v, n)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * float(n)

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

    for _ in range(30):
        if float(np.sum(np.clip(v - lo, 0.0, 1.0))) < target:
            lo -= max(1.0, abs(lo)) * 0.5 + 1.0
        else:
            break
    for _ in range(30):
        if float(np.sum(np.clip(v - hi, 0.0, 1.0))) > target:
            hi += max(1.0, abs(hi)) * 0.5 + 1.0
        else:
            break

    for _ in range(70):
        a = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - a, 0.0, 1.0)))
        if s > target:
            lo = a
        else:
            hi = a

    a = 0.5 * (lo + hi)
    out = np.clip(v - a, 0.0, 1.0)

    for _ in range(12):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-15:
            break
        free = (out > 1e-12) & (out < 1.0 - 1e-12)
        if np.any(free):
            out[free] += diff / float(np.sum(free))
        else:
            out += diff / float(n)
        out = np.clip(out, 0.0, 1.0)

    return out


def compute_lags(h, n):
    b = 1.0 - h
    c = np.correlate(h, b, mode='full')
    return c * (2.0 / float(n))


def compute_c5(h, n):
    return float(np.max(compute_lags(h, n)))


def add_lag_gradient(G, h, n, lag, weight):
    if weight == 0.0:
        return
    if lag == 0:
        G += weight * (1.0 - 2.0 * h)
        return
    m = abs(lag)
    if m >= n:
        return
    if lag > 0:
        G[:n - m] += weight * (1.0 - h[m:])
        G[m:] -= weight * h[:n - m]
    else:
        G[:n - m] -= weight * h[m:]
        G[m:] += weight * (1.0 - h[:n - m])


def weighted_gradient_from_ids(h, n, ids, weights):
    G = np.zeros(n, dtype=float)
    for r in range(len(ids)):
        w = float(weights[r])
        if w <= 1e-16:
            continue
        add_lag_gradient(G, h, n, int(ids[r]) - (n - 1), w)
    return G


def top_logsum_gradient(c, h, n, alpha, top_m):
    c = np.asarray(c, dtype=float)
    alpha = float(alpha)
    if alpha <= 0.0:
        alpha = 1.0
    L = int(c.size)
    m = int(min(max(1, int(top_m)), L))
    if m >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -m)[-m:]
    cmax = float(np.max(c))
    s = alpha * (c[idx] - cmax)
    s = np.maximum(s, -745.0)
    ex = np.exp(s)
    summ = float(np.sum(ex))
    if summ <= 0.0:
        w = np.full(m, 1.0 / float(m))
    else:
        w = ex / summ
    val = cmax + float(math.log(summ) if summ > 0.0 else 0.0) / alpha
    G = weighted_gradient_from_ids(h, n, idx, w)
    G *= (2.0 / float(n))
    return float(val), G

    a = 0.5 * (lo + hi)
    return sigmoid(z - a)


def z_from_h(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size != n:
        h = np.resize(h, n)
    hh = np.clip(h, 1e-10, 1.0 - 1e-10)
    z = np.log(hh / (1.0 - hh))
    z = np.clip(z, -15.0, 15.0)
    z = z - float(np.median(z))
    return z


def lbfgs_top_obj(z, n, alpha, top_m):
    h = calibrate_z(z, n)
    c = compute_lags(h, n)
    val, G = top_logsum_gradient(c, h, n, alpha, top_m)
    p = h * (1.0 - h)
    P = float(np.sum(p))
    if P <= 1e-14:
        grad = np.zeros_like(z)
    else:
        dot = float(np.dot(G, p))
        grad = p * (G - dot / P)
    return float(val), grad

def cvx_linear_step(h, n, trust, Gmat, cvals, t0):

        return None
    try:
        d = cp.Variable(n)
        t = cp.Variable()
        lb = np.maximum(-h, -trust)
        ub = np.minimum(1.0 - h, trust)
        cons = [
            d >= lb,
            d <= ub,
            cp.sum(d) == 0,
            t >= 0.0,
            t <= t0 + 1e-8,
            (Gmat @ d) + cvals <= t
        ]
        prob = cp.Problem(cp.Minimize(t), cons)
        dval = None
        tval = None
        solvers = [getattr(cp, 'GLOP', None), getattr(cp, 'CBC', None)]
        for solver in solvers:
            if solver is None:
                continue
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    if d.value is not None and t.value is not None:
                        dval = np.asarray(d.value, dtype=float).reshape(n)
                        tval = float(t.value)
                        break
            except Exception:
                pass
        if dval is None:
            return None
        h_new = project_feasible(h + dval, n)
        return h_new, compute_c5(h_new, n), tval
    except Exception:
        return None

        return cvxval

    h_new, t_pred = step_val
    h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new, n)
    return h_new, c_new, t_pred


def choose_lags_full(n):
    L = 2 * n - 1
    if n <= 330:
        return 0
    if n <= 520:
        return min(L, 320)
    return min(L, 260)


def choose_lags_small(n):
    L = 2 * n - 1
    if n <= 330:
        return min(L, 180)
    if n <= 520:
        return min(L, 160)
    return min(L, 120)


def lp_refine(h, n, deadline, max_iters, trust_start, n_lags):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    trust = float(trust_start)
    stall = 0

    for _ in range(max(1, int(max_iters))):
        if time.time() > deadline - 1.5:
            break
        step = linear_step(h, n, trust, n_lags)
        if step is None:
            trust *= 0.65
            stall += 1
            if stall >= 6 and trust < 0.004:
                break
            continue

        h_new, c_new, t_pred = step
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-13:
            best_h = h_new.copy()
            best_c = c_new

        accept = False
        if c_new < current_c - 1e-14:
            accept = True
        elif c_new <= current_c + 1e-11 and t_pred < current_c - 1e-9 and stall < 3:
            accept = True

        if accept:
            h = h_new
            current_c = c_new
            stall = 0
            trust = min(float(trust_start), trust * 1.04)
        else:
            trust *= 0.65
            stall += 1
            if stall >= 6 and trust < 0.004:
                break

    return best_h, best_c


def pair_search(h, n, deadline):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    ktop = min(5, 2 * n - 1)
    deltas = [0.05, 0.02, 0.01, 0.004, 0.0015, 0.0006]

    for delta in deltas:
        if time.time() > deadline:
            break
        for _pass in range(2):
            if time.time() > deadline:
                break
            improved = False
            c = compute_lags(h, n)
            top_idx = np.argsort(c)[-ktop:]
            for idx in top_idx:
                if time.time() > deadline:
                    break
                lag = int(idx) - (n - 1)
                m = abs(lag)
                if m == 0 or m >= n:
                    continue
                limit = n - m
                for i in range(limit):
                    if time.time() > deadline:
                        break
                    j = i + m
                    hi = float(h[i])
                    hj = float(h[j])
                    if hi > hj + 1e-10:
                        d = delta
                        maxd = 0.5 * (hi - hj)
                        if d > maxd:
                            d = maxd
                        if d > 1e-9:
                            h[i] = hi - d
                            h[j] = hj + d
                            cnew = compute_c5(h, n)
                            if cnew < current_c - 1e-14:
                                current_c = cnew
                                if cnew < best_c:
                                    best_c = cnew
                                improved = True
                            else:
                                h[i] = hi
                                h[j] = hj
                    elif hj > hi + 1e-10:
                        d = delta
                        maxd = 0.5 * (hj - hi)
                        if d > maxd:
                            d = maxd
                        if d > 1e-9:
                            h[i] = hi + d
                            h[j] = hj - d
                            cnew = compute_c5(h, n)
                            if cnew < current_c - 1e-14:
                                current_c = cnew
                                if cnew < best_c:
                                    best_c = cnew
                                improved = True
                            else:
                                h[i] = hi
                                h[j] = hj
            if not improved:
                break
            h = project_feasible(h, n)
    return project_feasible(h, n), best_c


def interpolate_h(hsrc, nsrc, n):
    if nsrc == n:
        return project_feasible(hsrc, n)
    x_src = np.linspace(0.0, 2.0, int(nsrc))
    x_new = np.linspace(0.0, 2.0, int(n))
    out = np.interp(x_new, x_src, np.asarray(hsrc, dtype=float))
    return project_feasible(out, n)


def generate_ar_binary(n, rng, rho):
    w = np.empty(n, dtype=float)
    x = 0.0
    coeff = math.sqrt(max(0.0, 1.0 - rho * rho))
    for i in range(n):
        x = rho * x + coeff * rng.randn()
        w[i] = x
    if n % 2 == 0:
        k = n // 2
        order = np.argsort(w)
        h = np.zeros(n, dtype=float)
        h[order[-k:]] = 1.0
    else:
        h = sigmoid(1.2 * w)
    return project_feasible(h, n)


def generate_best_start(n, rng):
    best = None
    best_c = float('inf')
    for rho in (0.90, 0.93, 0.95, 0.97, 0.98):
        h = generate_ar_binary(n, rng, rho)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best = h
    return best, best_c


def shift_candidates(h, n, deadline):
    base = project_feasible(h, n)
    best = base.copy()
    best_c = compute_c5(best, n)
    shifts = []
    raw = (-12, -6, -3, -1, 1, 3, 6, 12, int(n // 4), -int(n // 4))
    for s in raw:
        sm = int(s) % int(n)
        if 0 < sm < int(n) and sm not in shifts:
            shifts.append(sm)
    for sm in shifts:
        if time.time() > deadline:
            break
        hs = np.roll(base, sm)
        hs = project_feasible(hs, n)
        cs = compute_c5(hs, n)
        if cs < best_c - 1e-12:
            best = hs
            best_c = cs
    return best, best_c


def optimize_window(h0, n, time_limit, rng, quick=False):
    t_start = time.time()
    deadline = t_start + max(1.0, float(time_limit))
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if quick:
        if deadline - time.time() > 12:
            maxit = 25 if n <= 400 else 15
            h_new = lbfgs_once(h, n, 800.0, 18, maxit)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                h = h_new
                best_h = h_new.copy()
                best_c = c_new
        if deadline - time.time() > 6:
            h_new, c_new = lp_refine(h, n, deadline - 1, 10, 0.03, choose_lags_small(n))
            if c_new < best_c - 1e-13:
                h = h_new
                best_h = h_new.copy()
                best_c = c_new
        if deadline - time.time() > 4 and n <= 420:
            h_new, c_new = pair_search(h, n, deadline - 0.5)
            if c_new < best_c - 1e-13:
                best_h = h_new.copy()
                best_c = c_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
        return best_h, best_c

    if deadline - time.time() > 40:
        if n <= 350:
            base_iter = 80
        elif n <= 500:
            base_iter = 55
        else:
            base_iter = 35
        for alpha, top in ((30.0, 60), (100.0, 40), (400.0, 25)):
            if deadline - time.time() > 12:
                h_new = lbfgs_once(h, n, alpha, top, base_iter)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-13:
                    h = h_new
                    best_h = h_new.copy()
                    best_c = c_new

    if n <= 420 and deadline - time.time() > 35:
        maxit = 22 if n <= 320 else 14
        for alpha, top in ((500.0, 18), (2000.0, 10)):
            if deadline - time.time() > 12:
                h_new = slsqp_once(h, n, alpha, top, maxit)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-13:
                    h = h_new
                    best_h = h_new.copy()
                    best_c = c_new

    if deadline - time.time() > 12:
        hmaxit = 35 if n <= 320 else (20 if n <= 520 else 12)
        h_new, c_new = lp_refine(h, n, deadline - 2, hmaxit, 0.07, choose_lags_full(n))
        if c_new < best_c - 1e-13:
            h = h_new
            best_h = h_new.copy()
            best_c = c_new

    if deadline - time.time() > 12:
        if n <= 350:
            base_iter = 45
        elif n <= 500:
            base_iter = 30
        else:
            base_iter = 20
        for alpha, top in ((1500.0, 12), (6000.0, 8)):
            if deadline - time.time() > 8:
                h_new = lbfgs_once(h, n, alpha, top, base_iter)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-13:
                    h = h_new
                    best_h = h_new.copy()
                    best_c = c_new

    if n <= 360 and deadline - time.time() > 20:
        if deadline - time.time() > 10:
            h_new = slsqp_once(h, n, 3000.0, 8, 18)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                h = h_new
                best_h = h_new.copy()
                best_c = c_new

    if deadline - time.time() > 8:
        hmaxit = 25 if n <= 320 else 12
        h_new, c_new = lp_refine(h, n, deadline - 1, hmaxit, 0.02, choose_lags_small(n))
        if c_new < best_c - 1e-13:
            h = h_new
            best_h = h_new.copy()
            best_c = c_new

    if deadline - time.time() > 5 and n <= 520:
        h_new, c_new = pair_search(h, n, deadline - 0.5)
        if c_new < best_c - 1e-13:
            best_h = h_new.copy()
            best_c = c_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(60.0, float(budget_s))
    deadline = t0 + budget - 12.0
    rng = np.random.RandomState(int(seed))

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float).reshape(-1)
    except Exception:
        parent = None

    if parent is not None and parent.size >= 8:
        best_n = int(parent.size)
        best_h = project_feasible(parent, best_n)
    else:
        best_n = 300
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)

    best_c = compute_c5(best_h, best_n)

    if time.time() < deadline - 30:
        gen_h, gen_c = generate_best_start(best_n, rng)
        if gen_h is not None and gen_c < best_c:
            best_h = gen_h
            best_c = gen_c

    if time.time() < deadline - 60:
        alloc = min(430.0, (deadline - time.time()) * 0.45)
        if alloc > 40:
            h_opt, c_opt = optimize_window(best_h, best_n, alloc, rng, False)
            if c_opt < best_c - 1e-12:
                best_h = h_opt
                best_n = int(len(h_opt))
                best_c = c_opt

    if time.time() < deadline - 220 and best_n % 2 == 0:
        gen_h, gen_c = generate_best_start(best_n, rng)
        if gen_h is not None and gen_c < best_c + 0.05:
            alloc = min(160.0, (deadline - time.time()) * 0.22)
            if alloc > 45:
                h_opt, c_opt = optimize_window(gen_h, best_n, alloc, rng, alloc < 90)
                if c_opt < best_c - 1e-12:
                    best_h = h_opt
                    best_n = int(len(h_opt))
                    best_c = c_opt

    nearby = []
    for d in (-2, -1, 1, 2):
        nt = best_n + d
        if nt >= 32 and nt not in nearby:
            nearby.append(nt)
    for nt in nearby:
        if time.time() > deadline - 45:
            break
        alloc = min(35.0, (deadline - time.time()) * 0.10)
        if alloc < 18:
            break
        h0 = interpolate_h(best_h, best_n, nt)
        h_opt, c_opt = optimize_window(h0, int(nt), alloc, rng, True)
        if c_opt < best_c - 1e-12:
            best_h = h_opt
            best_n = int(len(h_opt))
            best_c = c_opt

    large = []
    for mult in (1.2, 1.5, 1.8, 2.2):
        nt = int(round(best_n * mult))
        if nt >= best_n + 10 and nt <= 900 and nt not in large:
            large.append(nt)
    for nt in (420, 560, 720):
        if nt >= best_n + 10 and nt <= 900 and nt not in large:
            large.append(nt)
    large.sort()
    if len(large) > 5:
        large = large[:5]

    allocs = [160.0, 120.0, 95.0, 75.0, 55.0]
    for rank, nt in enumerate(large):
        if time.time() > deadline - 70:
            break
        nt = int(nt)
        remaining = deadline - time.time()
        alloc = allocs[min(rank, len(allocs) - 1)]
        alloc = min(alloc, remaining * 0.35)
        if alloc < 35:
            break

        h_interp = interpolate_h(best_h, best_n, nt)
        c_interp = compute_c5(h_interp, nt)
        h0 = h_interp
        if nt % 2 == 0:
            h_gen, c_gen = generate_best_start(nt, rng)
            if h_gen is not None and c_gen < c_interp:
                h0 = h_gen

        quick = alloc < 75
        h_opt, c_opt = optimize_window(h0, nt, alloc, rng, quick)
        if c_opt < best_c - 1e-12:
            best_h = h_opt
            best_n = int(len(h_opt))
            best_c = c_opt

    if time.time() < deadline - 25:
        h_sh, c_sh = shift_candidates(best_h, best_n, deadline - 5)
        if c_sh < best_c - 1e-12:
            best_h = project_feasible(h_sh, best_n)
            best_c = c_sh
            if time.time() < deadline - 30:
                h_opt, c_opt = optimize_window(best_h, best_n, min(60.0, deadline - time.time() - 10), rng, True)
                if c_opt < best_c - 1e-12:
                    best_h = h_opt
                    best_n = int(len(h_opt))
                    best_c = c_opt

    if time.time() < deadline - 20:
        rem = deadline - time.time() - 5.0
        if rem > 18:
            h_opt, c_opt = optimize_window(best_h, best_n, rem, rng, rem < 80)
            if c_opt < best_c - 1e-12:
                best_h = h_opt
                best_n = int(len(h_opt))
                best_c = c_opt

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
