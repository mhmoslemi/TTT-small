           cmax = C.max()
           w = np.exp(alpha*(C - cmax))
           w /= w.sum()
           
           # Gradient
           grad = np.zeros(n)
           # Lag 0 contribution
           grad += w[n-1] * (1 - 2*h)
           
           # Other lags
           for m in range(2*n-1):
               if m == n-1: continue
               k = m - (n-1)
               if w[m] < 1e-10: continue  # skip negligible weights
               # dC[m]/dh[j] = (1-h[j+k])*I(0≤j+k<n) - h[j-k]*I(0≤j-k<n)
               if k > 0:
                   # (1-h[j+k]) for j=0..n-1-k
                   grad[:n-k] += w[m] * (1 - h[k:])  # h[j+k] for j=0..n-1-k → h[k:n]
                   # -h[j-k] for j=k..n-1
                   grad[k:] -= w[m] * h[:n-k]  # h[j-k] for j=k..n-1 → h[0:n-k]
               else:  # k < 0
                   k2 = -k  # positive
                   # (1-h[j+k]) = (1-h[j-k2]) for j=k2..n-1
                   grad[k2:] += w[m] * (1 - h[:n-k2])
                   # -h[j-k] = -h[j+k2] for j=0..n-1-k2
                   grad[:n-k2] -= w[m] * h[k2:]

           for _ in range(30):

               C_try = np.correlate(h_try, 1-h_try, 'full')
               c_try = C_try.max() * dx
               if c_try < best_c - 1e-12 * best_c or c_try < cmax * dx:
                   h = h_try
                   best_c = min(best_c, c_try)
                   break
               lr *= 0.5
           else:
               # No improvement, try next alpha
               break
       
       return h, best_c
