    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        exp_shifted = np.exp(shifted - mx)
        weights = exp_shifted / exp_shifted.sum()
    else:
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / exp_shifted.sum()
    return m + np.log(exp_shifted.sum()) / alpha, weights

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_full(h, n, alpha, maxiter=5000, ftol=1e-16):
    """SLSQP on the full smooth max."""
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        return smooth_max(c, alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n), n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def slsqp_active(h, n, active_idx, alpha, maxiter=3000, ftol=1e-15):
    """SLSQP on a subset of active lags."""
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        return smooth_max(c[active_idx], alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n), n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def outer_approx_step(h_curr, n, trust, top_k=10):
    """One step of outer approximation: linearize top_k lags, solve LP."""
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    
    # Select top_k most active lags
    top_idx = np.argsort(c)[-top_k:]
    
    # Build LP
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0,
            h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    
    for idx in top_idx:
        lag = int(idx - (n - 1))
        grad = compute_lag_gradient(h_curr, n, lag)
        # Linearization: c[idx] + dx * (grad @ (h - h_curr)) <= t
        # Wait, c is already scaled by dx. Let me be careful.
        # c[idx] = sum_i h[i] * (1-h[i+lag]) * dx  (approximately, with the correlate formula)
        # Actually, c = np.correlate(h, 1-h, mode='full') * dx
        # The linearization of the UNSCALED correlation at h_curr is:
        # corr[idx] + (grad_unscaled @ (h - h_curr))
        # where grad_unscaled is the gradient of the unscaled correlation
        # And c[idx] = corr[idx] * dx
        # So: c[idx] * (1/dx) + (grad @ (h - h_curr)) * (1/dx) ... 
        # Hmm, let me think about this differently.
        
        # The correlation value (unscaled) at h is: corr(h)[idx]
        # The C5 contribution is: corr(h)[idx] * dx
        # We want: corr(h)[idx] * dx <= t
        # Linearize corr(h)[idx] at h_curr:
        # corr(h)[idx] ≈ corr(h_curr)[idx] + grad_corr[idx] @ (h - h_curr)
        # where grad_corr[idx] is the gradient of the unscaled correlation w.r.t. h
        # 
        # The unscaled correlation is: sum_i h[i] * (1 - h[i+lag])
        # Its gradient w.r.t. h[j] is: (the gradient I computed above)
        # But wait, my compute_lag_gradient computes the gradient of the SCALED value?
        # Let me check: in the previous code, it's used as:
        # c[k] + dx * (grad @ (h_var - h_curr)) <= t_var
        # So c[k] is the scaled value (already has dx), and grad @ (h-h_curr) is unscaled
        # This means: c[k] + dx * (grad @ (h-h_curr)) <= t
        # Which is: corr_scaled[k] + dx * grad_unscaled @ (h-h_curr) <= t
        # = corr_unscaled[k]*dx + dx * grad_unscaled @ (h-h_curr) <= t
        # = dx * (corr_unscaled[k] + grad_unscaled @ (h-h_curr)) <= t
        # = dx * (linearized correlation) <= t
        # This is correct!
        
        cons.append(c[idx] + dx * (grad @ (h_var - h_curr)) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    
    # Try solvers in order
    for solver in [cp.CBC, cp.GLOP, cp.GLPK]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy(), compute_c5(h_curr, n)
    
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy(), compute_c5(h_curr, n)
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new, n)

def run_outer_approx(h, n, deadline, max_iters=50, top_k=10):
    """Run outer approximation until deadline."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        
        h_new, c_new = outer_approx_step(h_curr, n, trust, top_k)
        
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        
        step = np.max(np.abs(h_new - h_curr))
        h_curr = h_new
        
        if step < 1e-9:
            break
        trust = max(0.003, trust * 0.85)
    
    return best_h, best_c

def gradient_local_search(h, n, deadline, top_m=5):
    """Gradient-based local search focusing on active lags."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    lr = 0.001
    for _ in range(2000):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        max_c = c.max()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx - (n - 1))
            lgrad = compute_lag_gradient(h_curr, n, lag)
            # Weight by how close to max this lag is
            w = np.exp(-(max_c - c[idx]) * 50)
            grad += w * lgrad
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Project gradient to maintain sum constraint
        grad -= np.mean(grad)
        
        # Line search
        improved = False
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1]:
            h_try = h_curr - trial_lr * grad
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                h_curr = h_try
                best_c = c_try
                best_h = h_try.copy()
                improved = True
                break
            elif c_try < compute_c5(h_curr, n):
                h_curr = h_try
        
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break
        else:
            lr = min(lr * 1.1, 0.01)
    
    return best_h, best_c

def pairwise_local_search(h, n, deadline):
    """Pairwise swap local search (like the previous code but improved)."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            # Try all top lags
            top_lags = np.argsort(c)[-8:]
            for k_star in top_lags:
                if time.time() > deadline:
                    break
                lag = int(k_star - (n - 1))
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] == h[j]:
                        continue
                    if h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    else:
                        delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if delta < 1e-8:
                        continue
                    if h[i] > h[j]:
                        h[i] -= delta
                        h[j] += delta
                    else:
                        h[i] += delta
                        h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        if h[i] > h[j]:
                            h[i] += delta
                            h[j] -= delta
                        else:
                            h[i] -= delta
                            h[j] += delta
    
    return h, best_c
