import numpy as np
import time
from scipy.fft import fft, ifft

def compute_corr(h, n):
    """Compute full cross-correlation of h and 1-h."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c, float(np.max(c) * dx)

def compute_corr_fft(h, n):
    """Compute full cross-correlation using FFT."""
    g = 1.0 - h
    size = 2 * n - 1
    Fh = fft(h, size)
    Fg = fft(g, size)
    corr = np.real(ifft(Fh * np.conj(Fg)))
    return corr

def project(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary by taking top n/2 values as 1."""
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def swap_delta(h, corr, n, i, j):
    """Compute the change in correlation vector when swapping h[i] and h[j].
    i becomes 1, j becomes 0 (i was 0, j was 1)."""
    # This is for the case h[i]=0, h[j]=1
    # New: h[i]=1, h[j]=0
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # Effect of setting h[i] from 0 to 1:
    # corr[k] = sum_{valid} h[a]*(1-h[a+k])
    # When h[i] goes 0->1:
    #   As h[a] term: adds (1-h[i+k]) for k where i is in range
    #   As h[a+k] term: adds -h[i-k] for k where i+k is in range (i.e., a=i-k, h[a+k]=h[i])

    # h[i] appears as h[a+k] when a+k=i, so a=i-k, contributes -h[i-k] if 0<=i-k<n

    # As h[a]: delta += (1-h[i+k]) for valid k (i.e., 0 <= i+k < n)
    # As h[b] in (1-h[b]): delta -= h[i-k] for valid k (i.e., 0 <= i-k < n)
    
    for k in range(-(n-1), n):
        corr_idx = k + n - 1
        if 0 <= i + k < n:
            delta[corr_idx] += (1.0 - h[i + k])
        if 0 <= i - k < n:
            delta[corr_idx] -= h[i - k]
    
    # Setting h[j]: 1 -> 0

        corr_idx = k + n - 1
        if 0 <= j + k < n:
            delta[corr_idx] -= (1.0 - h[j + k])
        if 0 <= j - k < n:
            delta[corr_idx] += h[j - k]
    
    return delta, (corr + delta).max()

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c, float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def pos_delta(h, n, m_range, pos, dh):
    """Delta in corr when h[pos] changes by dh."""
    delta = np.zeros(2 * n - 1)
    im = pos + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += dh * (1.0 - h[im[msk]])
    imn = pos - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= dh * h[imn[msk]]
    return delta

def best_swap_search(h, corr, n, deadline, max_iters=1000):
    """Best-improvement 1-swap local search."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    one_mask = (h == 1)
    zero_mask = (h == 0)
    ones = np.where(one_mask)[0]
    zeros = np.where(zero_mask)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        n_zeros = len(zeros)
        n_ones = len(ones)
        if n_zeros == 0 or n_ones == 0:
            break
        
        # Precompute deltas for all zeros and ones
        zero_deltas = np.zeros((n_zeros, 2*n-1))
        for ii in range(n_zeros):
            zero_deltas[ii] = pos_delta(h, n, m_range, zeros[ii], 1.0)
        
        one_deltas = np.zeros((n_ones, 2*n-1))
        for jj in range(n_ones):
            one_deltas[jj] = pos_delta(h, n, m_range, ones[jj], -1.0)
        
        # Find best swap
        best_val = cur_c
        best_ii = -1
        best_jj = -1
        for ii in range(n_zeros):
            base = corr + zero_deltas[ii]
            # For each one, check max
            # new_max[jj] = (base + one_deltas[jj]).max()
            # Use: max of (base + one_deltas[jj]) for all jj
            tmp = base[None, :] + one_deltas  # shape (n_ones, 2n-1)
            maxes = tmp.max(axis=1)
            min_idx = np.argmin(maxes)
            if maxes[min_idx] * dx < best_val - 1e-15:
                best_val = maxes[min_idx] * dx
                best_ii = ii
                best_jj = min_idx
        
        if best_ii < 0:
            break
        
        # Apply swap
        i = zeros[best_ii]
        j = ones[best_jj]
        h[i] = 1.0
        h[j] = 0.0
        corr += zero_deltas[best_ii] + one_deltas[best_jj]
        cur_c = corr.max() * dx
        
        # Update masks
        zero_mask[i] = False
        zero_mask[j] = True
        one_mask[i] = True
        one_mask[j] = False
        zeros = np.where(zero_mask)[0]
        ones = np.where(one_mask)[0]
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
