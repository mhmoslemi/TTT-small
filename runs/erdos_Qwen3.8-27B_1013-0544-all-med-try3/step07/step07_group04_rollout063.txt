        c = np.correlate(h, 1 - h, mode='full') * dx_arr(n)
        c_raw = np.correlate(h, 1 - h, mode='full')
        # Top active lags
        top_m = 8
        top_idx = np.argsort(c_raw)[-top_m:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += compute_lag_gradient(h, n, int(k))
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        step = alpha / grad_norm
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            alpha = min(alpha * 1.05, 0.3)
        else:
            h_new = 0.5 * h + 0.5 * h_new
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            alpha *= 0.9
        h = h_new
    return best_h, best_c

def dx_arr(n):
    return 2.0 / n

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Subgradient
    sg_end = min(deadline - 20, time.time() + min(60, (deadline - time.time()) * 0.3))
    if time.time() < sg_end:
        h_sg, c_sg = subgradient_optimize(best_h, n, sg_end)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    
    # Phase 2: SLSQP with smooth max
    if time.time() < deadline - 15:
        for alpha in [10000, 50000, 200000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=3000, ftol=1e-14)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 3: Active set SLSQP
    if time.time() < deadline - 15:
        c = np.correlate(best_h, 1 - best_h, mode='full')
        top_idx = np.argsort(c)[-6:]
        for alpha in [50000, 200000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=3000, ftol=1e-14, top_idx=top_idx)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                # Update top set
                c = np.correlate(best_h, 1 - best_h, mode='full')
                top_idx = np.argsort(c)[-6:]
    
    # Phase 4: Coordinate descent
    if time.time() < deadline - 5:
        h_cd, c_cd = coordinate_descent(best_h, n, min(deadline - 2, time.time() + 40))
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    # Phase 5: Pair exchange
    if time.time() < deadline - 2:
        h_px, c_px = pair_exchange_search(best_h, n, deadline - 1)
        if c_px < best_c:
            best_c = c_px
            best_h = h_px
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def try_outer_approx(h, n, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if time.time() < deadline:
        h_oo, c_oo = outer_approx(best_h, n, min(deadline, time.time() + 80))
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # First: optimize at the parent resolution
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Try outer approximation at parent resolution
    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        h_oo, c_oo = try_outer_approx(best_h, n, min(deadline - 100, time.time() + 120))
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    
    # Try other resolutions
    n_candidates = [200, 400, 250, 350, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.22)
        if alloc < 40:
            break
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 35) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final refinement
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # High-alpha SLSQP
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=4000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Coordinate descent
        if time.time() < deadline - 3:
            h_cd, c_cd = coordinate_descent(best_h, n, deadline - 2)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd
        # Pair exchange
        if time.time() < deadline - 1:
            h_px, c_px = pair_exchange_search(best_h, n, deadline - 0.5)
            if c_px < best_c:
                best_c = c_px
                best_h = h_px
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
