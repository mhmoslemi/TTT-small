def compute_c5(h, n):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_full(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    T = 0.5 * float(n)

    for _ in range(1500):
        S = float(np.sum(h))
        diff = T - S
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(np.sum(room))
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = S
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h

    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        S = float(np.sum(h))
        diff = T - S
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            cap = room[idx]
            if cap <= 1e-15:
                break
            h[idx] = min(1.0, h[idx] + min(diff, cap))
        else:
            idx = int(np.argmax(h))
            cap = h[idx]
            if cap <= 1e-15:
                break
            h[idx] = max(0.0, h[idx] - min(-diff, cap))
    return h


def sym_jacobian(n):
    if n % 2 == 0:
        return np.ones(n // 2, dtype=float)
    m = (n - 1) // 2
    return np.concatenate((np.full(m, 2.0, dtype=float), np.array([1.0], dtype=float)))

    m = (n - 1) // 2
    return np.concatenate((v[:m], np.array([v[m]]), v[:m][::-1]))

def interpolate_h(h_old, n_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    if n_new == n_old:
        return h_old.copy()
    dx_old = 2.0 / float(n_old)
    dx_new = 2.0 / float(n_new)
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    return np.interp(x_new, x_old, h_old)


def full_h_loss_core(h, n, alpha):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx

    mf = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mf):
        return 1e100, np.zeros(n, dtype=float)

    s = (c - mf) * alpha
    smax = float(np.max(s))
    exps = np.exp(s - smax)
    zsum = float(np.sum(exps))

    if zsum <= 0.0 or not np.isfinite(zsum):
        w = np.full(c.size, 1.0 / float(c.size), dtype=float)
        f = mf
    else:
        f = mf + (smax + math.log(zsum)) / alpha
        w = exps / zsum

    p = np.zeros(n, dtype=float)
    q = np.zeros(n, dtype=float)
    p[0] = w[n - 1]
    if n > 1:
        p[1:] = w[n - 2::-1]
        q[1:] = w[n:2 * n - 1]

    r = p + q
    pr = np.empty(n + 1, dtype=float)
    pr[0] = 0.0
    pr[1:] = np.cumsum(p)
    qr = np.empty(n + 1, dtype=float)
    qr[0] = 0.0
    qr[1:] = np.cumsum(q)

    A = pr[n:0:-1] + qr[1:]
    C1 = np.correlate(h, r, mode="full")[n - 1:n - 1 + n]
    B = np.convolve(r, h, mode="full")[:n]
    g = dx * (A - C1 - B)

    if not np.all(np.isfinite(g)):
        g = np.nan_to_num(g, nan=0.0, posinf=1e6, neginf=-1e6)
    return float(f), g


def slsqp_full_loss(z, n, alpha):

def full_sum_constraint(z, n):
    return float(np.sum(z) - 0.5 * float(n))

def slsqp_full_step(z, n, alpha, maxiter):
    z = project_full(np.asarray(z, dtype=float).copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [
        {
            "type": "eq",
            "fun": full_sum_constraint,
            "jac": full_sum_jacobian,
            "args": (n,),
        }
    ]
    try:
        res = minimize(
            slsqp_full_loss,
            z,
            args=(n, alpha),
            method="SLSQP",
            jac=True,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-13, "iprint": 0},
        )
        x = getattr(res, "x", None)
        if x is None or not np.all(np.isfinite(np.asarray(x, dtype=float))):
            return z
        x = np.asarray(x, dtype=float).copy()
        return project_full(x, n)
    except Exception:
        return z

                best_h = h2.copy()
                best_v = None

    if time.time() < deadline - 150.0:
        n0 = int(best_n)
        if best_v is not None and len(best_v) == num_v(n0):
            v0 = project_v(best_v, n0)
        else:
            v0 = v_from_h(best_h, n0)

        alphas_high = [30.0, 200.0, 1000.0, 5000.0, 30000.0]
        maxit_high = 140 if num_v(n0) > 140 else 180
        pair_high = [0.02, 0.008, 0.003, 0.0012, 0.0005, 0.0002]
        v1, c1 = optimize_v(
            v0,
            n0,
            deadline,
            alphas_high,
            maxit_high,
            pair_high,
            use_lbfgs=(num_v(n0) > 150),
        )
        if c1 < best_c:
            best_c = c1
            best_v = v1.copy()
            best_h = make_sym(n0, best_v).copy()
            best_n = n0

    desired_ns = [
        max(60, n_init - 8),
        max(60, n_init - 4),
        n_init + 1,
        n_init - 1,
        n_init + 4,
        n_init + 8,
        240,
        300,
        200,
        350,
    ]
    ns = []
    seen = set()
    for x in desired_ns:
        x = int(x)
        if 60 <= x <= 550 and x not in seen:
            seen.add(x)
            ns.append(x)

    if best_v is not None:
        source_h = make_sym(best_n, project_v(best_v, best_n))
    else:
        source_h = best_h

    for n_try in ns:
        rem = deadline - time.time()
        if rem < 120.0:
            break
        if n_try == best_n:
            continue
        alloc = min(170.0, rem * 0.28)
        if alloc < 55.0:
            break
        h_new = interpolate_h(source_h, best_n, n_try)
        v_new = v_from_h(h_new, n_try)
        alphas_med = [50.0, 600.0, 5000.0, 30000.0]
        maxit_med = 90 if num_v(n_try) > 160 else 120
        pair_med = [0.02, 0.008, 0.003, 0.0012]
        v2, c2 = optimize_v(
            v_new,
            n_try,
            time.time() + alloc,
            alphas_med,
            maxit_med,
            pair_med,
            use_lbfgs=(num_v(n_try) > 160),
        )
        if c2 < best_c:
            best_c = c2
            best_v = v2.copy()
            best_h = make_sym(n_try, best_v).copy()
            best_n = n_try
            source_h = best_h

    rcount = 0
    while time.time() < deadline - 85.0:
        rem = deadline - time.time()
        alloc = min(80.0, rem * 0.35)
        if alloc < 32.0:
            break

        rcount += 1
        n_use = int(best_n)

        if best_v is not None and rcount % 3 == 0:
            base = project_v(best_v, n_use)
            if rcount % 6 == 0:
                h_base = make_sym(n_use, base)
                h_bin = (h_base > 0.5).astype(float)
                v_new = v_from_h(h_bin, n_use)
            else:
                noise_scale = 0.035 + 0.045 * float((rcount % 10) + 1) / 10.0
                v_new = project_v(base + noise_scale * rng.randn(base.size), n_use)
        elif rcount % 3 == 1:
            v_new = make_noise_seed(n_use, 0.08, rng)
        else:
            if rcount % 2 == 0:
                v_new = make_fourier_seed(n_use, rng, d=12, scale=0.28)
            else:
                v_new = make_block_seed(n_use, rng, blocks=5)

        alphas_restart = [100.0, 1800.0, 18000.0]
        maxit_restart = 80 if num_v(n_use) > 160 else 110
        pair_restart = [0.01, 0.004, 0.0015, 0.0006]
        v2, c2 = optimize_v(
            v_new,
            n_use,
            time.time() + alloc,
            alphas_restart,
            maxit_restart,
            pair_restart,
            use_lbfgs=False,
        )
        if c2 < best_c:
            best_c = c2
            best_v = v2.copy()
            best_h = make_sym(n_use, best_v).copy()
            best_n = n_use

    if best_v is not None and time.time() < deadline - 4.0:
        n_f = int(best_n)
        v_f = project_v(best_v, n_f)

        if time.time() < deadline - 8.0:
            v2 = slsqp_sym_step(v_f, n_f, 100000.0, 160)
            c2 = eval_v(v2, n_f)
            if c2 < best_c:
                best_c = c2
                v_f = project_v(v2, n_f)

        if time.time() < deadline - 2.0:
            v3, c3 = pair_search_sym(
                v_f,
                n_f,
                deadline - 1.0,
                [0.008, 0.003, 0.0012, 0.0005, 0.0002],
            )
            if c3 < best_c:
                best_c = c3
                v_f = project_v(v3, n_f)

        best_v = project_v(v_f, n_f)
        best_h = make_sym(n_f, best_v).copy()
        best_n = n_f
        best_c = eval_v(best_v, best_n)

    if best_h is None or best_n <= 0:
        best_n = 250
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c = compute_c5(best_h, best_n)

    best_h = project_full(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
