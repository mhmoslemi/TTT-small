    """
    Find optimal d such that h[p]+direction*d, h[q]-direction*d minimizes max(lags).
    direction is +1 or -1.
    Returns (best_d, best_max, new_lags).
    """
    # Compute coefficients for this pair
    v0_p = h[p]
    v0_q = h[q]
    
    # For m ≠ n-1: lags[m] changes by direction*d * (coeff_p[m] - coeff_q[m])
    # where coeff_p[m] is the coefficient of h[p] in lags[m] (for m≠n-1)
    # and coeff_q[m] similarly for h[q]

    # coeff_p[m]: h[p] appears in lags[m] with coefficient...

    # As second factor: -h[r1] where r1 = p-m+(n-1) if valid

    diff_coeff = np.zeros(2*n-1)

        diff_coeff[mlo:mhi+1][valid] += (1 - h[q_vals[valid]])
    # Second term: i = p-m+(n-1), valid if 0≤i<n → m in [-(n-1-p), p]
    mlo2 = max(0, p-(n-1))

        valid = (i_vals >= 0) & (i_vals < n)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jacobian(x, n):
    return np.ones_like(x)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    lags = np.correlate(x, 1 - x, mode='full') * dx
    m = np.max(lags)
    s = alpha * (lags - m)
    mx = np.max(s)
    val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return float(val)

def slsqp_optimize(h, n, alpha, deadline, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
