        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        
        grad = smooth_max_grad_full(h, n, alpha)
        # Project gradient to satisfy sum constraint: subtract mean
        grad -= grad.mean()
        # Take a small step
        step = 0.01
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)

            h_new = h - step * grad

            if compute_c5(h_new, n) < c:
                h = h_new
            else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def corr_scaled(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = corr_scaled(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        e = np.exp(shifted - mx)
        return float(m + (mx + np.log(np.sum(e))) / alpha)
    e = np.exp(shifted)
    return float(m + np.log(np.sum(e)) / alpha)

def smooth_max_grad(x, n, alpha):
    c = corr_scaled(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        e = np.exp(shifted - mx)
    else:
        e = np.exp(shifted)
    w = e / e.sum()  # length 2n-1, sums to 1
    
    # dF/dh[j] = (2/n) * (1 - term2[j] - term3[j])
    # term2[j] = sum_k w[k] * h[j+k-(n-1)] (0 if index out of [0,n-1])
    # term3[j] = sum_k w[k] * h[j-k+(n-1)] (0 if index out of [0,n-1])
    
    # Use FFT for efficiency
    # term2[j] = sum_k w[k] * h[j+k-(n-1)]
    # Let m = j+k-(n-1), k = m-j+(n-1)
    # term2[j] = sum_m w[m-j+(n-1)] * h[m] where m in [0,n-1] and m-j+(n-1) in [0,2n-2]
    # This is a cross-correlation of w (length 2n-1) with h (length n)
    
    # Use FFT: zero-pad both to length >= 3n-1
    Nfft = 1
    while Nfft < 4 * n:
        Nfft *= 2
    W = np.fft.rfft(w, Nfft)
    H = np.fft.rfft(h, Nfft)
    
    # term2[j] = sum_k w[k]*h[j+k-n+1] = (w circularly cross-correlated with h)
    # Let's compute: for each j in 0..n-1
    # term2[j] = sum_{k=0}^{2n-2} w[k] * h[j+k-(n-1)]
    # Define h_pad: h_pad[m] = h[m] for m in [0,n-1], else 0
    # term2[j] = sum_k w[k] * h_pad[j+k-(n-1)]
    # Let t = j-(n-1): term2[j] = sum_k w[k] * h_pad[t+k]
    # This is (w * h_pad)(t) where * denotes cross-correlation... 
    # Actually it's a convolution-like sum: sum_k w[k]*h_pad[k+t]
    # = (w flipped * h_pad)(t) for convolution
    # Let w_flip[k'] = w[2n-2-k']
    # sum_k w[k]*h_pad[k+t] = sum_k' w[2n-2-k']*h_pad[2n-2-k'+t]
    # Hmm, let me just do it directly for n<=500 since it's O(n^2) = 250K ops
    
    term2 = np.zeros(n)
    term3 = np.zeros(n)
    for j in range(n):
        # term2[j]: k from 0 to 2n-2, index into h: j+k-(n-1), valid if 0<=j+k-(n-1)<n
        idx = j - (n - 1)
        k_start = max(0, -(j - (n-1)))  # so that j+k-(n-1) >= 0 => k >= n-1-j
        k_end = min(2*n-2, n-1-j+n-1)   # so that j+k-(n-1) <= n-1 => k <= 2n-2-j
        k_start = max(0, n-1-j)
        k_end = min(2*n-2, 2*n-2-j)
        if k_start <= k_end:
            h_idx = j + np.arange(k_start, k_end+1) - (n-1)
            term2[j] = np.dot(w[k_start:k_end+1], h[h_idx])
        
        # term3[j]: k from 0 to 2n-2, index into h: j-k+(n-1), valid if 0<=j-k+(n-1)<n
        # j-k+(n-1) >= 0 => k <= j+n-1
        # j-k+(n-1) <= n-1 => k >= j
        k_start3 = max(0, j)
        k_end3 = min(2*n-2, j+n-1)
        if k_start3 <= k_end3:
            h_idx3 = j - np.arange(k_start3, k_end3+1) + (n-1)
            term3[j] = np.dot(w[k_start3:k_end3+1], h[h_idx3])
    
    dx = 2.0 / n
    grad = dx * (1.0 - term2 - term3)
    return grad

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha),
                       jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < smooth_max_obj(h, n, alpha):
            return project_feasible(res.x, n)
        return h
    except Exception:
        return h

def gradient_descent_polish(h, n, deadline, max_iter=20000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 5000000
    step_init = 0.05
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break
        
        grad = smooth_max_grad(h, n, alpha)
        # Project gradient to null space of sum constraint
        grad = grad - grad.mean()
        
        # Adaptive step
        step = step_init / (1.0 + it * 0.001)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < c - 1e-16:
            h = h_new
            step_init = min(step * 1.2, 0.1)
        else:
            step_init = max(step * 0.5, 1e-8)
            if step_init < 1e-9:
                break
            # Try smaller step
            h_new2 = h - step_init * grad
            h_new2 = project_feasible(h_new2, n)
            if compute_c5(h_new2, n) < c - 1e-16:
                h = h_new2
            else:
                no_improve += 1
                
    return best_h, best_c

def coordinate_descent_polish(h, n, deadline, max_iter=100000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 20000000
    idx = 0
