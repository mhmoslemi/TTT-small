    except Exception:
        h0 = np.full(200, 0.5)
        n0 = 200

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: SLSQP optimization at n0 with increasing alpha (250s)
    t1_end = t0 + 250
    alphas_p1 = [300, 1000, 5000, 20000, 100000, 500000, 2000000]
    for a in alphas_p1:
        if time.time() > t1_end - 25:
            break
        ht = slsqp_opt(best_h, n0, a, 2500)
        c = compute_c5(ht, n0)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht
        # Quick local refinement
        if time.time() < t1_end - 15:
            best_h, best_c = local_search(best_h, n0, time.time() + 12,
                                          [0.01, 0.005, 0.002], max_passes=30)

    # Phase 2: Extended local search at n0 (120s)
    t2_end = t0 + 370
    if n0 <= 300:
        best_h, best_c = local_search(best_h, n0, t2_end,
                                      [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005],
                                      max_passes=60)
    else:
        best_h, best_c = near_local_search(best_h, n0, t2_end,
                                           [0.02, 0.01, 0.005, 0.002, 0.001],
                                           radius=50, max_passes=40)

    # Phase 3: SLSQP with very high alpha (100s)
    t3_end = t0 + 470
    for a in [500000, 2000000, 10000000, 50000000]:
        if time.time() > t3_end - 10:
            break
        ht = slsqp_opt(best_h, n0, a, 3000)
        c = compute_c5(ht, n0)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht

    # Phase 4: Try alternative resolutions (250s)
    t4_end = t0 + 720
    n_best = len(best_h)
    for n_try in [200, 250, 300, 400]:
        if time.time() > t4_end - 60:
            break
        if n_try == n_best:
            continue
        alloc = min(60, t4_end - time.time() - 50)
        if alloc < 25:
            break
        h_interp = interp_to_n(best_h, n_best, n_try)
        h_interp = project_feasible(h_interp, n_try)
        alphas_p4 = [500, 5000, 50000, 500000, 5000000]
        ht, c_try = optimize_at_n(h_interp, n_try, time.time() + alloc - 15, alphas_p4, 1500)
        if time.time() < t4_end - 45:
            if n_try <= 300:
                ht, c_try = local_search(ht, n_try, time.time() + 15,
                                         [0.01, 0.005, 0.001], max_passes=30)
            else:
                ht, c_try = near_local_search(ht, n_try, time.time() + 15,
                                              [0.01, 0.005, 0.001], radius=40, max_passes=20)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = ht
            n_best = n_try

    # Phase 5: Restart with perturbation (80s)
    t5_end = t0 + 800
    if time.time() < t5_end - 30:
        n_best = len(best_h)
        for scale in [0.05, 0.10, 0.15]:
            if time.time() > t5_end - 20:
                break
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(40, t5_end - time.time() - 15)
            if alloc < 15:
                break
            ht, c_try = optimize_at_n(hp, n_best, time.time() + alloc,
                                      [20000, 200000, 2000000], 1500)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = ht

    # Phase 6: Final SLSQP + local search (remaining)
    n_best = len(best_h)
    t6 = time.time()
    if t6 < deadline - 30:
        ht = slsqp_opt(best_h, n_best, 100000000, 4000)
        c = compute_c5(ht, n_best)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht

    # Phase 7: Final local search (last 20-30s)
    if n_best <= 250:
        best_h, best_c = local_search(best_h, n_best, deadline - 1,
                                      [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                      max_passes=80)
    else:
        best_h, best_c = near_local_search(best_h, n_best, deadline - 1,
                                           [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                           radius=50, max_passes=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
