import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h.astype(np.float64), (1.0 - h).astype(np.float64), mode='full')
    return float(np.max(c)) * dx

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=np.float64)
    hb[idx[:n // 2]] = 1.0
    return hb

def compute_delta(h, i, j, n):
    """Compute correlation change when i: 0->1, j: 1->0."""
    n_corr = 2 * n - 1
    delta = np.zeros(n_corr)
    nm1 = n - 1
    # i: 0->1: +(1-h[i+k]) for valid i+k, -h[i-k] for valid i-k
    t1s = nm1 - i
    t1e = min(nm1 + nm1 - i, n_corr - 1)
    if t1s <= t1e:
        idx = np.arange(t1s, t1e + 1)
        delta[idx] += 1.0 - h[i + idx - nm1]
    t2s = i
    t2e = min(nm1 + i, n_corr - 1)
    if t2s <= t2e:
        idx = np.arange(t2s, t2e + 1)
        delta[idx] -= h[i - (idx - nm1)]
    # j: 1->0: -(1-h[j+k]) for valid j+k, +h[j-k] for valid j-k
    t3s = nm1 - j
    t3e = min(nm1 + nm1 - j, n_corr - 1)
    if t3s <= t3e:
        idx = np.arange(t3s, t3e + 1)
        delta[idx] -= 1.0 - h[j + idx - nm1]
    t4s = j
    t4e = min(nm1 + j, n_corr - 1)
    if t4s <= t4e:
        idx = np.arange(t4s, t4e + 1)
        delta[idx] += h[j - (idx - nm1)]
    return delta

def local_search_first_improve(h, n, deadline, rng, n_tries=30, max_sweeps=100000):
    """First-improvement local search with random move ordering."""
    dx = 2.0 / n
    h = to_binary(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c = float(corr.max()) * dx
    best_h = h.copy()
    best_c = cur_c
    n_corr = 2 * n - 1
    nm1 = n - 1

    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()

    sweeps = 0
    while time.time() < deadline and sweeps < max_sweeps:
        sweeps += 1
        improved = False
        perm = rng.permutation(len(zeros))
        for zi in perm:
            if time.time() > deadline:
                return best_h, best_c
            i = zeros[zi]
            found = False
            nt = min(len(ones), n_tries)
            if nt < len(ones):
                oj_arr = rng.choice(len(ones), size=nt, replace=False)
            else:
                oj_arr = np.arange(len(ones))
            rng.shuffle(oj_arr)
            for oj in oj_arr:
                j = ones[oj]
                delta = compute_delta(h, i, j, n)
                # Quick check: if delta at argmax(corr) is >= 0, can't improve
                t_max = np.argmax(corr)
                if delta[t_max] >= -1e-15:
                    continue
                nc = float(np.max(corr + delta)) * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += delta
                    cur_c = nc
                    zeros[zi] = j
                    ones[oj] = i
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    improved = True
                    found = True
                    break
        if not improved:
            break
    return best_h, best_c

def local_search_best_improve(h, n, deadline, rng, max_sweeps=500, n_tries=50):
    """Best-improvement local search (slower per sweep but more thorough)."""
    dx = 2.0 / n
    h = to_binary(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c = float(corr.max()) * dx
    best_h = h.copy()
    best_c = cur_c

    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()

    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        best_move = None
        best_nc = cur_c
        perm_z = rng.permutation(len(zeros))
        for zi in perm_z:
            if time.time() > deadline:
                break
            i = zeros[zi]
            nt = min(len(ones), n_tries)
            if nt < len(ones):

                delta = compute_delta(h, i, j, n)
                nc = float(np.max(corr + delta)) * dx

                    best_nc = nc
                    best_move = (zi, oj)
        if best_move is not None:
            zi, oj = best_move
            i = zeros[zi]
            j = ones[oj]

        else:

def perturb(h, n, n_swaps, rng):
    """Randomly perturb by swapping n_swaps pairs."""
    h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    for _ in range(n_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = rng.randint(len(ones))
        zi = rng.randint(len(zeros))
        i = zeros[zi]
        j = ones[oi]
        h[i], h[j] = 1.0, 0.0
        zeros[zi] = j
        ones[oi] = i
    return h

def ils(h, n, deadline, rng, n_restarts=50, base_perturb=5):
    """Iterated Local Search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c

    for r in range(n_restarts):
        if time.time() > deadline:
            break
        # Local search
        ls_deadline = min(deadline, time.time() + 30.0)
        cur_h, cur_c = local_search_first_improve(cur_h, n, ls_deadline, rng, n_tries=30)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
        # Perturb proportionally to how far from "perfect"
        perturb_size = max(2, base_perturb + r // 5)
        cur_h = perturb(best_h, n, perturb_size, rng)

    return best_h, best_c

def make_pattern(n, kind, rng):
    """Create various structured initializations."""
    h = np.zeros(n)
    half = n // 2
    if kind == 0:
        # Random
        pos = rng.choice(n, size=half, replace=False)
        h[pos] = 1.0
    elif kind == 1:

        alpha = 0.6180339887
        vals = np.floor(np.arange(n) * alpha)
        # Take first half
        h = (vals < alpha * (n // 2)).astype(np.float64)
        # Adjust to have exactly half
        cnt = int(h.sum())
        if cnt > half:
            ones = np.where(h == 1)[0]
            remove = rng.choice(ones, size=cnt - half, replace=False)
            h[remove] = 0
        elif cnt < half:
            zeros = np.where(h == 0)[0]
            add = rng.choice(zeros, size=half - cnt, replace=False)
            h[add] = 1
    elif kind == 2:
        # Spread out (evenly spaced)
        pos = np.linspace(0, n - 1, half).astype(int)
        h[pos] = 1.0
    elif kind == 3:
        # Two-scale pattern
        block = 4
        for start in range(0, n, block):
            h[start] = 1.0
            h[start + 2] = 1.0 if start + 2 < n else 0.0
        cnt = int(h.sum())
        if cnt > half:
            ones = np.where(h == 1)[0]
            remove = rng.choice(ones, size=cnt - half, replace=False)
            h[remove] = 0
        elif cnt < half:
            zeros = np.where(h == 0)[0]
            add = rng.choice(zeros, size=half - cnt, replace=False)
            h[add] = 1
    else:
        pos = rng.choice(n, size=half, replace=False)
        h[pos] = 1.0
    return h
