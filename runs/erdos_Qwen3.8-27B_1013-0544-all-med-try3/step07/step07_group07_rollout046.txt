    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Bisection on shift
    lo, hi = -max(s-target, 0)*1.5 - 0.01, max(target-s, 0)*1.5 + 0.01
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = hp < 1.0
            if mask.any():
                hp[mask] += diff / mask.sum() * (1.0 - hp[mask]) / (1.0 - hp[mask]).sum() if (1.0-hp[mask]).sum()>0 else diff
        else:
            mask = hp > 0.0
            if mask.any():

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_continuous(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return hp

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def compute_delta_swap(h, i, j, n):
    """Compute delta in correlation when h[i]:0->1 and h[j]:1->0"""
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def tabu_local_search(hb, n, deadline, tabu_len=150, max_iters=50000):
    """Tabu search for binary solution"""
    dx = 2.0 / n
    h = hb.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    tabu = {}  # (i,j) -> expiry_iter
    it = 0
    stagnation = 0
    while it < max_iters:
        if time.time() > deadline:
            break
        best_delta = None
        best_nc = cur_c
        best_pair = None
        # Sample a subset of pairs for efficiency
        n_samp = min(len(zeros_arr), 50)
        samp_idx = np.random.choice(len(zeros_arr), n_samp, replace=False)
        for si in samp_idx:
            i = zeros_arr[si]
            for j in ones_arr:
                key = (i, j)
                if key in tabu and tabu[key] > it:
                    # Aspiration: allow if better than best
                    pass
                delta = compute_delta_swap(h, i, j, n)
                nc = (corr + delta).max() * dx
                is_tabu = key in tabu and tabu[key] > it
                if not is_tabu and nc < best_nc:
                    best_nc = nc
                    best_delta = delta
                    best_pair = (i, j)
                elif is_tabu and nc < best_c:
                    # Aspiration criterion
                    best_nc = nc
                    best_delta = delta
                    best_pair = (i, j)
        if best_pair is None:
            stagnation += 1
            it += 1
            if stagnation > 20:
                break
            continue
        stagnation = 0
        i, j = best_pair
        h[i], h[j] = 1.0, 0.0
        corr += best_delta
        cur_c = best_nc
        tabu[(i, j)] = it + tabu_len
        tabu[(int(np.where(ones_arr == j)[0][0]), int(np.where(zeros_arr == i)[0][0]))] = it + tabu_len
        # Update arrays
        zi = int(np.where(zeros_arr == i)[0][0])
        oi = int(np.where(ones_arr == j)[0][0])
        zeros_arr[zi] = j
        ones_arr[oi] = i
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        it += 1
    return best_h, best_c

def anneal_binary(hb, n, deadline, seed=42, max_iters=3000000, T_init=0.04, cool=0.999995):
    """Simulated annealing for binary solution with better schedule"""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = hb.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = T_init
    iter_no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = compute_delta_swap(h, i, j, n)
        nc = (corr + delta).max() * dx
        diff = nc - cur_c
        if diff <= 0 or (T > 1e-12 and rng.random() < np.exp(-diff / (T * dx * 0.4))):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                iter_no_improve = 0
            else:
                iter_no_improve += 1
        else:
            iter_no_improve += 1
        T *= cool
        # Reheat if stuck too long
        if iter_no_improve > 200000 and T < T_init * 0.1:
            T = T_init * 0.5
            iter_no_improve = 0
    return best_h, best_c

def pgd_solve(h, n, alphas, deadline, max_iter=5000):
    """Projected gradient descent with soft-max-smoothed objective"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        cur_c = best_c
        step = min(float(n) / (4.0 * alpha), 0.5)
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            c = np.correlate(h, 1 - h, mode='full')
            a = alpha * c
            sm = a.max()
            exp_c = np.exp(a - sm)
            p = exp_c / exp_c.sum()
            grad = np.zeros(n)
            for k in range(-(n - 1), n):
                pk = p[k + n - 1]
                if pk < 1e-10:
                    continue
                i_min = max(0, k)
                i_max = min(n - 1, n - 1 + k)
                if i_min <= i_max:
                    grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
                i_min2 = max(0, -k)
                i_max2 = min(n - 1, n - 1 - k)
                if i_min2 <= i_max2:
                    grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            direction = -grad / gnorm
            accepted = False
            for ls in range(30):
                h_new = project_continuous(h + step * (0.5 ** ls) * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-14:
                    h = h_new
                    cur_c = c_new
                    step = step * (0.5 ** ls)
                    accepted = True
                    break
            if accepted:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
