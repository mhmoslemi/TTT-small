    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_shifts(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    return w  # weights for each shift

def grad_c5(h, n, alpha):
    c = compute_all_shifts(h, n)
    w = smooth_max_grad(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    # c[m] = sum_i h[i]*(1-h[i+k]) * dx where k = m - (n-1)
    # ∂c[m]/∂h[j] = dx * (1-h[j+k]) - dx * h[j-k] (with bounds)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_shifts(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha):
    c = compute_all_shifts(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(h, n, alpha, mu=0.0):
    c = compute_all_shifts(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n/2.0)**2
    return float(val)

def smooth_max_grad(h, n, alpha, mu=0.0):
    g = grad_c5(h, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (h.sum() - n/2.0)
    return g

def coordinate_descent_pass(h, n, max_sweeps=10):
    """Coordinate descent: for each i, find optimal h[i]."""
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            c = compute_all_shifts(h, n)
            dx = 2.0 / n
            
            # Compute g = ∂c/∂h[i]
            g = np.zeros(2*n - 1)
            for m_idx in range(2*n - 1):
                k = m_idx - (n - 1)
                if k >= 0:
                    if i < n - k:
                        g[m_idx] += dx * (1 - h[i + k])
                    if i >= k:
                        g[m_idx] += dx * (-h[i - k])
                else:
                    kk = -k
                    if i >= kk:
                        g[m_idx] += dx * (1 - h[i - kk])
                    if i <= n - 1 - kk:
                        g[m_idx] += dx * (-h[i + kk])
            
            nonzero = np.abs(g) > 1e-15
            if not nonzero.any():
                continue
            
            gn = g[nonzero]
            cn = c[nonzero]
            b = cn - h[i] * gn  # intercepts: f(t) = max_m gn[m]*t + b[m]
            
            # Bisection on subgradient
            lo_t, hi_t = 0.0, 1.0
            for _ in range(40):
                mid = (lo_t + hi_t) / 2.0
                vals = gn * mid + b
                mv = vals.max()
                active = vals >= mv - 1e-12
                slopes = gn[active]
                if len(slopes) == 0:
                    break
                if slopes.min() > 1e-12:
                    hi_t = mid
                elif slopes.max() < -1e-12:
                    lo_t = mid
                else:

            t_new = (lo_t + hi_t) / 2.0
            t_new = np.clip(t_new, 0.0, 1.0)
            
            if abs(t_new - h[i]) > 1e-10:

        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
        if not improved:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_shifts(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_coord_grad(h, n, i):
    dx = 2.0 / n
    g = np.zeros(2 * n - 1)
    center = n - 1
    max_k = n - 1 - i
    if max_k >= 0:
        ks = np.arange(max_k + 1)
        g[center + ks] += dx * (1.0 - h[i + ks])

    if max_k2 >= 0:
        ks = np.arange(max_k2 + 1)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_all_shifts(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_all_shifts(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):

            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    return x[k:n]
