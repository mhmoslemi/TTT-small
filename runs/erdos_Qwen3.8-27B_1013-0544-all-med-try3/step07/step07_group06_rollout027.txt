**Key insight**: Instead of trying all O(n²) pairs, we can:
1. Compute the correlation vector corr
2. Find the top few lags with highest correlation
3. For each such lag k, identify which positions contribute most to corr[k]
4. Try swaps that specifically reduce those contributions

For a binary solution, corr[k] = number of i where h[i]=1 and h[i+k]=0 (for positive k).
To reduce corr[k], we want to find i where h[i]=1 and h[i+k]=0, and change h[i] to 0 (by swapping with some j where h[j]=0).

The "cost" of also setting h[j]=1 is the increase in correlation at other lags. So we want j that minimizes the increase.

This is a promising approach but complex to implement efficiently. Let me just go with the simpler approach of:
- Random swaps with acceptance (simulated annealing style)
- Plus greedy swaps (try n random pairs per step, accept first improvement)

        delta = np.zeros(n1)
        
        im = i + mi
        msk = (im >= 0) & (im < n)
        delta[msk] += (1.0 - h[im[msk]])
        
        jm = j + mi
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= (1.0 - h[jm[msk]])
        
        imn = i - mi
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        
        jmn = j - mi
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        
        # Cross corrections
        m1 = j - i
        delta[n - 1 + m1] += 1.0
        m2 = i - j
        delta[n - 1 + m2] += 1.0
        
        new_corr = corr + delta
        new_max = new_corr.max()
        
        if new_max < cur_max - 1e-12:
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_max = new_max
            zero_list.remove(i)
            one_list.append(i)
            one_list.remove(j)
            zero_list.append(j)
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100000 and iters - best_stall_reset > 200000:
                # Perturb and reset
                n_pert = rng.randint(3, 10)
                for _ in range(n_pert):
                    a = zero_list[rng.randint(len(zero_list))]
                    b = one_list[rng.randint(len(one_list))]
                    h[a] = 1.0
                    h[b] = 0.0
                    zero_list.remove(a)
                    one_list.append(a)
                    one_list.remove(b)
                    zero_list.append(b)
                corr = np.correlate(h, 1 - h, mode='full')
                cur_max = corr.max()
                stall = 0
                best_stall_reset = iters
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n_best = n0
    n = n0
    h = best_h.copy()
    
    # Phase 1: SLSQP with analytical gradient (0-15%)
    p1_end = t0 + 0.15 * (budget_s - 4)
    alphas = [200, 800, 3000, 10000, 40000, 150000]
    for alpha in alphas:
        if time.time() > p1_end:
            break
        remaining = p1_end - time.time()
        if remaining < 5:
            break
        h = slsqp_run(h, n, alpha, 600, p1_end)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            n_best = n
    
    h = best_h.copy()
    
    # Phase 2: Binary local search (15-55%)
    p2_end = t0 + 0.55 * (budget_s - 4)
    hb, cb = fast_binary_search(h, n, p2_end, seed=seed, max_iters=30000000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        n_best = len(best_h)
    
    # Phase 3: Multi-resolution (55-78%)
    p3_end = t0 + 0.78 * (budget_s - 4)
    for nh in [400, 600, 1000]:
        if time.time() > p3_end - 20:
            break
        sub_budget = min(p3_end - 5, time.time() + (p3_end - time.time()) * 0.45)
        
        xo = np.linspace(0, 2, n_best)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        
        # SLSQP on new grid (only if n is manageable)
        if nh <= 600 and time.time() < sub_budget - 30:
            for alpha in [3000, 20000, 100000]:
                if time.time() > sub_budget - 30:
                    break
                hh = slsqp_run(hh, nh, alpha, 400, sub_budget - 25)
                ch = compute_c5(hh, nh)
                if ch < best_c:
                    best_c = ch
                    best_h = hh.copy()
                    n_best = nh
        
        # Binary search on new grid
        p3b_end = min(p3_end - 5, time.time() + 80)
        hb2, cb2 = fast_binary_search(hh, nh, p3b_end, seed=seed + 500 + nh, max_iters=5000000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n_best = nh
    
    # Phase 4: Extended binary search on best (78-95%)
    p4_end = deadline - 15
    if time.time() < p4_end:
        hb, cb = fast_binary_search(best_h, n_best, p4_end, seed=seed + 1000, max_iters=50000000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            n_best = len(best_h)
    
    # Phase 5: Final SLSQP polish (95-99%)
    if time.time() < deadline - 3 and n_best <= 500:
        h_final = slsqp_run(best_h, n_best, 500000, 400, deadline - 2)
        cf = compute_c5(h_final, n_best)
        if cf < best_c:
            best_c = cf
            best_h = h_final.copy()
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
