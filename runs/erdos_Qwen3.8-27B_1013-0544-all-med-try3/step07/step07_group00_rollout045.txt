import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def active_shifts(c_vec, nl, threshold_frac=0.001):
    mx = c_vec.max()
    thr = mx - max(1e-10, threshold_frac * mx)
    act = np.where(c_vec >= thr)[0]
    if len(act) > nl:
        order = np.argsort(-c_vec[act])
        act = act[order[:nl]]
    return list(act)

def smooth_obj(x, n, alpha, act):
    c = compute_c5_vec(x, n)
    vals = c[act]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += 0.005 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, act):
    c = compute_c5_vec(x, n)
    vals = c[act]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    na = n - 1
    for wi, idx in zip(w, act):
        if wi < 1e-15:
            continue
        k = idx - na
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 0.01 * (x.sum() - n / 2.0)
    return grad

def lbfgs_round(h, n, alpha, act, maxiter, deadline, ftol=1e-16, gtol=1e-12):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad,
                       args=(n, alpha, act), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': gtol, 'maxfun': maxiter * 10, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

def refine_pair_moves(h, n, deadline, steps=None, radius=40):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]
    for d in steps:

            break
        improved = True
        while improved:

            for i in range(n):

                        h[j] -= d

def coord_descent(h, n, deadline, sweeps=10):
    best_c = compute_c5(h, n)
    for s in range(sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            old = h[i]
            c_vec = compute_c5_vec(h, n)
            best_val = c_vec.max()
            for trial in [old - 0.05, old - 0.01, old - 0.001, old + 0.001, old + 0.01, old + 0.05]:
                if trial < 0 or trial > 1:
                    continue
                h[i] = trial
                adjust = best_c * n / 2 - h.sum()
                if abs(adjust) > 0.5:
                    h[i] = old

                nv = compute_c5(h, n)
                if nv < best_val - 1e-15:
                    best_val = nv
                    best_c = nv
                else:
                    h[i] = old
                    break
            h[i] = old
        h = project_feasible(h, n)
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)

    best_c = compute_c5(h, n)
    step = step_init
    na = n - 1

    cur_h, cur_c = refine_pair_moves(cur_h, cur_n, min(t7 + 20, deadline - 5),
        steps=[0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05], radius=25)

    # Phase 8: Try n=2000 if time allows (~60s)
    if time.time() < deadline - 60:
        n2 = 2000
        h2k = upsample(cur_h, cur_n, n2)
        c2k = compute_c5(h2k, n2)
        t8 = min(t0 + budget_s - 20, deadline)
        if c2k < cur_c + 0.005:
            h2k, c2k = focused_lbfgs(h2k, n2, min(t8 - 30, deadline - 10),
                alphas=[5000, 50000], top_k=8, inner_iters=3)
            h2k, c2k = refine_pair_moves(h2k, n2, deadline - 3,
                steps=[0.001, 0.0005, 0.0002, 0.0001], radius=20)
            if c2k < cur_c - 1e-12:
                cur_h = h2k
                cur_c = c2k
                cur_n = n2

    # Final validation
    cur_h = project_feasible(cur_h, cur_n)
    cur_c = compute_c5(cur_h, cur_n)

    return cur_h, cur_c, cur_n
