import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c_scaled = c_raw * dx
    shifted = alpha * c_scaled
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    sum_exp = exp_sh.sum()
    f = (max_sh + np.log(sum_exp)) / alpha
    p = exp_sh / sum_exp
    u = 1.0 - h
    grad = np.zeros(n)
    for i in range(n):
        lo = max(0, n - 1 - i)
        hi = min(2 * n - 2, n - 1 + i)
        if lo <= hi:
            js = i + np.arange(lo, hi + 1) - (n - 1)
            grad[i] += np.dot(p[lo:hi + 1], u[js])
        lo2 = max(0, i)
        hi2 = min(2 * n - 2, n - 1 + i)
        if lo2 <= hi2:
            js2 = (n - 1) + i - np.arange(lo2, hi2 + 1)
            grad[i] -= np.dot(p[lo2:hi2 + 1], h[js2])
    grad *= dx
    return f, grad

def lbfgs_smooth(h_init, n, alpha, deadline, maxiter=3000):
    def obj(x):
        f, _ = smooth_max_grad(x, alpha)
        return f
    def grad(x):
        _, g = smooth_max_grad(x, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxls': 50, 'maxfun': 200000})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

    dx = 2.0 / n

    best_h = h.copy()

    m_range = np.arange(-(n - 1), n).astype(np.int64)

    for restart in range(n_restarts):

                zeros_idx = np.where(h == 0)[0]
                if len(ones_idx) > 0 and len(zeros_idx) > 0:
                    a = rng.choice(ones_idx)
                    b = rng.choice(zeros_idx)
                    h[a], h[b] = 0.0, 1.0
            current_h = h.copy()

        h = current_h.copy()

        cur_c = float(corr.max() * dx)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        zeros_indices = np.where(h == 0)[0]
        no = len(ones_indices)
        nz = len(zeros_indices)

            if time.time() > deadline:
                break
            improved = False
            global_best_delta = np.inf
            global_pair = None

                ps = p + m_range
                valid1 = (ps >= 0) & (ps < n)
                term1 = np.zeros(two_n_1)
                term1[valid1] = 1.0 - h[ps[valid1]]
                ps2 = p - m_range
                valid2 = (ps2 >= 0) & (ps2 < n)
                term2 = np.zeros(two_n_1)
                term2[valid2] = h[ps2[valid2]]
                delta_p = -term1 + term2
                base_p = corr + delta_p

                qs = zeros_indices[:, None] + m_range[None, :]
                valid_q1 = (qs >= 0) & (qs < n)
                term1_q = np.zeros((nz, two_n_1))
                r = qs[valid_q1]
                term1_q[valid_q1] = 1.0 - h[r]

                qs2 = zeros_indices[:, None] - m_range[None, :]
                valid_q2 = (qs2 >= 0) & (qs2 < n)
                term2_q = np.zeros((nz, two_n_1))
                r2 = qs2[valid_q2]
                term2_q[valid_q2] = h[r2]

                delta_q = (term1_q - term2_q).T
                candidates = base_p[None, :] + delta_q

                k1_arr = n - 1 + zeros_indices - p
                valid_k1 = (k1_arr >= 0) & (k1_arr < two_n_1)
                for iq in range(nz):
                    if valid_k1[iq]:
                        candidates[iq, k1_arr[iq]] += 2.0
                k2_arr = n - 1 + p - zeros_indices
                valid_k2 = (k2_arr >= 0) & (k2_arr < two_n_1)
                for iq in range(nz):
                    if valid_k2[iq]:
                        candidates[iq, k2_arr[iq]] += 2.0

                cand_maxes = candidates.max(axis=1)
                min_idx = np.argmin(cand_maxes)
                if cand_maxes[min_idx] < global_best_delta:
                    global_best_delta = cand_maxes[min_idx]
                    global_pair = (p, zeros_indices[min_idx])
                    global_new_corr = candidates[min_idx].copy()

            if global_pair is not None and global_best_delta < cur_c / dx - 1e-14:
                p, q = global_pair
                h[p], h[q] = 0.0, 1.0
                corr = global_new_corr
                cur_c = global_best_delta * dx
                ones_indices = np.array([x for x in ones_indices if x != p] + [q])
                zeros_indices = zeros_indices[np.where(zeros_indices != q)[0]]
                no -= 1
                nz -= 1
                improved = True

            if not improved:
                break

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()

    return best_h, best_c
