    dx = 2.0 / n
    
    while time.time() < deadline:
        # Find the shift where max is achieved
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        # Convert full-correlation index to shift
        # full correlation index i corresponds to shift i-(n-1)
        k = max_idx - (n - 1)
        
        # Find the most contributing pairs at this shift
        if k >= 0:

            contributions = h[:n-k] * (1.0 - h[k:]) * dx

            k_abs = -k

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jacobian(x, n):
    return np.ones(n)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    stripped = alpha * (c - m)
    mx = stripped.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(stripped - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(stripped))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_local_search(h, n, deadline, deltas):
    """Focus moves on pairs contributing most to the peak overlap shift."""
    current_c = compute_c5(h, n)
    dx = 2.0 / n
    max_iter_no_improve = 0
    
    while time.time() < deadline and max_iter_no_improve < 5:
        max_iter_no_improve = 0
        for d in deltas:
            improved = False
            while time.time() < deadline:
                # Find peak shift
                corr_vec = np.correlate(h, 1.0 - h, mode='full') * dx
                max_idx = int(np.argmax(corr_vec))
                k = max_idx - (n - 1)
                
                # Compute contributions at this shift
                if k >= 0:
                    valid = np.arange(0, n - k)
                    contribs = h[valid] * (1.0 - h[valid + k]) * dx
                else:
                    ka = -k
                    valid = np.arange(ka, n)
                    contribs = h[valid] * (1.0 - h[valid - ka]) * dx
                
                # Get top contributing positions
                n_top = min(30, len(contribs))
                top_locals = np.argsort(-contribs)[:n_top]
                
                any_improve = False
                for li in top_locals:
                    if time.time() > deadline:
                        return h, current_c
                    if k >= 0:
                        i_src = int(valid[li])
                        i_dst = int(valid[li] + k)
                    else:
                        i_src = int(valid[li])
                        i_dst = int(valid[li] - ka)
                    
                    # Try moving mass from source (high h) to dest (low h)
                    if h[i_src] - d >= 0.0 and h[i_dst] + d <= 1.0:
                        h[i_src] -= d
                        h[i_dst] += d
                        c = compute_c5(h, n)
                        if c < current_c - 1e-14:
                            current_c = c
                            improved = True
                            any_improve = True
                        else:
                            h[i_src] += d
                            h[i_dst] -= d
                    
                    # Try moving mass from dest to source (if dest is high)
                    if h[i_dst] - d >= 0.0 and h[i_src] + d <= 1.0:
                        h[i_dst] -= d
                        h[i_src] += d
                        c = compute_c5(h, n)
                        if c < current_c - 1e-14:
                            current_c = c
                            improved = True
                            any_improve = True
                        else:
                            h[i_dst] += d
                            h[i_src] -= d

            if not improved:

            break

def full_pair_search(h, n, deadline, deltas, radius=None):
    """Standard pair search with optional radius limit."""
    current_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, current_c
                if radius is None:
                    j_lo, j_hi = i + 1, n
                else:
                    j_lo = max(0, i - radius)
                    j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < current_c - 1e-14:
                            current_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0 and not improved:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < current_c - 1e-14:
                            current_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, current_c

def sa_optimize(h, n, deadline, rng, n_steps=200000):
    """Simulated annealing with pair moves."""
    current_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = current_c
    
    t_start = 0.003
    t_end = 0.00001
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps)
        T = t_start * (t_end / t_start) ** frac
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.choice([-1, 1]) * rng.uniform(0.002, 0.04)
        
        new_i = h[i] + delta
        new_j = h[j] - delta
        if new_i < 0.0 or new_i > 1.0 or new_j < 0.0 or new_j > 1.0:
            continue
        
        h[i] = new_i
        h[j] = new_j
        new_c = compute_c5(h, n)
        
        if new_c < current_c:
            current_c = new_c
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = h.copy()
        else:
            de = new_c - current_c
            if de < 0.005 and rng.random() < np.exp(-de / max(T, 1e-12)):
                current_c = new_c
            else:
                h[i] = new_i - delta
                h[j] = new_j + delta
    
    return best_h, best_c
