def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

def top_k_smooth_max(c, k, alpha):
    """Smooth max of top k values using log-sum-exp"""
    # Sort and take top k
    idx = np.argsort(c)[-k:]

    m = c_top.max()
    if m > 500:
        val = m + (0 + np.log(np.sum(np.exp(c_top - m)))) / alpha
    else:
        val = np.log(np.sum(np.exp(c_top))) / alpha
    return val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha, top_k=None):
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_sel = c[idx]
    else:
        c_sel = c
    m = c_sel.max()
    if m * alpha > 500:
        shifted = c_sel - m
        return m + np.max(shifted) + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha
    else:
        return np.log(np.sum(np.exp(alpha * c_sel))) / alpha

def grad_smooth_max_correct(h, n, alpha, top_k=None):
    """Compute gradient of smooth max of top-k correlations (corrected version)."""
    c = compute_c5_all(h, n)
    total_shifts = 2 * n - 1
    if top_k is not None and top_k < total_shifts:
        idx = np.argsort(c)[-top_k:]
    else:
        idx = np.arange(total_shifts)
    
    c_sel = c[idx]
    m = c_sel.max()
    exponent = alpha * (c_sel - m)
    w = np.exp(exponent)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(idx)) / len(idx)
        w_sum = 1.0
    w /= w_sum
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for ii in range(len(idx)):
        wm = w[ii]
        if wm < 1e-14:
            continue
        m_idx = idx[ii]
        k = m_idx - (n - 1)
        weight = wm * dx
        
        if k >= 0:
            # For k >= 0: c = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
            # d/dh[j] = (1-h[j-k])*[k<=j<=n-1] - h[j+k]*[0<=j<=n-1-k]
            nv = n - k

            # j in [max(0,k), n-1-k]: contribution is (1-h[j-k]) - h[j+k]
            mid_lo = k
            mid_hi = nv  # = n-k
            if mid_lo < mid_hi:
                len_mid = mid_hi - mid_lo
                grad[mid_lo:mid_hi] += weight * ((1.0 - h[0:len_mid]) - h[k:k+len_mid])
            # j in [n-k, n-1]: contribution is (1-h[j-k])
            if k > 0:
                grad[n-k:n] += weight * (1.0 - h[0:k])
        else:
            kk = -k
            # For k < 0, kk = -k: c = sum_{n=kk}^{n-1} h[n-kk]*(1-h[n])
            # d/dh[j] = (1-h[j+kk])*[0<=j<=n-1-kk] - h[j-kk]*[kk<=j<=n-1]
            nv = n - kk
            if nv <= 0:
                continue
            # j in [0, kk-1]: contribution is (1-h[j+kk])
            if kk > 0:
                grad[0:kk] += weight * (1.0 - h[kk:n])
            # j in [kk, n-1-kk]: contribution is (1-h[j+kk]) - h[j-kk]
            mid_lo = kk
            mid_hi = nv
            if mid_lo < mid_hi:
                len_mid = mid_hi - mid_lo
                grad[mid_lo:mid_hi] += weight * ((1.0 - h[kk:kk+len_mid]) - h[0:len_mid])
            # j in [n-kk, n-1]: contribution is -h[j-kk]
            if kk > 0:
                grad[n-kk:n] += weight * (-h[0:kk])
    
    return grad

def _obj_pen(x, n, alpha, mu, top_k):
    c = compute_c5_all(x, n)
    m = c.max()
    if top_k is not None and top_k < 2*n-1:
        idx = np.argsort(c)[-top_k:]
        c_sel = c[idx]
    else:
        c_sel = c
    am = alpha * (c_sel - c_sel.max())
    val = c_sel.max() + np.log(np.sum(np.exp(am))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _obj_pen_grad(x, n, alpha, mu, top_k):
    g = grad_smooth_max_correct(x, n, alpha, top_k)
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

def lbfgs_optimize(h, n, deadline, alpha_start=100, top_k=20, max_time_per_config=60):
    """Optimize using L-BFGS-B with corrected gradient."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    configs = [
        (100, 0.0, 30),
        (500, 0.0, 30),
        (2000, 0.0, 30),
        (5000, 0.0, 20),
        (20000, 0.0, 20),
        (100, 100, 30),
        (500, 100, 30),
        (2000, 100, 20),
        (5000, 100, 20),

            break

        if t_sec < 5:
            break
        try:

                          options={'maxiter': maxiter, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxfun': maxiter * 10,
                                   'maxls': 40, 'disp': False,

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:

    return best_h, best_c

def projected_gradient_descent(h, n, deadline, alpha=500, top_k=20):
    """Projected gradient descent with backtracking line search."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.01
    
    for it in range(50000):
        remaining = deadline - time.time()
        if remaining < 2:
            break
        
        g = grad_smooth_max_correct(h, n, alpha, top_k)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g_dir = g / g_norm
        
        improved = False
        step = lr
        for _ in range(50):
            h_trial = project_feasible(h - step * g_dir, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                h = h_trial
                best_c = c_trial
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-15:
                break
        else:
            lr = min(lr * 1.1, 0.1)
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=80):
    """Local pair search with adaptive deltas."""
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_round = True
        while time.time() < deadline and improved_round:
            improved_round = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_round = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_round = True
                        else:
                            h[i] += d
                            h[j] -= d
    
    return h, best_c

                    if i == j:

                    move = step * (g[i] - g[j])
                    if abs(move) < 1e-10:

                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            rounds += 1
            if not improved:

    return h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max_correct(h, n, alpha, top_k=None):
    c = compute_c5_all(h, n)
    total_shifts = 2 * n - 1
    if top_k is not None and 0 < top_k < total_shifts:
        idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        idx = np.arange(total_shifts)
    c_sel = c[idx]
    m = c_sel.max()
    exponent = alpha * (c_sel - m)
    exponent = np.clip(exponent, -50, 50)
    w = np.exp(exponent)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(idx)) / len(idx)
