def compute_all_lags(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-15:

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

    """One outer approximation LP step."""
    dx = 2.0 / n

    hl = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hl) == n / 2.0, hl >= 0, hl <= 1,
            cp.norm_inf(hl - h) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = lag_grad(h, n, lag)  # gradient w.r.t. h (without dx)
        # c_k(h_new) ≈ c_k(h) + dx * g · (h_new - h)
        cons.append(c[k] + dx * (g @ (hl - h)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)

        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except:
            return h.copy()

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, m):
    grad = np.zeros(n)
    a = abs(m)
    if a == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0:
        grad[:n - a] -= h[a:]
        grad[a:] += (1.0 - h[:n - a])
    else:
        grad[:n - a] += (1.0 - h[a:])
        grad[a:] -= h[:n - a]
    return grad

def _sme(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 400:
        return m + (mx + np.log(s.size + np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sc(x, n):
    return float(x.sum() - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def run_slsqp(h, n, alpha, top_idx, maxiter, ftol):
    def f(x):
        return _sme(x, n, alpha, top_idx)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(f, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def outer_lp_step(h, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    hl = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hl) == n / 2.0, hl >= 0, hl <= 1,
            cp.norm_inf(hl - h) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = lag_grad(h, n, lag)
        cons.append(c[k] + dx * (g @ (hl - h)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except:
            return h.copy()
    if prob.status not in ('optimal', 'optimal_inaccurate') or tv.value is None:
        return h.copy()
    return project_feasible(np.array(hl.value), n)

def outer_approx(h, n, deadline, max_iter=35):
    hcur = h.copy()
    best_h = hcur.copy()
    best_c = compute_c5(hcur, n)
    trust = 0.25
    for _ in range(max_iter):
        if time.time() > deadline - 2:
            break
        hnew = outer_lp_step(hcur, n, trust)
        cn = compute_c5(hnew, n)
        if cn < best_c - 1e-14:
            best_c = cn
            best_h = hnew.copy()
        step = np.abs(hnew - hcur).max()
        if step < 1e-10:
            break
        hcur = hnew
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

def gradient_step_search(h, n, deadline, n_steps=3000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.02
    for s in range(n_steps):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = lag_grad(h, n, lag)
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.03, 0.06)
        else:
            step *= 0.75
            if step < 1e-8:
                break
    return best_h, best_c

def fine_local_search(h, n, deadline):

    best_c = compute_c5(h, n)

                    break
                i = int(np.argmax(np.abs(h - 0.5) * (1 - np.abs(h - 0.5)) * mid_mask.astype(float)))
                if h[i] == 0.5:
                    i = int(np.argmax((h * (1 - h)) * mid_mask.astype(float)))
                target = 0.0 if h[i] < 0.5 else 1.0
                delta = min(d * 2, abs(h[i] - target) * 0.5)
                if delta < 1e-9:
                    break
                sign = 1.0 if target > h[i] else -1.0
                h[i] += sign * delta
                best_j = -1
                best_cj = best_c
                candidates = np.where(np.abs(h - 0.5) < 0.45)[0]
                candidates = candidates[candidates != i]

                    cj = compute_c5(h_try, n)

                if best_j >= 0:

                            continue
                        h[i] -= delta
                        h[j] += delta
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-15:

                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                else:
                    if h[j] > h[i] + 1e-7:
                        delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.8, (1.0 - h[i]) * 0.8)
