    h = np.clip(x, 0, 1)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 500:
        val = m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    elif sm < -500:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    # Penalty for sum constraint
    val += penalty * (h.sum() - n / 2.0) ** 2
    return val

    # A = np.correlate(h, w, mode='full') has length n + 2n-1 - 1 = 3n-2
    # A[p] = sum_i h[i] * w[i+p-n] for valid i, i+p-n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(x, n, alpha, penalty):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 500:
        w = np.exp(s - sm)
        w_sum = w.sum()
        val = m + (sm + np.log(w_sum)) / alpha
    elif sm < -500:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        w_sum = 1.0
        val = m
    else:
        w = np.exp(s)
        w_sum = w.sum()
        val = m + np.log(w_sum) / alpha
    w = w / w_sum
    sum_diff = h.sum() - n / 2.0
    val += penalty * sum_diff * sum_diff
    d_vals = np.arange(-(n - 1), n)
    A = np.zeros(n)
    B = np.zeros(n)
    for j in range(n):
        p_lo = max(-(n - 1), -j)
        p_hi = min(n - 1, n - 1 - j)
        if p_lo <= p_hi:
            pr = np.arange(p_lo, p_hi + 1)
            A[j] = np.dot(w[pr + (n - 1)], h[j + pr])
        p_lo2 = max(-(n - 1), j - (n - 1))

    grad_corr = dx * (1.0 - A - B)
    grad_pen = 2.0 * penalty * sum_diff
    return val, grad_corr + grad_pen

        return x.sum() - target_sum

        return np.ones(n)

        m = c.max()

            return m
        return m + np.log(np.sum(np.exp(s))) / alpha
