            if len(j1) > 0:
                grad[j1] += pk * (1 - h[j1 + s])
            j2 = np.arange(s, n)
            if len(j2) > 0:
                grad[j2] += pk * (-h[j2 - s])
        else:
            s_abs = -s
            j1 = np.arange(s_abs, n)
            if len(j1) > 0:
                grad[j1] += pk * (1 - h[j1 - s_abs])
            j2 = np.arange(0, n - s_abs)
            if len(j2) > 0:
                grad[j2] += pk * (-h[j2 + s_abs])
    
    grad *= dx
    return obj, grad

def projected_grad_descent(h, n, deadline, alpha_start=100, alpha_end=50000, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / min(max_iter, 5000))
        obj, grad = compute_gradient(h, n, alpha)
        
        # Backtracking line search
        lr = 0.5
        for _ls in range(30):
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16 or c_new < obj * 0.99:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
            lr *= 0.5
        
        # Periodic check
        if it % 200 == 0:
            c_check = compute_c5(h, n)
            if c_check < best_c:
                best_c = c_check
                best_h = h.copy()
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=3000):
    def obj_grad(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sh_max = shifted.max()
        exp_s = np.exp(shifted - sh_max)
        p = exp_s / exp_s.sum()
        f = m + (sh_max + np.log(exp_s.sum())) / alpha
        
        g = np.zeros(n)
        shifts_arr = np.arange(2*n-1) - (n-1)
        for idx in range(2*n-1):
            pk = p[idx]
            if pk < 1e-14:
                continue
            s = shifts_arr[idx]
            if s >= 0:
                j1 = np.arange(0, n - s)
                if len(j1) > 0:
                    g[j1] += pk * (1 - x[j1 + s])
                j2 = np.arange(s, n)
                if len(j2) > 0:
                    g[j2] += pk * (-x[j2 - s])
            else:
                s_abs = -s
                j1 = np.arange(s_abs, n)
                if len(j1) > 0:
                    g[j1] += pk * (1 - x[j1 - s_abs])
                j2 = np.arange(0, n - s_abs)
                if len(j2) > 0:
                    g[j2] += pk * (-x[j2 + s_abs])
        g *= dx
        return f, g
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj_grad, h, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search_fast(h, n, deadline, deltas, max_passes=100):
    """Pair search with fast evaluation using precomputed correlation"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_full = compute_c5_array(h, n)
    dx = 2.0 / n
    shifts = np.arange(2*n-1) - (n-1)
    
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i+1, n):
                    if time.time() > deadline:
                        break
                    hi = best_h[i] + d
                    hj = best_h[j] - d
                    if hi > 1.0 or hj < 0.0:
                        continue
                    
                    # Fast evaluation
                    delta = np.zeros(2*n-1)
                    for idx in range(2*n-1):
                        s = shifts[idx]
                        term_i = 0.0
                        if 0 <= i+s < n:
                            term_i += (1 - best_h[i+s])
                        if 0 <= i-s < n:
                            term_i += -best_h[i-s]
                        term_j = 0.0
                        if 0 <= j+s < n:
                            term_j += (1 - best_h[j+s])
                        if 0 <= j-s < n:
                            term_j += -best_h[j-s]
                        delta[idx] = d * (term_i - term_j)
                    
                    c_new_vals = c_full + delta
                    c_new_max = float(np.max(c_new_vals))
                    
                    if c_new_max < best_c - 1e-15:
                        best_h[i] = hi
                        best_h[j] = hj
                        c_full = c_new_vals
                        best_c = c_new_max
                        improved_any = True
                    else:
                        # Try opposite direction
                        hi2 = best_h[i] - d
                        hj2 = best_h[j] + d
                        if hi2 < 0.0 or hj2 > 1.0:
                            continue
                        # delta is negated
                        c_new_vals2 = c_full - delta
                        c_new_max2 = float(np.max(c_new_vals2))
                        if c_new_max2 < best_c - 1e-15:
                            best_h[i] = hi2
                            best_h[j] = hj2
                            c_full = c_new_vals2
                            best_c = c_new_max2
                            improved_any = True
    
    return best_h, best_c
