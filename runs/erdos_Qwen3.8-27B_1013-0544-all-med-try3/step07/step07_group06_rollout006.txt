import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_add_at(h, n):
    p = np.arange(n)[:, None]
    lags = np.arange(-(n-1), n)[None, :]
    idx1 = p - lags
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = p + lags
    valid2 = (idx2 >= 0) & (idx2 < n)
    add_at = np.zeros((n, 2 * n - 1))
    add_at[valid1] = (1.0 - h[idx1[valid1]])
    add_at[valid2] -= h[idx2[valid2]]
    return add_at

def binary_local_search(h_init, n, deadline, max_sweeps=5000, n_restarts=6, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    current_h = h.copy()
    two_n1 = 2 * n - 1

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            n_swaps = rng.randint(3, 9)
            for _ in range(n_swaps):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(zi)
                    b = rng.choice(oi)
                    current_h[a] = 1.0
                    current_h[b] = 0.0
            h = current_h.copy()

        corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
        cur_c = float(corr.max()) * dx
        add_at = compute_add_at(h, n)

        stagnant = 0
        for sweep in range(max_sweeps):
            if time.time() > deadline - 0.5:
                break
            improved = False
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break

            nz = len(zeros)
            no = len(ones)
            threshold = corr.max()

            # Vectorized: try all (a,b) swaps at once
            # delta[a,b,m] = add_at[a,m] - add_at[b,m]
            # new_corr = corr + delta → new_max = max(new_corr)
            dz = zeros
            dc = ones
            # (nz, 1, 2n1) - (1, no, 2n1) → (nz, no, 2n1)
            delta = add_at[dz][:, None, :] - add_at[dc][None, :, :]
            new_corr = corr[None, None, :] + delta
            new_max = new_corr.max(axis=2)  # (nz, no)
            best_flat = int(new_max.argmin())
            best_a_i, best_b_i = divmod(best_flat, no)
            bm = new_max[best_a_i, best_b_i]

            if bm < threshold - 1e-14:
                a = dz[best_a_i]
                b = dc[best_b_i]
                h[a] = 1.0
                h[b] = 0.0
                corr = new_corr[best_a_i, best_b_i, :].copy()
                cur_c = float(corr.max()) * dx
                add_at = compute_add_at(h, n)
                improved = True
                stagnant = 0
            else:
                # Try 2-swap: swap two 1s with two 0s
                if stagnant < 3 and no >= 2 and nz >= 2:
                    found_2swap = False
                    # Sample a few pairs to keep time bounded
                    n_trials = min(no * (no - 1) // 2, 5000)
                    for _ in range(min(n_trials, 2000)):
                        if time.time() > deadline - 0.3:
                            break
                        bi = rng.randint(0, no)
                        bj = rng.randint(0, no - 1)
                        if bj >= bi:
                            bj += 1
                        ai = rng.randint(0, nz)
                        aj = rng.randint(0, nz - 1)
                        if aj >= ai:
                            aj += 1
                        b1, b2 = dc[bi], dc[bj]
                        a1, a2 = dz[ai], dz[aj]
                        delta2 = (add_at[a1] + add_at[a2] - add_at[b1] - add_at[b2])
                        nc2 = corr + delta2

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        if alpha < 1e6:
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        else:
            # Top-k smoothing: only smooth over top 20
            k_top = min(20, len(g))
            top_idx = np.argpartition(g, -k_top)[-k_top:]
            gt = g[top_idx]
            mt = gt.max()
            st = alpha * (gt - mt)
            st_max = st.max()
            return mt + (st_max + np.log(np.exp(st - st_max).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < np.inf:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def slsqp_topk(h, n, k_top, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        kk = min(k_top, len(g))
        top_idx = np.argpartition(g, -kk)[-kk:]
        gt = g[top_idx]
        mt = gt.max()
        # Smooth max over top-k with small smoothing
        eps = 0.01
        s = np.log(np.exp((gt - mt) / eps).sum()) * eps
        return mt + s

    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        kk = min(k_top, len(g))
        top_idx = np.argpartition(g, -kk)[-kk:]
        gt = g[top_idx].copy()
        mt = gt.max()
        eps = 0.01
        weights = np.exp((gt - mt) / eps)
        weights /= weights.sum()
        # Gradient of g[k] w.r.t. h:
        # g_k = sum_i h[i] * (1-h[i+k]) * dx (for lag k)
        # dg_k/dh[j] = (1-h[j+k])*dx - h[j-k]*dx (with boundary handling)
        grad = np.zeros(n)
        for k_t in top_idx:
            lag = int(k_t) - (n - 1)
            w = weights[list(top_idx).index(k_t)]
            # (1-h[j+lag]) for j where j+lag in [0,n)
            lo = max(0, -lag)
            hi = min(n, n - lag)
            j_idx = np.arange(lo, hi)
            jp = j_idx + lag
            grad[j_idx] += w * (1.0 - h[jp]) * dx
            # -h[j-lag] for j where j-lag in [0,n)
            lo2 = max(0, lag)
            hi2 = min(n, n + lag)
            j_idx2 = np.arange(lo2, hi2)
            jm = j_idx2 - lag
            grad[j_idx2] -= w * h[jm] * dx
        return grad

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n) / n}]
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        pass
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.fun < np.inf:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def coordinate_descent_smooth(h, n, deadline, max_iters=50000):
    """Projected coordinate descent on the smoothed objective."""
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    k_top = 30

    for it in range(max_iters):
        if time.time() > deadline - 1:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        kk = min(k_top, len(g))
        top_idx = np.argpartition(g, -kk)[-kk:]
        gt = g[top_idx].copy()
        mt = gt.max()
        eps = 0.05
        weights = np.exp((gt - mt) / eps)
        weights /= weights.sum()

        grad = np.zeros(n)
        for t_idx, k_t in enumerate(top_idx):
            lag = int(k_t) - (n - 1)
            w = weights[t_idx]
            lo = max(0, -lag)
            hi = min(n, n - lag)
            j_idx = np.arange(lo, hi)
            jp = j_idx + lag
            grad[j_idx] += w * (1.0 - h[jp]) * dx
            lo2 = max(0, lag)
            hi2 = min(n, n + lag)
            j_idx2 = np.arange(lo2, hi2)
            jm = j_idx2 - lag
            grad[j_idx2] -= w * h[jm] * dx

        # Projected gradient step with line search
        step = 0.01
        for ls in range(20):
            s = step / (1.0 + np.abs(grad).max() + 1e-10)
            h_new = h - s * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c + 1e-10:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
            step *= 0.5
        else:
            if it > 100:
                break

    return best_h, best_c
