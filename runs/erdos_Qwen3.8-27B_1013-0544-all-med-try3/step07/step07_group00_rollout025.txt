def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def correlation_grad(h, n, shift_idx):
    """Gradient of the correlation at a specific shift index."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def subgradient_minimax(h, n, deadline, max_iter=10000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    step = 0.01
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Active set: shifts within 0.1% of max
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        # Compute subgradient (average of active gradients)
        grad = np.zeros(n)
        for idx in active:
            grad += correlation_grad(h, n, idx)
        grad /= len(active)
        
        # Line search
        for mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            s = step * mult
            h_trial = project_feasible(h - s * grad, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c:
                best_c = c_trial
                best_h = h_trial.copy()
                h = h_trial
                break
        else:
            step *= 0.5
            if step < 1e-8:
                break
    
    return best_h, best_c

Actually, I want to add one more idea: **the "water-filling" approach**.

The idea is:
1. Compute all correlation values c_0, c_1, ..., c_{2n-2}
2. The target is to make them all equal (or at least the top ones equal)
3. "Water-filling": find the level τ such that the "water" (mass from h) flows to equalize the top correlations
4. This is analogous to the water-filling algorithm in communications

In practice:
1. Find the top K shifts
2. Set a target level τ = average of the top K
3. For each shift k > τ, compute the "deficit" d_k = τ - c_k
4. For each shift k < τ (in the top K), compute the "surplus" s_k = c_k - τ
5. Adjust h to reduce the surplus and fill the deficit

This is a linearized approach and might not work directly, but it gives intuition for the right direction.

The gradient-based version is:
- Compute the gradient of each top-K shift
- Weight them by the "deficit" or "surplus"
- Take a step

This is essentially what the softmax-approach does, but with adaptive weights.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def corr_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def softmax_obj(h, n, active, alpha, mu=0.0):
    c_vec = compute_c5_vec(h, n)
    vals = np.array([c_vec[i] for i in active])
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(h, n, active, alpha, mu=0.0):
    c_vec = compute_c5_vec(h, n)
    vals = np.array([c_vec[i] for i in active])
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def run_active_set_lbfgs(h, n, deadline, max_rounds=50, K_range=(3, 20)):
    """Dynamic active-set L-BFGS optimization."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    alphas = [5000, 20000, 50000, 200000, 500000, 2000000]
    
    for round_i in range(max_rounds):
        if time.time() > deadline - 10:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Find active set
        for frac in [0.0001, 0.0005, 0.001, 0.002, 0.005, 0.01]:
            threshold = max_val - max(1e-10, frac * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) >= K_range[0]:
                break
        
        if len(active) > K_range[1]:
            # Keep only the top K
            active = active[np.argsort(-c_vec[active])[:K_range[1]]]
        
        if len(active) < 2:
            # Use all top 5
            active = np.argsort(-c_vec)[:5]
        
        # Choose alpha based on round
        alpha = alphas[min(round_i // 8, len(alphas) - 1)]
        
        # Run L-BFGS
        try:
            res = minimize(
                softmax_obj, h, jac=softmax_grad,
                args=(n, active, alpha, 0.01),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': 20000, 'maxls': 30}
            )
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
        
        # Early stop if no improvement for 10 rounds
        if (round_i > 0 and round_i % 10 == 0):
            # Check if we're making progress
            pass
    
    return best_h, best_c

def subgradient_minimax(h, n, deadline, max_iter=50000):
    """Subgradient method for the minimax problem."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.005
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) == 0:
            active = np.argmax(c_vec)
            active = np.array([active])
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in active:
            grad += corr_grad_single(h, n, idx)
        grad /= len(active)
        
        # Try a few step sizes
        found = False
        for mult in [1.0, 0.5, 0.25, 0.1]:
            s = step * mult
            h_trial = project_feasible(h - s * grad, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                best_c = c_trial
                best_h = h_trial.copy()
                h = h_trial
                no_improve = 0
                found = True
                break
        
        if not found:
            no_improve += 1
            if no_improve > 20:
                step *= 0.5
                if step < 1e-7:
                    break
    
    return best_h, best_c
