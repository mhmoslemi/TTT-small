```python
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def f_values(h, i, n):
    """Compute f(i, L) for all L values, indexed by m = L + (n-1)"""
    result = np.zeros(2*n - 1)
    # (1 - h[i+L])_valid: valid when 0 <= i+L < n

        m_end = L_end + (n-1)
        idx = i + np.arange(L_start, L_end+1)
        result[m_start:m_end+1] += (1.0 - h[idx])

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
```

Actually, I want to add one more optimization to the local search: instead of trying a fixed set of deltas, I can do a "binary search" on the optimal delta for each pair. The idea is:
1. Compute d = fi - fj
2. The function g(δ) = max(c + δ*d) is convex in δ
3. If g(0) = max(c) and we want g(δ) < g(0), we need δ such that the max decreases
4. The optimal δ (for a given direction) can be found by:
   a. Compute the "slope" of g at δ=0: this is the max of d[m] for m where c[m] = max(c)
   b. If all slopes are ≤ 0, increasing δ helps; if all ≥ 0, decreasing helps
   c. The optimal δ is where the slope changes sign

For simplicity, I'll just use a geometric sequence of deltas: [0.05, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def f_values(h, i, n):
    result = np.zeros(2 * n - 1)
    L_start = max(-(n - 1), -i)
    L_end = min(n - 1, n - 1 - i)
    if L_start <= L_end:
        m_start = L_start + (n - 1)
        m_end = L_end + (n - 1)
        idx = i + np.arange(L_start, L_end + 1)
        result[m_start:m_end + 1] += (1.0 - h[idx])
    L_start2 = max(-(n - 1), i - n + 1)
    L_end2 = min(n - 1, i)
    if L_start2 <= L_end2:
        m_start2 = L_start2 + (n - 1)
        m_end2 = L_end2 + (n - 1)
        idx2 = i - np.arange(L_start2, L_end2 + 1)
        result[m_start2:m_end2 + 1] -= h[idx2]
    return result


def fast_local_search(h, n, deadline, deltas, radius=None, recompute_every=200):
    c = compute_corr(h, n).copy()
    best_c = float(c.max())
    count = n if radius is None else radius
    move_count = 0
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            fi = f_values(h, i, n)
            lo = max(0, i - count)
            hi = min(n, i + count + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                fj = f_values(h, j, n)
                d = fi - fj
                for delta in deltas:
                    if h[i] + delta <= 1.0 + 1e-15 and h[j] - delta >= -1e-15:
                        new_max = float(np.max(c + delta * d))
                        if new_max < best_c - 1e-16:
                            c = c + delta * d
                            h[i] += delta
                            h[j] -= delta
                            best_c = new_max
                            fi = f_values(h, i, n)
                            improved = True
                            move_count += 1
                            break
                    if h[i] - delta >= -1e-15 and h[j] + delta <= 1.0 + 1e-15:
                        new_max = float(np.max(c - delta * d))
                        if new_max < best_c - 1e-16:
                            c = c - delta * d
                            h[i] -= delta
                            h[j] += delta
                            best_c = new_max
                            fi = f_values(h, i, n)
                            improved = True
                            move_count += 1
                            break
                if move_count > 0 and move_count % recompute_every == 0:
                    c = compute_corr(h, n).copy()
                    best_c_verify = float(c.max())
                    if abs(best_c_verify - best_c) > 1e-12:
                        best_c = best_c_verify
        if not improved:
            break
    c_final = compute_corr(h, n)
    best_c = float(c_final.max())
    return h, best_c


def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def _sum_constraint(x, n):
    return float(x.sum() - n / 2.0)


def _sum_jacobian(x, n):
    return np.ones(len(x))


def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def optimize_from_start(h_init, n, deadline, rng):
    h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(h, n)

    for alpha in [2000, 20000, 200000]:
        if time.time() > deadline - 60:
            break
        remaining = deadline - time.time() - 60
        if remaining < 15:
            break
        maxiter = min(2000, int(remaining * 10))
        ht = slsqp_optimize(h, n, alpha, maxiter=max(500, maxiter))
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            h = ht

    if time.time() < deadline - 5:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        h, best_c = fast_local_search(h, n, deadline - 1, deltas, radius=n)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: SLSQP + local search on initial construction (budget: 350s)
    alloc1 = min(350, (deadline - t0) * 0.35)
    h1, c1 = optimize_from_start(h_init, n_init, t0 + alloc1, rng)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Try different resolutions (budget: 250s)
    n_candidates = [200, 300, 150, 350]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if n_try == n_init:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        remaining = deadline - time.time() - 80
        alloc = min(120, remaining * 0.3)
        if alloc < 30:
            break
        h_try, c_try = optimize_from_start(h_interp, n_try, time.time() + alloc, rng)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Phase 3: Multi-restart with perturbation (budget: 150s)
    n_best = len(best_h)
    if time.time() < deadline - 60:
        remaining = deadline - time.time() - 60
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(45, (deadline - time.time() - 30) * 0.4)
            if alloc < 15:
                break
            h_try, c_try = optimize_from_start(hp, n_best, time.time() + alloc, rng)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()

    # Phase 4: Aggressive fine-tuning on best (budget: remaining)
    if time.time() < deadline - 3:
        n_best = len(best_h)
        deltas_fine = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        best_h, best_c = fast_local_search(best_h, n_best, deadline - 1,
                                           deltas_fine, radius=n_best)
        # One more SLSQP polish
        if time.time() < deadline - 2:
            ht = slsqp_optimize(best_h, n_best, 500000, maxiter=2000)
            c = compute_c5(ht, n_best)
            if c < best_c:
                best_c = c
                best_h = ht
            # Final tiny local search
            if time.time() < deadline - 1:
                deltas_finest = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
                best_h, best_c = fast_local_search(best_h, n_best, deadline - 0.5,
                                                   deltas_finest, radius=n_best)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
