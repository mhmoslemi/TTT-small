def fast_len(m):
    m = int(m)
    if m <= 0:
        return 1
    return 1 << (m - 1).bit_length()


def grid_x(n):
    n = int(n)
    return (np.arange(n, dtype=float) + 0.5) * (2.0 / n)


def project_box_sum(v, target, iters=45):
    v = np.asarray(v, dtype=float)
    target = float(target)
    c = np.clip(v, 0.0, 1.0)
    if abs(float(c.sum()) - target) < 1e-13:
        return c
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(int(iters)):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    out = np.clip(v - lam, 0.0, 1.0)
    s = float(out.sum())
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0.0:
            mask = out < 1.0
        else:
            mask = out > 0.0
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            out[mask] += diff / cnt
    return np.clip(out, 0.0, 1.0)


def project_fast(h, target, iters=20):
    h = np.asarray(h, dtype=float).copy()
    target = float(target)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(int(iters)):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h


def fix_sum(h):
    h = np.asarray(h, dtype=float).copy()
    target = float(h.size) / 2.0
    for _ in range(12):
        s = float(np.sum(h, dtype=np.float64))
        diff = target - s
        if abs(diff) < 1e-15:

            idx = int(np.argmax(1.0 - h))

def corr_fast(h, v=None):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if v is None:
        v = 1.0 - h
    L = fast_len(2 * n - 1)
    A = np.fft.rfft(h, L)
    B = np.fft.rfft(v[::-1], L)
    out = np.fft.irfft(A * B, L)
    return out[:2 * n - 1]


def compute_c5_fast(h):
    n = len(h)
    if n == 0:
        return 0.0
    T = corr_fast(h, 1.0 - h)
    return float(np.max(T) * (2.0 / n))


def compute_c5_exact(h):
    n = len(h)
    if n == 0:
        return 0.0
    T = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(T) * (2.0 / n))


def symmetrize(h):
    h = np.asarray(h, dtype=float)
    h = 0.5 * (h + h[::-1])

def interpolate_h(h_old, n_old, n_new):

    if n_new == n_old:
        return h_old.copy()
    x_old = grid_x(n_old)
    x_new = grid_x(n_new)
    return np.interp(x_new, x_old, h_old)


def spectral_triangular_init(n, a=0.24):
    n = int(2 * (n // 2))
    if n <= 2:
        n = 4
    k = np.arange(n)
    dist = np.minimum(k, n - k)
    R = np.maximum(0.0, float(a) - dist / float(n))
    R[0] = float(a)
    P = np.fft.rfft(R)
    P = np.real(P)
    P[P < 0.0] = 0.0
    U = np.sqrt(P)
    u = np.fft.irfft(U, n)
    u = np.roll(u, n // 2)
    u = 0.5 * (u + u[::-1])
    u -= float(u.mean())
    t = np.arange(n, dtype=float) / float(max(1, n - 1))
    w = np.cos(np.pi * (t - 0.5)) ** 2
    wmax = float(np.max(w))
    if wmax > 0.0:
        w /= wmax
    u *= w
    u -= float(u.mean())
    m = float(np.max(np.abs(u)))
    if m > 1e-15:
        u *= (0.95 / m)
        if m > 0.0:
            u /= m
    h = 0.5 + 0.5 * u
    return project_box_sum(np.clip(h, 0.0, 1.0), n / 2.0)


def sinc_init(n, a=0.52):

    if n <= 2:
        n = 4
    x = grid_x(n)
    y = x - 1.0

def softmax_weights(c, alpha):
    c = np.asarray(c, dtype=float)
    if c.size == 0:
        return np.array([])
    m = float(np.max(c))
    z = float(alpha) * (c - m)
    e = np.exp(np.minimum(z, 0.0))
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        w = np.zeros_like(c, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def grad_T_weights(h, weights):
    h = np.asarray(h, dtype=float)
    weights = np.asarray(weights, dtype=float)
    n = len(h)
    L = fast_len(3 * n - 2)
    u = 1.0 - h
    Fu = np.fft.rfft(u, L)
    Fh = np.fft.rfft(h[::-1], L)
    Fw = np.fft.rfft(weights, L)
    t1 = np.fft.irfft(Fu * Fw, L)[n - 1:2 * n - 1]
    t2 = np.fft.irfft(Fh * Fw, L)[n - 1:2 * n - 1][::-1]
    return t1 - t2


def pgd_optimize(h, n, deadline):
    h = np.asarray(h, dtype=float)
    h = project_box_sum(h, n / 2.0)
    dx = 2.0 / n
    best = compute_c5_fast(h)
    alphas = [20.0, 100.0, 400.0, 1600.0, 6000.0, 20000.0]
    best_step = 0.01
    for alpha in alphas:
        no_improve = 0
        while time.time() < deadline:
            T = corr_fast(h, 1.0 - h)
            c = T * dx
            cl = float(np.max(c))
            if cl < best - 1e-13:
                best = cl
            if alpha > 1e7:
                w = np.zeros_like(T, dtype=float)
                w[int(np.argmax(T))] = 1.0
            else:
                w = softmax_weights(c, alpha)
            g = grad_T_weights(h, w)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
            direction = -g / gn
            cand_steps = [
                best_step,
                0.5 * best_step,
                2.0 * best_step,
                1e-4, 3e-4, 1e-3, 3e-3,
                1e-2, 3e-2, 0.1, 0.3, 0.8
            ]
            improved = False
            for eta in cand_steps:
                if time.time() > deadline:
                    break
                if eta < 1e-6 or eta > 2.0:
                    continue
                hnew = project_box_sum(h + eta * direction, n / 2.0, iters=30)
                cnew = compute_c5_fast(hnew)
                if cnew < best - 1e-12:
                    h = hnew
                    best = cnew
                    best_step = float(eta)
                    improved = True
                    break
            if improved:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 3:
                    break
        if time.time() > deadline:
            break
    h = project_box_sum(h, n / 2.0)
    return h, float(compute_c5_fast(h))


def sum_eq(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_eq_jac(x, n):
    return np.ones(len(x), dtype=float)


def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    T = corr_fast(x, 1.0 - x)
    c = T * dx
    m = float(np.max(c))
    z = float(alpha) * (c - m)
    e = np.exp(np.minimum(z, 0.0))
    s = float(e.sum())
    if s <= 0.0:
        return m
    return float(m + math.log(s) / alpha)


def smooth_jac(x, n, alpha):
    dx = 2.0 / n
    T = corr_fast(x, 1.0 - x)
    c = T * dx
    w = softmax_weights(c, alpha)
    g = grad_T_weights(x, w)
    return dx * g


def slsqp_smooth(h, n, alpha, maxiter=50):
    h = np.asarray(h, dtype=float)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac, "args": (n,)}]
        res = minimize(
            smooth_obj,
            h,
            args=(n, alpha),
            jac=smooth_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12},
        )
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        return h.copy()


def optimize_full(h, n, deadline):
    h = np.asarray(h, dtype=float)
    n = int(n)
    h = project_box_sum(h, n / 2.0)
    cfast = compute_c5_fast(h)
    if n <= 260 and time.time() < deadline - 25:
        for a in [500.0, 2000.0]:
            if time.time() > deadline - 10:
                break
            ht = slsqp_smooth(h, n, a, maxiter=50)
            ct = compute_c5_fast(ht)
            if ct < cfast - 1e-13:
                h = ht
                cfast = ct
    if time.time() < deadline - 5:
        h, cfast = pgd_optimize(h, n, deadline)
    h = project_box_sum(h, n / 2.0)
    cex = compute_c5_exact(h)
    return h, float(cex)


def grad_active_lag(h, k):
    n = len(h)
    u = 1.0 - np.asarray(h, dtype=float)
    g = np.zeros(n, dtype=float)
    k = int(k)
    if k >= 0:
        if k == 0:
            g += u
            g -= h
        else:
            g[k:] += u[:n - k]
            g[:n - k] -= h[k:]
    else:
        m = -k
        g[:n - m] += u[m:]
        g[m:] -= h[:-m]
    return g


def make_G(h, p):
    h = np.asarray(h, dtype=float)
    n = len(h)
    u = 1.0 - h
    G = np.zeros(2 * n - 1, dtype=float)
    p = int(p)
    G[p:p + n] += u[::-1]
    off = n - 1 - p
    G[off:off + n] -= h
    return G


def active_pair_polish(h, n, deadline):
    h = np.asarray(h, dtype=float)
    h = project_box_sum(h, n / 2.0)
    dx = 2.0 / n
    best = compute_c5_fast(h)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    if n <= 300:
        list_size = 35
        top_lags = 8
    elif n <= 800:
        list_size = 25
        top_lags = 6
    else:
        list_size = 18
        top_lags = 5
    m = 2 * n - 1
    temp = np.empty(m, dtype=float)
    for _cycle in range(2):
        for d in deltas:
            improved_any = True
            while improved_any and time.time() < deadline:
                improved_any = False
                T = corr_fast(h, 1.0 - h)
                if top_lags >= m:
                    idx = np.arange(m)
                else:
                    idx = np.argpartition(T, -top_lags)[-top_lags:]
                idx = idx[np.argsort(T[idx])[::-1]]
                found = False
                for kidx in idx:
                    if time.time() >= deadline:
                        break
                    k = int(kidx) - (n - 1)
                    g = grad_active_lag(h, k)
                    order = np.argsort(g)
                    P = order[:list_size]
                    Q = order[::-1][:list_size]
                    U = np.unique(np.concatenate((P, Q)))
                    Gmat = np.empty((len(U), m), dtype=float)
                    for ii, pp in enumerate(U):
                        Gmat[ii] = make_G(h, int(pp))
                    pos = {}
                    for ii, pp in enumerate(U):
                        pos[int(pp)] = ii
                    for pa in P:
                        ia = pos[int(pa)]
                        hav = float(h[pa])
                        for qb in Q:
                            if pa == qb:
                                continue
                            ib = pos[int(qb)]
                            np.subtract(Gmat[ia], Gmat[ib], out=temp)
                            np.multiply(temp, d, out=temp)
                            np.add(T, temp, out=temp)
                            mx = float(temp.max())
                            if mx * dx < best - 1e-14:
                                if 0.0 <= hav + d <= 1.0 and 0.0 <= h[qb] - d <= 1.0:
                                    hp = h.copy()
                                    hp[pa] = hav + d
                                    hp[qb] = float(h[qb]) - d
                                    cacc = compute_c5_fast(hp)
                                    if cacc < best - 1e-13:
                                        hp = project_box_sum(hp, n / 2.0, iters=20)
                                        cacc2 = compute_c5_fast(hp)
                                        if cacc2 < best - 1e-13:
                                            h = hp
                                            best = cacc2
                                            improved_any = True
                                            found = True
                                            break
                            if time.time() >= deadline:
                                break
                        if found or time.time() >= deadline:
                            break
                    if found or time.time() >= deadline:
                        break
                if found:
                    break
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break
    h = project_box_sum(h, n / 2.0)
    return h, float(compute_c5_fast(h))


def make_basis_matrix(n, M):
    x = grid_x(n)
    y = x - 1.0
    B = np.empty((M, n), dtype=float)
    for i in range(M):
        r = i + 1
        B[i] = np.cos(r * math.pi * y)
    return B


def basis_eval(params, B, target):
    params = np.asarray(params, dtype=float)
    u = params @ B
    h = np.clip(0.5 + 0.5 * u, 0.0, 1.0)
    return project_fast(h, float(target), iters=20)


def basis_objective(params, B, target):
    h = basis_eval(params, B, target)
    return compute_c5_fast(h)


def basis_coeffs_from_h(h, n, M):
    h = np.asarray(h, dtype=float)
    B = make_basis_matrix(n, M)
    u = h - 0.5
    return (2.0 / float(n)) * (B @ u)


def random_basis_search(n, M, deadline, seed, x0):
    B = make_basis_matrix(n, M)
    target = n / 2.0
    bounds_lo = -0.85
    bounds_hi = 0.85
    rng = np.random.RandomState(seed)
    x0 = np.clip(np.asarray(x0, dtype=float), bounds_lo, bounds_hi)
    bestx = x0.copy()
    bestc = basis_objective(bestx, B, target)
    while time.time() < deadline - 3:
        x = x0 + rng.uniform(-0.35, 0.35, M)
        x = np.clip(x, bounds_lo, bounds_hi)
        c = basis_objective(x, B, target)
        if c < bestc:
            bestx = x
            bestc = c
    return basis_eval(bestx, B, target)


def de_basis_search(n, M, deadline, seed, x0):
    B = make_basis_matrix(n, M)
    target = n / 2.0
    bounds = [(-0.85, 0.85)] * M
    x0 = np.asarray(x0, dtype=float)
    x0 = np.clip(x0, -0.85, 0.85)
    remaining = max(10.0, deadline - time.time())
    maxiter = int(max(6, remaining / 12.0))
    maxiter = min(maxiter, 30)
    if differential_evolution is not None:
        try:
            res = differential_evolution(
                basis_objective,
                bounds,
                x0=x0,
                args=(B, target),
                seed=seed,
                maxiter=maxiter,
                popsize=12,
                tol=1e-4,
                mutation=(0.5, 1.5),
                recomb=0.7,
                polish=False,
                updating="immediate",
                workers=1,
            )
            x0 = np.clip(res.x, -0.85, 0.85)
        except Exception:
            pass
    else:
        x0 = random_basis_search(n, M, deadline, seed, x0)
        return basis_eval(x0, B, target)
    if time.time() < deadline - 10:
        try:
            res2 = minimize(
                basis_objective,
                x0,
                args=(B, target),
                method="Nelder-Mead",
                options={"maxiter": 250, "xatol": 1e-8, "fatol": 1e-9},
            )
            x0 = np.clip(res2.x, -0.85, 0.85)
        except Exception:
            pass
    return basis_eval(x0, B, target)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 6.0
    rng = np.random.RandomState(seed)

    h_par = None
    try:
        h_par = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h_par = None

    if h_par is None or h_par.size == 0:
        n_par = 250
        h_par = spectral_triangular_init(n_par, 0.24)
    else:
        h_par = h_par.copy()
        n_par = int(h_par.size)

    h_par = project_box_sum(h_par, n_par / 2.0)
    best_h = h_par.copy()
    best_c = float(compute_c5_exact(best_h))
    best_n = int(best_h.size)

    hs = symmetrize(best_h.copy())
    cs = float(compute_c5_exact(hs))
    if cs < best_c:
        best_h = hs
        best_c = cs
        best_n = int(best_h.size)

    if time.time() < deadline - 40:
        parent_alloc = min(90.0, max(12.0, (deadline - t0) * 0.08))
        t_end = min(deadline - 25.0, t0 + parent_alloc)
        h_opt, c_opt = optimize_full(best_h.copy(), best_n, t_end)
        if c_opt < best_c:
            best_h = h_opt
            best_c = float(c_opt)
            best_n = int(best_h.size)

    base = max(int(best_n), 100)
    desired = []
    for mult in [1.35, 1.8, 2.4, 3.3, 4.5]:
        n = int(round(base * mult / 2.0) * 2)
        if n > base and n <= 2200:
            desired.append(n)
    for fixed in [900, 1300, 1800]:
        if fixed > base and fixed <= 2200:
            desired.append(fixed)
    desired = sorted(set(desired))
    if len(desired) > 5:
        idxs = [0, len(desired) // 3, 2 * len(desired) // 3, len(desired) - 2, len(desired) - 1]
        desired = [desired[int(i)] for i in sorted(set(idxs))]
    n_list = [int(2 * (x // 2)) for x in desired]
    n_list = [x for x in n_list if x > best_n]
    n_list.sort()

    for n in n_list:
        if time.time() > deadline - 50:
            break
        remaining = deadline - time.time()
        if n < 500:
            alloc = 45.0
        elif n < 900:
            alloc = 80.0
        elif n < 1400:
            alloc = 110.0
        else:
            alloc = 150.0
        alloc = min(alloc, 0.45 * remaining, remaining - 45.0)
        if alloc < 25.0:
            break

        cands = []
        h_int = interpolate_h(best_h, best_n, n)
        cands.append(h_int)
        for a in [0.20, 0.25, 0.30]:
            cands.append(spectral_triangular_init(n, a))
        cands.append(sinc_init(n, 0.52))

        scored = []
        for rank, hc in enumerate(cands):
            hc = project_box_sum(hc, n / 2.0, iters=35)
            if hc.size != n:
                continue
            cfs = float(compute_c5_fast(hc))
            if math.isfinite(cfs):
                scored.append((cfs, rank, hc))
        scored.sort()

        ntry = 1
        if len(scored) > 1 and n <= 900 and (deadline - time.time()) > max(alloc, 90.0):
            ntry = 2
        per = alloc / max(1, ntry)
        for rank in range(min(ntry, len(scored))):
            if time.time() > deadline - 40:
                break
            cfs, _, hc = scored[rank]
            if rank > 0 and cfs > best_c + 0.006:
                break
            t_end = min(deadline - 35.0, time.time() + per)
            h_opt, c_opt = optimize_full(hc, n, t_end)
            if c_opt < best_c:
                best_h = h_opt
                best_c = float(c_opt)
                best_n = int(best_h.size)

    if differential_evolution is not None and time.time() < deadline - 110:
        n_de = int(round(min(1024, max(512, best_n)) / 2.0) * 2)
        if n_de < 384:
            n_de = 384
        M = 10
        x0 = None
        if best_h is not None and best_n > 0:
            hd = interpolate_h(best_h, best_n, n_de)
            hd = project_box_sum(hd, n_de / 2.0)
            x0 = basis_coeffs_from_h(hd, n_de, M)
        de_alloc = min(120.0, max(35.0, (deadline - time.time()) * 0.25))
        t_end = min(deadline - 70.0, time.time() + de_alloc)
        h_de = de_basis_search(n_de, M, t_end, seed, x0)
        c_de = float(compute_c5_exact(h_de))
        if c_de < best_c:
            best_h = h_de
            best_c = c_de
            best_n = int(best_h.size)

        if time.time() < deadline - 90 and c_de < best_c + 0.005:
            n_up = int(round(min(1600, max(2 * n_de, best_n)) / 2.0) * 2)
            if n_up > best_n:
                h_up = interpolate_h(best_h, best_n, n_up)
                h_up = project_box_sum(h_up, n_up / 2.0)
                t_end = min(deadline - 40.0, time.time() + 80.0)
                h_opt, c_opt = optimize_full(h_up, n_up, t_end)
                if c_opt < best_c:
                    best_h = h_opt
                    best_c = float(c_opt)
                    best_n = int(best_h.size)

    if best_h is not None and time.time() < deadline - 30:
        h_pol, c_pol = active_pair_polish(best_h.copy(), best_n, deadline - 3.0)
        h_pol = project_box_sum(h_pol, best_n / 2.0)
        c_pol = float(compute_c5_exact(h_pol))
        if c_pol < best_c:
            best_h = h_pol
            best_c = c_pol
            best_n = int(best_h.size)

        hs = symmetrize(best_h.copy())
        cs = float(compute_c5_exact(hs))
        if cs < best_c:
            best_h = hs
            best_c = cs
            best_n = int(best_h.size)

        if time.time() < deadline - 20:
            for scale in [0.03, 0.015, 0.006]:
                if time.time() > deadline - 15:
                    break
                hp = best_h + rng.normal(0.0, scale, best_n)
                hp = project_box_sum(hp, best_n / 2.0)
                t_end = min(deadline - 2.0, time.time() + 25.0)
                ht, ct = active_pair_polish(hp, best_n, t_end)
                ht = project_box_sum(ht, best_n / 2.0)
                ct = float(compute_c5_exact(ht))
                if ct < best_c:
                    best_h = ht
                    best_c = ct
                    best_n = int(best_h.size)

    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c = 0.5

    best_h = np.asarray(best_h, dtype=float)
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = fix_sum(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = float(compute_c5_exact(best_h))
    best_n = int(best_h.size)
    return best_h, best_c, best_n
