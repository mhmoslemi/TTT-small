import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, active_idx, mu):
    c = compute_c5_vec(x, n)
    if active_idx is not None and len(active_idx) > 0:
        vals = c[active_idx]
    else:
        vals = c
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, active_idx, mu):
    c = compute_c5_vec(x, n)
    if active_idx is not None and len(active_idx) > 0:
        vals = c[active_idx]
        idxs = active_idx
    else:
        vals = c
        idxs = np.arange(len(c))
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_half = n - 1
    for wi, mm in zip(w, idxs):
        if wi < 1e-12:
            continue
        k = mm - n_half
        wm = wi * dx
        if k >= 0:
            nv = n - k
            # CORRECTED: grad[k:n] += (1-x[0:nv]) * wm
            #            grad[0:nv] += -x[k:n] * wm
            grad[k:] += (1.0 - x[:nv]) * wm
            grad[:nv] += (-x[k:]) * wm
        else:
            kk = -k
            nv = n - kk
            # CORRECTED: grad[0:nv] += (1-x[kk:n]) * wm
            #            grad[kk:n] += -x[0:nv] * wm
            grad[:nv] += (1.0 - x[kk:]) * wm
            grad[kk:] += (-x[:nv]) * wm
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active(c_vec, n, top_k=20, threshold_frac=0.001):
    m = c_vec.max()
    # Get shifts within threshold_frac of max, up to top_k
    thresh = m - threshold_frac * m
    active = np.where(c_vec >= thresh)[0]
    if len(active) > top_k:
        # Keep only the top_k
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    return active

def run_lbfgs(h, n, deadline, alpha_schedule, mu=0.0, top_k=20, label=""):
    """Run L-BFGS-B with adaptive active set."""
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    
    for alpha in alpha_schedule:
        if time.time() > deadline - 10:
            break
        # Refresh active set
        c_vec = compute_c5_vec(h_cur, n)
        active = get_active(c_vec, n, top_k=top_k)
        
        try:
            res = minimize(
                smooth_obj, h_cur,
                jac=smooth_grad,
                args=(n, alpha, active, mu),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': 10000, 'maxls': 40}
            )
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_c

def local_refine(h, n, deadline, step=0.001):
    """Fine local optimization with pair moves guided by gradient."""
    best_c = compute_c5(h, n)
    step_size = step
    n_stagnant = 0
    
    while time.time() < deadline and step_size > 1e-7:
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        thresh = m - 0.002 * m

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, active_idx, mu):
    c = compute_c5_vec(x, n)
    if active_idx is not None and len(active_idx) > 0:
        vals = c[active_idx]
    else:
        vals = c
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, active_idx, mu):
    c = compute_c5_vec(x, n)
    if active_idx is not None and len(active_idx) > 0:
        vals = c[active_idx]
        idxs = active_idx
    else:
        vals = c
        idxs = np.arange(len(c))
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_half = n - 1
    for wi, mm in zip(w, idxs):
        if wi < 1e-13:
            continue
        k = mm - n_half

        else:
            kk = -k
            nv = n - kk
            grad[:nv] += (1.0 - x[kk:]) * wm
            grad[kk:] += (-x[:nv]) * wm
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active(c_vec, top_k=20, threshold_frac=0.002):

    active = np.where(c_vec >= thresh)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    return active

def run_lbfgs_phase(h, n, deadline, alpha_schedule, mu, top_k, label=""):
    best_c = compute_c5(h, n)
