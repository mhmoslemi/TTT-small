            d_min = -h[i]
            d_max = min(1.0 - h[i], h[j])
            if d_max - d_min < 1e-14:
                continue
            d_star = optimal_pair_d(c, B, dx, d_min, d_max)
            if abs(d_star) > 1e-12:
                h[i] += d_star
                h[j] -= d_star
                h[i] = np.clip(h[i], 0.0, 1.0)
                h[j] = np.clip(h[j], 0.0, 1.0)
                c += dx * d_star * B
                cv = c.max()
                if cv < best_c - 1e-14:
                    best_c = cv
                    improved += 1
        sweeps += 1
        if improved == 0:
            stall_count += 1
            if stall_count >= 3:
                break
        else:
            stall_count = 0
        if sweeps >= 200:
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def full_pair_sweep(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    p = np.arange(n_corr) - n + 1
    c = np.correlate(h, 1 - h, mode='full') * dx
    rounds = 0
    while time.time() < deadline and rounds < 50:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                B = compute_B(h, i, j, p, n)
                d_min = -h[i]
                d_max = min(1.0 - h[i], h[j])
                if d_max - d_min < 1e-14:
                    continue
                d_star = optimal_pair_d(c, B, dx, d_min, d_max)
                if abs(d_star) > 1e-12:
                    h[i] += d_star
                    h[j] -= d_star
                    h[i] = np.clip(h[i], 0.0, 1.0)
                    h[j] = np.clip(h[j], 0.0, 1.0)
                    c += dx * d_star * B
                    cv = c.max()
                    if cv < best_c - 1e-14:
                        best_c = cv
                        improved = True
        rounds += 1
        if not improved:
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def gradient_phase(h, n, deadline, alpha=200):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    p_all = np.arange(n_corr) - n + 1
    h = h.copy()
    alpha_curr = alpha
    for _ in range(10000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c.max()
        if c_max < best_c:
            best_c = c_max
            h_best = h.copy()
        m_star = int(np.argmax(c))
        p_star = p_all[m_star]
        grad = dx * (safe_lookup(1 - h, np.arange(n) - p_star, n) -
                     safe_lookup(h, np.arange(n) + p_star, n))
        grad += grad.mean()
        alpha_ls = 1.0
        h_old = h.copy()
        c_old_max = c_max
        for _ in range(30):
            h_trial = h_old + alpha_ls * grad
            if h_trial.min() >= -1e-6 and h_trial.max() <= 1 + 1e-6:
                h_trial = np.clip(h_trial, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c_trial = np.correlate(h_trial, 1 - h_trial, mode='full') * dx
                if c_trial.max() < c_old_max - 0.01 * alpha_ls * np.abs(grad).sum() * 0.01:
                    break
            alpha_ls *= 0.5
        h = h_old + alpha_ls * grad
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            h_best = h.copy()
        else:
            alpha_curr = max(50, alpha_curr * 0.95)
    return h_best if 'h_best' in dir() else h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000):
    from scipy.optimize import minimize
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def _sum_c(x):
        return float(x.sum() - n / 2.0)
    def _sum_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def thue_morse(n):
    out = np.zeros(n)
    for i in range(n):
        out[i] = np.binary_repr(i).count('1') % 2
    return out.astype(float)

def generate_binary_start(n, rng):
    h = np.zeros(n)
    indices = rng.choice(n, size=n // 2, replace=False)
    h[indices] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    best_h = None
    best_c = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    h_init = project_feasible(h_init, n_init)
    c_init = compute_c5(h_init, n_init)
    best_h, best_c = h_init.copy(), c_init

    # Phase 1: Gradient-based optimization on initial (25% time)
    t_dead1 = t0 + (budget_s - 10) * 0.25
    h_g, c_g = gradient_phase(h_init.copy(), n_init, t_dead1, alpha=100)
    if c_g < best_c:
        best_c, best_h = c_g, h_g

    # Phase 2: SLSQP polish with increasing alpha (15% time)
    t_dead2 = t0 + (budget_s - 10) * 0.40
    ht = best_h.copy()
    for a in [200, 1000, 5000, 20000, 100000]:
        if time.time() > t_dead2 - 5:
            break
        ht = slsqp_smooth(ht, n_init, a, maxiter=800)
        c = compute_c5(ht, n_init)
        if c < best_c:
            best_c, best_h = c, ht.copy()

    # Phase 3: Random pair sweep (30% time)
    t_dead3 = t0 + (budget_s - 10) * 0.70
    h_ps, c_ps = random_pair_sweep(best_h.copy(), n_init, t_dead3, rng, n_pairs_per_sweep=5000)
    if c_ps < best_c:
        best_c, best_h = c_ps, h_ps

    # Phase 4: Near-pair full sweep (20% time)
    t_dead4 = t0 + (budget_s - 10) * 0.92
    h_fs, c_fs = full_pair_sweep(best_h.copy(), n_init, t_dead4, radius=40)
    if c_fs < best_c:
        best_c, best_h = c_fs, h_fs

    # Phase 5: Try alternative resolutions and starts (remaining time)
    if time.time() < deadline - 30:
        for n_alt in [128, 150, 200, 256]:
            if time.time() > deadline - 20:
                break
            if n_alt == n_init:
                continue
            # Interpolate best solution to new resolution
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_alt)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_alt)
            c_interp = compute_c5(h_interp, n_alt)

            alloc = min(60, max(10, deadline - time.time() - 15))
            h_alt, c_alt = random_pair_sweep(h_interp.copy(), n_alt, time.time() + alloc, rng, n_pairs_per_sweep=3000)
            if c_alt < best_c:
                best_c, best_h = c_alt, h_alt
                n_init = n_alt

            # Try binary start
            if time.time() < deadline - 15:
                h_bin = generate_binary_start(n_alt, rng)
                h_bin = project_feasible(h_bin, n_alt)
                alloc2 = min(30, max(5, deadline - time.time() - 10))
                h_b, c_b = random_pair_sweep(h_bin.copy(), n_alt, time.time() + alloc2, rng, n_pairs_per_sweep=2000)
                if c_b < best_c:
                    best_c, best_h = c_b, h_b
                    n_init = n_alt

    # Phase 6: Final polish
    if best_h is not None and time.time() < deadline - 3:
        n_final = len(best_h)
        h_final, c_final = full_pair_sweep(best_h.copy(), n_final, deadline - 1, radius=20)
        if c_final < best_c:
            best_c, best_h = c_final, h_final

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
