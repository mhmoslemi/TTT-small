def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_to_polytope(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    # Binary search for the shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any small numerical issues
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary by thresholding at the k-th largest value"""
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def lse_gradient(h, n, alpha):
    """Compute gradient of log-sum-exp(alpha * corr) w.r.t. h"""
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    a = alpha * corr
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:

        # d/dh[i] of corr[k+n-1] = (1-h[i+k]) for valid i+k, -h[i-k] for valid i-k
        # (with the k=0 correction handled implicitly)
        i_min = max(0, k)
        i_max = min(n-1, n-1+k)
        if i_min <= i_max:
            grad[i_min:i_max+1] += pk * (1.0 - h[i_min-k:i_max-k+1])
        i_min2 = max(0, -k)
        i_max2 = min(n-1, n-1-k)

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_to_polytope(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def precompute_swap_deltas(h, n):
    """Precompute delta_i for each position i (effect of changing h[i] from 0 to 1)"""
    # delta_i[m] = (1-h[i+k]) * 1[0<=i+k<n] - h[i-k] * 1[0<=i-k<n] where k = m-(n-1)
    # = (1-h[m-(n-1)+i]) for m in [n-1-i, 2(n-1)-i]
    #   - h[i+n-1-m] for m in [i, n-1+i]
    m_range = np.arange(-(n-1), n)
    deltas = np.zeros((n, 2*n-1))
    for i in range(n):
        im = i + m_range
        msk = (im >= 0) & (im < n)
        deltas[i][msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        deltas[i][msk] -= h[imn[msk]]
    return deltas

def lse_gradient(h, n, alpha):
    """Gradient of (1/alpha)*log(sum(exp(alpha*corr)))"""
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    a = alpha * corr
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n-1, n-1+k)
        if i_min <= i_max:
            grad[i_min:i_max+1] += pk * (1.0 - h[i_min-k:i_max-k+1])
        i_min2 = max(0, -k)
        i_max2 = min(n-1, n-1-k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2+1] -= pk * h[i_min2+k:i_max2+k+1]
    return grad

def pgd_refine(h, n, alphas, deadline, max_iter=5000):
    """Proximal gradient descent with log-sum-exp smoothing"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        cur_c = best_c
        step = min(float(n) / (4.0 * alpha), 0.3)
        
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            grad = lse_gradient(h, n, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            
            # Backtracking line search
            for ls in range(30):
                trial_step = step * (0.7 ** ls)
                h_new = project_to_polytope(h - trial_step * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    h = h_new
                    cur_c = c_new
                    step = trial_step
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    break
            else:
                # No improvement in line search
                step *= 0.5
                if step < 1e-10:
                    break
    
    return best_h, best_c
