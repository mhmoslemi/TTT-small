            alpha = min(100000.0, alpha * 1.3)
    return best_h, best_c

def slsqp_restricted(h, n, alpha, top_m, maxiter=5000, ftol=1e-15):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_max_obj, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_full(h, n, alpha, maxiter=4000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_full_smooth_obj, h, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def focused_lag_search(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        if lag > 0:
            contrib = h[m:n] * (1.0 - h[0:n-m])
        else:
            contrib = h[0:n-m] * (1.0 - h[m:n])
        top_pairs = np.argsort(contrib)[::-1]
        n_try = min(80, len(top_pairs))
        for pi in range(n_try):
            if time.time() > deadline:
                break
            idx = top_pairs[pi]
            if lag > 0:
                i = idx + m
                j = idx
            else:
                i = idx
                j = idx - m
            hi = h[i]
            hj = h[j]
            if hi <= hj + 1e-6:
                continue
            for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                delta = min(step, (hi - hj) * 0.3)
                if delta < 1e-8:
                    break
                h[i] -= delta
                h[j] += delta
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] += delta
                    h[j] -= delta
                    h = project_feasible(h, n)
        if not improved:
            # Try smaller random steps on contributing pairs
            for trial in range(30):
                if time.time() > deadline:
                    break
                idx = rng_focused.randint(0, len(top_pairs))
                if idx >= len(top_pairs):
                    continue
                pi2 = top_pairs[idx]
                if lag > 0:
                    i2 = pi2 + m
                    j2 = pi2
                else:
                    i2 = pi2
                    j2 = pi2 - m
                if h[i2] <= h[j2]:
                    continue
                delta = 0.0005 * rng_focused.random()
                if delta >= h[i2] or delta >= 1.0 - h[j2]:
                    continue
                h[i2] -= delta
                h[j2] += delta
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                else:
                    h[i2] += delta
                    h[j2] -= delta
                    h = project_feasible(h, n)
            if not improved:
                break
    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, t0_temp=0.02, t1_temp=0.001, max_iter=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    for it in range(max_iter):
        if time.time() > deadline:
            break
        temp = t0_temp * (t1_temp / t0_temp) ** (it / max_iter)
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        i = rng.randint(0, n - m)
        j = i + m
        if lag > 0:
            if h[i] > h[j]:
                delta = min(0.03 * rng.random(), h[i] * 0.2, (1-h[j]) * 0.2)
                h[i] -= delta
                h[j] += delta
            elif h[j] > h[i]:
                delta = min(0.03 * rng.random(), h[j] * 0.2, (1-h[i]) * 0.2)
                h[i] += delta
                h[j] -= delta
            else:
                continue
        else:
            if h[j] > h[i]:
                delta = min(0.03 * rng.random(), h[j] * 0.2, (1-h[i]) * 0.2)
                h[i] -= delta
                h[j] += delta
            elif h[i] > h[j]:
                delta = min(0.03 * rng.random(), h[i] * 0.2, (1-h[j]) * 0.2)
                h[i] += delta
                h[j] -= delta
            else:
                continue
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        delta_c = c_new - current_c
        if delta_c < 0 or rng.random() < np.exp(-delta_c / temp):
            current_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h = best_h.copy()
            current_c = best_c
    return best_h, best_c

def binarize_sort(h, n):
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def binary_local_search(h_bin, n, deadline, max_iters=100000):
    h_bin = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    rng_bin = np.random.RandomState(12345)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[rng_bin.randint(0, len(ones))]
        j = zeros[rng_bin.randint(0, len(zeros))]
        if i == j:
            continue
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c_new = compute_c5(h_bin, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_bin.copy()
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return best_h, best_c

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    global rng_focused
    t0 = time.time()
    rng = np.random.RandomState(seed)
    rng_focused = np.random.RandomState(seed + 1)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize at initial resolution
    if h_init is not None:
        h_work = project_feasible(h_init, n_init)
        best_c = compute_c5(h_work, n_init)
        best_h = h_work.copy()

        # Quick Nesterov momentum
        phase_dl = min(t0 + 120, deadline)
        h_new, c_new = nesterov_momentum(h_work, n_init, phase_dl, alpha=3000, lr=1.0, beta=0.85)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

        # SLSQP polish on top lags
        if time.time() < deadline - 20:
            for top_m in [2, 4, 6]:
                if time.time() > deadline - 15:
                    break
                h_sls = slsqp_restricted(best_h, n_init, 50000, top_m, maxiter=4000, ftol=1e-15)
                c_sls = compute_c5(h_sls, n_init)
                if c_sls < best_c:
                    best_c = c_sls
                    best_h = h_sls

        # Focused lag search
        if time.time() < deadline - 30:
            dl = min(deadline - 25, time.time() + 80)
            h_fl, c_fl = focused_lag_search(best_h, n_init, dl)
            if c_fl < best_c:
                best_c = c_fl
                best_h = h_fl

    else:
        h_work = np.full(n_init, 0.5)
        h_new, c_new = nesterov_momentum(h_work, n_init, min(t0 + 100, deadline), alpha=2000, lr=1.0)
        best_h, best_c = h_new, c_new

    # Phase 2: Try higher resolutions
    n_try_list = [600, 700, 800]
    for n_try in n_try_list:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 30
        if remaining < 90:
            break
        h_interp = interpolate_h(best_h, n_try)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(150, remaining * 0.3)
        if alloc < 50:
            break

        # Nesterov at new resolution
        h_opt, c_opt = nesterov_momentum(h_interp, n_try, time.time() + alloc * 0.5,
                                          alpha=5000, lr=1.2, beta=0.88)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

        # SLSQP at new resolution
        if time.time() < deadline - 60:
            h_s = slsqp_restricted(best_h, n_try, 80000, 4, maxiter=3000, ftol=1e-15)
            c_s = compute_c5(h_s, n_try)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

        # Focused search at new resolution
        if time.time() < deadline - 40:
            dl = min(deadline - 30, time.time() + 50)
            h_fl2, c_fl2 = focused_lag_search(best_h, n_try, dl)
            if c_fl2 < best_c:
                best_c = c_fl2
                best_h = h_fl2

    # Phase 3: Go back to best resolution and polish harder
    n_best = len(best_h)
    h_best = project_feasible(best_h, n_best)

    # Multiple SLSQP passes
    if time.time() < deadline - 40:
        for top_m in [2, 3, 5, 8]:
            if time.time() > deadline - 20:
                break
            h_s = slsqp_restricted(h_best, n_best, 100000, top_m, maxiter=5000, ftol=1e-16)
            c_s = compute_c5(h_s, n_best)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
                h_best = h_s

    # Focused lag search (longer)
    if time.time() < deadline - 20:
        dl = min(deadline - 10, time.time() + (deadline - time.time()) * 0.35)
        h_fl, c_fl = focused_lag_search(best_h, n_best, dl)
        if c_fl < best_c:
            best_c = c_fl
            best_h = h_fl

    # Simulated annealing on the best
    if time.time() < deadline - 15:
        sa_dl = min(deadline - 5, time.time() + 40)
        h_sa, c_sa = simulated_annealing(best_h, n_best, sa_dl, rng, t0_temp=0.01, t1_temp=0.0005)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Try binary solution near the optimum
    if time.time() < deadline - 10:
        h_bin = binarize_sort(best_h, n_best)
        c_bin = compute_c5(h_bin, n_best)
        if c_bin < best_c and time.time() < deadline - 10:
            h_bl, c_bl = binary_local_search(h_bin, n_best, min(deadline - 3, time.time() + 30))
            if c_bl < best_c:
                best_c = c_bl
                best_h = h_bl

    # Final: very fine local search
    if time.time() < deadline - 3:
        h_final = project_feasible(best_h, n_best)
        dl_final = deadline - 1
        # Small gradient steps
        for step in [0.0001, 0.00005]:
            if time.time() > dl_final:
                break
            c_all = compute_all_lags(h_final, n_best)
            k_top = int(np.argmax(c_all))
            lag_top = k_top - (n_best - 1)
            if lag_top == 0:
                break
            m_top = abs(lag_top)
            for i in range(0, n_best - m_top):
                if time.time() > dl_final:
                    break
                j = i + m_top
                if lag_top > 0:
                    if h_final[i] > h_final[j]:
                        d = min(step, (h_final[i] - h_final[j]) * 0.5)
                        h_final[i] -= d
                        h_final[j] += d
                else:
                    if h_final[j] > h_final[i]:
                        d = min(step, (h_final[j] - h_final[i]) * 0.5)
                        h_final[i] -= d
                        h_final[j] += d
                h_final = project_feasible(h_final, n_best)
                c_test = compute_c5(h_final, n_best)
                if c_test < best_c - 1e-16:
                    best_c = c_test
                    best_h = h_final.copy()

    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    return best_h, best_c, n_best
