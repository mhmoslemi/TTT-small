                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-12:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_orig = 200
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Adam at original resolution with coarse-to-fine alpha
    if time.time() < deadline - 400:
        t1 = min(time.time() + 150, deadline - 400)
        alpha_sched = [100, 500, 2000, 10000, 50000]
        lr_sched = [0.05, 0.03, 0.01, 0.005, 0.002]
        best_h, best_c = adam_optimize(best_h, n_orig, t1,
                                        alpha_sched, lr_sched,
                                        iters_per_phase=3000,
                                        active_recompute=300,
                                        max_active=60, mu=0.05)
    
    # Phase 2: L-BFGS with active set at original res
    if time.time() < deadline - 300:
        t2 = min(time.time() + 100, deadline - 300)
        h2, c2 = two_phase_lbfgs(best_h, n_orig, t2,
                                  alpha_list=(5000, 50000, 500000))
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to n=1000 for finer resolution
    n_fine = 1000
    h_fine = upsample_h(best_h, n_orig, n_fine)
    c_fine = compute_c5(h_fine, n_fine)
    
    if time.time() < deadline - 250:
        # Adam at n=1000
        t3 = min(time.time() + 120, deadline - 250)
        alpha_sched_f = [500, 2000, 10000, 50000]
        lr_sched_f = [0.02, 0.01, 0.005, 0.002]
        h_fine, c_fine = adam_optimize(h_fine, n_fine, t3,
                                        alpha_sched_f, lr_sched_f,
                                        iters_per_phase=2000,
                                        active_recompute=200,
                                        max_active=50, mu=0.02)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine
    else:
        # Stay at original resolution
        n_fine = n_orig
        h_fine = best_h
        c_fine = best_c
    
    # Phase 4: Subgradient descent at fine resolution
    if time.time() < deadline - 150:
        t4 = min(time.time() + 80, deadline - 150)
        h4, c4 = subgradient_descent(h_fine, n_fine, t4,
                                      steps=8000, lr_start=0.003, lr_end=0.00005)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Adam again with very high alpha (near-exact max)
    if time.time() < deadline - 80:
        t5 = min(time.time() + 50, deadline - 80)
        alpha_sched5 = [100000, 500000, 2000000]
        lr_sched5 = [0.003, 0.001, 0.0003]
        h5, c5 = adam_optimize(best_h, n_fine, t5,
                                alpha_sched5, lr_sched5,
                                iters_per_phase=3000,
                                active_recompute=100,
                                max_active=20, mu=0.005)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: L-BFGS polish on active set
    if time.time() < deadline - 40:
        t6 = min(time.time() + 30, deadline - 40)
        h6, c6 = two_phase_lbfgs(best_h, n_fine, t6,
                                  alpha_list=(100000, 1000000, 5000000))
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Subgradient fine polish
    if time.time() < deadline - 15:
        t7 = min(time.time() + 12, deadline - 15)
        h7, c7 = subgradient_descent(best_h, n_fine, t7,
                                      steps=3000, lr_start=0.001, lr_end=0.00001)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Fine pair descent
    if time.time() < deadline - 5:
        t8 = min(time.time() + 4, deadline - 5)
        h8, c8 = fine_pair_descent(best_h, n_fine, t8,
                                    deltas=(0.001, 0.0001), radius=30)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final projection and verification
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    # Also try at original resolution and keep the better
    if n_final != n_orig and n_orig >= 100:
        h_orig_test = upsample_h(best_h, n_final, n_orig)
        c_orig_test = compute_c5(h_orig_test, n_orig)
        if c_orig_test < best_c:
            best_h = h_orig_test
            best_c = c_orig_test
            n_final = n_orig
    
    return best_h, best_c, n_final
