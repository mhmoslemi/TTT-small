        try:
            def penalty_obj_and_grad(x):
                o, g = smooth_obj_and_grad(x, n, active_idx, alpha)
                s = x.sum() - n / 2.0
                lam = 1e5
                o += lam * s * s
                g += 2.0 * lam * s * np.ones(n)
                return o, g
            res = minimize(penalty_obj_and_grad, h, jac=True, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                  'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_m=5, n_rounds=6, alpha_start=1e5):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for rnd in range(n_rounds):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        alpha = alpha_start * (2 ** rnd)
        
        for method in ['trust-constr', 'slsqp']:
            if time.time() > deadline - 5:
                break
            h_new, c_new = optimize_round(h_curr, n, top_idx, alpha, 
                                          deadline - 5, method=method)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
                h_curr = h_new
            elif c_new < best_c + 1e-8:
                if np.max(np.abs(h_new - h_curr)) > 1e-10:
                    h_curr = h_new
    
    return best_h, best_c

def local_search(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                # Try to reduce lag 0 by pushing values toward 0 or 1
                for i in range(n):
                    if time.time() > deadline:
                        break
                    if 0.05 < h[i] < 0.95:
                        for direc in [1, -1]:
                            delta = min(d, 0.45, 0.5 - abs(h[i] - 0.5))
                            h[i] += direc * delta
                            h[i] = min(1.0, max(0.0, h[i]))
                            h = project_feasible(h, n)
                            c_new = compute_c5(h, n)
                            if c_new < best_c:
                                best_c = c_new
                                improved = True
                            else:
                                h = project_feasible(h - direc * delta * np.ones(n) / n, n)
                continue
            m = abs(lag)
            if m >= n:
                continue
            if lag > 0:
                pairs = [(i, i + m) for i in range(n - m)]
            else:
                pairs = [(i - m, i) for i in range(m, n)]
                pairs = [(i, i + m) for i in range(n - m)]
            
            n_checked = 0
            for i, j in pairs:
                if time.time() > deadline:
                    break
                if abs(h[i] - h[j]) < 1e-10:
                    continue
                # Try moving mass from higher to lower to reduce overlap
                if lag > 0:
                    # reduce h[i]*(1-h[j]) -> decrease h[i] or increase h[j]
                    if h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.8, h[i] * 0.9, (1 - h[j]) * 0.9)
                    else:
                        delta = min(d, (h[j] - h[i]) * 0.8, h[j] * 0.9, (1 - h[i]) * 0.9)
                        delta = -delta
                else:
                    if h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.8, h[j] * 0.9, (1 - h[i]) * 0.9)
                        delta = -delta
                    else:
                        delta = min(d, (h[i] - h[j]) * 0.8, h[i] * 0.9, (1 - h[j]) * 0.9)
                
                if abs(delta) < 1e-10:
                    continue
                h[i] += delta
                h[j] -= delta
                h[i] = min(1.0, max(0.0, h[i]))
                h[j] = min(1.0, max(0.0, h[j]))
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
                    h[i] = min(1.0, max(0.0, h[i]))
                    h[j] = min(1.0, max(0.0, h[j]))
                    h = project_feasible(h, n)
                n_checked += 1
    
    return h, best_c

def multi_alpha_sweep(h, n, deadline, top_m=5):
    """Sweep over different alpha values and active set sizes."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    c = compute_all_lags(h, n)
    for m in [3, 5, 7, 10]:
        if time.time() > deadline - 10:
            break
        top_idx = np.argsort(c)[-m:]
        for alpha in [5e4, 2e5, 1e6, 5e6, 2e7]:
            if time.time() > deadline - 5:
                break
            h_new, c_new = optimize_round(best_h, n, top_idx, alpha, 
                                          deadline - 5, method='slsqp')
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                c = compute_all_lags(best_h, n)
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Optimization at initial resolution (parent's n)
    if time.time() < deadline - 120:
        alloc1 = min(180, (deadline - time.time()) * 0.3)
        h1, c1 = active_set_optimize(best_h, n_init, t0 + alloc1, top_m=5, n_rounds=5, alpha_start=1e5)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Multi-alpha sweep at best resolution
    if time.time() < deadline - 100:
        alloc2 = min(80, (deadline - time.time()) * 0.2)
        h2, c2 = multi_alpha_sweep(best_h, n_init, t0 + alloc2, top_m=5)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Try other resolutions
    n_alternatives = [250, 320, 400, 200]
    for n_try in n_alternatives:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 40:
            break
        alloc = min(70, remaining * 0.2)
        if alloc < 25:
            break
        
        # Interpolate best solution to new resolution
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        
        h_opt, c_opt = active_set_optimize(h_start, n_try, time.time() + alloc, 
                                           top_m=5, n_rounds=4, alpha_start=1e5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 4: Go back to best resolution and do more rounds
    n_best = len(best_h)
    if time.time() < deadline - 60:
        alloc4 = min(80, (deadline - time.time()) * 0.3)
        h4, c4 = active_set_optimize(best_h, n_best, time.time() + alloc4, 
                                     top_m=7, n_rounds=4, alpha_start=5e5)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Noise restarts at best resolution
    if time.time() < deadline - 50:
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.008 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc_r = min(35, (deadline - time.time() - 25) * 0.35)
            if alloc_r < 15:
                break
            h_r, c_r = active_set_optimize(hp, n_best, time.time() + alloc_r,
                                           top_m=5, n_rounds=3, alpha_start=2e5)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    
    # Phase 6: Final polish
    if time.time() < deadline - 10:
        # Very tight SLSQP on the top 3 lags
        c = compute_all_lags(best_h, n_best)
        top3 = np.argsort(c)[-3:]
        for alpha in [5e6, 2e7]:
            if time.time() > deadline - 5:
                break
            h_new, c_new = optimize_round(best_h, n_best, top3, alpha,
                                          deadline - 3, method='slsqp')
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Local search
        if time.time() < deadline - 3:
            best_h, best_c = local_search(best_h, n_best, deadline - 1)
    
    # Final verification
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    
    return best_h, best_c, n_best
