import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def deterministic_pair_search(h, n, deadline, deltas, max_passes=80):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def randomized_pair_search(h, n, deadline, rng, deltas, samples=4000, max_passes=400):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for _ in range(samples):
                if time.time() > deadline:
                    return h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                sign = 1 if rng.rand() < 0.5 else -1
                hi = h[i] + sign * d
                hj = h[j] - sign * d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[j] = h[i] - sign * d, h[j] + sign * d
            if not improved:
                break
    return h, best_c

def band_structure_optimize(n, deadline, rng):
    """Optimize banded {0,1} structures by searching band positions."""
    best_h = None
    best_c = float('inf')
    dx = 2.0 / n
    for m in [3, 4, 5, 6, 7, 8, 10, 12, 15, 20, 25, 30]:
        if time.time() > deadline:
            break
        L = 1.0 / m
        L_idx = max(1, int(round(L / dx)))
        for trial in range(50):
            if time.time() > deadline:
                break
            # Try even spacing with random offset
            offset = rng.uniform(0, 2.0 / m)
            positions = [(offset + k * 2.0 / m) % 2.0 for k in range(m)]
            for _refine in range(50):
                if time.time() > deadline:
                    break
                improved = False
                for k in range(m):
                    for dp in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                        new_pos = (positions[k] + dp) % 2.0
                        valid = True
                        for l in range(m):
                            if l != k:
                                diff = abs(new_pos - positions[l])
                                diff = min(diff, 2.0 - diff)
                                if diff < L * 0.95:
                                    valid = False
                                    break
                        if not valid:
                            continue
                        old = positions[k]
                        positions[k] = new_pos
                        h = np.zeros(n)
                        for pos in positions:
                            s = int(round(pos / dx)) % n
                            e = s + L_idx
                            for idx in range(L_idx):
                                h[(s + idx) % n] = 1.0
                        h = project_feasible(h, n)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-12:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            positions[k] = old
                if not improved:
                    break
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_new, n_new)
    return h_new

def multi_alpha_slsqp(h, n, deadline, alphas, maxiter_each=3000):
    h = h.copy()
    best_c = compute_c5(h, n)
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_optimize(h, n, a, maxiter=maxiter_each, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht.copy()
    return h, best_c
