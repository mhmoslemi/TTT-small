    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def grad_active_set(h, n, n_top=10):
    c = compute_c5_vec(h, n)
    top_idx = np.argsort(c)[::-1][:n_top]
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in top_idx:
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += dx * (1.0 - h[k:n])
            grad[k:n] += dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += dx * (1.0 - h[0:nv])
            grad[0:nv] += dx * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_value(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (500, 5.0), (2000, 10.0), (5000, 50.0),
        (20000, 100.0), (100000, 500.0), (500000, 2000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def active_set_descent(h, n, deadline, n_top=5, lr_init=0.005):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stuck_count = 0
    for it in range(100000):
        if time.time() > deadline:
            break
        g = grad_active_set(h, n, n_top)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g /= g_norm
        improved = False
        for trial in range(40):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                stuck_count = 0
                break
            lr *= 0.5
        if not improved:
            lr *= 0.8
            stuck_count += 1
            if stuck_count > 50:
                lr = lr_init * 0.5
                stuck_count = 0
            if lr < 1e-13:
                break
    return best_h, best_c

def cyclic_coord_descent(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    order = list(range(n))
    np.random.RandomState(42).shuffle(order)
    for sweep in range(10000):
        if time.time() > deadline:
            break
        improved_sweep = False
        for i in order:
            if time.time() > deadline:
                return best_h, best_c
            # Try moving mass from i to a few other positions
            for j in [(i+1) % n, (i-1) % n, (i+n//3) % n, (i+2*n//3) % n]:
                if i == j:
                    continue
                # Try small moves in both directions
                for delta in [0.01, 0.005, 0.002, 0.001]:
                    if h[i] - delta >= 0 and h[j] + delta <= 1:
                        h[i] -= delta
                        h[j] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved_sweep = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    if h[i] + delta <= 1 and h[j] - delta >= 0:
                        h[i] += delta
                        h[j] -= delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved_sweep = True
                        else:
                            h[i] -= delta
                            h[j] += delta
        if not improved_sweep:
            break
    return best_h, best_c

def pair_search_smart(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    radius = max(20, n // 10)
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
        if not improved_any:
            continue
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=200000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    def _con_fun(x):
        return float(x.sum() - n / 2.0)
    def _con_jac(x):
        return np.ones(n)
    def _obj(x):
        c = compute_c5_vec(x, n)
        return float(smooth_max_value(c, alpha))
    def _obj_grad(x):
        return grad_smooth_max(x, n, alpha)
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def try_resolution(h_best, c_best, n_best, n_target, deadline):
    """Interpolate to n_target and optimize"""
    if time.time() > deadline - 30:
        return h_best, c_best, n_best
    x_old = np.linspace(0, 2, n_best)
    x_new = np.linspace(0, 2, n_target)
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_feasible(h_interp, n_target)
    c_inter = compute_c5(h_interp, n_target)
    # Phase A: L-BFGS
    t_a = min(time.time() + 80, deadline - 60)
    if t_a > time.time() + 10:
        h_new, c_new = lbfgs_phase(h_interp, n_target, t_a, max_iters=150)
    else:
        h_new, c_new = h_interp, c_inter
    # Phase B: Active set descent
    if time.time() < deadline - 40:
        t_b = min(time.time() + 60, deadline - 30)
        h_new, c_new = active_set_descent(h_new, n_target, t_b, n_top=5, lr_init=0.003)
    # Phase C: Pair search
    if time.time() < deadline - 15:
        h_new, c_new = pair_search_smart(h_new, n_target, deadline - 5,
                                         deltas=[0.01, 0.005, 0.002, 0.001, 0.0005])
    if c_new < c_best:
        return h_new, c_new, n_target
    return h_best, c_best, n_best

def random_restart(h_best, c_best, n, deadline, seed):
    """Try a random perturbation and see if it leads to improvement"""
    if time.time() > deadline - 60:
        return h_best, c_best
    rng = np.random.RandomState(seed)
    for attempt in range(5):
        if time.time() > deadline - 30:
            break
        # Random perturbation
        h_pert = h_best.copy()
        idx = rng.choice(n, size=n//5, replace=False)
        h_pert[idx] += rng.uniform(-0.1, 0.1, size=len(idx))
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        # Quick L-BFGS
        t_end = min(time.time() + 30, deadline - 20)
        if t_end < time.time() + 5:
            continue
        h_opt, c_opt = lbfgs_phase(h_pert, n, t_end, max_iters=50)
        if c_opt < c_best:
            return h_opt, c_opt
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    n_best = n_init

    # Phase 1: Quick L-BFGS polish at initial resolution
    t1 = min(t0 + 120, deadline - 400)
    best_h, best_c = lbfgs_phase(best_h, n_best, t1, max_iters=100)

    # Phase 2: Active set gradient descent
    if time.time() < deadline - 300:
        t2 = min(time.time() + 150, deadline - 300)
        best_h, best_c = active_set_descent(best_h, n_best, t2, n_top=5, lr_init=0.005)

    # Phase 3: Try higher resolutions
    resolutions_to_try = []
    if n_best < 500:
        resolutions_to_try.append(500)
    if n_best < 1000:
        resolutions_to_try.append(800)
    if n_best < 1500:
        resolutions_to_try.append(1200)

    for n_try in resolutions_to_try:
        if time.time() > deadline - 120:
            break
        t_end = min(time.time() + 200, deadline - 80)
        best_h, best_c, n_best = try_resolution(best_h, best_c, n_best, n_try, t_end)

    # Phase 4: SLSQP at best resolution
    if time.time() < deadline - 60:
        t4 = min(time.time() + 80, deadline - 40)
        for alpha in [100000, 500000]:
            if time.time() > t4:
                break
            h_s, c_s = slsqp_phase(best_h, n_best, min(time.time() + 40, t4),
                                   alpha=alpha, maxiter=800)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 5: Active set descent again
    if time.time() < deadline - 30:
        t5 = min(time.time() + 60, deadline - 20)
        best_h, best_c = active_set_descent(best_h, n_best, t5, n_top=3, lr_init=0.002)

    # Phase 6: Pair search for final polishing
    if time.time() < deadline - 10:
        t6 = min(time.time() + 50, deadline - 5)
        best_h, best_c = pair_search_smart(best_h, n_best, t6,
                                            deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    # Phase 7: One more active set pass if time remains
    if time.time() < deadline - 5:
        best_h, best_c = active_set_descent(best_h, n_best, deadline - 2, n_top=3, lr_init=0.001)

    # Final projection and evaluation
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)

    return best_h, best_c, n_best
