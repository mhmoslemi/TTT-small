    """Polish by optimizing pairs that reduce the top-k shifts."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    step_base = 0.01
    for round_num in range(max_rounds):
        if time.time() > deadline:
            break
        
        # Identify active shifts
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - 0.005 * max_val
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(-c_vec, kind='stable')[0]
            active = np.sort(active[np.argsort(-c_vec[active], kind='stable')[:top_k]])
        
        # Compute gradient contribution from active shifts
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)

                grad[0:nv] += dx * (1.0 - h[k:n])
                grad[k:n] += dx * (-h[0:nv])

                grad[0:nv] += dx * (-h[kk:n])

        # helps reduce the active correlations
        
        improved = False
        # Sort by gradient magnitude
        g_pos = np.where(grad > 0)[0]
        g_neg = np.where(grad < 0)[0]
        
        # Try moving from high-grad to low-grad positions
        # But we need to be selective - try the most promising pairs
        if len(g_pos) > 0 and len(g_neg) > 0:

            top_neg = g_neg[np.argsort(-grad[g_neg])[:20]]  # most negative first

            best_cj = 0

def lp_guided_step(h, n, active_shifts, max_step=0.01):
    """Use LP to find the best direction, then line search."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    c_vec = compute_c5_vec(h, n)
    
    # Compute gradients of c_k for active shifts
    grads = []
    for idx in active_shifts:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[0:nv] = dx * (1.0 - h[k:n])
            g[k:n] += dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:n] = dx * (1.0 - h[0:nv])
            g[0:nv] += dx * (-h[kk:n])
        grads.append(g)
    
    n_active = len(active_shifts)
    n_var = n + 1  # n for delta, 1 for t
    
    # Objective: minimize t = variable n
    c_obj = np.zeros(n_var)
    c_obj[n] = 1.0
    
    # Constraints: c_k(h) + grad_k^T delta <= t
    # i.e., grad_k^T delta - t <= -c_k(h)
    A_ub = np.zeros((n_active, n_var))
    b_ub = np.zeros(n_active)
    for i in range(n_active):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = -c_vec[active_shifts[i]]
    
    # Box constraints: -h_i <= delta_i <= 1-h_i
    # Also: sum(delta) = 0 (equality constraint)
    # And: |delta_i| <= max_step (to keep linearization valid)
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]

    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -max_step), min(1-h[i], max_step)))
    bounds.append((-np.inf, np.inf))  # t is unbounded
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return h, compute_c5(h, n)
    
    delta = res.x[:n]
    
    # Line search: find best alpha in [0, 1] such that h + alpha*delta is feasible
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
        h_try = h + alpha * delta
        h_try = np.clip(h_try, 0, 1)
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
    
    return best_h, best_c

This is a good idea! The LP gives us the theoretically best direction (given the linearization), and the line search ensures we don't overshoot.

But there's a concern: solving an LP with 1000+ variables might be slow. Let me check: HiGHS is very fast for LPs. An LP with 1000 variables and 3000 constraints should solve in < 1 second. So even if we do 100 LP iterations, that's < 100 seconds. Very feasible.

Actually, I realize there might be an issue with the LP. The constraint "sum(delta) = 0" combined with the box constraints might make the LP infeasible if max_step is too small. Let me use a softer constraint: instead of requiring |delta_i| ≤ max_step, I'll just rely on the box constraints and the line search.

Actually, the box constraints already limit delta: -h_i ≤ delta_i ≤ 1-h_i. The max_step is an additional constraint to keep the linearization valid. Let me keep it but make it generous (e.g., 0.05).

Let me also think about whether the LP approach is guaranteed to make progress. If the current point is a local minimum of the max (with the given active set), the LP will give delta = 0 (no improvement possible). In that case, we need to either:
1. Change the active set (add/remove shifts)
2. Use a different approach (like the softmax smoothing)

The LP approach is great for "polishing" when we're close to the optimum, but it might get stuck if the active set changes. So I'll use it in combination with the softmax approach.

OK, let me now write the final code. I'll combine:
1. Initial optimization with L-BFGS + softmax (to get close to optimum)
2. LP-guided steps (for final polishing)
3. Pair search (for additional fine-tuning)

1. Start at n=500 (from initial_h_values)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wmd = wi * dx
        if wmd < 1e-14:
            continue
        if k >= 0:
            nv = n - k
            grad[0:nv] += wmd * (1.0 - x[k:n])
            grad[k:n] += wmd * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wmd * (1.0 - x[0:nv])
            grad[0:nv] += wmd * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active_shifts(h, n, top_k=10, tol=0.002):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - tol * max_val
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.asarray(np.argsort(-c_vec, kind='stable')[0])
        active = np.sort(active[np.argsort(-c_vec[active], kind='stable')[:top_k]])
    return active

def optimize_topk_lbfgs(h, n, deadline, alpha_configs=None, mu=0.1):
    """L-BFGS optimization focusing on top shifts."""
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alpha_configs is None:
        alpha_configs = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    
    for alpha in alpha_configs:
        if time.time() > deadline - 10:
            break
        active = get_active_shifts(h, n, top_k=10, tol=0.003)
        if len(active) == 0:
            break
        try:
            res = minimize(topk_obj, h, jac=topk_grad,
                          args=(n, active, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def lp_polish(h, n, deadline, max_rounds=200):
    """LP-guided polishing step."""
    best_c = compute_c5(h, n)
    max_step = 0.02
    
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        
        active = get_active_shifts(h, n, top_k=8, tol=0.001)
        if len(active) < 2:
            break
        
        dx = 2.0 / n
        c_vec = compute_c5_vec(h, n)
        
        # Compute gradients
        grads = []
        for idx in active:
            k = idx - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                nv = n - k
                g[0:nv] = dx * (1.0 - h[k:n])
                g[k:n] += dx * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] = dx * (1.0 - h[0:nv])
                g[0:nv] += dx * (-h[kk:n])
            grads.append(g)
        
        # Set up LP: minimize t s.t. c_k + grad_k^T delta <= t, -h <= delta <= 1-h, sum(delta)=0, |delta|<=max_step
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        
        n_act = len(active)
        A_ub = np.zeros((n_act, n_var))
        b_ub = np.zeros(n_act)
        for i in range(n_act):
            A_ub[i, :n] = grads[i]
            A_ub[i, n] = -1.0
            b_ub[i] = -c_vec[active[i]]
        
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = [0.0]
        
        bounds = []
        for i in range(n):
            lo = max(-h[i], -max_step)
            hi = min(1.0 - h[i], max_step)
            bounds.append((lo, hi))
        bounds.append((-np.inf, np.inf))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 5})
        except Exception:
            break
        
        if res.status != 0:
            break
        
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-12:
            break
        
        # Line search
        improved = False
        for alpha_ls in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_try = h + alpha_ls * delta
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                best_c = c_try
                h = h_try
                improved = True
                break
        
        if not improved:
            max_step *= 0.5
            if max_step < 1e-6:
                break
    
    return h, best_c
