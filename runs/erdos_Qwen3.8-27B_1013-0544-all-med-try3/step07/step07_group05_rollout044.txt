        # Also need i-s valid: s in [i-(n-1), i]
        s_lo2 = max(i - (n - 1), -(n - 1))
        s_hi2 = min(i, n - 1)
        s_start = max(s_lo, s_lo2)
        s_end = min(s_hi, s_hi2)
        
        for s in range(s_start, s_end + 1):
            k = s + (n - 1)
            contrib = 0.0
            if 0 <= i + s < n:
                contrib += (1.0 - h_new[i + s])
            if 0 <= i - s < n:
                contrib -= h_new[i - s]
            corr[k] += dx * delta * contrib
    return corr


def efficient_coord_descent(h, n, deadline, max_iters=500000):
    """Coordinate descent with incremental correlation update."""
    h = h.copy()
    corr = compute_corr_full(h, n)
    best_c = float(corr.max())
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    dx = 2.0 / n
    
    it = 0
    while it < max_iters:
        if time.time() > deadline:
            break
        improved = False
        for d in deltas:
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Try h[i] + d
                for sign in [1, -1]:
                    hi = h[i] + sign * d
                    if hi < 0.0 or hi > 1.0:
                        continue
                    # Check sum constraint: we need sum(h) = n/2
                    # Changing h[i] by sign*d changes sum by sign*d
                    # So we need to compensate elsewhere, OR just allow it temporarily
                    # For simplicity, skip sum constraint in coord descent and project at end
                    old_val = h[i]
                    h[i] = hi
                    corr_new = fast_corr_update(corr.copy(), np.full(n, old_val if False else h[i] - sign*d), h, n)
                    # Actually simpler: just recompute the affected part
                    # Let me use a different approach: save old, set new, update corr
                    h[i] = old_val
                    # Recompute corr incrementally
                    old_h = h.copy()
                    h[i] = hi
                    # For all shifts, update:
                    for s in range(-(n-1), n):
                        k = s + (n - 1)
                        contrib = 0.0
                        if 0 <= i + s < n:
                            contrib += sign * d * (1.0 - h[i + s])
                        if 0 <= i - s < n:
                            contrib += -sign * d * h[i - s]
                        if abs(contrib) > 1e-15:
                            corr[k] += contrib
                    new_max = corr.max()
                    if new_max < best_c - 1e-13:
                        best_c = new_max
                        improved = True
                    else:
                        # Revert
                        for s in range(-(n-1), n):
                            k = s + (n - 1)
                            contrib = 0.0
                            if 0 <= i + s < n:
                                contrib += sign * d * (1.0 - h_new_val(i+s, h, hi, i, old_val))
                            if 0 <= i - s < n:
                                contrib += -sign * d * h_new_val(i-s, h, hi, i, old_val)
                            if abs(contrib) > 1e-15:
                                corr[k] -= contrib
                        h[i] = old_val
            if not improved:
                continue
        if not improved:
            break
        it += 1
    return h, best_c


def h_new_val(idx, h, new_val, changed_idx, old_val):
    if idx == changed_idx:
        return new_val
    return h[idx]


def coord_descent_v2(h, n, deadline, max_rounds=200):
    """Simpler coordinate descent: just recompute corr each time but try fewer things."""
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0003]
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(n):
                for sign in [1, -1]:
                    hi = h[i] + sign * d
                    if hi < 0 or hi > 1:
                        continue
                    h[i] = hi
                    h = project_feasible(h, n)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-12:
                        best_c = c
                        improved = True
                    else:
                        h[i] = hi - sign * d
            if not improved:
                break
    return h, best_c


def subgradient_method(h, n, deadline, iters=100000, lr0=0.001, lr_decay=0.9995):
    """Subgradient method on the active (max) constraint."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    lr = lr0
    
    for it in range(iters):
        if time.time() > deadline:
            break
        if it % 500 == 0:
            lr *= lr_decay
            c_cur = compute_c5(h, n)
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = corr.argmax()
        s = max_idx - (n - 1)  # shift
        
        # Gradient of c_s wrt h:
        g = np.zeros(n)
        if s == 0:
            g[:] = dx * (1 - 2 * h)
        elif s > 0:
            g[s:] += dx * (1 - h[:n - s])
            g[:n - s] -= dx * h[s:]
        else:
            sp = -s
            g[:n - sp] += dx * (1 - h[sp:])
            g[sp:] -= dx * h[:n - sp]
        
        # Projected gradient step
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new <= best_c + 1e-12:  # allow slight increase
            h = h_new
        # else: don't update h
    
    return best_h, best_c


def multi_scale_slsqp(h, n, deadline):
    """Run SLSQP with various alpha values."""
    h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [200, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    for a in alphas:
        if time.time() > deadline - 3:
            break
        ht = slsqp_optimize(h, n, a, maxiter=6000, ftol=1e-17)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht.copy()
    return h, best_c


def random_perturb_optimize(h_best, n, deadline, rng, n_trials=10):
    """Try random perturbations and re-optimize."""
    best_c = compute_c5(h_best, n)
    best_h = h_best.copy()
    
    for trial in range(n_trials):
        if time.time() > deadline - 10:
            break
        scale = rng.choice([0.001, 0.005, 0.01, 0.02, 0.05])
        hp = h_best + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        
        # Quick SLSQP
        alloc = min(15.0, (deadline - time.time() - 10) * 0.4)
        if alloc < 5:
            break
        ht = slsqp_optimize(hp, n, 10000, maxiter=3000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
    
    return best_h, best_c


def frequency_perturb_optimize(h_best, n, deadline, rng):
    """Try sinusoidal perturbations of various frequencies."""
    best_c = compute_c5(h_best, n)
    best_h = h_best.copy()
    x = np.linspace(0, 2, n)
    
    for freq in [2, 3, 5, 7, 11, 15]:
        if time.time() > deadline - 10:
            break
        for amp in [0.01, 0.03, 0.05]:
            if time.time() > deadline - 10:
                break
            phase = rng.uniform(0, 2 * np.pi)
            pert = amp * np.sin(2 * np.pi * freq * x / 2 + phase)
            hp = h_best + pert
            hp = project_feasible(hp, n)
            
            alloc = min(12.0, (deadline - time.time() - 10) * 0.35)
            if alloc < 4:
                break
            ht = slsqp_optimize(hp, n, 5000, maxiter=3000, ftol=1e-15)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht.copy()
    
    return best_h, best_c


def block_perturb_optimize(h_best, n, deadline, rng):
    """Try block (rectangular) perturbations."""
    best_c = compute_c5(h_best, n)
    best_h = h_best.copy()
    
    for _ in range(20):
        if time.time() > deadline - 10:
            break
        block_size = rng.randint(max(3, n // 20), max(5, n // 8))
        start = rng.randint(0, n - block_size)
        amp = rng.choice([-0.1, -0.05, 0.05, 0.1, -0.2, 0.2])
        
        hp = h_best.copy()
        hp[start:start + block_size] += amp
        hp = project_feasible(hp, n)
        
        alloc = min(10.0, (deadline - time.time() - 10) * 0.3)
        if alloc < 4:
            break
        ht = slsqp_optimize(hp, n, 2000, maxiter=2500, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
    
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    n = n_init
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: SLSQP optimization from initial (30% of time)
    alloc1 = min(250.0, (deadline - t0) * 0.30)
    h, best_c = multi_scale_slsqp(h, n, t0 + alloc1)
    best_h = h.copy()
    
    # Phase 2: Subgradient method (15% of time)
    if time.time() < deadline - 80:
        alloc2 = min(120.0, (deadline - time.time()) * 0.15)
        h2, c2 = subgradient_method(best_h, n, time.time() + alloc2,
                                     iters=50000, lr0=0.002, lr_decay=0.9998)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
        # Follow up with SLSQP
        if time.time() < deadline - 60:
            h3 = slsqp_optimize(best_h, n, 20000, maxiter=4000, ftol=1e-16)
            c3 = compute_c5(h3, n)
            if c3 < best_c:
                best_c = c3
                best_h = h3.copy()
    
    # Phase 3: Frequency perturbations (15% of time)
    if time.time() < deadline - 80:
        alloc3 = min(100.0, (deadline - time.time()) * 0.15)
        h4, c4 = frequency_perturb_optimize(best_h, n, time.time() + alloc3, rng)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
    
    # Phase 4: Block perturbations (10% of time)
    if time.time() < deadline - 60:
        alloc4 = min(70.0, (deadline - time.time()) * 0.10)
        h5, c5 = block_perturb_optimize(best_h, n, time.time() + alloc4, rng)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
    
    # Phase 5: Random perturbations (10% of time)
    if time.time() < deadline - 50:
        alloc5 = min(70.0, (deadline - time.time()) * 0.10)
        h6, c6 = random_perturb_optimize(best_h, n, time.time() + alloc5, rng, n_trials=12)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
    
    # Phase 6: Final SLSQP polish (15% of time)
    if time.time() < deadline - 30:
        alloc6 = min(100.0, (deadline - time.time()) * 0.15)
        h7, c7 = multi_scale_slsqp(best_h, n, time.time() + alloc6)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
    
    # Phase 7: Very fine coordinate descent (10% of time)
    if time.time() < deadline - 20:
        alloc7 = min(60.0, (deadline - time.time()) * 0.10)
        h8, c8 = coord_descent_v2(best_h, n, time.time() + alloc7, max_rounds=50)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()
    
    # Phase 8: Final high-precision SLSQP
    if time.time() < deadline - 8:
        h9 = slsqp_optimize(best_h, n, 50000000, maxiter=10000, ftol=1e-18)
        c9 = compute_c5(h9, n)
        if c9 < best_c:
            best_c = c9
            best_h = h9.copy()
    
    # Final cleanup
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    # Verify constraints
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        best_h = project_feasible(best_h, n)
    
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
