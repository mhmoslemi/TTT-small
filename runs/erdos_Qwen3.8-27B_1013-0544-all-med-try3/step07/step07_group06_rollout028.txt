def compute_F_counts(h, n):
    h = np.asarray(h, dtype=float)
    L = 2*n - 1
    if n == 1:
       total = float(h.sum())
       return np.array([total - h[0]*h[0]])
    N = 1
    while N < 2*n:
       N <<= 1
    f = np.fft.rfft(h, N)
    A = np.fft.irfft(f * f.conj(), N)[:n]
    A = np.ascontiguousarray(A, dtype=float)

    return out

def lse_obj_grad(h, alpha, n, dx):
    F = compute_F_counts(h, n)
    C = F * dx
    shift = float(C.max())
    g = alpha * (C - shift)
    gmax = float(g.max())
    e = np.exp(g - gmax)
    s = float(e.sum())
    obj = shift + (gmax + np.log(s)) / alpha
    total_w = s
    w = e / total_w
    u = n - 1
    w0 = w[u]
    if n == 1:
       grad_counts = np.array([w0 * (1 - 2*h[0])])
       return obj, grad_counts * dx, shift
    pos = w[u+1:]
    neg = w[:u][::-1]
    pos_cum = np.empty(n, dtype=float)
    pos_cum[0] = 0.0
    if n > 1:
        np.cumsum(pos, out=pos_cum[1:])
    neg_cum = np.empty(n, dtype=float)
    neg_cum[0] = 0.0
    np.cumsum(neg, out=neg_cum[1:])
    k = np.arange(n)
    Lterm = w0 + pos_cum[n - 1 - k] + neg_cum[k]
    V = np.zeros(2*n - 1, dtype=float)
    # for r 1..n-1

import numpy as np
import time
from scipy.optimize import minimize


_SLSQP_ALPHA = 1.0
_SLSQP_N = 1
_SLSQP_DX = 1.0
_SLSQP_TARGET = 0.0
_SLSQP_CACHE_VALID = False
_SLSQP_CACHE_X = None
_SLSQP_CACHE_OBJ = 0.0

def evaluate_c5(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_box_mean(x, target):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return x
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(55):
        nu = 0.5 * (lo + hi)
        y = np.clip(x - nu, 0.0, 1.0)
        s = float(y.sum())
        if s > target:
            hi = nu
        else:
            lo = nu
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    err = float(target - y.sum())
    if abs(err) > 1e-13:
        mask = (y > 0.0) & (y < 1.0)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += err / float(cnt)
        else:
            if err > 0:
                idx = np.nonzero(y < 1.0)[0]
                if idx.size:
                    y[idx[0]] += min(1.0 - y[idx[0]], err)
            else:
                idx = np.nonzero(y > 0.0)[0]
                if idx.size:
                    y[idx[0]] += min(-y[idx[0]], err)
    y = np.clip(y, 0.0, 1.0)
    return y


def compute_F_counts(h, n):
    h = np.asarray(h, dtype=float)
    if n == 1:
        total = float(h.sum())
        return np.array([total - h[0] * h[0]], dtype=float)

    N = 1
    while N < 2 * n:
        N <<= 1
    f = np.fft.rfft(h, N)
    A = np.fft.irfft(f * f.conj(), N)[:n]
    A = np.ascontiguousarray(A, dtype=float)
    A[np.abs(A) < 1e-15] = 0.0

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]

    L = 2 * n - 1
    u = n - 1
    out = np.empty(L, dtype=float)

    r = np.arange(n)
    out[u + r] = pref[n - r] - A
    if n > 1:
        rp = np.arange(1, n)
        out[u - rp] = (total - pref[rp]) - A[rp]
    return out


def lse_obj_grad(h, alpha, n, dx):
    h = np.asarray(h, dtype=float)
    F = compute_F_counts(h, n)
    C = F * dx
    shift = float(np.max(C))

    g = alpha * (C - shift)
    gmax = float(np.max(g))
    e = np.exp(g - gmax)
    s = float(e.sum())
    obj = shift + (gmax + float(np.log(s))) / float(alpha)

    w = e / s
    u = n - 1

    if n == 1:
        grad_counts = np.array([w[0] * (1.0 - 2.0 * h[0])], dtype=float)
        return obj, grad_counts * dx, shift

    pos = w[u + 1:]
    neg = w[:u][::-1]

    pos_cum = np.empty(n, dtype=float)
    pos_cum[0] = 0.0
    np.cumsum(pos, out=pos_cum[1:])

    neg_cum = np.empty(n, dtype=float)
    neg_cum[0] = 0.0
    np.cumsum(neg, out=neg_cum[1:])

    k = np.arange(n)
    Lterm = w[u] + pos_cum[n - 1 - k] + neg_cum[k]

    wr = w[u + 1:] + w[:u][::-1]
    V = np.zeros(2 * n - 1, dtype=float)
    V[u + 1:] = wr
    V[:u] = wr[::-1]

    need = 3 * n - 2
    Nc = 1
    while Nc < need:
        Nc <<= 1
    Hf = np.fft.rfft(h, Nc)
    Vf = np.fft.rfft(V, Nc)
    conv = np.fft.irfft(Hf * Vf, Nc)
    cross = conv[u:u + n]

    grad_counts = Lterm - 2.0 * w[u] * h - cross
    return obj, grad_counts * dx, shift


def continuous_adam(h, n, deadline, alpha_start, alpha_end, lr=0.02, seed=42):
    h = np.asarray(h, dtype=float)
    if n <= 0:
        return h, np.inf
    if time.time() > deadline - 1.0:
        return h, evaluate_c5(h, n)

    target = 0.5 * float(n)
    h = project_box_mean(h, target)
    best_h = h.copy()
    best_c = evaluate_c5(h, n)
    dx = 2.0 / float(n)

    mu = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)
    rng = np.random.RandomState(seed)
    alpha = float(alpha_start)
    lr_cur = float(lr)
    it = 0
    no_obj = 0
    no_c = 0

    while time.time() < deadline - 0.5:
        it += 1
        current_obj, current_grad, current_cmax = lse_obj_grad(h, alpha, n, dx)
        if current_cmax < best_c - 1e-12:
            best_c = current_cmax
            best_h = h.copy()

        b1c = 0.9
        b2c = 0.999
        mu = b1c * mu + (1.0 - b1c) * current_grad
        v = b2c * v + (1.0 - b2c) * (current_grad * current_grad)
        b1 = 1.0 - (b1c ** it)
        b2 = 1.0 - (b2c ** it)

        lr_eff = lr_cur * max(0.2, 1.0 - 0.5 * min(1.0, float(it) / 30000.0))
        step = lr_eff * (mu / b1) / (np.sqrt(v / b2) + 1e-8)
        snorm = float(np.linalg.norm(step))
        if snorm > 0.25 and snorm > 0.0:
            step *= 0.25 / snorm

        cand = project_box_mean(h - step, target)
        cand_obj, cand_grad, cand_cmax = lse_obj_grad(cand, alpha, n, dx)

        accept = False
        if cand_obj < current_obj - 1e-13:
            accept = (cand_cmax <= current_cmax + 0.003)
        if cand_cmax < current_cmax - 1e-10:
            accept = True

        if not accept:
            small = project_box_mean(h - 0.5 * step, target)
            small_obj, small_grad, small_cmax = lse_obj_grad(small, alpha, n, dx)
            if small_obj < current_obj - 1e-13 or small_cmax < current_cmax - 1e-10:
                accept = True
                cand = small
                cand_cmax = small_cmax

        if accept:
            h = cand
            no_obj = 0
            if cand_cmax < best_c - 1e-12:
                best_c = cand_cmax
                best_h = h.copy()
                no_c = 0
            else:
                no_c += 1
        else:
            no_obj += 1
            no_c += 1
            if no_obj > 90:
                lr_cur = max(1e-4, 0.65 * lr_cur)
                no_obj = 0
            if no_c > 700:
                if alpha < alpha_end:
                    alpha = min(float(alpha_end), alpha * 1.35)
                h = project_box_mean(h + 0.05 * rng.randn(n), target)
                mu.fill(0.0)
                v.fill(0.0)
                no_c = 0
                no_obj = 0

        if it % 50 == 0 and alpha < alpha_end:
            growth = (float(alpha_end) / max(alpha, 1e-9)) ** (50.0 / 10000.0)
            alpha = min(float(alpha_end), alpha * growth)

    h = project_box_mean(h, target)
    c_final = evaluate_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, float(best_c)


def interp_to_n(h, n_old, n_new):
    h = np.asarray(h, dtype=float)
    if n_old == n_new:
        return h.copy()
    dx_old = 2.0 / float(n_old)
    dx_new = 2.0 / float(n_new)
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    y = np.interp(x_new, x_old, h)
    return y


def binarize_preserve(h, n):
    target = int(n // 2) if n % 2 == 0 else int(round(0.5 * n))
    b = (np.asarray(h, dtype=float) >= 0.5).astype(float)
    diff = int(round(float(target - int(b.sum()))))
    if diff > 0:
        order = np.argsort(np.asarray(h, dtype=float))[::-1]
        zeros = np.nonzero(b == 0.0)[0]
        if zeros.size:
            pick = zeros[:diff]
            if pick.size < diff:
                order2 = order[b == 0.0]
                pick = order2[:diff]
            b[pick] = 1.0
    elif diff < 0:
        order = np.argsort(np.asarray(h, dtype=float))
        ones = np.nonzero(b == 1.0)[0]
        if ones.size:
            pick = ones[(-diff)::-1]
            if pick.size < -diff:
                order2 = order[b == 1.0]
                pick = order2[-diff:]
            b[pick] = 0.0
    return b


def perturb_binary(h, n, k, seed):
    h = np.rint(np.asarray(h, dtype=float)).astype(float).copy()
    h = binarize_preserve(h, n)
    rng = np.random.RandomState(seed)
    for _ in range(max(0, int(k))):
        ones = np.nonzero(h == 1.0)[0]
        zeros = np.nonzero(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
    return h


def flip_deltas_full(h, n):
    h = np.asarray(h, dtype=float)
    L = 2 * n - 1
    u = n - 1
    src = np.zeros((n, L), dtype=float)
    tgt = np.zeros((n, L), dtype=float)

    for idx in range(L):
        m = idx - u
        if m >= 0:
            cnt = n - m
            if cnt > 0:
                src[:cnt, idx] = 1.0 - h[m:n]
                tgt[m:, idx] = h[:cnt]
        else:
            r = -m
            cnt = n - r
            if cnt > 0:
                src[r:, idx] = 1.0 - h[:cnt]
                tgt[:cnt, idx] = h[r:]

    corr = np.zeros((n, L), dtype=float)
    corr[:, u] = -1.0
    F0_all = src - tgt + corr
    F1_all = -src + tgt + corr
    return F0_all, F1_all


def binary_global_search(h, n, deadline, max_steps=None, seed=42):
    h = binarize_preserve(np.asarray(h, dtype=float), n)
    u = n - 1
    L = 2 * n - 1
    dx = 2.0 / float(n)

    base = np.rint(compute_F_counts(h, n))
    best_h = h.copy()
    best_count = float(np.max(base))

    if max_steps is None:
        if n <= 250:
            max_steps = 350
        elif n <= 350:
            max_steps = 180
        elif n <= 500:
            max_steps = 90
        else:
            max_steps = 40

    step = 0
    while step < max_steps and time.time() < deadline - 1.0:
        Z = np.nonzero(h == 0.0)[0]
        O = np.nonzero(h == 1.0)[0]
        if Z.size == 0 or O.size == 0:
            break

        F0_all, F1_all = flip_deltas_full(h, n)
        Zc = int(Z.size)
        Oc = int(O.size)
        F0 = np.ascontiguousarray(F0_all[Z])
        F1 = np.ascontiguousarray(F1_all[O])

        maxmem = 2000000
        B = max(1, min(L, maxmem // max(1, Zc * Oc)))
        T = np.full((Zc, Oc), -1e100, dtype=float)

        for s in range(0, L, B):
            e = min(L, s + B)
            F0b = F0[:, s:e]
            F1b = F1[:, s:e]
            block = F0b[:, None, :] + F1b[None, :, :] + base[s:e][None, None, :]
            bmax = block.max(axis=2)
            T = np.maximum(T, bmax)

        zcol = np.arange(Zc)[:, None]
        orow = np.arange(Oc)[None, :]
        d = O[orow] - Z[zcol]

        idx1 = u + d
        idx2 = u - d

        F0g1 = F0[zcol, idx1]
        F1g1 = F1[orow, idx1.T].T
        X1 = base[idx1] + F0g1 + F1g1 + 1.0

        F0g2 = F0[zcol, idx2]
        F1g2 = F1[orow, idx2.T].T
        X2 = base[idx2] + F0g2 + F1g2 + 1.0

        vals = np.maximum(np.maximum(T, X1), X2)
        pos = int(np.argmin(vals))
        pi, pj = np.unravel_index(pos, vals.shape)
        new_count = float(vals[pi, pj])
        cur_count = float(np.max(base))

        if new_count < cur_count - 1e-7:
            zi = int(Z[pi])
            oj = int(O[pj])
            h[zi] = 1.0
            h[oj] = 0.0
            base = np.rint(compute_F_counts(h, n))
            if float(np.max(base)) < best_count - 1e-7:
                best_count = float(np.max(base))
                best_h = h.copy()
            step += 1
        else:
            break

    c = float(best_count) * dx
    cf = evaluate_c5(best_h, n)
    if cf < c + 1e-12:
        c = cf
    return best_h, float(c)


def binary_anneal(h, n, deadline, seed=42, temp_start=3.0, temp_end=0.12):
    h = binarize_preserve(np.asarray(h, dtype=float), n)
    u = n - 1
    dx = 2.0 / float(n)
    base = np.rint(compute_F_counts(h, n))
    cur_count = float(np.max(base))
    best_h = h.copy()
    best_count = cur_count
    start = time.time()
    rng = np.random.RandomState(seed)

    F0_all, F1_all = flip_deltas_full(h, n)
    Z = np.nonzero(h == 0.0)[0]
    O = np.nonzero(h == 1.0)[0]

    while time.time() < deadline - 1.0 and Z.size and O.size:
        i = int(rng.choice(Z))
        j = int(rng.choice(O))

        tmp = base + F0_all[i, :] + F1_all[j, :]
        new_count = float(np.max(tmp))
        d = j - i
        idx1 = u + d
        idx2 = u - d
        v1 = float(tmp[idx1] + 1.0)
        if v1 > new_count:
            new_count = v1
        if idx2 != idx1:
            v2 = float(tmp[idx2] + 1.0)
            if v2 > new_count:
                new_count = v2

        dc = new_count - cur_count
        if dc <= 1e-7:
            accept = True
        else:
            elapsed_frac = max(0.0, min(1.0, (time.time() - start) / max(1e-9, deadline - start)))
            temp = float(temp_end + (temp_start - temp_end) * (1.0 - elapsed_frac) ** 2)
            accept = temp > 0.0 and bool(rng.random() < np.exp(-dc / max(1e-6, temp)))

        if accept:
            h[i] = 1.0
            h[j] = 0.0
            base = np.rint(compute_F_counts(h, n))
            cur_count = float(np.max(base))
            F0_all, F1_all = flip_deltas_full(h, n)
            Z = np.nonzero(h == 0.0)[0]
            O = np.nonzero(h == 1.0)[0]
            if cur_count < best_count - 1e-7:
                best_count = cur_count
                best_h = h.copy()

    c = float(best_count) * dx
    cf = evaluate_c5(best_h, n)
    if cf < c + 1e-12:
        c = cf
    return best_h, float(c)


def slsqp_set_params(alpha, n, dx, target):
    global _SLSQP_ALPHA, _SLSQP_N, _SLSQP_DX, _SLSQP_TARGET, _SLSQP_CACHE_VALID
    _SLSQP_ALPHA = float(alpha)
    _SLSQP_N = int(n)
    _SLSQP_DX = float(dx)
    _SLSQP_TARGET = float(target)
    _SLSQP_CACHE_VALID = False


def slsqp_obj(x):
    global _SLSQP_CACHE_VALID, _SLSQP_CACHE_X, _SLSQP_CACHE_OBJ, _SLSQP_CACHE_GRAD
    if (not _SLSQP_CACHE_VALID) or (not np.array_equal(x, _SLSQP_CACHE_X)):
        obj, grad, _ = lse_obj_grad(x, _SLSQP_ALPHA, _SLSQP_N, _SLSQP_DX)
        _SLSQP_CACHE_X = np.asarray(x, dtype=float).copy()
        _SLSQP_CACHE_OBJ = float(obj)
        _SLSQP_CACHE_GRAD = np.asarray(grad, dtype=float).copy()
        _SLSQP_CACHE_VALID = True
    return _SLSQP_CACHE_OBJ


def slsqp_jac(x):
    global _SLSQP_CACHE_VALID, _SLSQP_CACHE_X, _SLSQP_CACHE_OBJ, _SLSQP_CACHE_GRAD
    if (not _SLSQP_CACHE_VALID) or (not np.array_equal(x, _SLSQP_CACHE_X)):
        obj, grad, _ = lse_obj_grad(x, _SLSQP_ALPHA, _SLSQP_N, _SLSQP_DX)
        _SLSQP_CACHE_X = np.asarray(x, dtype=float).copy()
        _SLSQP_CACHE_OBJ = float(obj)
        _SLSQP_CACHE_GRAD = np.asarray(grad, dtype=float).copy()
        _SLSQP_CACHE_VALID = True
    return _SLSQP_CACHE_GRAD


def slsqp_con(x):
    return float(np.sum(x) - _SLSQP_TARGET)


def slsqp_refine(h, n, deadline, alpha, maxiter=200):
    h = np.asarray(h, dtype=float)
    if time.time() > deadline - 2.0:
        return h, evaluate_c5(h, n)
    target = 0.5 * float(n)
    h = project_box_mean(h, target)
    dx = 2.0 / float(n)
    c0 = evaluate_c5(h, n)

    slsqp_set_params(alpha, n, dx, target)
    bounds = [(0.0, 1.0)] * int(n)
    cons = {"type": "eq", "fun": slsqp_con}
    try:
        res = minimize(
            slsqp_obj,
            h,
            jac=slsqp_jac,
            bounds=bounds,
            constraints=cons,
            method="SLSQP",
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False},
        )
        h2 = project_box_mean(res.x, target)
        c2 = evaluate_c5(h2, n)
        if c2 < c0:
            return h2, float(c2)
        return h, float(c0)
    except Exception:
        return h, float(c0)


def make_initial_h(n, seed):
    rng = np.random.RandomState(seed)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h1 = 0.5 + 0.48 * np.cos(np.pi * x)
    h2 = 0.5 + 0.25 * np.cos(3.0 * np.pi * x + 0.6) + 0.18 * np.sin(5.0 * np.pi * x)
    h = 0.55 * h1 + 0.45 * h2
    h = np.clip(h, 0.0, 1.0)
    h = project_box_mean(h, 0.5 * n)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + max(20.0, float(budget_s)) - 5.0

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 4:
            raise Exception("bad parent")
    except Exception:
        n0 = 250
        h0 = make_initial_h(n0, int(seed))

    target0 = 0.5 * float(n0)
    h0 = project_box_mean(h0, target0)
    best_n = int(n0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h, best_n)
    rng = np.random.RandomState(int(seed))

    T = max(25.0, float(budget_s) - 5.0)

    pA = t0 + 0.18 * T
    if best_n <= 500 and pA > time.time() + 8.0:
        hh, cc = continuous_adam(best_h, best_n, pA, 500.0, 50000.0, 0.020, int(seed) + 11)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
            best_n = int(best_n)

    pB = pA + 0.10 * T
    bin_n = best_n if best_n % 2 == 0 else best_n + 1
    if bin_n > 500:
        bin_n = 500
    if bin_n <= 500 and pB > time.time() + 12.0:
        base_bin = best_h.copy() if best_n == bin_n else interp_to_n(best_h, best_n, bin_n)
        base_bin = project_box_mean(base_bin, 0.5 * float(bin_n))
        hb, cb = binary_global_search(base_bin, bin_n, pB, max_steps=None, seed=int(seed) + 20)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            best_n = int(bin_n)
        if time.time() < pB - 20.0:
            ha, ca = binary_anneal(hb, bin_n, pB, seed=int(seed) + 21, temp_start=2.5, temp_end=0.10)
            if ca < best_c:
                best_c = ca
                best_h = ha.copy()
                best_n = int(bin_n)
            elif ca < cb:
                hb = ha.copy()
                cb = ca
        if time.time() < pB - 25.0:
            hp = perturb_binary(hb, bin_n, int(rng.randint(3, 9)), int(seed) + 22)
            hb2, cb2 = binary_global_search(hp, bin_n, pB, max_steps=60, seed=int(seed) + 23)
            if cb2 < cb:
                hb = hb2.copy()
                cb = cb2
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                best_n = int(bin_n)

    pC = pB + 0.18 * T
    if best_n <= 500 and pC > time.time() + 12.0:
        start = best_h.copy()
        start = project_box_mean(start + 0.02 * rng.randn(best_n), 0.5 * float(best_n))
        hh, cc = continuous_adam(start, best_n, pC, 5000.0, 200000.0, 0.012, int(seed) + 31)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
            best_n = int(best_n)

    pD = pC + 0.25 * T
    alt_candidates = []
    for cand in (300, 350, 500, 450):
        if cand != best_n and cand % 2 == 0 and cand <= 500:
            alt_candidates.append(cand)
        if len(alt_candidates) >= 2:
            break

    if alt_candidates and pD > time.time() + 40.0:
        alts = alt_candidates[:2]
        remain_to_D = max(0.0, pD - time.time())
        for ai, an in enumerate(alts):
            shrink = len(alts) - 1 - ai
            share = (remain_to_D / float(max(1, ai + 1)))
            seg_end = time.time() + share
            if seg_end > pD:
                seg_end = pD
            if seg_end - time.time() < 30.0:
                if pD - time.time() > 40.0:
                    seg_end = min(pD, time.time() + 40.0)
                else:
                    seg_end = time.time()
            if seg_end <= time.time() + 6.0:
                continue

            hh = interp_to_n(best_h, best_n, an)
            hh = project_box_mean(hh, 0.5 * float(an))
            tbin = time.time() + 0.35 * (seg_end - time.time())
            if tbin > seg_end - 20.0:
                tbin = seg_end - 20.0
            if tbin > time.time() + 8.0:
                hb, cb = binary_global_search(hh, an, tbin, max_steps=60, seed=int(seed) + 40 + an)
                if cb < best_c:
                    best_c = cb
                    best_h = hb.copy()
                    best_n = int(an)
                start = hb
            else:
                start = hh
            start = project_box_mean(start + 0.015 * rng.randn(an), 0.5 * float(an))
            if seg_end > time.time() + 8.0:
                hh2, cc = continuous_adam(start, an, seg_end, 3000.0, 120000.0, 0.010, int(seed) + 50 + an)
                if cc < best_c:
                    best_c = cc
                    best_h = hh2.copy()
                    best_n = int(an)

    rem = max(0.0, hard_deadline - time.time())
    if rem > 80.0 and best_n <= 350:
        t1 = min(hard_deadline - 25.0, time.time() + min(60.0, 0.20 * rem))
        if t1 > time.time() + 10.0:
            hh, cc = slsqp_refine(best_h, best_n, t1, 80000.0, 220)
            if cc < best_c:
                best_c = cc
                best_h = hh.copy()
                best_n = int(best_n)

    if hard_deadline - time.time() > 20.0:
        start = best_h.copy()
        if best_n <= 400:
            start = project_box_mean(start + 0.01 * rng.randn(best_n), 0.5 * float(best_n))
        else:
            start = project_box_mean(start, 0.5 * float(best_n))
        hh, cc = continuous_adam(
            start,
            best_n,
            hard_deadline - 3.0,
            50000.0,
            600000.0,
            0.008,
            int(seed) + 90,
        )
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
            best_n = int(best_n)

    rem = max(0.0, hard_deadline - time.time())
    if rem > 45.0 and best_n <= 300:
        t1 = min(hard_deadline - 3.0, time.time() + min(40.0, 0.5 * rem))
        if t1 > time.time() + 8.0:
            hh, cc = slsqp_refine(best_h, best_n, t1, 400000.0, 160)
            if cc < best_c:
                best_c = cc
                best_h = hh.copy()
                best_n = int(best_n)

    best_h = project_box_mean(best_h, 0.5 * float(best_n))
    rr = np.rint(best_h)
    if np.max(np.abs(best_h - rr)) < 1e-7 and abs(float(rr.sum()) - 0.5 * float(best_n)) < 1e-7:
        best_h = rr.astype(float)

    best_c = evaluate_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
