import numpy as np
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binned_signed_mag_smooth(h, n, alpha):
    # Smooth max of all lag values
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def top_lags_smooth(h, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-15, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    obj_func = top_lags_smooth if top_idx is not None else binned_signed_mag_smooth
    obj_args = (n, top_idx, alpha) if top_idx is not None else (n, alpha)
    try:
        res = minimize(obj_func, h, args=obj_args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _compute_lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1

        if time.time() > deadline - 3:

        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-8:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

def block_exchange_local_search(h, n, deadline):
    """Local search using block exchanges: swap blocks of consecutive positions."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    block_sizes = [1, 2, 3, 5, 10, 20]
    
    for bs in block_sizes:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0:
                break
            
            # Try block exchanges
            for i in range(0, n - m - bs + 1, max(1, n // 200)):
                if time.time() > deadline:
                    break
                # Try shifting block [i, i+bs) to position j
                for j in range(0, n - bs + 1, max(1, n // 100)):
                    if time.time() > deadline:
                        break
                    if abs(i - j) <= bs:
                        continue
                    # Check if blocks don't overlap
                    if i < j + bs and i + bs > j:
                        continue
                    # Swap
                    block_i = h[i:i+bs].copy()
                    block_j = h[j:j+bs].copy()
                    h[i:i+bs] = block_j
                    h[j:j+bs] = block_i
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i:i+bs] = block_i
                        h[j:j+bs] = block_j
                if improved:
                    break
    return best_h, best_c

def coordinate_descent(h, n, deadline):
    """Coordinate descent: try to improve one coordinate at a time."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for _ in range(3):  # Passes
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0:
                break
            
            # For the worst lag, try to adjust h[i] and h[i+m]
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                # Try to decrease the contribution at this lag
                # c_k = sum h[i]*(1-h[i+k]) for positive lag
                # To decrease: decrease h[i] or increase h[i+k]
                if h[i] > h[j]:
                    # Decrease h[i], increase h[j] (maintain sum)
                    delta = min(0.01, (h[i] - h[j]) * 0.5)
                    if delta < 1e-8:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(0.01, (h[j] - h[i]) * 0.5)
                    if delta < 1e-8:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return best_h, best_c

def generate_quasi_binary(n, rng, base_h=None):
    """Generate a quasi-binary candidate (values near 0 or 1)."""
    if base_h is not None:
        h = np.round(base_h)
        # Repair to have correct sum
        n_ones = int(np.sum(h == 1))
        target = n // 2
        if n_ones > target:
            # Remove some ones (those with lowest base_h value)
            one_idx = np.where(h == 1)[0]
            order = np.argsort(base_h[one_idx])
            for idx in order[:n_ones - target]:
                h[one_idx[idx]] = 0
        elif n_ones < target:
            zero_idx = np.where(h == 0)[0]
            order = np.argsort(-base_h[zero_idx])
            for idx in order[:target - n_ones]:
                h[zero_idx[idx]] = 1
        # Add small perturbation
        h += rng.normal(0, 0.05, n)
    else:
        h = generate_binary_candidate(n, rng)
        h += rng.normal(0, 0.05, n)
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def top_lags_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-14, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        obj_func = top_lags_obj
        obj_args = (n, top_idx, alpha)
    else:
        obj_func = smooth_max_obj
        obj_args = (n, alpha)
    try:
        res = minimize(obj_func, h, args=obj_args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _compute_lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

                    continue
                if h[i] > h[j] + 1e-6:
                    delta = min(d, (h[i] - h[j]) * 0.4)
                    if delta < 1e-8:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        tabu.append(key)
                        if len(tabu) > tabu_size:
                            tabu.pop(0)
                        no_improve = 0
                        found = True
                        break
                    else:
                        h[i] += delta
                        h[j] -= delta
                        no_improve += 1
                elif h[j] > h[i] + 1e-6:
                    delta = min(d, (h[j] - h[i]) * 0.4)
                    if delta < 1e-8:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        tabu.append(key)
                        if len(tabu) > tabu_size:
                            tabu.pop(0)
                        no_improve = 0
                        found = True
                        break
                    else:
                        h[i] -= delta
                        h[j] += delta
                        no_improve += 1
            if not found:
                no_improve += 1
    return best_h, best_c

def block_perturb(h, n, rng, n_blocks=3, block_len=5):
    h = h.copy()
    for _ in range(n_blocks):
        i = rng.randint(0, n - block_len)
        j = rng.randint(0, n - block_len)
        if abs(i - j) < block_len * 2:
            continue
        if i < j + block_len and i + block_len > j:
            continue
        bi = h[i:i+block_len].copy()
        bj = h[j:j+block_len].copy()
        h[i:i+block_len] = bj
        h[j:j+block_len] = bi
    return project_feasible(h, n)

def multi_start_refine(h_start, n, deadline, rng, n_starts=4):
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    
    initial_alloc = min(60, (deadline - time.time()) * 0.3)
    h_opt, c_opt = _single_refine(h_start, n, time.time() + initial_alloc)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    
    for s in range(n_starts):
        if time.time() > deadline - 30:
            break
        h_try = block_perturb(best_h, n, rng, n_blocks=2+s, block_len=3+s*2)
        h_try += rng.normal(0, 0.03*(1+s*0.5), n)
        h_try = project_feasible(h_try, n)
        alloc = min(50, (deadline - time.time() - 20) * 0.25)
        if alloc < 20:
            break
        h_opt, c_opt = _single_refine(h_try, n, time.time() + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    
    return best_h, best_c

def _single_refine(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    if time.time() < deadline - 15:
        h_new = slsqp_optimize(best_h, n, alpha=5000, maxiter=2000, ftol=1e-12)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    
    if time.time() < deadline - 20:
        oo_dl = min(deadline - 15, time.time() + 30)
        h_new, c_new = outer_approx(best_h, n, oo_dl, max_iters=25)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    
    if time.time() < deadline - 15:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-8:]
        h_new = slsqp_optimize(best_h, n, alpha=30000, maxiter=4000, ftol=1e-13, top_idx=top_idx)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-6:]
        h_new = slsqp_optimize(best_h, n, alpha=100000, maxiter=4000, ftol=1e-14, top_idx=top_idx)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    
    if time.time() < deadline - 5:
        h_new, c_new = tabu_local_search(best_h, n, deadline - 2, tabu_size=30)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    
    return project_feasible(best_h, n), best_c

def generate_binary_start(n, rng):
    pos = rng.choice(n, size=n//2, replace=False)
    h = np.zeros(n)
    h[pos] = 1.0
    h += rng.normal(0, 0.08, n)
    return project_feasible(h, n)

def generate_structured_start(n, rng, base=None):
    if base is not None:
        h = base.copy()
        h = np.where(h > 0.5, 1.0, 0.0)
        n_ones = int(np.sum(h))
        target = n // 2
        if n_ones > target:
            one_idx = np.where(h == 1)[0]
            order = np.argsort(base[one_idx])
            for idx in order[:n_ones - target]:
                h[one_idx[idx]] = 0
        elif n_ones < target:
            zero_idx = np.where(h == 0)[0]
            order = np.argsort(-base[zero_idx])
            for idx in order[:target - n_ones]:
                h[zero_idx[idx]] = 1
        h += rng.normal(0, 0.06, n)
    else:
        h = generate_binary_start(n, rng)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize from parent
    if h_init is not None:
        alloc = min(200, (deadline - t0) * 0.25)
        h_opt, c_opt = _single_refine(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    
    n = len(best_h) if best_h is not None else 300
    
    # Phase 2: Multi-start with diverse initializations at same n
    if time.time() < deadline - 100:
        # Try structured (near-binary) starts
        for s in range(3):
            if time.time() > deadline - 80:
                break
            if h_init is not None:
                h_start = generate_structured_start(n, rng, base=h_init)
            else:
                h_start = generate_structured_start(n, rng)
            h_start += rng.normal(0, 0.04*(1+s), n)
            h_start = project_feasible(h_start, n)
            alloc = min(55, (deadline - time.time() - 60) * 0.3)
            if alloc < 25:
                break
            h_opt, c_opt = _single_refine(h_start, n, time.time() + alloc)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
        
        # Pure binary starts
        for s in range(2):
            if time.time() > deadline - 60:
                break
            h_start = generate_binary_start(n, rng)
            h_start += rng.normal(0, 0.04, n)
            h_start = project_feasible(h_start, n)
            alloc = min(50, (deadline - time.time() - 50) * 0.3)
            if alloc < 20:
                break
            h_opt, c_opt = _single_refine(h_start, n, time.time() + alloc)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    
    # Phase 3: Multi-start refinement from best
    if best_h is not None and time.time() < deadline - 80:
        h_opt, c_opt = multi_start_refine(best_h, n, time.time() + min(150, (deadline - time.time())*0.4), rng, n_starts=3)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    
    # Phase 4: Try different resolutions
    n_candidates = [250, 350, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 70:
            break
        if n_try == n:
            continue
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        alloc = min(55, (deadline - time.time() - 50) * 0.25)
        if alloc < 25:
            break
        h_opt, c_opt = _single_refine(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    
    # Phase 5: Final intensive refinement
    if best_h is not None and time.time() < deadline - 20:
        n_best = len(best_h)
        
        # Progressive alpha increase on active lags
        for alpha in [50000, 100000, 300000, 500000]:
            if time.time() > deadline - 10:
                break
            c = compute_all_lags(best_h, n_best)
            top_idx = np.argsort(c)[-10:]
            h_new = slsqp_optimize(best_h, n_best, alpha=alpha, maxiter=5000, ftol=1e-15, top_idx=top_idx)
            c_new = compute_c5(h_new, n_best)
            if c_new < best_c - 1e-15:
                best_c, best_h = c_new, h_new
            c = compute_all_lags(best_h, n_best)
            top_idx = np.argsort(c)[-5:]
            h_new = slsqp_optimize(best_h, n_best, alpha=alpha*2, maxiter=3000, ftol=1e-15, top_idx=top_idx)
            c_new = compute_c5(h_new, n_best)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
        
        # Tabu local search
        if time.time() < deadline - 5:
            h_new, c_new = tabu_local_search(best_h, n_best, deadline - 3, tabu_size=50)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
        
        # Final SLSQP polish
        if time.time() < deadline - 3:
            c = compute_all_lags(best_h, n_best)
            top_idx = np.argsort(c)[-4:]
            h_new = slsqp_optimize(best_h, n_best, alpha=1000000, maxiter=5000, ftol=1e-16, top_idx=top_idx)
            c_new = compute_c5(h_new, n_best)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
