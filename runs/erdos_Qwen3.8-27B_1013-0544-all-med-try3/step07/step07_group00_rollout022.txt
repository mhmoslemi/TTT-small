    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:k+nv])
                grad[k:k+nv] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:kk+nv] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:kk+nv])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def find_active(c_vec, n, fraction=0.01, max_count=20):
    """Find shifts that are within 'fraction' of the max."""
    m = c_vec.max()
    threshold = m - max(1e-8, fraction * m)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > max_count:
        order = np.argsort(-c_vec[active])
        active = active[order[:max_count]]
    return active, m

def adam_subgrad(h, n, deadline, lr=0.01, beta1=0.9, beta2=0.99, eps=1e-8, active_frac=0.01):
    """Adam-style subgradient method for the max function."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        m = c.max()
        active, _ = find_active(c, n, active_frac, 15)
        
        # Subgradient: average of gradients of active shifts
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += (1.0 - h[k:k+nv])
                    grad[k:k+nv] += -h[0:nv]
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:kk+nv] += (1.0 - h[0:nv])
                    grad[0:nv] += -h[kk:kk+nv]
        grad *= dx / len(active)
        
        t += 1
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * grad ** 2
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Projected step
        h_new = h - step
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # Accept if it's not too much worse (annealing)
        if c_new < best_c + 1e-4:
            h = h_new
        else:
            h = h  # reject
            lr *= 0.95
        
        if time.time() > deadline:
            break
    
    return best_h, best_c

def pair_search(h, n, deadline, radius=40, steps=None):

                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def project_feasible(h, n):
