import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def lse_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)
    s2 = w.sum()
    if s2 < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return w / s2

def grad_weighted(h, n, w):
    dx = 2.0 / n
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-12:
            continue
        lag = ki - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if lag == 0:
            g += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            g[:n-m] += wi * (1.0 - h[m:])
            g[m:] += -wi * h[:n-m]
        else:
            g[m:] += wi * (1.0 - h[:n-m])
            g[:n-m] += -wi * h[m:]
    return g * dx

def gradient_descent(h, n, deadline, max_alpha=200, start_alpha=20, lr=0.5, momentum=0.92):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    alpha = start_alpha
    lr_curr = lr
    for it in range(200000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        w = lse_weights(c, alpha)
        g = grad_weighted(h, n, w)
        v = momentum * v + lr_curr * g
        h_new = project_feasible(h - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if it % 200 == 199:
            top = np.sort(c)[-5:]
            gap = top[-1] - top[0]
            alpha = min(max_alpha, max(start_alpha, start_alpha / (1 + gap * 200)))
            if it % 2000 == 1999:
                lr_curr *= 0.9
    return best_h, best_c

def swap_delta_c5(h, n, i, j, d):
    """Compute exact delta in all lags for swap h[i]-=d, h[j]+=d. Returns new max corr * dx."""
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    # Apply swap temporarily
    old_i, old_j = h[i], h[j]
    h[i] -= d
    h[j] += d
    h = np.clip(h, 0, 1)
    c_new = compute_all_lags(h, n)
    # Restore
    h[i] = old_i
    h[j] = old_j
    return float(np.max(c_new))

def slsqq_polish(h, n, deadline, top_m=5, alpha=100000, maxiter=2000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        cc = compute_all_lags(x, n)
        active = cc[top_idx]
        return lse_obj(active, alpha)
    
    def jac(x):
        cc = compute_all_lags(x, n)
        active = cc[top_idx]
        w = lse_weights(active, alpha)
        g = np.zeros(n)
        for mi, ki in enumerate(top_idx):
            if w[mi] < 1e-12:
                continue
            lag = ki - (n - 1)
            m = abs(lag)
            if m >= n:
                continue
            wi = w[mi]
            if lag == 0:
                g += wi * (1.0 - 2.0 * x)
            elif lag > 0:
                g[:n-m] += wi * (1.0 - x[m:])
                g[m:] += -wi * x[:n-m]
            else:
                g[m:] += wi * (1.0 - x[:n-m])
                g[:n-m] += -wi * x[m:]
        return g * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def multi_start_gradient(h0, n, rng, deadline, n_starts=5):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        scale = 0.01 + 0.01 * s
        hp = h0 + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (n_starts - s))
        h_new, c_new = gradient_descent(hp, n, sub_dl, start_alpha=30, lr=0.4, momentum=0.9)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

            best_h = h_sl
        # Second SLSQP with different top lags
        if time.time() < deadline - 40:
            h_sl2 = slsqq_polish(best_h, n_best, deadline - 20, top_m=10, alpha=100000, maxiter=1500)
            c_sl2 = compute_c5(h_sl2, n_best)
            if c_sl2 < best_c:
                best_c = c_sl2
                best_h = h_sl2
    
    # Phase 4: Another round of local search after SLSQP
    if time.time() < deadline - 80:
        phase4_dl = min(deadline - 50, time.time() + 80)
        h_ls2, c_ls2 = binary_local_search(best_h, n_best, phase4_dl, rng)
        if c_ls2 < best_c:
            best_c = c_ls2
            best_h = h_ls2
            n_best = len(best_h)
    
    # Phase 5: Try different resolution
    if time.time() < deadline - 100:
        for n_try in [600, 450]:
            if time.time() > deadline - 60:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            sub_dl = min(deadline - 30, time.time() + 60)
            h_gd2, c_gd2 = gradient_descent(h_interp, n_try, sub_dl, start_alpha=20, max_alpha=80, lr=0.4, momentum=0.9)
            if c_gd2 < best_c:
                best_c = c_gd2
                best_h = h_gd2
                n_best = n_try
                # Quick local search at new resolution
                if time.time() < deadline - 40:
                    h_ls3, c_ls3 = binary_local_search(best_h, n_best, deadline - 20, rng)
                    if c_ls3 < best_c:
                        best_c = c_ls3
                        best_h = h_ls3
                        n_best = len(best_h)
    
    # Phase 6: Multi-start at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 50:
        remaining = deadline - time.time() - 20
        n_starts = max(2, int(remaining / 40))
        for s in range(n_starts):
            if time.time() > deadline - 25:
                break
            scale = 0.005 + 0.005 * s
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            sub_dl = min(deadline - 15, time.time() + min(25, remaining / n_starts))
            h_new, c_new = gradient_descent(hp, n_best, sub_dl, start_alpha=50, lr=0.3, momentum=0.93)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                # Quick local search
                if time.time() < deadline - 10:
                    h_ls4, c_ls4 = binary_local_search(best_h, n_best, deadline - 5, rng)
                    if c_ls4 < best_c:
                        best_c = c_ls4
                        best_h = h_ls4
    
    # Phase 7: Final polish
    n_final = len(best_h)
    if time.time() < deadline - 10:
        h_final = slsqq_polish(best_h, n_final, deadline - 5, top_m=5, alpha=200000, maxiter=1000)
        c_final = compute_c5(h_final, n_final)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    if time.time() < deadline - 2:
        # Very last local search with remaining time
        h_final2, c_final2 = binary_local_search(best_h, n_final, deadline - 0.5, rng)
        if c_final2 < best_c:
            best_c = c_final2
            best_h = h_final2
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
