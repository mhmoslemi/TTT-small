    zeros = np.where(h == 0)[0]

        best_a = -1

        best_new_corr = None

def compute_delta(h, n, a, b):
    """Compute the change in correlation when h[a] goes 0→1 and h[b] goes 1→0."""
    delta = np.zeros(2*n-1)
    # h[a]: was 0, now 1
    # As first factor in c_m: h[a]*(1-h[a+m]) → adds (1-h[a+m]) for valid m
    # valid: 0 <= a+m < n → -a <= m < n-a
    m_lo = max(-a, -(n-1))
    m_hi = min(n-a, n) - 1
    for m in range(m_lo, m_hi+1):
        delta[n-1+m] += (1.0 - h[a+m])
    # As second factor in c_m: (1-h[a-m])*h[a-m]... wait
    # c_m = sum_i h[i]*(1-h[i+m])
    # h[a] appears as (1-h[a]) when i+m = a, i.e., i = a-m
    # term: h[a-m]*(1-h[a]) → was h[a-m]*1, now h[a-m]*0, change: -h[a-m]
    # valid: 0 <= a-m < n → a-n+1 <= m <= a
    m_lo2 = max(a-n+1, -(n-1))
    m_hi2 = min(a, n-1)
    for m in range(m_lo2, m_hi2+1):
        delta[n-1+m] -= h[a-m]
    
    # h[b]: was 1, now 0
    # As first factor: h[b]*(1-h[b+m]) → was (1-h[b+m]), now 0, change: -(1-h[b+m])
    m_lo = max(-b, -(n-1))
    m_hi = min(n-b, n) - 1
    for m in range(m_lo, m_hi+1):
        delta[n-1+m] -= (1.0 - h[b+m])
    # As second factor: h[b-m]*(1-h[b]) → was h[b-m]*0, now h[b-m]*1, change: +h[b-m]
    m_lo2 = max(b-n+1, -(n-1))
    m_hi2 = min(b, n-1)
    for m in range(m_lo2, m_hi2+1):
        delta[n-1+m] += h[b-m]
    
    return delta

def project_feasible(h, n):
    target = n / 2.0

                       options={'maxiter': maxiter, 'ftol': 1e-16})

def fast_greedy_binary_ls(h, n, deadline, max_iters=50000, seed=42):
    """Fast greedy binary local search focusing on the worst lag."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    cur_max = float(corr.max())
    best_max = cur_max

    stuck_count = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        m_star = int(np.argmax(corr) - (n-1))
        
        # Generate candidate swaps that reduce c[m_star]
        candidates = []
        
        if m_star >= 0:
            # c[m_star] = sum_{i=0}^{n-1-m_star} h[i+m_star]*(1-h[i])
            # Swaps that reduce it:
            # Type 1: a=i (0→1), b=i+m_star (1→0) where h[i]=0, h[i+m_star]=1
            for i in range(n - m_star):
                if h[i] == 0 and h[i + m_star] == 1:
                    candidates.append((i, i + m_star))
        else:
            k = -m_star
            # c[m_star] = c[-k] = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
            for i in range(k, n):
                if h[i] == 0 and h[i - k] == 1:
                    candidates.append((i, i - k))
        
        if not candidates:
            stuck_count += 1
            if stuck_count > 20:
                # Random perturbation
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    a = rng.choice(zeros)
                    b = rng.choice(ones)
                    h[a], h[b] = 1, 0
                    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
                    cur_max = float(corr.max())
                continue
        else:
            stuck_count = 0
        
        # Evaluate each candidate
        best_impr = 0
        best_pair = None
        for (a, b) in candidates:
            # Compute delta
            delta = np.zeros(2*n-1)
            # h[a]: 0→1
            m_lo = max(-a, -(n-1))
            m_hi = min(n-a, n) - 1
            for m in range(m_lo, m_hi+1):
                delta[n-1+m] += (1.0 - h[a+m])
            m_lo2 = max(a-n+1, -(n-1))
            m_hi2 = min(a, n-1)

                delta[n-1+m] -= h[a-m]
            # h[b]: 1→0

                delta[n-1+m] -= (1.0 - h[b+m])
            m_lo2 = max(b-n+1, -(n-1))
            m_hi2 = min(b, n-1)
            for m in range(m_lo2, m_hi2+1):
                delta[n-1+m] += h[b-m]
            
            new_corr = corr + delta
            new_max = float(new_corr.max())
            impr = cur_max - new_max
            if impr > best_impr:
                best_impr = impr
                best_pair = (a, b, delta, new_corr)
        
        if best_pair is None or best_impr < 1e-12:
            # No improvement found; try random perturbation
            if rng.random() < 0.3:

                zeros = np.where(h == 0)[0]

    T0 = 2.0
    T1 = 0.005

    while time.time() < deadline:
        progress = min((time.time() - t_start) / t_total, 1.0)
        T = T0 * (T1 / T0) ** progress

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = int(rng.choice(zeros))
        b = int(rng.choice(ones))
        d = vec_delta(h, n, a, b)
        nc = corr + d
        nv = float(nc.max())
        dv = nv - cur_val

        if dv < 0 or rng.random() < np.exp(-dv / max(T, 1e-10)):
            h[a], h[b] = 1, 0
            corr = nc
            cur_val = nv
            if nv < best_val:
                best_val = nv
                best_h = h.copy()

    return best_h, best_val * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Brief SLSQP for smooth init
    p1 = t0 + 25
    h = best_h.copy()
    for alpha in [500, 5000]:
        if time.time() > p1 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 300, p1 - 5)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Round to binary
    h_bin = to_binary(h, n)
    c_bin = compute_c5(h_bin, n)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()

    # Phase 2: Greedy binary LS
    p2 = t0 + 100
    hb, cb = greedy_binary_ls(best_h, n, p2, seed=seed, max_iters=300000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: Simulated annealing
    p3 = t0 + 450
    hb, cb = sa_binary(best_h, n, p3, seed=seed + 100)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 4: Greedy LS after SA
    p4 = t0 + 600
    hb, cb = greedy_binary_ls(best_h, n, p4, seed=seed + 200, max_iters=500000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 5: SLSQP polish from best
    p5 = t0 + 720
    h = best_h.copy() + 0.01 * rng.randn(n)
    h = project_feasible(h, n)
    for alpha in [100000, 1000000]:
        if time.time() > p5 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 1500, p5 - 5)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Binary from SLSQP + greedy
    if time.time() < dl - 60:
        h_bin2 = to_binary(h, n)
        c_b2 = compute_c5(h_bin2, n)
        if c_b2 < best_c:
            best_c = c_b2
            best_h = h_bin2.copy()
        p6 = dl - 5
        hb, cb = greedy_binary_ls(best_h, n, p6, seed=seed + 300, max_iters=500000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 7: Try n=300
    if time.time() < dl - 40:
        nh = 300
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = to_binary(hh, nh)
        c_hh = compute_c5(hh, nh)
        p7 = dl - 3
        hb, cb = greedy_binary_ls(hh, nh, p7, seed=seed + 400, max_iters=100000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            n = nh

    # Return best
    if best_h.max() > 1.0 or best_h.min() < 0.0:
        best_h = np.clip(best_h, 0, 1)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
