def outer_lp_step(h_curr, n, trust_radius):
    dx = 2.0 / n
    c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons_list = [cp.sum(h_var) == n / 2.0,
                 h_var >= 0, h_var <= 1,
                 cp.norm(h_var - h_curr, 2) <= trust_radius]
    
    lags = np.arange(-(n-1), n)
    for k in range(2 * n - 1):
        lag = lags[k]
        m = abs(lag)
        grad = np.zeros(n)
        if m == 0:
            grad[:] = 1.0 - 2.0 * h_curr
        elif lag > 0:
            grad[:n-m] -= h_curr[m:]
            grad[m:] += (1.0 - h_curr[:n-m])
        else:
            grad[:n-m] += (1.0 - h_curr[m:])
            grad[m:] -= h_curr[:n-m]
        cons_list.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons_list)
    for solver in [cp.GLOP, cp.CBC, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    
    if h_var.value is None:
        return h_curr.copy()
    return project_feasible(h_var.value, n)

                                h[j] += delta
    return h, best_c

def swap_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_all_lags(h, n)
    for k_star in np.argsort(c)[-4:]:
        if time.time() > deadline:
            break
        lag = int(k_star) - (n - 1)
        if lag == 0:
            continue
        m = abs(lag)
        for i in range(0, n - m, 2):
            if time.time() > deadline:
                break
            j = i + m
            if h[i] != h[j]:
                swap_i = int(np.argmax(h[[i, j]]))
                swap_j = int(np.argmin(h[[i, j]]))
                new_h = h.copy()
                new_h[i], new_h[j] = h[j], h[i]
                new_h = project_feasible(new_h, n)
                c_new = compute_c5(new_h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    h = new_h
                    c = compute_all_lags(h, n)
                    if time.time() > deadline:
                        break
    return h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Progressive full-smoothing L-BFGS
    for alpha in [30, 100, 500, 2000]:
        if time.time() > deadline - 15:
            break
        h_new = lbfgs_solve(h, n, alpha, maxiter=250)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    
    # Phase 2: Outer LP approximation
    if time.time() < deadline - 40:
        h_temp = best_h.copy()
        trust = 0.25
        for _ in range(12):
            if time.time() > deadline - 20:
                break
            h_next = outer_lp_step(h_temp, n, trust)
            c_next = compute_c5(h_next, n)
            if c_next < best_c - 1e-12:
                best_c = c_next
                best_h = h_next.copy()
            step = np.max(np.abs(h_next - h_temp))
            if step < 1e-8:
                break
            h_temp = h_next
            trust = max(0.005, trust * 0.82)
    
    # Phase 3: Active set L-BFGS (top 6 lags)
    if time.time() < deadline - 20:
        h_temp = best_h.copy()
        for rnd in range(12):
            if time.time() > deadline - 10:
                break
            c = compute_all_lags(h_temp, n)
            top_idx = np.argsort(c)[-6:]
            for alpha in [3000, 20000]:
                if time.time() > deadline - 10:
                    break
                h_new = lbfgs_solve(h_temp, n, alpha, top_idx=top_idx, maxiter=200)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    best_h = h_new.copy()
                    h_temp = h_new
    
    # Phase 4: Very high alpha on top 3 lags
    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-3:]
        h_new = lbfgs_solve(best_h, n, 100000, top_idx=top_idx, maxiter=300)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
    
    # Phase 5: SLSQP polish on top 4 lags
    if time.time() < deadline - 12:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-4:]
        h_new = scg_solve(best_h, n, 50000, top_idx=top_idx, maxiter=1500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
    
    # Phase 6: Expensive local search
    if time.time() < deadline - 3:
        best_h, best_c = expensive_local_search(best_h, n, deadline - 1, top_lags=5)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Optimize at the initial resolution
    if h_init is not None:
        alloc = min(130, (deadline - t0) * 0.18)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Explore higher resolutions
    n_candidates = [400, 500, 650]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 90
        if remaining < 70:
            break
        alloc = min(110, remaining * 0.35)
        if alloc < 45:
            break
        if best_h is not None and len(best_h) == n_try:
            continue
        
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Restarts at best resolution for fine-tuning
    if best_h is not None and time.time() < deadline - 35:
        n = len(best_h)
        n_restarts = 0
        while time.time() < deadline - 25 and n_restarts < 6:
            n_restarts += 1
            scale = 0.006 + 0.004 * n_restarts
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.35)
            if alloc < 18:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final ultra-fine polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # One more active set pass
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-2:]
        h_new = lbfgs_solve(best_h, n, 500000, top_idx=top_idx, maxiter=200)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Final local search
        best_h, best_c = expensive_local_search(best_h, n, deadline - 1, top_lags=3)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        fallback_n = n_init if h_init is not None else 300
        best_h = project_feasible(np.full(fallback_n, 0.5), fallback_n)
        best_c = compute_c5(best_h, fallback_n)
    
    return best_h, best_c, len(best_h)
