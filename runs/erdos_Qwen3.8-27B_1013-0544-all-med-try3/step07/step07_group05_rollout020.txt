import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    """Convert continuous h to binary, preserving sum."""
    h_sorted_idx = np.argsort(-h)  # indices from largest to smallest
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:n//2]] = 1.0
    return h_bin

def grad_smooth_max(h, n, alpha):
    """Compute gradient of smooth max of correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    v = 1.0 - 2.0 * h
    # grad[j] = sum_i w[j+i] * v[i] for i=0..n-1
    corr = np.correlate(w, v, mode='full')
    grad = corr[:n]
    return grad, m

def proj_grad_descent(h, n, deadline, max_iter=5000):
