import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_shift(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_logsumexp(h, n, alpha, active_idx=None):
    c = compute_c5_vec(h, n)
    if active_idx is not None:
        vals = c[active_idx]
    else:
        vals = c
    ac = alpha * vals
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    log_sum = ac_max + np.log(exp_ac.sum())
    val = log_sum / alpha
    w = exp_ac / exp_ac.sum()
    return val, w, c

def objective_smoothed(h, n, alpha, active_idx=None):
    val, w, c = smooth_logsumexp(h, n, alpha, active_idx)
    return val

def gradient_smoothed(h, n, alpha, active_idx=None):
    c = compute_c5_vec(h, n)
    if active_idx is not None:
        vals = c[active_idx]
        indices = active_idx
    else:
        vals = c
        indices = np.arange(len(c))
    ac = alpha * vals
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    w = exp_ac / exp_ac.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, indices):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def linsq(x, n):
    return float(x.sum() - n / 2.0)

def linsq_jac(x, n):
    return np.ones(n)

def get_active_set(c_vec, n, top_frac=0.05, min_count=5):
    n_corr = len(c_vec)
    max_val = c_vec.max()
    threshold = max_val - max(1e-8, 0.01 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(-c_vec)[:min_count]
    elif len(active) > top_frac * n_corr:
        active = np.argsort(-c_vec[active])
        active = active[:top_frac * n_corr]
    return active

def run_lbfgs_stage(h, n, deadline, alpha, maxiter=3000, top_updates=50, n_top=10):

def pair_local_search(h, n, deadline, max_rounds=2000):

def slqp_polish(h, n, deadline, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': linsq, 'jac': linsq_jac, 'args': (n,)}]

def coordinate_descent(h, n, deadline):
    """Coordinate descent: vary one h[i] and its compensating partner."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    target = n / 2.0
    rng = np.random.default_rng(456)

                return best_h if 'best_h' in dir() else h, best_c
            # Find a complementary j to move with i
            # Choose j with similar value (to minimize disruption)
            candidates = np.where(np.abs(h - h[i]) < 0.3)[0]
            candidates = np.delete(candidates, i)
            if len(candidates) == 0:
                continue
            j = candidates[rng.integers(0, len(candidates))]
            
            # Vary delta in [-h[i], 1-h[i]] and [-h[j], 1-h[j]]
            # h[i] += delta, h[j] -= delta (or vice versa)
            best_delta = 0

            for delta in [-0.02, -0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01, 0.02]:

                hj_new = h[j] - delta
                if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:

                h[i] = hi_new

                h[j] -= best_delta

                improved = True

        if not improved:

    if 'best_h' in dir():
