    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    """Compute gradient of smooth max of correlation values."""
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()

        w = np.exp(shifted - mx)

        w = np.exp(shifted)

        lag = k - (n-1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)

            grad[0:n-lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n-idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n-idx])
    return grad * dx

    """Optimize using L-BFGS-B with penalty for sum constraint."""
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        
        return obj_val, grad

    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                      bounds=bounds, jac=True,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50})
        return project_feasible(np.clip(res.x, 0, 1), n)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)

def binary_relax(h, n, deadline, rng, alpha=200000, maxiter=2000):
    """Threshold to binary, then relax back to continuous."""
    # Threshold
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        # Remove excess ones (remove the smallest h values among those > 0.5)
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1

    # Now relax from binary using L-BFGS-B
    h_relaxed = lbfgsb_optimize(h_bin.astype(float), n, alpha, deadline, maxiter=maxiter)
    return h_relaxed

    maxiter_phases = [300, 400, 500, 600, 700, 800, 1000, 1200]

    for alpha, maxit in zip(alpha_phases, maxiter_phases):
        if time.time() > deadline - 300:
            break
        remaining = deadline - 300 - time.time()
        if remaining < 5:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + min(remaining, 60),
                                maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Simulated annealing
    if time.time() < deadline - 200:
        sa_deadline = min(deadline - 120, time.time() + 250)
        h_opt, c_opt = simulated_annealing(best_h, n, sa_deadline, rng,
                                           alpha_sa=80000, t_init=0.008,
                                           t_min=0.00003, cool=0.99985)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: L-BFGS-B polish after SA
    if time.time() < deadline - 150:
        for alpha in [500000, 5000000, 50000000]:
            if time.time() > deadline - 60:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 30, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Binary relaxation as perturbation
    if time.time() < deadline - 150:
        h_bin = binary_relax(best_h, n, time.time() + 60, rng, alpha=500000, maxiter=2000)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
            # Polish the binary-relaxed solution
            h_opt = lbfgsb_optimize(best_h, n, 10000000, time.time() + 30, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Try higher resolution
    for n_try in [700, 1000]:
        if time.time() > deadline - 150:
            break
        remaining = deadline - 100 - time.time()
        if remaining < 40:
            break
        h_interp = expand_to_resolution(best_h, len(best_h), n_try)
        # Optimize at new resolution
        h_opt = lbfgsb_optimize(h_interp, n_try, 100000, time.time() + min(30, remaining * 0.3),
                                maxiter=1500)
        h_opt = lbfgsb_optimize(h_opt, n_try, 5000000, time.time() + min(20, remaining * 0.2),
                                maxiter=2000)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Intensive local search
    if time.time() < deadline - 40:
        n_final = len(best_h)
        ls_deadline = min(deadline - 10, time.time() + 100)
        h_opt, c_opt = intensive_local_search(best_h, n_final, ls_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Another SA round at the best resolution
    if time.time() < deadline - 50:
        n_final = len(best_h)
        sa_deadline2 = min(deadline - 20, time.time() + 80)
        h_opt, c_opt = simulated_annealing(best_h, n_final, sa_deadline2, rng,
                                           alpha_sa=200000, t_init=0.003,
                                           t_min=0.00002, cool=0.9999)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 15:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 50000000, deadline - 5, maxiter=3000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final local search
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_opt, c_opt = intensive_local_search(best_h, n_final, deadline - 1, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
