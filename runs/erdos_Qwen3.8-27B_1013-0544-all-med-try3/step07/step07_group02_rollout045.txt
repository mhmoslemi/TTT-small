    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    grad[:n-m] += (1.0 - h[m:])
    grad[m:] += -h[:n-m]
    return grad

def sqp_step(h, n, top_m, deadline):
    """One SQP iteration using LP subproblem."""
    dx = 2.0 / n
    c_all = np.correlate(h, 1 - h, mode='full') * dx
    
    top_idx = np.argsort(c_all)[-top_m:]
    
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = 1.0
    
    n_cons = len(top_idx)
    A_ub = np.zeros((n_cons, nv))
    b_ub = np.zeros(n_cons)
    
    for idx, k_idx in enumerate(top_idx):
        lag = k_idx - (n - 1)
        g = lag_gradient(h, n, lag)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = np.dot(g, h) - c_all[k_idx]
    
    A_eq = np.ones((1, nv))
    A_eq[0, n] = 0.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(c_all.min() * 0.5, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            h_lp = res.x[:n]
            return h_lp, res.x[n]
    except Exception:
        pass
    return h, c_all.max()

def smooth_max_grad(h, n, alpha):
    """Gradient of smooth max of all lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        lag = k - (n-1)
        grad += w[k] * lag_gradient(h, n, lag)
    return grad * dx
