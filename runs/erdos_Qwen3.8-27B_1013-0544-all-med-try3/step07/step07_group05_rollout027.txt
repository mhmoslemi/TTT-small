    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def precompute_patterns(h, n):
    """Precompute P_i for each i: P_i[k] = (1-h[i-k])*I[0<=i-k<n] - h[i+k]*I[0<=i+k<n]"""
    k = np.arange(-(n-1), n)  # shift values, length 2n-1
    patterns = np.zeros((n, 2*n-1))
    for i in range(n):
        i_minus_k = i - k
        i_plus_k = i + k
        valid_minus = (i_minus_k >= 0) & (i_minus_k < n)
        valid_plus = (i_plus_k >= 0) & (i_plus_k < n)
        patterns[i] = np.where(valid_minus, 1.0 - h[i_minus_k], 0.0) - np.where(valid_plus, h[i_plus_k], 0.0)
    return patterns

def fast_local_search(h, n, deadline, deltas, max_passes=10000):
    """Fast local search using precomputed patterns."""
    h = h.copy()
    dx = 2.0 / n
    c = compute_correlation(h, n)
    best_c5 = c.max() * dx
    
    for delta in deltas:
        patterns = precompute_patterns(h, n)
        n_passes = 0

                            h[j] -= delta
                            c += dc
                            best_c5 = new_max
                            improved = True
                            patterns = precompute_patterns(h, n)  # Need to update!
                            n_passes += 1
                            break
                    # Try h[i] -= delta, h[j] += delta
                    if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                        dc = delta * (patterns[j] - patterns[i])
                        new_max = (c + dc).max() * dx
                        if new_max < best_c5 - 1e-15:

                            c += dc
                            best_c5 = new_max
                            improved = True
                            patterns = precompute_patterns(h, n)
                            n_passes += 1
                            break
                if improved:
                    break
            if not improved:
                break
    return h, best_c5
