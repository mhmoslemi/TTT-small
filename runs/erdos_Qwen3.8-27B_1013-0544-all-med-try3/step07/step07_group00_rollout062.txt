    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def l2_obj(x, n, mu):
    c = compute_c5_vec(x, n)
    val = float(np.sum(c * c))
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def l2_grad(x, n, mu):
    c = compute_c5_vec(x, n)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        k = mm - (n - 1)
        wm = 2.0 * c[mm] * dx
        if wm < 1e-15:
            continue
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_lse(h, n, deadline, alpha, mu, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, jac=lse_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-15, 'maxfun': maxiter * 10, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def lbfgs_l2(h, n, deadline, mu=100.0, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(l2_obj, h, jac=l2_grad,
                      args=(n, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 10, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def cyclic_cd(h, n, deadline, max_sweeps=80):
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    n_golden = 22

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            a, b = 0.0, 1.0
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            h[i] = c_pt
            fc = compute_c5(h, n)
            h[i] = d_pt
            fd = compute_c5(h, n)
            for _g in range(n_golden):
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    h[i] = c_pt
                    fc = compute_c5(h, n)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    h[i] = d_pt
                    fd = compute_c5(h, n)
            h[i] = (a + b) / 2.0
        h = project_feasible(h, n)
        c_now = compute_c5(h, n)
        if c_now < best_c - 1e-15:
            best_c = c_now
            improved = True
        if sweep > 2 and not improved:
            break
    return h, best_c

def top_shifts_cd(h, n, deadline, max_sweeps=60, n_top=8):
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    n_golden = 20

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.005 * max_val)
        active_shifts = np.where(c_vec >= threshold)[0]
        if len(active_shifts) > n_top:
            active_shifts = active_shifts[np.argsort(-c_vec[active_shifts])[:n_top]]

        # Find indices involved in active shifts
        n_half = n - 1
        involved = set()
        for s in active_shifts:
            k = s - n_half
            if k >= 0:
                involved.update(range(n - k))
                involved.update(range(k, n))
            else:
                kk = -k
                involved.update(range(n - kk))
                involved.update(range(kk, n))
        sorted_involved = sorted(involved)[:min(len(involved), 200)]

        improved = False
        for i in sorted_involved:
            if time.time() > deadline:
                return h, best_c
            a, b = 0.0, 1.0
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            h[i] = c_pt
            fc = compute_c5(h, n)
            h[i] = d_pt
            fd = compute_c5(h, n)
            for _g in range(n_golden):
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    h[i] = c_pt
                    fc = compute_c5(h, n)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    h[i] = d_pt
                    fd = compute_c5(h, n)
            h[i] = (a + b) / 2.0
        h = project_feasible(h, n)
        c_now = compute_c5(h, n)
        if c_now < best_c - 1e-15:
            best_c = c_now
            improved = True
        if sweep > 2 and not improved:
            break
    return h, best_c

def limited_pair_search(h, n, deadline, radius=30, steps=None):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.001, 0.0005, 0.0002, 0.0001]

    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[j] - d >= -1e-12 and h[i] + d <= 1.0 + 1e-12:
                        h[i] = min(h[i] + d, 1.0)
                        h[j] = max(h[j] - d, 0.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c

def slqp_polish(h, n, deadline, alpha=200000):
    bounds = [(0.0, 1.0)] * n
    try:
        cons_fun = lambda x: x.sum() - n / 2.0
        cons_jac = lambda x: np.ones(len(x))
        res = minimize(lse_obj, h, args=(n, alpha, 100.0),
                      method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                      options={'maxiter': 3000, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def nesterov_subgrad(h, n, deadline, lse_alpha=10000, n_steps=50000):
    h = project_feasible(h.copy(), n)
    h_prev = h.copy()
    best_c = compute_c5(h, n)
    mu = 50.0
    lr = 0.5
    mu2 = 10.0

    for step in range(n_steps):
        if time.time() > deadline:
            break
        t = step + 1
        momentum = (t - 1) / (t + 2)
        y = h + momentum * (h - h_prev)
        np.clip(y, 0.0, 1.0, out=y)

        grad = lse_grad(y, n, lse_alpha, mu)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        step_size = lr / (1.0 + 0.001 * t)
        h_try = y - step_size * grad
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)

        if c_try < best_c:
            best_c = c_try
            h_prev = h.copy()
            h = h_try
            lr = min(lr * 1.01, 2.0)
        else:
            h_prev = h.copy()
            lr *= 0.8
            if lr < 0.001:
                lr = 0.1
                lse_alpha = max(1000, int(lse_alpha * 0.5))

    return h, best_c

def random_restart_opt(h, n, deadline, n_tries=3, noise=0.005):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(777)

    for t in range(n_tries):
        if time.time() > deadline - 40:
            break
        h_pert = h + rng.normal(0, noise, size=n)
        h_pert = project_feasible(h_pert, n)
        t_sub = max(15, (deadline - time.time()) - 40 - (n_tries - 1 - t) * 30)
        h_opt, c_opt = lbfgs_lse(h_pert, n, time.time() + t_sub,
                                   alpha=5000, mu=500.0, maxiter=2000)
        if c_opt < best_c - 1e-9:
            best_c = c_opt
            best_h = h_opt
            h = h_opt
    return best_h, best_c

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(250, 0.5)

    n = len(h_init)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS at original resolution with progressive alpha
    phase1_end = min(80, budget_s * 0.07)
    if time.time() < t0 + phase1_end:
        configs = [(200, 20.0), (1000, 200.0), (5000, 2000.0), (20000, 20000.0)]
        for alpha, mu in configs:
            if time.time() > t0 + phase1_end:
                break
            remaining = t0 + phase1_end - time.time()
            if remaining < 10:
                break
            h_new, c_new = lbfgs_lse(best_h, n, time.time() + remaining,
                                      alpha=alpha, mu=mu, maxiter=3000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 2: L2 objective for different landscape
    if time.time() < deadline - 500:
        t2_end = min(time.time() + 60, deadline - 500)
        h_new, c_new = lbfgs_l2(best_h, n, t2_end, mu=200.0, maxiter=2000)
        if c_new < best_c - 1e-8:
            best_c = c_new
            best_h = h_new

    # Phase 3: Cyclic coordinate descent (full)
    if time.time() < deadline - 350:
        t3_end = min(time.time() + 120, deadline - 350)
        best_h, best_c = cyclic_cd(best_h, n, t3_end, max_sweeps=60)

    # Phase 4: Top-shifts focused CD
    if time.time() < deadline - 250:
        t4_end = min(time.time() + 80, deadline - 250)
        best_h, best_c = top_shifts_cd(best_h, n, t4_end, max_sweeps=50, n_top=10)

    # Phase 5: L-BFGS with very high alpha (tight LSE)
    if time.time() < deadline - 200:
        for alpha, mu in [(50000, 50000.0), (200000, 200000.0)]:
            if time.time() > deadline - 200:
                break
            h_new, c_new = lbfgs_lse(best_h, n, deadline - 200,
                                      alpha=alpha, mu=mu, maxiter=5000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 6: Nesterov accelerated subgradient
    if time.time() < deadline - 150:
        t6_end = min(time.time() + 80, deadline - 150)
        h_new, c_new = nesterov_subgrad(best_h, n, t6_end,
                                         lse_alpha=20000, n_steps=40000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 7: More cyclic CD
    if time.time() < deadline - 100:
        t7_end = min(time.time() + 60, deadline - 100)
        best_h, best_c = cyclic_cd(best_h, n, t7_end, max_sweeps=40)

    # Phase 8: L2 optimization round
    if time.time() < deadline - 70:
        h_new, c_new = lbfgs_l2(best_h, n, deadline - 70, mu=500.0, maxiter=2000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 9: Limited pair search
    if time.time() < deadline - 50:
        t9_end = min(time.time() + 35, deadline - 50)
        best_h, best_c = limited_pair_search(best_h, n, t9_end,
                                              radius=25, steps=[0.0005, 0.0002, 0.0001])

    # Phase 10: Top-shifts CD again
    if time.time() < deadline - 30:
        t10_end = min(time.time() + 20, deadline - 30)
        best_h, best_c = top_shifts_cd(best_h, n, t10_end, max_sweeps=20, n_top=5)

    # Phase 11: SLSQP polish
    if time.time() < deadline - 8:
        h_new, c_new = slqp_polish(best_h, n, deadline - 5, alpha=500000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 12: Final micro-cd
    if time.time() < deadline - 2:
        best_h, best_c = cyclic_cd(best_h, n, deadline - 1, max_sweeps=10)

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
