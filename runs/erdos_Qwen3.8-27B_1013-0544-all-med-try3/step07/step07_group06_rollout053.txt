def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

                    # Let k = m-(n-1), c[n-1+k] = sum_p h[p]*(1-h[p+k])
                    #
                    # When h[i] changes 0->1 and h[j] changes 1->0:
                    # delta_c[n-1+k] = 1*(1-h_new[i+k]) - 0*(1-h_old[i+k])   [i term in h[p]]
                    #                  - 1*(1-h_new[j+k]) + 0*(1-h_old[j+k])  [j term in h[p]]
                    #                  + h_new[i-k]*1 - h_old[i-k]*1          [i term in (1-h[p+k])]
                    #                  - (h_new[j-k]*1 - h_old[j-k]*1)*(-1)   [j term in (1-h[p+k])]
                    #
                    # Wait this is getting confusing. Let me use the direct formula:
                    # Before: h[i]=0, h[j]=1
                    # After: h[i]=1, h[j]=0
                    #
                    # c_k = sum_p h[p]*(1-h[p+k])
                    # Change at p=i: was 0*(1-h[i+k]), now 1*(1-h_new[i+k])
                    #   = (1-h_new[i+k]) if i+k valid, else 0
                    # Change at p=j: was 1*(1-h_new[j+k]) [h[j] now 0 so this term vanishes]
                    #   Actually was 1*(1-h[j+k]) with h[j]=1, now 0*(1-h[j+k])=0
                    #   Change: -(1-h_new[j+k]) if j+k valid
                    #
                    # Also, i appears as p+k when p=i-k: h[i-k]*(1-h[i]) changes from h[i-k]*1 to h[i-k]*0
                    #   Change: -h_new[i-k] if i-k valid
                    # j appears as p+k when p=j-k: h[j-k]*(1-h[j]) changes from h[j-k]*0 to h[j-k]*1
                    #   Change: +h_new[j-k] if j-k valid
                    #
                    # So delta_c[n-1+k] = (1-h_new[i+k])_{valid} - (1-h_new[j+k])_{valid}
                    #                     - h_new[i-k]_{valid} + h_new[j-k]_{valid}
                    #
                    # Where h_new differs from h_old only at i and j.
                    # If i+k == i: h_new[i+k] = 1 (k=0)
                    # If i+k == j: h_new[i+k] = 0
                    # If j+k == i: h_new[j+k] = 1
                    # If j+k == j: h_new[j+k] = 0 (k=0)
                    # If i-k == i: h_new[i-k] = 1 (k=0)
                    # If i-k == j: h_new[i-k] = 0
                    # If j-k == i: h_new[j-k] = 1
                    # If j-k == j: h_new[j-k] = 0 (k=0)
                    
                    delta = np.zeros(2 * n - 1)
                    
                    # + (1-h_new[i+k]) for valid i+k
                    ik = i
                    for offset in m_range:
                        pass  # Too slow to loop
                    
                    # Let me vectorize:
                    # Term 1: (1-h_new[i+k]) for k where 0<=i+k<n
                    # k range: -i to n-1-i
                    ks1_start = max(-(n-1), -i)
                    ks1_end = min(n-1, n-1-i)
                    if ks1_end >= ks1_start:
                        idx1 = np.arange(ks1_start, ks1_end + 1)
                        vals1 = 1.0 - h[i + idx1]
                        vals1[i + idx1 == i] = 0.0  # h_new[i]=1, so 1-1=0
                        vals1[i + idx1 == j] = 1.0  # h_new[j]=0, so 1-0=1
                        delta[(n-1) + idx1] += vals1
                    
                    # Term 2: -(1-h_new[j+k]) for valid j+k
                    ks2_start = max(-(n-1), -j)
                    ks2_end = min(n-1, n-1-j)
                    if ks2_end >= ks2_start:
                        idx2 = np.arange(ks2_start, ks2_end + 1)
                        vals2 = 1.0 - h[j + idx2]
                        vals2[j + idx2 == i] = 0.0  # h_new[i]=1
                        vals2[j + idx2 == j] = 1.0  # h_new[j]=0
                        delta[(n-1) + idx2] -= vals2
                    
                    # Term 3: -h_new[i-k] for valid i-k
                    ks3_start = max(-(n-1), i-(n-1))
                    ks3_end = min(n-1, i)
                    if ks3_end >= ks3_start:
                        idx3 = np.arange(ks3_start, ks3_end + 1)
                        vals3 = h[i - idx3]
                        vals3[i - idx3 == i] = 1.0  # h_new[i]=1
                        vals3[i - idx3 == j] = 0.0  # h_new[j]=0
                        delta[(n-1) + idx3] -= vals3
                    
                    # Term 4: +h_new[j-k] for valid j-k
                    ks4_start = max(-(n-1), j-(n-1))
                    ks4_end = min(n-1, j)
                    if ks4_end >= ks4_start:
                        idx4 = np.arange(ks4_start, ks4_end + 1)
                        vals4 = h[j - idx4]
                        vals4[j - idx4 == i] = 1.0  # h_new[i]=1
                        vals4[j - idx4 == j] = 0.0  # h_new[j]=0
                        delta[(n-1) + idx4] += vals4
                    
                    new_corr = corr + delta
                    nc = new_corr.max() * (2.0 / n)
                    
                    if nc < cur_c - 1e-14:
                        h[i] = 1.0
                        h[j] = 0.0
                        corr = new_corr
                        cur_c = nc
                        ones.add(i)
                        ones.discard(j)
                        zeros.discard(i)
                        zeros.add(j)
                        ones_list = list(ones)
                        zeros_list = list(zeros)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c


def slsqp_refine(h, n, deadline, maxiter=2000):
    """Use SLSQP with softmax smoothing for final polish."""
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        alpha = 1e5
        sh = alpha * (g - m)
        sm = sh.max()
        return m + np.log(np.exp(sh - sm).sum()) / alpha
    
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    n = n0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Coordinate Chebyshev descent (fast, ~250s)
    p1_end = t0 + 0.25 * (budget_s - 3)
    h1, c1 = coordinate_chebyshev(best_h, n, p1_end, max_sweeps=500, tol=0.002)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()
    
    # Phase 2: Binary local search (extensive, ~350s)
    p2_end = t0 + 0.57 * (budget_s - 3)
    hb, cb = binary_local_search(best_h, n, p2_end, max_sweeps=3000, n_restarts=8, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 3: Continuous refinement from binary (start unrounded, ~150s)
    p3_end = t0 + 0.72 * (budget_s - 3)
    if time.time() < p3_end - 30:
        h3 = best_h.copy()
        # Un-round: add small random perturbation then project
        rng = np.random.RandomState(seed + 100)
        h3 = project_feasible(h3 + 0.1 * rng.randn(n), n)
        h3, c3 = coordinate_chebyshev(h3, n, p3_end, max_sweeps=300, tol=0.0005)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    
    # Phase 4: Multi-resolution attempts (~150s)
    p4_end = t0 + 0.86 * (budget_s - 3)
    for nh in [200, 300, 350, 400]:
        if time.time() > p4_end - 30:
            break
        if nh == n:
            continue
        # Interpolate best_h to new resolution
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        # Quick binary search at new resolution
        hb2, cb2 = binary_local_search(hh, nh, min(time.time() + 40, p4_end - 10),
                                        max_sweeps=500, n_restarts=3, seed=seed + nh)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n = nh
    
    # Phase 5: Final polish - coordinate descent + binary search
    p5_end = deadline - 5
    if time.time() < p5_end - 60:
        # Continuous coordinate descent
        h5 = project_feasible(best_h + 0.05 * np.random.RandomState(seed+200).randn(n), n)
        h5, c5 = coordinate_chebyshev(h5, n, min(time.time() + 80, p5_end - 40),
                                        max_sweeps=200, tol=0.0003)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
    
    # Final binary search
    if time.time() < p5_end - 10:
        hb5, cb5 = binary_local_search(best_h, n, p5_end, max_sweeps=1000, n_restarts=3, seed=seed+300)
        if cb5 < best_c:
            best_c = cb5
            best_h = hb5.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
