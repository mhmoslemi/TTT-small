import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def local_search(h, n, deadline, max_sweeps=2000):
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h, n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_list):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr = compute_corr(h, n)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def il_s(h_start, n, deadline, rng, max_rounds=30):
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h, n)
    best_h = cur_h.copy()
    best_c = cur_c
    dx = 2.0 / n
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        cur_h, cur_c = local_search(cur_h, n, min(deadline, time.time() + 8))
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
        n_pert = rng.randint(2, 5)
        pert_h = cur_h.copy()
        ones = np.where(pert_h == 1)[0]
        zeros = np.where(pert_h == 0)[0]
        used_i = []
        used_j = []
        for _ in range(n_pert):
            if len(ones) < 1 or len(zeros) < 1:
                break
            ii = rng.randint(len(zeros))
            jj = rng.randint(len(ones))
            i = zeros[ii]
            j = ones[jj]
            if i in used_i or j in used_j:
                continue
            pert_h[i], pert_h[j] = 1.0, 0.0
            used_i.append(i)
            used_j.append(j)
            ones = np.where(pert_h == 1)[0]
            zeros = np.where(pert_h == 0)[0]
        cur_h = pert_h
    return best_h, best_c

def anneal_single(h_start, n, deadline, rng, T_start=0.008, T_end=0.0003, cool=0.99995):
    dx = 2.0 / n
    h = h_start.copy()
    corr = compute_corr(h, n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    m_range = np.arange(-(n - 1), n)
    T = T_start
    while time.time() < deadline and T > T_end:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = swap_delta(h, n, i, j, m_range)
        nc = (corr + delta).max() * dx
        accept = False
        if nc <= cur_c:
            accept = True
        else:
            diff = nc - cur_c
            if diff > 0:
                ratio = np.exp(-diff / T)
                if rng.random() < ratio:
                    accept = True
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def anneal_multi(h_start, n, deadline, rng):
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    n_restarts = 5
    time_per = max(5.0, (deadline - time.time()) / n_restarts)
    for r in range(n_restarts):
        dl_r = min(deadline, time.time() + time_per)
        T_s = 0.005 + 0.003 * rng.random()
        cool_r = 0.9999 + 0.0001 * rng.random()
        h_r, c_r = anneal_single(h_start, n, dl_r, rng, T_start=T_s, T_end=0.0002, cool=cool_r)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
    return best_h, best_c

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

def pgd_solve(h_start, n, alphas, deadline, max_iter=10000):
    dx = 2.0 / n
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        cur_c = best_c
        step = min(float(n) / (4.0 * alpha), 0.5)
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            grad = lse_grad(h, n, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            direction = -grad / gnorm
            accepted = False
            for ls in range(30):
                h_new = project(h + step * (0.5 ** ls) * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    h = h_new
                    cur_c = c_new
                    step = step * (0.5 ** ls)
                    accepted = True
                    break
            if accepted:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                break
        c = compute_c5(best_h, n)
        best_h = h.copy()
    return best_h, best_c
