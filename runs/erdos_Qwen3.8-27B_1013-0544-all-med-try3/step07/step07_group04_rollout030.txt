            grad[j2] -= p[k] * dx * h[:n - abs_m]
        else:
            # m_lag < 0, abs_m = -m_lag
            # dc/dh_j = dx * [(1-h[j-abs_m]) for j in [abs_m, n-1]] - [h[j+abs_m] for j in [0, n-abs_m-1]]
            j1 = slice(abs_m, n)
            j2 = slice(0, n - abs_m)
            grad[j1] += p[k] * dx * (1.0 - h[:n - abs_m])
            grad[j2] -= p[k] * dx * h[abs_m:]
    
    # Add penalty gradient
    if penalty_mu > 0:
        diff = h.sum() - n / 2.0
        obj += penalty_mu * diff * diff
        grad += 2.0 * penalty_mu * diff
    
    return obj, grad

def lbfgsb_optimize(h, n, alpha, penalty_mu, deadline, maxiter=2000):
    """Optimize using L-BFGS-B."""
    bounds = [(0.0, 1.0)] * n
    
    def wrapped_obj(x):
        obj, _ = smooth_max_obj_and_grad(x, n, alpha, penalty_mu)
        return obj
    
    def wrapped_grad(x):
        _, grad = smooth_max_obj_and_grad(x, n, alpha, penalty_mu)
        return grad
    
    try:
        res = minimize(wrapped_obj, h, jac=wrapped_grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16,
                                             'gtol': 1e-14, 'maxfun': 50000})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    """SLSQP with exact sum constraint - used for final polish."""
    bounds = [(0.0, 1.0)] * n
    
    def sum_fun(x, n):
        return float(np.sum(x) - n / 2.0)
    
    def sum_jac(x, n):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    
    def obj(x):
        return smooth_max_obj_and_grad(x, n, alpha, 0.0)[0]
    
    def grad(x):
        return smooth_max_obj_and_grad(x, n, alpha, 0.0)[1]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_optimize(h, n, deadline, top_m, alpha_seq):

def binary_push(h, n, deadline):
    """Push h values toward 0 or 1 to reduce C(0) while keeping other lags bounded."""
    h = h.copy()
    best_c = compute_c5(h, n)
    max_rounds = 200
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        # Find the "most fractional" h[i] (furthest from 0 and 1)
        frac = 1.0 - 2.0 * np.abs(h - 0.5)  # higher = more fractional... wait
        # Actually, we want to push values that are close to 0 or 1
        # The benefit of pushing h[i] toward 0 or 1 is (h[i] - h[i]^2) decrease
        # = h[i]*(1-h[i]) decrease
        # The most "beneficial" push is on the h[i] with largest h[i]*(1-h[i])
        # Wait no, we want to REDUCE C(0) = dx*(n/2 - sum h^2)
        # So we want to INCREASE sum h^2
        # Pushing h[i] from v to v+dv (toward 1) increases h^2 by 2v+1 * dv
        # Pushing h[i] from v to v-dv (toward 0) increases h^2 by (2v-1) * dv... 
        # d(h^2)/dh = 2h
        # If h > 0.5: pushing toward 1 gives d(h^2) = 2h * dv (positive)
        # If h < 0.5: pushing toward 0 gives d(h^2) = -2h * dv (negative!) 
        # Wait: if h < 0.5 and we push toward 0: h -> h - dv, h^2 -> (h-dv)^2 = h^2 - 2h*dv + dv^2
        # So h^2 decreases! That's bad for C(0).
        # If h < 0.5 and we push toward 1: h -> h + dv, h^2 -> h^2 + 2h*dv
        # So h^2 increases. Good!
        # Similarly if h > 0.5 and we push toward 1: h^2 increases by 2h*dv (even more)
        # If h > 0.5 and we push toward 0: h^2 decreases.
        
        # So the best move is to push h[i] toward 1 always (for any h[i] in (0,1))
        # and push h[j] toward 0 (for any h[j] in (0,1))
        # Net effect on sum h^2: +2*h[i]*dv - 2*h[j]*dv = 2*(h[i]-h[j])*dv
        # This is positive if h[i] > h[j]
        
        # So: find pair (i,j) with h[i] > h[j], both in (0,1), and transfer mass from j to i
        best_gain = 0
        best_i, best_j = -1, -1
        # To be efficient, sort by h value
        idx = np.argsort(h)
        # j should be low, i should be high
        for j_idx in range(min(20, n//4)):  # bottom 20
            for i_idx in range(max(n//4, n-20), n):  # top 20
                i, j = i_idx, j_idx
                if 0 < h[j] < 1 and 0 < h[i] < 1:
                    gain = 2 * (h[i] - h[j])
                    if gain > best_gain:
                        best_gain = gain
                        best_i, best_j = i, j
        if best_i < 0:
            break
        d = min(0.01, h[best_j] * 0.5, 1 - h[best_i] * 0.5)
        h[best_i] += d
        h[best_j] -= d
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c + 1e-10:  # Allow tiny increase
            best_c = min(best_c, c_new)
        else:
            # Revert
            h[best_i] -= d
            h[best_j] += d
            h = project_feasible(h, n)
    return h, best_c

def outer_approx_step(h_curr, n, trust):
    """LP-based outer approximation step."""
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m = abs(lag)
        if m == 0:
            # c[0] at h_curr: sum h(1-h) = n/2 - sum h^2
            # Linearization: n/2 - sum h^2 ≈ n/2 - (2*h_curr - 1) @ (h_var - h_curr) - sum h_curr^2
            # = n/2 - sum h_curr^2 + (1-2*h_curr) @ (h_var - h_curr)
            # = c[lag] + (1-2*h_curr) @ (h_var - h_curr)  [since c = dx*(n/2 - sum h^2)]
            # Actually c[k] * (1/dx) = n/2 - sum h_curr^2
            # Gradient of (n/2 - sum h^2) w.r.t. h = -2*h + ... 
            # d/dh_j (n/2 - sum h_i^2) = -2*h_j
            # So linearization: c[k] + dx * (-2*h_curr) @ (h_var - h_curr)
            grad = -2.0 * h_curr
        elif lag > 0:
            # c = dx * sum_{i=0}^{n-m-1} h[i]*(1-h[i+m])
            # grad_j = dx * (1-h[j+m]) for j < n-m; grad_j = -dx*h[j-m] for j >= m
            grad = np.zeros(n)
            grad[:n-m] = (1.0 - h_curr[m:])
            grad[m:] -= h_curr[:n-m]
        else:
            # c = dx * sum_{i=0}^{n-m-1} h[i+m]*(1-h[i])
            grad = np.zeros(n)
            grad[m:] = (1.0 - h_curr[:n-m])
            grad[:n-m] -= h_curr[m:]
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=50):
    h_curr = h.copy()
    best_h = h_curr.copy()
