    h_curr = h.copy()

        for mm in range(2*n-1):

                grad[mm, 0:nv] = (1.0 - h_curr[k:n]) * dx
                grad[mm, k:n] += (-h_curr[0:nv]) * dx

def slp_iteration(h_curr, n, active_idx, trust_r):
    """One SLP iteration using linprog."""
    from scipy.optimize import linprog
    
    c_vec = compute_c5_vec(h_curr, n)
    dx = 2.0 / n
    
    # Compute gradients for active shifts
    m = len(active_idx)
    grads = np.zeros((m, n))
    for i, mm in enumerate(active_idx):
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grads[i, 0:nv] = (1.0 - h_curr[k:n]) * dx
            grads[i, k:n] += (-h_curr[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grads[i, kk:n] = (1.0 - h_curr[0:nv]) * dx
            grads[i, 0:nv] += (-h_curr[kk:n]) * dx
    
    # LP: min t
    #   s.t. -t + grads[i] @ h <= -c_vec[active_idx[i]] + grads[i] @ h_curr  for i
    #        0 <= h <= 1 (and trust region)
    #        sum(h) = n/2
    
    # Variables: [t, h_0, h_1, ..., h_{n-1}]
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[0] = 1.0  # minimize t
    
    # Inequality constraints
    A_rows = []
    b_rows = []
    
    for i in range(m):
        row = np.zeros(n_vars)
        row[0] = -1.0
        row[1:] = grads[i]
        A_rows.append(row)
        b_rows.append(-c_vec[active_idx[i]] + np.dot(grads[i], h_curr))
    
    # h_i <= 1
    for i in range(n):
        row = np.zeros(n_vars)
        row[i+1] = 1.0
        A_rows.append(row)
        b_rows.append(1.0)
    
    # -h_i <= 0
    for i in range(n):
        row = np.zeros(n_vars)
        row[i+1] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
    
    # Trust region
    for i in range(n):
        row = np.zeros(n_vars)
        row[i+1] = 1.0
        A_rows.append(row)
        b_rows.append(h_curr[i] + trust_r)
        
        row = np.zeros(n_vars)
        row[i+1] = -1.0
        A_rows.append(row)
        b_rows.append(-(h_curr[i] - trust_r))
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n/2.0])
    
    # Bounds
    bounds = [(None, None)] + [(0, 1)] * n  # t is free, h in [0,1]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    h_new = result.x[1:]
    return np.clip(h_new, 0, 1)

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_single(h, n, mm):
    """Gradient of c_mm with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = mm - (n - 1)
    if k >= 0:
        nv = n - k
        grad[:nv] = (1.0 - h[k:]) * dx
        grad[k:] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:] = (1.0 - h[:nv]) * dx
        grad[:nv] += (-h[kk:]) * dx
    return grad

def slp_optimize(h, n, deadline, max_iter=500, trust_r_init=0.5, 
                 active_tol_init=0.005, max_active=30):
    """Successive Linear Programming."""
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    trust_r = trust_r_init
    active_tol = active_tol_init
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h_curr, n)
        max_val = c_vec.max()
        
        # Active set
        active_mask = c_vec >= max_val - active_tol
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > max_active:
            order = np.argsort(-c_vec[active_idx])
            active_idx = active_idx[order[:max_active]]
        elif len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c_vec))])
        
        # Solve LP
        h_new = solve_slp_lp(h_curr, n, active_idx, c_vec, trust_r)
        if h_new is None:
            trust_r *= 0.5
            if trust_r < 0.01:
                break
            continue
        
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-12:
            best_c = c_new
            h_curr = h_new
            stagnant = 0
            trust_r = min(trust_r * 1.2, 1.0)  # Slightly expand
        else:
            # Try damped step
            improved = False
            for t in [0.7, 0.5, 0.3, 0.1]:
                h_d = h_curr + t * (h_new - h_curr)
                h_d = project_feasible(h_d, n)
                c_d = compute_c5(h_d, n)
                if c_d < best_c - 1e-12:
                    best_c = c_d
                    h_curr = h_d
                    improved = True
                    break
            if not improved:
                stagnant += 1
                trust_r *= 0.7
                active_tol *= 1.5  # Expand active set
                if stagnant > 10:
                    break
    
    return h_curr, best_c

def solve_slp_lp(h_curr, n, active_idx, c_vec, trust_r):
    """Solve the LP subproblem for one SLP iteration."""
    dx = 2.0 / n
    m = len(active_idx)
    
    # Compute gradients

        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grads[i, :nv] = (1.0 - h_curr[k:]) * dx
            grads[i, k:] += (-h_curr[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grads[i, kk:] = (1.0 - h_curr[:nv]) * dx
            grads[i, :nv] += (-h_curr[kk:]) * dx
    
    # Build LP
    # Variables: [t, h_0, ..., h_{n-1}]
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[0] = 1.0
    
    # Inequality rows
    n_ineq = m + 4 * n  # active constraints + bounds + trust region
    A_ub = np.zeros((n_ineq, n_vars))
    b_ub = np.zeros(n_ineq)
    row = 0
    
    # Active constraints: -t + grads[i] @ h <= -c_k + grads[i] @ h_curr
    for i in range(m):
        A_ub[row, 0] = -1.0
        A_ub[row, 1:] = grads[i]
        b_ub[row] = -c_vec[active_idx[i]] + np.dot(grads[i], h_curr)
        row += 1
    
    # h_i <= 1
    for i in range(n):
        A_ub[row, i+1] = 1.0
        b_ub[row] = 1.0
        row += 1
    
    # -h_i <= 0
    for i in range(n):
        A_ub[row, i+1] = -1.0
        b_ub[row] = 0.0
        row += 1
    
    # Trust region: h_i <= h_curr_i + trust_r
    for i in range(n):
        A_ub[row, i+1] = 1.0
        b_ub[row] = h_curr[i] + trust_r
        row += 1
    
    # -h_i <= -(h_curr_i - trust_r)
    for i in range(n):
        A_ub[row, i+1] = -1.0
        b_ub[row] = -(h_curr[i] - trust_r)
        row += 1
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    bounds = [(None, None)] + [(0.0, 1.0)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 30})
        if result.status == 0:
            return result.x[1:]
    except Exception:
        pass
    
    return None

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    n_shifts = 2 * n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(n_shifts):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            wm = w[mm] * dx
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            wm = w[mm] * dx
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_pass(h, n, deadline, alphas):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_moves(h, n, deadline, deltas=None):
    """Fine pair moves."""
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved_any = False
        stride = max(1, n // 200)
        for i in range(0, n, stride):
            if time.time() > deadline:
                return h, best_c
            # Try a few partner indices
            partners = [(i + d * n) % n, (i - d * n) % n if i >= d*n else (i + n - d*n) % n,
                       i + 1 if i+1 < n else 0, i - 1 if i > 0 else 1,
                       (i + n//2) % n]
            for j in partners:
                if j < 0 or j >= n or j == i:
                    continue
                if h[i] >= d and h[j] <= 1.0 - d:
                    h[i] -= d; h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-13:
                        best_c = c; improved_any = True
                    else:
                        h[i] += d; h[j] -= d
                if h[j] >= d and h[i] <= 1.0 - d:
                    h[i] += d; h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-13:
                        best_c = c; improved_any = True
                    else:
                        h[i] -= d; h[j] += d
        if not improved_any:
            break
    return h, best_c
