import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    """Gradient of overlap at given lag w.r.t. h (includes dx factor)."""
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    else:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return dx * grad

def _smooth_obj_and_grad(h, n, alpha, top_idx=None):
    """Smooth max of overlaps with analytical gradient."""
    dx = 2.0 / n
    c_all = np.correlate(h, 1 - h, mode='full') * dx
    if top_idx is not None:
        c_act = c_all[top_idx]
        idx = top_idx
    else:
        c_act = c_all
        idx = np.arange(len(c_all))
    
    m = c_act.max()
    shifted = alpha * (c_act - m)
    shifted -= shifted.max()
    exp_sh = np.exp(shifted)
    Z = exp_sh.sum()
    w = exp_sh / Z
    
    obj = m + np.log(Z) / alpha
    
    grad = np.zeros(n)
    for i in range(len(c_act)):
        if w[i] < 1e-12:
            continue
        lag = idx[i] - (n - 1)
        grad += w[i] * _lag_gradient(h, n, lag)
    
    return obj, grad

def _sum_con(h, n):
    return np.sum(h) - n / 2.0

def _sum_con_jac(h, n):
    return np.ones(n)

def slsqp_with_grad(h, n, alpha, maxiter=3000, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        val, _ = _smooth_obj_and_grad(x, n, alpha, top_idx)
        return val
    
    def grad(x):
        _, g = _smooth_obj_and_grad(x, n, alpha, top_idx)
        return g
    
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    
    try:
        res = minimize(obj, h, jac=grad, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

2. For a proposed exchange (move δ from position i to position j), compute the change in each overlap analytically.
3. Check if the max overlap decreases.

The change in overlap[k] when h[i] -= δ and h[j] += δ:
For each lag k, the overlap changes by:
Δ_k = δ * [∂overlap_k/∂h[j] - ∂overlap_k/∂h[i]]

And ∂overlap_k/∂h[j] is just the j-th component of the gradient of overlap_k, which I can compute.

But computing this for all lags k is O(n) per proposal, and we want to try O(n) proposals, so O(n²) per iteration. For n=300, that's 90,000 per iteration. With maybe 100 iterations per step size and 6 step sizes, that's 54 million operations. Should take a few seconds in numpy.

Actually, I can vectorize this better. Let me think...

For a fixed pair (i, j), the change in overlap at lag k is:
Δ_k = δ * (grad_k[j] - grad_k[i])

where grad_k is the gradient of overlap_k.

I can precompute all gradients at once (a matrix of shape (2n-1, n)), but that's 599 * 300 = 179,700 values, which is fine.

Then for a given pair (i,j), the changes are: changes = δ * (G[:, j] - G[:, i]) where G is the gradient matrix.
The new max overlap is max(c + changes).

This is O(n) per pair evaluation (for the max). With O(n²) pairs, total O(n³) ≈ 27 million for n=300. Doable in a few seconds.

But we don't need to try all pairs. We can be smarter:
- Focus on pairs where h[i] > h[j] (moving mass from high to low).
- Focus on the worst lag and find pairs that most reduce it.

For the worst lag k*, the gradient component at position p is grad_{k*}[p]. To reduce overlap_{k*}, we want to move mass in the -grad direction. So we want to decrease h at positions where grad is positive and increase h where grad is negative.

A simple heuristic: find i where grad_{k*}[i] is maximally positive (h[i] should decrease) and j where grad_{k*}[j] is maximally negative (h[j] should increase).

This is essentially a steepest descent step for the single worst lag.

OK I think I've been overthinking this. Let me just write a clean implementation with:
1. Progressive alpha SLSQP (main workhorse)
2. Active set SLSQP (refinement)
3. Simple gradient-based local search (final polish)
4. A few different n values

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    else:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return dx * grad

def _smooth_obj_eval(x, n, alpha, top_idx):
    dx = 2.0 / n
    c_all = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c_act = c_all[top_idx]
    else:
        c_act = c_all
    m = c_act.max()
    shifted = alpha * (c_act - m)
    shifted -= shifted.max()
    exp_sh = np.exp(shifted)
    Z = exp_sh.sum()
    return m + np.log(Z) / alpha

def _smooth_grad_eval(x, n, alpha, top_idx):
    dx = 2.0 / n
    c_all = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c_act = c_all[top_idx]
        idx = top_idx
    else:
        c_act = c_all
        idx = np.arange(len(c_all))
    m = c_act.max()
    shifted = alpha * (c_act - m)
    shifted -= shifted.max()
    exp_sh = np.exp(shifted)
    w = exp_sh / exp_sh.sum()
    grad = np.zeros(n)
    for i in range(len(c_act)):
        if w[i] < 1e-13:
            continue
        lag = int(idx[i]) - (n - 1)
        grad += w[i] * _lag_gradient(x, n, lag)
    return grad

def _sum_eq(x, n):
    return np.sum(x) - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_eval, h, args=(n, alpha, top_idx),
                       jac=_smooth_grad_eval, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def progressive_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    schedule = [
        (30, 200), (50, 400), (100, 600), (200, 800),
        (500, 1500), (1000, 2000), (5000, 2500),
        (10000, 3000), (50000, 3000), (100000, 3000),
        (500000, 3000),
    ]
    for alpha, mi in schedule:
        if time.time() > deadline - 20:
            break
        h_new = slsqp_opt(best_h, n, alpha, mi)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def active_set_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for top_k in [3, 5, 8, 12]:
        if time.time() > deadline - 20:
            break
        for alpha in [10000, 50000, 200000]:
            if time.time() > deadline - 20:
                break
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-top_k:]
            h_new = slsqp_opt(best_h, n, alpha, 2000, top_idx=top_idx)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
    return best_h, best_c

def gradient_local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for ss in step_sizes:
        if time.time() > deadline:
            break
        max_inner = 500
        for _ in range(max_inner):
            if time.time() > deadline:
                break
            c = compute_all_lags(best_h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            g = _lag_gradient(best_h, n, lag)
            d = -ss * g
            d -= d.mean()
            h_new = best_h + d
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            else:
                ss2 = ss * 0.5
                d = -ss2 * g
                d -= d.mean()
                h_new = best_h + d
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                else:
                    break
    return best_h, best_c

def smooth_grad_local_search(h, n, deadline, alpha=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]
    for ss in step_sizes:
        if time.time() > deadline:
            break
        for _ in range(300):
            if time.time() > deadline:
                break
            g = _smooth_grad_eval(best_h, n, alpha, None)
            d = -ss * g
            d -= d.mean()
            h_new = best_h + d
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            else:
                break
    return best_h, best_c

def pair_exchange_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_all_lags(best_h, n)
    deltas = [0.01, 0.005, 0.001, 0.0005, 0.0001]
    for d_val in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(best_h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            g = _lag_gradient(best_h, n, lag)
            pos_idx = np.argsort(g)[-1]
            neg_idx = np.argsort(g)[0]
            if best_h[pos_idx] <= best_h[neg_idx]:
                break
            delta = min(d_val, (best_h[pos_idx] - best_h[neg_idx]) * 0.5, best_h[pos_idx], 1.0 - best_h[neg_idx])
            if delta < 1e-10:
                break
            best_h[pos_idx] -= delta
            best_h[neg_idx] += delta
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                improved = True
            else:
                best_h[pos_idx] += delta
                best_h[neg_idx] -= delta
    return best_h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 60:
        alloc = min(time.time() + (deadline - time.time()) * 0.5, deadline - 60)
        best_h, best_c = progressive_optimize(best_h, n, alloc)
    if time.time() < deadline - 30:
        alloc = min(time.time() + (deadline - time.time()) * 0.4, deadline - 30)
        best_h, best_c = active_set_optimize(best_h, n, alloc)
    if time.time() < deadline - 10:
        best_h, best_c = gradient_local_search(best_h, n, min(deadline - 5, time.time() + (deadline - time.time()) * 0.3))
    if time.time() < deadline - 5:
        best_h, best_c = smooth_grad_local_search(best_h, n, min(deadline - 2, time.time() + (deadline - time.time()) * 0.3))
    if time.time() < deadline - 3:
        best_h, best_c = pair_exchange_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
