```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def pnorm_objective(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return np.sum(c**p) ** (1.0/p)

def pnorm_gradient(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    s = np.sum(c**p)
    if s < 1e-30:
        return np.zeros(n)
    coeff = s**(1.0/p - 1.0)
    w = c**(p-1)  # shape (2n-1,)
    
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        k_idx = n - 1 + lag
        wk = w[k_idx]
        if abs(wk) < 1e-30:
            continue
        j_min = max(0, lag)
        j_max = min(n-1, n-1+lag)
        for j in range(j_min, j_max+1):
            g = 0.0
            if 0 <= j-lag < n:
                g += (1.0 - x[j-lag]) * dx
            if 0 <= j+lag < n:
                g -= x[j+lag] * dx
            grad[j] += coeff * wk * g
    return grad
```

Let me precompute:
- For lag >= 0: j ∈ [lag, n-1]

  - j+lag ∈ [2*lag, 2*lag+n-1-lag] = [2*lag, n-1+lag], valid when 2*lag <= n-1, i.e., lag <= (n-1)/2
  - Actually j+lag < n when j < n-lag, i.e., j ∈ [lag, n-1-lag]
  - So for j ∈ [lag, n-1-lag]: both terms active
  - For j ∈ [n-lag, n-1] (if n-lag > n-1-lag, i.e., lag > 0): only term1 active
  
- For lag < 0 (let m = -lag > 0): j ∈ [0, n-1-m]
  - j-lag = j+m ∈ [m, n-1] (always valid since j >= 0)
  - j+lag = j-m, valid when j >= m, i.e., j ∈ [m, n-1-m]
  - So for j ∈ [0, m-1]: only term1 active (with j-lag = j+m)
  - For j ∈ [m, n-1-m]: both terms active

```python
def pnorm_gradient(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    s = np.sum(c**p)
    if s < 1e-30:
        return np.zeros(n)
    coeff = s**(1.0/p - 1.0)
    
    # Build the gradient vectorized
    # For each lag, compute the contribution
    # grad[j] = coeff * sum_lag w[lag] * ((1-x[j-lag])*dx - x[j+lag]*dx)
    #         = coeff * dx * sum_lag w[lag] * (1 - x[j-lag] - x[j+lag])
    
    # Let A = sum_lag w[lag] (constant)
    # Let B[j] = sum_lag w[lag] * x[j-lag] (cross-correlation)
    # Let C[j] = sum_lag w[lag] * x[j+lag] (correlation)
    # grad[j] = coeff * dx * (A - B[j] - C[j])
    
    # We need to compute B and C efficiently
    # w is indexed by lag from -(n-1) to n-1, stored in array of length 2n-1
    # w_arr[k_idx] = w[k_idx - (n-1)] for k_idx in 0..2n-2
    
    w_full = c**(p-1)  # length 2n-1, indexed k_idx = n-1+lag
    
    # B[j] = sum_lag w[lag] * x[j-lag]
    # = sum_{k_idx} w_full[k_idx] * x[j - (k_idx - (n-1))]
    # = sum_{k_idx} w_full[k_idx] * x[j + (n-1) - k_idx]
    # Let m = j + (n-1) - k_idx, so k_idx = j + (n-1) - m
    # B[j] = sum_m x[m] * w_full[j + (n-1) - m]
    # This is the cross-correlation of x and w_full!
    # B = np.correlate(x, w_full, mode='full') but with appropriate shifting
    
    # Actually: np.correlate(a, v, 'full')[k] = sum_n v[n] * a[n+k]
    # B[j] = sum_lag w_full[n-1+lag] * x[j-lag]
    # Let k = lag, n' = n-1+lag = n-1+k
    # B[j] = sum_k w_full[n-1+k] * x[j-k]
    # This is like a cross-correlation where we correlate x with a shifted w_full
    
    # Simpler: just compute directly
    # B[j] = sum_{lag=-(n-1)}^{n-1} w_full[n-1+lag] * x[j-lag] (where x[j-lag] is 0 if out of range)
    
    # For j from 0 to n-1:
    #   lag ranges over [-(n-1), n-1] but x[j-lag] is nonzero only when 0<=j-lag<n
    #   i.e., j-n+1 <= lag <= j
    #   Combined: max(j-n+1, -(n-1)) <= lag <= min(j, n-1)
    
    B = np.zeros(n)
    C = np.zeros(n)
    for lag in range(-(n-1), n):
        wk = w_full[n-1+lag]
        if abs(wk) < 1e-30:
            continue
        # B[j] += wk * x[j-lag] for j where 0 <= j-lag < n
        # j from lag to lag+n-1 (if lag >= 0: j from lag to n-1)
        # (if lag < 0: j from 0 to n-1+lag)
        j_start = max(0, lag)
        j_end = min(n, n+lag)  # exclusive
        if j_start < j_end:
            B[j_start:j_end] += wk * x[j_start-lag : j_end-lag]
        
        # C[j] += wk * x[j+lag] for j where 0 <= j+lag < n
        # j from -lag to n-1-lag (if lag >= 0: j from 0 to n-1-lag)
        # (if lag < 0: j from -lag to n-1)
        j_start_c = max(0, -lag)
        j_end_c = min(n, n-lag)
        if j_start_c < j_end_c:
            C[j_start_c:j_end_c] += wk * x[j_start_c+lag : j_end_c+lag]
    
    A = np.sum(w_full)
    grad = coeff * dx * (A - B - C)
    return grad
```
