    return float(np.max(c) * dx)

def corr_array(h, n):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def project_to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def make_delta_j(h, n, ones):
    """Precompute delta for turning each 'one' position OFF. Shape: (len(ones), 2n-1)."""
    nj = 2 * n - 1
    m_range = np.arange(-(n - 1), n)
    delta_j = np.zeros((len(ones), nj))
    for idx, j in enumerate(ones):
        jm = j + m_range
        mask = (jm >= 0) & (jm < n)
        delta_j[idx, mask] = -(1.0 - h[jm[mask]])
        jmn = j - m_range
        mask = (jmn >= 0) & (jmn < n)
        delta_j[idx, mask] += h[jmn[mask]]
    return delta_j

def make_delta_i(h, n, i):
    """Compute delta for turning position i ON (0->1). Shape: (2n-1,)."""
    nj = 2 * n - 1
    m_range = np.arange(-(n - 1), n)
    di = np.zeros(nj)
    im = i + m_range
    mask = (im >= 0) & (im < n)
    di[mask] = 1.0 - h[im[mask]]
    imn = i - m_range
    mask = (imn >= 0) & (imn < n)
    di[mask] -= h[imn[mask]]
    return di

def vectorized_local_search(h, n, deadline, max_sweeps=500):
    """Vectorized binary local search with single swaps."""
    dx = 2.0 / n
    h = h.copy()
    corr = corr_array(h, n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    nj = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Precompute delta_j for all one positions
        delta_j = make_delta_j(h, n, ones)  # shape (n1, nj)
        
        for i in zeros:
            if time.time() > deadline:
                break
            
            di = make_delta_i(h, n, i)  # shape (nj,)
            base = corr + di  # shape (nj,)
            
            # For each j, new_corr = base + delta_j[idx], then add special terms
            # Special terms: -2 at position n-1, +1 at n-1+(j-i), +1 at n-1+(i-j)
            # First compute base max without special terms
            new_maxes = (base[np.newaxis, :] + delta_j).max(axis=1)  # shape (n1,)
            
            # Apply corrections for special terms
            # -2 at n-1: subtract 2 from base[n-1] for all j
            base_corr_n1 = base[n - 1] - 2.0
            # +1 at n-1+(j-i): for each j, position n-1+(j[i]-i)
            k1 = n - 1 + (ones - i)
            valid1 = (k1 >= 0) & (k1 < nj)
            for idx in range(len(ones)):
                val = new_maxes[idx]
                # Check the three special positions
                sp1 = base_corr_n1 + delta_j[idx, n - 1]
                if sp1 > val:
                    val = sp1
                if valid1[idx]:
                    sp2 = base[k1[idx]] + delta_j[idx, k1[idx]] + 1.0
                    if sp2 > val:
                        val = sp2
                k2_val = n - 1 + (i - ones[idx])
                if 0 <= k2_val < nj:
                    sp3 = base[k2_val] + delta_j[idx, k2_val] + 1.0
                    if sp3 > val:
                        val = sp3
                new_maxes[idx] = val
            
            best_idx = int(new_maxes.argmin())
            best_new_max = new_maxes[best_idx]
            cur_max = corr.max()
            
            if best_new_max < cur_max - 1e-12:
                j = int(ones[best_idx])
                h[i], h[j] = 1.0, 0.0
                corr = corr_array(h, n)
                cur_c = corr.max() * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, seed=42, T0=0.05, cool=0.99995, min_T=1e-8):
    """Simulated annealing on binary solutions."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    corr = corr_array(h, n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    nj = 2 * n - 1
    m_range = np.arange(-(n - 1), n)
    T = T0
    iters = 0
    stuck = 0
    B = 5000
    
    while time.time() < deadline and T > min_T:
        if iters % 10000 == 0 and stuck > 20000:
            # Restart from best with perturbation
            h = best_h.copy()
            corr = corr_array(h, n)
            cur_c = corr.max() * dx
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(len(zeros) // 10):
                ai = rng.randint(len(zeros))
                bi = rng.randint(len(ones))
                h[zeros[ai]], h[ones[bi]] = 1.0, 0.0
                zeros[ai], ones[bi] = ones[bi], zeros[ai]
            corr = corr_array(h, n)
            cur_c = corr.max() * dx
            T = T0 * 0.5
            stuck = 0
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Adaptive: occasionally do a "double swap"
        if rng.random() < 0.1 and len(zeros) >= 2 and len(ones) >= 2:
            a1, a2 = rng.choice(len(zeros), 2, replace=False)
            b1, b2 = rng.choice(len(ones), 2, replace=False)
            i1, i2 = zeros[a1], zeros[a2]
            j1, j2 = ones[b1], ones[b2]
            # Compute delta for double swap
            h_old = h.copy()
            h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0
            new_corr = corr_array(h, n)
            nc = new_corr.max() * dx
            if nc <= cur_c or (T > min_T and rng.random() < np.exp(-800.0 * (nc - cur_c) / T)):
                corr = new_corr
                cur_c = nc
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    stuck = 0
                else:
                    stuck += 1
            else:
                h[i1], h[i2], h[j1], h[j2] = 0.0, 0.0, 1.0, 1.0
                stuck += 1
        else:
            ai = rng.randint(len(zeros))
            bi = rng.randint(len(ones))
            i, j = int(zeros[ai]), int(ones[bi])
            
            # Compute delta
            delta = np.zeros(nj)
            im = i + m_range
            mask = (im >= 0) & (im < n)
            delta[mask] += 1.0 - h[im[mask]]
            imn = i - m_range
            mask = (imn >= 0) & (imn < n)
            delta[mask] -= h[imn[mask]]
            jm = j + m_range
            mask = (jm >= 0) & (jm < n)
            delta[mask] -= 1.0 - h[jm[mask]]
            jmn = j - m_range
            mask = (jmn >= 0) & (jmn < n)
            delta[mask] += h[jmn[mask]]
            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            if 0 <= k1 < nj:
                delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            if 0 <= k2 < nj:
                delta[k2] += 1.0
            
            nc = (corr + delta).max() * dx
            if nc <= cur_c or (T > min_T and rng.random() < np.exp(-800.0 * (nc - cur_c) / T)):
                h[i], h[j] = 1.0, 0.0
                corr = corr + delta
                zeros[ai] = j
                ones[bi] = i
                cur_c = nc
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    stuck = 0
                else:
                    stuck += 1
            else:
                stuck += 1
        
        T *= cool
        iters += 1
    
    return best_h, best_c

def soft_max_pgd(h, n, deadline, alphas=None, max_iter=3000):
    """Soft-max PGD for continuous relaxation."""
    if alphas is None:
        alphas = [20, 100, 500, 2000, 10000, 50000]
    
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        cur_c = best_c
        step = min(float(n) / (4.0 * alpha), 0.5)
        
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            c = np.correlate(h, 1 - h, mode='full')
            a = alpha * c
            a_max = a.max()
            exp_a = np.exp(a - a_max)
            total = exp_a.sum()
            if total < 1e-30:
                break
            p = exp_a / total
            
            # Gradient computation
            grad = np.zeros(n)
            for k in range(-(n - 1), n):
                pk = p[k + n - 1]
                if pk < 1e-14:
                    continue
                i_min = max(0, k)
                i_max = min(n - 1, n - 1 + k)
                if i_min <= i_max:
                    grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
                i_min2 = max(0, -k)
                i_max2 = min(n - 1, n - 1 - k)
                if i_min2 <= i_max2:
                    grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            direction = -grad / gnorm
            
            accepted = False
            for ls in range(30):
                s = step * (0.5 ** ls)
                h_try = np.clip(h + s * direction, 0.0, 1.0)
                # Enforce sum constraint softly
                target = n / 2.0
                s_curr = h_try.sum()
                if abs(s_curr - target) > 1e-10:
                    diff = target - s_curr
                    room_up = (1.0 - h_try).sum()
                    room_dn = h_try.sum()
                    if diff > 0 and room_up > 1e-15:
                        h_try += (diff / room_up) * (1.0 - h_try)
                    elif diff < 0 and room_dn > 1e-15:
                        h_try += (diff / room_dn) * h_try
                    h_try = np.clip(h_try, 0.0, 1.0)
                
                c_new = compute_c5(h_try, n)
                if c_new < cur_c - 1e-14:
                    h = h_try
                    cur_c = c_new
                    step = s
                    accepted = True
                    break
            
            if not accepted:
                break
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def multi_start_local(h_start, n, deadline, n_starts=5, seed=0):
    """Multiple local search starts from perturbed versions of h_start."""
    rng = np.random.RandomState(seed)
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    
    for s in range(n_starts):
        if time.time() > deadline:
            break
        if s == 0:
            h_try = h_start.copy()
        else:
            h_try = h_start.copy()
            n_perturb = n // (4 + s)
            h_bin = project_to_binary(h_try, n)
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            for _ in range(n_perturb):
                ai = rng.randint(len(zeros))
                bi = rng.randint(len(ones))
                h_bin[zeros[ai]], h_bin[ones[bi]] = 1.0, 0.0
                zeros[ai], ones[bi] = ones[bi], zeros[ai]
            h_try = h_bin
        
        phase_dl = min(deadline, time.time() + (deadline - time.time()) / (n_starts - s))
        h_res, c_res = vectorized_local_search(h_try, n, phase_dl, max_sweeps=200)
        if c_res < best_c:
            best_c = c_res
            best_h = h_res.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    hard_deadline = t0 + total
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    
    # Convert to binary for consistent starting point
    h0_bin = project_to_binary(h0, n)
    best_h = h0_bin.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Soft PGD to get a good continuous solution (15% time)
    p1 = t0 + 0.15 * total
    if time.time() < p1:
        h_soft = h0.copy()
        h_soft_pgd, c_soft = soft_max_pgd(h_soft, n, p1, alphas=[50, 200, 1000, 5000, 20000], max_iter=2000)
        # Use the soft solution's "倾向" to create a binary starting point
        h_from_soft = project_to_binary(h_soft_pgd, n)
        c_from_soft = compute_c5(h_from_soft, n)
        if c_from_soft < best_c:
            best_c = c_from_soft
            best_h = h_from_soft.copy()
    
    # Phase 2: Vectorized local search (30% time)
    p2 = t0 + 0.45 * total
    h_ls, c_ls = vectorized_local_search(best_h, n, p2, max_sweeps=500)
    if c_ls < best_c:
        best_c = c_ls
        best_h = h_ls.copy()
    
    # Phase 3: Simulated annealing (25% time)
    p3 = t0 + 0.70 * total
    h_anneal = project_to_binary(best_h, n)
    h_an, c_an = simulated_annealing(h_anneal, n, p3, seed=seed, T0=0.04, cool=0.99994)
    if c_an < best_c:
        best_c = c_an
        best_h = h_an.copy()
    
    # Phase 4: Local search after annealing (10% time)
    p4 = t0 + 0.80 * total
    h_ls2, c_ls2 = vectorized_local_search(best_h, n, p4, max_sweeps=400)
    if c_ls2 < best_c:
        best_c = c_ls2
        best_h = h_ls2.copy()
    
    # Phase 5: Second annealing with different params (10% time)
    p5 = t0 + 0.90 * total
    h_an2, c_an2 = simulated_annealing(best_h, n, p5, seed=seed + 100, T0=0.02, cool=0.99997)
    if c_an2 < best_c:
        best_c = c_an2
        best_h = h_an2.copy()
    
    # Phase 6: Final local search + multi-resolution (10% time)
    p6 = hard_deadline - 2
    if time.time() < p6 - 10:
        h_ls3, c_ls3 = vectorized_local_search(best_h, n, p6 - 15, max_sweeps=300)
        if c_ls3 < best_c:
            best_c = c_ls3
            best_h = h_ls3.copy()
    
    # Try a different resolution if time permits
    if time.time() < hard_deadline - 30:
        for nh in [n + 50, n + 100]:
            if time.time() > hard_deadline - 25:
                break
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = np.interp(xn, xo, best_h)
            hh_bin = project_to_binary(hh, nh)
            c_try = compute_c5(hh_bin, nh)
            if c_try < best_c or time.time() < hard_deadline - 40:
                h_try, c_try2 = vectorized_local_search(hh_bin, nh, min(hard_deadline - 20, time.time() + 20), max_sweeps=100)
                if c_try2 < best_c:
                    best_c = c_try2
                    best_h = h_try.copy()
                    n = nh
    
    # Final: ensure valid binary solution
    best_h = project_to_binary(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
