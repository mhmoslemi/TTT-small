def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_swap_delta(h, n, i, j, m_range):
    """Compute delta in correlation when swapping h[i]:0->1 and h[j]:1->0"""
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

    best_h = h.copy()

    for step in range(max_steps):
        if time.time() > deadline:
            break
        T = t0 * (t1 / t0) ** (steps_done / max_steps)
        oi = np.where(h == 1)[0]

            cur_c = new_c

    best_c = compute_c5(best_h, n)

    # Phase 1: Quick SLSQP at original resolution
    p1_end = t0 + 50
    h = best_h.copy()
    for alpha in [2000, 20000, 200000]:
        if time.time() > p1_end:
            break
        h = slsqp_smooth(h, n, alpha, 1500, p1_end)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Binary refine at original resolution
    p2_end = t0 + 150
    hb, cb = binary_local_search(best_h, n, p2_end, max_sweeps=8000, n_restarts=4, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: SLSQP polish from binary
    p3_end = t0 + 220
    if time.time() < p3_end:
        h = slsqp_smooth(best_h.copy(), n, 2000000, 2000, p3_end)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Multi-resolution optimization (main phase)
    p4_end = t0 + budget_s * 0.72
    best_h, best_c, cur_n = multi_resolution_optimize(best_h, cur_n, p4_end, seed=seed)

    # Phase 5: Extended simulated annealing at best resolution
    p5_end = t0 + budget_s * 0.88
    if time.time() < p5_end:
        # Multiple SA runs with different parameters
        sa_configs = [
            (0.05, 0.001, 200000, seed + 100),
            (0.02, 0.0005, 150000, seed + 200),
            (0.1, 0.002, 150000, seed + 300),
        ]
        for (t0sa, t1sa, nsteps, sasa) in sa_configs:
            if time.time() > p5_end:
                break
            remaining = p5_end - time.time()
            hs, cs = simulated_annealing(best_h, cur_n, min(time.time() + remaining * 0.4, p5_end),
                                         t0=t0sa, t1=t1sa, max_steps=nsteps, seed=sasa)
            if cs < best_c:
                best_c = cs
                best_h = hs.copy()
                # Quick binary polish after SA improvement
                if time.time() < p5_end - 5:
                    hr, cr = binary_local_search(best_h, cur_n, min(time.time() + 15, p5_end - 2),
                                                 max_sweeps=3000, n_restarts=1, seed=seed)
                    if cr < best_c:
                        best_c = cr
                        best_h = hr.copy()

    # Phase 6: Final SLSQP + binary polish
    if time.time() < deadline - 10:
        h = best_h.copy()
        if not np.all((h == 0) | (h == 1)):
            h = slsqp_smooth(h, cur_n, 5000000, 2000, deadline - 5)
        else:
            h = slsqp_smooth(h + 0.02 * rng.randn(cur_n), cur_n, 5000000, 2000, deadline - 5)
            h = project_feasible(h, cur_n)
        c = compute_c5(h, cur_n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if time.time() < deadline - 3:
            hb, cb = binary_local_search(best_h, cur_n, deadline, max_sweeps=5000, n_restarts=2, seed=seed+500)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

    # Final projection and exact evaluation
    best_h = project_feasible(best_h, cur_n)
    best_c = compute_c5(best_h, cur_n)
    return best_h, best_c, cur_n
