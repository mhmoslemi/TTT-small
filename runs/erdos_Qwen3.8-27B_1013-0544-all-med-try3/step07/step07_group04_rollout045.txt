def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:

            if tot < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m >= n:
        pass
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_active(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, alpha, maxiter=5000, ftol=1e-15, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    def obj(x):
        if top_idx is not None:
            return smooth_obj_active(x, n, top_idx, alpha)
        return smooth_obj_full(x, n, alpha)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_descent(h, n, deadline, top_m=20, base_lr=0.005):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    it = 0
    patience = 0
    max_patience = 500
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        if c.max() < best_c - 1e-13:
            best_c = c.max()
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        if patience > max_patience:
            break
        top_vals = np.sort(c)[-top_m:]
        active_mask = c >= top_vals[0] - 1e-15
        # Use weights based on how close to max
        weights = np.zeros(len(c))
        thr = top_vals[0] * 0.95
        weights[c >= thr] = 1.0
        if weights.sum() > top_m:
            top_idx = np.argsort(c)[-top_m:]
            weights = np.zeros(len(c))
            weights[top_idx] = 1.0
        subgrad = np.zeros(n)
        active_lags = np.where(weights > 0)[0]
        for idx in active_lags:
            lag = idx - (n - 1)
            subgrad += weights[idx] * lag_gradient(h, n, lag)
        subgrad /= max(1.0, weights.sum())
        # Adaptive learning rate
        lr = base_lr / (1.0 + 0.05 * it)
        # Progressively reduce
        lr *= math.exp(-0.002 * it)
        h_try = h - lr * subgrad
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try <= c.max() + 1e-14:
            h = h_try
        else:
            # Halve step
            h_try = h - 0.5 * lr * subgrad
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            h = h_try
        it += 1
    h = project_feasible(h, n)
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h
    return best_h, best_c

    # Phase 1: Full smooth with moderate alpha

                h = h_trial

        c_trial = compute_c5(h_trial, n)

def targeted_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    # Targeted coordinate adjustments on the worst lag
    for _ in range(3):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            # Minimize sum of h*(1-h): push h toward binary
            for i in range(n):
                if h[i] < 0.01:
                    h[i] = 0.0
                elif h[i] > 0.99:
                    h[i] = 1.0
                    h[i] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
            continue
        m = abs(lag)
        # For this lag, we want to reduce sum of h[i]*(1-h[i+lag])
        # Move mass from high h to low h within the paired positions
        if lag > 0:
            pairs = [(i, i + m) for i in range(n - m) if 0 <= i + m < n]
        else:
            ml = -lag
            pairs = [(i, i - ml) for i in range(ml, n) if 0 <= i - ml < n]
        # Sort pairs by "dissimilarity" and try to reduce it
        for i, j in pairs:
            if time.time() > deadline:
                break
            if h[i] > h[j]:
                # Moving mass from i to j reduces h[i]*(1-h[j]) for this pair
                delta = 0.01
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0, 1)
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new >= best_c + 1e-14:
                    h[i] += delta
                    h[j] -= delta
                    h = np.clip(h, 0, 1)
                    h = project_feasible(h, n)

                delta = 0.01
                h[j] -= delta
                h[i] += delta
                h = np.clip(h, 0, 1)
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new >= best_c + 1e-14:
                    h[j] += delta
                    h[i] -= delta
                    h = np.clip(h, 0, 1)
                    h = project_feasible(h, n)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
