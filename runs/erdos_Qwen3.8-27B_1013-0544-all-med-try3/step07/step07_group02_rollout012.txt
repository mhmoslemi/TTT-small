def grad_smooth_max(h, n, alpha, top_k=None):
    """Compute gradient of smooth max of lag values."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    
    if top_k is not None:
        # Only consider top_k lags
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_active = c[idx]
        weights = np.exp(alpha * (c_active - c_active.max()))
        weights /= weights.sum()
        # Build full weight vector
        w_full = np.zeros(2*n-1)
        w_full[idx] = weights
    else:
        w_full = np.exp(alpha * (c - c.max()))
        w_full /= w_full.sum()
    
    # Compute gradient as weighted sum of lag gradients
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w_full[k] < 1e-10:
            continue
        lag = k - (n-1)
        m = abs(lag)
        if m >= n:
            continue
        if lag == 0:
            grad += w_full[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[m:] += w_full[k] * (1.0 - h[:n-m])
            grad[:n-m] += w_full[k] * (-h[m:])
        else:
            grad[:n-m] += w_full[k] * (1.0 - h[m:])
            grad[m:] += w_full[k] * (-h[:n-m])

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += (-h[m:n])
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += (-h[0:n-m])
    return grad

def smooth_max_grad(h, n, alpha, top_k=5):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_k < 2 * n - 1:
        idx = np.argpartition(c, -(top_k))[-(top_k):]
        c_top = c[idx]
        s = alpha * (c_top - c_top.max())
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        grad = np.zeros(n)
        for ki, ii in enumerate(idx):
            if w[ki] < 1e-10:
                continue
            lag = int(ii) - (n - 1)
            grad += w[ki] * grad_lag(h, n, lag)
    else:
        s = alpha * (c - c.max())
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[k] * grad_lag(h, n, lag)
    return grad * dx, c

def gradient_phase(h, n, deadline, rng, alpha_start=30.0):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_start
    lr = 0.08
    mom = 0.91
    v = np.zeros(n)
    top_k = 3
    stall = 0
    it = 0

    while time.time() < deadline:
        g, c_vals = smooth_max_grad(h, n, alpha, top_k)
        v = mom * v + lr * g
        h_new = project_feasible(h - v, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1

        if stall > 150:
            lr *= 0.65
            v *= 0.4
            stall = 0
            if lr < 0.008:
                alpha = min(alpha * 1.4, 5000.0)
                lr = 0.02
                top_k = min(top_k + 2, 12)

        if stall > 800:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            v = np.zeros(n)
            lr = 0.04
            alpha = max(20.0, alpha * 0.7)
            stall = 0

        it += 1
    return best_h, best_c, it

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.04, 0.02, 0.01, 0.005, 0.002]

    for rnd in range(500):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-6:]
        rnd_improved = 0
        for k_star in top_idx:
            lag = int(k_star) - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            for d in deltas:
                if time.time() > deadline:
                    break
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-7:
                        delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.5)
                        if delta < 1e-9:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            rnd_improved += 1
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-7:
                        delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.5)
                        if delta < 1e-9:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            rnd_improved += 1
                        else:
                            h[i] -= delta
                            h[j] += delta
                if rnd_improved > 80:
                    break
            if rnd_improved > 80:
                break
        if rnd_improved == 0:
            break
    return best_h, best_c

def paired_swap_search(h, n, deadline, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_c_all = compute_all_lags(best_h, n)

    for it in range(200000):
        if time.time() > deadline:
            break
        if it % 500 == 0:
            best_c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(best_c_all))
        lag = k_star - (n - 1)
        m = abs(lag)
        if m == 0 or m >= n:
            break
        i = rng.randint(0, n - m)
        j = i + m
        if abs(h[i] - h[j]) < 1e-8:
            continue
        if h[i] > h[j]:
            delta = min(0.06 * rng.random(), h[i] * 0.4, (1.0 - h[j]) * 0.4)
        else:
            delta = -min(0.06 * rng.random(), h[j] * 0.4, (1.0 - h[i]) * 0.4)
        if abs(delta) < 1e-9:
            continue
        h[i] -= delta
        h[j] += delta
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
        else:
            h = best_h.copy()
    return best_h, best_c

def multi_resolution_opt(h_best, c_best, n_best, deadline, rng):
    n_tries = [400, 500, 600, 700]
    if n_best not in n_tries:
        n_tries.insert(0, n_best)

    for n_try in n_tries:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(150, remaining * 0.3)
        if alloc < 40:
            break

        if n_try == n_best:
            h_start = h_best.copy()
        else:
            x_old = np.linspace(0, 2, len(h_best))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, h_best)
            h_start = project_feasible(h_start, n_try)

        grad_deadline = time.time() + alloc * 0.6
        h_g, c_g, _ = gradient_phase(h_start, n_try, grad_deadline, rng)
        c_g = compute_c5(h_g, n_try)

        ls_deadline = time.time() + min(alloc * 0.35, deadline - time.time() - 10)
        h_ls, c_ls = local_search(h_g, n_try, ls_deadline)

        if c_ls < c_best:
            c_best = c_ls
            h_best = h_ls.copy()
            n_best = n_try

    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Main gradient optimization at initial resolution
    grad_deadline = t0 + min(350, (deadline - t0) * 0.4)
    h_g, c_g, _ = gradient_phase(best_h, n_init, grad_deadline, rng)
    if c_g < best_c:
        best_c = c_g
        best_h = h_g

    # Phase 2: Local search
    if time.time() < deadline - 60:
        ls_deadline = min(deadline - 40, time.time() + 200)
        h_ls, c_ls = local_search(best_h, n_init, ls_deadline)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Phase 3: Paired swap random search
    if time.time() < deadline - 30:
        ms_deadline = min(deadline - 20, time.time() + 80)
        h_ms, c_ms = paired_swap_search(best_h, n_init, ms_deadline, rng)
        if c_ms < best_c:
            best_c = c_ms
            best_h = h_ms

    # Phase 4: Multi-resolution
    if time.time() < deadline - 100:
        best_h, best_c, n_best = multi_resolution_opt(
            best_h, best_c, len(best_h), deadline, rng)

    # Phase 5: Final gradient + local search polish
    n_final = len(best_h)
    if time.time() < deadline - 20:
        gdl = min(deadline - 10, time.time() + 60)
        h_g, c_g, _ = gradient_phase(best_h, n_final, gdl, rng, alpha_start=50.0)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g

    if time.time() < deadline - 5:
        ldl = deadline - 2
        h_ls, c_ls = local_search(best_h, n_final, ldl)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Final projection and validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
