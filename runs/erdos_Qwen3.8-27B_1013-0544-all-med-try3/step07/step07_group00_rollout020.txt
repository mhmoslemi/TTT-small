                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': 10000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
                    global_best_c = min(global_best_c, c_new)
                    global_best_h = h.copy()
                    global_best_n = n
            except Exception:
                pass
    
    # Phase 4: Local line search
    if time.time() < deadline - 40:
        t_local = min(time.time() + 30, deadline - 30)
        h, best_c = local_line_search(h, n, t_local, base_step=0.005)
        global_best_c = min(global_best_c, best_c)
        global_best_h = h.copy()
        global_best_n = n
    
    # Phase 5: Very fine local search
    if time.time() < deadline - 20:
        t_fine = min(time.time() + 15, deadline - 5)
        h, best_c = finer_local_search(h, n, t_fine)
        global_best_c = min(global_best_c, best_c)
        global_best_h = h.copy()
        global_best_n = n
    
    # Phase 6: Try perturbations and re-optimize
    if time.time() < deadline - 15:
        for trial in range(5):
            if time.time() > deadline - 8:
                break
            h_pert = h.copy()
            mask = rng.random(n) < 0.05
            h_pert[mask] += rng.normal(0, 0.02, mask.sum())
            h_pert = project_feasible(h_pert, n)
            c_pert = compute_c5(h_pert, n)
            
            if c_pert < best_c:
                h = h_pert
                best_c = c_pert
            else:
                # Still try to optimize from perturbed point
                pass
            
            # Quick L-BFGS from perturbed point
            try:
                res = minimize(smooth_obj, h_pert, jac=smooth_grad,
                              args=(n, 100000, 0.1), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 1000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 5000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
                    global_best_c = min(global_best_c, c_new)
                    global_best_h = h.copy()
                    global_best_n = n
            except Exception:
                pass
    
    # Final: also check at original resolution
    h_orig_final = project_feasible(h_init, n_init)
    c_orig_final = compute_c5(h_orig_final, n_init)
    if c_orig_final < global_best_c:
        global_best_c = c_orig_final
        global_best_h = h_orig_final
        global_best_n = n_init
    
    # Final cleanup
    global_best_h = project_feasible(global_best_h, global_best_n)
    global_best_c = compute_c5(global_best_h, global_best_n)
    
    return global_best_h, global_best_c, global_best_n
