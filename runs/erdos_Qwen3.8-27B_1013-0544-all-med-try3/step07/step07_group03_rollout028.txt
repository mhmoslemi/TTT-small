from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = c - m
    return float(m + np.log(np.sum(np.exp(alpha * shifted))) / alpha)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def bottleneck_subgradient_step(h, n, eta, n_steps=10):
    """Take steps using subgradient of the max lag."""
    best_c = compute_c5(h, n)
    for step in range(n_steps):
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(c)
        # lag s = k_max - (n-1)
        s = k_max - (n - 1)
        # Compute gradient of c[k_max] w.r.t. h
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            # h[j] appears as h[i] in c: contributes -h[j+s] if valid
            if 0 <= j + s < n:
                val -= h[j + s]
            # h[j] appears as h[i+s] in c: contributes (1-h[j-s]) if valid
            if 0 <= j - s < n:
                val += (1 - h[j - s])
            grad[j] = val
        grad *= dx
        # Take step
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
            eta *= 0.95  # slightly reduce step size
        else:
            eta *= 0.5  # reduce step size more
            if eta < 1e-10:
                break
    return h, best_c

def simulated_annealing(h, n, deadline, rng, T0=0.01, T_end=0.0001):
    """Simulated annealing with pair moves."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    T = T0
    
    # Determine number of steps we can do (rough estimate)
    test_start = time.time()
    for _ in range(100):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.05, 0.05)
        if h[i] + d <= 1.0 and h[j] - d >= 0.0:
            h[i] += d; h[j] -= d
            _ = compute_c5(h, n)
            h[i] -= d; h[j] += d
    elapsed = time.time() - test_start
    steps_per_sec = 100.0 / max(elapsed, 1e-6)
    
    total_steps = int(steps_per_sec * (deadline - time.time()) * 0.8)
    total_steps = max(total_steps, 100)
    
    for step in range(total_steps):
        if time.time() > deadline:
            break
        # Update temperature
        progress = step / max(total_steps, 1)
        T = T0 * (T_end / T0) ** progress
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try a few delta sizes
        d = rng.choice([0.01, 0.02, 0.05, 0.1]) * rng.choice([1, -1])
        if h[i] + d > 1.0 or h[j] - d < 0.0:
            continue
        h[i] += d; h[j] -= d
        c = compute_c5(h, n)
        delta = c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            current_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas):

    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):

                lo = max(0, i - radius)

                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
