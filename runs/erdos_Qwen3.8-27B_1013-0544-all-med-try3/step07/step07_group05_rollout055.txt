import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_onto_box_sum(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    h = h.copy().clip(0, 1)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-10
            if not free.any():
                break
            h[free] += diff / free.sum()
        else:
            free = h > 1e-10
            if not free.any():
                break
            h[free] += diff / free.sum()
        h.clip(0, 1, out=h)
    return h

def grad_c_at_lag(h, n, lag_idx):
    """
    Gradient of c[lag_idx] (the raw correlation before *dx) w.r.t. h.
    
    For output index i (0 to 2n-2), the lag k = i - (n-1).
    c[i] = sum over valid n of h[n+k] * (1-h[n])
    
    For k >= 0: c[i] = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
    For k < 0: c[i] = sum_{n=-k}^{n-1} h[n+k]*(1-h[n])
    """
    k = lag_idx - (n - 1)
    grad = np.zeros(n)
    
    # For each valid pair (n, n+k) where both in [0, n-1]:
    if k >= 0:
        for n_idx in range(n - k):
            # term: h[n_idx + k] * (1 - h[n_idx])
            # d/dh[j]: if j == n_idx+k: (1-h[n_idx])
            #          if j == n_idx: -h[n_idx+k]
            grad[n_idx + k] += (1.0 - h[n_idx])
            grad[n_idx] -= h[n_idx + k]
    else:
        for n_idx in range(-k, n):
            # term: h[n_idx + k] * (1 - h[n_idx])
            j1 = n_idx + k  # >= 0 since n_idx >= -k
            j2 = n_idx
            grad[j1] += (1.0 - h[j2])
            grad[j2] -= h[j1]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    # Log-sum-exp approximation to max
    shifted = alpha * (c - m)
    logger = np.log(np.sum(np.exp(shifted))) / alpha
    return m + logger

def slsqp_run(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda: np.ones(n)}]
    res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_feasible(res.x, n)

def subgrad_descent(h, n, n_iter=5000, lr_init=1e-4, deadline=None):
    """Projected subgradient descent on max correlation"""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    dx = 2.0 / n
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        # Find active lags (top few)
        k_star = int(np.argmax(c))
        
        # Compute gradient at the worst lag
        k = k_star - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
        else:
            kp = -k
            grad[:n-kp] += (1.0 - h[kp:])
            grad[kp:] -= h[:n-kp]
        
        # Average with second-worst for stability
        c_sorted = np.sort(c)[::-1]
        if c_sorted[1] > c_sorted[0] - 1e-6:
            k_star2 = int(np.argsort(c)[-2])
            k2 = k_star2 - (n - 1)
            grad2 = np.zeros(n)
            if k2 >= 0:
                grad2[k2:] += (1.0 - h[:n-k2])
                grad2[:n-k2] -= h[k2:]
            else:
                kp2 = -k2
                grad2[:n-kp2] += (1.0 - h[kp2:])
                grad2[kp2:] -= h[:n-kp2]
            grad = 0.5 * (grad + grad2)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            # Slightly increase lr
            lr *= 1.01
        else:
            # Decrease lr
            lr *= 0.95
            h = h_new  # still move (to escape plateaus)
            # Actually, let's not move if it's worse
            h = h  # revert
            if lr < 1e-8:
                break
    
    return best_h, best_c

Penalty method:
min_h max_k c[k] + μ * (sum(h) - n/2)² + ν * ∑ max(0, -h[i])² + ν * ∑ max(0, h[i]-1)²

For the box constraints, we can use a barrier: -∑ log(h[i]) - ∑ log(1-h[i])

This gives an unconstrained problem that's smooth, suitable for L-BFGS-B (which we can run for many iterations).

def barrier_obj(x, n, mu, barrier):
    """Penalty + barrier objective"""
    # Smooth max of correlation
    c = np.correlate(x, 1.0 - x, mode='full')
    # But x might not be in [0,1] or have correct sum
    # Apply soft clipping
    x_soft = x  # We'll handle via barrier

    # Sum penalty
    sum_pen = mu * (np.sum(x) - n/2.0)**2
    # Barrier
    x_clip = np.clip(x, 1e-8, 1-1e-8)

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def project_feasible(h, n):
    target = n/2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            mask = h < 1.0 - 1e-12

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
    m = c.max()
    s = c - m
    # Numerically stable log-sum-exp
    if alpha > 1e6:
        # Approximate: just return max + tiny correction
        top5 = np.argsort(c)[-5:]
        top_vals = c[top5] - m
        w = np.exp(alpha * top_vals)
        lse = np.log(np.sum(w)) / alpha
        return m + lse
    return m + np.log(np.sum(np.exp(alpha * s))) / alpha

def coordinate_polish(h, n, deadline, deltas):
    """Polish using local coordinate moves"""
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Try moving mass from i to i±1 (and a few others)
                for di in [-3, -2, -1, 1, 2, 3]:
                    j = i + di
                    if j < 0 or j >= n:
                        continue
                    # Move: h[i] += d, h[j] -= d
                    hi, hj = h[i] + d, h[j] - d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c
