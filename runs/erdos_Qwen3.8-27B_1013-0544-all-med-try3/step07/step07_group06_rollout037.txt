def vectorized_binary_search(h_init, n, deadline, max_sweeps=100000):

    target = n // 2

    cur_c = corr.max() * dx

    A = np.zeros((n, 2*n-1))

        ones = np.where(h == 1)[0]

                if 0 <= k2 < 2*n-1:
                    diffs[bi, k2] += 2

            best_val = new_maxes[best_bi]

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h to {h : sum(h) = n/2, 0 <= h <= 1}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    def objective(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def constraint(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constraint}]
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h

        k_star = int(np.argmax(corr))

def golden_ratio_init(n):
    """Initialize using golden ratio spacing for binary solution."""
    h = np.zeros(n)
    target = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.array([int((i * phi) % n) for i in range(target)])

            h = project_feasible(h, n)
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            A = compute_A_matrix(h, n)
            swaps += 1
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = golden_ratio_init(n0)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    
    # Phase 1: SLSQP annealing from initial (n=250)
    p1 = t0 + 0.18 * (budget_s - 3)
    h_sol, c_sol = annealed_slsqp(best_h, n0, p1, alpha_start=200, alpha_end=5000000, n_alphas=6)
    if c_sol < best_c:
        best_c = c_sol
        best_h = h_sol.copy()
    
    # Phase 2: Top-constraint SLSQP
    p2 = t0 + 0.28 * (budget_s - 3)
    if time.time() < p2:
        for top_k in [3, 5, 8]:
            if time.time() > p2 - 10:
                break
            h_tc = slsqp_top_constraints(best_h, n0, top_k=top_k, margin=2e-4, maxiter=500, deadline=p2 - 5)
            c_tc = compute_c5(h_tc, n0)
            if c_tc < best_c:
                best_c = c_tc
                best_h = h_tc.copy()
    
    # Phase 3: Subgradient polish
    p3 = t0 + 0.38 * (budget_s - 3)
    if time.time() < p3:
        h_sg, c_sg = subgradient_polish(best_h, n0, max_iters=30000, deadline=p3)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg.copy()
    
    # Phase 4: SLSQP after subgradient
    p4 = t0 + 0.45 * (budget_s - 3)
    if time.time() < p4:
        h_sol, c_sol = annealed_slsqp(best_h, n0, p4, alpha_start=5000, alpha_end=10000000, n_alphas=5)
        if c_sol < best_c:
            best_c = c_sol
            best_h = h_sol.copy()
    
    # Phase 5: Multi-resolution exploration
    p5 = t0 + 0.62 * (budget_s - 3)
    for nh in [200, 300, 220, 280]:
        if time.time() > p5 - 30:
            break
        if nh == n0:
            continue
        # Interpolate to new resolution
        xo = np.linspace(0, 2, n0)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        
        # Optimize at new resolution
        rem = min(p5 - 20 - time.time(), 120)
        if rem < 10:
            continue
        h_r, c_r = annealed_slsqp(hh, nh, time.time() + rem, alpha_start=500, alpha_end=2000000, n_alphas=4)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
            best_n = nh
            n0 = nh
    
    # Phase 6: Binary local search for good starting point
    p6 = t0 + 0.75 * (budget_s - 3)
    hb, cb = vectorized_binary_search(best_h, n0, p6, max_swaps=100000)
    
    # Use binary solution as a starting point for SLSQP
    if time.time() < p6 + 30:
        h_binarized = project_feasible(hb + 0.3, n0)
        h_from_bin, c_from_bin = annealed_slsqp(h_binarized, n0, min(deadline - 5, time.time() + 60), alpha_start=500, alpha_end=3000000, n_alphas=4)
        if c_from_bin < best_c:
            best_c = c_from_bin
            best_h = h_from_bin.copy()
    
    # Also try binary solution directly
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 7: Extended SLSQP on best
    p7 = t0 + 0.90 * (budget_s - 3)
    if time.time() < p7:
        h_sol, c_sol = annealed_slsqp(best_h, n0, p7, alpha_start=10000, alpha_end=20000000, n_alphas=6)
        if c_sol < best_c:
            best_c = c_sol
            best_h = h_sol.copy()
    
    # Phase 8: Continuous local search
    p8 = t0 + 0.95 * (budget_s - 3)
    if time.time() < p8:
        h_cl, c_cl = continuous_local_search(best_h, n0, p8, max_swaps=50000)
        if c_cl < best_c:
            best_c = c_cl
            best_h = h_cl.copy()
    
    # Phase 9: Final polish
    if time.time() < deadline - 2:
        # Quick SLSQP
        h_final = slsqp_optimize(best_h, n0, 5000000, 2000, deadline - 1)
        c_final = compute_c5(h_final, n0)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
        
        # Quick binary search
        if time.time() < deadline - 1:
            hb2, cb2 = vectorized_binary_search(best_h, n0, deadline - 0.5, max_swaps=50000)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n0)
    best_c = compute_c5(best_h, n0)
    
    return best_h, best_c, n0
