    """Find optimal h[i] given others fixed, minimizing max over active shifts."""
    # c[k=0] as function of t: C0_rest + t*(1-t)
    # c[k] for k≠0 as function of t: base_k + b_k*t
    
    # Precompute for all active shifts
    # k=0 (shift index n-1 in correlate)
    c0_rest = float(np.sum(h*(1-h))) - h[i]*(1-h[i])
    # c0(t) = c0_rest + t*(1-t)
    
    # For other shifts
    shift_indices = active_shifts  # indices in the correlate array
    # For shift index s, the actual shift is k = s - (n-1)
    # c[s] involves h[i] as:
    #   h[i]*(1-h[i-k]) if 0<=i-k<n  [h[i] appears as h[n+k] with n=i-k]
    #   h[i+k]*(1-h[i]) if 0<=i+k<n  [h[i] appears in (1-h[n]) with n=i]
    
    bases = []
    slopes = []
    for s in shift_indices:
        k = s - (n-1)
        if k == 0:
            continue  # handled separately
        base = 0
        slope = 0
        # term: h[i]*(1-h[i-k]) if 0<=i-k<n
        if 0 <= i-k < n:
            slope += (1 - h[i-k])
        # term: h[i+k]*(1-h[i]) if 0<=i+k<n
        if 0 <= i+k < n:
            base += h[i+k]
            slope += (-h[i+k])
        bases.append(base)
        slopes.append(slope)
    
    # Now minimize max(c0_rest + t*(1-t), bases[0]+slopes[0]*t, ..., bases[m-1]+slopes[m-1]*t)
    # subject to 0<=t<=1
    
    def obj(t):
        val = c0_rest + t*(1-t)  # k=0
        for b, sl in zip(bases, slopes):
            v = b + sl*t
            if v > val:
                val = v
        return val
    
    # Golden section search on [0,1]
    # But first, we also need to include the shift k=0's actual index
    # Actually k=0 corresponds to shift index s = n-1 in the correlate
    # Let me make sure I include it in active_shifts or handle it separately
    
    # Ternary search
    a, b = 0.0, 1.0
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    c = b - gr*(b-a)
    d = a + gr*(b-a)
    fc = obj(c)
    fd = obj(d)
    for _ in range(60):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr*(b-a)
            fc = obj(c)
        else:
            a = c; c = d; fc = fd
            d = a + gr*(b-a)
            fd = obj(d)
    
    t_opt = (a + b) / 2
    return t_opt

def compute_c5(h, n):

def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def find_active_shifts(h, n, k=10, rel_thresh=0.005):

    c = compute_c5_vec(h, n)
    max_val = c.max()
    threshold = max_val * (1 - rel_thresh)
    active = np.where(c >= threshold)[0]
    if len(active) > k:
        active = active[np.argsort(-c[active])][:k]
    return active, max_val

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n/2.0)**2

    m = c.max()

import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def find_active_shifts(h, n, top_k=15, rel_thresh=0.01):
    c = compute_c5_vec(h, n)
    max_val = c.max()
    threshold = max_val * (1.0 - rel_thresh)
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c[active])
        active = active[order[:top_k]]
    active_set = set(active.tolist())
    active_set.add(n - 1)  # always include shift 0
    return sorted(active_set), float(max_val)

def coord_min_1d(i, h, n, active_shifts, c0_total):
    """Find t in [0,1] minimizing max over active shifts of c_k(t)."""
    c0_exclude_i = c0_total - h[i] * (1.0 - h[i])
    
    shift_info = []
    for s in active_shifts:
        k = s - (n - 1)
        if k == 0:
            continue
        base = 0.0
        slope = 0.0
        ik = i - k
        if 0 <= ik < n:
            slope += (1.0 - h[ik])
        ipk = i + k
        if 0 <= ipk < n:
            base += h[ipk]
            slope -= h[ipk]
        shift_info.append((base, slope))
    
    ns = len(shift_info)
    
    def evaluate(t):
        val = c0_exclude_i + t * (1.0 - t)
        for b, sl in shift_info:
            v = b + sl * t
            if v > val:
                val = v
        return val
    
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = evaluate(c)
    fd = evaluate(d)
    for _ in range(45):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = evaluate(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = evaluate(d)
    
    t_opt = (a + b) / 2.0
    # Clamp
    if t_opt < 0.0:
        t_opt = 0.0
    elif t_opt > 1.0:
        t_opt = 1.0
    return t_opt

def coordinate_descent(h, n, max_sweeps=2000, time_limit=None, top_k=15, rel_thresh=0.01):
    """Coordinate descent with active-set tracking."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    no_improve_count = 0
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        
        active_shifts, max_val = find_active_shifts(h, n, k=top_k, rel_thresh=rel_thresh)
        c0_total = float(np.sum(h * (1.0 - h)))
        
        moved = False
        for i in range(n):
            t_opt = coord_min_1d(i, h, n, active_shifts, c0_total)
            if abs(t_opt - h[i]) > 1e-13:
                h[i] = t_opt
                c0_total = c0_total - (h[i] * (1.0 - h[i])) + (t_opt * (1.0 - t_opt))
                moved = True
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count > 30:
            # Perturb to escape local optimum
            top_k = min(top_k + 5, 35)
            rel_thresh = max(rel_thresh * 1.5, 0.05)
            # Small random perturbation
            rng = np.random.RandomState(sweep * 7919)
            perturb = rng.randn(n) * 0.01
            h = project_feasible(h + perturb, n)
            no_improve_count = 0
    
    return best_h, best_c

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_refine(h, n, deadline, alphas=(5000, 20000, 100000), mu=10.0):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def subgradient_descent(h, n, time_limit=None, learning_rate=0.001, decay=0.95):
    """Subgradient method on the true max."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = learning_rate
    n_iter = 0
    
    while time.time() < time_limit:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Active set: shifts within 0.1% of max
        thresh = max_val * 0.999
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        # Subgradient
        dx = 2.0 / n
        grad = np.zeros(n)
        for s in active:
            k = s - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        lr *= decay
        n_iter += 1
        if lr < 1e-8:
            lr = 0.001  # reset
            # Add small noise
            rng = np.random.RandomState(n_iter)
            h = project_feasible(h + rng.randn(n) * 0.005, n)
    
    return best_h, best_c
