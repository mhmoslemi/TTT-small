    S = np.zeros((n, 2*n-1))
    for L_col in range(2*n-1):
        L = L_col - (n-1)
        # (1-h[p-L])_valid for p in range(n)
        # p-L in [0, n-1] => L <= p <= n-1+L
        p_lo = max(0, L)

            S[idx, L_col] += 1.0 - h[idx - L]

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_sensitivity(h, n):
    """S[p, L_col] = (1-h[p-L])_valid - h[p+L]_valid where L = L_col - (n-1)"""
    S = np.zeros((n, 2*n-1))
    for L_col in range(2*n-1):
        L = L_col - (n-1)
        # (1-h[p-L])_valid: p-L in [0,n-1] => max(0,L) <= p <= min(n-1, n-1+L)
        p_lo = max(0, L)
        p_hi = min(n, n+L)  # exclusive
        if p_hi > p_lo:
            idx = np.arange(p_lo, p_hi)
            S[idx, L_col] += 1.0 - h[idx - L]
        # -h[p+L]_valid: p+L in [0,n-1] => max(0,-L) <= p <= min(n-1, n-1-L)
        p_lo2 = max(0, -L)
        p_hi2 = min(n, n-L)  # exclusive
        if p_hi2 > p_lo2:
            idx2 = np.arange(p_lo2, p_hi2)
            S[idx2, L_col] -= h[idx2 + L]
    return S

    return m + np.log(np.sum(np.exp(shifted))) / alpha
