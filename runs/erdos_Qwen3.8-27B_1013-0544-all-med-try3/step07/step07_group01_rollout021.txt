def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)

    return grad

            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    
    return best_h, best_c

def do_small_pair_search(h, n, deadline, radius=15):
    """Very limited pair search for final polish."""
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = False
        max_fails = 5000
        fails = 0
        while not improved and fails < max_fails and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    fails += 1
                    if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                        old_i, old_j = h[i], h[j]
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            fails = 0
                        else:
                            h[i], h[j] = old_i, old_j
                    if 0 <= h[i] - d and h[j] + d <= 1:
                        old_i, old_j = h[i], h[j]
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            fails = 0
                        else:
                            h[i], h[j] = old_i, old_j
                            fails += 1
            if not improved:
                fails += 1
    
    return h, best_c
