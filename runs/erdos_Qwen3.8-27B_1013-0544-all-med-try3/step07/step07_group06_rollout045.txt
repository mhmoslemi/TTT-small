def full_recompute(h, n):
    corr = np.correlate(h, 1 - h, mode='full')
    return corr, float(corr.max()) * (2.0 / n)

def simulated_annealing(h_bin, n, deadline, max_iters, seed=42):
    """Simulated annealing on binary representation."""
    rng = np.random.RandomState(seed)
    target = n // 2
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    corr, cur_c = full_recompute(h_bin, n)
    best_c = cur_c
    best_h = h_bin.copy()
    
    # Temperature schedule
    T_start = max(cur_c * 2.0 / n, 0.01)  # initial temperature
    T_end = T_start * 1e-6
    n_trials = max_iters
    dx = 2.0 / n
    
    h = h_bin.copy()
    
    iter_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 1000 == 0 and it > 0:
            # Check time more frequently
            pass
        
        # Temperature
        frac = it / max_iters
        T = T_start * (T_end / T_start) ** frac
        
        # Pick a random 1 and a random 0 to swap
        o_list = list(ones)
        z_list = list(zeros)
        if len(o_list) == 0 or len(z_list) == 0:
            break
        i = o_list[rng.randint(len(o_list))]
        j = z_list[rng.randint(len(z_list))]
        
        # Try the swap
        h[i], h[j] = 0.0, 1.0
        corr_new, c_new = full_recompute(h, n)
        delta = c_new - cur_c
        
        if delta < -1e-15:
            # Accept
            cur_c = c_new
            corr = corr_new
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # Metropolis acceptance
            if T > 1e-15 and rng.random() < np.exp(-delta / T):
                cur_c = c_new
                corr = corr_new
                ones.discard(i)
                ones.add(j)
                zeros.discard(j)
                zeros.add(i)
            else:
                # Reject
                h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c

def hill_climb_binary(h_bin, n, deadline, max_sweeps=500, seed=42):
    """Greedy hill climbing on binary representation."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr, cur_c = full_recompute(h_bin, n)
    best_c = cur_c
    best_h = h_bin.copy()
    h = h_bin.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Randomize order
        z_order = list(zeros)
        rng.shuffle(z_order)
        for j in z_order:
            if time.time() > deadline:
                break
            o_order = list(ones)
            rng.shuffle(o_order)
            best_swap = None
            best_nc = cur_c
            for i in o_order:
                h[i], h[j] = 0.0, 1.0
                _, nc = full_recompute(h, n)
                h[i], h[j] = 1.0, 0.0
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_swap = i
            if best_swap is not None:
                h[best_swap], h[j] = 0.0, 1.0
                corr, cur_c = full_recompute(h, n)
                ones.discard(best_swap)
                ones.add(j)
                zeros.discard(j)
                zeros.add(best_swap)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:

    elif pattern == 'block':
