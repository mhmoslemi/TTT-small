def fine_line_search(h, n, deadline):
    """Fine line search along gradient direction."""
    best_c = compute_c5(h, n)
    for step_scale in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            if time.time() > deadline:
                break
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            # Find the worst lag
            k_worst = np.argmax(c_vec)
            k = k_worst - (n-1)
            # Compute gradient for this lag
            dx = 2.0/n
            grad = np.zeros(n)
            if k >= 0:
                nv = n - k
                grad[0:nv] = (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] = (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
            # Try moves in negative gradient direction
            # Find i where we can decrease h[i] (grad[i] > 0 means decreasing h[i] helps)
            # and j where we can increase h[j]
            pos = np.where(grad > 0.01)[0]  # decreasing these helps
            neg = np.where(grad < -0.01)[0]  # increasing these helps
            
            if len(pos) == 0 or len(neg) == 0:
                break
            
            # Try the most extreme pairs
            pos_sorted = pos[np.argsort(-grad[pos])]
            neg_sorted = neg[np.argsort(grad[neg])]
            
            for i in pos_sorted[:20]:
                if time.time() > deadline:
                    return h, best_c
                for j in neg_sorted[:20]:
                    step = step_scale
                    if h[i] - step >= 0 and h[j] + step <= 1:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def adam_optimize(h, n, deadline, lr=0.01, beta1=0.9, beta2=0.999, alpha=50000, mu=100):
    """Adam optimizer with smooth max objective."""
    best_c = compute_c5(h, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    while time.time() < deadline:
        t += 1
        grad = smooth_grad(h, n, alpha, mu)
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Adaptive step size
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Clip step to prevent violating constraints
        step = np.clip(step, -h*0.5, (1-h)*0.5)
        
        h_new = h + step
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            # Revert
            pass
    return h, best_c

        best_h, best_c = gradient_line_search(best_h, len(best_h), t5_end)

    # Phase 6: Final active set polish (40s)
    t6_end = deadline - 5
    if time.time() < t6_end:
        best_h, best_c = run_active_round(best_h, len(best_h), t6_end)

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
