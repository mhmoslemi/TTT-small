import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_binary(h, n):
    """Project h to binary with exactly n/2 ones."""
    n_ones = n // 2
    if h.sum() == n_ones:
        return (h > 0.5).astype(float)
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n_ones]] = 1.0
    return hb

def compute_delta1(h, n, i, j):
    """Compute correlation delta for swapping h[i]:0→1 and h[j]:1→0."""
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def compute_delta2(h, n, i1, i2, j1, j2):
    """Compute correlation delta for swapping h[i1],h[i2]:0→1 and h[j1],h[j2]:1→0.
    i1, i2 are currently 0; j1, j2 are currently 1."""
    # Compute delta for first swap
    d1 = compute_delta1(h, n, i1, j1)
    # Temporarily apply first swap
    h[i1] = 1.0
    h[j1] = 0.0
    # Compute delta for second swap (with updated h)
    d2 = compute_delta1(h, n, i2, j2)
    # Restore
    h[i1] = 0.0
    h[j1] = 1.0
    return d1 + d2

def anneal_single(h, n, deadline, rng, T0=0.01, cool=0.99995, max_iters=500000):
    """Annealing with single swaps (1-for-1)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    T = T0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)

            corr += delta

            ones_arr[bi] = i
            if cur_c < best_c:

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n_ones]] = 1.0
    return hb

def delta_single(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def local_search_single(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_list):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = delta_single(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    n_ones = len(ones_arr)
    n_zeros = len(zeros_arr)
    m_range = np.arange(-(n - 1), n)
    T = T0
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h, period=2)
    return project_binary(h_interp, n_new)

def make_periodic(n, period, ones_in_period):
    h = np.zeros(n)
    for i in range(n):
        if i % period < ones_in_period:
            h[i] = 1.0
    return project_binary(h, n)

def make_quadratic_residue(n, p):
    h = np.zeros(n)
    for i in range(n):
        if (i * i) % p == 0 or (i * i) % p == 1 or (i * i) % p == (p - 1):
            h[i] = 1.0
    return project_binary(h, n)
