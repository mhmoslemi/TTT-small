from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            # Need to increase sum
            room = 1.0 - h
            s_room = room.sum()
            if s_room < 1e-15:
                break
            h = h + (diff / s_room) * room
            h = np.clip(h, 0.0, 1.0)
        else:
            # Need to decrease sum
            s_h = h.sum()
            if s_h < 1e-15:
                break
            h = h + (diff / s_h) * h
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    """Compute gradient of log-sum-exp approximation of max(corr*dx)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    vals = corr * dx
    # Clip for numerical stability
    max_val = vals.max()
    exp_vals = np.exp(alpha * (vals - max_val))
    Z = exp_vals.sum()
    weights = exp_vals / Z * alpha * dx  # d/dh of (1/alpha)*log(sum(exp(alpha*corr*k*dx)))
    
    # Gradient: for each j, sum over m of weights[m] * d(corr[m])/d(h[j])
    grad = np.zeros(n)
    nn = len(corr)  # 2n-1
    mid = n - 1
    
    for j in range(n):
        # corr[m] = sum_i h[i+m](1-h[i]) for m>=0 (valid range)

        g = 0.0
        # m >= 0: m ranges 0 to n-1, corr[mid+m]
        for m in range(n):
            cm = weights[mid + m]
            # d(corr[mid+m])/d(h[j])
            # corr[mid+m] = sum_{i=0}^{n-1-m} h[i+m](1-h[i])
            # From h[i+m] term: if j = i+m, i = j-m, valid if 0 <= j-m <= n-1-m, i.e., m <= j <= n-1

            # From (1-h[i]) term: if j = i, valid if 0 <= j <= n-1-m
            if j <= n-1-m:
                g -= cm * h[j + m]
            # m < 0: p = -m, corr[mid-p]
        for p in range(1, n):
            cm = weights[mid - p]
            # corr[mid-p] = sum_{i=p}^{n-1} h[i](1-h[i-p])

        grad[j] = g
    return grad

def project_feasible(h, n):

                val = 0.0
                if 0 <= idx_p < n:
                    val += 1.0 - h[idx_p]
                if 0 <= idx_m < n:
                    val -= h[idx_m]
                delta[k + n - 1] -= val
            # Cross terms
            delta[n-1] -= 2.0
            if 0 <= n-1 + (dj-di) < 2*n-1:
                delta[n-1 + (dj-di)] += 1.0
            if 0 <= n-1 + (di-dj) < 2*n-1:
                delta[n-1 + (di-dj)] += 1.0
            
            new_max = np.max(corr + delta) * dx
            if new_max < cur_c - 1e-15:
                h[di] = 1.0
                h[dj] = 0.0
                corr += delta
                cur_c = new_max
                best_c = min(best_c, cur_c)
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    
    return best_h, best_c
```

Wait, this local search is going to be WAY too slow. Each swap evaluation involves looping over 2n-1 lags in Python. For n=500, that's 1000 iterations in Python per swap, and we need maybe 50000 swaps. That's 50M Python iterations - way too slow.

Let me vectorize the delta computation:

```python
def compute_swap_delta(h, n, i, j):
    """Compute the change in correlation when h[i] goes 0->1 and h[j] goes 1->0"""
    dx_idx = np.arange(-(n-1), n)  # lags
    
    # partial for position i (0->1): adds (1-h[i+k]) - h[i-k] at lag k
    ip = i + dx_idx  # i+k
    im = i - dx_idx  # i-k
    val_i = np.zeros(2*n-1)
    mask_p = (ip >= 0) & (ip < n)
    mask_m = (im >= 0) & (im < n)
    val_i[mask_p] += 1.0 - h[ip[mask_p]]
    val_i[mask_m] -= h[im[mask_m]]
    
    # partial for position j (1->0): subtracts (1-h[j+k]) - h[j-k] at lag k
    jp = j + dx_idx
    jm = j - dx_idx
    val_j = np.zeros(2*n-1)
    mask_p2 = (jp >= 0) & (jp < n)
    mask_m2 = (jm >= 0) & (jm < n)
    val_j[mask_p2] += 1.0 - h[jp[mask_p2]]
    val_j[mask_m2] -= h[jm[mask_m2]]
    
    delta = val_i - val_j
    # Cross terms
    delta[n-1] -= 2.0
    k1 = n-1 + (j-i)
    k2 = n-1 + (i-j)
    if 0 <= k1 < 2*n-1:
        delta[k1] += 1.0
    if 0 <= k2 < 2*n-1:
        delta[k2] += 1.0
    
    return delta
```

This is O(n) per swap in numpy, which is fast. But we still need many swaps.

Actually, for the local search, let me try a different approach: instead of random pair sampling, do a **systematic** search over the most promising pairs.

Key insight: the lags that matter most are those where corr[k] is near the maximum. Let's call them the "active set". A swap that decreases the active lags' correlation values is the most beneficial.

For a given active lag k, the swap (i,j) affects corr[k] by:
Δcorr[k] = (1-h[i+k])*[valid] - h[i-k]*[valid] - (1-h[j+k])*[valid] + h[j-k]*[valid] + cross

To decrease corr[k], we want Δcorr[k] < 0 for the active lags.

This gives us a way to prioritize which swaps to try.

But this is getting very complex. Let me simplify and focus on what's likely to work:

1. **Projected gradient descent** with the vectorized gradient (this is the main improvement)
2. **Shorter but more focused local search** (only try ~10000 swaps per sweep)
3. **Multiple resolutions**

Let me also consider: maybe the best approach is to use CVXPY to solve a related convex problem, then use the solution as initialization.

The problem min max_k corr[k]*dx s.t. 0≤h≤1, sum(h)=n/2 is non-convex (because of the max and the bilinear h*(1-h) terms).

But we can relax: if we fix the "1" set S (where h=1), then corr[k] = |S| - |S ∩ (S+k)| = n/2 - |S ∩ (S+k)| for a binary function. Wait, that's not quite right.

For binary h with sum(h)=n/2:
corr[k] = Σ_i h[i]*(1-h[i+k]) = Σ_{i∈S} (1-h[i+k]) = |S| - |{i∈S : i+k∈S}| = n/2 - |S ∩ (S-k)|

So corr[k] = n/2 - autocorrelation_of_S[k]

And C₅ = max_k (n/2 - autoS[k]) * dx = (n/2 - min_k autoS[k]) * dx

So minimizing C₅ is equivalent to maximizing min_k autoS[k], where autoS[k] = |{i : i∈S and i+k∈S}|.

This is a combinatorial problem: find a subset S of {0,...,n-1} of size n/2 that maximizes the minimum autocorrelation.

This is related to the "balanced autocorrelation" problem in signal processing!

For the binary case, let me think about what structure minimizes C₅:
- We want autoS[k] to be as uniform as possible (all close to n/4 for large k)
- The minimum over k of autoS[k] should be maximized

For n=250, |S|=125. The average autocorrelation is (n/2)²/n = n/4 = 62.5. If we could make all autoS[k] = 62.5, then C₅ = (125-62.5)*0.008 = 0.5. But we can't achieve perfect uniformity.

The theoretical lower bound for C₅ (in the continuous case) is related to the Erdős problem. The best known lower bound is around 0.25 (for the continuous version), but discrete approximations give higher values.

Anyway, for the binary case, the optimization is:
maximize min_k autoS[k] s.t. |S| = n/2, S ⊂ {0,...,n-1}

This is a max-min problem. We can approach it with:
- Binary search on the target t: can we find S with min_k autoS[k] ≥ t?
- Or: iteratively improve the minimum autocorrelation

The local search I described (flipping elements) is exactly the right approach for this combinatorial problem. The question is just making it efficient.

Let me now think about efficiency. For n=500:
- |S| = 250, |zeros| = 250
- Number of possible swaps: 250*250 = 62500
- Each swap evaluation: O(n) numpy ops ≈ 500 ops
- Total per sweep: 62500 * 500 = 31M ops in numpy ≈ 0.3 seconds

So a full sweep over all pairs takes ~0.3 seconds. In 1000 seconds, I can do ~3000 sweeps. That should be enough to converge.

But wait, I don't need to check ALL pairs. I can be smarter:
- Only try swaps where i is a zero and j is a one
- Precompute partial_i for all zeros (250 * O(n) = 125K ops)
- Then for each (i,j) pair, delta = precomputed_i - precomputed_j + cross (O(1) per pair)
- Check if max(corr + delta) < cur_c: this is O(n) per pair unless we can shortcut

The shortcut: if the current max of corr is at positions K_max (the active set), then:
max(corr + delta) = max(max_{k∈K_max} (corr[k]+delta[k]), max_{k∉K_max} (corr[k]+delta[k]))

For k∉K_max, corr[k] < cur_c/dx (the max), so corr[k]+delta[k] < cur_c/dx + max(delta[k]).
If max(delta) < 0, then all non-active positions are safe.
If max(delta) > 0, we need to check.

Actually, the most efficient approach: precompute the top-3 correlation values and their positions. For a swap, only check those positions (plus the positions where delta is largest).

Let me implement this:

```python
def efficient_local_search(h, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    target = n // 2
    # Ensure binary
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    mid = n - 1
    lags = np.arange(-(n-1), n)
    
    # Precompute for all zeros and ones
    def precompute_partials(h, positions):
        # Returns array of shape (len(positions), 2n-1)
        partials = np.zeros((len(positions), 2*n-1))
        for idx, pos in enumerate(positions):
            ip = pos + lags
            im = pos - lags
            mask_p = (ip >= 0) & (ip < n)
            mask_m = (im >= 0) & (im < n)
            partials[idx, mask_p] += 1.0 - h[ip[mask_p]]
            partials[idx, mask_m] -= h[im[mask_m]]
        return partials
    
    # Get initial sets
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
            
        # Precompute partials for all zeros and ones
        zero_partials = precompute_partials(h, zeros)  # (n_zeros, 2n-1)
        one_partials = precompute_partials(h, ones)  # (n_ones, 2n-1)
        
        # For each zero i and one j, check swap
        # delta = zero_partials[i_idx] - one_partials[j_idx] + cross
        # We want max(corr + delta) < cur_c/dx
        # Equivalently: max over k of (corr[k] + zero_partials[i_idx,k] - one_partials[j_idx,k] + cross_k) < cur_c/dx
        
        # This is O(n_zeros * n_ones * n) = O(n^3) which is too slow for n=500
        # (250 * 250 * 1000 = 62.5M * some factor)
        
        # Better: for each pair, the check is O(n) using numpy max
        # Total: 250*250 = 62500 pairs, each O(n) = O(62500*1000) = 62.5M ops
        # In numpy, this should take a few seconds
        
        # Actually, let's vectorize: compute delta matrix (n_zeros, n_ones, 2n-1)
        # That's 250 * 250 * 1000 = 62.5M elements = 500MB. Too much!
        
        # Alternative: loop over zeros, for each zero compute vs all ones
        for zi, i in enumerate(zeros):
            if time.time() > deadline:
                break
            zp = zero_partials[zi]  # (2n-1,)
            
            for oj, j in enumerate(ones):
                op = one_partials[oj]
                delta = zp - op
                delta[mid] -= 2.0
                k1 = mid + (j - i)
                k2 = mid + (i - j)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                
                new_max = np.max(corr + delta) * dx
                if new_max < cur_c - 1e-14:
                    # Accept swap
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += delta
                    cur_c = new_max
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    improved = True
                    # Update sets
                    zeros = np.delete(zeros, zi)
                    ones = np.delete(ones, oj)
                    zeros = np.append(zeros, j).astype(int)
                    ones = np.append(ones, i).astype(int)
                    break  # restart sweep after a swap
            
            if improved:
                break
    
    return best_h, best_c
```

This is still O(n³) per sweep (n zeros × n ones × n lags for max check). For n=500: 500*500*1000 = 250M per sweep. At ~1ns per numpy op, that's 0.25s per sweep. In 500s, we can do ~2000 sweeps. But we only need maybe 50-100 sweeps to converge.

Actually wait, I'm looping over zeros and ones in Python, which adds overhead. Let me vectorize the inner loop:

For a fixed zero i, I want to find the one j that minimizes max(corr + zp - op_j + cross_j).

This is hard to vectorize because of the max. But I can vectorize the computation of corr + zp - op_j for all j simultaneously:

base = corr + zp  # (2n-1,)
For all ones: deltas = base[None,:] - one_partials  # (n_ones, 2n-1)
Add cross terms...

Then new_maxes = deltas.max(axis=1) * dx  # (n_ones,)
Best j = argmin(new_maxes)

This is O(n_ones * 2n) = O(n²) per zero, and O(n³) per sweep total.
For n=500: 500 * 500 * 1000 = 250M per sweep. Memory: 500*1000 = 500K per zero iteration. Fine.

Let me implement this:

```python
def efficient_local_search_v2(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    mid = n - 1
    lags = np.arange(-(n-1), n)
    Ncorr = 2*n - 1
    
    def compute_partials(positions):
        # (len(positions), 2n-1)
        result = np.zeros((len(positions), Ncorr))
        for pi, pos in enumerate(positions):
            ip = pos + lags
            im = pos - lags
            mp = (ip >= 0) & (ip < n)
            mm = (im >= 0) & (im < n)
            result[pi, mp] += 1.0 - h[ip[mp]]
            result[pi, mm] -= h[im[mm]]
        return result
    
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        
        # Precompute partials
        zp_all = compute_partials(zeros)  # (n_z, 2n-1)
        op_all = compute_partials(ones)   # (n_o, 2n-1)
        
        for zi in range(len(zeros)):
            if time.time() > deadline:
                break
            i = zeros[zi]
            zp = zp_all[zi]  # (2n-1,)
            base = corr + zp  # (2n-1,)
            
            # For each one j, compute new correlation max
            # delta_j = base - op_all[j] + cross_j
            # cross terms: -2 at mid, +1 at mid+(j-i), +1 at mid+(i-j)
            
            all_deltas = base[None, :] - op_all  # (n_o, 2n-1)
            all_deltas[:, mid] -= 2.0
            
            # Cross terms
            for oj in range(len(ones)):
                j = ones[oj]
                k1 = mid + (j - i)
                k2 = mid + (i - j)
                if 0 <= k1 < Ncorr:
                    all_deltas[oj, k1] += 1.0
                if 0 <= k2 < Ncorr:
                    all_deltas[oj, k2] += 1.0
            
            new_maxes = (corr + all_deltas).max(axis=1) * dx
            best_oj = new_maxes.argmin()
            
            if new_maxes[best_oj] < cur_c - 1e-14:
                j = ones[best_oj]
                delta = all_deltas[best_oj]
                h[i] = 1.0
                h[j] = 0.0
                corr += delta
                cur_c = new_maxes[best_oj]
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
