def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:

        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])

        if i_min2 <= i_max2:
