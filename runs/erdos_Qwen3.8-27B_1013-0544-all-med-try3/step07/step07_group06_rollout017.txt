import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary: top n/2 values become 1"""
    idx = np.argsort(h)[::-1][:n//2]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def vectorized_local_search(h, n, deadline, max_iters=5000, seed=42):
    """Efficient binary local search using vectorized swap evaluation"""
    dx = 2.0 / n
    target = n // 2
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        n1 = len(ones_idx)
        n0 = len(zeros_idx)
        
        if n1 == 0 or n0 == 0:
            break
        
        # Precompute add contributions for each zero position
        # add_i[k] for k in [0, 2n-2], m = k-(n-1)
        # Contribution when h[i] goes 0→1:
        #   For valid i+m: add (1-h[i+m])
        #   For valid i-m: add -h[i-m] (because 1-h[i] goes from 1 to 0)
        
        # For sub contributions for each one position j:
        # Contribution when h[j] goes 1→0:

        #   For valid j-m: add h[j-m] (because 1-h[j] goes from 0 to 1)

        # add[i, k] where k corresponds to lag m = k-(n-1)
        m_range = np.arange(-(n-1), n)  # shape (2n-1,)
        
        add_mat = np.zeros((n0, 2*n-1))
        sub_mat = np.zeros((n1, 2*n-1))

            msk = (im >= 0) & (im < n)
            add_mat[ii, msk] += (1.0 - h[im[msk]])
            # i-m valid
            imn = i - m_range
            msk2 = (imn >= 0) & (imn < n)
            add_mat[ii, msk2] -= h[imn[msk2]]
        
        for jj, j in enumerate(ones_idx):
            # j+m valid
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            sub_mat[jj, msk] -= (1.0 - h[jm[msk]])
            # j-m valid
            jmn = j - m_range
            msk2 = (jmn >= 0) & (jmn < n)
            sub_mat[jj, msk2] += h[jmn[msk2]]
        
        # Now for each pair (i, j): delta = add_mat[i] + sub_mat[j] + cross
        # C_ij = corr + add_mat[i] + sub_mat[j]
        # Apply cross terms and find max
        
        # Batch compute: for each i, compute C_i = (n1, 2n-1)
        # C_i = corr[None,:] + add_mat[i][None,:] + sub_mat
        
        # Memory: n0 * n1 * (2n-1) = 125 * 125 * 499 ≈ 7.8M floats ≈ 62MB
        # This is OK for n=250
        
        k0 = n - 1  # position for m=0
        
        best_delta_c = -np.inf
        best_i = -1
        best_j = -1
        
        for ii in range(n0):
            C_i = corr[None, :] + add_mat[ii][None, :] + sub_mat  # (n1, 2n-1)
            
            # Apply cross terms
            C_i[:, k0] -= 2.0
            
            # Cross at lag j-i: position k0 + (ones_idx[jj] - zeros_idx[ii])
            for jj in range(n1):
                k1 = k0 + (ones_idx[jj] - zeros_idx[ii])
                k2 = k0 + (zeros_idx[ii] - ones_idx[jj])
                if 0 <= 2*n-2 and 0 <= k1 < 2*n-1:
                    C_i[jj, k1] += 1.0
                if 0 <= k2 < 2*n-1:
                    C_i[jj, k2] += 1.0
            
            maxes = C_i.max(axis=1)
            # Note: k1 and k2 might be same as k0 when i==j, but i≠j here
            # Also k1 == k0 when ones_idx[jj] == zeros_idx[ii], impossible since h[i]=0, h[j]=1
            
            # Find best
            min_idx = maxes.argmin()
            if maxes[min_idx] * dx < best_delta_c:
                # Wait, I want to minimize the max, so I want the pair with smallest max
                best_delta_c = maxes[min_idx] * dx
                best_i = ii
                best_j = min_idx
        
        if best_i < 0:
            break
        
        # Check if this is an improvement
        if best_delta_c >= cur_c - 1e-12:
            break
        
        # Apply the swap
        i_pos = zeros_idx[best_i]
        j_pos = ones_idx[best_j]
        h[i_pos] = 1.0
        h[j_pos] = 0.0
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        cur_c = corr.max() * dx
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

_constraint_n = 250
_alpha_global = 1000.0
_dx_global = 0.008

def _sum_constraint(x):
    return x.sum() - _constraint_n / 2.0

def _slsqp_obj(x):
    g = np.correlate(x, 1 - x, mode='full') * _dx_global
    m = g.max()
    sh = _alpha_global * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / _alpha_global

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1][:n // 2]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

    h = to_binary(h, n).copy()

    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    k0 = n - 1
    two_n_1 = 2 * n - 1
    n_stagnant = 0
    t_sa = 0.008
    t_min = 0.0003
    C_buf = np.empty((n + 1, two_n_1))

    for it in range(max_iters):
        if time.time() > deadline:
            break

        zeros_idx = np.where(h == 0)[0]

        n0 = len(zeros_idx)
        if n1 == 0 or n0 == 0:
            break

        msk_a = (im >= 0) & (im < n)
        im_safe = np.clip(im, 0, n - 1)
        add_mat = (1.0 - h[im_safe].astype(float)) * msk_a

        imn = zeros_idx[:, None] - m_range[None, :]
        msk_b = (imn >= 0) & (imn < n)

        msk_c = (jm >= 0) & (jm < n)
        jm_safe = np.clip(jm, 0, n - 1)
        sub_mat = -(1.0 - h[jm_safe].astype(float)) * msk_c

        msk_d = (jmn >= 0) & (jmn < n)

        best_ii = -1
        best_jj = -1

        cur_c = new_c
        n_stagnant = 0

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    global _constraint_n, _alpha_global, _dx_global
    _constraint_n = n
    _alpha_global = alpha
    _dx_global = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h
