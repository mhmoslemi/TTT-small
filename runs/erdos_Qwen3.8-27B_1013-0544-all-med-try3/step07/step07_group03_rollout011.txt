```python
import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n):
    grad = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        p = m - (n-1)
        if p >= 0:
            if n > p:
                grad[m, p:] += (1.0 - h[:n-p])
            if p > 0:
                grad[m, :n-p] -= h[p:]
        else:
            q = -p
            if n > q:
                grad[m, :n-q] += (1.0 - h[q:])
            if n > q:
                grad[m, q:] -= h[:n-q]
    return grad

def solve_lp(h0, n, solver):
    c_all = np.correlate(h0, 1 - h0, mode='full')
    grad = compute_gradient(h0, n)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    rhs = -c_all + grad @ h0
    A = np.hstack([grad, -np.ones((2*n-1, 1))])
    x = cp.hstack([h_var, t_var])
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        A @ x <= rhs,
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    
    if h_var.value is None:
        return None, None
    
    h_new = np.clip(np.array(h_var.value, dtype=float), 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new, float(t_var.value)

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)

                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:

    return h, best_c
