```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, idx):
    dx = 2.0 / n
    k = idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            g[0:nv] += (1.0 - h[k:n]) * dx
            if k > 0:
                g[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            g[kk:n] += (1.0 - h[0:nv]) * dx
            g[0:nv] += (-h[kk:n]) * dx
    return g

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n_c = len(c)
    grad = np.zeros(n)
    for mm in range(n_c):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                if k > 0:
                    grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def topk_grad(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                if k > 0:
                    grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 20000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def topk_optimize(h, n, deadline, top_k=5):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for _ in range(15):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            top_in_active = active[np.argsort(-c_vec[active])[:top_k]]
            active = top_in_active
        if len(active) == 0:
            break
        for alpha in [5000, 50000, 500000, 5000000]:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(topk_obj, h, jac=topk_grad,
                              args=(n, active, alpha), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 15000, 'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def subgradient_step(h, n, step_size, n_active=8):
    """One subgradient step on the exact max."""
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    sorted_idx = np.argsort(-c_vec)
    active = sorted_idx[:n_active]
    # Only include shifts that are close to max
    close_thresh = max_val - 1e-4
    active = active[c_vec[active] >= close_thresh]
    if len(active) == 0:
        active = sorted_idx[:1]
    grad = np.zeros(n)
    for idx in active:
        grad += shift_grad(h, n, idx)
    grad /= len(active)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def subgradient_method(h, n, deadline, init_step=0.01, max_iter=5000):
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    step = init_step
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        h_new = subgradient_step(h_cur, n, step)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            h = h_new
            h_cur = h_new
            step *= 1.05
            patience = 0
        elif c_new < best_c:
            h_cur = h_new
            step *= 0.95
            patience += 1
        else:
            step *= 0.7
            patience += 1
        if step < 1e-8:
            break
        if patience > 200:
            break
    return h, best_c

def fine_pair_search(h, n, deadline, max_radius=50):
    best_c = compute_c5(h, n)
    h = h.copy()
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline - 2:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i]+=d, h[j]-=d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i]-=d, h[j]+=d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def gradient_fine_search(h, n, deadline):
    """Use gradient to guide fine pair moves."""
    best_c = compute_c5(h, n)
    h = h.copy()
    base_step = 0.001
    for round_idx in range(30):
        if time.time() > deadline - 2:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 10:
            active = active[np.argsort(-c_vec[active])[:10]]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad(h, n, idx)
        grad /= len(active)
        # Find best pair move
        top_idx = np.argsort(-np.abs(grad))[:60]
        best_improve = 0
        best_move = None
        for ai in range(len(top_idx)):
            if time.time() > deadline - 2:
                break
            i = top_idx[ai]
            for aj in range(len(top_idx)):
                if ai == aj:
                    continue
                j = top_idx[aj]
                gi, gj = grad[i], grad[j]
                if gi < gj:
                    # decrease h[i], increase h[j]
                    for mult in [1.0, 0.5, 0.25]:
                        step = base_step * mult
                        if h[i] - step >= 0 and h[j] + step <= 1:
                            h[i] -= step
                            h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_improve = best_c - c
                                best_c = c
                                best_move = True
                                break
                            else:
                                h[i] += step
                                h[j] -= step
                elif gi > gj:
                    # increase h[i], decrease h[j]
                    for mult in [1.0, 0.5, 0.25]:
                        step = base_step * mult
                        if h[i] + step <= 1 and h[j] - step >= 0:
                            h[i] += step
                            h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_improve = best_c - c
                                best_c = c
                                best_move = True
                                break
                            else:
                                h[i] -= step
                                h[j] += step
        if best_move is None:
            base_step *= 0.7
            if base_step < 1e-7:
                break
    return h, best_c

def perturb_reoptimize(h, n, deadline, n_perturbs=5, sigma=0.01):
    """Try random perturbations and re-optimize from each."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for p in range(n_perturbs):
        if time.time() > deadline - 60:
            break
        h_p = h + np.random.randn(n) * sigma
        h_p = project_feasible(h_p, n)
        # Quick optimization from perturbed start
        t_sub = min(time.time() + 40, deadline - 60)
        h_opt, c_opt = lbfgs_optimize(h_p, n, t_sub,
            [(1000, 10.0), (5000, 50.0), (20000, 500.0)])
        if c_opt < best_c - 1e-15:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def multi_resolution(h, n, deadline, target_ns=(600, 750)):
    """Try optimization at different resolutions."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x_fine = np.linspace(0, 2, n)
    for n_target in target_ns:
        if time.time() > deadline - 120:
            break
        x_target = np.linspace(0, 2, n_target)
        h_t = np.interp(x_target, x_fine, h)
        h_t = project_feasible(h_t, n_target)
        t_sub = min(time.time() + 80, deadline - 60)
        h_t, c_t = lbfgs_optimize(h_t, n_target, t_sub,
            [(500, 20.0), (2000, 100.0), (10000, 1000.0), (50000, 5000.0)])
        # Also try top-k
        if time.time() < deadline - 30:
            h_t, c_t = topk_optimize(h_t, n_target, time.time() + min(20, deadline - time.time() - 5), 4)
        if c_t < best_c - 1e-15:
            best_c = c_t
            best_h = h_t
    return best_h, best_c, len(best_h)

def single_point_search(h, n, deadline):
    """Try adjusting single points to fine-tune."""
    best_c = compute_c5(h, n)
    h = h.copy()
    delta = 0.0002
    for _ in range(20):
        if time.time() > deadline - 2:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 2:
                break
            for d in [delta, -delta]:
                new_val = h[i] + d
                if 0 <= new_val <= 1:
                    # Adjust a compensating point to maintain sum
                    if d > 0:
                        # Need to decrease some other point
                        room = h - 0.0
                        candidates = np.where(room > delta / 2)[0]
                    else:
                        # Need to increase some other point
                        room = 1.0 - h
                        candidates = np.where(room > delta / 2)[0]
                    if len(candidates) == 0:
                        continue
                    # Pick the one with largest gradient contribution
                    # Just try a few
                    for j in np.random.choice(candidates, size=min(10, len(candidates)), replace=False):
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i] -= d
                            h[j] += d
            if improved:
                break
        if not improved:
            delta *= 0.5
            if delta < 1e-6:
                break
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds,
                      options={'maxiter': 2000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
```
