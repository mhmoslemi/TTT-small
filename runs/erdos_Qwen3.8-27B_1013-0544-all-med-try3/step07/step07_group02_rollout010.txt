                h[j] += delta
            else:
                delta = min(0.01 * rng.random(), h[j] * 0.2, (1 - h[i]) * 0.2)
                h[j] -= delta
                h[i] += delta
        
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            c_all = compute_all_lags(h, n)
    
    return best_h, best_c

def slsqp_polish(h, n, alpha, top_m, maxiter=2000):
    dx = 2.0 / n
    c_all = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c_all)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1.0 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    
    def obj_jac(x):
        cc = np.correlate(x, 1.0 - x, mode='full') * dx
        active = cc[top_idx]
        w = smooth_max_weights(active, alpha)
        grad = np.zeros(n)
        for ii, wi in zip(top_idx, w):
            if wi < 1e-8:
                continue
            grad += wi * grad_lag_idx(x, n, int(ii))
        return grad * dx
    
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or (hasattr(res, 'x') and res.x is not None):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def binary_push(h, n, threshold_lo=0.1, threshold_hi=0.9):
    """Push values towards 0 or 1"""
    h_new = h.copy()
    h_new[h_new < threshold_lo] *= 0.5
    h_new[h_new > threshold_hi] = 1.0 - (1.0 - h_new[h_new > threshold_hi]) * 0.5
    h_new[h_new < 0.3] = h_new[h_new < 0.3] * 0.6
    h_new[h_new > 0.7] = 0.7 + (h_new[h_new > 0.7] - 0.7) * 0.6
    return project_feasible(h_new, n)

def interpolate_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def _slsqp_con_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _slsqp_con_jac(x, n):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Gradient descent at initial resolution
    phase1_dl = min(t0 + 120, deadline - 200)
    if time.time() < phase1_dl:
        h1, c1 = gradient_descent_v2(h_init, n_init, phase1_dl, top_m=3)
        if c1 < best_c:
            best_c, best_h = c1, h1
    
    # Phase 2: Try interpolation to higher resolution and optimize
    resolutions = [600, 750, 900]
    for n_try in resolutions:
        if time.time() > deadline - 150:
            break
        remaining = deadline - time.time() - 50
        if remaining < 80:
            break
        
        h_start = interpolate_h(best_h, len(best_h), n_try)
        alloc = min(130, remaining * 0.25)
        if alloc < 50:
            break
        
        # Gradient optimize
        h_opt, c_opt = gradient_descent_v2(h_start, n_try, time.time() + alloc * 0.6, top_m=3)
        
        # SLSQP polish
        if time.time() < deadline - 40:
            h_opt = slsqp_polish(h_opt, n_try, alpha=500, top_m=3, maxiter=1000)
            c_opt = compute_c5(h_opt, n_try)
        
        # Quick local search
        if time.time() < deadline - 20:
            h_opt, c_opt = targeted_local_search(h_opt, n_try, deadline - 10)
        
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    
    # Phase 3: Aggressive targeted local search at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 20:
        ls_dl = min(deadline - 5, time.time() + (deadline - time.time()) * 0.45)
        best_h, best_c = targeted_local_search(best_h, n_best, ls_dl)
    
    # Phase 4: Pair swap random search
    if time.time() < deadline - 15:
        ms_dl = min(deadline - 5, time.time() + 80)
        h4, c4 = pair_swap_search(best_h, n_best, ms_dl, rng)
        if c4 < best_c:
            best_c, best_h = c4, h4
    
    # Phase 5: SLSQP final polish
    if time.time() < deadline - 8:
        n = len(best_h)
        for top_m in [2, 3, 5]:
            if time.time() > deadline - 5:
                break
            h5 = slsqp_polish(best_h, n, alpha=1000, top_m=top_m, maxiter=1500)
            c5 = compute_c5(h5, n)
            if c5 < best_c:
                best_c, best_h = c5, h5
    
    # Phase 6: Final micro local search
    if time.time() < deadline - 2:
        best_h, best_c = targeted_local_search(best_h, len(best_h), deadline - 0.5)
    
    # Final projection and verification
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
