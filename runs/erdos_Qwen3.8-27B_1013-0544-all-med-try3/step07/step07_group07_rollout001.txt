import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def corr_of(h):

def pgd_soft(h, n, alpha, deadline, max_iter=5000):
    """PGD with log-sum-exp smoothing of the max."""
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    def project(v):
        lo, hi = -2.0, 2.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            v2 = np.clip(v - mid, 0.0, 1.0)
            if v2.sum() > target:
                lo = mid
            else:
                hi = mid
        v2 = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
        # Fix sum
        diff = target - v2.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                room = 1.0 - v2
                if room.sum() > 1e-15:
                    v2 += (diff / room.sum()) * room
            else:
                if v2.sum() > 1e-15:
                    v2 += (diff / v2.sum()) * v2
        return np.clip(v2, 0.0, 1.0)
    
    step = min(float(n) / (4.0 * alpha), 0.1)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c = corr_of(h)
        a = alpha * (c - c.max())
        exp_a = np.exp(a)
        p = exp_a / exp_a.sum()
        
        # Gradient computation
        grad = np.zeros(n)
        for k in range(-(n-1), n):
            pk = p[k + n - 1]
            if pk < 1e-15:
                continue
            # c[k+n-1] = sum_i h[i]*(1-h[i-k])
            # d/dh[i]: (1-h[i-k]) for the first term, -h[i] for the second... 
            # Actually: c[m] = sum_i h[i] * (1-h[i-s]) where s = m-(n-1)
            # d c[m] / d h[j] = (1-h[j-s]) [0<=j-s<n] - h[j+s] [0<=j+s<n] * ... 
            # Wait no. c[m] = sum_i h[i]*(1-h[i-s])
            # ∂c[m]/∂h[j] = (1-h[j-s]) if 0≤j-s<n (from the h[j] factor)
            #                + (-1)*h[j+s] if 0≤j+s<n (from the -(h[j+s]) factor in (1-h[(j+s)-s]))
            # Hmm wait: for the term i=j: h[j]*(1-h[j-s]), derivative w.r.t. h[j] is (1-h[j-s])
            # For the term i=j+s: h[j+s]*(1-h[j+s-s]) = h[j+s]*(1-h[j])
            #   derivative w.r.t. h[j] is h[j+s] * (-1) = -h[j+s]
            
            # So ∂c[m]/∂h[j] = pk * [(1-h[j-s]) * I(0≤j-s<n) - h[j+s] * I(0≤j+s<n)]
            
            # Vectorized:
            # For all j: contribution is pk * [(1-h[j-s]) - h[j+s]] (with bounds)
            # j-s in [0,n-1] means j in [s, s+n-1], intersected with [0,n-1]
            # j+s in [0,n-1] means j in [-s, n-1-s], intersected with [0,n-1]
            
            j_min1 = max(0, s) if (s := k) < n-1 else 0  # j >= s for j-s >= 0
            j_max1 = min(n-1, n-1+s) if s >= 0 else min(n-1, -1)  # j-s <= n-1 means j <= n-1+s
            # Actually: j-s >= 0 means j >= s (only if s > 0)
            # j-s < n means j < n+s (always true if j < n and s >= 0)
            # j-s < n and j-s >= 0: j in [max(0,s), min(n-1, n-1+s)]
            # If s >= 0: j in [s, n-1] (since n-1+s > n-1, so min is n-1)
            #   Wait: j-s < n means j < n+s, always true for j < n
            #   j-s >= 0 means j >= s
            #   So j in [s, n-1] if s >= 0, or j in [0, n-1] if s <= 0
            # j+s >= 0 means j >= -s (only if s < 0)
            # j+s < n means j < n-s (only if s > 0)
            # If s >= 0: j in [0, n-1-s]
            # If s < 0: j in [-s, n-1]
            
            if s >= 0:
                j1_lo, j1_hi = s, n - 1
                j2_lo, j2_hi = 0, n - 1 - s
            else:
                j1_lo, j1_hi = 0, n - 1
                j2_lo, j2_hi = -s, n - 1
            
            if j1_lo <= j1_hi:
                idx1 = np.arange(j1_lo, j1_hi + 1)
                grad[idx1] += pk * (1.0 - h[idx1 - s])
            if j2_lo <= j2_hi:
                idx2 = np.arange(j2_lo, j2_hi + 1)
                grad[idx2] -= pk * h[idx2 + s]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Backtracking line search
        accepted = False
        for ls in range(30):
            h_new = project(h - step * (0.5 ** ls) * grad / gnorm)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                step = step * (0.5 ** ls)
                accepted = True
                break
        if not accepted:
            break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def local_search_1swap(h, n, deadline, max_sweeps=200):
    """Local search with single 0-1 swap moves."""
    dx = 2.0 / n
    corr = corr_of(h)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    m_range = np.arange(-(n-1), n)
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for zi in range(len(zeros)):
            i = zeros[zi]
            best_j = -1
            best_nc = cur_c
            for j in ones:
                delta = _swap_delta_fast(i, j, h, n, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += _swap_delta_fast(i, best_j, h, n, m_range)  # recompute after swap
                # Actually need to recompute properly
                corr = corr_of(h)
                cur_c = corr.max() * dx
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def _swap_delta_fast(i, j, h, n, m_range):
    """Delta in correlation array when i goes 0→1 and j goes 1→0."""
    # This was before the swap
    delta = np.zeros(2 * n - 1)
    # i: 0→1
    a = i - m_range
    msk_a = (a >= 0) & (a < n)
    delta[msk_a] += (1.0 - h[a[msk_a]])
    b = i + m_range
    msk_b = (b >= 0) & (b < n)
    delta[msk_b] -= h[b[msk_b]]
    # j: 1→0 (negate)
    a2 = j - m_range
    msk_a2 = (a2 >= 0) & (a2 < n)
    delta[msk_a2] -= (1.0 - h[a2[msk_a2]])
    b2 = j + m_range
    msk_b2 = (b2 >= 0) & (b2 < n)
    delta[msk_b2] += h[b2[msk_b2]]
    return delta

def anneal_1swap(h, n, deadline, seed=42, T_init=0.01, cool=0.99995):
    """Simulated annealing with 1-swap moves."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = corr_of(h)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    zeros_arr = np.where(h == 0)[0].copy()
    ones_arr = np.where(h == 1)[0].copy()
    m_range = np.arange(-(n-1), n)
    T = T_init
    n_ones = len(ones_arr)
    n_zeros = len(zeros_arr)
    
    it = 0
    while time.time() < deadline:
        it += 1
        ai = rng.randint(n_zeros)
        bi = rng.randint(n_ones)
        i = zeros_arr[ai]
        j = ones_arr[bi]
        
        delta = _swap_delta_fast(i, j, h, n, m_range)
        nc = (corr + delta).max() * dx
        
        accept = nc <= cur_c
        if not accept and T > 1e-12:
            diff = nc - cur_c
            if diff < 0.1:  # Don't accept absurdly bad moves
                accept = rng.random() < np.exp(-200.0 * diff / T)
        
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        T *= cool
        # Reheat occasionally
        if it % 50000 == 0 and T < 0.005:
            T = 0.005
    
    return best_h, best_c

def _swap2_delta_fast(i1, i2, j1, j2, h, n, m_range):
    """Delta when i1,i2 go 0→1 and j1,j2 go 1→0."""
    delta = np.zeros(2 * n - 1)
    for (pos, sign) in [(i1, 1), (i2, 1), (j1, -1), (j2, -1)]:
        a = pos - m_range
        msk_a = (a >= 0) & (a < n)
        delta[msk_a] += sign * (1.0 - h[a[msk_a]])
        b = pos + m_range
        msk_b = (b >= 0) & (b < n)
        delta[msk_b] += sign * (-h[b[msk_b]])
    return delta
