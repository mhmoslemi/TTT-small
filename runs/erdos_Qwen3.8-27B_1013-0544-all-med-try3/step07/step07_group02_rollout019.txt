import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def grad_top_lags(h, n, top_k=3):

    c = np.correlate(h, 1 - h, mode='full')
    top_idx = np.argsort(c)[-top_k:]

    for k in top_idx:
        lag = int(k) - (n - 1)
        grad += lag_gradient(h, n, lag)
    return (grad / top_k) * dx

def smooth_max_obj(h, n, alpha):

def slsqp_step(h, n, alpha, maxiter=2000, top_m=5):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    
    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for t, idx in enumerate(top_idx):
            if w[t] < 1e-10:
                continue
            lag = int(idx) - (n - 1)
            grad += w[t] * lag_gradient(x, n, lag)
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def binary_push(h, n, deadline):

    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()

                best_c = c_new
                best_h = h.copy()
            else:
                h[i] = hi

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def grad_weighted_lags(h, n, top_k=3):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        lag = int(k) - (n - 1)
        grad += lag_gradient(h, n, lag)
    return (grad / top_k) * dx

def gradient_descent(h, n, deadline, alpha_top=3, lr_init=1.0, momentum=0.9, decay=0.95):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    patience = 0
    lr = lr_init
    stuck_large = 0
    
    for it in range(200000):
        if time.time() > deadline:
            break
        g = grad_weighted_lags(h, n, top_k=alpha_top)
        v = momentum * v + lr * g
        h_new = project_feasible(h - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
            stuck_large = 0
        else:
            patience += 1
            stuck_large += 1
        h = h_new
        if patience > 200:
            lr *= 0.8
            v *= 0.5
            patience = 0
        if stuck_large > 1000:
            lr = max(0.02, lr * 0.7)
            v *= 0.3
            stuck_large = 0
        if it % 500 == 499:
            lr = max(0.01, lr * decay)
    
    return best_h, best_c

def local_search(h, n, deadline, max_pairs=60, small_delta=0.001):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    stuck = 0
    
    for rnd in range(1000):
        if time.time() > deadline - 1:
            break
        c_all = compute_all_lags(h, n)
        top_lags = np.argsort(c_all)[-2:]
        any_improved = False
        
        for k_idx in top_lags:
            if time.time() > deadline - 1:
                break
            lag = int(k_idx) - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            
            diffs = np.abs(h[0:n-m] - h[m:n])
            sort_idx = np.argsort(diffs)[::-1]
            top_pairs = sort_idx[:max_pairs]
            
            for pi in top_pairs:
                if time.time() > deadline - 1:
                    break
                i = int(pi)
                j = i + m
                hi, hj = h[i], h[j]
                if abs(hi - hj) < 1e-9:
                    continue

                    max_d = (hi - hj) * 0.45
                else:
                    max_d = (hj - hi) * 0.45

                    else:

                    else:

                            continue

def _sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def _active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _active_smooth_grad(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for t in range(len(top_idx)):
        if w[t] < 1e-12:
            continue
        idx = int(top_idx[t])
        lag = idx - (n - 1)
        grad += w[t] * lag_gradient(x, n, lag)
    return grad * dx

def slsqp_step(h, n, alpha, maxiter=3000, top_m=3):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_active_smooth_obj, h, args=(n, top_idx, alpha),
                      jac=_active_smooth_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_full(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    
    def _full_obj(x, n, alpha):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def _full_grad(x, n, alpha):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-12:
                continue
            lag = k - (n - 1)
            grad += w[k] * lag_gradient(x, n, lag)
        return grad * dx
    
    try:
        res = minimize(_full_obj, h, args=(n, alpha),
                      jac=_full_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_push(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    mid_mask = (h > 0.1) & (h < 0.9)
    mid_idx = np.where(mid_mask)[0]
    if len(mid_idx) == 0:
        return best_h, best_c
    
    mid_order = np.argsort(np.abs(h[mid_idx] - 0.5))
    
    for idx in mid_order:
        if time.time() > deadline:
            break
        i = int(mid_idx[idx])
        hi = h[i]
        
        for target in [0.0, 1.0]:
            h[i] = target
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
            else:
                h[i] = hi
                h = project_feasible(h, n)
    
    return best_h, best_c

def random_restart_search(h_best, n, deadline, rng, n_restarts=2):
    best_h = h_best.copy()
    best_c = compute_c5(best_h, n)
    
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        alloc = min(40, (deadline - time.time() - 10) * 0.5)
        if alloc < 20:
            break
        
        scale = 0.015 * (1.0 + 0.4 * r)
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        
        # Quick gradient descent
        gd_end = time.time() + alloc * 0.5
        hp, c_h = gradient_descent(hp, n, min(gd_end, deadline - 40), alpha_top=3, lr_init=0.8, momentum=0.9)
        
        # Quick local search
        if time.time() < deadline - 35:
            hp, c_h = local_search(hp, n, min(deadline - 30, time.time() + 8), max_pairs=30, small_delta=0.002)
        
        if c_h < best_c:
            best_c = c_h
            best_h = hp
    
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    remaining = deadline - time.time()
    
    # Phase 1: Gradient descent (40% of time)
    if remaining > 40:
        gd_end = time.time() + remaining * 0.35
        h_new, c_new = gradient_descent(h, n, gd_end, alpha_top=3, lr_init=1.0, momentum=0.9)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    
    # Phase 2: SLSQP with top 3 (15% of time)
    if time.time() < deadline - 20:
        h_new = slsqp_step(h, n, alpha=50000, maxiter=3000, top_m=3)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    
    # Phase 3: Local search (35% of time)
    if time.time() < deadline - 10:
        ls_end = deadline - 5
        h_new, c_new = local_search(best_h, n, ls_end, max_pairs=80, small_delta=0.001)
