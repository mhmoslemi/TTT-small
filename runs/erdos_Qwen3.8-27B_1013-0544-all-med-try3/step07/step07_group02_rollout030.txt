            break
        remaining = deadline - time.time() - 40
        if remaining < 80:
            break
        alloc = min(150, remaining * 0.3)
        if alloc < 50:
            break
        h_interp = interpolate_solution(best_h, len(best_h), n_try)
        h_opt, c_opt = optimize_single(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Also try lower resolution (sometimes easier to optimize)
    if time.time() < deadline - 80:
        for n_low in [300, 400]:
            if time.time() > deadline - 50:
                break
            remaining = deadline - time.time() - 30
            if remaining < 40:
                break
            alloc = min(60, remaining * 0.4)
            h_interp = interpolate_solution(best_h, len(best_h), n_low)
            h_opt, c_opt = optimize_single(h_interp, n_low, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 60:
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scales = [0.01, 0.02, 0.03, 0.05, 0.08]
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(45, (deadline - time.time() - 20) * 0.3)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_single(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Binary SA at best resolution
    if time.time() < deadline - 30:
        sa_deadline = min(deadline - 5, time.time() + 25)
        h_sa, c_sa = simulated_annealing_binary(best_h, n_best, sa_deadline, rng, max_iter=30000)
        if c_sa < best_c:
            best_c = c_sa
            # Convert binary back to fractional with small smoothing
            best_h = h_sa.copy()
    
    # Phase 6: Final intensive polishing
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Final gradient descent with high alpha
        if time.time() < deadline - 15:
            alpha_sched = [5000, 10000, 20000, 50000]
            h_new, c_new = grad_momentum(best_h, n, alpha_sched, min(deadline - 10, time.time() + 20), lr=0.5, momentum=0.85)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Final SLSQP on top-3
        if time.time() < deadline - 10:
            c_all = compute_all_lags(best_h, n)
            top_idx = np.argsort(c_all)[-3:]
            h_new = slsqp_topk(best_h, n, top_idx, 100000, maxiter=5000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Final local search
        if time.time() < deadline - 3:
            best_h, best_c = focused_local_search(best_h, n, deadline - 1, max_rounds=100)
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
