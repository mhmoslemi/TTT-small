def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val_grad(h, n, alpha):
    dx = 2.0 / n
    raw_c = np.correlate(h, 1 - h, mode='full')
    m = raw_c.max()
    shifted = raw_c - m
    shifted = np.clip(shifted, -50.0, 0.0)
    e = np.exp(alpha * shifted)
    e_sum = e.sum()
    p = e / e_sum
    val = dx * (m + np.log(e_sum) / alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += pk * (1.0 - h[s:n])
            grad[s:n] -= pk * h[0:n-s]
        else:
            t = -s
            grad[t:n] += pk * (1.0 - h[0:n-t])
            grad[0:n-t] -= pk * h[t:n]
    grad *= dx
    return val, grad

def sgd_optimize(h, n, deadline, alpha_start=50.0, alpha_end=50000.0,
                 lr_start=0.03, lr_end=0.0005, momentum=0.85, n_iter_max=20000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    patience = 0
    patience_max = 2000
    
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        progress = min(1.0, it / n_iter_max)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        val, grad = smooth_max_val_grad(h, n, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            h = h_new
            patience += 1
            if patience > patience_max:
                v *= 0.5
                patience = patience_max // 2
    return best_h, best_c

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = np.max(c)
    shifted = c - m
    shifted = np.clip(shifted, -50, 0)
    e = np.exp(alpha * shifted)
    es = e.sum()
    if es > 1e30:
        return float(m)
    return float(m + np.log(es) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        c_res = compute_c5(res.x, n)
        if c_res < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def gradient_guided_search(h, n, deadline, rng, deltas=(0.005, 0.002, 0.001, 0.0005)):

    best_c = compute_c5(h, n)
    for d in deltas:

            return h, best_c
        for _pass in range(500):
            if time.time() > deadline:

            improved = False
            _, grad = smooth_max_val_grad(h, n, 10000.0)
            pos_idx = np.argsort(grad)[::-1]

            for i in pos_idx[:80]:

                    return h, best_c
                if grad[i] < 0:
                    break
                for j in neg_idx[:80]:
                    if j == i:

                    if grad[j] > 0:

                    hi = h[i] - d
                    hj = h[j] + d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

            if time.time() > t6 - 2:

            h_opt = slsqp_optimize(best_h, n, alpha, maxiter=2000)
            c = compute_c5(h_opt, n)
            if c < best_c:
                best_c = c
                best_h = h_opt

    # Phase 7: Fine pair search (15%)
    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, n, deadline - 2,
                                      [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                      max_passes=400)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
