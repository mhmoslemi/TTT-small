```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            if not mask.any():
                break
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h_curr, n, lag):
    """Correct gradient of g_lag(h) = dx * sum_i h[i]*(1-h[(i+lag)]) over valid range."""
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad * dx
    if lag > 0:
        grad[:n - m] += 1.0
        grad[:n - m] -= h_curr[m:]
        grad[m:] -= h_curr[:n - m]
    else:
        grad[m:] += 1.0
        grad[:n - m] -= h_curr[m:]
        grad[m:] -= h_curr[:n - m]
    return grad * dx

def _smooth_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    if top_idx is not None:
        c = np.correlate(x, 1 - x, mode='full') * dx
        c = c[top_idx]
    else:
        c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    def obj(x):
        return _smooth_obj(x, n, alpha, top_idx)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def outer_approx(h, n, deadline, max_iters=200, init_trust=0.2):
    """Outer approximation with corrected gradient."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = init_trust
    min_trust = 0.003

    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        c_all = compute_all_lags(h_curr, n)
        # Solve LP: min t s.t. g_k(h_curr) + grad_k(h_curr)·(h-h_curr) <= t for all k
        # with 0<=h<=1, sum(h)=n/2, |h-h_curr|<=trust
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0,
                h_var >= 0, h_var <= 1,
                cp.abs(h_var - h_curr) <= trust]
        # Add linearized constraints for all lags (or top ones for efficiency)
        num_lags = len(c_all)
        # Use all lags for accuracy
        for k in range(num_lags):
            lag = k - (n - 1)
            grad = _lag_gradient(h_curr, n, lag)
            cons.append(c_all[k] + grad @ (h_var - h_curr) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solver = None
        result_h = None
        for solver in ['GLOP', 'CBC', 'GLPK']:
            try:
                prob.solve(solver=cp.SOLVER_MAP[solver] if hasattr(cp, 'SOLVER_MAP') else solver,
                           **({'time_limit': 5} if solver != 'GLOP' else {}))
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    if h_var.value is not None:
                        result_h = h_var.value
                    break
            except Exception:
                continue
        if result_h is None:
            try:
                prob.solve(solver='GLOP')
                if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
                    result_h = h_var.value
            except Exception:
                pass
        if result_h is None:
            break
        h_new = project_feasible(result_h, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:

        h_curr = h_new
        if c_new <= best_c + 1e-12:
            trust = max(min_trust, trust * 0.85)
        else:
            trust = min(init_trust, trust * 1.3)

def local_search(h, n, deadline):

    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c_all = compute_all_lags(h, n)
            k_star = int(np.argmax(c_all))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            # Try moves that reduce the worst lag
            if lag > 0:
                # c = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
                # To reduce: move mass from high h[i+m] to low h[i] (for i+m > i)
                # Or move mass out of positions that contribute to the sum
                for i in range(n - m):
                    if time.time() > deadline - 1:
                        break
                    j = i + m
                    # Moving mass from j to i: h[i] increases, h[j] decreases
                    # c changes by: (1-h[j]-delta)*(delta) - h[i]*(1-h[j]-delta) ... complex
                    # Just try it
                    if h[j] > 0.01:
                        delta = min(d, h[j] * 0.4, 1.0 - h[i])
                        if delta < 1e-9:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                    # Also try moving mass from i to j (opposite direction)
                    # This might help other lags
                    if h[i] > 0.01:
                        delta = min(d, h[i] * 0.4, 1.0 - h[j])
                        if delta < 1e-9:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
            else:
                L = -lag
                for i in range(L, n):
                    if time.time() > deadline - 1:
                        break
                    j = i - L
                    if h[i] > 0.01:
                        delta = min(d, h[i] * 0.4, 1.0 - h[j])
                        if delta < 1e-9:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    if h[j] > 0.01:
                        delta = min(d, h[j] * 0.4, 1.0 - h[i])
                        if delta < 1e-9:
                            continue
                        h[j] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
    return h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    """Full optimization pipeline at a given resolution."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Step 1: Outer approximation
    if time.time() < deadline - 20:
        oo_end = time.time() + min(100, (deadline - time.time()) * 0.4)
        h_oo, c_oo = outer_approx(best_h, n, oo_end)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo

    # Step 2: Active set SLSQP
    if time.time() < deadline - 15:
        as_end = time.time() + min(60, (deadline - time.time()) * 0.35)
        h_as, c_as = active_set_slsqp(best_h, n, as_end)
        if c_as < best_c:
            best_c = c_as
            best_h = h_as

    # Step 3: Full SLSQP with high alpha
    if time.time() < deadline - 10:
        h_new = slsqp_optimize(best_h, n, 200000, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Step 4: Local search
    if time.time() < deadline - 3:
        h_new, c_new = local_search(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at initial resolution (300) - allocate ~35%
    if time.time() < deadline - 50:
        phase1_end = t0 + budget_s * 0.35
        h_opt, c_opt = optimize_at_resolution(best_h, n_init, min(phase1_end, deadline - 60), rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Try higher resolutions - allocate ~40%
    n_candidates = [350, 400, 500, 600]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 60
        if remaining < 40:
            break
        alloc = min(100, remaining * 0.3)
        if alloc < 35:
            break
        # Interpolate from best solution
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Try lower resolutions too (sometimes coarser is easier to optimize well)
    if time.time() < deadline - 100:
        for n_try in [200, 250]:
            if time.time() > deadline - 80:
                break
            remaining = deadline - time.time() - 50
            if remaining < 35:
                break
            alloc = min(70, remaining * 0.3)
            if alloc < 30:
                break
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
            h_start = project_feasible(h_start, n_try)
            h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Restart from best with perturbations
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final polish at best resolution
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # High-precision SLSQP
        if time.time() < deadline - 10:
            h_new = slsqp_optimize(best_h, n, 500000, maxiter=4000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Local search
        if time.time() < deadline - 2:
            h_new, c_new = local_search(best_h, n, deadline - 1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)

    return best_h, best_c, len(best_h)
```
