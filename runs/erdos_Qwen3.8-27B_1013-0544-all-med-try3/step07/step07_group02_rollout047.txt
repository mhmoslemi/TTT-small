def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h, n, deadline, max_iters=100000):
    """Local search on binary solutions: swap a 1 and a 0, accept if max lag decreases."""
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones = np.sum(h_bin)
    if ones != n // 2:
        if ones > n // 2:
            # Flip some 1s to 0s
            one_idx = np.where(h_bin == 1)[0]
            np.random.shuffle(one_idx)
            h_bin[one_idx[:int(ones - n//2)]] = 0
        else:
            zero_idx = np.where(h_bin == 0)[0]
            np.random.shuffle(zero_idx)
            h_bin[zero_idx[:int(n//2 - ones)]] = 1
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    c_all = compute_all_lags(h_bin, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Find the worst lag
        k_star = np.argmax(c_all)
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        # Find a 1 and a 0 that are m apart
        if lag > 0:
            # h[i]=1, h[i+m]=0 would reduce c[lag]
            candidates_1 = np.where(h_bin[:n-m] == 1)[0]
            candidates_0 = np.where(h_bin[m:] == 0)[0] + m
            if len(candidates_1) == 0 or len(candidates_0) == 0:
                break
            # Try a random candidate
            i = candidates_1[np.random.randint(len(candidates_1))]
            j = np.random.randint(n)  # random position
            # Actually, let's try swapping i (which is 1) with some random 0
            zeros = np.where(h_bin == 0)[0]
            if len(zeros) == 0:
                break
            j = zeros[np.random.randint(len(zeros))]
            if j == i:
                continue
            # Swap
            h_bin[i] = 0
            h_bin[j] = 1
            c_new = compute_c5(h_bin, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_bin.copy()
                c_all = compute_all_lags(h_bin, n)
            else:
                h_bin[i] = 1
                h_bin[j] = 0
    
    return best_h, best_c

def swap_delta(h, p, q, n):
    """Compute the change in all c[k] when h[p] goes 1→0 and h[q] goes 0→1."""
    # p: 1→0, q: 0→1
    delta_c = np.zeros(2*n-1)
    
    # Position p was 1, now 0
    # As first factor (h[p]): contributes h[p]*(1-h[p+lag]) = 1*(1-h[p+lag])
    # After: 0. Change: -(1-h[p+lag])
    # Valid: 0 <= p+lag < n, i.e., -p <= lag <= n-1-p
    lags_p_first = np.arange(max(-p, -(n-1)), min(n-1-p, n-1) + 1)
    for lag in lags_p_first:
        idx = lag + (n-1)  # index in full correlation
        delta_c[idx] -= (1 - h[p + lag])
    
    # As second factor ((1-h[p])): contributes h[p-lag]*(1-h[p]) = h[p-lag]*0 = 0
    # After: h[p-lag]*(1-0) = h[p-lag]. Change: +h[p-lag]
    # Valid: 0 <= p-lag < n, i.e., p-n+1 <= lag <= p
    lags_p_second = np.arange(max(p-n+1, -(n-1)), min(p, n-1) + 1)
    for lag in lags_p_second:
        idx = lag + (n-1)
        delta_c[idx] += h[p - lag]
    
    # Position q was 0, now 1
    # As first factor (h[q]): was 0, now (1-h[q+lag])
    # Change: +(1-h[q+lag])
    lags_q_first = np.arange(max(-q, -(n-1)), min(n-1-q, n-1) + 1)
    for lag in lags_q_first:
        idx = lag + (n-1)
        delta_c[idx] += (1 - h[q + lag])
    
    # As second factor ((1-h[q])): was h[q-lag]*(1-0) = h[q-lag], now h[q-lag]*0 = 0
    # Change: -h[q-lag]
    lags_q_second = np.arange(max(q-n+1, -(n-1)), min(q, n-1) + 1)
    for lag in lags_q_second:
        idx = lag + (n-1)
        delta_c[idx] -= h[q - lag]
    
    return delta_c
