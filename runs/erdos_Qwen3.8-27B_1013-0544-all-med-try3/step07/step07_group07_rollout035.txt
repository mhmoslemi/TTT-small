        chunk_size = 20

            chunk = zeros[start:start+chunk_size]
            # Shape: (chunk_size, len(ones), 2n-1)
            M = corr[None, None, :] + flip[chunk][:, None, :] - flip[ones][None, :, :]
            results = M.max(axis=2)  # (chunk_size, len(ones))

            break

        h[best_i] = 1.0
        h[best_j] = 0.0

        if cur_max < best_max:
