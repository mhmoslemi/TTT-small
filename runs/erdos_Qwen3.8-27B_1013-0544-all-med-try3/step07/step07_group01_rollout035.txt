    """Gradient of smooth max over only target_lags"""
    c = compute_c5_vec(h, n)
    # Focus on specific lags
    vals = c[target_lags]
    sh = alpha * (vals - vals.max())
    mx = sh.max()
    w = np.exp(sh - mx)
