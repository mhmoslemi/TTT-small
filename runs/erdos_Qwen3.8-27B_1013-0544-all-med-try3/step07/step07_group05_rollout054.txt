import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlation(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = compute_correlation(x, n) * dx
    return smooth_max(c, alpha)

def _sum_con_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, deadline=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        if deadline is not None and time.time() > deadline - 2:
            return h.copy()
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_obj_1d(t, c0, slope, B_n1, C_n1):
    vals = c0 + slope * t
    vals[n_minus_1] = -t * t + B_n1 * t + C_n1
    return vals.max()

n_minus_1 = 0  # global, set per call

def coord_descent(h, n, deadline, max_passes=100000):
    global n_minus_1
    n_minus_1 = n - 1
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target_sum = n / 2.0
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return best_h, best_c
            old_val = h[j]
            h[j] = 0.0
            c0 = compute_correlation(h, n)
            h[j] = 1.0
            c1 = compute_correlation(h, n)
            
            slope = c1 - c0
            B_n1 = c1[n_minus_1] - c0[n_minus_1] + 1.0
            C_n1 = c0[n_minus_1]
            
            res = minimize_scalar(coord_obj_1d, args=(c0, slope, B_n1, C_n1),
                                  bounds=(0, 1), method='bounded',
                                  options={'xatol': 1e-10})
            t_opt = res.x
            h[j] = t_opt
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-14:

        else:

            if not improved and p > 5:

def basin_hopping(h, n, deadline, rng, num_trials=30):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for trial in range(num_trials):
        if time.time() > deadline:
            break
        scale = 0.02 * (1.0 + 0.5 * np.sin(trial * 1.7))
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n)
        
        # Quick SLSQP
        if time.time() > deadline - 5:
            break
        alloc = min(30, (deadline - time.time()) / max(1, num_trials - trial - 1))
        h_opt = slsqp_optimize(h_pert, n, 100000, maxiter=1000, deadline=time.time() + alloc)
        c_opt = compute_c5(h_opt, n)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

                                c = compute_c5(best_h, n)
                                if c < best_c - 1e-15:
                                    best_c = c
                                    improved = True
                                else:
                                    best_h[i] -= sign * d
                                    best_h[j] += sign * d
            if not improved:
                break
    
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def optimize_resolution(h_start, n, deadline, rng):
    """Full optimization pipeline for a given resolution"""
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: SLSQP with moderate alpha
    if time.time() < deadline - 60:
        alloc1 = min(120, (deadline - time.time()) * 0.3)
        ht = slsqp_optimize(best_h, n, 5000, maxiter=3000, deadline=time.time() + alloc1)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Phase 2: SLSQP with higher alpha
    if time.time() < deadline - 60:
        alloc2 = min(100, (deadline - time.time()) * 0.25)
        ht = slsqp_optimize(best_h, n, 50000, maxiter=3000, deadline=time.time() + alloc2)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Phase 3: SLSQP with very high alpha
    if time.time() < deadline - 50:
        alloc3 = min(80, (deadline - time.time()) * 0.2)
        ht = slsqp_optimize(best_h, n, 500000, maxiter=4000, deadline=time.time() + alloc3)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Phase 4: Coordinate descent
    if time.time() < deadline - 30:
        alloc4 = min(200, (deadline - time.time()) * 0.4)
        ht, ct = coord_descent(best_h, n, time.time() + alloc4)
        if ct < best_c:
            best_c = ct
            best_h = ht
    
    # Phase 5: Focused coordinate descent
    if time.time() < deadline - 20:
        alloc5 = min(100, (deadline - time.time()) * 0.35)
        ht, ct = focused_coord_descent(best_h, n, time.time() + alloc5)
        if ct < best_c:
            best_c = ct
            best_h = ht
    
    # Phase 6: Basin hopping
    if time.time() < deadline - 20:
        alloc6 = min(60, (deadline - time.time()) * 0.4)
        ht, ct = basin_hopping(best_h, n, time.time() + alloc6, rng)
        if ct < best_c:
            best_c = ct
            best_h = ht
    
    # Phase 7: Final SLSQP
    if time.time() < deadline - 10:
        ht = slsqp_optimize(best_h, n, 5000000, maxiter=5000, deadline=time.time() + 50)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Phase 8: Final coordinate descent
    if time.time() < deadline - 5:
        ht, ct = coord_descent(best_h, n, deadline - 2)
        if ct < best_c:
            best_c = ct
            best_h = ht
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    
    best_h = None
    best_c = float('inf')
    
    # Strategy: try multiple resolutions, allocate time based on remaining budget
    
    # Resolution list: try n_init, then a higher resolution
    resolutions = list(dict.fromkeys([n_init, 400, 500]))  # deduplicate, keep order
    # Remove n_init if it's 250 (parent) and we'll try it first
    resolutions = [r for r in resolutions if r >= 100 and r <= 600]
    
    # Allocate time: give more to the first (parent) resolution
    n_res = len(resolutions)
    time_allocs = []
    remaining = deadline - t0
    if n_res == 1:
        time_allocs = [remaining - 10]
    elif n_res == 2:
        time_allocs = [remaining * 0.5 - 5, remaining * 0.5 - 5]
    else:
        time_allocs = [remaining * 0.4 - 5, remaining * 0.35 - 5, remaining * 0.25 - 10]
    
    for idx, n_try in enumerate(resolutions):
        if time.time() > deadline - 20:
            break
        alloc = min(time_allocs[idx], deadline - time.time() - 10)
        if alloc < 30:
            break
        
        # Get starting point for this resolution
        if n_try == n_init:
            h_start = h_init.copy()
        else:
            if best_h is not None:
                h_start = interpolate_h(best_h, len(best_h), n_try)
            else:
                h_start = interpolate_h(h_init, n_init, n_try)
        
        h_opt, c_opt = optimize_resolution(h_start, n_try, time.time() + alloc, rng)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    
    # Final refinement on best solution
    if best_h is not None and time.time() < deadline - 10:
        n_final = len(best_h)
        remaining = deadline - time.time() - 5
        if remaining > 20:
            # One more round of CD + SLSQP
            ht, ct = coord_descent(best_h, n_final, time.time() + min(60, remaining * 0.5))
            if ct < best_c:
                best_c = ct
                best_h = ht
            if time.time() < deadline - 5:
                ht = slsqp_optimize(best_h, n_final, 10000000, maxiter=3000, deadline=deadline - 3)
                c = compute_c5(ht, n_final)
                if c < best_c:
                    best_c = c
                    best_h = ht
            best_h = project_feasible(best_h, n_final)
            best_c = compute_c5(best_h, n_final)
    
    if best_h is None:
        best_h = project_feasible(np.full(200, 0.5), 200)
        best_c = compute_c5(best_h, 200)
    
    return best_h, best_c, len(best_h)
