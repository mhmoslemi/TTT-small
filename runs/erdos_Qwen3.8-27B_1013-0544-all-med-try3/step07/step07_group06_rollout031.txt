def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def to_binary(h, n):

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_binary(h_init, n, deadline, max_sweeps=500):
    """Greedy local search: try all pairs, take best improving move."""
    h = h_init.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c = float(corr.max() * dx)
    best_h = h.copy()
    best_c = cur_c
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = 1e-14
        best_pair = None
        best_new_corr = None
        
        for i in zeros:
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                new_corr = np.correlate(h, 1 - h, mode='full')

                h[i] = 0.0
                h[j] = 1.0

                    best_improve = improve
                    best_pair = (i, j)
                    best_new_corr = new_corr

            break

        h[i] = 1.0
        h[j] = 0.0
        corr = best_new_corr
        cur_c = float(corr.max() * dx)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_binary_fast(h_init, n, deadline, max_sweeps=300):
    """Greedy local search using O(n) incremental updates."""
    h = h_init.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c = float(corr.max() * dx)
    best_h = h.copy()
    best_c = cur_c
    n_full = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = 1e-14
        best_pair = None
        best_new_corr = None
        
        # Precompute effect vectors for all zeros
        # effect_zero[i][k] = change in corr[k] if h[i] goes 0->1 (ignoring interaction with j)
        # effect_one[j][k] = change in corr[k] if h[j] goes 1->0 (ignoring interaction with i)
        
        # For zero at position i, flipping to 1:
        # Δc[k] for shift s=k-(n-1):
        #   + (1-h[i+s]) if 0<=i+s<n
        #   - h[i-s] if 0<=i-s<n
        # For one at position j, flipping to 0:
        #   - (1-h[j+s]) if 0<=j+s<n  
        #   + h[j-s] if 0<=j-s<n
        
        # Combined for pair (i,j): Δc = effect_i + effect_j (approximately, 
        # exact except at s=j-i where we need +2 correction)
        
        # Build effect matrix for all zeros: shape (len(zeros), n_full)
        ez = np.zeros((len(zeros), n_full))
        for zi, i in enumerate(zeros):
            # + (1-h[i+s]) for valid i+s
            s_min = max(0, -i)
            s_max = min(n - 1 - i, n - 1)
            for s in range(s_min, s_max + 1):
                k = s + n - 1
                ez[zi, k] += 1.0 - h[i + s]
            # - h[i-s] for valid i-s
            s_min2 = max(-(n-1), i - (n-1))
            s_max2 = min(i, n - 1)
            for s in range(s_min2, s_max2 + 1):
                k = s + n - 1
                ez[zi, k] -= h[i - s]
        
        # Build effect matrix for all ones: shape (len(ones), n_full)
        eo = np.zeros((len(ones), n_full))
        for oi, j in enumerate(ones):
            # - (1-h[j+s]) for valid j+s
            s_min = max(0, -j)
            s_max = min(n - 1 - j, n - 1)
            for s in range(s_min, s_max + 1):
                k = s + n - 1
                eo[oi, k] -= (1.0 - h[j + s])
            # + h[j-s] for valid j-s
            s_min2 = max(-(n-1), j - (n-1))
            s_max2 = min(j, n - 1)
            for s in range(s_min2, s_max2 + 1):
                k = s + n - 1
                eo[oi, k] += h[j - s]
        
        # For each pair (zi, oi), the approximate new corr is corr + ez[zi] + eo[oi]
        # The max is max(corr + ez[zi] + eo[oi])
        # Add interaction correction at s=j-i: +2
        # And at s=i-j: 0 (net correction is 0 as derived)
        
        # Vectorized: for all zi simultaneously, compute corr + ez[zi] + eo[oi] for each oi
        # This is (len(zeros), len(ones), n_full) which might be too large
        # For n=250: 125*125*499 = 7.8M elements, ~62MB. OK.
        
        # Actually let's do it in batches to save memory
        batch_size = 125
        for zi in range(len(zeros)):
            # For this zero, compute new_max for all ones
            # new_corr_approx = corr + ez[zi] + eo[oi] for each oi
            # new_max = max(corr + ez[zi] + eo[oi])
            # = max((corr + ez[zi]) + eo[oi])
            
            base = corr + ez[zi]  # shape (n_full,)
            # For each oi: max(base + eo[oi])
            # eo is (len(ones), n_full)
            # base[None, :] + eo has shape (len(ones), n_full)
            candidates_max = np.max(base[None, :] + eo, axis=1)  # shape (len(ones),)
            
            for oi in range(len(ones)):
                nc = candidates_max[oi] * dx
                # Apply interaction correction at s = j - i
                j = ones[oi]
                i = zeros[zi]
                s_int = j - i
                k_int = s_int + n - 1
                if 0 <= k_int < n_full:
                    # The correction is +2 at this position
                    val_at_int = (base[k_int] + eo[oi, k_int] + 2.0) * dx
                    nc = max(nc, val_at_int)
                
                improve = cur_c - nc
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (zi, oi, i, j)
        
        if best_pair is None:
            break
        
        zi, oi, i, j = best_pair
        h[i] = 1.0
        h[j] = 0.0
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c = float(corr.max() * dx)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
