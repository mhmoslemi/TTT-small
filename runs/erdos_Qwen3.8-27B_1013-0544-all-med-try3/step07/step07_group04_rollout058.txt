def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha, top_k=None):
    if top_k is not None and len(c) > top_k:
        idx = np.argsort(c)[-top_k:]
        c = c[idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _obj(x, n, alpha, top_k):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c, alpha, top_k)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, top_k, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj, h, args=(n, alpha, top_k), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

def projected_grad_descent(h, n, alpha, top_k, deadline, lr0=0.01, momentum=0.9):
    h_curr = h.copy()
    v = np.zeros(n)
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    lr = lr0
    step = 0
    while time.time() < deadline:
        grad = compute_grad(h_curr, n, alpha, top_k)
        v = momentum * v - lr * grad
        h_new = project_feasible(h_curr + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            lr = min(lr * 1.05, 0.1)  # increase learning rate slowly
        else:
            v = np.zeros(n)
            lr *= 0.7  # reduce learning rate
            h_curr = project_feasible(h_curr - lr * grad, n)
        step += 1
        if step % 100 == 0:
            alpha = min(alpha * 1.5, 1e6)  # increase sharpness
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha, top_k):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    N = 2 * n - 1
    if top_k and top_k < N:
        idx = np.argsort(c)[-top_k:]
        c = c[idx]
    m = c.max()
    a = alpha
    shifted = a * (c - m)
    mx = shifted.max()
    if mx > 400:
        shifted = shifted - mx
        return float((m + mx) + np.log(np.sum(np.exp(shifted))) / a)
    return float(m + np.log(np.sum(np.exp(shifted))) / a)
