import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def logsumexp_shifts_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in active])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 300:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def logsumexp_shifts_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in active])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [50000, 200000, 1000000, 5000000]
    
    for rnd in range(max_rounds):

        max_val = c_vec.max()

        active_candidates = np.where(c_vec >= threshold)[0]
        
        if len(active_candidates) == 0:

        # Limit to top 8 for computational efficiency

            sorted_active = active_candidates[np.argsort(-c_vec[active_candidates])]

def fine_directional_search(h, n, deadline, max_iters=20000):
    """Very fine local search using gradient information."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in steps:
        if time.time() > deadline - 5:
            break
        iter_count = 0
        for _ in range(max_iters):
            if time.time() > deadline - 5:
                return best_h, best_c
            if iter_count % 500 == 0:
                best_h, best_c = h.copy(), compute_c5(h, n)
            
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-6, 0.001 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) == 0:
                break
            
            # Compute average gradient over active shifts
            grad = np.zeros(n)
            for idx in active:
                grad += shift_grad_single(h, n, idx)
            grad /= len(active)
            
            # Find best pair move
            gi_sorted = np.argsort(-np.abs(grad))[:30]
            best_improve = -np.inf
            best_move = None
            
            for ii in range(len(gi_sorted)):
                i = gi_sorted[ii]
                for jj in range(len(gi_sorted)):
                    if ii == jj:
                        continue
                    j = gi_sorted[jj]
                    gi, gj = grad[i], grad[j]
                    if gi > gj:
                        di, dj = -d, d
                    else:
                        di, dj = d, -d
                    if h[i] + di < 0 or h[i] + di > 1:
                        continue
                    if h[j] + dj < 0 or h[j] + dj > 1:
                        continue
                    h[i] += di
                    h[j] += dj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        imp = best_c - c
                        if imp > best_improve:
                            best_improve = imp
                            best_move = (i, j, di, dj)
                        best_c = c
                    else:
                        h[i] -= di
                        h[j] -= dj
            
            if best_move is None:
                break
            iter_count += 1
    
    return best_h, best_c

def block_adjustment(h, n, deadline, block_size=3):
    """Try adjusting small blocks of consecutive values."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for _ in range(200):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        active = np.where(c_vec >= max_val - 0.001 * max_val)[0]
        
        if len(active) == 0:
            break
        
        # Compute gradient
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad_single(best_h, n, idx)
        grad /= len(active)
        
        # Try small block adjustments
        improved = False
        for i in range(n - block_size + 1):
            if time.time() > deadline - 5:
                break
            # Center of block
            g_center = grad[i:i+block_size].mean()
            if abs(g_center) < 1e-8:
                continue
            # Try shifting mass in/out of this block
            direction = -np.sign(g_center)
            for step in [0.001, 0.0005, 0.0002]:
                delta = direction * step
                if (best_h[i:i+block_size] + delta).min() < 0 or (best_h[i:i+block_size] + delta).max() > 1:
                    continue
                # Compensate elsewhere (use least gradient location)
                others = np.arange(n)
                others = others[(np.arange(n) < i) | (np.arange(n) >= i + block_size)]
                g_others = grad[others]
                comp_idx = others[np.argmin(-np.abs(g_others))]  # least gradient magnitude
                comp_delta = -delta * block_size
                if best_h[comp_idx] + comp_delta < 0 or best_h[comp_idx] + comp_delta > 1:
                    continue
                best_h[i:i+block_size] += delta
                best_h[comp_idx] += comp_delta
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                else:
                    best_h[i:i+block_size] -= delta
                    best_h[comp_idx] -= comp_delta
        
        if not improved:
            break
    
    return best_h, best_c

        if n_fine < 100:

        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_fine)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = project_feasible(h_fine, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        
        if time.time() > deadline - 80:
            break
        
        # Optimize at fine resolution
        t3 = min(time.time() + 80, deadline - 80)
        h_fine, c_fine = active_set_optimize(h_fine, n_fine, t3, max_rounds=20)
        
        if c_fine < best_c:
            best_h = h_fine
            best_c = c_fine
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick L-BFGS warm-up (60s)
    bounds = [(0.0, 1.0)] * n_orig
    c_vec = compute_c5_vec(best_h, n_orig)
    active = np.where(c_vec >= c_vec.max() - 0.002 * c_vec.max())[0]
    if len(active) > 8:
        active = active[np.argsort(-c_vec[active])][:8]
    
    if time.time() < deadline - 400:
        t1 = min(time.time() + 60, deadline - 400)
        for alpha in [10000, 100000, 1000000]:
            if time.time() > t1 - 5:
                break
            try:
                res = minimize(
                    logsumexp_shifts_obj, best_h, jac=logsumexp_shifts_grad,
                    args=(n_orig, active, alpha, 0.001),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxfun': 30000, 'maxls': 50}
                )
                h_new = project_feasible(res.x, n_orig)
                c_new = compute_c5(h_new, n_orig)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 2: Multi-scale optimization
    if time.time() < deadline - 100:
        best_h, best_c = multi_scale_optimize(best_h, len(best_h), deadline)
    
    # Phase 3: Very fine local search
    n_final = len(best_h)
    if time.time() < deadline - 30:
        t3 = min(time.time() + 60, deadline - 30)
        best_h, best_c = fine_directional_search(best_h, n_final, t3)
    
    # Phase 4: Block adjustments
    if time.time() < deadline - 20:
        t4 = min(time.time() + 30, deadline - 20)
        best_h, best_c = block_adjustment(best_h, n_final, t4)
    
    # Phase 5: Final active-set polish
    if time.time() < deadline - 5:
        best_h, best_c = active_set_optimize(best_h, n_final, deadline, max_rounds=10)
    
    # Final validation
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
