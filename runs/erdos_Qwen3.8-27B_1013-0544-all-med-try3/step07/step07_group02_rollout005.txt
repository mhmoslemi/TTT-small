def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def compute_all_lags(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad

    else:
        grad[0:n-m] += (1.0 - h[m:n])

    return grad

overlap(m) = |{i : h[i]=1, h[i+m]=0, i+m < n}|

We want to minimize max_m overlap(m).

This is a combinatorial optimization problem. The key insight is that for the binary case, we can do exact local search:
- For each pair of positions (i,j), try swapping h[i] and h[j]
- Compute the change in all lags efficiently
- Accept if it reduces the max

But with n=500, there are 500*499/2 ≈ 125K pairs, and for each we'd need to compute the change in ~500 lags. That's 62.5M operations per round. With 100 rounds, that's 6.25B operations. Too slow in pure Python.

But we can be smarter: only try swaps that involve the worst lag.

OK let me just write a solid implementation that focuses on:
1. Good gradient optimization
2. Targeted local search
3. Multiple resolutions
4. Simulated annealing for escape

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-8:
            continue
        lag = k - (n-1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def grad_descent_improved(h, n, alpha, deadline, lr=1.0, momentum=0.9):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    patience = 0
    for it in range(50000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h_curr, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        if patience > 800:
            lr *= 0.6
            v *= 0.3
            patience = 0
        if patience > 2500:
            break
    return best_h, best_c

def _sum_con_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _top_k_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _full_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=3000, top_m=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        if top_m is not None:
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]
            res = minimize(_top_k_obj, h, args=(n, top_idx, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            return project_feasible(res.x, n)
        else:
            res = minimize(_full_obj, h, args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_local_search(h, n, deadline, rng, max_rounds=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        
        # Gradient step on worst lag
        g = _lag_gradient(h, n, lag)
        for ss in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
            h_try = project_feasible(h - ss * g, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                c_all = compute_all_lags(h, n)
                improved = True
                break
        
        if not improved:
            # Targeted pair swaps at worst lag
            for attempt in range(300):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n - m)
                j = i + m
                if h[i] > h[j] + 1e-8:
                    delta = min(0.02 * (1 + 0.5 * rng.random()), (h[i] - h[j]) * 0.5)
                    h[i] -= delta
                    h[j] += delta
                    h = project_feasible(h, n)
                    c_try = compute_c5(h, n)
                    if c_try < best_c - 1e-14:
                        best_c = c_try
                        c_all = compute_all_lags(h, n)
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                        h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-8:
                    delta = min(0.02 * (1 + 0.5 * rng.random()), (h[j] - h[i]) * 0.5)
                    h[i] += delta
                    h[j] -= delta
                    h = project_feasible(h, n)
                    c_try = compute_c5(h, n)
                    if c_try < best_c - 1e-14:
                        best_c = c_try
                        c_all = compute_all_lags(h, n)
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
                        h = project_feasible(h, n)
            
            # Also try top-3 lags
            if not improved:
                top3 = np.argsort(c_all)[-3:]
                for k2 in top3:
                    if time.time() > deadline:
                        break
                    lag2 = k2 - (n - 1)
                    if lag2 == 0:
                        continue
                    m2 = abs(lag2)
                    for attempt in range(100):
                        i = rng.randint(0, n - m2)
                        j = i + m2
                        if abs(h[i] - h[j]) < 1e-8:
                            continue
                        if h[i] > h[j]:
                            delta = 0.01 * rng.random()
                        else:
                            delta = -0.01 * rng.random()
                        h[i] -= delta
                        h[j] += delta
                        h = project_feasible(h, n)
                        c_try = compute_c5(h, n)
                        if c_try < best_c - 1e-14:
                            best_c = c_try
                            c_all = compute_all_lags(h, n)
                            improved = True
                            break
                        else:
                            h[i] += delta
                            h[j] -= delta
                            h = project_feasible(h, n)
                    if improved:
                        break
            
            if not improved:
                break
    return h, best_c

def simulated_annealing(h, n, deadline, rng, T_init=0.008, T_min=0.00005, cooling=0.9997):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    T = T_init
    n_iter = 0
    for it in range(300000):
        if time.time() > deadline:
            break
        if T < T_min:
            break
        # Generate perturbation: pick 2 random positions
        idx = rng.randint(n, 2)
        i, j = idx[0], idx[1]
        if i == j:
            continue
        magnitude = rng.uniform(0.01, 0.08) * (1.0 + 0.3 * T / T_init)
        delta = rng.choice([-1, 1]) * magnitude
        h_new = h_curr.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        delta_c = c_new - best_c
        n_iter += 1
        if delta_c < 0:
            h_curr = h_new
            best_h = h_new.copy()
            best_c = c_new
            T = max(T * cooling ** 0.5, T_min)
        else:
            accept_prob = np.exp(-delta_c / T)
            if rng.random() < accept_prob:
                h_curr = h_new
        T = max(T * cooling, T_min)
    return best_h, best_c

def multi_start_gradient(h, n, deadline, rng, n_starts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for s in range(n_starts):
        if time.time() > deadline - 20:
            break
        scale = 0.015 * (1.0 + 0.5 * s)
        hp = h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        alloc = (deadline - time.time() - 20) / (n_starts - s)
        pd = time.time() + alloc * 0.9
        h_new, c_new = grad_descent_improved(hp, n, 500, pd, lr=1.0, momentum=0.9)
        h_new2, c_new2 = targeted_local_search(h_new, n, time.time() + alloc * 0.1, rng, max_rounds=100)
        if c_new2 < best_c:
            best_c = c_new2
            best_h = h_new2
    return best_h, best_c

def optimize_pipeline(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    phase_time = max(deadline - time.time(), 10)
    
    # Phase 1: Multi-alpha gradient descent
    alphas = [(100, 0.12), (300, 0.12), (800, 0.12), (1500, 0.12), (3000, 0.10)]
    for alpha, frac in alphas:
        if time.time() > deadline - 40:
            break
        pd = time.time() + frac * phase_time
        h_new, c_new = grad_descent_improved(h, n, alpha, pd, lr=1.0, momentum=0.9)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    
    # Phase 2: SLSQP on top lags
    if time.time() < deadline - 30:
        for tm in [3, 5, 7]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_optimize(h, n, 50000, maxiter=2000, top_m=tm)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
    
    # Phase 3: Simulated annealing
    if time.time() < deadline - 25:
        sa_alloc = min(phase_time * 0.15, deadline - time.time() - 15)
        sa_deadline = time.time() + sa_alloc
        h_new, c_new = simulated_annealing(best_h, n, sa_deadline, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Do targeted search after SA
        if time.time() < deadline - 10:
            h_new2, c_new2 = targeted_local_search(best_h, n, deadline - 5, rng, max_rounds=200)
            if c_new2 < best_c:
                best_c = c_new2
                best_h = h_new2
    
    # Phase 4: Final targeted local search
    if time.time() < deadline - 5:
        h_new, c_new = targeted_local_search(best_h, n, deadline - 1, rng, max_rounds=1000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
