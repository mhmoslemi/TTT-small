    # Find top-k lags

    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
        np.clip(h, 0.0, 1.0, out=h)
    return h

def subgrad_max(h, n, n_active=10):
    corr = get_corr(h, n)
    m = 2 * n - 1
    shifts = np.arange(m) - (n - 1)
    top_idx = np.argsort(corr)[-n_active:]
    active_shifts = shifts[top_idx].astype(int)
    grad = np.zeros(n)
    for k in active_shifts:
        k = int(k)
        j = np.arange(n)
        m1 = (j + k >= 0) & (j + k < n)
        grad[m1] += (1.0 - h[j[m1] + k])
        m2 = (j - k >= 0) & (j - k < n)
        grad[m2] -= h[j[m2] - k]
    grad /= n_active
    return float(corr.max() * 2.0 / n), grad

def active_set_gradient(h, n, deadline, n_active=10, base_lr=0.15):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = base_lr
    stuck = 0
    vel = np.zeros(n)
    momentum = 0.7
    
    while time.time() < deadline:
        c_val, grad = subgrad_max(h, n, n_active)
        
        vel = momentum * vel - lr * grad
        improved = False
        for factor in [1.0, 0.7, 0.5, 0.3, 0.15]:
            h_new = project_feasible(h - factor * vel, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.02, 1.5)
                improved = True
                break
            elif c_new < c_val * (2.0 / n) - 1e-10:
                h = h_new
                improved = True
                break
        if not improved:
            lr *= 0.85
            stuck += 1
            vel *= 0.5
            if stuck > 80:
                break
        else:
            stuck = 0
    
    return best_h, best_c

def slsqp_smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def slsqp_sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def slsqp_sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_sum_con, 'jac': slsqp_sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def pair_search(h, n, deadline, deltas, max_passes=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50, max_passes=300):

def triple_search(h, n, deadline, d, max_passes=200):
    """Try moving mass from one position to two others (or vice versa)"""
    h = h.copy()
    best_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            return h, best_c
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return h, best_c
                for k in range(j + 1, n):
                    if time.time() > deadline:
                        return h, best_c
                    # Move d/2 from i to j and d/2 from i to k
                    hi2 = h[i] - d
                    hj2 = h[j] + d / 2
                    hk2 = h[k] + d / 2
                    if 0.0 <= hi2 and 0.0 <= hj2 <= 1.0 and 0.0 <= hk2 <= 1.0:
                        h[i], h[j], h[k] = hi2, hj2, hk2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j], h[k] = hi2 + d, hj2 - d / 2, hk2 - d / 2
        if not improved:
            break
    return h, best_c

def sim_anneal(h, n, deadline, rng, T0=0.003, T_end=0.00005, n_iter=15000):

    best_c = compute_c5(h, n)

    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / n_iter)

        d = rng.uniform(0.002, 0.04) * (1 if rng.rand() < 0.5 else -1)
        hi = h[i] + d
        hj = h[j] - d

            continue

        h = project_feasible(h, n)

            if c_new < best_c:

        else:

def spectral_init(n, rng, n_modes=8):
    """Create initial h with specific spectral structure"""
    h = np.full(n, 0.5)
    freqs = np.arange(1, n_modes + 1)
    for m in freqs:
        amp = rng.uniform(0.05, 0.2) / m
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * m * np.arange(n) / n + phase)
    h = project_feasible(h, n)
    return h

def sawtooth_init(n, direction=1):
    """Create a sawtooth-like initial condition"""

    h = np.where(x < 0.5, 0.5 + direction * x * 0.8, 0.5 - direction * (1 - x) * 0.8)
