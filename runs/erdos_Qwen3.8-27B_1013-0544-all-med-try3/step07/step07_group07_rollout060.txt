            if time.time() > deadline:
                break
            grad = lse_subgrad(h, n, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            direction = -grad / gnorm
            accepted = False
            for ls in range(35):
                h_new = project(h + step * (0.5 ** ls) * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    h = h_new
                    cur_c = c_new
                    step = step * (0.5 ** ls)
                    accepted = True
                    break
            if accepted:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                break
    return best_h, best_c

def binary_local_search(hb, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h = hb.copy()
    corr = get_corr(h, n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    m_range = np.arange(-(n - 1), n)
    n_half = 1
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for idx_i in range(len(zeros_list)):
            if time.time() > deadline:
                break
            i = zeros_list[idx_i]
            if h[i] != 0:
                continue
            best_j = -1
            best_nc = cur_c
            im = i + m_range
            msi = (im >= 0) & (im < n)
            imn = i - m_range
            msin = (imn >= 0) & (imn < n)
            for j in ones_list:
                if h[j] != 1:
                    continue
                delta = np.zeros(2 * n - 1)
                delta[msi] += 1.0 - h[im[msi]]
                delta[msin] -= h[imn[msin]]
                jm = j + m_range
                mskj = (jm >= 0) & (jm < n)
                jmn = j - m_range
                mskjn = (jmn >= 0) & (jmn < n)
                delta[mskj] -= 1.0 - h[jm[mskj]]
                delta[mskjn] += h[jmn[mskjn]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr = get_corr(h, n)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                ones_list.append(i)
                zeros_list[idx_i] = best_j
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def binary_2swap_search(hb, n, deadline, max_rounds=300):
    """Two sequential 1-swaps as a compound move."""
    dx = 2.0 / n
    h = hb.copy()
    corr = get_corr(h, n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n - 1), n)
    
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        ones_set = set(np.where(h == 1)[0].tolist())
        zeros_set = set(np.where(h == 0)[0].tolist())
        if len(ones_set) < 2 or len(zeros_set) < 2:
            break
        improved = False
        
        # Randomly pick a zero to start
        i1_candidates = list(zeros_set)
        np.random.shuffle(i1_candidates)
        for i1 in i1_candidates[:20]:
            if time.time() > deadline:
                break
            if h[i1] != 0:
                continue
            im1 = i1 + m_range
            msi1 = (im1 >= 0) & (im1 < n)
            imn1 = i1 - m_range
            msin1 = (imn1 >= 0) & (imn1 < n)
            
            # Find best first swap
            best_j1 = -1
            best_nc1 = cur_c
            best_delta1 = None
            for j1 in list(ones_set)[:40]:
                if h[j1] != 1:
                    continue
                delta1 = np.zeros(2 * n - 1)
                delta1[msi1] += 1.0 - h[im1[msi1]]
                delta1[msin1] -= h[imn1[msin1]]
                jm1 = j1 + m_range
                mskj1 = (jm1 >= 0) & (jm1 < n)
                jmn1 = j1 - m_range
                mskjn1 = (jmn1 >= 0) & (jmn1 < n)
                delta1[mskj1] -= 1.0 - h[jm1[mskj1]]
                delta1[mskjn1] += h[jmn1[mskjn1]]
                delta1[n - 1] -= 2.0
                k11 = n - 1 + (j1 - i1)
                if 0 <= k11 < 2 * n - 1:
                    delta1[k11] += 1.0
                k12 = n - 1 + (i1 - j1)
                if 0 <= k12 < 2 * n - 1:
                    delta1[k12] += 1.0
                nc1 = (corr + delta1).max() * dx
                if nc1 < best_nc1 - 1e-14:
                    best_nc1 = nc1
                    best_j1 = j1
                    best_delta1 = delta1
            
            if best_j1 < 0:
                continue
            
            # Apply first swap, find best second swap
            h[i1] = 1.0
            h[best_j1] = 0.0
            corr_tmp = corr + best_delta1
            cur_c_tmp = corr_tmp.max() * dx
            
            best_i2 = -1
            best_j2 = -1
            best_nc2 = cur_c_tmp
            best_delta2 = None
            for i2 in zeros_set:
                if i2 == i1:
                    continue
                if h[i2] != 0:
                    continue
                im2 = i2 + m_range
                msi2 = (im2 >= 0) & (im2 < n)
                imn2 = i2 - m_range
                msin2 = (imn2 >= 0) & (imn2 < n)
                for j2 in ones_set:
                    if j2 == best_j1 or j2 == i1:
                        continue
                    if h[j2] != 1:
                        continue
                    delta2 = np.zeros(2 * n - 1)
                    delta2[msi2] += 1.0 - h[im2[msi2]]
                    delta2[msin2] -= h[imn2[msin2]]
                    jm2 = j2 + m_range
                    mskj2 = (jm2 >= 0) & (jm2 < n)
                    jmn2 = j2 - m_range
                    mskjn2 = (jmn2 >= 0) & (jmn2 < n)
                    delta2[mskj2] -= 1.0 - h[jm2[mskj2]]
                    delta2[mskjn2] += h[jmn2[mskjn2]]
                    delta2[n - 1] -= 2.0
                    k21 = n - 1 + (j2 - i2)
                    if 0 <= k21 < 2 * n - 1:
                        delta2[k21] += 1.0
                    k22 = n - 1 + (i2 - j2)
                    if 0 <= k22 < 2 * n - 1:
                        delta2[k22] += 1.0
                    nc2 = (corr_tmp + delta2).max() * dx
                    if nc2 < best_nc2 - 1e-14:
                        best_nc2 = nc2
                        best_i2 = i2
                        best_j2 = j2
                        best_delta2 = delta2
            
            # Revert first swap
            h[i1] = 0.0
            h[best_j1] = 1.0
            
            if best_nc2 < cur_c - 1e-14:
                # Apply both swaps
                h[i1] = 1.0
                h[best_j1] = 0.0
                h[best_i2] = 1.0
                h[best_j2] = 0.0
                corr = get_corr(h, n)
                cur_c = corr.max() * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
            elif best_nc1 < cur_c - 1e-14:
                # Just apply first swap
                h[i1] = 1.0
                h[best_j1] = 0.0
                corr = get_corr(h, n)
                cur_c = corr.max() * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        
        if not improved:
            break
    return best_h, best_c

def anneal_binary(hb, n, deadline, seed=42, n_iters=3000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = hb.copy()
    corr = get_corr(h, n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    T = 0.015
    cool = 0.99999
    stuck = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        accept = False
        if nc <= cur_c:
            accept = True
        elif T > 1e-10:
            ratio = nc - cur_c
            if ratio < T / 1500.0:
                accept = rng.random() < np.exp(-1500.0 * ratio / T)
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            stuck = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            stuck += 1
        T *= cool
        if stuck > 50000:
            stuck = 0
            T = max(T, 0.005)
    return best_h, best_c

def peak_shave(h, n, deadline, n_iter=10000):
    """Target the argmax shift with subgradient descent."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    step = 0.05
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = get_corr(h, n)
        km = np.argmax(corr)
        k_star = km - (n - 1)
        cur_c = corr[km] * (2.0 / n)
        
        # Subgradient of corr[km] w.r.t. h
        # corr[n-1+k] = sum_i (1-h[i]) * h[i+k] for valid i
        # d/dh[j]: (1-h[j-k]) [from h[j]=h[i+k], i=j-k] + (-h[j+k]) [from 1-h[j] where i=j]
        subgrad = np.zeros(n)
        for j in range(n):
            v = 0.0
            if 0 <= j - k_star < n:
                v += (1.0 - h[j - k_star])
            if 0 <= j + k_star < n:
                v -= h[j + k_star]
            subgrad[j] = v
        
        h_new = project(h - step * subgrad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            best_h = h.copy()
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.7
            if step < 1e-10:
                break
    return best_h, best_c

def multi_start_search(h_best, n, deadline, seed=123, n_starts=5):
    """Generate perturbations and re-optimize."""
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c = compute_c5(h_best, n)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        pert = h_best.copy()
        # Randomly flip a few bits
        n_flip = rng.randint(3, 8)
        flip_idx = rng.choice(n, size=n, replace=False)
        for fi in flip_idx[:n_flip]:
            pert[int(fi)] = 1.0 - pert[int(fi)]
        pert = project(pert, n)
        hb = to_binary(pert, n)
        # Quick local search
        dl_sub = min(deadline, time.time() + 30)
        hb, cb = binary_local_search(hb, n, dl_sub, max_sweeps=100)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    np.random.seed(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    
    n = n0
    h0 = project(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: PGD with LSE (30% of time)
    p1 = t0 + 0.25 * total
    alphas1 = [10, 50, 200, 1000, 5000, 20000, 100000]
    best_h, best_c = pgd_solve(best_h, n, alphas1, p1, max_iter=6000)
    
    # Phase 2: Binary local search (15%)
    p2 = t0 + 0.40 * total
    hb = to_binary(best_h, n)
    hb, cb = binary_local_search(hb, n, p2, max_sweeps=600)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 3: 2-swap binary search (15%)
    p3 = t0 + 0.55 * total
    hb2 = to_binary(best_h, n)
    hb2, cb2 = binary_2swap_search(hb2, n, p3, max_rounds=400)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()
    
    # Phase 4: Annealing (20%)
    p4 = t0 + 0.75 * total
    ha = to_binary(best_h, n)
    ha, ca = anneal_binary(ha, n, p4, seed=seed, n_iters=4000000)
    if ca < best_c:
        best_c = ca
        best_h = ha.copy()
    
    # Phase 5: Peak shaving (continuous refinement) (10%)
    p5 = t0 + 0.85 * total
    hs, cs = peak_shave(best_h, n, p5, n_iter=15000)
    if cs < best_c:
        best_c = cs
        best_h = hs.copy()
    
    # Phase 6: Multi-start (7%)
    p6 = t0 + 0.92 * total
    if time.time() < p6 - 15:
        hm, cm = multi_start_search(best_h, n, p6, seed=seed + 7, n_starts=6)
        if cm < best_c:
            best_c = cm
            best_h = hm.copy()
    
    # Phase 7: Final binary polish (5%)
    p7 = dl - 3
    if time.time() < p7 - 10:
        hb3 = to_binary(best_h, n)
        hb3, cb3 = binary_local_search(hb3, n, p7, max_sweeps=300)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()
    
    # Final: try one more peak shave if time
    if time.time() < dl - 2:
        hs2, cs2 = peak_shave(best_h, n, dl - 1, n_iter=5000)
        if cs2 < best_c:
            best_c = cs2
            best_h = hs2.copy()
    
    # Final evaluation
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
