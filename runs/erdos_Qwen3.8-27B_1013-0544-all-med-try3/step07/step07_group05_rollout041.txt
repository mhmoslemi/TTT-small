    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=8000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(logsumexp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def worst_shift_search(h, n, deadline, max_iter=50000):
    """Targeted local search focusing on worst shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_corr = 2 * n - 1
    center = n - 1
    
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = compute_c5_all(best_h, n)
        # Find worst shift
        worst_idx = np.argmax(c_all)
        worst_s = c_all[worst_idx]
        
        # Find second worst to ensure we're making progress
        c_sorted = np.sort(c_all)[::-1]
        second_worst = c_sorted[1] if len(c_sorted) > 1 else 0
        
        # Shift index to physical index offset
        shift_idx = worst_idx - center  # positive means h shifted right
        
        improved = False
        # Try moves that reduce the worst shift
        if shift_idx > 0:
            # h[i] and h[i+shift_idx]
            for i in range(n - shift_idx):
                if time.time() > deadline:
                    break
                if best_h[i] > 0.01 and best_h[i + shift_idx] < 0.99:
                    # Move mass from i to i+shift_idx
                    d = min(best_h[i] * 0.5, (1.0 - best_h[i + shift_idx]) * 0.5, 0.02)
                    if d < 1e-6:
                        continue
                    new_h = best_h.copy()
                    new_h[i] -= d
                    new_h[i + shift_idx] += d
                    new_c = compute_c5(new_h, n)
                    if new_c < best_c - 1e-15:
                        best_h = new_h
                        best_c = new_c
                        improved = True
                        break
                if best_h[i] < 0.99 and best_h[i + shift_idx] > 0.01:
                    d = min((1.0 - best_h[i]) * 0.5, best_h[i + shift_idx] * 0.5, 0.02)
                    if d < 1e-6:
                        continue
                    new_h = best_h.copy()
                    new_h[i] += d
                    new_h[i + shift_idx] -= d
                    new_c = compute_c5(new_h, n)
                    if new_c < best_c - 1e-15:
                        best_h = new_h
                        best_c = new_c
                        improved = True
                        break
        elif shift_idx < 0:
            s = -shift_idx
            for i in range(s, n):
                if time.time() > deadline:
                    break
                if best_h[i - s] > 0.01 and best_h[i] < 0.99:
                    d = min(best_h[i - s] * 0.5, (1.0 - best_h[i]) * 0.5, 0.02)
                    if d < 1e-6:
                        continue
                    new_h = best_h.copy()
                    new_h[i - s] -= d
                    new_h[i] += d
                    new_c = compute_c5(new_h, n)
                    if new_c < best_c - 1e-15:
                        best_h = new_h
                        best_c = new_c
                        improved = True
                        break
                if best_h[i - s] < 0.99 and best_h[i] > 0.01:
                    d = min((1.0 - best_h[i - s]) * 0.5, best_h[i] * 0.5, 0.02)
                    if d < 1e-6:
                        continue
                    new_h = best_h.copy()
                    new_h[i - s] += d
                    new_h[i] -= d
                    new_c = compute_c5(new_h, n)
                    if new_c < best_c - 1e-15:
                        best_h = new_h
                        best_c = new_c
                        improved = True
                        break
        else:
            # Shift 0: h[i](1-h[i]) = 0 for binary, so this shouldn't be worst
            # Try random pair moves
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                d = 0.01 * (1 if best_h[i] > best_h[j] else -1)
                new_h = best_h.copy()
                new_h[i] += d
                new_h[j] -= d
                new_h = np.clip(new_h, 0, 1)
                new_c = compute_c5(new_h, n)
                if new_c < best_c - 1e-15:
                    best_c = new_c
                    best_h = new_h
                    improved = True
        
        if not improved:
            # Try random perturbation to escape
            if it % 100 == 99:
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j:
                    d = 0.03 * (1 if np.random.rand() < 0.5 else -1)
                    new_h = best_h.copy()
                    new_h[i] += d
                    new_h[j] -= d
                    new_h = np.clip(new_h, 0, 1)
                    new_h = project_feasible(new_h, n)
                    new_c = compute_c5(new_h, n)
                    if new_c < best_c + 0.005:  # Allow some worsening
                        best_h = new_h
                        if new_c < best_c:
                            best_c = new_c
    return best_h, best_c

def pair_search_focused(h, n, deadline, deltas, max_passes=200):
    """Pair exchange search focusing on improving the maximum."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_corr = 2 * n - 1
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            c_all = compute_c5_all(best_h, n)
            # Focus on positions that contribute to worst shifts
            worst_idx = np.argmax(c_all)
            shift_idx = worst_idx - (n - 1)
            
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                # Prioritize positions near the worst shift
                j_candidates = []
                if shift_idx != 0:
                    j_target = i + shift_idx if shift_idx > 0 else i + shift_idx
                    if 0 <= j_target < n and j_target != i:
                        j_candidates.append(j_target)
                
                for j in j_candidates:
                    hi = best_h[i] + d
                    hj = best_h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        best_h[i], best_h[j] = hi, hj
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i], best_h[j] = hi - d, hj + d
                    hi2 = best_h[i] - d
                    hj2 = best_h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        best_h[i], best_h[j] = hi2, hj2
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                
                # Also try near pairs
                for j in range(max(0, i-5), min(n, i+6)):
                    if j == i or j in j_candidates:
                        continue
                    hi = best_h[i] + d
                    hj = best_h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        best_h[i], best_h[j] = hi, hj
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i], best_h[j] = hi - d, hj + d
            if not improved:
                break
    return best_h, best_c

def iterative_reweighting(h, n, deadline, n_rounds=5):
    """Iteratively reweight shifts and optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_c5_all(best_h, n)
    weights = np.ones(len(c_all))
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        # Update weights: emphasize worst shifts
        c_max = c_all.max()
        if c_max > 0:
            weights = np.exp(5.0 * c_all / c_max)
            weights /= weights.sum()
        
        # Define weighted objective
        def weighted_obj(x):
            dx = 2.0 / n
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            return float(np.sum(weights * c))
        
        def weighted_grad(x):
            # Numerical gradient would be too slow; use SLSQP without gradient
            return None
        
        # Use SLSQP with weighted sum objective (no log-sum-exp)
        # Actually, let's use a hybrid: weighted log-sum-exp
        def hybrid_obj(x):
            dx_val = 2.0 / n
            c = np.correlate(x, 1.0 - x, mode='full') * dx_val
            # Weighted version: emphasize worst
            w_c = c * weights * (2 * n - 1)  # Renormalize
            m = w_c.max()
            shifted = 100.0 * (w_c - m)
            mx = shifted.max()
            if mx > 500:
                val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / 100.0
            else:
                val = m + np.log(np.sum(np.exp(shifted))) / 100.0
            return float(val)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
        remaining = deadline - time.time()
        maxiter = min(4000, int(remaining * 100))
        try:
            res = minimize(hybrid_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            best_h = project_feasible(res.x, n)
        except Exception:
            pass
        
        c_new = compute_c5(best_h, n)
        if c_new < best_c:
            best_c = c_new
        c_all = compute_c5_all(best_h, n)
    
    return best_h, best_c

def multistart_optimize(h_start, n, deadline, rng, max_starts=3):
    """Optimize from multiple starting points."""
    best_h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(best_h, n)
    
    for start_idx in range(max_starts):
        if time.time() > deadline - 30:
            break
        if start_idx == 0:
            cur_h = best_h.copy()
        else:
            # Perturbed start
            scale = 0.05 * (start_idx + 1)
            perturbed = best_h + rng.normal(0, scale, n)
            cur_h = project_feasible(perturbed, n)
        
        # SLSQP with increasing alpha
        cur_h = cur_h.copy()
        alphas = [2000, 10000, 50000, 200000, 1000000]
        for a in alphas:
            if time.time() > deadline - 20:
                break
            remaining = deadline - time.time() - 15
            maxiter = min(5000, int(remaining * 50))
            if maxiter < 100:
                break
            cur_h = slsqp_optimize(cur_h, n, a, maxiter=maxiter, ftol=1e-17)
            c = compute_c5(cur_h, n)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        
        # Targeted search
        if time.time() < deadline - 10:
            cur_h, c = worst_shift_search(cur_h, n, deadline - 5)
            if c < best_c:
                best_c = c
                best_h = cur_h
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize at initial resolution
    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        c_init = compute_c5(h_init, n_init)
        best_h, best_c = h_init, c_init
        
        alloc1 = min(250, (deadline - t0) * 0.35)
        h_opt, c_opt = multistart_optimize(h_init, n_init, t0 + alloc1, rng, max_starts=2)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: Try higher resolutions
    if time.time() < deadline - 100:
        for n_try in [500, 750]:
            if time.time() > deadline - 60:
                break
            remaining = deadline - time.time() - 30
            alloc = min(180, remaining * 0.4)
            if alloc < 40:
                break
            
            # Upsample from best solution
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            c_interp = compute_c5(h_interp, n_try)
            
            # Also try a random binary initialization
            h_rand = np.zeros(n_try)
            ones = rng.choice(n_try, size=n_try // 2, replace=False)
            h_rand[ones] = 1.0
            
            # Optimize both
            h_opt1, c_opt1 = multistart_optimize(h_interp, n_try, time.time() + alloc * 0.6, rng, max_starts=2)
            if c_opt1 < best_c:
                best_c = c_opt1
                best_h = h_opt1
            
            if time.time() < deadline - 40:
                h_opt2, c_opt2 = multistart_optimize(h_rand, n_try, time.time() + (deadline - 40 - time.time()) * 0.8, rng, max_starts=3)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2
    
    # Phase 3: Final refinement at best resolution
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        
        # SLSQP with very high alpha
        if time.time() < deadline - 15:
            h_final = slsqp_optimize(best_h, n, 5000000, maxiter=8000, ftol=1e-18)
            c_final = compute_c5(h_final, n)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        
        # Focused pair search
        if time.time() < deadline - 5:
            deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
            best_h, best_c = pair_search_focused(best_h, n, deadline - 2, deltas, max_passes=100)
        
        # Final worst-shift search
        if time.time() < deadline - 3:
            best_h, best_c = worst_shift_search(best_h, n, deadline - 1, max_iter=30000)
        
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        n = n_init
        best_h = np.full(n, 0.5)
        best_c = compute_c5(best_h, n)
    
    return best_h, best_c, len(best_h)
