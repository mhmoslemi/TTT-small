            if tot < 1e-15:

        val = mx + (mx2 + np.log(np.sum(np.exp(shifted - mx2)))) / alpha

        else:
            stagnant = 0
        if lr < 1e-14:
            break
    return best_h, best_c

def worst_lag_desc(h, n, deadline, lr_init=0.004):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stagnant = 0
    for _ in range(100000):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h, n)
        m_star = int(np.argmax(overlaps))
        g = grad_single_lag(h, n, m_star)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break
        g = g / gnorm
        step = lr
        improved = False
        for _ in range(40):
            if step < 1e-16:
                break
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-17:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                lr = min(step * 1.02, 0.2)
                break
            step *= 0.5
        if not improved:
            stagnant += 1
            lr *= 0.7
            if stagnant > 80:
                break
        else:
            stagnant = 0
        if lr < 1e-14:
            break
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def resample_func(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturb_and_reopt(h, n, rng, deadline, noise_scale=0.02, opt_time=30):
    h_pert = h + rng.randn(n) * noise_scale
    h_pert = project_feasible(h_pert, n)
    c_pert = compute_c5(h_pert, n)
    if c_pert > compute_c5(h, n) + 0.01:
        return h, compute_c5(h, n)
    t_opt = time.time() + min(opt_time, deadline - time.time() - 5)
    if t_opt <= time.time():
        return h, compute_c5(h, n)
    h_new, c_new = lbfgs_optimize(h_pert, n, t_opt, alpha=50000, maxiter=80)
    if c_new < compute_c5(h, n):
        return h_new, c_new
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS-B initial polish
    t1 = min(time.time() + 70, deadline - 500)
    if time.time() < t1:
        h, c = lbfgs_optimize(h, n, t1, alpha=50000, maxiter=150)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Top-K gradient descent (multiple K values)
    for K in [5, 10, 20, 40]:
        if time.time() > deadline - 300:
            break
        alloc = min(40, deadline - time.time() - 250)
        if alloc < 10:
            break
        h2, c2 = topk_grad_desc(best_h, n, time.time() + alloc, K=K, lr_init=0.006)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: Worst-lag focused descent
    if time.time() < deadline - 250:
        alloc = min(80, deadline - time.time() - 200)
        if alloc > 10:
            h3, c3 = worst_lag_desc(best_h, n, time.time() + alloc, lr_init=0.003)
            if c3 < best_c:
                best_c = c3
                best_h = h3.copy()

    # Phase 4: Perturbation restarts (simulated annealing style)
    if time.time() < deadline - 200:
        perturb_times = 6
        for p_i in range(perturb_times):
            if time.time() > deadline - 100:
                break
            noise_scale = 0.015 * (1.0 - p_i / float(perturb_times))
            opt_t = min(45, (deadline - time.time() - 100) / float(perturb_times - p_i))
            if opt_t < 15:
                break
            h_p, c_p = perturb_and_reopt(best_h, n, rng, deadline,
                                         noise_scale=noise_scale, opt_time=opt_t)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p.copy()
                # Follow up with gradient descent
                if time.time() < deadline - 80:
                    h_f, c_f = topk_grad_desc(best_h, n, time.time() + 20, K=8, lr_init=0.004)
                    if c_f < best_c:
                        best_c = c_f
                        best_h = h_f.copy()

    # Phase 5: Pair search (moderate radius)
    if time.time() < deadline - 120:
        alloc = min(100, deadline - time.time() - 100)
        if alloc > 20:
            h5, c5 = pair_search(best_h, n, time.time() + alloc,
                                 [0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0001],
                                 radius=100)
            if c5 < best_c:
                best_c = c5
                best_h = h5.copy()

    # Phase 6: Multi-resolution exploration
    if time.time() < deadline - 100:
        nf = len(best_h)
        for n_try in [400, 500, 250, 350, 450]:
            if time.time() > deadline - 50:
                break
            remaining = deadline - time.time() - 30
            if remaining < 35:
                break
            h_interp = resample_func(best_h, nf, n_try)
            alloc1 = min(35, remaining * 0.35)
            if alloc1 < 12:
                continue
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc1,
                                      alpha=50000, maxiter=100)
            if time.time() < deadline - 25:
                alloc2 = min(25, deadline - time.time() - 15)
                if alloc2 > 8:
                    h_t, c_t = topk_grad_desc(h_t, n_try, time.time() + alloc2, K=10, lr_init=0.004)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t.copy()
                nf = n_try

    # Phase 7: Final L-BFGS + top-K gradient polish
    nf = len(best_h)
    if time.time() < deadline - 15:
        alloc = min(30, deadline - time.time() - 10)
        if alloc > 5:
            h7, c7 = lbfgs_optimize(best_h, nf, time.time() + alloc * 0.5,
                                    alpha=200000, maxiter=80)
            if c7 < best_c:
                best_c = c7
                best_h = h7.copy()
    if time.time() < deadline - 8:
        alloc = min(20, deadline - time.time() - 5)
        if alloc > 5:
            h7, c7 = topk_grad_desc(best_h, nf, time.time() + alloc, K=5, lr_init=0.003)
            if c7 < best_c:
                best_c = c7
                best_h = h7.copy()

    # Phase 8: Final fine pair search
    if time.time() < deadline - 3:
        h8, c8 = pair_search(best_h, nf, deadline - 2,
                             [0.0005, 0.0003, 0.0001, 0.00005, 0.00002],
                             radius=60)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
