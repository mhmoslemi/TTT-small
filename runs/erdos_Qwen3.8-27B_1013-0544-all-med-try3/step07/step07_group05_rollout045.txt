import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    s = shifted.max()
    return float(c_max + (s + np.log(np.sum(np.exp(shifted - s)))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    s = shifted.max()
    w = np.exp(shifted - s)
    w /= w.sum()
    
    ones_minus_h = 1.0 - h
    grad = np.zeros(n)
    for tau in range(-(n-1), n):
        wt = w[tau + n - 1]
        if wt < 1e-13:
            continue
        j_lo = max(0, -tau)
        j_hi = min(n, n - tau)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += wt * ones_minus_h[j_lo+tau:j_hi+tau]
        j_lo2 = max(0, tau)
        j_hi2 = min(n, tau + n)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wt * h[j_lo2-tau:j_hi2-tau]
    return grad * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            tot = mask.sum()
            h[mask] += diff / tot
        else:
            mask = h > 0.0
            if not mask.any():
                break
            tot = mask.sum()
            h[mask] += diff / tot
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_with_grad(h0, n, alpha, deadline):
    def fun(x):
        return smooth_max_val(x, n, alpha)
    def jac(x):
        return smooth_max_grad(x, n, alpha)
    def con_fun(x):
        return float(np.sum(x) - n/2.0)
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0,1)] * n
    cons = [{'type':'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h0.copy()

def projected_gradient(h, n, alpha, deadline, lr_init=0.001, n_max_iter=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stuck = 0
    for it in range(n_max_iter):
        if time.time() > deadline:
            break
        f, g = smooth_max_val(h, n, alpha), smooth_max_grad(h, n, alpha)
        # Projected gradient step
        h_new = h - lr * g
        # Project onto sum constraint and box
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stuck = 0
            lr = min(lr * 1.05, 0.01)
        else:
            stuck += 1
            lr *= 0.95
            if stuck > 200:
                break
    return best_h, best_c

def active_set_projected_grad(h, n, deadline, K=8, n_iter=30000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_c_array(h, n)
        # Find top-K active lags
        top_idx = np.argsort(c)[-K:]
        # Compute gradient as weighted sum of active lag gradients
        g = np.zeros(n)
        for tau_idx in top_idx:
            tau = tau_idx - (n - 1)
            ones_minus_h = 1.0 - h
            j_lo = max(0, -tau)
            j_hi = min(n, n - tau)
            if j_lo < j_hi:
                g[j_lo:j_hi] += ones_minus_h[j_lo+tau:j_hi+tau]
            j_lo2 = max(0, tau)
            j_hi2 = min(n, tau + n)
            if j_lo2 < j_hi2:
                g[j_lo2:j_hi2] -= h[j_lo2-tau:j_hi2-tau]
        g /= K
        # Gradient step
        step = 0.001 * dx
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.5
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas, max_passes=200):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i+1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi-d, hj+d
                # also try negative direction
                if time.time() > deadline:
                    break
                for j in range(i+1, n):
                    hi = h[i] - d
                    hj = h[j] + d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi+d, hj-d
            if not improved:
                break
    return h, best_c
