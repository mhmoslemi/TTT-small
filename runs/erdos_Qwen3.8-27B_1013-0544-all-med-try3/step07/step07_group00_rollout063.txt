```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n/2.0)**2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    K = 2*n - 1
    grad = np.zeros(n)
    for mm in range(K):
        wm = w[mm]
        if wm < 1e-14:
            continue
        k = mm - (n-1)
        wdx = wm * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wdx * (1.0 - x[k:])
            grad[k:] += wdx * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wdx * (1.0 - x[:nv])
            grad[:nv] += wdx * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def subgradient_step(h, n, step_size, top_frac=0.02):
    """One subgradient step: move in direction that reduces the top correlations."""
    c = compute_c5_vec(h, n)
    m = c.max()
    # Active set: shifts within top_frac of max
    thresh = m - top_frac * m
    active = c >= thresh
    n_active = active.sum()
    if n_active == 0:
        return h
    # Average gradient over active shifts
    dx = 2.0/n
    K = 2*n - 1
    grad = np.zeros(n)
    for mm in range(K):
        if not active[mm]:
            continue
        k = mm - (n-1)
        wdx = dx / n_active
        if k >= 0:
            nv = n - k
            grad[:nv] += wdx * (1.0 - h[k:])
            grad[k:] += wdx * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wdx * (1.0 - h[:nv])
            grad[:nv] += wdx * (-h[kk:])
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def subgradient_anneal(h, n, deadline, init_step=0.01, min_step=1e-6, cooldown=0.995):
    """Subgradient method with annealing."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = init_step
    while time.time() < deadline:
        h_new = subgradient_step(h, n, step)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = best_h.copy()
        step *= cooldown
        if step < min_step:
            break
    return best_h, best_c
```

```python
def coord_descent_sweep(h, n, mu, n_sweeps=100):
    """Coordinate descent for the LSE-regularized problem."""
    target_sum = n / 2.0
    dx = 2.0 / n
    K = 2 * n - 1
    
    for sweep in range(n_sweeps):
        for i in range(n):
            # Compute D_k: coefficient of h[i] in c_k
            # c_k = dx * sum_j h[j](1-h[j+k])
            # For shift k (0-indexed, actual shift = k - (n-1)):
            #   if shift s >= 0: c_s = dx * sum_{j=0}^{n-1-s} h[j](1-h[j+s])
            #     h[i] appears as h[i](1-h[i+s]) if i <= n-1-s
            #                or as h[i-s](1-h[i]) if i-s >= 0
            #   if shift s < 0 (let kk=-s): c_{-kk} = dx * sum_{j=kk}^{n-1} h[j](1-h[j-kk])
            #     h[i] appears as h[i](1-h[i-kk]) if i >= kk
            #                or as h[i+kk](1-h[i]) if i+kk <= n-1
            
            # D[i contribution to c_mm] where mm = k_shift + (n-1):
            D = np.zeros(K)
            for mm in range(K):

                if k >= 0:

                    if i <= n-1-k:

                        D[mm] += dx * (1.0 - h[i+k])
                    if i-k >= 0:

                        D[mm] += dx * (-h[i-k])
                else:
```

```python
def compute_D_matrix(h, n):

    dx = 2.0 / n
    K = 2 * n - 1
    D = np.zeros((n, K))
    
    for mm in range(K):
        k = mm - (n-1)
        if k >= 0:
            nv = n - k
            # First term: h[i](1-h[i+k]) for i=0..nv-1
            D[:nv, mm] += (1.0 - h[k:k+nv])
            # Second term: h[i-k](1-h[i]) for i=k..n-1
            D[k:, mm] += (-h[:nv])

            kk = -k
            nv = n - kk
            # First term: h[i](1-h[i-kk]) for i=kk..n-1
            D[kk:, mm] += (1.0 - h[:nv])
            # Second term: h[i+kk](1-h[i]) for i=0..nv-1
            D[:nv, mm] += (-h[kk:kk+nv])
    
    D *= dx
    return D
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_D_matrix(h, n):
    dx = 2.0 / n
    K = 2 * n - 1
    D = np.zeros((n, K))
    for mm in range(K):
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            D[:nv, mm] += (1.0 - h[k:k+nv])
            D[k:, mm] += (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            D[kk:, mm] += (1.0 - h[:nv])
            D[:nv, mm] += (-h[kk:kk+nv])
    D *= dx
    return D

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    K = 2 * n - 1
    grad = np.zeros(n)
    for mm in range(K):
        wm = w[mm]
        if wm < 1e-15:
            continue
        k = mm - (n - 1)
        wdx = wm * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wdx * (1.0 - x[k:k+nv])
            grad[k:k+nv] += wdx * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:kk+nv] += wdx * (1.0 - x[:nv])
            grad[:nv] += wdx * (-x[kk:kk+nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sum_eq_fun(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def run_lbfgs(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 4000, 'ftol': 1e-19,
                                   'gtol': 1e-15, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def coord_descent(h, n, mu, deadline, n_sweeps=200, n_grid=500, batch_size=8):

    K = 2 * n - 1
    target_sum = n / 2.0
    x_grid = np.linspace(0.0, 1.0, n_grid)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for sweep in range(n_sweeps):
        if time.time() > deadline - 2:

            targets = target_sum - (h_sum - h[start:end])

            f_vals = vals.max(axis=1) + mu * (x_grid[None, :] - targets[:, None]) ** 2
            best_idx = np.argmin(f_vals, axis=1)

def run_slqp(h, n, deadline, alpha=500000, maxiter=3000):

def local_pair_search(h, n, deadline):

    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for d in deltas:
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)

                        best_h[j] += d

        x_fine = np.linspace(0, 2, n_hi)
        h_hi = np.interp(x_fine, np.linspace(0, 2, n), best_h)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()
        else:
            h_hi = best_h.copy()  # keep best at n=500

        if len(h_hi) == 1000:
            # Phase 6: L-BFGS at n=1000
            if time.time() < deadline - 180:
                t6 = min(time.time() + 100, deadline - 180)
                h_hi, c_hi = run_lbfgs(h_hi, 1000, t6,
                    [(500, 50.0), (5000, 500.0), (50000, 5000.0)])
                if c_hi < best_c:
                    best_c = c_hi
                    best_h = h_hi.copy()

            # Phase 7: Coordinate descent at n=1000
            if time.time() < deadline - 100:
                t7 = min(time.time() + 120, deadline - 100)
                h_hi, c_hi = coord_descent(h_hi, 1000, mu=10.0, deadline=t7,
                                            n_sweeps=80, n_grid=400, batch_size=4)
                if c_hi < best_c:
                    best_c = c_hi
                    best_h = h_hi.copy()

            # Phase 8: SLSQP at n=1000
            if time.time() < deadline - 50:
                t8 = min(time.time() + 40, deadline - 50)
                h_hi, c_hi = run_slqp(h_hi, 1000, t8, alpha=2000000, maxiter=2000)
                if c_hi < best_c:
                    best_c = c_hi
                    best_h = h_hi.copy()

            # Phase 9: Final coordinate descent at n=1000
            if time.time() < deadline - 15:
                h_hi, c_hi = coord_descent(h_hi, 1000, mu=50.0, deadline=deadline - 14,
                                            n_sweeps=30, n_grid=1000, batch_size=2)
                if c_hi < best_c:
                    best_c = c_hi
                    best_h = h_hi.copy()

    # Phase 10: Final local search at best resolution
    if time.time() < deadline - 5:
        n_fin = len(best_h)
        t10 = deadline - 3
        best_h, best_c = local_pair_search(best_h, n_fin, t10)

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
```
