        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_moves(h, n, n_steps, deadline, base_lr):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        lr = base_lr / (1.0 + 0.01 * step)
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxima = corr.max()
        active = np.where(corr >= maxima - 1e-6)[0]
        k_star = active[0]
        offset = k_star - (n - 1)
        g = np.zeros(n)
        if offset == 0:
            g[:] = 1.0 - 2.0 * h
        elif offset > 0:
            g[offset:] += (1.0 - h[:n - offset])
            g[:n - offset] += (-h[offset:])
        else:
            o = -offset
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] += (-h[o:])
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
    return h, best_c

def pair_search_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            idx = np.random.permutation(n)
            for ii in range(n):
                i = idx[ii]
                if h[i] + d > 1.0 and h[i] - d < 0.0:
                    continue
                lo = max(0, i - 60)
                hi = min(n, i + 61)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def full_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_with_knots(n_knots, n, deadline, rng, h_init):
    x_knots = np.linspace(0.0, 2.0, n_knots)
    x_samples = np.linspace(0.0, 2.0, n)
    
    def knots_to_h(knots):
        h = np.interp(x_samples, x_knots, knots)
        return project_feasible(h, n)
    
    def obj_knots(knots):
        h = knots_to_h(knots)
        return compute_c5(h, n)
    
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    
    h_init_knots = np.interp(x_knots, x_samples, h_init)
    
    starts = [h_init_knots.copy()]
    for freq in [1, 2, 3, 4]:
        starts.append(0.5 + 0.3 * np.sin(2 * np.pi * freq * x_knots + rng.uniform(0, 2*np.pi)))
    starts.append(0.5 + 0.3 * np.cos(2 * np.pi * x_knots))
    starts.append(0.5 + 0.2 * np.sin(4 * np.pi * x_knots) + 0.1 * np.sin(8 * np.pi * x_knots))
    for _ in range(5):
        starts.append(np.clip(rng.uniform(0.2, 0.8, n_knots), 0, 1))
    
    bounds_k = [(0.0, 1.0)] * n_knots
    
    for s, k0 in enumerate(starts):
        if time.time() > deadline:
            break
        k0 = np.clip(k0, 0, 1)
        alloc = max(2, (deadline - time.time()) / (len(starts) - s))
        try:
            res = minimize(obj_knots, k0, method='Nelder-Mead', bounds=bounds_k,
                          options={'maxiter': int(alloc * 80), 'xatol': 1e-7, 'fatol': 1e-9})
            if res.x is not None:
                h_test = knots_to_h(res.x)
                c_test = compute_c5(h_test, n)
                if c_test < best_c:
                    best_c = c_test
                    best_h = h_test
        except Exception:
            pass
    return best_h, best_c

def structured_start(n, rng):
    x = np.linspace(0, 2, n)
    candidates = []
    for amp in [0.25, 0.3, 0.35, 0.4]:
        for freq in [1, 2, 3]:
            for phase in [0, np.pi/4, np.pi/2]:
                h = 0.5 + amp * np.sin(2*np.pi*freq*x + phase)
                candidates.append(np.clip(h, 0, 1))
    h_sturm = np.array([1.0 if ((i * 0.618033988749895) % 1.0) < 0.5 else 0.0 for i in range(n)])
    candidates.append(h_sturm)
    for _ in range(3):
        h = np.clip(rng.normal(0.5, 0.3, n), 0, 1)
        candidates.append(h)
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: SLSQP on initial (n_init)
    if time.time() < deadline - 30:
        for a in [2000, 10000, 50000]:
            if time.time() > deadline - 30:
                break
            ht = slsqp_optimize(best_h, n_init, a, maxiter=1500)
            c = compute_c5(ht, n_init)
            if c < best_c:
                best_c = c
                best_h = ht
    
    # Phase 2: Local pair search on initial
    if time.time() < deadline - 60:
        if n_init <= 200:
            d_end = min(deadline - 30, time.time() + 120)
            best_h, best_c = full_pair_search(best_h, n_init, d_end,
                                              [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            d_end = min(deadline - 30, time.time() + 120)
            best_h, best_c = pair_search_local(best_h, n_init, d_end,
                                               [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
    
    # Phase 3: Subgradient + pair search
    if time.time() < deadline - 50:
        d_end = min(deadline - 30, time.time() + 40)
        best_h, c_sg = subgradient_moves(best_h, n_init, 500, d_end, base_lr=0.02)
        if c_sg < best_c:
            best_c = c_sg
        if time.time() < deadline - 30:
            d_end2 = min(deadline - 20, time.time() + 40)
            if n_init <= 200:
                best_h, best_c = full_pair_search(best_h, n_init, d_end2,
                                                  [0.01, 0.005, 0.002, 0.001, 0.0005])
            else:
                best_h, best_c = pair_search_local(best_h, n_init, d_end2,
                                                   [0.01, 0.005, 0.002, 0.001, 0.0005])
    
    # Phase 4: Try different n values with knot optimization
    n_candidates = []
    for nc in [150, 180, 200]:
        if nc != n_init:
            n_candidates.append(nc)
    if n_init not in [200, 250]:
        n_candidates.append(250)
    
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        alloc = min(100, (deadline - time.time()) * 0.3)
        if alloc < 30:
            break
        
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        # Knot optimization
        nk = max(15, n_try // 8)
        h_opt, c_opt = optimize_with_knots(nk, n_try, time.time() + alloc * 0.5, rng, h_interp)
        
        # Refine with SLSQP
        if time.time() < deadline - 40:
            ht = slsqp_optimize(h_opt, n_try, 20000, maxiter=1000)
            c = compute_c5(ht, n_try)
            if c < c_opt:
                c_opt = c
                h_opt = ht
        
        # Local search
        if time.time() < deadline - 30:
            d_end = min(deadline - 30, time.time() + alloc * 0.3)
            if n_try <= 200:
                h_opt, c_opt = full_pair_search(h_opt, n_try, d_end,
                                                [0.01, 0.005, 0.002, 0.001, 0.0005])
            else:
                h_opt, c_opt = pair_search_local(h_opt, n_try, d_end,
                                                 [0.01, 0.005, 0.002, 0.001, 0.0005])
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 5: Multi-restart with perturbation on best
    if best_h is not None and time.time() < deadline - 60:
        n_best = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(3, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(40, (deadline - time.time() - 20) * 0.4)
            if alloc < 15:
                break
            ht = slsqp_optimize(hp, n_best, 30000, maxiter=800)
            c = compute_c5(ht, n_best)
            if n_best <= 200:
                ht, c = full_pair_search(ht, n_best, time.time() + alloc * 0.4,
                                         [0.005, 0.002, 0.001, 0.0005])
            else:
                ht, c = pair_search_local(ht, n_best, time.time() + alloc * 0.4,
                                          [0.005, 0.002, 0.001, 0.0005])
            if c < best_c:
                best_c = c
                best_h = ht
    
    # Phase 6: Final intensive polished local search
    if best_h is not None and time.time() < deadline - 15:
        n_best = len(best_h)
        # SLSQP polish
        if time.time() < deadline - 20:
            ht = slsqp_optimize(best_h, n_best, 100000, maxiter=2000)
            c = compute_c5(ht, n_best)
            if c < best_c:
                best_c = c
                best_h = ht
                n_best = len(best_h)
        
        # Final pair search with very fine deltas
        if time.time() < deadline - 5:
            if n_best <= 200:
                best_h, best_c = full_pair_search(best_h, n_best, deadline - 2,
                                                  [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
            else:
                best_h, best_c = pair_search_local(best_h, n_best, deadline - 2,
                                                   [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    # Phase 7: Final subgradient refinement
    if best_h is not None and time.time() < deadline - 3:
        h_sg, c_sg = subgradient_moves(best_h, len(best_h), 200, deadline - 2, base_lr=0.005)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
