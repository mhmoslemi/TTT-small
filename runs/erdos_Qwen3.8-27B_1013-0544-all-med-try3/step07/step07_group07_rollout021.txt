```
def project(h,n):
    target=n/2.0
    hh=np.asarray(h,float)
    lo,hi=-2.0,2.0
    for _ in range(60):
       mid=(lo+hi)/2
       s=np.clip(hh-mid,0,1).sum()
       if s>target: lo=mid
       else: hi=mid
```

```
def pair_new_c(h,c,i,j,delta,n):
    tmp=c.copy()
    d=float(delta)
    if d == 0.0: return tmp
    hrev=h[::-1]
    tmp[n-1-i : 2*n-1-i] += d * (1.0 - h)
    tmp[i : i+n] += -d * hrev
    tmp[n-1-j : 2*n-1-j] += -d * (1.0 - h)
    tmp[j : j+n] += d * hrev
    r1 = n-1+j-i
    if 0 <= r1 < 2*n-1: tmp[r1] += d*d
    r2 = n-1+i-j
    if 0 <= r2 < 2*n-1: tmp[r2] += d*d
    ui=float(h[i]); uj=float(h[j])
    xi=ui+d; yj=uj-d
    tmp[n-1] = c[n-1] - (ui*(1-ui)+uj*(1-uj)) + (xi*(1-xi)+yj*(1-yj))
    return tmp
```
If d=1 and i's h=0,j h=1, c0 exact same. If h and c consistent. Good.

```
   a=alpha*c

   f=sm + pen*dev*dev
```

```python
def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project(h, n):
    target = 0.5 * float(n)
    x = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(x - tau, 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = out < 1.0
            cnt = float(mask.sum())
            if cnt > 0.0:
                out[mask] += diff / cnt
        else:
            mask = out > 0.0
            cnt = float(mask.sum())
            if cnt > 0.0:
                out[mask] += diff / cnt
        out = np.clip(out, 0.0, 1.0)
    return out


def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=float)
    hb[idx[: n // 2]] = 1.0
    return hb


def swap_new_c(h, c, i, j, n):
    tmp = c.copy()
    # finite difference for i: 0 -> 1
    tmp[n - 1 - i : 2 * n - 1 - i] += 1.0 - h
    tmp[i : i + n] -= h[::-1]
    # finite difference for j: 1 -> 0
    tmp[n - 1 - j : 2 * n - 1 - j] += h - 1.0
    tmp[j : j + n] += h[::-1]
    # interaction of the two swapped coordinates
    r1 = n - 1 + j - i
    if 0 <= r1 < 2 * n - 1:
        tmp[r1] += 1.0
    r2 = n - 1 + i - j
    if 0 <= r2 < 2 * n - 1:
        tmp[r2] += 1.0
    # c_0 does not change for binary swaps
    tmp[n - 1] = c[n - 1]
    return tmp

def anneal_binary(hb, n, deadline, seed=42, T0=2.5, cool=0.99996, max_iters=1000000):
    h = np.asarray(hb, dtype=float)
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
    cur = float(c.max())
    best_raw = cur
    best_h = h.copy()
    dx = 2.0 / float(n)

    zeros = []
    ones = []
    for k, v in enumerate(h):
        if v > 0.5:
            ones.append(int(k))
        else:
            zeros.append(int(k))

    rng = np.random.RandomState(seed)
    T = float(T0)
    rec = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if len(zeros) == 0 or len(ones) == 0:
            break
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        if i == j:
            continue
        tmp = swap_new_c(h, c, i, j, n)
        new = float(tmp.max())
        accept = False
        if new < cur - 1e-12:
            accept = True
        elif abs(new - cur) <= 1e-12:
            if rng.random() < 0.35:
                accept = True
        else:
            if T > 1e-5 and (new - cur) <= 4.0:
                p = math.exp(-1.0 * (new - cur) / max(T, 1e-5))
                if rng.random() < p:
                    accept = True

        if accept:
            c = tmp
            cur = new
            h[i] = 1.0
            h[j] = 0.0
            zeros[ai] = j
            ones[bi] = i
            if cur < best_raw:
                best_raw = cur
                best_h = h.copy()
            rec += 1
            if rec % 800 == 0:
                c = np.correlate(h, 1.0 - h, mode="full").astype(float)
                cur = float(c.max())
                if cur < best_raw:
                    best_raw = cur
                    best_h = h.copy()
        T *= cool

    return best_h, best_raw * dx


def pair_new_c(h, c, i, j, delta, n):
    # Increase h[i] by delta, decrease h[j] by delta.
    tmp = c.copy()
    d = float(delta)
    if d == 0.0:
        return tmp

    hrev = h[::-1]
    # first-order change at i
    tmp[n - 1 - i : 2 * n - 1 - i] += d * (1.0 - h)
    tmp[i : i + n] += -d * hrev
    # first-order change at j
    tmp[n - 1 - j : 2 * n - 1 - j] += -d * (1.0 - h)
    tmp[j : j + n] += d * hrev

    # exact quadratic corrections for the two mutual lags
    r1 = n - 1 + j - i
    if 0 <= r1 < 2 * n - 1:
        tmp[r1] += d * d
    r2 = n - 1 + i - j
    if 0 <= r2 < 2 * n - 1:
        tmp[r2] += d * d

    # exact correction for c_0
    ui = float(h[i])
    uj = float(h[j])
    xi = ui + d
    yj = uj - d
    tmp[n - 1] = (
        c[n - 1]
        - (ui * (1.0 - ui) + uj * (1.0 - uj))
        + (xi * (1.0 - xi) + yj * (1.0 - yj))
    )
    return tmp


def targeted_polish(

    cur = float(c.max())
    best_raw = cur
    best_h = h.copy()
    dx = 2.0 / float(n)

    for _ in range(iterations):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode="full").astype(float)
        cur = float(c.max())
        if cur < best_raw:
            best_raw = cur
            best_h = h.copy()

        act = np.argsort(c)[::-1][: int(active_count)]
        cand = []
        seen = set()
        for r in act:
            r = int(r)
            d = r - (n - 1)
            if d == 0:

            if d > 0:
                a = np.arange(0, n - d)
            else:
                a = np.arange(-d, n)
            if len(a) == 0:

            b = a + d
            term = h[a] * (1.0 - h[b])

                    j = bb
                if i == j:

                if abs(float(h[i]) - float(h[j])) < 1e-8:
                    continue
                key = (int(i), int(j))
                if key in seen:
                    continue
                seen.add(key)
                cand.append((float(h[j] - h[i]), int(i), int(j)))

        if len(cand) == 0:

        cand.sort(reverse=True)

        improved = False
        for _, i, j in cand[: int(candidate_limit)]:
            if time.time() > deadline:

            if h[i] >= h[j]:
                i, j = j, i
            maxd = min(1.0 - float(h[i]), float(h[j]))
            if maxd < 1e-6:

            best_delta = 0.0
            best_val = cur
            best_tmp = None
            for frac in (1.0, 0.62, 0.37, 0.19, 0.09):
                delta = frac * maxd
                if delta <= 1e-7:

                tmp = pair_new_c(h, c, int(i), int(j), delta, n)
                val = float(tmp.max())
                if val < best_val - 1e-12:
                    best_val = val
                    best_delta = delta
                    best_tmp = tmp

            if best_delta > 0.0 and best_tmp is not None:
                xi = float(h[i]) + best_delta
                yj = float(h[j]) - best_delta
                h[i] = min(1.0, max(0.0, xi))
                h[j] = min(1.0, max(0.0, yj))
                if abs(h.sum() - 0.5 * n) > 1e-8:
                    h = project(h, n)
                    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
                    cur = float(c.max())
                else:
                    c = best_tmp
                    cur = best_val
                if cur < best_raw:
                    best_raw = cur
                    best_h = h.copy()
                improved = True
                break

        if not improved:
            break

    return best_h, best_raw * dx


def smooth_grad(p, h, n):
    P = p + p[::-1]
    ones = np.ones(n, dtype=float)
    s_full = np.correlate(ones, p, mode="full")
    h_full = np.correlate(h, P, mode="full")
    idx = 2 * n - 2
    return s_full[idx : n - 2 : -1] - h_full[idx : n - 2 : -1]


def _smooth_objective(x):
    n = int(_OPT["n"])
    alpha = float(_OPT["alpha"])
    pen = float(_OPT["pen"])
    c = np.correlate(x, 1.0 - x, mode="full")
    a = alpha * c
    m = float(a.max())
    e = np.exp(a - m)
    p = e / e.sum()
    sm = (m + math.log(float(e.sum()))) / alpha
    g = smooth_grad(p, x, n)
    dev = float(x.sum() - 0.5 * n)
    f = sm + pen * dev * dev
    g = g + 2.0 * pen * dev
    return f, g


def smooth_refine(h, n, deadline, alphas, pen=20.0, maxiter=40):
    best_h = project(np.asarray(h, dtype=float), n).copy()
    best_c = compute_c5(best_h, n)
    if not _HAS_SCIPY_OPT:
        return best_h, best_c

    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        x0 = project(best_h, n).copy()
        _OPT["n"] = int(n)
        _OPT["alpha"] = float(alpha)
        _OPT["pen"] = float(pen)
        try:
            res = _minimize(
                _smooth_objective,
                x0,
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxls": 50,
                    "disp": False,
                },
            )
            x = np.asarray(res.x, dtype=float)
        except Exception:
            break
        x = project(x, n)
        cx = compute_c5(x, n)
        if cx < best_c - 1e-12:
            best_c = cx
            best_h = x.copy()
    return best_h, best_c


def deadline_slice(now, hard, frac, total):
    rem = hard - now
    if rem <= 0.0:
        return now
    return now + min(rem, frac * total)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(20.0, float(budget_s) - 25.0)
    hard = t0 + total
    rng = np.random.RandomState(seed)

    n = 400
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
        if n < 8:
            raise Exception("bad parent size")
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    if n % 2 == 1:
        n2 = n + 1
        x0 = np.linspace(0.0, 2.0, n)
        x1 = np.linspace(0.0, 2.0, n2)
        h0 = np.interp(x1, x0, h0)
        n = n2

    h0 = project(h0, n)
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_n = int(n)
    best_c = compute_c5(best_h, best_n)

    hb0 = to_binary(best_h, best_n)
    cb0 = compute_c5(hb0, best_n)
    if cb0 < best_c:
        best_c = cb0
        best_h = hb0.copy()

    # Cheap alternative low-discrepancy / random starts, only if they beat the current best.
    if best_n > 0:
        irr = (1.0 + math.sqrt(5.0)) / 2.0
        vals = np.mod(np.arange(best_n, dtype=float) * irr, 1.0)
        hst = to_binary(vals, best_n)
        cst = compute_c5(hst, best_n)
        if cst < best_c:
            best_c = cst
            best_h = hst.copy()

        hrd = to_binary(rng.rand(best_n), best_n)
        crd = compute_c5(hrd, best_n)
        if crd < best_c:
            best_c = crd
            best_h = hrd.copy()

    # Phase 1: strong binary local search.
    now = time.time()
    d1 = deadline_slice(now, hard, 0.30, total)
    hb1 = to_binary(best_h, best_n)
    hb1, cb1 = local_search_binary(
        hb1, best_n, d1, active_count=70, top_k=10, max_sweeps=8000
    )
    if cb1 < best_c:
        best_c = cb1
        best_h = hb1.copy()

    # Phase 2: short binary annealing to escape the local basin.
    now = time.time()
    d2 = deadline_slice(now, hard, 0.12, total)
    hb2 = to_binary(best_h, best_n)
    hb2, cb2 = anneal_binary(hb2, best_n, d2, seed=seed)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()

    # Phase 3: exact sum-preserving fractional coordinate polishing.
    now = time.time()
    d3 = deadline_slice(now, hard, 0.18, total)
    hp3, cp3 = targeted_polish(
        best_h,
        best_n,
        d3,
        active_count=14,
        worst_per_lag=18,
        candidate_limit=380,
        iterations=1500,
    )
    if cp3 < best_c:
        best_c = cp3
        best_h = hp3.copy()

    # Phase 4: binarize again and do another local search.
    now = time.time()
    d4 = deadline_slice(now, hard, 0.16, total)
    hb4 = to_binary(best_h, best_n)
    hb4, cb4 = local_search_binary(
        hb4, best_n, d4, active_count=85, top_k=12, max_sweeps=6000
    )
    if cb4 < best_c:
        best_c = cb4
        best_h = hb4.copy()

    # Phase 5: fractional polish again.
    now = time.time()
    d5 = deadline_slice(now, hard, 0.08, total)
    hp5, cp5 = targeted_polish(
        best_h,
        best_n,
        d5,
        active_count=18,
        worst_per_lag=22,
        candidate_limit=450,
        iterations=1000,
    )
    if cp5 < best_c:
        best_c = cp5
        best_h = hp5.copy()

    # Phase 6: optional smooth L-BFGS refinement.
    now = time.time()
    if now < hard - 80 and _HAS_SCIPY_OPT and best_n <= 700:
        d6 = deadline_slice(now, hard, 0.05, total)
        hs6, cs6 = smooth_refine(
            best_h, best_n, d6, alphas=[30, 150, 600], pen=20.0, maxiter=40
        )
        if cs6 < best_c:
            best_c = cs6
            best_h = hs6.copy()

    # Phase 7: optional higher-resolution escape if time remains.
    now = time.time()
    if now < hard - 170 and best_n < 800:
        nh = min(800, max(800, 2 * best_n))
        if nh % 2 == 1:
            nh += 1
        x_old = np.linspace(0.0, 2.0, best_n)
        x_new = np.linspace(0.0, 2.0, nh)
        hh = np.interp(x_new, x_old, best_h)
        hh = project(hh, nh)
        hbh = to_binary(hh, nh)
        cbh = compute_c5(hbh, nh)

        dl = min(hard - 70.0, time.time() + 95.0)
        hbh, cbh = local_search_binary(
            hbh, nh, dl, active_count=85, top_k=5, max_sweeps=3000
        )
        dl2 = min(hard - 50.0, time.time() + 55.0)
        hp_h, cp_h = targeted_polish(
            hbh, nh, dl2,
            active_count=12,
            worst_per_lag=14,
            candidate_limit=180,
            iterations=300,
        )
        if cp_h < cbh:
            hbh, cbh = hp_h, cp_h
        if cbh < best_c:
            best_c = cbh
            best_h = hbh.copy()
            best_n = nh

    # Final tiny polish using remaining safe time.
    now = time.time()
    if now < hard - 8:
        hp_f, cp_f = targeted_polish(
            best_h,
            best_n,
            hard - 5.0,
            active_count=22,
            worst_per_lag=26,
            candidate_limit=500,
            iterations=300,
        )
        if cp_f < best_c:
            best_c = cp_f
            best_h = hp_f.copy()

    if abs(float(best_h.sum()) - 0.5 * best_n) > 1e-8:
        best_h = project(best_h, best_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, best_n)

    return best_h, float(best_c), int(best_n)
```
