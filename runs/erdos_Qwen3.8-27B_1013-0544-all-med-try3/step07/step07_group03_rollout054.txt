def compute_c5(h, n):

def gradient_obj(h, n, alpha):

    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    m = a.max()
    exp_a = np.exp(a - m)
    p = exp_a / exp_a.sum()

    # k=0 term
    grad += p[n - 1] * (1.0 - 2.0 * h)

        pr = p[n - 1 + k]
        if pr < 1e-15:
            continue
        # (1-h[m-k]) for m in [k, n-1]
        grad[k:] += pr * (1.0 - h[:-k])
        # -h[m+k] for m in [0, n-1-k]
        grad[:n - k] -= pr * h[k:]

    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Optimize at initial n
    alloc1 = min(300, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_at_n(best_h, n_init, t0 + alloc1)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    
    # Phase 2: Try other resolutions
    x_init = np.linspace(0, 2, n_init)
    n_trials = [int(n_init * 0.8), int(n_init * 1.2), 200, 300, 400]
    n_trials = sorted(set([n for n in n_trials if 100 <= n <= 500 and n != n_init]))
    
    for n_try in n_trials:
        if time.time() > deadline - 60:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = np.clip(h_interp, 0, 1)
        # Rescale to meet constraint
        h_interp = project_feasible(h_interp, n_try)
        
        alloc = min(150, (deadline - time.time()) * 0.25)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_at_n(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Multi-restart with structured perturbations
    if best_h is not None and time.time() < deadline - 80:
        n_best = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 60))
        
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            # Different perturbation strategies
            if r % 3 == 0:
                # Gaussian perturbation
                scale = 0.02 + 0.05 * (r / max(1, n_restarts - 1))
                hp = best_h + rng.normal(0, scale, n_best)
            elif r % 3 == 1:
                # Sinusoidal perturbation
                x = np.linspace(0, 2, n_best)
                freq = rng.randint(2, 8)
                phase = rng.uniform(0, 2 * np.pi)
                amp = 0.03 + 0.04 * (r / max(1, n_restarts - 1))
                hp = best_h + amp * np.sin(freq * np.pi * x + phase)
            else:
                # Random block flips
                hp = best_h.copy()
                for _ in range(rng.randint(3, 8)):
                    si = rng.randint(0, n_best - 10)
                    sn = rng.randint(5, 20)
                    hp[si:si+sn] += rng.uniform(-0.05, 0.05)
            
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 30) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n_best, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Final intensive polish on best solution
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        # SLSQP with high alpha
        if time.time() < deadline - 10:
            ht = slsqp_opt(best_h, n, 100000, maxiter=2000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
        
        # Fine coordinate descent
        if time.time() < deadline - 5:
            h, best_c = targeted_coord_descent(best_h, n, deadline - 3,
                                                [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        
        # Fine pair descent
        if time.time() < deadline - 2:
            radius = 40 if n > 200 else 70
            best_h, best_c = pair_descent(best_h, n, deadline - 1,
                                          [0.001, 0.0005, 0.0002, 0.0001], radius=radius)
    
    # Final SLSQP polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 200000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
