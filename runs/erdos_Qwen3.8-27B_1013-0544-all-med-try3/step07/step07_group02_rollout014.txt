import numpy as np
from scipy.optimize import minimize
import time

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_subgrad(h, n, top_m=1, tol=1e-6):
    """Compute subgradient of max_k c_k, averaging over near-max lags."""
    c = compute_all_lags(h, n)
    c_max = c.max()
    # Find lags within tol of max
    active = np.where(c > c_max - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    g = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        g += _single_lag_grad(h, n, lag)
    g /= len(active)
    return g, c

def _single_lag_grad(h, n, lag):
    """Gradient of c_lag w.r.t. h."""
    g = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return g
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # c = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        # dc/dh[i] for i < n-m: (1-h[i+m])
        # dc/dh[i] for i >= m and i < n: -h[i-m]
        # These overlap for m <= i < n-m
        g[0:n-m] += (1.0 - h[m:n])
        g[m:n] -= h[0:n-m]
    else:
        # lag < 0, m = -lag
        # c = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        # = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        # dc/dh[i] for i >= m: (1-h[i-m])
        # dc/dh[i] for i < n-m: -h[i+m]
        g[m:n] += (1.0 - h[0:n-m])
        g[0:n-m] -= h[m:n]
    return g

def subgradient_method(h, n, deadline, lower_bound=0.35, top_m=3, tol=1e-5):
    h_curr = h.copy()
    h_nest = h.copy()  # Nesterov point
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    # Adaptive parameters
    lr = 5.0
    decay = 0.9995
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(50000):
        if time.time() > deadline:
            break
        
        # Nesterov step
        h_eval = h_curr + momentum * (h_curr - h_nest)
        h_nest = h_curr.copy()
        
        g, c_all = exact_subgrad(h_eval, n, top_m=top_m, tol=tol)
        c_max = c_all.max()
        
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h_eval.copy()
        
        # Polyak-like step
        gap = c_max - lower_bound
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-12:
            break
        step = lr * gap / (g_norm_sq * n)  # Scale by n to keep reasonable
        step = min(step, 0.5)  # Cap step size
        
        v = momentum * v - step * g
        h_curr = project_feasible(h_curr + v, n)
        
        lr *= decay
        if it % 5000 == 4999:
            # Reset momentum periodically
            v *= 0.5
    
    return best_h, best_c

    c_max = c_all.max()
    active = np.where(c_all >= c_max - top_eps)[0]

    for k in active:
        lag = k - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if lag == 0:
            g += 1.0 - 2.0 * h
        elif lag > 0:
            g[0:n-m] += (1.0 - h[m:n])
            g[m:n] -= h[0:n-m]
        else:
            g[m:n] += (1.0 - h[0:n-m])
            g[0:n-m] -= h[m:n]
    g /= len(active)
    return g, c_max

    h_nest = h0.copy()

    best_c = compute_c5(h0, n)

        if time.time() > deadline:
            break
        if it == 0 or it % 10 == 0:
            h_nest = h.copy()
        
        # Evaluate at Nesterov point
        h_eval = h + beta * (h - h_nest)
        h_nest = h.copy()
        c_all = compute_all_lags(h_eval, n)
        c_max = c_all.max()

            best_c = c_max
            best_h = h_eval.copy()

        gamma = (c_max - lower_bound) / gn

        h = h_new
        
        # Periodically decrease top_eps to focus on the true max
        if it % 2000 == 1999:

The key observation: we're minimizing max_k c_k where c_k is a quadratic form in h. At the optimum, KKT conditions say:

2. Drop constraints that are not binding
3. Solve the reduced problem (fewer quadratic constraints → easier)
4. Check feasibility of dropped constraints

But solving even 3 quadratic constraints is hard (non-convex QCQP).

    c_all = compute_all_lags(h0, n)
    c_max = c_all.max()
    
    # Build LP: minimize t
    # Variables: h[0..n-1], t → total n+1 variables
    # c[0] * x + ... : minimize t → c_obj = [0, ..., 0, 1]
    
    # Constraints: for each active lag k:
    #   ∇c_k(h0)^T h + (c_k(h0) - ∇c_k(h0)^T h0) ≤ t
    #   → ∇c_k(h0)^T h - t ≤ c_k(h0) - ∇c_k(h0)^T h0
    # Equivalently: ∇c_k(h0)^T h - t ≤ c_k(h0) - ∇c_k(h0)^T h0
    # But at h0: ∇c_k(h0)^T h0 = linearization value
    # Actually: c_k(h) ≈ c_k(h0) + ∇c_k(h0)^T (h - h0)
    # So: c_k(h0) + ∇c_k(h0)^T h - ∇c_k(h0)^T h0 ≤ t
    # → ∇c_k(h0)^T h - t ≤ ∇c_k(h0)^T h0 - c_k(h0)
    
    A_rows = []
    b_vals = []
    for k in active_lags:
        g = _single_lag_grad(h0, n, k)
        rhs = np.dot(g, h0) - c_all[k]
        row = np.concatenate([g, [-1.0]])
        A_rows.append(row)
        b_vals.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Sum constraint: sum(h) = n/2
    # 1^T h ≤ n/2 AND 1^T h ≥ n/2 → -1^T h ≤ -n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    bounds = [(0, 1)] * n + [(c_max * 0.5, None)]  # t ≥ c_max/2
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        return project_feasible(h_new, n)
    return h0.copy()

    return np.correlate(h, 1.0 - h, mode='full') * dx

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += d / s * h

    g = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return g
    if lag == 0:

        g[m:n] += (1.0 - h[0:n-m])
        g[0:n-m] -= h[m:n]
    return g

def lp_local_step(h, n, top_m=5, tol=1e-6, step_frac=1.0):
    """One step of LP-based local optimization."""
    c_all = compute_all_lags(h, n)
    c_max = c_all.max()
    
    # Get top active lags
    sorted_idx = np.argsort(c_all)[::-1]
    active = sorted_idx[:top_m]
    # Also include lags within tol of max
    near_max = np.where(c_all >= c_max - tol)[0]
    active = np.unique(np.concatenate([active, near_max]))
    
    # Build LP
    n_vars = n + 1  # h[0..n-1] + t
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    A_rows = []
    b_vals = []
    for k in active:
        lag = k - (n - 1)
        g = single_lag_grad(h, n, lag)
        # Linearized: c_k(h0) + g^T (h - h0) <= t
        # g^T h - t <= g^T h0 - c_k(h0)
        rhs = np.dot(g, h) - c_all[k]
        row = np.zeros(n_vars)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(rhs)
    
    A_ub = np.array(A_rows) if A_rows else np.empty((0, n_vars))
    b_ub = np.array(b_vals) if b_vals else np.empty(0)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000})
        if res.success:
            h_lp = res.x[:n]
            # Damped step
            h_new = h + step_frac * (h_lp - h)
            h_new = project_feasible(h_new, n)
            return h_new, c_max
    except Exception:
        pass
    return h.copy(), c_max

                best_h = h_try.copy()
                h_curr = h_try
            else:

                break

    if len(active) == 0:
        active = np.array([np.argmax(c_all)])
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    A_rows = []
    b_vals = []
    for k in active:
        lag = k - (n - 1)
        g = single_lag_grad(h, n, lag)
        rhs = np.dot(g, h) - c_all[k]
        row = np.zeros(n_vars)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(rhs)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 10000})
        if res.success:
            h_lp = res.x[:n]
            h_new = h + step_frac * (h_lp - h)
            return project_feasible(h_new, n), c_max
    except Exception:
        pass
    return h.copy(), c_max

def iterative_lp(h, n, deadline, top_m=5, tol=1e-5, step_frac=0.9):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    stuck_count = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        h_new, c_max = lp_local_step(h_curr, n, top_m, tol, step_frac)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            stuck_count = 0
        else:
            h_try = h_curr + 0.4 * (h_new - h_curr)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_h = h_try.copy()
                h_curr = h_try
                stuck_count = 0
            else:
                stuck_count += 1
                if stuck_count > 20:
                    break
    return best_h, best_c

def nesterov_subgrad(h0, n, deadline, lower_bound=0.30, top_eps=1e-4):
    h = h0.copy()
    h_nest = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    beta = 0.9
    lr = 3.0
    decay = 0.9998
    for it in range(100000):
        if time.time() > deadline:
            break
        h_eval = h + beta * (h - h_nest)
        h_nest = h.copy()
        h_eval = np.clip(h_eval, 0, 1)
        c_all = compute_all_lags(h_eval, n)
        c_max = c_all.max()
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h_eval.copy()
        active = np.where(c_all >= c_max - top_eps)[0]
        g = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            g += single_lag_grad(h_eval, n, lag)
        g /= len(active)
        gn = np.dot(g, g)
        if gn < 1e-15:
            break
        gamma = lr * (c_max - lower_bound) / (gn * n)
        gamma = min(gamma, 0.05)
        v = beta * v + gamma * g
        h_new = project_feasible(h - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr *= decay
        if it % 5000 == 4999:
            v *= 0.3
            top_eps *= 0.9
    return best_h, best_c

def fast_local_search(h, n, deadline, max_rounds=300):
    h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_max = c_all.max()
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        m = abs(lag)
        if m == 0 or m >= n:
            break
        found = False
        deltas = [0.04, 0.02, 0.01, 0.005, 0.002]
        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(n - m):
                j = i + m
                if h[i] >= h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4)
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        c_all = compute_all_lags(h, n)
                        found = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                else:
                    delta = min(d, (h[j] - h[i]) * 0.4)
                    if delta < 1e-10:
                        continue
                    h[j] -= delta
                    h[i] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        c_all = compute_all_lags(h, n)
                        found = True
                    else:
                        h[j] += delta
                        h[i] -= delta
        if not found:
            top_lags = np.argsort(c_all)[-8:]
            found_any = False
            for k2 in top_lags:
                lag2 = k2 - (n - 1)
                m2 = abs(lag2)
                if m2 == 0 or m2 >= n:
                    continue
                step_i = max(1, (n - m2) // 80)
                for i in range(0, n - m2, step_i):
                    j2 = i + m2
                    if h[i] > h[j2]:
                        delta = 0.01
                        h[i] -= delta
                        h[j2] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-13:
                            best_c = c_new
                            c_all = compute_all_lags(h, n)
                            found_any = True
                        else:
                            h[i] += delta
                            h[j2] -= delta
                    elif h[j2] > h[i]:
                        delta = 0.01
                        h[j2] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-13:
                            best_c = c_new
                            c_all = compute_all_lags(h, n)
                            found_any = True
                        else:
                            h[j2] += delta
                            h[i] -= delta
                if time.time() > deadline:
                    break
            if not found_any:
                break
    return h, best_c

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return smooth_max_value(c, alpha)

def smooth_max_obj_active(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    active = c[top_idx]
    return smooth_max_value(active, alpha)

def _sum_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter, ftol, top_m=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        if top_m is not None:
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]
            res = minimize(smooth_max_obj_active, h, args=(n, alpha, top_idx),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        else:
            res = minimize(smooth_max_obj, h, args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        if res.success:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def optimize_single(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase A: LP iterations
    if time.time() < deadline - 30:
        lp_deadline = min(deadline - 30, time.time() + 60)
        h_new, c_new = iterative_lp(h, n, lp_deadline, top_m=5, tol=1e-5, step_frac=0.85)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    # Phase B: Nesterov subgradient
    if time.time() < deadline - 20:
        ns_deadline = min(deadline - 20, time.time() + 40)
        h_new, c_new = nesterov_subgrad(h, n, ns_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    # Phase C: SLSQP
    if time.time() < deadline - 15:
        for tm in [3, 5]:
            if time.time() > deadline - 10:
                break
            h_new = slsqp_run(h, n, 100000, 2000, 1e-14, top_m=tm)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
    # Phase D: Local search
    if time.time() < deadline - 5:
        ls_deadline = min(deadline - 2, time.time() + (deadline - time.time()) * 0.5)
        h_new, c_new = fast_local_search(best_h, n, ls_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interpolate_h(h_from, n_to):
    n_from = len(h_from)
    if n_from == n_to:
        return h_from.copy()
    x_from = np.linspace(0, 2, n_from)
    x_to = np.linspace(0, 2, n_to)
    h_interp = np.interp(x_to, x_from, h_from)
    return project_feasible(h_interp, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Main optimization at initial resolution
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.30)
        h_opt, c_opt = optimize_single(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_init = np.full(500, 0.5)
        n_init = 500
        alloc = min(150, (deadline - t0) * 0.2)
        h_opt, c_opt = optimize_single(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Try LP-focused optimization on the best
    if time.time() < deadline - 60:
        lp_deadline = min(deadline - 60, time.time() + 80)
        h_new, c_new = iterative_lp(best_h, n_init, lp_deadline, top_m=8, tol=1e-6, step_frac=0.95)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Multi-resolution: try higher n
    n_candidates = [700, 1000, 1500]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 40
        if remaining < 80:
            break
        alloc = min(100, remaining * 0.30)
        if alloc < 50:
            break
        h_start = interpolate_h(best_h, n_try)
        h_opt, c_opt = optimize_single(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Also try lower resolution for potentially different structure
    for n_try in [400, 350]:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 30
        if remaining < 50:
            break
        alloc = min(60, remaining * 0.35)
        if alloc < 30:
            break
        h_start = interpolate_h(best_h, n_try)
        h_opt, c_opt = optimize_single(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.015 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(55, (deadline - time.time() - 25) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_single(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final polish: LP + local search
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Final LP pass
        if time.time() < deadline - 20:
            h_new, c_new = iterative_lp(best_h, n, deadline - 15, top_m=10, tol=1e-7, step_frac=0.99)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Final local search
        if time.time() < deadline - 4:
            best_h, best_c = fast_local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
