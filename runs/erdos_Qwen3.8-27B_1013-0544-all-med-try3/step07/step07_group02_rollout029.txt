def evaluate_swap_change(h, n, a, b, corr):
    """Compute the change in corr for swap h[a]:1->0, h[b]:0->1.
    Returns the new corr array (or just the change)."""

    # Δc[k] = -f(a,k) + f(b,k) for each k
    
    # We need to compute this for all k = -(n-1), ..., n-1
    # But it's 0 for most k (only non-zero when p-k or p+k is in [0,n-1])
    
    # For position a, f(a,k) is non-zero when:
    # 0 ≤ a-k ≤ n-1, i.e., a-(n-1) ≤ k ≤ a
    # or 0 ≤ a+k ≤ n-1, i.e., -a ≤ k ≤ n-1-a
    
    # So f(a,k) is non-zero for k in [a-(n-1), a] ∪ [-a, n-1-a]
    # = [a-n+1, a] ∪ [-a, n-1-a]
    
    # Similarly for b.
    
    # The union of non-zero ranges for both a and b covers at most 2n indices.
    # But we can just compute for all 2n-1 indices in O(n).
    
    delta = np.zeros(2*n-1)

        fa = 0.0
        if 0 <= a - lag <= n-1:
            fa += (1 - h[a - lag])
        if 0 <= a + lag <= n-1:
            fa -= h[a + lag]
        # f(b, lag)
        fb = 0.0
        if 0 <= b - lag <= n-1:
            fb += (1 - h[b - lag])
        if 0 <= b + lag <= n-1:
            fb -= h[b + lag]
        delta[k] = -fa + fb
    return delta

def evaluate_swap_change_vec(h, n, a, b):
    """Vectorized computation of swap effect."""
    lags = np.arange(-(n-1), n)  # size 2n-1
    
    # f(a, lag)
    fa = np.zeros(2*n-1)
    idx1 = (lags >= a-(n-1)) & (lags <= a)  # 0 <= a-lag <= n-1
    fa[idx1] += (1 - h[a - lags[idx1]])
    idx2 = (lags >= -a) & (lags <= n-1-a)  # 0 <= a+lag <= n-1
    fa[idx2] -= h[a + lags[idx2]]
    
    # f(b, lag)
    fb = np.zeros(2*n-1)
    idx1 = (lags >= b-(n-1)) & (lags <= b)
    fb[idx1] += (1 - h[b - lags[idx1]])
    idx2 = (lags >= -b) & (lags <= n-1-b)
    fb[idx2] -= h[b + lags[idx2]]
    
    return -fa + fb

```python
def swap_delta(h, n, a, b):
    """Compute change in correlation array when h[a]:1->0 and h[b]:0->1."""
    # c[k] = sum_n h[n+k] * (1-h[n]) for k = -(n-1), ..., n-1
    # Change from h[a]: 1->0:
    #   -c gets contribution from: -(1-h[a-k]) for valid k where a-k in [0,n-1]
    #                              +h[a+k] for valid k where a+k in [0,n-1]
    #   Wait, I had: ∂c[k]/∂h[a] = (1-h[a-k])*I[valid] - h[a+k]*I[valid]
    #   So change = -∂c[k]/∂h[a] = -(1-h[a-k])*I[valid] + h[a+k]*I[valid]
    
    # Change from h[b]: 0->1:
    #   +∂c[k]/∂h[b] = (1-h[b-k])*I[valid] - h[b+k]*I[valid]
    
    # Total: Δc[k] = -(1-h[a-k])*I[0≤a-k≤n-1] + h[a+k]*I[0≤a+k≤n-1]
    #               + (1-h[b-k])*I[0≤b-k≤n-1] - h[b+k]*I[0≤b+k≤n-1]
    
    delta = np.zeros(2*n-1)
    k_vals = np.arange(-(n-1), n)  # lag values
    
    # For position a (1->0):
    # Term 1: -(1-h[a-k]) for k where 0 <= a-k <= n-1, i.e., a-(n-1) <= k <= a
    mask1 = (k_vals >= a-(n-1)) & (k_vals <= a)
    if np.any(mask1):
        idx = a - k_vals[mask1]  # these are in [0, n-1]
        delta[mask1] += -(1 - h[idx])
    
    # Term 2: +h[a+k] for k where 0 <= a+k <= n-1, i.e., -a <= k <= n-1-a
    mask2 = (k_vals >= -a) & (k_vals <= n-1-a)
    if np.any(mask2):
        idx = a + k_vals[mask2]
        delta[mask2] += h[idx]
    
    # For position b (0->1):
    # Term 3: +(1-h[b-k]) for k where 0 <= b-k <= n-1, i.e., b-(n-1) <= k <= b
    mask3 = (k_vals >= b-(n-1)) & (k_vals <= b)
    if np.any(mask3):
        idx = b - k_vals[mask3]
        delta[mask3] += (1 - h[idx])
    
    # Term 4: -h[b+k] for k where 0 <= b+k <= n-1, i.e., -b <= k <= n-1-b
    mask4 = (k_vals >= -b) & (k_vals <= n-1-b)
    if np.any(mask4):
        idx = b + k_vals[mask4]
        delta[mask4] -= h[idx]
    
    return delta
```

This is O(n) per swap and vectorized. For n=500, this should be very fast (maybe 5-10μs per call).
