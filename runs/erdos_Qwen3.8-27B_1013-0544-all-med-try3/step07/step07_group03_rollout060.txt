    """Find d in [d_lo, d_hi] that minimizes max_k (c[k] + d*A[k])
    c: current correlation values (2n-1 values)
    A: change coefficients for each c_k when h[i]+=d, h[j]-=d

    # We can evaluate f(d) quickly: f(d) = max(c + d*A)

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_influence(h, n):
    m_total = 2 * n - 1
    infl = np.zeros((n, m_total))
    m = np.arange(m_total)
    for i in range(n):
        mask1 = (m >= i) & (m <= i + n - 1) & (m < m_total)
        if np.any(mask1):
            q1 = i - m[mask1] + n - 1
            infl[i, mask1] += (1.0 - h[q1])
        mask2 = (m >= n - 1 - i) & (m <= 2 * n - 2 - i) & (m < m_total)
        if np.any(mask2):
            k2 = i + m[mask2] - n + 1
            infl[i, mask2] += (-h[k2])
    return infl

def ternary_min_convex(c, A, d_lo, d_hi, niter=30):
    lo, hi = d_lo, d_hi
    if hi - lo < 1e-14:
        return (lo + hi) / 2
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c + m1 * A)
        f2 = np.max(c + m2 * A)
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def pair_line_search(h, n, deadline, max_sweeps=50, radius=None, threshold=0.01):
    best_c = float(np.max(compute_c5_vec(h, n)))
    c = compute_c5_vec(h, n)
    infl = compute_influence(h, n)
    npts = len(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        if radius is None:
            pairs = [(i, j) for i in range(npts) for j in range(i + 1, npts)]
        else:
            pairs = []
            for i in range(npts):
                lo_i = max(0, i - radius)
                hi_i = min(npts, i + radius + 1)
                for j in range(lo_i, hi_i):
                    if j != i:
                        pairs.append((i, j))
        for (i, j) in pairs:
            if time.time() > deadline:
                return h, best_c
            A = infl[i] - infl[j]
            amax = A[np.argmax(c)]
            # Quick filter: if the active lag's A is too small in both directions, skip
            if abs(amax) < threshold:
                continue
            # Direction 1: h[i] += d, h[j] -= d, d in [0, min(1-h[i], h[j])]
            d_hi_1 = min(1.0 - h[i], h[j])
            if d_hi_1 > 1e-12:
                d_opt = ternary_min_convex(c, A, 0.0, d_hi_1, niter=25)
                if d_opt > 1e-10:
                    h[i] += d_opt
                    h[j] -= d_opt

                    if new_c < best_c - 1e-15:
                        best_c = new_c
                        improved = True
                        c = compute_c5_vec(h, n)
                        infl = compute_influence(h, n)
                    else:
                        h[i] -= d_opt
                        h[j] += d_opt
            # Direction 2: h[i] -= d, h[j] += d, d in [0, min(h[i], 1-h[j])]

                    else:

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    val = np.log(np.sum(np.exp(shifted))) / alpha
    return float(m + val)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000, deadline=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if deadline is not None and time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
