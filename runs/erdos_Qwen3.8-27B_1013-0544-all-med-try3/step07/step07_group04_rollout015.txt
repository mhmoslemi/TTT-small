So g_0 = dx*m - 2*g_m.

If we want to minimize max(g_0, g_m), we set g_0 = g_m:
dx*m - 2*g_m = g_m → g_m = dx*m/3
Then g_0 = g_m = dx*m/3 = (2/n)*m/3 = 2*m/(n*3) = 2*(n/2)/(n*3) = 1/3.

So if the only relevant lags are 0 and m, the optimal C₅ would be 1/3. But there are other lags too, so the actual optimum is higher.

This analysis suggests that for anti-periodic h, we should optimize to make ∑h[0..m-1]² = (dx*m)/3 = 1/3 to equalize g_0 and g_m.

But other lags (k ≠ 0, ±m) will also matter. Let me not over-constrain and just use the symmetry as initialization.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(80):
        mu = 0.5 * (lo + hi)
        s = np.clip(h - mu, 0.0, 1.0).sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    mu = 0.5 * (lo + hi)
    result = np.clip(h - mu, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-13:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += diff / cnt
    return np.clip(result, 0.0, 1.0)

def _smooth_max_obj(x, n, active_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    ac = c[active_idx]
    amax = ac.max()
    s = alpha * (ac - amax)
    return float(amax + np.log(np.sum(np.exp(s))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_active(h, n, active_idx, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_max_obj, h, args=(n, active_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.asarray(res.x, dtype=float), n)
    except Exception:
        return h.copy()

def balanced_search(h, n, deadline, top_k=1, base_step=0.015, max_rounds=3000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_k:]
        step = base_step / (1.0 + rnd * 0.015)
        if step < 5e-7:
            step = 5e-7
        h_save = h.copy()
        for k_idx in top_idx:
            m = int(k_idx - (n - 1))
            if m == 0:
                continue
            if m > 0:
                diff = h[:n - m] - h[m:]
                move = np.clip(diff * 0.3, -step, step)
                h[:n - m] -= move
                h[m:] += move
            else:
                m2 = -m
                diff = h[m2:] - h[:n - m2]
                move = np.clip(diff * 0.3, -step, step)
                h[m2:] -= move
                h[:n - m2] += move
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
        else:
            h = h_save
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_passes=80):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for p in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        m = k_star - (n - 1)
        if m == 0:
            break
        step = 0.002 / (1.0 + p * 0.08)
        if step < 1e-7:
            step = 1e-7
        improved = False
        if m > 0:
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                diff = h[i] - h[j]
                if abs(diff) < 1e-12:
                    continue
                if diff > 0:
                    delta = min(step, diff * 0.4)
                    h[i] -= delta
                    h[j] += delta
                    h[i] = max(0.0, min(1.0, h[i]))
                    h[j] = max(0.0, min(1.0, h[j]))
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                else:
                    delta = min(step, (-diff) * 0.4)
                    h[i] += delta
                    h[j] -= delta
                    h[i] = max(0.0, min(1.0, h[i]))
                    h[j] = max(0.0, min(1.0, h[j]))
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
        else:
            m2 = -m
            for i in range(m2, n):
                if time.time() > deadline:
                    break
                j = i - m2
                diff = h[i] - h[j]
                if abs(diff) < 1e-12:
                    continue
                if diff > 0:
                    delta = min(step, diff * 0.4)
                    h[i] -= delta
                    h[j] += delta
                    h[i] = max(0.0, min(1.0, h[i]))
                    h[j] = max(0.0, min(1.0, h[j]))
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                else:
                    delta = min(step, (-diff) * 0.4)
                    h[i] += delta
                    h[j] -= delta
                    h[i] = max(0.0, min(1.0, h[i]))
                    h[j] = max(0.0, min(1.0, h[j]))
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
        if not improved:
            break
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 40:
        c = compute_all_lags(best_h, n)
        top10 = np.argsort(c)[-10:]
        h_new = slsqp_active(best_h, n, top10, alpha=5000, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 20:
        h_ns, c_ns = balanced_search(best_h, n, ts=time.time() + min(40, (deadline - time.time()) * 0.3),
                                     top_k=1, base_step=0.02, max_rounds=2000)
        if c_ns < best_c:
            best_c = c_ns
            best_h = h_ns
    if time.time() < deadline - 20:
        h_new = slsqp_active(best_h, n, np.argsort(compute_all_lags(best_h, n))[-5:],
                            alpha=20000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 10:
        h_ns, c_ns = balanced_search(best_h, n, ts=time.time() + min(30, (deadline - time.time()) * 0.4),
                                     top_k=3, base_step=0.008, max_rounds=1500)
        if c_ns < best_c:
            best_c = c_ns
            best_h = h_ns
    if time.time() < deadline - 5:
        h_new = slsqp_active(best_h, n, np.argsort(compute_all_lags(best_h, n))[-3:],
                            alpha=100000, maxiter=4000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 2:
        best_h, best_c = coordinate_descent(best_h, n, deadline - 1, max_passes=60)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
