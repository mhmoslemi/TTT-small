    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    
    for it in range(10000):

            break
        
        c = compute_all_lags(h, n)
        # Smooth max gradient
        m = c.max()
        shifted = alpha * (c - m)

        # Gradient of smooth max
        grad = np.zeros(n)

                grad += p[k_idx] * lag_gradient(h, n, lag)

            h = h_new

        if it % 100 == 0:
            lr = min(lr * 1.05, 1e-3)

    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
        else:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def projected_gradient_descent(h, n, deadline, alpha_start=2000, alpha_end=200000,
                                max_iter=50000, top_m=None, verbose=False):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    prev_grad = None
    prev_h = None
    prev_f = None
    stalled = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # Adaptive alpha
        frac = min(1.0, it / (max_iter * 0.5))
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        c = compute_all_lags(h, n)

        # Compute smooth max gradient
        m_max = c.max()
        shifted = alpha * (c - m_max)
        shifted = np.clip(shifted, -40, 40)
        p = np.exp(shifted)
        p_sum = p.sum()
        p /= p_sum

        # Gradient as weighted sum of lag gradients
        grad = np.zeros(n)
        # Only compute for lags with significant weight
        thresh = max(1e-6, p.max() * 0.001)
        active = np.where(p > thresh)[0]
        for idx in active:
            lag = idx - (n - 1)
            grad += p[idx] * lag_gradient(h, n, lag)

        # Barzilai-Borwein step size
        if prev_grad is not None and prev_h is not None:
            s = h - prev_h
            y = grad - prev_grad
            ys = np.dot(y, s)
            ss = np.dot(s, s)
            yy = np.dot(y, y)
            if ss > 1e-20 and ys > 1e-20:
                lr_bb = ss / ys
            elif yy > 1e-20:
                lr_bb = ys / yy
            else:
                lr_bb = lr
            lr_bb = np.clip(lr_bb, 1e-6, 2.0)
            lr = 0.7 * lr + 0.3 * lr_bb

        # Line search with backtracking
        f_curr = m_max + np.log(p_sum) / alpha
        accepted = False
        for trial in range(20):
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c_new_arr = compute_all_lags(h_new, n)
            f_new = c_new_arr.max() + np.log(np.sum(np.exp(alpha * (c_new_arr - c_new_arr.max())))) / alpha
            if f_new < f_curr - 1e-12 * abs(f_curr):
                accepted = True
                break
            lr *= 0.5

        if not accepted:
            lr *= 0.1
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)

        c_val = compute_c5(h_new, n)
        if c_val < best_c - 1e-14:
            best_c = c_val
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1

        prev_grad = grad.copy()
        prev_h = h.copy()
        h = h_new

        if stalled > 5000:
            break
        if it % 5000 == 0 and verbose:
            print(f"  PGD iter {it}, c5={best_c:.6f}, lr={lr:.2e}, alpha={alpha:.0f}")

    return best_h, best_c

def active_set_slsqp(h, n, top_m=8, alpha=50000, maxiter=3000, deadline=None):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n

    def objective(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        top_idx = np.argsort(c)[-top_m:]
        c_top = c[top_idx]
        m = c_top.max()
        shifted = alpha * (c_top - m)
        shifted = np.clip(shifted, -40, 40)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    def gradient(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        top_idx = np.argsort(c)[-top_m:]
        c_top = c[top_idx]
        m = c_top.max()
        shifted = alpha * (c_top - m)
        shifted = np.clip(shifted, -40, 40)
        p = np.exp(shifted)
        p /= p.sum()
        grad = np.zeros(n)
        for i, idx in enumerate(top_idx):
            lag = idx - (n - 1)
            grad += p[i] * lag_gradient(x, n, lag)
        return grad

    def sum_fun(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]

    try:
        res = minimize(objective, h, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
    except Exception:
        return h.copy()
    c_new = compute_c5(h_new, n)
    if c_new <= compute_c5(h, n) + 1e-10:
        return h_new
    return h.copy()

def directed_local_search(h, n, deadline, block_sizes=(1, 2, 4, 8)):
    h = h.copy()
    best_c = compute_c5(h, n)

    for block_size in block_sizes:
        if time.time() > deadline:
            break
        improved = True
        iter_count = 0
        while improved and time.time() < deadline:
            improved = False
            iter_count += 1
            if iter_count > 500:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break

            # Find pair of blocks that would most reduce the active lag
            if lag > 0:
                # We want to reduce h[i]*1 and increase 1-h[j] for j=i+m
                # So move mass from high h[i] to low h[j] where j=i+m
                # Or move mass from h[j] (high) to h[i] (low) where i=j-m
                if np.random.random() < 0.5:
                    # Move mass from source to target at offset m
                    for i in range(0, n - m, max(1, (n - m) // 40)):
                        j = i + m
                        if h[i] > h[j] + 1e-6:
                            delta = min(0.01, (h[i] - h[j]) * 0.4)
                            delta = max(delta, 1e-5)
                            delta = min(delta, h[i] - h[j] + 1e-9)
                            h[i] -= delta
                            h[j] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] += delta
                                h[j] -= delta
                else:
                    for j in range(m, n, max(1, (n - m) // 40)):
                        i = j - m
                        if h[j] > h[i] + 1e-6:
                            delta = min(0.01, (h[j] - h[i]) * 0.4)
                            delta = max(delta, 1e-5)
                            delta = min(delta, h[j] - h[i] + 1e-9)
                            h[j] -= delta
                            h[i] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                h[j] += delta
                                h[i] -= delta
            else:
                for i in range(0, n - m, max(1, (n - m) // 40)):
                    j = i + m
                    if h[i] > h[j] + 1e-6:
                        delta = min(0.01, (h[i] - h[j]) * 0.4)
                        delta = max(delta, 1e-5)
                        delta = min(delta, h[i] - h[j] + 1e-9)
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta

            # Also try random pairwise moves
            if time.time() > deadline:
                break
            for _ in range(20):
                if time.time() > deadline:
                    break
                i = np.random.randint(0, n)
                j = np.random.randint(0, n)
                if i == j:
                    continue
                if h[i] > h[j] + 1e-4:
                    delta = min(0.005, (h[i] - h[j]) * 0.3)
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-4:
                    delta = min(0.005, (h[j] - h[i]) * 0.3)
                    h[j] -= delta
                    h[i] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[j] += delta
                        h[i] -= delta

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Projected gradient (main workhorse)
    t1 = min(time.time() + 25, deadline - 30)
    if t1 > time.time() + 5:
        h_pg, c_pg = projected_gradient_descent(h, n, t1, alpha_start=1000, alpha_end=100000)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg

    # Phase 2: Active set SLSQP refinement
    if time.time() < deadline - 20:
        for top_m in [6, 12, 20]:
            if time.time() > deadline - 15:
                break
            h_as = active_set_slsqp(best_h, n, top_m=top_m, alpha=50000, maxiter=2000)
            c_as = compute_c5(h_as, n)
            if c_as < best_c - 1e-12:
                best_c = c_as
                best_h = h_as

    # Phase 3: More projected gradient with higher alpha
    t3 = min(time.time() + 20, deadline - 15)
    if t3 > time.time() + 3:
        h_pg2, c_pg2 = projected_gradient_descent(best_h, n, t3, alpha_start=50000, alpha_end=500000)
        if c_pg2 < best_c:
            best_c = c_pg2
            best_h = h_pg2

    # Phase 4: Local search
    if time.time() < deadline - 3:
        h_ls, c_ls = directed_local_search(best_h, n, deadline - 1)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize at initial resolution
    if h_init is not None:
        alloc = min(80, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Phase 2: Multi-resolution exploration
    n_candidates = list(range(200, 601, 50))
    # Sort by priority: close to initial n first, then finer
    if h_init is not None:
        n_candidates.sort(key=lambda x: abs(x - n_init))
    else:
        n_candidates = [250, 300, 350, 400, 450, 500, 550]

    for n_try in n_candidates:
        if time.time() > deadline - 60:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 40
        if remaining < 40:
            break
        alloc = min(60, remaining * 0.3)
        if alloc < 20:
            continue

        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)

        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 30) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.02 + 0.01 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 20) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Final projected gradient with very high alpha
        if time.time() < deadline - 10:
            h_final, c_final = projected_gradient_descent(
                best_h, n, deadline - 5,
                alpha_start=200000, alpha_end=1000000, max_iter=20000)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final

        # Final local search
        if time.time() < deadline - 2:
            best_h, best_c = directed_local_search(best_h, n, deadline - 1)

        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    # Fallback
    if best_h is None:
        n_fb = 300
        best_h = project_feasible(np.full(n_fb, 0.5), n_fb)
        best_c = compute_c5(best_h, n_fb)

    return best_h, best_c, len(best_h)
