import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def topk_grad(h, n, alpha, top_idx):
    c_vec = compute_c5_vec(h, n)
    c_active = np.array([c_vec[i] for i in top_idx])
    m = c_active.max()
    sh = alpha * (c_active - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w = w / (w.sum() + 1e-300)
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wi * (1.0 - h[k:n]) * dx
            grad[k:n] += wi * (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wi * (1.0 - h[0:nv]) * dx
            grad[0:nv] += wi * (-h[kk:n]) * dx
    return grad

def full_smooth_grad(h, n, alpha):
    c_vec = compute_c5_vec(h, n)
    m = c_vec.max()
    sh = alpha * (c_vec - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w = w / (w.sum() + 1e-300)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def full_smooth_obj(h, n, alpha):
    c_vec = compute_c5_vec(h, n)
    m = c_vec.max()
    sh = alpha * (c_vec - m)
    mx = sh.max()
    val = m + np.log(np.sum(np.exp(sh - mx)) + 1e-300) / alpha
    return float(val)

def topk_smooth_obj(h, n, alpha, top_idx):

    c_active = np.array([c_vec[i] for i in top_idx])
    m = c_active.max()
    sh = alpha * (c_active - m)

    return float(val)

def sum_constraint(h, n):
    return float(h.sum() - n / 2.0)

def sum_constraint_jac(h, n):
    return np.ones(n)

def adam_step(h, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    t_new = t + 1
    m_new = beta1 * m + (1.0 - beta1) * grad
    v_new = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m_new / (1.0 - beta1 ** t_new)

    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)

def run_lbfgs_topk(h, n, deadline, top_k=5, alphas=None):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [2000, 10000, 50000, 200000, 1000000]

            h_new = project_feasible(res.x, n)

def run_slqp_opt(h, n, deadline, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]

def fine_pair_search(h, n, deadline, center_radius=40):
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True

    return h, best_c
