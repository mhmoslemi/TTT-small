- g = grad(f(y))
- v = momentum * v + lr * g
- x = y - v

This often converges faster than plain momentum.

Also, I'll add a **line search** step: after computing the gradient direction, do a few trial steps with different learning rates to find the best one.

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    # Final exact fix
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Add/subtract uniformly from elements that have room
        if diff > 0:
            room = 1.0 - h
            idx = np.argsort(room)[::-1]
            for i in idx:
                add = min(diff, room[i])
                h[i] += add
                diff -= add
                if diff < 1e-15:
                    break
        else:
            diff = -diff
            idx = np.argsort(h)[::-1]
            for i in idx:
                sub = min(diff, h[i])
                h[i] -= sub
                diff -= sub
                if diff < 1e-15:
                    break
    return h

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def nesterov_optimize(h, n, alpha, deadline, lr=1.0, momentum=0.9):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    patience = 0
    max_patience = 500
    
    for it in range(100000):
        if time.time() > deadline:
            break
        
        # Nesterov lookahead
        y = h_curr + momentum * v
        y = np.clip(y, 0.0, 1.0)
        
        # Gradient at lookahead point
        g = grad_smooth_max(y, n, alpha)
        
        # Update velocity
        v = momentum * v + lr * g
        
        # Update position
        h_new = h_curr - v
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h_curr = h_new
        
        if patience > max_patience:
            lr *= 0.7
            v *= 0.5
            patience = 0
            max_patience = int(max_patience * 1.2)
        
        if it % 10000 == 9999:
            lr = max(0.05, lr * 0.9)
    
    return best_h, best_c
