def subgrad_c5(h, n):
    """Compute subgradient of max_k corr[k]*dx w.r.t. h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c = corr * dx
    # Find indices where c achieves the max
    max_val = c.max()
    mask = c >= max_val - 1e-12
    # Subgradient of c[k] w.r.t. h:
    # corr[k] = sum_i h[i] * (1-h[i+k]) for valid (i, i+k)
    # d(corr[k])/dh[i] = (1-h[i+k]) - h[i-k] (where indices are valid)
    # More carefully using the correlation definition:
    # np.correlate(a, b, 'full')[n-1+k] = sum_i a[i+k]*b[i] for valid i
    # So corr[n-1+k] = sum_i h[i+k]*(1-h[i]) for i where both i and i+k are in [0,n-1]
    
    grad = np.zeros(n)
    for k in range(-n+1, n):
        if not mask[n-1+k]:
            continue
        # corr[n-1+k] = sum_i h[i+k]*(1-h[i]) for valid i
        # d/dh[j]: 
        #   term where j = i+k: (1-h[j-k]) if j-k valid
        #   term where j = i: -h[j+k] if j+k valid
        i_min = max(0, -k)
        i_max = min(n-1, n-1-k)
        # For i in [i_min, i_max]: term is h[i+k]*(1-h[i])
        # d/dh[i+k] += (1-h[i]) => for j=i+k, j in [i_min+k, i_max+k]
        # d/dh[i] += -h[i+k] => for j=i, j in [i_min, i_max]
        for i in range(i_min, i_max+1):
            grad[i+k] += (1.0 - h[i])
            grad[i] -= h[i+k]
    return grad * dx

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

    im = i + m_range
    msk = (im >= 0) & (im < n)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def swap_delta(corr, h, i, j, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_ls(hb, n, deadline, max_sweeps=400):
    dx = 2.0 / n
    h = hb.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = float(corr.max()) * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones:
                d = swap_delta(corr, h, i, j, n, m_range)
                nc = float((corr + d).max()) * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
                    best_d = d
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += best_d
                cur_c = float(corr.max()) * dx
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True

def pgd_refine(h, n, deadline, alphas=None, max_iter=8000):

        alphas = [15, 60, 250, 800, 3000, 12000, 50000, 200000]
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        cur_c = best_c
        step = min(1.0 / (6.0 * alpha), 3.0 / n)
        for _ in range(max_iter):
            if time.time() > deadline:

            corr = np.correlate(h, 1 - h, mode='full')
            a = alpha * corr
            a_max = a.max()

            p = exp_a / exp_a.sum()
            grad = np.zeros(n)
            active = np.where(p > 1e-9)[0]
            for k_idx in active:
                pk = p[k_idx]
                k = int(k_idx) - (n - 1)

                if i_min > i_max:

                grad[i_min + k:i_max + 1 + k] += pk * (1.0 - h[i_min:i_max + 1])
                grad[i_min:i_max + 1] -= pk * h[i_min + k:i_max + 1 + k]
            grad *= dx
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:

                if step < 1e-14:
                    break
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

    if hap[1] < best_c:
        best_c = hap[1]
        best_h = hap[0].copy()
    ha2 = to_binary(best_h, n)
    ha2, ca2 = binary_anneal(ha2, n, p4, seed=seed + 13)
    if ca2 < best_c:
        best_c = ca2
        best_h = ha2.copy()

    # Phase 5: Continuous PGD refinement (20%)
    p5 = t0 + 0.80 * total
    hr = best_h.copy()
    hr, cr = pgd_refine(hr, n, p5, max_iter=6000)
    if cr < best_c:
        best_c = cr
        best_h = hr.copy()

    # Phase 6: Try n=480 (10%)
    p6 = t0 + 0.90 * total
    if time.time() < p6 - 30:
        nh = 480
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project(hh, nh)
        hhb = to_binary(hh, nh)
        hhb, chb = binary_ls(hhb, nh, t0 + 0.87 * total, max_sweeps=100)
        hha = binary_anneal(hhb, nh, t0 + 0.89 * total, seed=seed + 23)
        hh_best = hha[0].copy() if hha[1] < chb else hhb.copy()
        hh_best, hh_c = pgd_refine(hh_best, nh, p6 - 2, alphas=[50, 300, 2000, 15000, 80000], max_iter=3000)
        if hh_c < best_c:
            best_c = hh_c
            best_h = hh_best.copy()
            n = nh

    # Phase 7: Final polish (last 10%)
    p7 = dl - 1
    if time.time() < p7 - 5:
        hb4 = to_binary(best_h, n)
        hb4, cb4 = binary_ls(hb4, n, p7 - 3, max_sweeps=80)
        if cb4 < best_c:
            best_c = cb4
            best_h = hb4.copy()
        hp = project(best_h, n)
        cp = compute_c5(hp, n)
        if cp < best_c:
            best_c = cp
            best_h = hp

    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
