def gradient_lag(h, n, lag_idx):
    """Compute gradient of the overlap at correlation index lag_idx w.r.t. h."""
    shift = lag_idx - (n - 1)
    grad = np.zeros(n)
    m = abs(shift)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        # overlap = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        grad[m:] += (1.0 - h[:n-m])  # d/dh[j] of h[j]*... for j >= m
        grad[:n-m] -= h[m:]  # d/dh[j] of ...(1-h[j]) for j < n-m
    else:
        # shift < 0, m = -shift
        # overlap = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        grad[:n-m] += (1.0 - h[m:])  # d/dh[j] of h[j]*... for j < n-m
        grad[m:] -= h[:n-m]  # d/dh[j] of ...(1-h[j]) for j >= m
    return grad

def grad_active_lags(h, n, top_idx, c):
    """Weighted gradient of the top lags, weighted by their softmax values."""
    vals = c[top_idx]
    m = vals.max()
    weights = np.exp(vals - m)
    weights /= weights.sum()
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        grad += weights[i] * gradient_lag(h, n, idx)
    return grad * (2.0 / n)

                h_curr[j] -= delta
            c_new = compute_c5(h_curr, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_curr.copy()
                improved = True
            else:
                if h_curr[i] > h_curr[j] - 2 * delta:
                    pass
                if h_curr[i] + delta > 1.0 or h_curr[j] - delta < -delta:
                    pass
                # undo
                if h_curr[i] < h_curr[j]:
                    h_curr[i] -= delta
                    h_curr[j] += delta
                else:
                    h_curr[i] += delta
                    h_curr[j] -= delta

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_h, best_c

def binary_push(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    for thresh in [0.005, 0.02, 0.05]:
        if time.time() > deadline:
            break
        mask = (h_curr > 0) & (h_curr < thresh)
        for i in np.where(mask)[0]:
            if time.time() > deadline:
                break
            h_try = h_curr.copy()
            h_try[i] = 0.0
            h_try = project_feasible(h_try, n)
            ct = compute_c5(h_try, n)
            if ct < best_c - 1e-12:
                best_c = ct
                best_h = h_try.copy()
                h_curr = h_try
        mask = (h_curr > 1 - thresh) & (h_curr < 1)
        for i in np.where(mask)[0]:
            if time.time() > deadline:
                break
            h_try = h_curr.copy()
            h_try[i] = 1.0
            h_try = project_feasible(h_try, n)
            ct = compute_c5(h_try, n)
            if ct < best_c - 1e-12:
                best_c = ct
                best_h = h_try.copy()
                h_curr = h_try
    return best_h, best_c

def outer_lp_step(h_curr, n, trust, time_limit=8):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(h_var - h_curr, 2) <= trust]
    for k in range(2 * n - 1):
        grad = gradient_lag(h_curr, n, k)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, time_limit=time_limit)
        except Exception:
            return h_curr.copy()
    if prob.status in ['optimal', 'optimal_inaccurate']:
        h_new = h_var.value
        if h_new is not None:
            return project_feasible(h_new, n)
    return h_curr.copy()

def optimize_resolution(h_init, n, deadline):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Moderate alpha, all lags
    if time.time() < deadline - 60:
        h_new = solve_slqsp(h, n, 800, maxiter=3000, ftol=1e-13)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h, h = c_new, h_new, h_new

    # Phase 2: Higher alpha, all lags
    if time.time() < deadline - 50:
        h_new = solve_slqsp(h, n, 5000, maxiter=5000, ftol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h, h = c_new, h_new, h_new

    # Phase 3: Top-lag focus
    c = compute_all_lags(h, n)
    top15 = np.argsort(c)[-15:]
    if time.time() < deadline - 40:
        h_new = solve_slqsp(h, n, 30000, top_idx=top15, maxiter=6000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h, h = c_new, h_new, h_new

    # Phase 4: Very high alpha, top 8
    c = compute_all_lags(h, n)
    top8 = np.argsort(c)[-8:]
    if time.time() < deadline - 35:
        h_new = solve_slqsp(h, n, 200000, top_idx=top8, maxiter=8000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h, h = c_new, h_new, h_new

    # Phase 5: Ultra high alpha, top 4
    c = compute_all_lags(h, n)
    top4 = np.argsort(c)[-4:]
    if time.time() < deadline - 30:
        h_new = solve_slqsp(h, n, 1000000, top_idx=top4, maxiter=10000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h, h = c_new, h_new, h_new

    # Phase 6: Gradient-based local refinement
    if time.time() < deadline - 20:
        h_new, c_new = fast_gradient_refine(h, n, deadline - 10)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 7: Swap local search
    if time.time() < deadline - 10:
        h_new, c_new = swap_refine(best_h, n, deadline - 3, max_rounds=40)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 8: Binary push
    if time.time() < deadline - 3:
        h_new, c_new = binary_push(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')

    # Load parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    # Main optimization at parent resolution
    if h_init is not None:
        alloc = min(500, deadline - t0 - 100)
        h_opt, c_opt = optimize_resolution(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt

    # Try a slightly different resolution
    if best_h is not None and time.time() < deadline - 100:
        n2 = len(best_h) + 25
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n2)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n2)
        alloc2 = min(80, deadline - time.time() - 50)
        if alloc2 > 40:
            h2, c2 = optimize_resolution(h_interp, n2, time.time() + alloc2)
            if c2 < best_c:
                best_c, best_h = c2, h2

    # Another try at higher resolution
    if best_h is not None and time.time() < deadline - 80:
        n3 = 400
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n3)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n3)
        alloc3 = min(70, deadline - time.time() - 40)
        if alloc3 > 35:
            h3, c3 = optimize_resolution(h_interp, n3, time.time() + alloc3)
            if c3 < best_c:
                best_c, best_h = c3, h3

    # Perturbation restarts on best
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(1, int(remaining / 40))
        for r in range(min(n_restarts, 4)):
            if time.time() > deadline - 30:
                break
            scale = 0.008 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(35, (deadline - time.time() - 20) * 0.5)
            if alloc < 20:
                break
            h_r, c_r = optimize_resolution(hp, n, time.time() + alloc)
            if c_r < best_c:
                best_c, best_h = c_r, h_r

    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        c = compute_all_lags(best_h, n)
        top2 = np.argsort(c)[-2:]
        h_new = solve_slqsp(best_h, n, 2000000, top_idx=top2, maxiter=12000, ftol=1e-18)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        h_new, c_new = fast_gradient_refine(best_h, n, deadline - 1, max_iters=50000)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)

    return best_h, best_c, len(best_h)
