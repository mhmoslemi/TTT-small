import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_set_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_set_grad(x, n, active_idx, alpha, mu):

    vals = c[active_idx]
    m = vals.max()
    sh = alpha * (vals - m)

def sq_constraint(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def lbfgs_round(h, n, alpha, mu, deadline, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-12, 'maxfun': maxiter*10, 'maxls': 50,
                               'disp': False})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

def projected_gradient_step(h, n, grad, step_size):
    """Take a projected gradient step with line search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
        s = step_size * factor
        h_trial = h - s * grad
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c:
            best_h = h_trial
            best_c = c_trial
            break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_constraint, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, mm in zip(w, idx):
        if wi < 1e-13:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def interp_to_n(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def run_lbfgs_alpha(h, n, alpha, mu, deadline, maxiter, bounds):
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16,
                               'gtol': 1e-12, 'maxfun': maxiter * 12,
                               'maxls': 40, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run_multi_alpha(h, n, deadline, alphas, mu, bounds, maxiter):
    h_work = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h_new = run_lbfgs_alpha(h_work, n, alpha, mu, deadline, maxiter, bounds)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_work = h_new
    return best_h, best_c

def run_active_set(h, n, deadline, bounds, top_frac=0.005, min_active=3, max_rounds=8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, top_frac * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) < min_active:
            active = np.argsort(-c_vec)[:min_active]
        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxfun': 20000, 'maxls': 30,
                                       'disp': False})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    return best_h, best_c

def run_slqp(h, n, deadline, alpha, bounds, maxiter=1500):
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run_perturb(h, n, deadline, bounds, seed, n_pert=6, strength=0.015):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for i in range(n_pert):
        if time.time() > deadline - 20:
            break
        h_pert = h + rng.uniform(-strength, strength, size=n)
        h_pert = project_feasible(h_pert, n)
        h_new = run_lbfgs_alpha(h_pert, n, 20000, 0.005, deadline - 15, 1500, bounds)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def coord_descent(h, n, deadline, radius=6, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] - d >= 0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if best_h[j] - d >= 0 and best_h[i] + d <= 1.0:
                        best_h[j] -= d
                        best_h[i] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Multi-alpha L-BFGS at original resolution
    bounds0 = [(0.0, 1.0)] * n0
    alphas1 = [30, 200, 1000, 5000, 20000, 100000, 500000]
    t1_end = min(time.time() + 90, deadline - 350)
    if t1_end > time.time() + 10:
        best_h, best_c = run_multi_alpha(best_h, best_n, t1_end, alphas1, 0.003, bounds0, 2000)
        best_n = n0

    # Phase 2: Optimize at n=800
    n2 = 800
    t2_end = min(time.time() + 160, deadline - 200)
    if t2_end > time.time() + 15:
        h2 = interp_to_n(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        bounds2 = [(0.0, 1.0)] * n2
        alphas2 = [50, 500, 3000, 15000, 80000, 400000, 2000000]
        h2, c2 = run_multi_alpha(h2, n2, t2_end, alphas2, 0.003, bounds2, 1500)
        c2 = compute_c5(h2, n2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = n2

    # Phase 3: Try n=1000
    n3 = 1000
    t3_end = min(time.time() + 120, deadline - 120)
    if t3_end > time.time() + 15:
        h3 = interp_to_n(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        bounds3 = [(0.0, 1.0)] * n3
        alphas3 = [50, 500, 3000, 15000, 80000, 400000]
        h3, c3 = run_multi_alpha(h3, n3, t3_end, alphas3, 0.003, bounds3, 1200)
        c3 = compute_c5(h3, n3)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
            best_n = n3

    # Phase 4: Multi-resolution cycling (coarse refine then back)
    t4_end = min(time.time() + 80, deadline - 80)
    if t4_end > time.time() + 15:
        n_coarse = max(200, best_n // 2)
        h_co = interp_to_n(best_h, best_n, n_coarse)
        h_co = project_feasible(h_co, n_coarse)
        bounds_co = [(0.0, 1.0)] * n_coarse
        h_co, c_co = run_multi_alpha(h_co, n_coarse, t4_end, [200, 2000, 10000, 50000], 0.005, bounds_co, 1500)
        c_co = compute_c5(h_co, n_coarse)
        h_up = interp_to_n(h_co, n_coarse, best_n)
        h_up = project_feasible(h_up, best_n)
        c_up = compute_c5(h_up, best_n)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up.copy()

    # Phase 5: Active-set refinement
    n5 = best_n
    bounds5 = [(0.0, 1.0)] * n5
    t5_end = min(time.time() + 70, deadline - 50)
    if t5_end > time.time() + 10:
        best_h, best_c = run_active_set(best_h, n5, t5_end, bounds5,
                                         top_frac=0.004, min_active=3, max_rounds=6)

    # Phase 6: SLSQP polish
    t6_end = min(time.time() + 40, deadline - 30)
    if t6_end > time.time() + 10:
        n6 = best_n
        bounds6 = [(0.0, 1.0)] * n6
        for a in [200000, 1000000]:
            if time.time() > t6_end - 5:
                break
            h6 = run_slqp(best_h, n6, t6_end, a, bounds6, maxiter=1500)
            c6 = compute_c5(h6, n6)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Perturbation restart
    t7_end = min(time.time() + 40, deadline - 15)
    if t7_end > time.time() + 10:
        n7 = best_n
        bounds7 = [(0.0, 1.0)] * n7
        best_h, best_c = run_perturb(best_h, n7, t7_end, bounds7, seed, n_pert=5, strength=0.01)

    # Phase 8: Final active-set pass
    t8_end = deadline - 3
    if t8_end > time.time() + 8:
        n8 = best_n
        bounds8 = [(0.0, 1.0)] * n8
        best_h, best_c = run_active_set(best_h, n8, t8_end, bounds8,
                                         top_frac=0.002, min_active=2, max_rounds=4)
        # Quick SLSQP
        if time.time() < deadline - 2:
            h8 = run_slqp(best_h, n8, deadline - 1, 1000000, bounds8, maxiter=800)
            c8 = compute_c5(h8, n8)
            if c8 < best_c:
                best_c = c8
                best_h = h8.copy()
                best_n = n8

    # Coordinate descent for final micro-optimization
    if deadline - time.time() > 5:
        best_h, best_c = coord_descent(best_h, best_n, deadline - 1, radius=5,
                                        deltas=[0.002, 0.001, 0.0005, 0.0002])

    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
