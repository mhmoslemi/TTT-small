def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_box_sum(h, target_sum):
    """Efficient projection onto {x: 0<=x<=1, sum(x)=target_sum} via bisection."""
    n = len(h)
    
    def f(lam):
        return np.clip(h - lam, 0.0, 1.0).sum() - target_sum
    
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    if f(lo) < 0:
        return np.clip(h, 0.0, 1.0)
    if f(hi) > 0:
        return np.clip(h, 0.0, 1.0)
    
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def _compute_lag_grad(h, n, m):

def smooth_gradient(h, n, alpha):

    x = h_curr.copy()  # momentum variable

        step = 0.01

            obj_trial = smooth_obj(h_trial, n, alpha)

        h_oo, c_oo = outer_approx_adaptive(best_h, n, oo_end, max_iters=100)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    
    # Phase C: SLSQP with smooth objective
    if time.time() < deadline - 30:
        for alpha in [5000, 50000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase D: Active set SLSQP
    if time.time() < deadline - 25:
        as_end = time.time() + min(40, (deadline - time.time()) * 0.4)
        h_as, c_as = active_set_slsqp(best_h, n, as_end, top_m=10, max_rounds=4)
        if c_as < best_c:
            best_c = c_as
            best_h = h_as
    
    # Phase E: More gradient descent with higher alpha
    if time.time() < deadline - 20:
        gd_end = time.time() + min(30, (deadline - time.time()) * 0.3)
        h_gd2, c_gd2 = gradient_descent(best_h, n, gd_end, alpha=200.0, max_iter=30000)
        if c_gd2 < best_c:
            best_c = c_gd2
            best_h = h_gd2
    
    # Phase F: Local search
    if time.time() < deadline - 5:
        ls_end = min(deadline - 2, time.time() + 25)
        best_h, best_c = local_search(best_h, n, ls_end)
    
    best_h = project_box_sum(best_h, n / 2.0)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize at initial resolution
    if h_init is not None:
        alloc1 = min(250, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt
    
    # Phase 2: Try other resolutions
    n_candidates = [200, 400, 500, 600]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.3)
        
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        
        h_start = project_box_sum(h_start, n_try / 2.0)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Multi-start perturbation at best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            # Different perturbation strategies
            if r == 0:
                hp = best_h + rng.normal(0, 0.01, n)
            elif r == 1:
                # Structured perturbation (Fourier mode)
                freq = rng.randint(3, n // 4)
                phase = rng.uniform(0, 2 * np.pi)
                pert = 0.01 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                hp = best_h + pert
            elif r == 2:
                # Block perturbation
                block = n // 5
                hp = best_h.copy()
                block_idx = rng.randint(0, n - block)
                hp[block_idx:block_idx + block] += rng.uniform(-0.03, 0.03)
            else:
                scale = 0.005 + 0.005 * r
                hp = best_h + rng.normal(0, scale, n)
            
            hp = project_box_sum(hp, n / 2.0)
            alloc = min(50, (deadline - time.time() - 30) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # High-alpha SLSQP
        if time.time() < deadline - 20:
            h_new = slsqp_optimize(best_h, n, 100000, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Final local search
        if time.time() < deadline - 3:
            best_h, best_c = local_search(best_h, n, deadline - 1)
        
        best_h = project_box_sum(best_h, n / 2.0)
        best_c = compute_c5(best_h, n)
    
    # Fallback
    if best_h is None:
        best_h = project_box_sum(np.full(300, 0.5), 150.0)
        best_c = compute_c5(best_h, 300)
    
    return best_h, best_c, len(best_h)
