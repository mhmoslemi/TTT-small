def compute_c5(h, n):

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    
    initial_time = time.time()
    final_time = deadline
    total_time = final_time - initial_time
    
    for it in range(100000):
        if time.time() > deadline:
            break
        progress = (time.time() - initial_time) / max(total_time, 1)
        temperature = t_start * (t_end / t_start) ** progress
        
        # Random move: pick two positions and swap part of their values
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 2 * temperature
        if h[i] + delta < 0 or h[i] + delta > 1:
            continue
        if h[j] - delta < 0 or h[j] - delta > 1:
            continue
        h[i] += delta
        h[j] -= delta
        c_new = compute_c5(h, n)
        
        if c_new < current_c or rng.random() < np.exp(-(c_new - current_c) / max(temperature, 1e-10)):
            current_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    
    return best_h, best_c
