        return float(m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha)

        h_new = slsqp_opt(h, n, top_k, 50000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    
    # Phase C: Fine local search
    if time.time() < deadline - 10:
        dl = min(deadline - 3, time.time() + (deadline - time.time()) * 0.5)
        h_new, c_new = fine_local_search(best_h, n, dl)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Optimize at parent resolution (250s)
    if time.time() < deadline - 800:
        alloc = min(250, (deadline - t0) * 0.3)
        h1, c1 = optimize_at_resolution(h_init, n_init, t0 + alloc, rng)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Binary projection and binary local search (150s)
    if time.time() < deadline - 500:
        n_cur = len(best_h)
        h_bin = np.zeros(n_cur)
        idx = np.argsort(best_h)[::-1][:n_cur // 2]
        h_bin[idx] = 1.0
        c_bin = compute_c5(h_bin, n_cur)
        
        if time.time() < deadline - 400:
            bl_dl = min(deadline - 400, time.time() + 120)
            h_b, c_b = binary_local_search(h_bin, n_cur, bl_dl, rng)
            if c_b < best_c:
                best_c = c_b
                best_h = h_b.copy()
            elif c_bin < best_c:
                best_c = c_bin
                best_h = h_bin.copy()
        
        # Also try: use binary as starting point for continuous optimization
        if c_bin < best_c + 0.01 and time.time() < deadline - 350:
            h_from_bin = h_bin.astype(float)
            # Perturb slightly
            h_pert = h_from_bin + rng.normal(0, 0.01, n_cur)
            h_pert = project_feasible(h_pert, n_cur)
            alloc = min(60, deadline - time.time() - 300)
            if alloc > 20:
                h_co, c_co = optimize_at_resolution(h_pert, n_cur, time.time() + alloc, rng)
                if c_co < best_c:
                    best_c = c_co
                    best_h = h_co.copy()
    
    # Phase 3: Try higher resolutions (300s)
    n_best = len(best_h)
    for n_try in [600, 700, 550, 650]:
        if time.time() > deadline - 200:
            break
        remaining = deadline - time.time() - 50
        if remaining < 80:
            break
        alloc = min(100, remaining * 0.3)
        if alloc < 40:
            break
        
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        h_r, c_r = optimize_at_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
            n_best = n_try
    
    # Phase 4: Multi-start at best resolution (150s)
    n_final = len(best_h)
    if time.time() < deadline - 100:
        n_restarts = 3
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_final)
            hp = project_feasible(hp, n_final)
            alloc = min(45, (deadline - time.time() - 50) * 0.25)
            if alloc < 20:
                break
            h_r, c_r = optimize_at_resolution(hp, n_final, time.time() + alloc, rng)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r.copy()
    
    # Phase 5: Final ultra-fine polish (remaining time)
    if best_h is not None and time.time() < deadline - 5:
        n_fin = len(best_h)
        # SLSQP top-2
        if time.time() < deadline - 12:
            hf = slsqp_opt(best_h, n_fin, 2, 100000, maxiter=4000)
            cf = compute_c5(hf, n_fin)
            if cf < best_c:
                best_c = cf
                best_h = hf
        # SLSQP top-1
        if time.time() < deadline - 10:
            hf2 = slsqp_opt(best_h, n_fin, 1, 200000, maxiter=3000)
            cf2 = compute_c5(hf2, n_fin)
            if cf2 < best_c:
                best_c = cf2
                best_h = hf2
        # Final local search
        if time.time() < deadline - 2:
            hl, cl = fine_local_search(best_h, n_fin, deadline - 1)
            if cl < best_c:
                best_c = cl
                best_h = hl
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
