import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, L):
    """Compute gradient of c_L w.r.t. h for lag L (0 ≤ L ≤ n-1)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if L == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    else:
        # ∂c_L/∂h[j] = dx*(1-h[j-L]) for j in [L, n-1]
        #              - dx*h[j+L] for j in [0, n-1-L]
        grad[L:] += dx * (1.0 - h[:n-L])
        grad[:n-L] -= dx * h[L:]
    return grad

def smooth_max_obj(x, n, alpha, top_indices=None):
    """Smooth max of top lags (or all lags)."""
    c_all = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    if top_indices is not None:
        c = c_all[top_indices]
    else:
        c = c_all
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-15, top_indices=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        if top_indices is not None:
            res = minimize(lambda x: smooth_max_obj(x, n, alpha, top_indices), h,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        else:
            res = minimize(lambda x: smooth_max_obj(x, n, alpha), h,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, L):
    dx = 2.0 / n
    grad = np.zeros(n)
    if L == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    else:
        grad[L:n] += dx * (1.0 - h[0:n-L])
        grad[0:n-L] -= dx * h[L:n]
    return grad

def obj_smooth_max(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def obj_smooth_max_top(x, n, alpha, top_idx):
    c_all = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    c = c_all[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def eq_con(x, n):
    return np.sum(x) - n / 2.0

def eq_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=3000, ftol=1e-15, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac, 'args': (n,)}]
    try:
        if top_idx is None:
            res = minimize(obj_smooth_max, h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        else:
            res = minimize(obj_smooth_max_top, h, args=(n, alpha, top_idx), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_refine(h, n, deadline, iters_per_chunk=200):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    n_lags = 2 * n - 1
    lags_arr = np.arange(-n+1, n)
    
    for phase in range(20):
        if time.time() > deadline:
            break
        alpha_sg = 50 + 20 * phase  # increasing sharpness
        m_top = min(3 + phase // 3, 10)
        lr = 0.01 * (0.95 ** phase)
        
        for it in range(iters_per_chunk):
            if time.time() > deadline:
                break
            c = compute_all_lags(h_curr, n)
            sorted_idx = np.argsort(c)[::-1]
            top_idx = sorted_idx[:m_top]

def coordinate_descent(h, n, deadline):
    """Aggressive coordinate descent focusing on the worst lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    delta_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in delta_sizes:

            break
        improved = True
        stall_count = 0
        while improved and time.time() < deadline:
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            L = abs(k_star - (n - 1))
            
            if L == 0:

                # Move mass from mid-values to extremes
                h_sorted = np.argsort(h)

                        h[j] += delta

                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 2*d:
                        delta = min(d, (h[j]-h[i])*0.3)

                        c_new = compute_c5(h, n)

                            best_c = c_new

                        else:
                            h[i] -= delta
                            h[j] += delta

                    stall_count += 1
                    if stall_count > 3:
                        break
                continue
            
            # For L > 0: c_L = dx * sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
            # To decrease: increase h[i] where h[i+L] is high, or decrease h[i] where h[i+L] is low

            pos = np.where(grad > 0)[0]
            neg = np.where(grad < 0)[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, L):
    dx = 2.0 / n
    grad = np.zeros(n)
    if L == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    else:
        grad[L:n] += dx * (1.0 - h[0:n-L])
        grad[0:n-L] -= dx * h[L:n]
    return grad

def obj_smooth_max(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_smooth_max_top(x, n, alpha, top_idx):
    c_all = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    c = c_all[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def eq_con(x, n):
    return np.sum(x) - n / 2.0

def eq_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=3000, ftol=1e-15, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac, 'args': (n,)}]
    try:
        if top_idx is None:
            res = minimize(obj_smooth_max, h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        else:
            res = minimize(obj_smooth_max_top, h, args=(n, alpha, top_idx), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_gradient_refine(h, n, deadline, n_phases=8):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        alpha_sg = 30 + 15 * phase
        m_top = min(3 + phase // 2, 8)
        lr = 0.02 * (0.92 ** phase)
        n_iters = 100
        for it in range(n_iters):
            if time.time() > deadline:
                break
            c = compute_all_lags(h_curr, n)
            sorted_idx = np.argsort(c)[::-1]
            top_idx = sorted_idx[:m_top]
            w_raw = np.exp(alpha_sg * (c[top_idx] - c[top_idx[0]]))
            w = w_raw / w_raw.sum()
            grad = np.zeros(n)
            for wi, idx in zip(w, top_idx):
                L = abs(int(idx - (n - 1)))
                grad += wi * lag_gradient(h_curr, n, L)
            grad -= grad.mean()
            norm_g = np.sqrt(np.sum(grad**2))
            if norm_g < 1e-16:
                break
            h_new = h_curr - lr * grad / np.sqrt(norm_g) * 0.05
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
            h_curr = h_new
    return best_h, best_c

def fast_coordinate_descent(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    delta_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    rng_local = np.random.RandomState(12345)
    for d in delta_sizes:
        if time.time() > deadline:
            break
        stall = 0
        max_stall = 8
        while stall < max_stall and time.time() < deadline:
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            L = abs(k_star - (n - 1))
            if L == 0:
                h_idx = np.argsort(h)
                for attempt in range(40):
                    i = h_idx[attempt % n]
                    j = h_idx[n - 1 - attempt % n]
                    if h[j] - h[i] < 2 * d + 1e-10:
                        break
                    delta = min(d, (h[j] - h[i]) * 0.25)
                    h[j] -= delta
                    h[i] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[j] += delta
                        h[i] -= delta
                if not improved:
                    stall += 1
                else:
                    stall = 0
                continue
            grad = lag_gradient(h, n, L)
            grad -= grad.mean()
            pos_mask = grad > 1e-10
            neg_mask = grad < -1e-10
            if not np.any(pos_mask) or not np.any(neg_mask):
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            for attempt in range(40):
                ip = pos_idx[rng_local.randint(len(pos_idx))]
                ineg = neg_idx[rng_local.randint(len(neg_idx))]
                delta = min(d, h[ip] * 0.9, (1.0 - h[ineg]) * 0.9)
                if delta < 1e-10:
                    continue
                h[ip] -= delta
                h[ineg] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h[ip] += delta
                    h[ineg] -= delta
            if not improved:
                stall += 1
            else:
                stall = 0
    return h, best_c

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha
    alphas_p1 = [5000, 20000, 50000]
    for a in alphas_p1:
        if time.time() > deadline - 40:
            break
        h_new = slsqp_run(best_h, n, a, maxiter=2500, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 2: Active set SLSQP
    if time.time() < deadline - 35:
        for m in [3, 5, 7, 10]:
            if time.time() > deadline - 35:
                break
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-m:]
            for a in [100000, 300000, 1000000]:
                if time.time() > deadline - 35:
                    break
                h_new = slsqp_run(best_h, n, a, maxiter=2000, ftol=1e-15, top_idx=top_idx)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                c = compute_all_lags(best_h, n)
                top_idx = np.argsort(c)[-m:]

    # Phase 3: Projected gradient
    if time.time() < deadline - 25:
        h_pg, c_pg = projected_gradient_refine(best_h, n, min(deadline - 25, time.time() + 50))
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg

    # Phase 4: Coordinate descent
    if time.time() < deadline - 5:
        h_cd, c_cd = fast_coordinate_descent(best_h, n, deadline - 2)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

    return best_h, best_c, len(best_h)
