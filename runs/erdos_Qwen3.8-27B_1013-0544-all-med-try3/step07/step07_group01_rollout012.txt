                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def perturbation_restart(h, n, deadline, rng, n_perturb=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(n_perturb):
        if time.time() > deadline - 60:
            break
        noise_scale = 0.02 / (p + 1)
        h_pert = h + rng.normal(0, noise_scale, size=n)
        h_pert = project_feasible(h_pert, n)
        h_pert, c_pert = lbfgs_phase(h_pert, n, time.time() + min(40, deadline - time.time() - 30), max_configs=4)
        if c_pert < best_c - 1e-10:
            best_c = c_pert
            best_h = h_pert.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    n_best = n_init

    # Phase 1: L-BFGS-B optimization (0-120s)
    t1 = min(t0 + 120, deadline - 600)
    best_h, best_c = lbfgs_phase(best_h, n_best, t1, max_configs=8)

    # Phase 2: SLSQP (120-200s)
    if time.time() < deadline - 500:
        t2 = min(time.time() + 80, deadline - 500)
        for a in [100000, 500000, 2000000]:
            if time.time() > t2:
                break
            h2, c2 = slqp_phase(best_h, n_best, min(time.time() + 30, t2), alpha=a, maxiter=1000)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Gradient-guided pair search, coarse (200-350s)
    if time.time() < deadline - 400:
        t3 = min(time.time() + 150, deadline - 400)
        best_h, best_c = grad_guided_pair_search(
            best_h, n_best, t3,
            deltas=[0.01, 0.005, 0.002, 0.001], frac=0.5)

    # Phase 4: Fine gradient-guided pair search (350-550s)
    if time.time() < deadline - 300:
        t4 = min(time.time() + 200, deadline - 300)
        best_h, best_c = grad_guided_pair_search(
            best_h, n_best, t4,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005], frac=0.4)

    # Phase 5: Very fine pair search (550-700s)
    if time.time() < deadline - 200:
        t5 = min(time.time() + 150, deadline - 200)
        best_h, best_c = full_pair_search(
            best_h, n_best, t5,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001], radius=80)

    # Phase 6: Perturbation restarts (700-800s)
    if time.time() < deadline - 150:
        t6 = min(time.time() + 100, deadline - 150)
        h6, c6 = perturbation_restart(best_h, n_best, t6, rng, n_perturb=2)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try higher resolution (800-950s)
    if time.time() < deadline - 120:
        for n_try in [400, 500, 600]:
            if time.time() > deadline - 80:
                break
            x_old = np.linspace(0, 2, n_best)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 40)
            if alloc < 20:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5, max_configs=4)
            if time.time() < deadline - 25:
                h_t, c_t = slqp_phase(h_t, n_try, time.time() + min(20, deadline - time.time() - 15),
                                      alpha=500000, maxiter=500)
            if time.time() < deadline - 15:
                h_t, c_t = grad_guided_pair_search(
                    h_t, n_try, min(time.time() + 20, deadline - 10),
                    deltas=[0.01, 0.005, 0.001, 0.0005], frac=0.5)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n_best = n_try

    # Phase 8: Final L-BFGS + SLSQP (950-1050s)
    if time.time() < deadline - 50:
        n_final = len(best_h)
        best_h, best_c = lbfgs_phase(best_h, n_final, min(time.time() + 50, deadline - 30), max_configs=4)
        if time.time() < deadline - 15:
            h8, c8 = slqp_phase(best_h, n_final, min(time.time() + 30, deadline - 10),
                                alpha=2000000, maxiter=2000)
            if c8 < best_c:
                best_c = c8
                best_h = h8

    # Phase 9: Final ultra-fine pair search (1050-1095s)
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = full_pair_search(
            best_h, nf, deadline - 2,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005], radius=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
