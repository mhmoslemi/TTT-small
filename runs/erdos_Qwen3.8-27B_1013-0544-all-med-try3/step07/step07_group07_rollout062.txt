def swap_delta(h, n, i, j, m_range):
    """Vectorized delta for swapping i(0->1) and j(1->0)."""
    delta = np.zeros(2*n-1)
    # i goes 0->1
    im = i + m_range
    msk = (im >= 0) & (im < n)

    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]

    jm = j + m_range

    delta[msk] -= 1.0 - h[jm[msk]]

    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2*n-1:

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def local_search_binary(h, n, deadline, max_sweeps=200):
    """Greedy best-improvement local search for binary h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n-1), n)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        cur_max = corr.max()
        for i in zeros_list[:]:
            if time.time() > deadline + 2:  # small grace
                break
            # For this i, find best j to swap with
            best_j = -1
            best_new_max = cur_max
            for j in ones_list:
                # Compute delta
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                new_max = (corr + delta).max()
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_j = j
            if best_j >= 0:
                # Apply swap
                h[i], h[best_j] = h[best_j], h[i]
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c
