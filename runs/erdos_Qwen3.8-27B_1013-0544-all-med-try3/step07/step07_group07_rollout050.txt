def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def binary_swap_search(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_h = h.copy()
    best_c = cur_c
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    m_range = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            im = i + m_range
            msk = (im >= 0) & (im < n)
            imn = i - m_range
            msk2 = (imn >= 0) & (imn < n)
            
            for j in list(ones):
                delta = np.zeros(2*n-1)
                delta[msk] += 1.0 - h[im[msk]]
                delta[msk2] -= h[imn[msk2]]
                jm = j + m_range
                msk3 = (jm >= 0) & (jm < n)
                jm2 = j - m_range
                msk4 = (jm2 >= 0) & (jm2 < n)
                delta[msk3] -= 1.0 - h[jm[msk3]]
                delta[msk4] += h[jm2[msk4]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max()
                if nc < best_nc - 1e-12:
                    best_nc = nc
                    best_j = j
            
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                # Recompute corr (safer than incremental for few moves)
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur_c = corr.max()
                zeros.discard(i)
                ones.discard(best_j)
                zeros.add(best_j)
                ones.add(i)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c * dx

def anneal_binary(h, n, deadline, seed=42, max_iters=1000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    m_range = np.arange(-(n-1), n)
    
    T = 0.05
    cool = 0.99998
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 50000 == 0 and it > 0:
            T = max(T, 0.01)  # Restart temperature if stuck
        
        ai = rng.randint(n_zeros)
        bi = rng.randint(n_ones)
        i = zeros_list[ai]
        j = ones_list[bi]
        
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk2 = (imn >= 0) & (imn < n)
        delta[msk2] -= h[imn[msk2]]
        jm = j + m_range
        msk3 = (jm >= 0) & (jm < n)
        delta[msk3] -= 1.0 - h[jm[msk3]]
        jm2 = j - m_range
        msk4 = (jm2 >= 0) & (jm2 < n)
        delta[msk4] += h[jm2[msk4]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n-1 + (i-j)
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        
        nc = (corr + delta).max()
        delta_c = nc - cur_c
        if delta_c <= 0 or (T > 1e-10 and rng.random() < np.exp(-delta_c * 500.0 / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr += delta
            cur_c = nc
            zeros_list[ai] = j
            ones_list[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    
    return best_h, best_c * dx
