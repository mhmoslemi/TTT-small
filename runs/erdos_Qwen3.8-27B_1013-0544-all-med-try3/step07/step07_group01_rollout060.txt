def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

    nn = len(c)

        k = m_idx - (n - 1)
        wm = w[m_idx] * dx

                grad[k:n] += wm * (-h[0:nv])

    return g

    return float(x.sum() - n/2.0)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(len(c)):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _obj_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _cons_fun(x, n):
    return float(x.sum() - n / 2.0)

def _cons_jac(x, n):
    return np.ones(n)

def do_lbfgs_phase(h, n, deadline, alpha_list=None, mu_list=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if alpha_list is None:
        alpha_list = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    if mu_list is None:
        mu_list = [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 10000.0, 50000.0]
    for idx in range(len(alpha_list)):
        if time.time() > deadline - 3:
            break
        a = alpha_list[idx]
        mu = mu_list[idx]
        t_alloc = min(35, deadline - time.time() - 2)
        if t_alloc < 4:
            break
        try:
            res = minimize(_obj, h, jac=_obj_grad, args=(n, a, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': 2000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def do_slsqp(h, n, deadline, alpha=500000, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(lambda z: smooth_max_val(compute_c5_vec(z, n), alpha), h,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), best_c

def subgrad_descent(h, n, deadline, max_iter=10000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.005
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        k_max = int(np.argmax(c_vec))
        k = k_max - (n - 1)
        dx = 2.0 / n
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[0:nv] += (1.0 - h[k:n]) * dx
                g[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:n] += (1.0 - h[0:nv]) * dx
                g[0:nv] += (-h[kk:n]) * dx
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            h = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.8
            if lr < 1e-9:
                break
    return best_h, best_c

def targeted_local_search(h, n, deadline):
    """Local search targeting the currently-maximizing shifts."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        passes = 0
        while time.time() < deadline:
            passes += 1
            improved = False
            c_vec = compute_c5_vec(h, n)
            top_k = np.argsort(c_vec)[-3:]
            for m_idx in top_k:
                k = m_idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    for i in range(nv):
                        if time.time() > deadline:
                            return best_h, best_c
                        j = i + k
                        for (a, b, sa, sb) in [(i, j, 1, -1), (j, i, -1, 1),
                                                 (i, j, -1, 1), (j, i, 1, -1)]:
                            if a == b:
                                continue
                            if h[a] + sa * d >= 0 and h[a] + sa * d <= 1 and \
                               h[b] + sb * d >= 0 and h[b] + sb * d <= 1:
                                h[a] += sa * d
                                h[b] += sb * d
                                c = compute_c5(h, n)
                                if c < best_c - 1e-14:
                                    best_c = c
                                    best_h = h.copy()
                                    improved = True
                                else:
                                    h[a] -= sa * d
                                    h[b] -= sb * d
                else:
                    kk = -k
                    nv = n - kk
                    for i in range(nv):
                        if time.time() > deadline:
                            return best_h, best_c
                        j = i + kk
                        for (a, b, sa, sb) in [(i, j, 1, -1), (j, i, -1, 1),
                                                 (i, j, -1, 1), (j, i, 1, -1)]:
                            if a == b:
                                continue
                            if h[a] + sa * d >= 0 and h[a] + sa * d <= 1 and \
                               h[b] + sb * d >= 0 and h[b] + sb * d <= 1:
                                h[a] += sa * d
                                h[b] += sb * d
                                c = compute_c5(h, n)
                                if c < best_c - 1e-14:
                                    best_c = c
                                    best_h = h.copy()
                                    improved = True
                                else:
                                    h[a] -= sa * d
                                    h[b] -= sb * d
            if not improved:
                break
    return best_h, best_c

def random_local_search(h, n, deadline, n_trials_per_pass=5000):
    """Random pair search - fast and broad."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for _ in range(n_trials_per_pass):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                sign = 1 if rng.rand() < 0.5 else -1
                a, b = i, j
                sa, sb = sign, -sign
                if h[a] + sa * d >= 0 and h[a] + sa * d <= 1 and \
                   h[b] + sb * d >= 0 and h[b] + sb * d <= 1:
                    h[a] += sa * d
                    h[b] += sb * d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[a] -= sa * d
                        h[b] -= sb * d
            if not improved:
                break
    return best_h, best_c

def fourier_optimize(n_eval, n_modes, deadline, seed=42):
    """Optimize h in Fourier basis: h(x) = 0.5 + sum(a_j cos(j*pi*x) + b_j sin(j*pi*x))"""
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n_eval, endpoint=False)
    n_params = 2 * n_modes
    coeffs = rng.randn(n_params) * 0.05
    
    def build_h(c):
        h = np.full(n_eval, 0.5)
        for j in range(n_modes):
            h += c[2*j] * np.cos((j+1) * np.pi * x)
            h += c[2*j+1] * np.sin((j+1) * np.pi * x)
        return h
    
    def fourier_obj(c):
        h = build_h(c)
        h = np.clip(h, 0, 1)
        n = n_eval
        target = n / 2.0
        diff = target - h.sum()
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-12:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-12:
                h += (diff / tot) * h
        h = np.clip(h, 0, 1)
        c_vec = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return smooth_max_val(c_vec, 5000) + 0.1 * (h.sum() - n/2.0)**2
    
    best_c = 1.0
    best_coeffs = coeffs.copy()
    
    for a_init in [0.05, 0.1, 0.2]:
        coeffs[:] = rng.randn(n_params) * a_init
        try:
            res = minimize(fourier_obj, coeffs, method='L-BFGS-B',
                          bounds=[(-0.5, 0.5)] * n_params,
                          options={'maxiter': 1000, 'ftol': 1e-14, 'maxls': 30})
            h = build_h(res.x)
            h = project_feasible(h, n_eval)
            c = compute_c5(h, n_eval)
            if c < best_c:
                best_c = c
                best_coeffs = res.x.copy()
        except Exception:
            pass
        if time.time() > deadline:
            break
    
    h = build_h(best_coeffs)
    h = project_feasible(h, n_eval)
    return h, compute_c5(h, n_eval)

            best_c, best_h = c8b, h8b
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
