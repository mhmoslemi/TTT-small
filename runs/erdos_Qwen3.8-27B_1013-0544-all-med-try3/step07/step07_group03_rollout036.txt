import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_rounds=100, tol=1e-10):
    """Coordinate descent exploiting the linearity of c_k in each h[i] for k!=0."""
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    current_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        old_c = current_c
        for i in range(n):
            # Compute coefficients a_k for the linear part
            # c_k(t) = c_k + (t - h[i]) * (1 - h[i+k] - h[i-k]) for k != 0

            best_t = hi

                h[i] = t
                # Compute new c values

                # But this requires knowing which k are valid...

                if c < best_val:
                    best_val = c
                    best_t = t
            h[i] = best_t

        h = project_feasible(h, n)

```python
def coord_descent_symmetric(h_half, m, max_rounds=300, tol=1e-12, deadline=None, grid_size=31):
    h_half = np.clip(h_half.copy(), 0, 1)
    n = 2 * m
    dx = 1.0 / m
    ts = np.linspace(0, 1, grid_size)
    
    def eval_half(hh):
        hf = np.concatenate([hh, 1.0 - hh])
        return float(np.max(np.correlate(hf, 1.0 - hf, mode='full'))) * dx
    
    current_c = eval_half(h_half)
    
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        old_c = current_c
        for i in range(m):
            hi = h_half[i]
            best_t = hi
            best_val = current_c
            for t in ts:
                if t == hi:
                    continue
                h_half[i] = t
                c = eval_half(h_half)
                if c < best_val:
                    best_val = c
                    best_t = t
            h_half[i] = best_t
            current_c = best_val
        
        h_half = np.clip(h_half, 0, 1)
        current_c = eval_half(h_half)
        if old_c - current_c < tol:
            break
    
    return np.concatenate([h_half, 1.0 - h_half]), current_c
```
