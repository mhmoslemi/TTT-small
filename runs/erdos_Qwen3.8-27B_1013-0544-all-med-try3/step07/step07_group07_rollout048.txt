import numpy as np
import time

def compute_c5(h, n):
    """Compute C5 value."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - hp
            sr = room.sum()
            if sr > 1e-15:
                hp += (diff / sr) * room
        else:
            sh = hp.sum()
            if sh > 1e-15:
                hp += (diff / sh) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary by taking top n/2 entries."""
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def lse_grad_correct(h, n, alpha):
    """Correct gradient of logsumexp(alpha * correlate(h, 1-h))."""
    c = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_a = np.exp(a - sm)
    p = exp_a / exp_a.sum()
    # p[m] is the weight for correlation at index m (lag = m - (n-1))
    # d L / d h[i] = alpha * sum_m p[m] * dc[m]/dh[i]
    # dc[m]/dh[i] = (1-h[i+lag])*I(0<=i+lag<n) - h[i-lag]*I(0<=i-lag<n)
    # where lag = m - (n-1)
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        m = n - 1 + lag
        pk = alpha * p[m]
        if pk < 1e-18:
            continue
        # (1-h[i+lag]) for i where 0 <= i+lag < n
        i_lo = max(0, -lag)
        i_hi = min(n-1, n-1-lag)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi+1)
            grad[idx] += pk * (1.0 - h[idx + lag])
        # -h[i-lag] for i where 0 <= i-lag < n
        i_lo2 = max(0, lag)
        i_hi2 = min(n-1, n-1+lag)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2+1)
            grad[idx2] -= pk * h[idx2 - lag]
    return grad

def adam_continuous(h, n, alphas, deadline, lr=0.01):
    """Adam-style continuous optimization with increasing alpha."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
        t = 0
        for _ in range(5000):
            if time.time() > deadline:
                break
            t += 1
            grad = lse_grad_correct(h, n, alpha)
            m_adam = 0.9 * m_adam + 0.1 * grad
            v_adam = 0.999 * v_adam + 0.001 * grad**2
            m_hat = m_adam / (1 - 0.9**t)
            v_hat = v_adam / (1 - 0.999**t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_new = project(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            # Accept if it improves or is close (for exploration)
            if c_new <= best_c + 1e-10:
                h = h_new
            # Check convergence
            if np.linalg.norm(grad) < 1e-14 * alpha:
                break
    return best_h, best_c
