def compute_c5(h, n):

def compute_all_c(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

def fast_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    preps = compute_all_c(h, n).copy()
    n_full = 2 * n - 1
    s_vals = np.arange(n_full) - (n - 1)
    
    moves_since_recount = 0
    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        for d in deltas:
            sweep_improved = False
            for i in range(n):
                if h[i] + d > 1.0 + 1e-12:
                    continue
                for j in range(n):
                    if i == j or h[j] - d < -1e-12:
                        continue
                    ti = np.zeros(n_full)
                    tf = i + s_vals
                    tb = i - s_vals
                    m_f = (tf >= 0) & (tf < n)
                    m_b = (tb >= 0) & (tb < n)
                    ti[m_f] = 1.0 - h[tf[m_f]]
                    if m_b.any():
                        ti[m_b] -= h[tb[m_b]]
                    tj = np.zeros(n_full)
                    tf = j + s_vals
                    tb = j - s_vals
                    m_f = (tf >= 0) & (tf < n)
                    m_b = (tb >= 0) & (tb < n)
                    tj[m_f] = 1.0 - h[tf[m_f]]
                    if m_b.any():
                        tj[m_b] -= h[tb[m_b]]
                    delta_c = dx * d * (ti - tj)
                    if (preps + delta_c).max() < best_c - 1e-14:
                        h[i] += d
                        h[j] -= d
                        preps = preps + delta_c
                        best_c = preps.max()
                        improved_any = True
                        sweep_improved = True
                        moves_since_recount += 1
                        if moves_since_recount >= 100:
                            preps = compute_all_c(h, n).copy()
                            moves_since_recount = 0
                if time.time() > deadline:
                    return h, best_c
            if not sweep_improved:
                break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    preps_check = preps.max()
    if abs(preps_check - best_c) > 1e-10:
        pass
    return h, best_c

def gradient_guided_search(h, n, deadline, deltas, top_k=10):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n_full = 2 * n - 1
    s_vals = np.arange(n_full) - (n - 1)
    
    for _ in range(100):
        if time.time() > deadline:
            break
        preps = compute_all_c(h, n)
        max_val = preps.max()
        sorted_idx = np.argsort(preps)[::-1]
        active = sorted_idx[:top_k]
        
        grad = np.zeros(n)
        for t in active:
            s = s_vals[t]
            if s >= 0:
                idx_f = np.arange(max(0, 0), min(n, n - s))
                idx_b = np.arange(max(0, s), n)
                grad[idx_f] += dx * (1.0 - h[idx_f + s])
                grad[idx_b] -= dx * h[idx_b - s]
            else:
                a = -s
                idx_f = np.arange(a, n)
                idx_b = np.arange(0, n - a)
                grad[idx_f] += dx * (1.0 - h[idx_f - a])
                grad[idx_b] -= dx * h[idx_b + a]
        
        improved = False
        for d in deltas:
            order = np.argsort(grad)
            candidates = []
            for ii in range(min(30, n)):
                for jj in range(n - 1 - min(30, n), n):
                    i_idx = order[ii]
                    j_idx = order[jj]
                    if i_idx == j_idx:
                        continue
                    if h[i_idx] + d <= 1.0 + 1e-12 and h[j_idx] - d >= -1e-12:
                        candidates.append((i_idx, j_idx))
            for (i_idx, j_idx) in candidates[:500]:
                if time.time() > deadline:
                    return h, best_c
                h[i_idx] += d
                h[j_idx] -= d
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    h[i_idx] -= d
                    h[j_idx] += d
            if not improved:
                break
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def try_resolution(h_best, n_best, n_try, deadline, rng):
    x_src = np.linspace(0, 2, n_best)
    x_dst = np.linspace(0, 2, n_try)
    h_interp = np.interp(x_dst, x_src, h_best)
    h_interp = project_feasible(h_interp, n_try)
    best_c = compute_c5(h_interp, n_try)
    for alpha in [5000, 50000, 500000]:
        if time.time() > deadline - 20:
            break
        ht = slsqp_solve(h_interp, n_try, alpha, maxiter=3000)
        c = compute_c5(ht, n_try)
        if c < best_c - 1e-14:
            best_c = c
            h_interp = ht
    if time.time() < deadline - 5:
        h_interp, best_c = fast_pair_search(h_interp, n_try, deadline - 2,
                                            [0.005, 0.002, 0.001, 0.0005])
    return h_interp, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100
    
    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    n_best = n_init
    
    # Phase 1: Progressive SLSQP on initial
    for alpha in [500, 2000, 10000, 100000, 500000]:
        if time.time() > deadline - 200:
            break
        alloc_iters = 4000
        ht = slsqp_solve(best_h, n_best, alpha, maxiter=alloc_iters)
        c = compute_c5(ht, n_best)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    
    # Phase 2: Fast pair search (main local optimization)
    if time.time() < deadline - 100:
        best_h, best_c = fast_pair_search(best_h, n_best, deadline - 120,
                                          [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
    
    # Phase 3: Threshold guided search
    if time.time() < deadline - 60:
        best_h, best_c = gradient_guided_search(best_h, n_best, deadline - 80,
                                                 [0.005, 0.002, 0.001, 0.0005], top_k=8)
    
    # Phase 4: Another SLSQP pass
    if time.time() < deadline - 50:
        ht = slsqp_solve(best_h, n_best, 2000000, maxiter=6000)
        c = compute_c5(ht, n_best)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    
    # Phase 5: Try other resolutions
    for n_try in [200, 300, 250]:
        if time.time() > deadline - 60 or n_try == n_best:
            continue
        alloc = min(80, (deadline - time.time()) * 0.3)
        if alloc < 30:
            break
        ht, ct = try_resolution(best_h, n_best, n_try, time.time() + alloc, rng)
        if ct < best_c - 1e-12:
            best_c = ct
            best_h = ht
            n_best = n_try
    
    # Phase 6: Final intensive pair search
    if time.time() < deadline - 15:
        deltas_fine = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        best_h, best_c = fast_pair_search(best_h, n_best, deadline - 8, deltas_fine)
    
    # Phase 7: Final SLSQP polish + pair search
    if time.time() < deadline - 8:
        ht = slsqp_solve(best_h, n_best, 5000000, maxiter=4000)
        c = compute_c5(ht, n_best)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
        if time.time() < deadline - 3:
            best_h, best_c = fast_pair_search(best_h, n_best, deadline - 1,
                                              [0.0005, 0.0002, 0.0001, 0.00005])
    
    # Final projection
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    
    return best_h, best_c, n_best
