def add_lag_gradient(g, k, h, n, w):

def smooth_max_value(h, n, alpha):
    if not np.all(np.isfinite(h)):
       return 1e100
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def line_search_direction(h,n,g,best_c,cap=0.08):
    g = np.asarray(g, dtype=float).copy()
    g = g - np.mean(g)
    norm = float(np.linalg.norm(g))
    if norm < 1e-13:
       return h,best_c,False
    d = -g / norm
    sbox = 1.0
    pos = d > 1e-12
    if np.any(pos):
       vals = (1.0 - h[pos]) / d[pos]
       mv = float(np.min(vals))
       if np.isfinite(mv): sbox = min(sbox, mv)
    neg = d < -1e-12
    if np.any(neg):
       vals = h[neg] / (-d[neg])
       mv = float(np.min(vals))
       if np.isfinite(mv): sbox = min(sbox, mv)
    if not np.isfinite(sbox) or sbox <= 1e-12:
       sbox = cap
    sbox = min(float(cap), max(float(sbox), 1e-8))

import numpy as np
import time
from scipy.optimize import minimize


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if n <= 0:
        return h
    target = 0.5 * float(n)
    a = np.clip(h, 0.0, 1.0)
    s = float(a.sum())
    if abs(s - target) < 1e-15:
        return a

    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        ss = float(np.clip(a - mid, 0.0, 1.0).sum())
        if ss > target:
            lo = mid
        else:
            hi = mid

    nu = 0.5 * (lo + hi)
    p = np.clip(a - nu, 0.0, 1.0)
    diff = target - float(p.sum())

    if abs(diff) > 1e-12:
        if diff > 0.0:
            room = 1.0 - p
            mask = room > 1e-13
            if np.any(mask):
                tot = float(room[mask].sum())
                if tot > 0.0:
                    p[mask] += (diff / tot) * room[mask]
        else:
            mask = p > 1e-13
            if np.any(mask):
                tot = float(p[mask].sum())
                if tot > 0.0:
                    p[mask] += (diff / tot) * p[mask]

    diff = target - float(p.sum())
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - p))
            val = p[idx] + diff
            p[idx] = 1.0 if val > 1.0 else val
        else:
            idx = int(np.argmax(p))
            val = p[idx] + diff
            p[idx] = 0.0 if val < 0.0 else val

    return np.clip(p, 0.0, 1.0)


def c5_vals(h, n):
    raw = np.correlate(h, 1.0 - h, mode='full')
    return raw[::-1] * (2.0 / float(n))


def c5_max(h, n):
    return float(np.max(c5_vals(h, n)))


def add_lag_gradient(g, k, h, n, w):
    if w == 0.0:
        return
    if k >= 0:
        nv = n - k
        if nv > 0:
            g[:nv] += w * (1.0 - h[k:n])
            g[k:n] += -w * h[:nv]
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            g[kk:n] += w * (1.0 - h[:nv])
            g[:nv] += -w * h[kk:n]


def smooth_max_value(h, n, alpha):
    if not np.all(np.isfinite(h)):
        return 1e100
    c = c5_vals(h, n)
    m = float(np.max(c))
    z = alpha * (c - m)
    zm = float(np.max(z))
    z = z - zm
    se = float(np.sum(np.exp(z)))
    if se <= 0.0 or not np.isfinite(se):
        return float(m)
    return float(m + (zm + np.log(se)) / alpha)


def grad_smooth_max(h, n, alpha):
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)

    total = 2 * n - 1
    c = c5_vals(h, n)
    m = float(np.max(c))
    z = alpha * (c - m)
    zm = float(np.max(z))
    z = z - zm
    p = np.exp(z)
    s = float(np.sum(p))

    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(total, dtype=float)
        mask = c >= (m - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            mask[int(np.argmax(c))] = True
            cnt = 1
        p[mask] = 1.0 / float(cnt)
    else:
        p /= s

    if alpha < 200.0:
        max_active = min(total, 1200 if n <= 1000 else 700)
    elif alpha < 1000.0:
        max_active = min(total, 900 if n <= 1000 else 550)
    elif alpha < 5000.0:
        max_active = min(total, 600 if n <= 1000 else 400)
    else:
        max_active = min(total, 350 if n <= 1000 else 250)

    thresh = 1e-12 if alpha < 1000.0 else 1e-14
    nz = np.nonzero(p > thresh)[0]
    if len(nz) > max_active:
        idx = np.argpartition(c, -max_active)[-max_active:]
    else:
        idx = nz

    g = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    for mi in idx:
        w = float(p[mi]) * dx
        if w <= 1e-18:
            continue
        add_lag_gradient(g, int(mi - (n - 1)), h, n, w)
    return g


def obj_pen(x, n, alpha, mu):
    val = smooth_max_value(x, n, alpha)
    if not np.isfinite(val):
        return 1e100
    diff = float(x.sum() - 0.5 * n)
    return float(val + mu * diff * diff)


def grad_pen(x, n, alpha, mu):
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - 0.5 * n)
    return g


def slsqp_obj(x, n, alpha):
    return smooth_max_value(x, n, alpha)


def slsqp_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)


def slsqp_cons(x, n):
    return float(x.sum() - 0.5 * n)


def slsqp_jac(x, n):
    return np.ones(n, dtype=float)

def slsqp_phase(h, n, deadline, alpha=10000.0, maxiter=80):

def top_grad(h, n, j):
    c = c5_vals(h, n)
    total = c.size
    j = max(1, min(int(j), total))

    if j == total:
        idx = np.arange(total)
    elif j == 1:
        idx = np.array([int(np.argmax(c))], dtype=int)
    else:
        idx = np.array(np.argpartition(c, -j)[-j:], dtype=int)

    g = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    w = dx / float(j)
    for mi in idx:
        add_lag_gradient(g, int(mi - (n - 1)), h, n, w)
    return g


def line_search_direction(h, n, g, best_c, cap=0.08):
    g = np.asarray(g, dtype=float).copy()
    g = g - float(np.mean(g))
    norm = float(np.linalg.norm(g))
    if norm < 1e-13:
        return h, best_c, False

    d = -g / norm
    sbox = 1.0

    pos = d > 1e-12
    if np.any(pos):
        vals = (1.0 - h[pos]) / d[pos]
        mv = float(np.min(vals))
        if np.isfinite(mv):
            sbox = min(sbox, mv)

    neg = d < -1e-12
    if np.any(neg):
        vals = h[neg] / (-d[neg])
        mv = float(np.min(vals))
        if np.isfinite(mv):
            sbox = min(sbox, mv)

    if not np.isfinite(sbox) or sbox <= 1e-12:
        sbox = cap
    sbox = min(float(cap), max(float(sbox), 1e-8))

    fracs = (1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01)
    for f in fracs:
        s = f * sbox
        if s <= 1e-15:
            continue
        h2 = project_feasible(h + s * d, n)
        c2 = c5_max(h2, n)
        if c2 < best_c - 1e-14:
            return h2, c2, True
    return h, best_c, False


def gradient_descent_top(h, n, deadline, cap=0.08):
    h = project_feasible(h, n)
    best_c = c5_max(h, n)
    js = [1, 2, 3, 5, 8, 13, 21]
    cap_cur = float(cap)
    stuck = 0

    for j in js:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            if stuck > 8:
                break
            g = top_grad(h, n, j)
            h2, c2, ok = line_search_direction(h, n, g, best_c, cap_cur)
            if ok:
                h = h2
                best_c = c2
                stuck = 0
                if cap_cur < cap:
                    cap_cur = min(cap, cap_cur * 1.05)
            else:
                stuck += 1
                cap_cur *= 0.7
                if cap_cur < 1e-6:
                    break
        if stuck > 8:
            break
    return h, best_c

        if rem < 5.0:
            break

        g = top_grad(h, n, top_j)
        g = g - float(np.mean(g))
        if float(np.linalg.norm(g)) < 1e-13:
            break

        high = np.argpartition(g, -P)[-P:]
        low = np.argpartition(g, P)[:P]
        hi_order = rng.permutation(high)
        lo_order = rng.permutation(low)

        rate = 8.0 if n <= 600 else 3.0
        max_attempts = max(30, min(P * P, int(rem * rate)))
        attempts = 0
        improved = False

        for i in hi_order:
            if improved or attempts >= max_attempts:
                break
            for j in lo_order:
                attempts += 1
                if attempts >= max_attempts:
                    break
                if i == j:
                    continue
                if g[i] > g[j]:
                    s = i
                    t = j
                else:
                    s = j
                    t = i
                limit = min(h[s], 1.0 - h[t])
                if limit <= 1e-10:
                    continue
                for f in fracs:
                    d = f * limit
                    if d <= 1e-12:
                        continue
                    h[s] -= d
                    h[t] += d
                    c2 = c5_max(h, n)
                    if c2 < best_c - 1e-14:
                        best_c = c2
                        improved = True
                        break
                    h[t] -= d
                    h[s] += d
            if improved:
                break

        if not improved:
            rem = deadline - time.time()
            if rem > 4.0:
                max_rand = max(20, min(100, int(rem * (2.0 if n <= 600 else 1.0))))
                for _r in range(max_rand):
                    rem = deadline - time.time()
                    if rem < 2.0:
                        break
                    i = int(rng.randint(0, n - 1))
                    j = int(rng.randint(0, n - 1))
                    if i == j:
                        continue
                    if g[i] >= g[j]:
                        s = i
                        t = j
                    else:
                        s = j
                        t = i
                    limit = min(h[s], 1.0 - h[t])
                    if limit <= 1e-10:
                        continue
                    for f in (0.05, 0.02, 0.01, 0.005):
                        d = f * limit
                        if d <= 1e-12:
                            continue
                        h[s] -= d
                        h[t] += d
                        c2 = c5_max(h, n)
                        if c2 < best_c - 1e-14:
                            best_c = c2
                            improved = True
                            break
                        h[t] -= d
                        h[s] += d
                    if improved:
                        break

        if not improved:
            break

    h = project_feasible(h, n)
    best_c = min(best_c, c5_max(h, n))
    return h, best_c


def resample_h(h_old, n_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    if n_new == n_old:
        return project_feasible(h_old.copy(), n_new)
    if n_new > n_old:
        x_old = np.linspace(0.0, 2.0, n_old)
        x_new = np.linspace(0.0, 2.0, n_new)
        h_new = np.interp(x_new, x_old, h_old)
    else:
        h_new = np.empty(n_new, dtype=float)
        for i in range(n_new):
            lo = (i * n_old) // n_new
            hi = ((i + 1) * n_old) // n_new
            if hi <= lo:
                hi = lo + 1
            if hi > n_old:
                hi = n_old
            if lo >= n_old:
                h_new[i] = h_old[-1]
            else:
                h_new[i] = float(np.mean(h_old[lo:hi]))
    return project_feasible(h_new, n_new)


def default_h(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5, dtype=float)
    for f in range(1, 8):
        amp = 0.25 / float(f)
        h += amp * float(rng.uniform(-1.0, 1.0)) * np.sin(np.pi * f * x + float(rng.uniform(0.0, 2.0 * np.pi)))
        h += amp * float(rng.uniform(-1.0, 1.0)) * np.cos(np.pi * f * x + float(rng.uniform(0.0, 2.0 * np.pi)))
    return project_feasible(h, n)


def sharpen_transform(h, n, gamma):
    h2 = 0.5 + gamma * (h - 0.5)
    return project_feasible(h2, n)


def try_sharpen(h, n, deadline):
    h = project_feasible(h, n)
    best = h.copy()
    best_c = c5_max(h, n)
    improved = False

    gammas = (1.6, 2.0, 2.6, 3.5)
    for gamma in gammas:
        if time.time() > deadline:
            break
        h2 = sharpen_transform(best, n, float(gamma))
        c2 = c5_max(h2, n)
        if c2 < best_c - 1e-13:
            best = h2
            best_c = c2
            improved = True

    if improved and time.time() < deadline - 5.0:
        t_end = min(deadline - 2.0, time.time() + 20.0)
        best, best_c = gradient_descent_top(best, n, t_end, cap=0.03)
        if time.time() < deadline - 2.0:
            best, best_c = pair_polish(best, n, deadline)

    best = project_feasible(best, n)
    best_c = min(best_c, c5_max(best, n))
    return best, best_c


def make_candidates(base):
    base_mults = [1.0, 0.9, 1.1, 1.25, 1.5, 1.8, 2.2, 2.8]
    raw = []
    for m in base_mults:
        raw.append(int(round(base * float(m))))
    for nt in (200, 250, 300, 350, 450, 600, 800, 1000, 1300):
        raw.append(nt)

    uniq = []
    for nt in raw:
        if 80 <= nt <= 2600 and nt not in uniq:
            uniq.append(nt)

    keys = []
    for nt in uniq:
        keys.append((abs(nt - base), nt, nt))
    keys.sort()

    ordered = []
    for item in keys:
        ordered.append(item[2])
        if len(ordered) >= 9:
            break
    return ordered


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 5.0

    try:
        iv = initial_h_values
        if iv is None:
            raise ValueError
        h0 = np.asarray(iv, dtype=float).ravel()
        n0 = int(len(h0))
        if n0 <= 0 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n0 = 600
        h0 = default_h(n0, int(seed))

    if n0 < 10:
        n0 = 300
        h0 = default_h(n0, int(seed))

    best_h = project_feasible(h0, n0)
    best_c = c5_max(best_h, n0)
    best_n = int(n0)

    if budget <= 25.0:
        best_h = project_feasible(best_h, best_n)
        best_c = c5_max(best_h, best_n)
        return best_h, float(best_c), best_n

    if deadline - time.time() > 90.0:
        t_end = min(deadline - 140.0, time.time() + 120.0)
        best_h, best_c = lbfgs_phase(best_h, best_n, t_end)
        if time.time() < deadline - 140.0:
            t_end2 = min(deadline - 140.0, time.time() + 35.0)
            best_h, best_c = pair_polish(best_h, best_n, t_end2)

    candidates = make_candidates(best_n)

    for idx, nt in enumerate(candidates):
        rem_all = deadline - time.time()
        if rem_all < 80.0:
            break

        if idx < 2:
            alloc = int(min(220, max(65, rem_all * 0.22)))
        else:
            alloc = int(min(180, max(55, rem_all * 0.15)))
        if alloc > rem_all - 80:
            alloc = int(max(0, rem_all - 80))
        if alloc < 50:
            break

        t_end = time.time() + float(alloc)
        h_t = resample_h(best_h, best_n, nt)
        c_t = c5_max(h_t, nt)

        if time.time() < t_end - 25.0:
            t_lb_end = min(t_end - 30.0, time.time() + min(float(alloc) * 0.55, 110.0))
            h_t, c_t = lbfgs_phase(h_t, nt, t_lb_end)

        rem_trial = t_end - time.time()
        if rem_trial > 45.0 and nt <= 1500:
            t_sl = min(t_end - 30.0, time.time() + min(45.0, rem_trial * 0.35))
            h_t2, c_t2 = slsqp_phase(h_t, nt, t_sl, alpha=10000.0, maxiter=80)
            if c_t2 < c_t:
                h_t, c_t = h_t2, c_t2

        rem_trial = t_end - time.time()
        if rem_trial > 30.0:
            t_dsc = min(t_end - 15.0, time.time() + min(40.0, rem_trial * 0.35))
            h_t, c_t = gradient_descent_top(h_t, nt, t_dsc, cap=0.06)

        rem_trial = t_end - time.time()
        if rem_trial > 15.0:
            t_pair = min(t_end - 3.0, time.time() + rem_trial * 0.6)
            h_t, c_t = pair_polish(h_t, nt, t_pair)

        h_t = project_feasible(h_t, nt)
        c_t = c5_max(h_t, nt)
        if c_t < best_c - 1e-15:
            best_c = c_t
            best_h = h_t
            best_n = int(nt)

    rem = deadline - time.time()
    if rem > 100.0:
        t_sh_end = min(deadline - 60.0, time.time() + 40.0)
        h_sh, c_sh = try_sharpen(best_h, best_n, t_sh_end)
        if c_sh < best_c - 1e-15:
            best_c = c_sh
            best_h = h_sh

    rem = deadline - time.time()
    if rem > 70.0:
        t_end = min(deadline - 40.0, time.time() + 50.0)
        best_h, best_c = lbfgs_phase(best_h, best_n, t_end)

    rem = deadline - time.time()
    if rem > 35.0:
        t_end = min(deadline - 20.0, time.time() + 25.0)
        best_h, best_c = gradient_descent_top(best_h, best_n, t_end, cap=0.05)

    rem = deadline - time.time()
    if rem > 18.0:
        t_end = min(deadline - 2.0, time.time() + min(15.0, rem - 16.0))
        best_h, best_c = pair_polish(best_h, best_n, t_end)

    best_h = project_feasible(best_h, best_n)
    best_c = c5_max(best_h, best_n)
    return best_h, float(best_c), int(best_n)
