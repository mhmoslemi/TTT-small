    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_gradient(h, n, alpha):
    """Compute gradient of smoothed max correlation."""
    c = compute_c5_vec(h, n)
    c_max = c.max()
    # Soft-max weights
    shifted = alpha * (c - c_max)
    weights = np.exp(shifted - shifted.max())
    weights /= weights.sum()
    
    # Gradient of c[k] w.r.t. h[i]
    # c[k] = sum_i h[i] * (1-h[i+k-(n-1)]) * dx for valid indices
    # dc[k]/dh[i] = (1-h[i+k-(n-1)]) * dx - h[i-k+(n-1)] * dx (if valid)
    
    grad = np.zeros(n)
    for m in range(2*n - 1):
        if weights[m] < 1e-12:
            continue
        k = m - (n - 1)  # actual lag
        w = weights[m]  # already includes dx factor since c includes dx
        
        if k >= 0:
            # c[m] = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k]) * dx
            # dc[m]/dh[i] for i in [0, n-1-k]: (1-h[i+k]) * dx
            # dc[m]/dh[i] for i in [k, n-1]: -h[i-k] * dx
            for i in range(n - k):
                grad[i] += w * (1 - h[i + k])
                grad[i + k] += w * (-h[i])
        else:
            kk = -k
            # c[m] = sum_{i=kk}^{n-1} h[i] * (1-h[i-kk]) * dx
            # dc[m]/dh[i] for i in [kk, n-1]: (1-h[i-kk]) * dx
            # dc[m]/dh[i] for i in [0, n-1-kk]: -h[i+kk] * dx
            for i in range(kk, n):
                grad[i] += w * (1 - h[i - kk])
                grad[i - kk] += w * (-h[i])
    
    return grad

If I change only h[i] by δ (and h[j] by -δ), I can compute the exact change in each C[k] without recomputing the full correlation. This is because:

C[k] = ∑_{valid i} h[i]·(1-h[i+k])·dx

If I change h[i] → h[i]+δ and h[j] → h[j]-δ:
ΔC[k] = δ·(1-h[i+k])·dx - δ·h[i-k]·dx - δ·(1-h[j+k])·dx + δ·h[j-k]·dx
       = δ·dx·[(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
       = δ·dx·[h[j+k] - h[i+k] - h[i-k] + h[j-k]]

(where we treat h[·] = 0 outside [0,n-1])

This is O(1) per (k, i, j) triple, but we need to check all k. So for a given (i,j), computing the effect on all correlations is O(n). But finding the best (i,j) is still O(n²) in the worst case.

However, we can precompute the "neighborhood" of each index. For a given i, the indices that affect C[k] when h[i] changes are i+k and i-k for all k. So the effect is:

For each k: ΔC[k] = δ·dx·[(1-h[i+k]) - h[i-k]] (for the i part)

This is the column of the Jacobian corresponding to h[i]. If I precompute all these columns (an n×(2n-1) matrix), I can find the best pair much faster.

    for n_try in [400, 500, 700]:
        if time.time() > deadline - 100:
            break
        x_old = np.linspace(0.0, 2.0, best_n)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc = min(100, deadline - time.time() - 50)
        if alloc < 30:
            break
        
        # L-BFGS at new resolution
        t_sub1 = time.time() + min(alloc * 0.5, 50)
        h_t, c_t = lbfgs_opt(h_interp, n_try, t_sub1, alpha=2000, mu=5.0, maxiter=400)
        
        # Gradient search
        if time.time() < deadline - 20:
            t_sub2 = time.time() + min(alloc * 0.4, 40)
            h_t, c_t = grad_guided_search(h_t, n_try, t_sub2, seed=seed + n_try)
        
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try
    
    # Phase 5: Final intensive search at best resolution
    if time.time() < deadline - 5:
        remaining = deadline - time.time() - 3
        if remaining > 30:
            best_h, best_c = grad_guided_search(best_h, best_n, time.time() + remaining * 0.6,
                                                seed=seed + 1000)
        if time.time() < deadline - 4:
            best_h, best_c = lbfgs_opt(best_h, best_n, deadline - 3, alpha=5000, mu=10.0, maxiter=200)
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
