def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, eta, deadline):
    """One step of projected subgradient descent on the min-max problem."""
    dx = 2.0 / n
    c = compute_all_c(h, n)
    idx = np.argmax(c)
    # Compute subgradient of c[idx] w.r.t. h
    m = idx - (n - 1)  # convert to actual lag
    grad = np.zeros(n)
    # ∂c[m]/∂h[j] = (1-h[j-m]) * 1{0≤j-m<n} - h[j+m] * 1{0≤j+m<n}
    # For j-m valid:
    valid_jm = np.arange(n) - m
    mask1 = (valid_jm >= 0) & (valid_jm < n)
    grad[mask1] += (1 - h[valid_jm[mask1]])
    # For j+m valid:
    valid_jp = np.arange(n) + m
    mask2 = (valid_jp >= 0) & (valid_jp < n)
    grad[mask2] -= h[valid_jp[mask2]]
    grad *= dx
    # Projected gradient step
    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)
    return h_new, c.max()

def slsqp_optimize(h, n, alpha, maxiter=500):
    """SLSQP optimization with soft-max approximation."""
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_feasible_fast(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s < 1e-14:
            return h
        h = h + diff / s * room
    else:
        s = h.sum()
        if s < 1e-14:
            return h
        h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def swap_delta(h, n, i, j, m_range):
    """Compute delta in correlation when swapping h[i] from 0 to 1 and h[j] from 1 to 0."""
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    delta = np.zeros(2 * n - 1)
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def sa_1swap(h_init, n, deadline, T0=0.004, Tend=0.000005, max_iters=500000, seed=42):
    """Simulated annealing with 1-swap moves only (efficient delta computation)."""
    target = n // 2
    rng = np.random.RandomState(seed)
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    one_idx = np.where(h == 1)[0]
    zero_idx = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        T = T0 * (Tend / T0) ** (it / max_iters)
        i = zero_idx[rng.randint(len(zero_idx))]
        j = one_idx[rng.randint(len(one_idx))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        dc = nc - cur_c
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-15)):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = nc
            zero_idx = np.where(h == 0)[0]
            one_idx = np.where(h == 1)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

                    best_h = h.copy()
        else:

            bs = int(rng.choice(block_sizes))
            if bs > len(one_idx):
                continue

            valid_runs = [r for r in runs if len(r) >= bs]
            if not valid_runs:

            accepted = False

                dest_start = rng.randint(0, n - bs + 1)
                dest = np.arange(dest_start, dest_start + bs)

                if len(overlap) > 0 and not np.array_equal(sorted(overlap.tolist()), sorted(src.tolist())):
                    continue
                if np.any(h[dest] == 1):
                    continue
                # Apply

                h[dest] = 1.0
                new_corr = np.correlate(h, 1 - h, mode='full')

                        best_c = cur_c
                        best_h = h.copy()
                    accepted = True
                else:
                    h[dest] = 0.0
                    h[src] = 1.0
                break

    best_c = cur_c
    best_h = h.copy()
    one_idx = np.where(h == 1)[0].tolist()
    zero_idx = np.where(h == 0)[0].tolist()
    m_range = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):
        if time.time() >= deadline:
            break
        improved = False
        for i in list(zero_idx):
            if time.time() >= deadline:

            for j in one_idx:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx

                    cur_c = nc

                    zero_idx.remove(i)
                    zero_idx.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_refine(h, n, deadline, eta_init=0.001, eta_end=0.00001, max_iters=50000, momentum=0.9):
    """Projected subgradient method with momentum on the min-max problem."""
    dx = 2.0 / n
    h_curr = h.copy()
    v = np.zeros(n)
    best = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        eta = eta_init * (eta_end / eta_init) ** (it / max_iters)
        c = compute_all_c(h_curr, n)
        idx = np.argmax(c)
        m = idx - (n - 1)
        grad = np.zeros(n)
        jm = np.arange(n) - m
        mask1 = (jm >= 0) & (jm < n)
        grad[mask1] += (1.0 - h_curr[jm[mask1]])
        jp = np.arange(n) + m
        mask2 = (jp >= 0) & (jp < n)
        grad[mask2] -= h_curr[jp[mask2]]
        grad *= dx
        v = momentum * v - eta * grad
        h_new = h_curr + v
        h_new = project_feasible_fast(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best = h_new.copy()
        h_curr = h_new
    return best, best_c

                best_h = hb.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
