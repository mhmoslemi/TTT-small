def compute_c5(h, n):

def compute_all_lags(h, n):

def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_dlf_lp(h_curr, n, top_k=None):
    """Build DLF linearized LP. Returns (h_var, t_var, cons, prob)."""
    m = 2 * n - 1
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    cons = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
    ]
    
    # Precompute raw correlate
    raw_corr = np.correlate(h_curr, 1 - h_curr, mode='full')
    
    # If top_k is given, only include top-k lags
    if top_k is not None and top_k < m:
        top_indices = np.argsort(raw_corr)[-top_k:]
    else:
        top_indices = range(m)
    
    for k in top_indices:
        lag = k - (n - 1)
        coeff = np.zeros(n)
        const = 0.0
        
        if lag == 0:
            coeff = 1.0 - 2.0 * h_curr
            const = np.sum(h_curr * h_curr)
        elif lag > 0:
            s = lag
            coeff[:n - s] -= h_curr[s:]
            coeff[s:] += (1.0 - h_curr[:n - s])
            const = np.sum(h_curr[s:] * (1.0 - h_curr[:n - s]))
        else:
            s = -lag
            coeff[:n - s] += (1.0 - h_curr[s:])
            coeff[s:] -= h_curr[:n - s]
            const = np.sum(h_curr[:n - s] * (1.0 - h_curr[s:]))
        
        cons.append(cp.sum(coeff * h_var) + const <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    return h_var, t_var, prob

def solve_lp(h_curr, n, top_k=None, deadline=None):
    """Solve the DLF LP and return new h."""
    h_var, t_var, prob = build_dlf_lp(h_curr, n, top_k=top_k)
    
    solvers = [cp.GLOP, cp.CBC, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            if deadline is not None and time.time() > deadline - 3:
                return h_curr.copy()
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if h_var.value is not None:
                    return np.clip(h_var.value, 0.0, 1.0)
        except Exception:
            continue
    return h_curr.copy()

def successive_lp(h_curr, n, max_iters=40, tol=1e-10, deadline=None, top_k=None):
    """Run successive DLF-LP iterations."""
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline - 3:
            break
        
        h_new = solve_lp(h_curr, n, top_k=top_k, deadline=deadline)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        
        step = np.max(np.abs(h_new - h_curr))
        if step < tol:
            break
        h_curr = h_new
    
    return best_h, best_c

def snap_to_binary(h, n, threshold=0.05):
    """Snap values close to 0 or 1 to exactly 0 or 1, then re-project."""
    h = h.copy()
    h[h < threshold] = 0.0
    h[h > 1.0 - threshold] = 1.0
    h = project_feasible(h, n)
    return h

def pair_swap_search(h, n, deadline, max_sweeps=5):
    """Local search: at each sweep, find worst lag and try swaps."""
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.03, 0.015, 0.008, 0.004, 0.002]
    
    for d in deltas:
        for _ in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            # Try top 3 lags
            top_lags = np.argsort(c)[-3:]
            for k_star in top_lags:
                if time.time() > deadline:
                    break
                lag = int(k_star) - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                for i in range(0, n - m, max(1, (n - m) // 500)):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] < h[j]:
                        delta = min(d, (h[j] - h[i]) * 0.4, h[j]*0.5, 0.5 - h[i]*0.5 + d)
                        if delta < 1e-7:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                    elif h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.4, h[i]*0.5, 0.5 - h[j]*0.5 + d)
                        if delta < 1e-7:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
            if not improved:
                break
    
    return h, best_c

def stochastic_hill_climb(h, n, deadline, n_iter=2000, rng=None):
    """Random perturbation + acceptance."""
    if rng is None:
        rng = np.random.RandomState(123)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    temperature = 0.02
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Pick 2 random positions and swap some mass
        i, j = rng.choice(n, 2, replace=False)
        delta = rng.uniform(0.005, 0.03) * rng.choice([1, -1])
        
        hi = h[i] + delta
        hj = h[j] - delta
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i] = hi
            h[j] = hj
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                temperature = max(0.002, temperature * 0.99)
            else:
                # Revert with some probability (accept worse occasionally)
                accept_prob = np.exp(-(c_new - best_c) / (temperature * 0.01))
                if rng.random() < accept_prob:
                    temperature = max(0.002, temperature * 0.95)
                else:
                    h[i], h[j] = hi - delta, hj + delta
                    h = project_feasible(h, n)
    
    return best_h, best_c

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h, n, alpha=50000, maxiter=2000):
    """SLSQP refinement with smooth max objective."""
    bounds = [(0.0, 1.0)] * n
    
    def sum_con(x):
        return np.sum(x) - n / 2.0
    
    def sum_con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    # Phase 1b: LP with top-k only (faster, might find different solution)

        if c_top < best_c:

            best_h = h_top
    
    # Phase 2: SLSQP refinement

            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
    
    # Phase 3: Binary snapping
    if time.time() < deadline - 10:
        h_snap = snap_to_binary(best_h, n, threshold=0.03)

        c_new = compute_c5(h_new, n)

            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)

    return best_h, best_c
