import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, pen_weight):
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    if alpha < 1:
        smax = m
    else:
        shifted = alpha * (c - m)
        shifted = np.clip(shifted, -700, 700)
        mx = shifted.max()
        if mx > 500:
            smax = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        else:
            smax = m + np.log(np.sum(np.exp(shifted))) / alpha
    s = h.sum() - n / 2.0
    return float(smax + pen_weight * s * s)

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000, pen_weight=0.0):
    h = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    if pen_weight == 0:
        # Use constraint
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
        try:
            res = minimize(lambda x: smooth_obj(x, n, alpha, 0), h,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            return project_feasible(res.x, n)
        except:
            return h
    else:
        # Use penalty
        try:
            res = minimize(lambda x: smooth_obj(x, n, alpha, pen_weight), h,
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            return project_feasible(res.x, n)
        except:
            return h

    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()

    stuck = 0

        j = rng.randint(n)
        if i == j:

        d = step * (1.0 if rng.rand() < 0.5 else -1.0)

        hj = h[j] - d
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
