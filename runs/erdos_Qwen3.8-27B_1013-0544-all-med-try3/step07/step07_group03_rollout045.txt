from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jacobian(x, n):
    return np.ones(x.shape[0])

def objective_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    s = np.sum(np.exp(shifted))
    if s > 0:
        return m + np.log(s) / alpha
    return m

def gradient_smooth(x, n, alpha):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        mx = shifted.max()
        e = np.exp(shifted - mx)
        p = e / e.sum()
    else:
        e = np.exp(shifted)
        p = e / e.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum over k where 0<=j+k<n of p[k+(n-1)]*(1-x[j+k])
        k_lo = max(0, -j)
        k_hi = min(n - 1 - j, n - 1)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            grad[j] += dx * np.dot(p[ks + (n - 1)], 1.0 - x[j + ks])
        # Second term: -sum over k where 0<=j-k<n of p[k+(n-1)]*x[j-k]
        k_lo2 = max(0, j - (n - 1))
        k_hi2 = min(j, n - 1)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            grad[j] -= dx * np.dot(p[ks2 + (n - 1)], x[j - ks2])
    return grad

def slsqp_optimize(h0, n, alpha, maxiter=3000):
    dx = 2.0 / n
    constraints = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective_smooth, h0.copy(), jac=lambda x: gradient_smooth(x, n, alpha),
                       args=(), method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n), res
    except Exception:
        return project_feasible(h0.copy(), n), None

def subgradient_step(h, n, alpha, step_scale=1.0):
    """One subgradient step: average gradient of top-k correlations."""
    dx = 2.0 / n
    c = compute_corr(h, n)
    n_corr = len(c)
    m_val = c.max()
    # Find all near-max lags
    threshold = m_val - 0.005
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    k_vals = np.arange(n_corr) - (n - 1)  # lag for each correlation index
    
    grad = np.zeros(n)
    for idx in active:
        k = k_vals[idx]
        for j in range(n):
            if 0 <= j + k < n:
                grad[j] += dx * (1.0 - h[j + k])
            if 0 <= j - k:
                grad[j] -= dx * h[j - k]
    grad /= len(active)
    
    # Step size
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return h.copy()
    step = step_scale * alpha / gnorm
    h_new = h - step * grad
    return project_feasible(h_new, n)
