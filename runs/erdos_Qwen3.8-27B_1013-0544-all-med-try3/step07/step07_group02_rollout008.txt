def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0 - 1e-12, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-12
            h_orig = h.copy()
            h = h_orig * mask
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    remaining = target_sum - h.sum()
    if abs(remaining) > 1e-13:
        if remaining > 0:
            candidates = np.where(h < 1.0)[0]
        else:
            candidates = np.where(h > 0.0)[0]
        if len(candidates) > 0:
            h[candidates] += remaining / len(candidates)
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    else:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    return grad

    grad *= dx
    return grad

def smooth_max_forward(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def slsqp_active_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def slsqp_full_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max_forward(c, alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16, top_m=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        if top_m is not None:

            res = minimize(slsqp_active_obj, h, args=(n, top_idx, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        else:
            res = minimize(slsqp_full_obj, h, args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def momentum_descent(h, n, deadline, alpha, lr_start=0.5, momentum=0.9, max_iter=20000):
    h_curr = project_feasible(h.copy(), n)
    best_h = h_curr.copy()
    best_c = compute_c5(best_h, n)
    v = np.zeros(n)
    patience = 0
    lr = lr_start
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        w = smooth_max_weights(c_all, alpha)
        
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] > 1e-10:
                lag = k - (n-1)
                grad += w[k] * lag_gradient(h_curr, n, lag)
        dx = 2.0 / n
        grad *= dx
        grad -= grad.mean()
        
        v = momentum * v + lr * grad
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 500:
                lr *= 0.7
                v *= 0.3
                patience = 0
                alpha = min(alpha * 2, 1000000)
        
        h_curr = h_new
        if it % 5000 == 4999:
            lr = max(0.05, lr * 0.8)
    
    return best_h, best_c

def interpolate_h(h_in, n_in, n_out):
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    h_out = np.interp(x_out, x_in, h_in)
    return project_feasible(h_out, n_out)
