def c_top(h, top_idx, n):
    dx=2.0/n
    out=np.empty(len(top_idx))
    one_minus=1.0-h
    for t,idx in enumerate(top_idx):
       lag=idx-(n-1)
       if lag>0:
          out[t]=np.dot(h[lag:], one_minus[:-lag])*dx
       elif lag<0:
          m=-lag
          out[t]=np.dot(h[:-m], one_minus[m:])*dx
       else:
          out[t]=np.dot(h, one_minus)*dx
    return out

def grad_top(h, top_idx, n, weights):
    dx=2.0/n
    g=np.zeros(n)
    for idx, w in zip(top_idx, weights):
       if w<1e-14: continue
       lag=idx-(n-1)
       if lag>0:
          m=lag
          # from first factor h_i for i=m..n-1: +(1-h_{i-m})
          g[m:] += w*(1.0 - h[:-m])
          # from second factor h_i for i=0..n-m-1: -h_{i+m}
          g[:-m] += -w*h[m:]
       elif lag<0:
          m=-lag
          # c=∑ h_i(1-h_{i+m})
          g[:-m] += w*(1.0 - h[m:])
          g[m:] += -w*h[:-m]
       else:
          g += w*(1.0 - 2.0*h)
    return g*dx

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot <= 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h,0,1)
    # final correction
    diff = target - h.sum()
    if diff > 1e-15:
       idx = int(np.argmax(1.0 - h))
       h[idx] = min(1.0, h[idx] + diff)
    elif diff < -1e-15:
       idx = int(np.argmax(h))
       h[idx] = max(0.0, h[idx] + diff)
    return h
```

import numpy as np
import time
import math
from scipy.optimize import minimize


def sanitize(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    return h


def project_feasible(h, n):
    y = sanitize(h)
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    else:
        y = y.copy()

    target = 0.5 * float(n)
    x = np.clip(y, 0.0, 1.0)
    s = float(np.sum(x))

    if abs(s - target) > 0.25:
        lo = float(np.min(y)) - 1.0
        hi = float(np.max(y)) + 1.0
        tval = x
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            tval = np.clip(y - mid, 0.0, 1.0)
            s = float(np.sum(tval))
            if s > target:
                lo = mid
            else:
                hi = mid
            if hi - lo < 1e-13:
                break
        x = tval

    for _ in range(80):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - x
            tot = float(np.sum(room))
            if tot <= 1e-15:
                break
            x = x + (diff / tot) * room
        else:
            tot = float(np.sum(x))
            if tot <= 1e-15:
                break
            x = x + (diff / tot) * x
        x = np.clip(x, 0.0, 1.0)

    for _ in range(3):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - x))
            if 1.0 - x[idx] > 1e-12:
                x[idx] += diff
        else:
            idx = int(np.argmax(x))
            if x[idx] > 1e-12:
                x[idx] += diff
        x = np.clip(x, 0.0, 1.0)

    return x


def compute_lags(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h):
    return float(np.max(compute_lags(h)))


def c_top_h_lags(h, lags, n):
    h = np.asarray(h, dtype=float)
    om = 1.0 - h
    out = np.empty(len(lags), dtype=float)
    dx = 2.0 / float(n)
    for t in range(len(lags)):
        lag = int(lags[t])
        if lag > 0:
            m = lag
            out[t] = float(np.dot(h[m:n], om[0:n - m])) * dx
        elif lag < 0:
            m = -lag
            out[t] = float(np.dot(h[0:n - m], om[m:n])) * dx
        else:
            out[t] = float(np.dot(h, om)) * dx
    return out


def smooth_max_value(vals, alpha):
    if len(vals) == 0:
        return 0.0
    m = float(np.max(vals))
    if alpha <= 0.0:
        return m
    shifted = alpha * (vals - m)
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    if s <= 0.0:
        return m
    return float(m + math.log(s) / alpha)


def smooth_max_weights(vals, alpha):
    L = len(vals)
    if L == 0:
        return np.ones(0, dtype=float)
    m = float(np.max(vals))
    shifted = alpha * (vals - m)
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    if s <= 0.0:
        w = np.zeros(L, dtype=float)
        w[int(np.argmax(vals))] = 1.0
        return w
    return exps / s


def grad_top_h_lags(h, lags, n, weights):
    h = np.asarray(h, dtype=float)
    g = np.zeros(n, dtype=float)
    if len(lags) == 0:
        return g

    for t in range(len(lags)):
        w = float(weights[t])
        if abs(w) <= 1e-12:
            continue
        lag = int(lags[t])
        if lag > 0:
            m = lag
            g[m:] += w * (1.0 - h[:-m])
            g[:-m] -= w * h[m:]
        elif lag < 0:
            m = -lag
            g[:-m] += w * (1.0 - h[m:])
            g[m:] -= w * h[:-m]
        else:
            g += w * (1.0 - 2.0 * h)

    return g * (2.0 / float(n))


def active_lags_from_full(full, n, top_k, margin):
    L = len(full)
    if top_k > L:
        top_k = L
    if top_k < 1:
        top_k = 1

    order = np.argsort(full)[::-1]
    selected = order[:top_k]
    maxv = float(full[order[0]])
    cap = max(2 * top_k, 120)

    if margin > 0.0:
        mask = full >= (maxv - margin)
        extra = np.nonzero(mask)[0]
        combined = np.unique(np.concatenate((selected, extra)))
        if len(combined) > cap:
            vals = full[combined]
            keep = np.argsort(vals)[::-1][:cap]
            combined = combined[keep]
        selected = combined

    return (selected - int(n - 1)).astype(int)


def sum_con(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_con_jac(x, n):
    return np.ones(len(x), dtype=float)


def obj_slsqp_active(x, n, lags, alpha):
    h = np.asarray(x, dtype=float)
    return smooth_max_value(c_top_h_lags(h, lags, n), alpha)


def jac_slsqp_active(x, n, lags, alpha):
    h = np.asarray(x, dtype=float)
    vals = c_top_h_lags(h, lags, n)
    w = smooth_max_weights(vals, alpha)
    return grad_top_h_lags(h, lags, n, w)


def obj_full_smax(x, n, alpha):
    h = np.asarray(x, dtype=float)
    return smooth_max_value(compute_lags(h), alpha)


def jac_full_smax(x, n, alpha):
    h = np.asarray(x, dtype=float)
    full = compute_lags(h)
    w = smooth_max_weights(full, alpha)
    mask = w > 1e-10
    if not bool(np.any(mask)):
        return np.zeros(len(h), dtype=float)
    lags = (np.arange(-(n - 1), n, dtype=int))[mask]
    w = w[mask]
    return grad_top_h_lags(h, lags, n, w)


def slsqp_active(h, n, lags, alpha, maxiter, ftol):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(
            obj_slsqp_active,
            h0,
            jac=jac_slsqp_active,
            args=(n, lags, alpha),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': float(ftol)}
        )
        if hasattr(res, 'x'):
            x = sanitize(res.x)
            if x.size == n and np.all(np.isfinite(x)):
                x = project_feasible(x, n)
                if np.all(np.isfinite(x)):
                    return x
    except Exception:
        pass
    return h0


def slsqp_full(h, n, alpha, maxiter, ftol):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(
            obj_full_smax,
            h0,
            jac=jac_full_smax,
            args=(n, alpha),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': float(ftol)}
        )
        if hasattr(res, 'x'):
            x = sanitize(res.x)
            if x.size == n and np.all(np.isfinite(x)):
                x = project_feasible(x, n)

                    return x
    except Exception:
        pass
    return h0


def projected_active(h, n, lags, alpha, deadline, max_iter, lr0):

    best_h = h.copy()
    best_c = compute_c5(h)

    vals = c_top_h_lags(h, lags, n)
    f = smooth_max_value(vals, alpha)
    weights = smooth_max_weights(vals, alpha)
    g = grad_top_h_lags(h, lags, n, weights)

    s_prev = None
    prev_g = None
    lr = float(lr0)
    active_set = set([int(x) for x in lags])

    for it in range(int(max_iter)):
        if time.time() > deadline - 0.5:
            break

        gn = float(np.linalg.norm(g))
        if gn < 1e-11:
            break

        if s_prev is not None:
            yvec = g - prev_g
            sy = float(np.dot(s_prev, yvec))
            if sy > 1e-12:
                bb = float(np.dot(s_prev, s_prev)) / sy
                if bb > 1e-4 and bb < 1.0:
                    lr = 0.5 * lr + 0.5 * bb

        lr = float(np.clip(lr, 1e-4, 0.5))
        accepted = False
        stop = False

        for fac in (1.0, 0.5, 0.25, 0.125):
            step = (lr * fac / gn) * g
            new_h = project_feasible(h - step, n)
            vals_n = c_top_h_lags(new_h, lags, n)
            f_n = smooth_max_value(vals_n, alpha)

            if f_n < f - 1e-13:
                s_prev = new_h - h
                prev_g = g
                h = new_h
                f = f_n
                accepted = True

                weights = smooth_max_weights(vals_n, alpha)
                g = grad_top_h_lags(h, lags, n, weights)

                if it % 8 == 0:
                    full = compute_lags(h)
                    cm = float(np.max(full))
                    cur = int(np.argmax(full)) - (n - 1)
                    if cm < best_c - 1e-14:
                        best_c = cm
                        best_h = h.copy()
                    if cm > best_c + 5e-4 or cur not in active_set:
                        stop = True
                break

        if stop:
            break
        if not accepted:
            lr *= 0.5
            if lr < 1e-4:
                break

    full = compute_lags(h)
    cm = float(np.max(full))
    if cm < best_c - 1e-14:
        best_c = cm
        best_h = h.copy()

    return best_h, best_c

def antisymmetric_make(h, n):
    h = project_feasible(h, n).copy()
    for i in range(n // 2):
        a = float(h[i])
        b = float(h[n - 1 - i])
        na = 0.5 + 0.5 * (a - b)
        na = max(0.0, min(1.0, na))
        nb = 1.0 - na
        h[i] = na
        h[n - 1 - i] = nb
    if n % 2 == 1:

    return project_feasible(h, n)

                    h[hi] = old_hi

                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:

                        d = min(0.015, 0.4 * (sv - tv), 0.5 * sv, 0.5 * (1.0 - tv))
                    else:
                        d = min(0.008, 0.5 * sv, 0.5 * (1.0 - tv))

                    if d < 1e-7:
                        continue

                    old_s = float(h[src])
                    old_t = float(h[sink])
                    h[src] = old_s - d
                    h[sink] = old_t + d
                    if h[src] < -1e-12 or h[sink] > 1.0 + 1e-12:
                        h[src] = old_s
                        h[sink] = old_t
                        continue

                    c_new = compute_lags(h)
                    cm = float(np.max(c_new))
                    budget -= 1
                    if cm < best_c - 1e-15:
                        best_c = cm
                        best_h = h.copy()
                        c = c_new
                        improved = True
                        break
                    else:
                        h[src] = old_s
                        h[sink] = old_t

                if improved:
                    break

            if improved:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 6:
                    break

    h = project_feasible(best_h, n)
    c = compute_lags(h)
    cm = float(np.max(c))
    if cm < best_c - 1e-14:
        best_c = cm
    return h, best_c


def fallback_initial(n, rng):
    x = np.arange(n) * (2.0 / float(n))
    base = 0.5 + 0.25 * np.sin(np.pi * x) + 0.10 * np.sin(2.0 * np.pi * x)
    base = base + rng.normal(0.0, 0.02, n)
    return project_feasible(base, n)


def interpolate_resolution(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return project_feasible(h, n_new)
    x_old = np.arange(n_old) * (2.0 / float(n_old))
    x_new = np.arange(n_new) * (2.0 / float(n_new))
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(y, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = max(10.0, float(budget_s))
    deadline = t0 + B - 8.0
    rng = np.random.RandomState(int(seed))

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

    if h_init is None or len(h_init) == 0:
        n0 = 512
        h0 = fallback_initial(n0, rng)
    else:
        n0 = len(h_init)
        h0 = project_feasible(h_init, n0)

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h)

    main_budget = deadline - 8.0 - time.time()
    if n0 <= 560:
        alloc_main = min(400.0, max(120.0, main_budget * 0.35))
    else:
        alloc_main = min(300.0, max(100.0, main_budget * 0.30))
    main_deadline = time.time() + alloc_main

    h_main, c_main = polish_resolution(best_h, n0, main_deadline, rng, allow_slsqp=(n0 <= 640))
    if c_main < best_c - 1e-14:
        best_h, best_c = h_main, c_main

    if time.time() < deadline - 220.0 and len(best_h) <= 640:
        n_alt = len(best_h)
        h_alt = antisymmetric_make(best_h, n_alt)
        c_alt = compute_c5(h_alt)
        if c_alt < best_c + 0.03:
            full_alt = compute_lags(h_alt)
            lags_alt = active_lags_from_full(full_alt, n_alt, 30, 3e-4)
            alt_deadline = min(deadline - 150.0, time.time() + 90.0)
            h_alt2, c_alt2 = projected_active(h_alt, n_alt, lags_alt, 1500.0, alt_deadline, 1000, 0.05)
            if c_alt2 < best_c - 1e-14:
                best_h, best_c = h_alt2, c_alt2

    base_n = len(best_h)
    candidates = []
    if base_n < 640:
        candidates.append(640)
    if base_n < 768:
        candidates.append(768)

    for n_try in candidates:
        if time.time() > deadline - 150.0:
            break
        remaining = deadline - time.time()
        alloc = min(170.0, remaining * 0.35)
        if alloc < 80.0:
            break

        h_try = interpolate_resolution(best_h, n_try)
        c_try = compute_c5(h_try)
        if c_try > best_c + 0.012:
            continue

        try_deadline = time.time() + alloc
        h_try2, c_try2 = polish_resolution(h_try, n_try, try_deadline, rng, allow_slsqp=(n_try <= 640))
        if c_try2 < best_c - 1e-14:
            best_h, best_c = h_try2, c_try2

    n_best = len(best_h)
    if time.time() < deadline - 100.0:
        scale = 0.03
        h_p = project_feasible(best_h + rng.normal(0.0, scale, n_best), n_best)
        c_p = compute_c5(h_p)
        if c_p < best_c + 0.02:
            full_p = compute_lags(h_p)
            lags_p = active_lags_from_full(full_p, n_best, 30, 2e-4)
            pert_deadline = min(deadline - 40.0, time.time() + min(100.0, (deadline - time.time()) * 0.3))
            h_p2, c_p2 = projected_active(h_p, n_best, lags_p, 2000.0, pert_deadline, 1500, 0.04)
            if c_p2 < best_c - 1e-14:
                best_h, best_c = h_p2, c_p2
                n_best = len(best_h)

    if time.time() < deadline - 30.0 and len(best_h) <= 640:
        time_left = deadline - time.time()
        if time_left > 40.0:
            maxit = min(25, max(10, int(time_left / 20)))
            h_f = slsqp_full(best_h, len(best_h), 5000.0, maxit, 1e-13)
            c_f = compute_c5(h_f)
            if c_f < best_c - 1e-14:
                best_h, best_c = h_f, c_f

    if time.time() < deadline - 5.0:
        h_loc, c_loc = local_polish(best_h, len(best_h), deadline - 2.0, rng)
        if c_loc < best_c - 1e-14:
            best_h, best_c = h_loc, c_loc

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)

    return best_h, float(best_c), int(len(best_h))
