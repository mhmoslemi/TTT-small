        grad += w[k] * lag_gradient(h, n, lag)
    return grad * dx

def grad_momentum(h, n, alpha, deadline, lr=1.5, momentum=0.9):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    patience = 0
    while time.time() < deadline:
        g = grad_smooth_max(h_curr, n, alpha)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        if patience > 500:
            lr *= 0.7
            v *= 0.5
            patience = 0
    return best_h, best_c

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=5000, top_m=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        if top_m is not None:
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]
            res = minimize(_active_smooth_obj, h, args=(n, top_idx, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        else:
            res = minimize(lambda x: smooth_max_value(np.correlate(x, 1-x, mode='full')*2.0/n, alpha),
                          h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def swap_delta_c5(h, n, i, j, delta):
    """Compute change in C5 when h[i]-=delta, h[j]+=delta. O(n) computation."""
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return compute_c5(h_new, n)

def aggressive_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    
    step_sizes = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    ss_idx = 0
    stuck_count = 0
    
    while time.time() < deadline:
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        
        for frac_try in [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            if time.time() > deadline:
                break
            for d in step_sizes[min(ss_idx, len(step_sizes)-1):min(ss_idx+3, len(step_sizes))]:
                if time.time() > deadline:
                    break
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-12:
                        delta = frac_try * min(d, (h[i]-h[j])*0.5)
                        if delta < 1e-12:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            c_all = compute_all_lags(h, n)
                            improved = True
                            stuck_count = 0
                            ss_idx = min(ss_idx + 1, len(step_sizes)-1)
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-12:
                        delta = frac_try * min(d, (h[j]-h[i])*0.5)
                        if delta < 1e-12:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            c_all = compute_all_lags(h, n)
                            improved = True
                            stuck_count = 0
                            ss_idx = min(ss_idx + 1, len(step_sizes)-1)
                        else:
                            h[i] -= delta
                            h[j] += delta
        
        if not improved:
            stuck_count += 1
            if stuck_count > 3:
                break
            # Try other top lags
            top_lags = np.argsort(c_all)[-5:]
            for k2 in top_lags:
                if time.time() > deadline:
                    break
                lag2 = k2 - (n-1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for d in [0.001, 0.0005, 0.0002]:
                    for i in range(n - m2):
                        if time.time() > deadline:
                            break
                        j2 = i + m2
                        if h[i] > h[j2] + 1e-12:
                            delta = min(d, (h[i]-h[j2])*0.3)
                            h[i] -= delta
                            h[j2] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h.copy()
                                c_all = compute_all_lags(h, n)
                                improved = True
                            else:
                                h[i] += delta
                                h[j2] -= delta
                        elif h[j2] > h[i] + 1e-12:
                            delta = min(d, (h[j2]-h[i])*0.3)
                            h[i] += delta
                            h[j2] -= delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h.copy()
                                c_all = compute_all_lags(h, n)
                                improved = True
                            else:
                                h[i] -= delta
                                h[j2] += delta
        ss_idx = min(ss_idx + 1, len(step_sizes)-1)
    
    return best_h, best_c

def micro_local_search(h, n, deadline):
    """Very fine-grained search with tiny steps."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    
    epsilons = [1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5]
    
    for eps in epsilons:
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(len(epsilons) and 0 or 0):
            pass
        # Multiple passes at this epsilon
        for _pass in range(3):
            if time.time() > deadline:
                break
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-6:]
            for k_idx in top_k:
                if time.time() > deadline:
                    break
                lag = k_idx - (n-1)
                if lag == 0:
                    continue
                m = abs(lag)
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-12:
                        delta = min(eps, (h[i]-h[j])*0.4)
                        if delta < 1e-14:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved_any = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-12:
                        delta = min(eps, (h[j]-h[i])*0.4)
                        if delta < 1e-14:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved_any = True
                        else:
                            h[i] -= delta
                            h[j] += delta
        if not improved_any:
            break
    
    return best_h, best_c

def random_perturb_search(h, n, deadline, rng, base_noise=0.01):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    while time.time() < deadline:
        noise_level = base_noise * (0.5 + 0.5 * rng.random())
        h_trial = h + rng.normal(0, noise_level, n)
        h_trial = project_feasible(h_trial, n)
        c_new = compute_c5(h_trial, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_trial.copy()
            h = h_trial
            base_noise = max(0.001, noise_level * 0.8)
        else:
            base_noise = max(0.0005, noise_level * 0.9)
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Gradient momentum from initial (25% time)
    if h_init is not None:
        h_work = project_feasible(h_init, n_init)
    else:
        n_init = 500
        h_work = np.full(n_init, 0.5)
    n = len(h_work)
    
    t_phase1_end = t0 + 0.25 * budget_s
    alpha_schedule = [200, 800, 3000, 10000, 30000]
    for alpha in alpha_schedule:
        if time.time() > t_phase1_end:
            break
        seg_deadline = min(t_phase1_end, time.time() + 0.8 * (t_phase1_end - t0))
        h_new, c_new = grad_momentum(h_work, n, alpha, seg_deadline, lr=1.0, momentum=0.88)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_work = h_new
    
    # Phase 2: SLSQP polishing (12% time)
    t_phase2_end = t0 + 0.37 * budget_s
    if time.time() < t_phase2_end:
        for top_m in [3, 5]:
            if time.time() > t_phase2_end - 5:
                break
            h_slsqp = slsqp_optimize(h_work, n, 200000, maxiter=4000, top_m=top_m)
            c_slsqp = compute_c5(h_slsqp, n)
            if c_slsqp < best_c:
                best_c = c_slsqp
                best_h = h_slsqp
            h_work = h_slsqp
    
    # Phase 3: Aggressive local search (35% time)
    t_phase3_end = t0 + 0.72 * budget_s
    if time.time() < t_phase3_end:
        h_ls, c_ls = aggressive_local_search(best_h, n, min(t_phase3_end, deadline - 20))
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
    
    # Phase 4: Micro local search (12% time)
    t_phase4_end = t0 + 0.84 * budget_s
    if time.time() < t_phase4_end:
        h_mls, c_mls = micro_local_search(best_h, n, min(t_phase4_end, deadline - 15))
        if c_mls < best_c:
            best_c = c_mls
            best_h = h_mls
    
    # Phase 5: Random perturbation + gradient restarts (8% time)
    t_phase5_end = t0 + 0.92 * budget_s
    while time.time() < t_phase5_end:
        noise = 0.005 + 0.01 * rng.random()
        h_pert = best_h + rng.normal(0, noise, n)
        h_pert = project_feasible(h_pert, n)
        # Brief gradient descent from perturbed point
        seg_dl = min(t_phase5_end, time.time() + 15)
        h_gs, c_gs = grad_momentum(h_pert, n, 50000, seg_dl, lr=0.5, momentum=0.85)
        if c_gs < best_c - 1e-14:
            best_c = c_gs
            best_h = h_gs
    
    # Phase 6: Final SLSQP + micro polish (remaining time)
    if time.time() < deadline - 5:
        h_final = slsqp_optimize(best_h, n, 500000, maxiter=3000, top_m=4)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    if time.time() < deadline - 1:
        best_h, best_c = micro_local_search(best_h, n, deadline - 0.5)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
