def subgradient_step(h, n, lr):
    """One subgradient step: move to reduce the worst lag."""
    lags = compute_all_lags(h, n)
    max_idx = np.argmax(lags)
    k = max_idx - (n - 1)
    
    # Subgradient of g_k(h) = sum_i h[i](1-h[i+k])*dx
    # dg_k/dh[i] = (1-h[i+k])*dx - h[i-k]*dx (for appropriate indices)
    # More carefully:
    # g_k = sum_i h[i](1-h[i+k]) * dx  (for k >= 0, i = 0..n-1-k)
    # dg_k/dh[j] = dx * (1-h[j+k]) for j = 0..n-1-k (first factor)
    #            - dx * h[j-k] for j = k..n-1 (second factor, when i+k=j)
    #            = dx * (1-h[j+k]) - dx * h[j-k] (with appropriate 0s at boundaries)
    
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            grad[i] += (1 - h[i+k]) * (2.0/n)
            if i + k < n:
                grad[i+k] += h[i] * (-(2.0/n))

            grad[i+abs_k] += (1 - h[i]) * (2.0/n)
            grad[i] += h[i+abs_k] * (-(2.0/n))
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)

    h = h.copy()

    best_h = h.copy()
    best_t = time.time()

    patience = 0

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    result = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return result

        h_init = np.full(200, 0.5)
        n_init = 200
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    # Phase 1: Main optimization at initial resolution
    alloc1 = min(350, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_block(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    # Phase 2: Antipodal start (1-h)
    if time.time() < deadline - 100:
        h_antipodal = 1.0 - best_h
        h_antipodal = project_feasible(h_antipodal, len(best_h))
        alloc2 = min(100, (deadline - time.time()) * 0.15)
        if alloc2 > 30:
            h_opt, c_opt = optimize_block(h_antipodal, len(best_h), time.time() + alloc2, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 3: Multi-resolution
    x_best = np.linspace(0, 2, len(best_h))
    n_cur = len(best_h)
    res_candidates = []
    for d in [-15, -10, -5, -3, -1, 1, 3, 5, 10, 15, 25, 50, 100]:
        nt = n_cur + d
        if 80 <= nt <= 600 and nt != n_cur:
            res_candidates.append(nt)
    for nt in [100, 120, 150, 180, 220, 280, 320, 400, 500]:
        if nt not in res_candidates:
            res_candidates.append(nt)
    res_candidates = sorted(set(res_candidates))
    for nt in res_candidates:
        if time.time() > deadline - 50:
            break
        x_new = np.linspace(0, 2, nt)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, nt)
        alloc = min(70, (deadline - time.time()) * 0.12)
        if alloc < 25:
            break
        h_opt, c_opt = optimize_block(h_interp, nt, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 4: Multi-restart with diverse perturbations
    if time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(4, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            if r % 5 == 0:
                scale = 0.02 + 0.08 * r / max(1, n_restarts)
                hp = best_h + rng.normal(0, scale, n)
            elif r % 5 == 1:
                hp = best_h.copy()
                for _ in range(4):
                    f = rng.randint(1, max(2, n // 4))
                    amp = rng.uniform(0.02, 0.07)
                    ph = rng.uniform(0, 2 * np.pi)
                    hp += amp * np.sin(2.0 * np.pi * f * np.arange(n) / n + ph)
            elif r % 5 == 2:
                hp = best_h.copy()
                bs = rng.randint(max(5, n // 12), max(10, n // 6))
                st = rng.randint(0, max(1, n - bs))
                hp[st:st + bs] = 1.0 - hp[st:st + bs]
            elif r % 5 == 3:
                hp = best_h.copy()
                mask = rng.random(n) < 0.3
                hp[mask] = np.clip(hp[mask] + rng.normal(0, 0.15, mask.sum()), 0, 1)
            else:
                hp = best_h.copy()
                # Swap-based perturbation: swap values at random positions
                perm = rng.permutation(n)
                for _ in range(n // 5):
                    i, j = perm[2 * _], perm[2 * _ + 1]
                    hp[i], hp[j] = hp[j], hp[i]
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 15) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_block(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 5: Intensive final refinement
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        ht, c = targeted_lag_reduction(best_h, n, min(deadline - 5, time.time() + (deadline - time.time()) * 0.4))
        if c < best_c:
            best_c = c
            best_h = ht
    # Phase 6: Final SLSQP + micro local search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = run_slsqp(best_h, n, 10000000, maxiter=4000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 2:
            ht, c = local_pair_search(best_h, n, deadline - 1,
                                       [0.0001, 0.00005, 0.00002], 25)
            if c < best_c:
                best_c = c
                best_h = ht
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
