    """Optimize h using gradient-guided pair moves with exact line search."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for step in range(max_steps):
        c_full = np.correlate(best_h, 1-best_h, mode='full')
        max_idx = np.argmax(c_full)

        # g_k(h) = sum_m h[m] * (1-h[m+k-(n-1)])
        # ∂g_k/∂h[i] = (1-h[i+k-(n-1)]) - h[i-(k-(n-1))]
        #             = (1-h[i+l]) - h[i-l]  where l = max_idx - (n-1)

        grad = np.zeros(n)
        for i in range(n):
            il = i + l
            h_il = best_h[il] if 0 <= il < n else 0.0
            i_neg_l = i - l
            h_il_neg = best_h[i_neg_l] if 0 <= i_neg_l < n else 0.0
            grad[i] = (1 - h_il) - h_il_neg

        i = np.argmin(grad)
        j = np.argmax(grad)
        if i == j:
            break
        if grad[i] >= 0 and grad[j] <= 0:
            break  # No improvement possible
        
        # Exact line search for pair (i, j): h[i] += d, h[j] -= d
        d_opt = exact_pair_d(best_h, i, j, n, c_full)

        vals = c_full + mid * a
        max_val = vals.max()
        close_idx = np.where(np.abs(vals - max_val) < 1e-10)[0]
        if len(close_idx) == 0:
            close_idx = np.array([np.argmax(vals)])
        amin = a[close_idx].min()
        amax = a[close_idx].max()
        if amin > 1e-10:
            hi = mid
        elif amax < -1e-10:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def gradient_pair_search(h, n, deadline, top_k=5, max_rounds=500):
    """Gradient-guided pair search using exact line search."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    n1 = n - 1
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_full = np.correlate(best_h, 1 - best_h, mode='full')
        # Find top-k active constraints
        sorted_idx = np.argsort(c_full)[::-1][:top_k]
        improved = False
        for max_idx in sorted_idx:
            l = max_idx - n1
            # Gradient of g_{max_idx}
            grad = np.zeros(n)
            for i in range(n):
                ip = i + l
                ineg = i - l
                h_ip = best_h[ip] if 0 <= ip < n else 0.0
                h_ineg = best_h[ineg] if 0 <= ineg < n else 0.0
                grad[i] = (1.0 - h_ip) - h_ineg
            # Best pair: most negative grad, most positive grad
            i_cand = np.argmin(grad)
            j_cand = np.argmax(grad)
            if i_cand == j_cand:
                continue
            if grad[i_cand] >= -1e-12 and grad[j_cand] <= 1e-12:
                continue
            B = compute_B_vec(best_h, n)
            d_opt = exact_pair_d(best_h, i_cand, j_cand, n, c_full, B)
            if abs(d_opt) < 1e-14:
                continue
            new_h = best_h.copy()
            new_h[i_cand] += d_opt
            new_h[j_cand] -= d_opt
            new_h = np.clip(new_h, 0, 1)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-15:
                best_h = new_h
                best_c = new_c
                improved = True
                c_full = np.correlate(best_h, 1 - best_h, mode='full')
        if not improved:
            break
    return best_h, best_c

def random_pair_search(h, n, deadline, rng, n_max_pairs=50000):
    """Random pair moves with exact line search (SA-like)."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    T0 = 0.002
    T_end = 0.00001
    count = 0
    for it in range(n_max_pairs):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (float(it) / n_max_pairs)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        c_full = np.correlate(best_h, 1 - best_h, mode='full')
        B = compute_B_vec(best_h, n)
        d_opt = exact_pair_d(best_h, i, j, n, c_full, B)
        if abs(d_opt) < 1e-14:
            continue
        new_h = best_h.copy()
        new_h[i] += d_opt
        new_h[j] -= d_opt
        new_h = np.clip(new_h, 0, 1)
        new_c = compute_c5(new_h, n)
        delta = new_c - best_c
        if delta < 0 or (T > 0 and np.log(rng.uniform(1e-15, 1.0)) < -delta / T):
            if new_c < best_c:
                best_c = new_c
                best_h = new_h
    return best_h, best_c
