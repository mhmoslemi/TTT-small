import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    """Gradient of c_k with respect to h, where k is the actual shift (not correlate index)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)  # momentum
    lr_curr = lr
    
    while time.time() < deadline:

def linprog_active_set(h, n, top_m=20):
    """Solve one LP iteration of the linearized active-set method."""
    c = compute_c5_vec(h, n)
    idx = np.argpartition(c, -top_m)[-top_m:]
    
    # Build matrix for: minimize t
    # subject to: g_k · h + b_k ≤ t for each k in active set
    #             0 ≤ h[i] ≤ 1
    #             sum(h) = n/2
    
    # Variables: [h[0], ..., h[n-1], t]
    # Objective: [0, ..., 0, 1]
    
    # Constraints for each active k:
    #   ∇c_k · h ≤ t - c_k(h)  =>  ∇c_k · h - t ≤ -c_k(h)
    
    m_len = len(idx)
    A_ub = np.zeros((m_len, n + 1))
    b_ub = np.zeros(m_len)
    
    for i, m in enumerate(idx):
        k = m - (n - 1)
        g = shift_gradient(h, n, k)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -c[m]
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n]
    return None

So: δc_k = δ * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]] * dx
         = δ * [h[j+k] - h[i+k] - h[i-k] + h[j-k]] * dx  (with appropriate zero terms)

This is linear in δ! So for each k, c_k changes linearly with δ. The max of linear functions is convex in δ. I can find the δ that minimizes max_k c_k by finding the point where the top two c_k values are equal (or at a boundary).

This gives an exact line search for each pair! Let me implement this.

```python
def pair_line_search(h, n, i, j, c_old):
    """Find optimal δ for transferring mass from j to i."""
    dx = 2.0 / n
    
    # Compute α_k for all k
    # δc_k = δ * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]] * dx
    # (with terms = 0 if index out of range)
    
    n_shifts = 2 * n - 1
    alpha = np.zeros(n_shifts)
    for mi in range(n_shifts):
        k = mi - (n - 1)
        term = 0.0
        # (1-h[i+k]) if 0 ≤ i+k < n
        if 0 <= i + k < n:
            term += (1.0 - h[i+k])
        # -h[i-k] if 0 ≤ i-k < n
        if 0 <= i - k < n:
            term -= h[i-k]
        # -(1-h[j+k]) if 0 ≤ j+k < n
        if 0 <= j + k < n:
            term -= (1.0 - h[j+k])
        # +h[j-k] if 0 ≤ j-k < n
        if 0 <= j - k < n:
            term += h[j-k]
        alpha[mi] = term * dx
    
    # f(δ) = max_k (c_old[k] + alpha[k] * δ)
    # Minimize over δ ∈ [h[j]-1+... , 1-h[j]+...] Wait:
    # h[i] + δ ∈ [0, 1] => δ ∈ [-h[i], 1-h[i]]
    # h[j] - δ ∈ [0, 1] => δ ∈ [h[j]-1, h[j]]
    # Combined: δ ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]
    
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    
    if delta_max <= delta_min:
        return 0.0
    
    # Convex piecewise linear minimization
    # Try a few strategies:
    # 1. Evaluate at boundaries and at "crossing" points
    # 2. Use bisection on the subgradient
    
    # Bisection on subgradient:
    # At δ, the subgradient is the set of α[k] for k that achieve the max
    # If min subgradient > 0, optimal is to the left
    # If max subgradient < 0, optimal is to the right
    # Otherwise, δ is optimal
    
    lo = delta_min
    hi = delta_max
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = c_old + alpha * mid
        imax = vals.argmax()
        # Subgradient at mid: all α[k] where c_old[k] + alpha[k]*mid == vals[max]
        # Use tolerance
        active = np.isclose(vals, vals.max(), atol=1e-10)
        if active.any():
            sg_min = alpha[active].min()
            sg_max = alpha[active].max()
            if sg_min > 1e-10:
                hi = mid
            elif sg_max < -1e-10:
                lo = mid
            else:
                break
        else:
            break
    
    delta_opt = (lo + hi) / 2
    
    # Verify
    f_lo = np.max(c_old + alpha * delta_min)
    f_hi = np.max(c_old + alpha * delta_max)
    f_mid = np.max(c_old + alpha * delta_opt)
    
    if f_mid <= f_lo and f_mid <= f_hi:
        return delta_opt
    elif f_lo <= f_hi:
        return delta_min
    else:
        return delta_max
```

    n_shifts = 2 * n - 1
    G = np.zeros((n, n_shifts))

        # First term: (1-h[p+k]) for k where 0 ≤ p+k < n

        k_range1 = np.arange(-p, n - p)

        vals2 = h[p - k_range2]

    return G
