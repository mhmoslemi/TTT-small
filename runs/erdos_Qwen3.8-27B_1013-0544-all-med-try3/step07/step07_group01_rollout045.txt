        if abs(diff) < 1e-14:

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def _obj_l(b, n, alpha, mu):
    c = np.correlate(b, 1 - b, mode='full') * (2.0 / n)
    val = smooth_max_val(c, alpha)
    val += mu * (b.sum() - n / 2.0) ** 2
    return float(val)

def _grad_l(b, n, alpha, mu):
    g = grad_smooth_max(b, n, alpha)
    g += 2.0 * mu * (b.sum() - n / 2.0)
    return g

    c = np.correlate(b, 1 - b, mode='full') * (2.0 / n)
    return float(smooth_max_val(c, alpha))

def _cons_s(b, n):
    return float(b.sum() - n / 2.0)

    return np.ones(n)

                            best_h = h.copy()
                            improved = True
                            g[i] = 0.0  # recompute locally
                        else:
                            h[i] -= move
                            h[j] += move
                if not improved:
                    break
                passes += 1
                h = best_h.copy()
    return best_h, best_c

def random_restart(h, n, deadline, rng, n_restarts=20):
    best_c = compute_c5(h)
    best_h = h.copy()
    for _ in range(n_restarts):
        if time.time() > deadline - 30:
            break
        h_pert = h.copy()
        # Small local perturbation
        n_pert = rng.randint(5, 20)
        idx = rng.choice(n, size=n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.02, 0.02, size=n_pert)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert.copy()
        # Try a quick local search from the perturbation
        t_rem = deadline - time.time() - 5
        if t_rem > 20:
            h_local, c_local = pair_search(h_pert, n, time.time() + min(15, t_rem * 0.5),
                                           [0.005, 0.002, 0.001, 0.0005], radius=30)
            if c_local < best_c:
                best_c = c_local
                best_h = h_local
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h)
    best_n = n_init

    # Phase 1: L-BFGS-B optimization on initial resolution (180s)
    t1_deadline = min(t0 + 180, deadline - 500)
    best_h, best_c = lbfgs_opt(best_h, best_n, t1_deadline, max_iters_per_alpha=150)
    best_n = len(best_h)

    # Phase 2: SLSQP refinement (80s)
    if time.time() < deadline - 400:
        t2_deadline = min(time.time() + 80, deadline - 400)
        for a in [100000, 500000, 2000000, 5000000]:
            if time.time() > t2_deadline - 5:
                break
            h2, c2 = slsqp_opt(best_h, best_n, min(time.time() + 25, t2_deadline),
                              alpha=a, maxiter=1000)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = len(best_h)

    # Phase 3: Gradient-guided pair search (120s)
    if time.time() < deadline - 300:
        t3_deadline = min(time.time() + 120, deadline - 300)
        best_h, best_c = gradient_pair(best_h, best_n, t3_deadline,
                                       [0.01, 0.005, 0.002, 0.001, 0.0005])
        best_n = len(best_h)

    # Phase 4: Try higher resolution (n=400, 500) (200s)
    if time.time() < deadline - 250:
        for n_try in [400, 500]:
            if time.time() > deadline - 80:
                break
            h_try = upsample(best_h, best_n, n_try)
            alloc = min(80, deadline - time.time() - 60)
            if alloc < 30:
                break
            # L-BFGS on new resolution
            h_try, c_try = lbfgs_opt(h_try, n_try, min(time.time() + alloc * 0.5, deadline - 60),
                                     max_iters_per_alpha=100)
            # Quick SLSQP
            if time.time() < deadline - 40:
                h_try, c_try = slsqp_opt(h_try, n_try, min(time.time() + 20, deadline - 40),
                                        alpha=2000000, maxiter=800)
            # Quick pair search
            if time.time() < deadline - 20:
                h_try, c_try = pair_search(h_try, n_try, min(time.time() + 15, deadline - 20),
                                          [0.005, 0.002, 0.001, 0.0005], radius=40)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
                best_n = n_try

    # Phase 5: Aggressive local pair search (150s)
    if time.time() < deadline - 80:
        t5_deadline = min(time.time() + 150, deadline - 80)
        nf = len(best_h)
        best_h, best_c = pair_search(best_h, nf, t5_deadline,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                     radius=50)
        best_n = len(best_h)

    # Phase 6: Random restarts (80s)
    if time.time() < deadline - 40:
        t6_deadline = min(time.time() + 80, deadline - 40)
        nf = len(best_h)
        best_h, best_c = random_restart(best_h, nf, t6_deadline, rng, n_restarts=15)
        best_n = len(best_h)

    # Phase 7: Final L-BFGS + SLSQP on best (60s)
    if time.time() < deadline - 15:
        nf = len(best_h)
        t7_deadline = min(time.time() + 40, deadline - 15)
        best_h, best_c = lbfgs_opt(best_h, nf, t7_deadline, max_iters_per_alpha=80)
        best_n = len(best_h)
        if time.time() < deadline - 8:
            best_h, best_c = slsqp_opt(best_h, best_n, deadline - 8, alpha=5000000, maxiter=500)
            best_n = len(best_h)

    # Phase 8: Ultra-fine pair search on the final solution
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = pair_search(best_h, nf, deadline - 1,
                                     [0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                     radius=30)

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
