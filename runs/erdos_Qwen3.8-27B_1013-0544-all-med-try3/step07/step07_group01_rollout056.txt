    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):

            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _eq_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _eq_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alphas, max_iters=150):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for idx, (alpha, mu) in enumerate(alphas):
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-15, 'maxfun': max_iters * 8,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)

            if c_new < best_c - 1e-16:

                h = h_new

    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=1000000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jac, 'args': (n,)}]
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})

        c_new = compute_c5(h_new, n)
        return h_new, c_new

                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
        if not improved:
            delta *= 0.5
            alpha *= 2.0
            if delta < 0.00005:
                break
    return h, best_c

def exhaustive_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline - 2:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def fine_coord_search(h, n, deadline, steps):
    best_c = compute_c5(h, n)
    alpha = 2000.0
    for step in steps:
        improved_outer = True
        while improved_outer and time.time() < deadline - 2:
            improved_outer = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return h, best_c
                for direction in [step, -step]:
                    if h[i] + direction < 0 or h[i] + direction > 1:
                        continue
                    h[i] += direction
                    h_new = project_feasible(h, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        h = h_new
                        improved_outer = True
                    else:
                        h[i] -= direction
        alpha *= 2.0
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400

    # Interpolate to n=400 for more degrees of freedom
    n_target = 400
    if n_init != n_target:
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_target)
        h_init = np.interp(x_new, x_old, h_init)

    best_h = project_feasible(h_init, n_target)
    best_c = compute_c5(best_h, n_target)
    best_n = n_target

    # Phase 1: L-BFGS-B with progressive alpha (main optimization)
    alphas_p1 = [
        (50, 5.0), (200, 10.0), (1000, 20.0),
        (5000, 50.0), (50000, 100.0), (200000, 500.0),
        (1000000, 2000.0)
    ]
    t1 = min(t0 + 280, deadline - 420)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1, alphas_p1, max_iters=180)

    # Phase 2: Higher alpha L-BFGS for sharper max
    if time.time() < deadline - 350:
        alphas_p2 = [(5000000, 5000.0), (20000000, 10000.0)]
        t2 = min(time.time() + 80, deadline - 350)
        best_h, best_c = lbfgs_optimize(best_h, best_n, t2, alphas_p2, max_iters=120)

    # Phase 3: SLSQP refinement
    if time.time() < deadline - 300:
        for a in [2000000, 10000000]:
            if time.time() > deadline - 300:
                break
            t3 = min(time.time() + 50, deadline - 300)
            h3, c3 = slsqp_optimize(best_h, best_n, t3, alpha=a, maxiter=800)
            if c3 < best_c:
                best_c = c3
                best_h = h3

    # Phase 4: Gradient descent (fast, many iterations)
    if time.time() < deadline - 250:
        t4 = min(time.time() + 100, deadline - 250)
        h4, c4 = gradient_descent_phase(best_h, best_n, t4, lr_init=0.01, alpha_init=500.0)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Smart gradient-guided local search (key improvement)
    if time.time() < deadline - 180:
        t5 = min(time.time() + 150, deadline - 180)
        best_h, best_c = smart_local_search(best_h, best_n, t5)

    # Phase 6: More gradient descent after local search
    if time.time() < deadline - 130:
        t6 = min(time.time() + 50, deadline - 130)
        h6, c6 = gradient_descent_phase(best_h, best_n, t6, lr_init=0.005, alpha_init=2000.0)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Exhaustive pair search with small steps
    if time.time() < deadline - 80:
        t7 = min(time.time() + 60, deadline - 80)
        best_h, best_c = exhaustive_pair_search(best_h, best_n, t7,
                                                 [0.001, 0.0005, 0.0002, 0.0001], radius=35)

    # Phase 8: Try n=500 variant
    if time.time() < deadline - 100:
        n_alt = 500
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_alt)
        h_alt = np.interp(x_new, x_old, best_h)
        h_alt = project_feasible(h_alt, n_alt)
        alloc = min(80, deadline - time.time() - 20)
        if alloc > 35:
            alts = [(1000, 20.0), (10000, 50.0), (100000, 200.0),
                    (1000000, 1000.0), (5000000, 5000.0)]
            h_alt, c_alt = lbfgs_optimize(h_alt, n_alt, time.time() + alloc * 0.6, alts, max_iters=100)
            if time.time() < deadline - 15 and c_alt < best_c + 0.002:
                h_alt, c_alt = slsqp_optimize(h_alt, n_alt, time.time() + min(25, deadline - time.time() - 10),
                                               alpha=5000000, maxiter=500)
            if c_alt < best_c:
                best_c = c_alt
                best_h = h_alt
                best_n = n_alt

    # Phase 9: Final polish - smart local + fine coord
    if time.time() < deadline - 10:
        t9a = min(time.time() + 30, deadline - 5)
        best_h, best_c = smart_local_search(best_h, best_n, t9a)
    if time.time() < deadline - 3:
        t9b = min(time.time() + 20, deadline - 2)
        best_h, best_c = fine_coord_search(best_h, best_n, t9b,
                                            [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
