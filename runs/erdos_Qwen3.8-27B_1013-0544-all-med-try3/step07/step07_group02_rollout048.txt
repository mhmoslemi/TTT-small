        active = cc[top_idx]

        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]

        grad = np.zeros(n)

            if abs(cc[idx] - max_val) < 1e-8:
                lag = idx - (n - 1)
                grad += lag_gradient(x, n, lag)
        grad /= np.sum([1 for idx in top_idx if abs(cc[idx] - max_val) < 1e-8])
        return grad * dx

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    return w

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

    c = np.correlate(h, 1 - h, mode='full') * dx

    patience = 0
    max_patience = 800
    
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 3000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        current_lr = lr * (0.92 ** (it // 400))
        
        g = grad_smooth_max(h_curr, n, alpha)
        v = momentum * v + current_lr * g
        
        # Limit step size

        active = cc[top_idx]

        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]

        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)

            w = np.exp(shifted)

        return grad * dx
    
    bounds = [(0.0, 1.0)] * n

    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
