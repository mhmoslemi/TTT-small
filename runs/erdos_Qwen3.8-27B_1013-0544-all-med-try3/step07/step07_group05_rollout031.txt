def binarize(h, n):
    h = h.copy()
    # Sort by distance from 0.5
    dist = np.abs(h - 0.5)
    order = np.argsort(dist)
    # Start with the most "decided" values
    # Try to set them to 0 or 1
    target_sum = n / 2.0
    # Greedily assign
    h_new = np.zeros(n)
    assigned = np.zeros(n, dtype=bool)
    
    # First pass: assign values far from 0.5
    for idx in order:
        if h[idx] > 0.5:
            h_new[idx] = 1.0
        else:
            h_new[idx] = 0.0
        assigned[idx] = True
        if h_new.sum() >= target_sum:

    # Fill remaining with fractional values
    remaining = target_sum - h_new.sum()
    unassigned = np.where(~assigned)[0]
    if len(unassigned) > 0 and remaining > 0:
        h_new[unassigned[0]] = remaining  # put all remainder in one spot
    
    h_new = project_feasible(h_new, n)

def compute_c5(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact projection
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-14:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights = weights / weights.sum()
    # Gradient of log-sum-exp wrt c is weights
    # Gradient of c[k] wrt x[i]:
    # c[k] = sum_j x[j] * (1-x[j+k-n+1]) * dx
    # dc[k]/dx[i] = (1-x[i+k-n+1])*dx - x[i+k-n+1+1]*dx ... (boundary aware)
    # Actually dc[k]/dx[i] = (1-x[i+k-n+1])*dx for the first term
    #                         + x[j]*(-(1)*(1))*dx for the second term where j+k-n+1=i
    # = (1-x[i+k-n+1])*dx - x[i-k+n-1]*dx
    # where terms are 0 if index out of range
    grad_c = np.zeros(n)
    for i in range(n):
        # First term: (1-x[i+k-n+1]) for valid k, i.e., i+k-n+1 in [0,n-1]
        # k ranges from n-1-i to n-1+i, clamped to [0, 2n-2]
        k_lo = max(0, n - 1 - i)
        k_hi = min(2 * n - 2, n - 1 + i)
        for k in range(k_lo, k_hi + 1):
            idx2 = i + k - n + 1
            if 0 <= idx2 < n:
                grad_c[i] += weights[k] * (1 - x[idx2]) * dx
        # Second term: -x[i'] where i'+k-n+1 = i, i.e., i' = i - k + n - 1
        for k in range(k_lo, k_hi + 1):
            idx1 = i - k + n - 1
            if 0 <= idx1 < n:
                grad_c[i] -= weights[k] * x[idx1] * dx
    return grad_c

def fast_gradient(x, n, alpha):
    """Fast gradient using the full correlation vector."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    # grad[k] = sum_i w[k] * dc[k]/dx[i]
    # dc[k]/dx[i] = dx * [(1 - x[i+k-n+1]) - x[i-k+n-1]] (with boundary=0)
    # So grad[i] = dx * sum_k w[k] * [(1 - x[i+k-n+1]) - x[i-k+n-1]]
    # = dx * sum_k w[k] * (1 - x[i+k-n+1]) - dx * sum_k w[k] * x[i-k+n-1]
    # First sum: for fixed i, sum over k where i+k-n+1 in [0,n-1]
    # Let a = i+k-n+1, so k = a - i + n - 1, a in [0, n-1]
    # First sum = sum_a w[a-i+n-1] * (1-x[a]) for a-i+n-1 in [0, 2n-2]
    # Second sum: for fixed i, sum over k where i-k+n-1 in [0,n-1]
    # Let b = i-k+n-1, so k = i-b+n-1, b in [0, n-1]
    # Second sum = sum_b w[i-b+n-1] * x[b] for i-b+n-1 in [0, 2n-2]
    #
    # First sum = sum_a x[a] * w[a-i+n-1] (with w out of range = 0)
    # This is a flipped cross-correlation of x and w!
    # Actually: sum_a x[a] * w[a + (n-1-i)] = (x * w_flipped)[i] type thing
    # Let me use: sum_a x[a] * w[a + offset] where offset = n-1-i
    # This is (x correlate w) at lag (n-1-i) ... let me just use np.correlate
    
    # First term: for each i, sum_a (1-x[a]) * w[a - i + n - 1]
    # = sum_a (1-x[a]) * w[(n-1) - (i-a)]
    # Let u = i-a, then = sum_u (1-x[i-u]) * w[n-1-u]
    # This is the cross-correlation of (1-x) and w at lag... 
    # Actually np.correlate(a, v)[k] = sum_i a[i] * v[i+k-len(v)+1]
    # Let me just compute it directly with the full correlation
    # 
    # f1[i] = sum_a (1-x[a]) * w[a - i + n - 1] for valid indices
    # = sum_a (1-x[a]) * w[n-1 + (a-i)]
    # Let d = a-i, then f1[i] = sum_d (1-x[i+d]) * w[n-1+d]
    # This is the cross-correlation of (1-x) and w (reversed) at lag i
    
    # Use np.correlate: np.correlate(a, v)[k] = sum_i a[i+k] * v[i]
    # I want: sum_d (1-x[i+d]) * w_rev[n-1-d] where w_rev[j] = w[len-1-j]
    # Hmm, let me just do it with two correlates
    
    # f1[i] = sum_d (1-x[i+d]) * w[n-1+d]
    # Let a[d] = (1-x[d]) and b[d] = w[n-1+d] (with b=0 out of range)
    # f1[i] = sum_d a[i+d] * b[n-1-d] ... this is getting confusing
    
    # Simpler: just compute the gradient numerically or use the direct formula
    # Actually, the easiest is:
    # grad[i] = dx * (sum_k w[k] * (1-x[i+k-n+1]) - sum_k w[k] * x[i-k+n-1])
    # First part: sum_k w[k] * (1-x[i+k-n+1]) 
    #   = (1-x) correlated with w at lag (n-1-i)
    #   Let's define: p1[i] = sum_k w[k] * (1-x[i+k-n+1])
    #   = sum_k w[k] * (1-x)[i+k-n+1]
    #   This is the cross-correlation of (1-x) and w at a specific lag
    #   np.correlate(1-x, w, 'full')[j] = sum_k (1-x)[k] * w[k+j-len(w)+1]
    #   = sum_k (1-x)[k] * w[k+j-2n+3]  ... ugh
    
    # Let me just use a loop-free approach with the correlation
    # p1[i] = sum_k w[k] * (1-x[i+k-n+1])
    # Let m = n-1, then p1[i] = sum_k w[k] * (1-x[i+k-m])
    # = sum_k w[k] * (1-x)[i+k-m]
    # Define u[k] = w[k], v[j] = (1-x)[j]
    # p1[i] = sum_k u[k] * v[i+k-m]
    # This is (v * u_flipped)[i-m+2n-2] in full correlation notation
    # np.correlate(v, u, 'full')[r] = sum_k v[k] * u[k+r-n_u+1] where n_u = len(u) = 2n-1
    # = sum_k v[k] * u[k+r-2n+2]
    # We want: sum_k u[k] * v[i+k-m] = sum_k u[k] * v[k+(i-m)]
    # Let's set r-2n+2 = i-m, so r = i-m+2n-2 = i+n-1
    # p1[i] = np.correlate(1-x, w, 'full')[i+n-1]
    
    # p2[i] = sum_k w[k] * x[i-k+n-1]
    # = sum_k u[k] * x[k+(n-1-i)]
    # Same form: p2[i] = np.correlate(x, w, 'full')[n-1-i+n-1] = np.correlate(x, w, 'full')[2n-2-i]
    # Wait let me redo: p2[i] = sum_k w[k] * x[i-k+n-1]
    # Let j = i-k+n-1, so k = i-j+n-1
    # = sum_j w[i-j+n-1] * x[j]
    # = sum_j x[j] * w[i+n-1-j]
    # np.correlate(x, w, 'full')[r] = sum_j x[j] * w[j+r-(2n-2)]
    # We want w[i+n-1-j] = w[j + (i+n-1) - 2j]... no this doesn't work cleanly.
    
    # Let me just be explicit:
    # np.correlate(a, v, 'full')[k] = sum_i a[i] * v[i + k - (len(v)-1)]
    # For a=x (len n), v=w (len 2n-1):
    # np.correlate(x, w, 'full')[k] = sum_i x[i] * w[i + k - (2n-2)]
    
    # p1[i] = sum_k w[k] * (1-x[i+k-n+1])
    # Let's flip: = sum_k w[k] * (1-x)[i+k-n+1]
    # np.correlate(1-x, w, 'full')[k'] = sum_j (1-x)[j] * w[j + k' - (2n-2)]
    # I want: sum_k w[k] * (1-x)[i+k-n+1]
    # Let j = i+k-n+1, so k = j-i+n-1
    # = sum_j (1-x)[j] * w[j-i+n-1]
    # = sum_j (1-x)[j] * w[j + (n-1-i)]
    # From the formula: sum_j (1-x)[j] * w[j + k' - (2n-2)] where k' - (2n-2) = n-1-i
    # So k' = 3n-3-i... that's out of range for 'full' mode (max is 2n-2+n-1 = 3n-3)
    # Actually 'full' mode has 2n-1 + 2n-1 - 1 = 4n-3 values? No.
    # For a of length n_a and v of length n_v, 'full' has n_a + n_v - 1 values.
    # n_a = n, n_v = 2n-1, so full length = n + 2n-1 - 1 = 3n-2.
    # k' ranges from 0 to 3n-3.
    # k' = 3n-3-i, which is in [0, 3n-3] for i in [0, n-1]. Good.
    
    # p2[i] = sum_k w[k] * x[i-k+n-1]
    # Let j = i-k+n-1, so k = i-j+n-1
    # = sum_j x[j] * w[i-j+n-1]
    # np.correlate(x, w, 'full')[k''] = sum_j x[j] * w[j + k'' - (2n-2)]
    # I need w[i-j+n-1] = w[j + k'' - (2n-2)]
    # So i-j+n-1 = j + k'' - 2n + 2
    # 2j = i + n - 1 - k'' + 2n - 2 = i + 3n - 3 - k''
    # This doesn't factor nicely because of the 2j.
    
    # Alternative: flip w
    # w_flip[k] = w[2n-2-k]
    # p2[i] = sum_k w[k] * x[i-k+n-1]
    # Let m = 2n-2-k, so k = 2n-2-m
    # = sum_m w[2n-2-m] * x[i-2n+2+m+n-1]
    # = sum_m w_flip[m] * x[i+m-n+1]
    # np.correlate(x, w_flip, 'full')[k'''] = sum_j x[j] * w_flip[j + k''' - (2n-2)]
    # I want: sum_m w_flip[m] * x[i+m-n+1]
    # Let j = i+m-n+1, so m = j-i+n-1
    # = sum_j x[j] * w_flip[j-i+n-1]
    # From formula: w_flip[j + k''' - (2n-2)] = w_flip[j - i + n - 1]
    # k''' - 2n + 2 = -i + n - 1, so k''' = n - 1 - i + 2n - 2 = 3n - 3 - i
    # Same as p1!
    
    # So: p1[i] = np.correlate(1-x, w, 'full')[3n-3-i]
    #     p2[i] = np.correlate(x, w_flip, 'full')[3n-3-i]
    
    w_flip = w[::-1]
    c1 = np.correlate(1 - x, w, mode='full')
    c2 = np.correlate(x, w_flip, mode='full')
    idx = 3 * n - 3 - np.arange(n)
    p1 = c1[idx]
    p2 = c2[idx]
    return dx * (p1 - p2)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda: np.ones(n)}]
    try:
        res = minimize(smooth_max_obj, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bbounds(n), constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < smooth_max_obj(h, n, alpha):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def bbounds(n):
    return [(0.0, 1.0)] * n

def pair_search(h, n, deadline, deltas, max_passes=100):
    h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i], h[j] = hi2 + d, hj2 - d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(80):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def targeted_pair_search(h, n, deadline, deltas):
    """Focus on pairs involving coordinates at the max correlation."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                return h, best_c
            improved = False
            # Find which lags are at the max
            c_vec = np.correlate(h, 1 - h, mode='full') * dx
            max_val = c_vec.max()
            active_lags = np.where(c_vec >= max_val - 1e-10)[0]
            # For each active lag, find the contributing coordinates
            # and try to move mass away
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    if abs(i - j) < 2 or abs(i - j) > n - 1:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    if not improved and c_new >= best_c:
                        # Try reverse
                        h[i], h[j] = h[i] - d if h[i] >= d else h[i], h[j] + d if h[j] <= 1-d else h[j]
                        break
            if not improved:
                break
    return h, best_c

def gradient_descent(h, n, deadline, lr_init=0.01, lr_end=0.0001, n_iter=None):
    """Subgradient descent on the smooth max objective."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if n_iter is None:
        n_iter = 5000
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = it / n_iter
        lr = lr_init * (lr_end / lr_init) ** t
        # Compute gradient of smooth max
        alpha = 50000
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w = w / w.sum()
        # Gradient
        w_flip = w[::-1]
        c1 = np.correlate(1 - h, w, mode='full')
        c2 = np.correlate(h, w_flip, mode='full')
        idx = 3 * n - 3 - np.arange(n)
        grad = dx * (c1[idx] - c2[idx])
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline, rng, do_pair_search=True):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=4000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    
    # Phase 2: Gradient descent
    if time.time() < deadline - 30:
        ht, ct = gradient_descent(best_h, n, time.time() + min(60, (deadline - time.time() - 30) * 0.5),
                                  lr_init=0.02, lr_end=0.0001, n_iter=3000)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
    
    # Phase 3: SLSQP polish
    if time.time() < deadline - 20:
        ht = slsqp_optimize(best_h, n, 5000000, maxiter=5000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
    
    # Phase 4: Pair search
    if do_pair_search and time.time() < deadline - 5:
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002], max_passes=200)
        elif n <= 350:
            best_h, best_c = near_pair_search(best_h, n, deadline - 2,
                                              [0.005, 0.002, 0.001, 0.0005], radius=40)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 2,
                                              [0.005, 0.002, 0.001], radius=30)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    
    # Phase A: Optimize at original resolution
    alloc_a = min(250, (deadline - t0) * 0.30)
    h_a, c_a = optimize_at_resolution(h_init, n_init, t0 + alloc_a, rng)
    if c_a < best_c:
        best_c, best_h = c_a, h_a
    
    # Phase B: Try intermediate resolutions
    resolutions = [400, 500]
    for n_try in resolutions:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc_b = min(250, (deadline - time.time()) * 0.30)
        h_b, c_b = optimize_at_resolution(h_interp, n_try, time.time() + alloc_b, rng,
                                          do_pair_search=(n_try <= 400))
        if c_b < best_c:
            best_c, best_h = c_b, h_b
    
    # Phase C: Try a different resolution for diversity
    if time.time() < deadline - 60:
        n_c = 600
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_c)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_c)
        alloc_c = min(150, (deadline - time.time()) * 0.25)
        h_c, c_c = optimize_at_resolution(h_interp, n_c, time.time() + alloc_c, rng,
                                          do_pair_search=False)
        if c_c < best_c:
            best_c, best_h = c_c, h_c
    
    # Phase D: Random restarts from best solution
    if time.time() < deadline - 40:
        n_best = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(2, int(remaining / 60))
        scales = [0.003, 0.008, 0.02, 0.05, 0.1]
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 20) * 0.4)
            if alloc < 15:
                break
            h_r, c_r = optimize_at_resolution(hp, n_best, time.time() + alloc, rng,
                                              do_pair_search=(n_best <= 300))
            if c_r < best_c:
                best_c, best_h = c_r, h_r
    
    # Phase E: Final fine polishing
    n_final = len(best_h)
    if time.time() < deadline - 3:
        if n_final <= 300:
            best_h, best_c = near_pair_search(best_h, n_final, deadline - 1,
                                              [0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=60)
        else:
            ht = slsqp_optimize(best_h, n_final, 20000000, maxiter=8000, ftol=1e-16)
            ct = compute_c5(ht, n_final)
            if ct < best_c:
                best_c, best_h = ct, ht.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
