import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 300:
        e = np.exp(shifted - mx)
        return float(m + (mx + np.log(np.sum(e))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(shifted))))  # /alpha already applied
    # Actually let me redo this properly
    # log-sum-exp: m + log(sum(exp(alpha*(c-m))))/alpha
    # = m + (max(shifted) + log(sum(exp(shifted - max(shifted)))))/alpha

def smooth_max_val2(c, alpha):

    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 300:

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 300:
        e = np.exp(shifted - mx)
    else:
        e = np.exp(shifted)
    w = e / e.sum()
    return w

def lag_gradient_raw(h, n, lag):
    """Gradient of correlate component (without dx factor)."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_max_gradient(h, n, c_vals, alpha):
    """Compute gradient of smooth max of all lags."""
    dx = 2.0 / n
    w = smooth_max_weights(c_vals, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        grad += w[k] * dx * lag_gradient_raw(h, n, lag)
    return grad

def smooth_max_gradient_active(h, n, c_vals, alpha, active_idx):
    """Compute gradient of smooth max over active lags only."""
    dx = 2.0 / n
    c_act = c_vals[active_idx]
    m = c_act.max()
    shifted = alpha * (c_act - m)
    mx = shifted.max()
    if mx > 300:
        e = np.exp(shifted - mx)
    else:
        e = np.exp(shifted)
    w = e / e.sum()
    grad = np.zeros(n)
    for i, k in enumerate(active_idx):
        if w[i] < 1e-15:
            continue
        lag = k - (n - 1)
        grad += w[i] * dx * lag_gradient_raw(h, n, lag)
    return grad

def fista_optimize(h0, n, alpha, max_iter, deadline):
    """FISTA (accelerated projected gradient) on smooth max."""
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Lipschitz constant estimate: bound on ||Hessian||
    # For smooth max, L ≈ alpha * max_w sum_k w_k * ||A_k|| 
    # Each ||A_k|| is at most 2*dx*(something)... let's just use a safe bound
    # Actually for the smooth max, the Lipschitz constant of the gradient is bounded by
    # alpha * max_k (sum of |Hessian of C_k|) which is roughly alpha * 4 * dx * (n-1) / (2n-1)
    # Let's use L = alpha * 4.0 as a rough bound and adjust
    
    L = alpha * 4.0
    step = 1.0 / L
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vals = compute_all_lags(h, n)
        f_val = smooth_max_val2(c_vals, alpha)
        g = smooth_max_gradient(h, n, c_vals, alpha)
        
        h_prev = h.copy()
        h = h - step * g
        h = project_feasible(h, n)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = h + ((t - 1) / t_new) * (h - h_prev)
        y = project_feasible(y, n)
        h = y
        t = t_new
        
        c_curr = compute_c5(h, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_smooth_obj(x, n, alpha, active_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_act = c[active_idx]
    return smooth_max_val2(c_act, alpha)

def slsqp_sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def slsqp_sum_jac(x, n):
    return np.ones_like(x)

def slsqp_active_optimize(h, n, active_idx, alpha, maxiter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_sum_con, 'jac': slsqp_sum_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_smooth_obj, h, args=(n, alpha, active_idx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_lag_reduction(h, n, deadline, max_rounds=50):
    """Targeted reduction of worst lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-5:]
        worst_k = top_idx[-1]
        worst_lag = worst_k - (n - 1)
        
        # Compute gradient of worst lag
        g = dx * lag_gradient_raw(h, n, worst_lag)
        
        # Try to move in negative gradient direction
        # But also check other top lags don't increase too much
        for trial_step in [0.001, 0.002, 0.005, 0.01]:
            h_trial = h - trial_step * g
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial, n)
            if c_new < best_c - 1e-14:
                h = h_trial
                best_c = c_new
                break
    
    return h, best_c

def local_search_coarse(h, n, deadline, max_evals=50000):

    h = h.copy()

            k_star = int(np.argmax(c))

            if m >= n:
                break

                if lag > 0:

                    # If we decrease h[j] and increase h[i] where j=i+m:
                    # h[j] decreases → reduces terms where j appears as h[i'+m]
                    # h[i] increases → reduces (1-h[i]) term
                    # Net effect depends on context
                    if h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.3, h[j] * 0.8, (1.0 - h[i]) * 0.8)
                        if delta < 1e-10:

                        h[i] += delta

                        c_new = compute_c5(h, n)

                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                else:
                    # lag < 0, m = -lag
                    # C = dx * sum h[i]*(1-h[i+m]) for i=0..n-1-m
                    # (from the negative lag formula in correlate)
                    if h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.8, (1.0 - h[j]) * 0.8)

                            h[j] -= delta
                            h[i] += delta
    return h, best_c

def binary_round_and_search(h, n, deadline):
    """Round to binary, then do local search on binary."""
    h_bin = (h > 0.5).astype(float)
    # Fix sum constraint
    s = h_bin.sum()
    target = n / 2.0
    if s > target:
        # Need to flip (s - target) ones to zeros
        ones = np.where(h_bin == 1)[0]
        # Flip the ones with smallest h values (closest to 0.5)
        order = np.argsort(h[ones])[:int(s - target)]
        h_bin[ones[order]] = 0
    elif s < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:int(target - s)]
        h_bin[zeros[order]] = 1
    
    best_c = compute_c5(h_bin, n)
    
    # Local search on binary: flip pairs
    rng_local = np.random.RandomState(999)
    evals = 0
    max_evals = min(100000, int((deadline - time.time()) * 50))
    
    for _ in range(max_evals):
        if time.time() > deadline:
            break
        # Try flipping a pair (i,j): h[i] 1->0, h[j] 0->1
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng_local.randint(len(ones))]
        j = zeros[rng_local.randint(len(zeros))]
        h_bin[i] = 0
        h_bin[j] = 1
        c_new = compute_c5(h_bin, n)
        evals += 1
        if c_new < best_c - 1e-14:
            best_c = c_new
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    
    return h_bin, best_c

def optimize_pipeline(h_start, n, deadline, n_fista_iter=3000):

    # Phase 1: FISTA with moderate alpha

        if c_f < best_c:
            best_c = c_f
            best_h = h_f
        h = h_f
    
    # Phase 2: FISTA with higher alpha
    alpha2 = 2000
    fista_end2 = min(deadline - 40, time.time() + 30)
    if fista_end2 > time.time() + 10:
        h_f2, c_f2 = fista_optimize(h, n, alpha2, n_fista_iter, fista_end2)
        if c_f2 < best_c:
            best_c = c_f2
            best_h = h_f2
        h = h_f2
    
    # Phase 3: Active set SLSQP
    c = compute_all_lags(h, n)
    top_m = min(20, 2 * n - 1)
    active_idx = np.argsort(c)[-top_m:]
    
    for alpha_slsqp in [5000, 20000, 100000]:
        slsqp_end = time.time() + min(60, (deadline - time.time()) * 0.3)
        if slsqp_end < time.time() + 10:

        h_sls = slsqp_active_optimize(h, n, active_idx, alpha_slsqp, maxiter=2000, ftol=1e-14)
        c_sls = compute_c5(h_sls, n)
        if c_sls < best_c:
            best_c = c_sls
            best_h = h_sls
        h = h_sls
        # Update active set
        c = compute_all_lags(h, n)
        active_idx = np.argsort(c)[-top_m:]
    
    # Phase 4: Targeted lag reduction
    targ_end = min(deadline - 20, time.time() + 40)

        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
        h = h_ls
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
