    # = (1-x[p-k]) * [p in S(k)] - x[p] * [p = i-k for some i in S(k)]
    # = (1-x[p-k]) * [p in S(k)] - x[p] * [p+k in S(k)]
    # Wait: p = i-k means i = p+k. So the second term is -x[p] * [p+k in S(k)].
    
    # For k >= 0: S(k) = {k, ..., n-1}
    #   [p in S(k)] = [p >= k and p < n] = [p >= k] (since p < n always)
    #   [p+k in S(k)] = [p+k >= k and p+k < n] = [p+k < n] = [p < n-k]
    #   So d/dx[p] = (1-x[p-k]) * I[p>=k] - x[p] * I[p < n-k]
    #   But p-k must be valid: p-k >= 0, i.e., p >= k. So (1-x[p-k]) is only defined when p >= k.
    #   Actually x[p-k] is only defined for 0 <= p-k < n. If p < k, p-k < 0, undefined.
    #   So the first term is (1-x[p-k]) if p >= k, else 0.
    #   And the second term: p+k < n means p < n-k.
    #   Combined: if k >= 0:
    #     d/dx[p] = I[p>=k]*(1-x[p-k]) - I[p<n-k]*x[p]
    
    # For k < 0, let q = -k > 0: S(-q) = {0, 1, ..., n-1-q}
    #   [p in S(-q)] = [p <= n-1-q] = [p < n-q]
    #   [p+(-q) in S(-q)]: p-q in {0,...,n-1-q}, i.e., 0 <= p-q <= n-1-q, i.e., q <= p <= n-1
    #   d/dx[p] = I[p < n-q]*(1-x[p+q]) - I[p>=q]*x[p]
    
    # This is getting complex. Let me just use the formula:
    # grad[p] = sum_m w[m] * d/dx[p] corr[m]
    # = sum_k w[n-1+k] * d/dx[p] corr at lag k
    
    # For each k, d/dx[p] corr_k:
    # If k >= 0: I[p>=k]*(1-x[p-k]) - I[p<n-k]*x[p]
    # If k < 0 (k=-q): I[p<n-q]*(1-x[p+q]) - I[p>=q]*x[p]
    
    # So grad[p] = sum_{k>=0, k<=n-1} w[n-1+k] * (I[p>=k]*(1-x[p-k]) - I[p<n-k]*x[p])
    #             + sum_{k=1}^{n-1} w[n-1-k] * (I[p<n-k]*(1-x[p+k]) - I[p>=k]*x[p])
    
    # = sum_{k=0}^{n-1} w[n-1+k] * I[p>=k]*(1-x[p-k]) - sum_{k=0}^{n-1} w[n-1+k] * I[p<n-k]*x[p]
    #   + sum_{k=1}^{n-1} w[n-1-k] * I[p<n-k]*(1-x[p+k]) - sum_{k=1}^{n-1} w[n-1-k] * I[p>=k]*x[p]
    
    # This can be computed efficiently with cumulative sums!
    
    dx = 2.0 / n
    grad = np.zeros(n)
    W = w * dx  # Include the dx factor since corr = raw_corr * dx
    
    # Term 1: sum_{k=0}^{n-1} W[n-1+k] * I[p>=k] * (1-x[p-k])
    # For fixed p: k ranges from 0 to p (I[p>=k])
    # (1-x[p-k]) for k=0..p: this is (1-x[p]), (1-x[p-1]), ..., (1-x[0])
    # So term1[p] = sum_{k=0}^{p} W[n-1+k] * (1-x[p-k])
    # = sum_{j=0}^{p} W[n-1+p-j] * (1-x[j])  [substituting j=p-k]
    # This is a convolution!
    
    # Actually let me just do it directly for n=300, it's fast enough.
    # n is at most ~500, and 2n-1 lags, so 500*1000 = 500,000 operations. Fine.
    
    for m in range(2*n - 1):
        if W[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = W[m]
        if k >= 0:
            # d/dx[p] = I[p>=k]*(1-x[p-k]) - I[p<n-k]*x[p]
            # First part: p from k to n-1: (1-x[p-k])
            if k < n:
                grad[k:n] += wm * (1.0 - x[0:n-k])
            # Second part: p from 0 to n-k-1: -x[p]
            grad[0:n-k] -= wm * x[0:n-k]
        else:
            # k = -q, q = -k > 0
            q = -k
            # d/dx[p] = I[p<n-q]*(1-x[p+q]) - I[p>=q]*x[p]
            # First part: p from 0 to n-q-1: (1-x[p+q])
            if q < n:
                grad[0:n-q] += wm * (1.0 - x[q:n])
            # Second part: p from q to n-1: -x[p]
            grad[q:n] -= wm * x[q:n]
    
    return grad

def slsqp_run(h, n, alpha, maxiter, deadline):
    """Run SLSQP optimization"""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0, 
             'jac': lambda x, n=n: np.ones(n)}]
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    def obj_grad(x):
        return smooth_max_grad(x, n, alpha)
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def l_bfgs_run(h, n, alpha, mu, maxiter, deadline):
    """Run L-BFGS-B with penalty for constraint"""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        val = smooth_max_obj(x, n, alpha)
        val += mu * (x.sum() - n/2.0)**2
        return val
    
    def obj_grad(x):
        g = smooth_max_grad(x, n, alpha)
        g += 2*mu*(x.sum() - n/2.0) * np.ones(n)
        return g

                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                               'maxfun': maxiter*20, 'maxls': 80})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

**Vectorized approach for pair search:**
For a fixed delta d, and for each i, I want to find all j such that the move (h[i]+=d, h[j]-=d) reduces the max correlation.

For a fixed i and delta d, the change in corr at lag k is:
Δcorr_k = d * (1-h[i-k]) * I[i∈S(k)] - d * (1-h[j-k]) * I[j∈S(k)]

For the move to be beneficial, we need max_k (c_k + Δcorr_k) < c_max for all k.
This means: for all k, c_k + d*(1-h[i-k])*I[i∈S(k)] - d*(1-h[j-k])*I[j∈S(k)] < c_max

This is a constraint on j for each k. Hard to vectorize directly.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
