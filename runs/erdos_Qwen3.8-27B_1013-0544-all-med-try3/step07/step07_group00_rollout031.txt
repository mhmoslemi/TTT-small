            if tot < 1e-15:

    m = c.max()

    best_h = h.copy()
    v = np.zeros(n)
    bounds_lo = np.zeros(n)
    bounds_hi = np.ones(n)

        if time.time() > deadline:
            break
        # Adaptive learning rate decay
        progress = i / n_steps
        lr_t = lr * (1.0 - 0.5 * progress)

    return best_h, best_c

        configs=[(200, 20.0), (1000, 100.0), (5000, 500.0),
                 (20000, 2000.0), (50000, 10000.0)])
    
    # Phase 2: L-BFGS with high alpha (200s)
    if time.time() < deadline - 600:
        t2 = min(time.time() + 200, deadline - 600)
        best_h, best_c = run_lbfgs(best_h, n, t2,
            configs=[(100000, 5000.0), (300000, 20000.0),
                     (1000000, 50000.0), (3000000, 100000.0),
                     (10000000, 200000.0)])
    
    # Phase 3: Top-shifts focused optimization (150s)
    if time.time() < deadline - 400:
        t3 = min(time.time() + 150, deadline - 400)
        best_h, best_c = run_top_shifts(best_h, n, t3, top_k=5,
                                         alpha_list=[50000, 500000, 5000000])
    
    # Phase 4: Momentum optimizer (150s)
    if time.time() < deadline - 250:
        t4 = min(time.time() + 150, deadline - 250)
        h_mom, c_mom = run_momentum(best_h, n, t4, alpha=2000000,
                                     mu=50.0, lr=0.003, beta=0.9,
                                     n_steps=30000)
        if c_mom < best_c:
            best_c = c_mom
            best_h = h_mom
    
    # Phase 5: SLSQP polish (100s)
    if time.time() < deadline - 150:
        t5 = min(time.time() + 50, deadline - 150)
        h5, c5 = run_slqp(best_h, n, t5, alpha=5000000, maxiter=5000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: More top-shifts with tighter focus (80s)
    if time.time() < deadline - 80:
        t6 = min(time.time() + 80, deadline - 80)
        best_h, best_c = run_top_shifts(best_h, n, t6, top_k=3,
                                         alpha_list=[2000000, 10000000])
    
    # Phase 7: Gradient local search (60s)
    if time.time() < deadline - 40:
        t7 = min(time.time() + 60, deadline - 40)
        best_h, best_c = gradient_local_search(best_h, n, t7,
                                                base_step=0.001, top_k=5)
    
    # Phase 8: Fine pair search (40s)
    if time.time() < deadline - 15:
        t8 = min(time.time() + 40, deadline - 15)
        best_h, best_c = local_pair_search(best_h, n, t8, radius=15,
                                            deltas=[0.0005, 0.0002, 0.0001])
    
    # Phase 9: Final SLSQP
    if time.time() < deadline - 3:
        h9, c9 = run_slqp(best_h, n, deadline - 1, alpha=10000000, maxiter=3000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    # Compare with original resolution
    if c_orig < best_c:
        return h_orig, c_orig, n_orig
    
    return best_h, best_c, n
