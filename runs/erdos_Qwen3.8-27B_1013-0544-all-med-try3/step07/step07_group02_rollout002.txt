def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target_sum - h.sum()
    if abs(diff) > 1e-15:
        # Spread over all non-boundary elements
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

    s = exps.sum()
    val = m + (mx + np.log(s)) / alpha

            if prob.status in ('optimal', 'optimal_inaccurate') and h_var.value is not None:
                h_new = project_feasible(h_var.value, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c + 1e-8:
                    h_curr = h_new
                    best_c = c_new
                    # Update active lags
                    c_all = compute_all_lags(h_curr, n)
                    top_indices = np.argsort(c_all)[-8:]
                    active_lags = [int(k - (n-1)) for k in top_indices]
                else:
                    break
            else:
                break
        except Exception:
            break
    
    return h_curr

def gradient_guided_local_search(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            # Try all lags
            sorted_lags = np.argsort(c_all)[::-1][:20]
            improved_any = False
            for k2 in sorted_lags:
                if time.time() > deadline:
                    break
                lag2 = int(k2 - (n - 1))
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for i in range(0, n - m2):
                    j = i + m2
                    if h[i] > h[j] + 1e-6:
                        delta = min(0.01, (h[i] - h[j]) * 0.25)
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-12:
                            best_c = c_new
                            best_h = h.copy()
                            improved_any = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-6:
                        delta = min(0.01, (h[j] - h[i]) * 0.25)
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-12:
                            best_c = c_new
                            best_h = h.copy()
                            improved_any = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                if improved_any:
                    break
            if not improved_any:
                break
            continue
        
        m = abs(lag)
        improved = False
        for d in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if time.time() > deadline:
                break
            for i in range(0, n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-8:
                    delta = min(d, (h[i] - h[j]) * 0.3)
                    if delta < 1e-12:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-12:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-8:
                    delta = min(d, (h[j] - h[i]) * 0.3)
                    if delta < 1e-12:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-12:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
        if not improved:
            # Try multi-lag
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[::-1][:6]
            improved2 = False
            for k2 in top_k:
                if time.time() > deadline:
                    break
                lag2 = int(k2 - (n - 1))
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for i in np.random.RandomState(rnd).choice(n - m2, min(50, n-m2), replace=False):
                    if time.time() > deadline:
                        break
                    j2 = i + m2
                    if h[i] > h[j2] + 1e-8:
                        delta = 0.003
                        h[i] -= delta
                        h[j2] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-12:
                            best_c = c_new
                            best_h = h.copy()
                            improved2 = True
                        else:
                            h[i] += delta
                            h[j2] -= delta
                    elif h[j2] > h[i] + 1e-8:
                        delta = 0.003
                        h[i] += delta
                        h[j2] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-12:
                            best_c = c_new
                            best_h = h.copy()
                            improved2 = True
                        else:
                            h[i] -= delta
                            h[j2] += delta
            if not improved2:
                break
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, n_iters=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_curr = best_c
    T = 0.005
    T_min = 0.0001
    cooling = (T_min / T) ** (1.0 / n_iters)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = max(T * cooling, T_min)
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            top_k = np.argsort(c_all)[::-1][1:6]
            lag = int(top_k[0] - (n - 1))
        m = abs(lag)
        if m >= n:
            continue
        i = rng.randint(0, n - m)
        j = i + m
        if abs(h[i] - h[j]) < 1e-8:
            continue
        if h[i] > h[j]:
            delta = 0.005 * rng.random() + 0.001
            delta = min(delta, (h[i] - h[j]) * 0.5)
            h[i] -= delta
            h[j] += delta
        else:
            delta = 0.005 * rng.random() + 0.001
            delta = min(delta, (h[j] - h[i]) * 0.5)
            h[i] += delta
            h[j] -= delta
        
        c_new = compute_c5(h, n)
        if c_new < c_curr or rng.random() < np.exp(-(c_new - c_curr) / max(T, 1e-10)):
            c_curr = c_new
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
        else:
            if h[i] < h[j]:
                h[i] += delta
                h[j] -= delta
            else:
                h[i] += delta
                h[j] -= delta
    
    return best_h, best_c

def momentum_grad(h, n, alpha, deadline, lr=1.5, momentum=0.9, max_iter=5000):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        val, g = smooth_max_obj_grad(h_curr, n, alpha)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        if patience > 500:
            lr *= 0.7
            v *= 0.3
            patience = 0
    return best_h, best_c

def optimize_phase(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Gradient momentum with multiple alphas
    alphas = [50, 200, 500, 1500, 3000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h_new, c_new = momentum_grad(best_h, n, alpha, deadline - 5)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 2: L-BFGS-B
    for alpha in [500, 2000, 5000]:
        if time.time() > deadline - 10:
            break
        h_new = lbfgs_optimize(best_h, n, alpha, maxiter=2000, deadline=deadline - 5)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 3: SLSQP on active lags
    if time.time() < deadline - 15:
        c_all = compute_all_lags(best_h, n)
        top_idx = np.argsort(c_all)[-5:]
        h_new = slsqp_smooth(best_h, n, 5000, top_idx, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Also try top 8
        if time.time() < deadline - 10:
            c_all = compute_all_lags(best_h, n)
            top_idx = np.argsort(c_all)[-8:]
            h_new = slsqp_smooth(best_h, n, 3000, top_idx, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 4: CVXPy LP relaxation
    if time.time() < deadline - 20:
        c_all = compute_all_lags(best_h, n)
        top_k = np.argsort(c_all)[-10:]
        active_lags = [int(k - (n-1)) for k in top_k if k != n-1]
        h_new = cvxpy_lp_iteration(best_h, n, active_lags, max_iter=8, deadline=deadline - 5)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 5: Local search
    if time.time() < deadline - 5:
        h_new, c_new = gradient_guided_local_search(best_h, n, deadline - 2)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interpolate_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Phase A: Optimize at parent resolution
    alloc_a = min(400, (deadline - t0) * 0.4)
    h_opt, c_opt = optimize_phase(h_init, n_init, t0 + alloc_a)
    best_h, best_c = h_opt, c_opt
    
    # Phase B: Try nearby resolutions
    n_tries = [n_init + 50, n_init - 50, n_init + 100]
    n_tries = [nx for nx in n_tries if 300 <= nx <= 800]
    for n_try in n_tries:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 30
        if remaining < 80:
            break
        alloc = min(150, remaining * 0.3)
        if alloc < 50:
            break
        h_start = interpolate_to_resolution(best_h, len(best_h), n_try)
        h_opt, c_opt = optimize_phase(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase C: Multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 20
        n_restarts = max(1, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.015 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(70, (deadline - time.time() - 20) * 0.3)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_phase(hp, n_best, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase D: Simulated annealing polish
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        sa_deadline = min(deadline - 5, time.time() + 120)
        h_new, c_new = simulated_annealing(best_h, n, sa_deadline, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase E: Final local search
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        h_new, c_new = gradient_guided_local_search(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
