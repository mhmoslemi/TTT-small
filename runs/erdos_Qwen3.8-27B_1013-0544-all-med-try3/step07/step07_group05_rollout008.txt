import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=4000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def weighted_obj(x, n, weights):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return float(np.sum(weights * c))

def iterative_reweight(h, n, deadline, n_rounds=8, temp=80.0):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_shifted = c - c.max()
        w = np.exp(c_shifted * temp)
        w = w / w.sum()
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(weighted_obj, h, args=(n, w), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 300, 'gtol': 1e-12,
                                                   'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
        except Exception:
            break
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def lag_targeted_search(h, n, deadline, top_m=4):

def coordinate_descent(h, n, deadline, max_rounds=50):
    """For each coordinate, try a small range of values."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.001, 0.0005, 0.0002, 0.0001]
    
    for step in step_sizes:
        if time.time() > deadline:
            break
        for _round in range(max_rounds):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                # Try increasing h[i] (decrease h[j] for some j)
                # Use the coordinate with most room
                for direction in [1, -1]:
                    delta = direction * step
                    hi = best_h[i] + delta
                    if 0.0 <= hi <= 1.0:
                        # Need to compensate: decrease another coordinate
                        # Pick the j that, when decreased, most helps
                        c_old = compute_c5_vec(best_h, n)
                        c_max_idx = c_old.argmax()
                        offset = c_max_idx - (n - 1)
                        # Find partner j
                        if offset >= 0:
                            j = min(n - 1 - offset, i + offset) if i + offset < n else max(0, i - offset)
                        else:
                            abs_off = -offset
                            j = min(n - 1, i + abs_off) if i + abs_off < n else max(0, i - abs_off)
                        if j == i:
                            j = (i + 1) % n
                        hj = best_h[j] - delta
                        if 0.0 <= hj <= 1.0:
                            best_h[i], best_h[j] = hi, hj
                            best_h = project_feasible(best_h, n)
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                best_h[i] = best_h[i] - direction * step
                                best_h[j] = best_h[j] + direction * step
                                best_h = project_feasible(best_h, n)
            if not improved:
                break
    return best_h, best_c
