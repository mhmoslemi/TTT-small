    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, alpha):
    """Smooth maximum using log-sum-exp."""
    m = vals.max()
    return float(m + np.log(np.sum(np.exp(vals - m))) / alpha)

    """Minimize smooth max of correlation * dx."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c, alpha)

def sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_jacobian(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, alpha, maxiter=3000, gtol=1e-10):

    try:
        res = minimize(objective_lg, h, args=(n, alpha), method='L-BFGS-B',
                       bounds=bounds,

def try_periodic(n, period, num_ones_per_period):
    """Try all binary patterns of given period and find the best."""
    # For period p, we need num_ones_per_period ones
    # Total ones: num_ones_per_period * (n // period) should be n/2
    # So num_ones_per_period * n / period = n/2
    # num_ones_per_period = period / 2
    
    if n % period != 0:
        return None, float('inf')
    
    reps = n // period
    ones_needed = n // 2
    if num_ones_per_period * reps != ones_needed:
        return None, float('inf')
    
    best_pattern = None
    best_c = float('inf')
    
    # Enumerate all C(period, num_ones_per_period) patterns
    from itertools import combinations
    for positions in combinations(range(period), num_ones_per_period):
        pattern = np.zeros(period)
        for p_idx in positions:
            pattern[p_idx] = 1.0
        h = np.tile(pattern, reps)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_pattern = pattern.copy()
    
    return best_pattern, best_c

import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def compute_c5_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smooth_max_c(c, alpha):
    m = c.max()
    return float(m + np.log(np.sum(np.exp(c - m))) / alpha)


def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return smooth_max_c(c, alpha)


def sum_con(x, n):
    return float(x.sum() - n / 2.0)


def sum_con_jac(x, n):
    return np.ones(n)


def slsqp_opt(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def enum_periodic(n, period):
    if n % period != 0:
        return None, float('inf')
    reps = n // period
    ones_per = period // 2
    if ones_per * reps != n // 2:
        return None, float('inf')
    best_h = None
    best_c = float('inf')
    for positions in combinations(range(period), ones_per):
        pat = np.zeros(period)
        for p_idx in positions:
            pat[p_idx] = 1.0
        h = np.tile(pat, reps)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def make_binary(h, n, seed=0):
    h = (h > 0.5).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for _ in range(n_ones - target):
            idx = ones[rng.randint(len(ones))]
            h[idx] = 0
            ones = np.where(h == 1)[0]
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for _ in range(target - n_ones):
            idx = zeros[rng.randint(len(zeros))]
            h[idx] = 1
            zeros = np.where(h == 0)[0]
    return h


def binary_swap_search(h, n, deadline, max_passes=500):
    h = make_binary(h, n, seed=42)
    best_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 1e-14
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h, n)
                h[i], h[j] = 1.0, 0.0
                d = best_c - c
                if d > best_delta:

        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0

        else:
