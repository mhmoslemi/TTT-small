```python
def project_feasible(h, n):
    target_sum = n / 2.0

            room = 1.0 - h

            if tot < 1e-14:

            h += (diff / tot) * room

            if tot < 1e-14:

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:

    else:
        grad[0:n-m] += (1.0 - h[m:n])
```

Actually wait, the overlap at lag k (positive) is: #{i : 0≤i<n-k, h[i]=1, h[i+k]=0}
This is NOT the same as the number of pairs at distance k. It's specifically (1,0) pairs, not (0,1) pairs.

For a binary sequence: overlap at lag k = #{i: h[i]=1, h[i+k]=0}
And overlap at lag -k = #{i: h[i]=0, h[i+k]=1} = #{i: h[i+k]=1, h[i]=0} = #{i: h[i]=0, h[i+k]=1}

Note: overlap(k) + overlap(-k) = #{i: h[i]≠h[i+k]} for i in appropriate range.

For the full correlation: correlate(h, 1-h, mode='full')
At position n-1+k (lag k), this is Σ_{i=0}^{n-1-k} h[i]*(1-h[i+k]) for k≥0
For k<0, it's Σ_{i=-k}^{n-1} h[i]*(1-h[i+k]) = Σ_{i=0}^{n-1+k} h[i+k]*(1-h[i])

So for binary h, this is:
- Lag k≥0: #{i∈[0,n-1-k] : h[i]=1, h[i+k]=0}
- Lag k<0: #{i∈[0,n-1+k] : h[i-k]=1, h[i]=0} = #{i∈[0,n-1+k] : h[i]=0, h[i-k]=1} = #{j∈[|k|,n-1] : h[j]=1, h[j-|k|]=0}

OK so both positive and negative lags count (1,0) patterns, just in different directions.

```python
        improved = False

            # For lag > 0: overlap term at position i is h[i]*(1-h[i+m])
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        # Distribute to elements with room
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        rem = diff
        for i in idx:
            if rem > 0:
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
            else:
                sub = min(-rem, h[i])
                h[i] -= sub
                rem += sub
            if abs(rem) < 1e-14:
                break
    return h

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def softmax_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def softmax_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-30:
        return np.full(len(c), 1.0/len(c))
    return w / s

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = softmax_weights(c, alpha)
    grad = np.zeros(n)
    n_full = 2*n - 1
    for k in range(n_full):
        if w[k] < 1e-10:
            continue
        lag = k - (n-1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad

def nesterov_descent(h, n, alpha, deadline, lr=0.3, momentum=0.9, max_iter=100000):
    h_curr = h.copy()
    v = np.zeros(n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Nesterov lookahead
        h_look = h_curr + momentum * v
        h_look = np.clip(h_look, 0.0, 1.0)
        g = grad_smooth_max(h_look, n, alpha)
        v = momentum * v - lr * g
        h_new = project_feasible(h_curr + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        if patience > 200:
            lr *= 0.8
            v *= 0.5
            patience = 0
        if it % 5000 == 4999:
            lr = max(0.01, lr * 0.7)
    return best_h, best_c

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return softmax_value(c, alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha, maxiter=2000, top_m=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        if top_m is not None:
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]
            def obj(x):
                cc = compute_all_lags(x, n)
                active = cc[top_idx]
                return softmax_value(active, 2.0 * alpha)
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13})
        else:
            res = minimize(lambda x: _smooth_obj(x, n, alpha), h,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_local_search(h, n, deadline, max_rounds=2000):
    """Very efficient local search: O(20) evaluations per round."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    step_idx = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if step_idx >= len(step_sizes):
            break
            
        d = step_sizes[step_idx]
        top_k = np.argsort(c_all)[-4:]
        improved = False
        
        for k_star in top_k:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            if lag > 0:
                h_l = h[0:n-m]
                h_r = h[m:n]
                scores = h_l * (1.0 - h_r)
            else:
                h_l = h[0:n-m]
                h_r = h[m:n]
                scores = h_r * (1.0 - h_l)
            
            # Process top 15 positions
            top_pos = np.argsort(scores)[-15:]
            for pos in top_pos:
                if time.time() > deadline:
                    break
                i = pos
                j = pos + m
                if lag > 0:
                    dd = min(d, h[i] * 0.2, (1.0 - h[j]) * 0.2)
                    if dd < 1e-8:
                        continue
                    h[i] -= dd
                    h[j] += dd
                else:
                    dd = min(d, h[j] * 0.2, (1.0 - h[i]) * 0.2)
                    if dd < 1e-8:
                        continue
                    h[j] -= dd
                    h[i] += dd
                
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    c_all = compute_all_lags(h, n)
                else:
                    if lag > 0:
                        h[i] += dd
                        h[j] -= dd
                    else:
                        h[j] += dd
                        h[i] -= dd
        
        if not improved:
            step_idx += 1
            if step_idx < len(step_sizes):
                h = best_h.copy()
                c_all = compute_all_lags(h, n)
    
    return best_h, best_c

    # Phase 1: Nesterov with annealed alpha
    for alpha in [30, 100, 400, 2000]:

        phase_end = min(deadline - 10, time.time() + (deadline - time.time()) * 0.25)

        if c_new < best_c:

def upscale(h, n_new):

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_rng_global = np.random.RandomState(12345)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        rem = diff
        for i in idx:
            if rem > 0:
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
            else:
                sub = min(-rem, h[i])
                h[i] -= sub
                rem += sub
            if abs(rem) < 1e-13:
                break
    return h

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def softmax_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def softmax_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    return w / s

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = softmax_weights(c, alpha)
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for k in range(n_full):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad

def nesterov_descent(h, n, alpha, deadline, lr=0.3, momentum=0.9, max_iter=200000):
    h_curr = h.copy()
    v = np.zeros(n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_look = h_curr + momentum * v
        h_look = np.clip(h_look, 0.0, 1.0)
        g = grad_smooth_max(h_look, n, alpha)
        v = momentum * v - lr * g
        h_new = project_feasible(h_curr + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        if patience > 300:
            lr *= 0.75
            v *= 0.4
            patience = 0
        if it % 8000 == 7999:
            lr = max(0.005, lr * 0.7)
    return best_h, best_c

def _slsqp_obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return softmax_value(c, alpha)

def _slsqp_obj_top(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    return softmax_value(active, 2.0 * alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha, maxiter=2000, top_m=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        if top_m is not None:
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]
            res = minimize(_slsqp_obj_top, h, args=(n, alpha, top_idx),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13})
        else:
            res = minimize(_slsqp_obj_full, h, args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_local_search(h, n, deadline, max_rounds=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    step_idx = 0
    stagcount = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if step_idx >= len(step_sizes):
            break
        d = step_sizes[step_idx]
        top_k = np.argsort(c_all)[-4:]
        improved = False
        for k_star in top_k:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            if lag > 0:
                h_l = h[0:n-m]
                h_r = h[m:n]
                scores = h_l * (1.0 - h_r)
            else:
                h_l = h[0:n-m]
                h_r = h[m:n]
                scores = h_r * (1.0 - h_l)
            top_pos = np.argsort(scores)[-20:]
            for pos in sorted(top_pos, key=lambda p: -scores[p]):
                if time.time() > deadline:
                    break
                i = pos
                j = pos + m
                if lag > 0:
                    dd = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                    if dd < 1e-9:
                        continue
                    h[i] -= dd
                    h[j] += dd
                else:
                    dd = min(d, h[j] * 0.3, (1.0 - h[i]) * 0.3)
                    if dd < 1e-9:
                        continue
                    h[j] -= dd
                    h[i] += dd
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    c_all = compute_all_lags(h, n)
                else:
                    if lag > 0:
                        h[i] += dd
                        h[j] -= dd
                    else:
                        h[j] += dd
                        h[i] -= dd
        if not improved:
            stagcount += 1
            if stagcount > 50:
                step_idx += 1
                stagcount = 0
                if step_idx < len(step_sizes):
                    h = best_h.copy()
                    c_all = compute_all_lags(h, n)
            if step_idx >= len(step_sizes):
                break
        else:
            stagcount = 0
    return best_h, best_c

def binary_push_search(h, n, deadline, max_rounds=300):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        above = np.where(h > 0.52)[0]
        below = np.where(h < 0.48)[0]
        if len(above) == 0 or len(below) == 0:
            break
        improved = False
        attempts = min(40, len(above) * len(below))
        for _ in range(attempts):
            if time.time() > deadline:
                break
            ia = _rng_global.randint(0, len(above))
            ib = _rng_global.randint(0, len(below))
            a = above[ia]
            b = below[ib]
            if a == b:
                continue
            dd = 0.002
            if h[a] + dd > 1.0 or h[b] - dd < 0.0:
                continue
            h[a] += dd
            h[b] -= dd
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                improved = True
                above = np.where(h > 0.52)[0]
                below = np.where(h < 0.48)[0]
                if len(above) == 0 or len(below) == 0:
                    break
            else:
                h[a] -= dd
                h[b] += dd
        if not improved:
            break
    return best_h, best_c

def anneal_shuffle(h, n, deadline, rng, max_rounds=200):
    """Simulated annealing with structure-aware moves."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    T = 0.02
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        T *= 0.995
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        if m >= n:
            break
        i = rng.randint(0, n - m)
        j = i + m
        if h[i] > h[j]:
            dd = min(0.03 * rng.random(), h[i] * 0.3, (1 - h[j]) * 0.3)
            h[i] -= dd
            h[j] += dd
        elif h[j] > h[i]:
            dd = min(0.03 * rng.random(), h[j] * 0.3, (1 - h[i]) * 0.3)
            h[i] += dd
            h[j] -= dd
        else:
            continue
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        delta = c_new - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            if h[i] > h[j]:
                h[i] += dd
                h[j] -= dd
            else:
                h[i] -= dd
                h[j] += dd
            h = project_feasible(h, n)
    return best_h, best_c

def optimize_pipeline(h_start, n, deadline, rng):
```
