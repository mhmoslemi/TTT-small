import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def symmetrize(h, n):
    return 0.5 * (h + h[::-1])

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_sym_feasible(h, n):

    h = project_feasible(h, n)
    h = symmetrize(h, n)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / n * np.ones(n)
        h = np.clip(h, 0.0, 1.0)
    return symmetrize(h, n)

def shifted_grad(h, n, idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def find_active(c_vec, tol_frac=0.0005, max_count=10):
    mx = c_vec.max()
    threshold = mx - max(tol_frac * mx, 1e-10)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > max_count:
        order = np.argsort(-c_vec[active])
        active = active[order[:max_count]]
    return active

def softmax_active_obj(x, n, alpha, active_idx, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_active_grad(x, n, alpha, active_idx, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):
        if wi < 1e-15:
            continue
        grad += wi * shifted_grad(x, n, idx)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad

def run_lbfgs_active(h, n, alpha, active_idx, mu, maxiter, deadline):

def subgradient_step(h, n, active_idx, lr, deadline):
    """Exact subgradient descent on the max."""
    c_vec = compute_c5_vec(h, n)
    mx = c_vec.max()
    threshold = mx - 1e-9
    true_active = np.where(c_vec >= threshold)[0]
    if len(true_active) == 0:
        true_active = active_idx
    grad = np.zeros(n)
    for idx in true_active:
        grad += shifted_grad(h, n, idx)
    grad /= len(true_active)
    # Projected gradient step maintaining sum constraint
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new, n)
    return h_new, c_new

        smx = sh.max()

            return float(mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / a)
        else:
            return float(mx + np.log(np.sum(np.exp(sh))) / a)

    def sq_jac_full(x, nn, a):
        c = compute_c5_vec(x, nn)
        mx = c.max()
        sh = a * (c - mx)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        grad = np.zeros(nn)
        for wi, idx in zip(w, range(len(c))):
            if wi < 1e-12:
                continue
            grad += wi * shifted_grad(x, nn, idx)
        return grad

    def cons_fun(x, nn):
        return x.sum() - nn / 2.0

    def cons_jac(x, nn):
        return np.ones(nn)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}]

        return h_new, c_new
