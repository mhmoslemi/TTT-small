            if time.time() > deadline:
                break
            k = s - (n - 1)
            i_lo = max(0, -k)
            i_hi = min(n, n - k)
            if i_lo >= i_hi:
                continue
            i_vals = np.arange(i_lo, i_hi)
            j_vals = i_vals + k
            contrib = h[j_vals] * (1.0 - h[i_vals])
            
            top_pairs = np.argsort(contrib)[-6:]
            for tp in top_pairs:
                if time.time() > deadline:
                    break
                xi = j_vals[tp]
                xj = i_vals[tp]
                for delta in [0.002, 0.005, 0.01, 0.02, 0.03]:
                    if h[xi] > delta and (1.0 - h[xj]) > delta:
                        h[xi] -= delta
                        h[xj] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[xi] += delta
                            h[xj] -= delta
                    if (1.0 - h[xi]) > delta and h[xj] > delta:
                        h[xi] += delta
                        h[xj] -= delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[xi] -= delta
                            h[xj] += delta
        
        if not improved:
            n_random = 2000
            for _ in range(n_random):
                if time.time() > deadline:
                    return h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = 0.001 * rng.rand() + 0.0005
                sign = 1 if rng.rand() < 0.5 else -1
                hi_v = h[i] + sign * d
                hj_v = h[j] - sign * d
                if 0.0 <= hi_v <= 1.0 and 0.0 <= hj_v <= 1.0:
                    h[i], h[j] = hi_v, hj_v
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[j] = hi_v - sign * d, hj_v + sign * d
            if not improved:
                break
    return h, best_c

def fine_random_swaps(h, n, deadline, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.0003, 0.0005, 0.0008, 0.001, 0.0015, 0.002]
    stuck = 0
    max_stuck = 5000
    
    while time.time() < deadline and stuck < max_stuck:
        d = deltas[rng.randint(len(deltas))]
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sign = 1 if rng.rand() < 0.5 else -1
        hi_v = h[i] + sign * d
        hj_v = h[j] - sign * d
        if 0.0 <= hi_v <= 1.0 and 0.0 <= hj_v <= 1.0:
            h[i], h[j] = hi_v, hj_v
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                stuck = 0
            else:
                h[i], h[j] = hi_v - sign * d, hj_v + sign * d
                stuck += 1
    return h, best_c

def neighborhood_swaps(h, n, deadline, radius=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(999)
    improved_any = False
    
    for delta in [0.005, 0.002, 0.001, 0.0005]:
        if time.time() > deadline:
            break
        for _pass in range(50):
            if time.time() > deadline:
                return h, best_c
            imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    for sign in [1, -1]:
                        hi_v = h[i] + sign * delta
                        hj_v = h[j] - sign * delta
                        if 0.0 <= hi_v <= 1.0 and 0.0 <= hj_v <= 1.0:
                            h[i], h[j] = hi_v, hj_v
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                imp = True
                            else:
                                h[i], h[j] = hi_v - sign * delta, hj_v + sign * delta
            if not imp:
                break
            improved_any = True
        if not improved_any:
            break
    return h, best_c

def multi_start_optimize(h_init, n, deadline, rng):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    
    # SLSQP round 1
    if time.time() < deadline - 30:
        ht = slsqp_optimize(best_h, n, 2000, maxiter=3000)
        ct = compute_c5(ht, n)
        if ct < best_c:
            best_c = ct
            best_h = ht
    
    # SLSQP round 2
    if time.time() < deadline - 20:
        ht = slsqp_optimize(best_h, n, 10000, maxiter=4000)
        ct = compute_c5(ht, n)
        if ct < best_c:
            best_c = ct
            best_h = ht
    
    # Bottleneck swaps
    if time.time() < deadline - 15:
        best_h, best_c = bottleneck_swaps(best_h, n, time.time() + min(80, deadline - time.time() - 10), max_rounds=200)
    
    # SLSQP round 3 (higher alpha)
    if time.time() < deadline - 15:
        ht = slsqp_optimize(best_h, n, 50000, maxiter=5000)
        ct = compute_c5(ht, n)
        if ct < best_c:
            best_c = ct
            best_h = ht
    
    # Neighborhood swaps
    if time.time() < deadline - 10 and n <= 500:
        best_h, best_c = neighborhood_swaps(best_h, n, time.time() + min(60, deadline - time.time() - 5), radius=25)
    
    # Fine random swaps
    if time.time() < deadline - 3:
        best_h, best_c = fine_random_swaps(best_h, n, deadline - 2, rng)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize at initial resolution
    h1 = project_feasible(h_init, n_init)
    c1 = compute_c5(h1, n_init)
    best_h, best_c = h1.copy(), c1
    
    alloc1 = min(200, (deadline - t0) * 0.35)
    if alloc1 > 40:
        h1, c1 = multi_start_optimize(h1, n_init, t0 + alloc1, rng)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
    
    # Phase 2: Try at n=400
    n2 = 400
    if time.time() < deadline - 120:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_feasible(h2, n2)
        
        alloc2 = min(300, deadline - time.time() - 60)
        if alloc2 > 50:
            h2, c2 = multi_start_optimize(h2, n2, time.time() + alloc2, rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
    
    # Phase 3: Try at n=500 if time allows
    n3 = 500
    if time.time() < deadline - 100:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n3)
        h3 = np.interp(x_new, x_old, best_h)
        h3 = project_feasible(h3, n3)
        
        alloc3 = min(200, deadline - time.time() - 40)
        if alloc3 > 40:
            # Quick SLSQP + swaps at n=500
            c3 = compute_c5(h3, n3)
            if time.time() < deadline - 40:
                ht = slsqp_optimize(h3, n3, 5000, maxiter=3000)
                ct = compute_c5(ht, n3)
                if ct < c3:
                    c3 = ct
                    h3 = ht
            if time.time() < deadline - 30:
                h3, c3 = bottleneck_swaps(h3, n3, time.time() + min(60, deadline - time.time() - 25), max_rounds=100)
            if time.time() < deadline - 20:
                ht = slsqp_optimize(h3, n3, 30000, maxiter=4000)
                ct = compute_c5(ht, n3)
                if ct < c3:
                    c3 = ct
                    h3 = ht
            if time.time() < deadline - 10:
                h3, c3 = fine_random_swaps(h3, n3, deadline - 5, rng)
            
            h3 = project_feasible(h3, n3)
            c3 = compute_c5(h3, n3)
            if c3 < best_c:
                best_c = c3
                best_h = h3.copy()
    
    # Phase 4: Random restarts at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 50 and best_h is not None:
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        scales = [0.008, 0.02, 0.04, 0.06]
        for r in range(min(n_restarts, 6)):
            if time.time() > deadline - 30:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            cp = compute_c5(hp, n_best)
            # Quick SLSQP
            if time.time() < deadline - 25:
                hp = slsqp_optimize(hp, n_best, 10000, maxiter=3000)
                cp = compute_c5(hp, n_best)
            # Quick swaps
            if time.time() < deadline - 15 and cp < best_c + 0.001:
                hp, cp = bottleneck_swaps(hp, n_best, time.time() + min(30, deadline - time.time() - 10), max_rounds=80)
            if cp < best_c:
                best_c = cp
                best_h = hp.copy()
    
    # Phase 5: Final fine-tuning
    if best_h is not None and time.time() < deadline - 5:
        n_f = len(best_h)
        h_f = best_h.copy()
        c_f = best_c
        
        if time.time() < deadline - 15:
            h_f = slsqp_optimize(h_f, n_f, 100000, maxiter=5000)
            c_f = compute_c5(h_f, n_f)
        
        if time.time() < deadline - 3:
            h_f, c_f = fine_random_swaps(h_f, n_f, deadline - 2, rng)
        
        best_h = project_feasible(h_f, n_f)
        best_c = compute_c5(best_h, n_f)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
