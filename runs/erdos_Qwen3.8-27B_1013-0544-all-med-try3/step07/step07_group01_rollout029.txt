        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        v = beta * v + (1.0 - beta) * g / g_norm
        h_new = project_feasible(h - lr * v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1
            if stall > 80:
                lr *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
    return best_h, best_c

def opt_critical_pair(h, n, deadline, top_k=10):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            top_idx = np.argsort(c_vec)[-top_k:]
            candidates = set()
            for m_idx in top_idx:
                k = m_idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    for i in range(0, nv, max(1, nv // 50)):
                        candidates.add((i, i + k))
                else:
                    kk = -k
                    nv = n - kk
                    for i in range(0, nv, max(1, nv // 50)):
                        candidates.add((i, i + kk))
            cand_list = list(candidates)
            np.random.shuffle(cand_list)
            for (i, j) in cand_list:
                if time.time() > deadline:
                    return h, best_c
                if i >= n or j >= n:
                    continue
                if h[j] - d >= -1e-12 and h[i] + d <= 1.0 + 1e-12:
                    h[j] -= d
                    h[i] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[j] += d
                        h[i] -= d
                if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                    h[i] -= d
                    h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
            if not improved:
                break
    return h, best_c

def grad_guided_pair(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    alpha = 50000
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for step in steps:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            order = np.argsort(g)
            n_try = min(80, n // 2)
            low_candidates = order[:n_try]
            high_candidates = order[-n_try:]
            for i in low_candidates:
                if time.time() > deadline:
                    return h, best_c
                for j in high_candidates:
                    if i == j:
                        continue
                    if h[i] - step < -1e-12 or h[j] + step > 1.0 + 1e-12:
                        move = min(h[i], 1.0 - h[j])
                        if move < 1e-10:
                            continue
                    else:
                        move = step
                    h[i] -= move
                    h[j] += move
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                        g = grad_smooth_max(h, n, alpha)
                        order = np.argsort(g)
                        low_candidates = order[:n_try]
                        high_candidates = order[-n_try:]
                    else:
                        h[i] += move
                        h[j] -= move
            if not improved:
                break
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def random_restart(h, n, scale=0.05):
    x = np.linspace(0, 2, n)
    noise = np.random.randn(n) * scale
    h_new = project_feasible(h + noise, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Optimize at initial resolution
    t1 = min(t0 + 200, deadline - 550)
    best_h, best_c = opt_lbfgs(best_h, best_n, t1, min_time=60)

    if time.time() < deadline - 450:
        t2 = min(time.time() + 80, deadline - 450)
        h2, c2 = opt_slsqp(best_h, best_n, t2, alpha=500000, maxiter=2500)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 2: Upsample to n=600 and optimize
    n_600 = 600
    if time.time() < deadline - 350:
        h_600 = upsample(best_h, best_n, n_600)
        c_600 = compute_c5(h_600, n_600)

        t3 = min(time.time() + 180, deadline - 300)
        h_600, c_600 = opt_lbfgs(h_600, n_600, t3, min_time=60)

        if time.time() < deadline - 250:
            t4 = min(time.time() + 70, deadline - 250)
            h_600, c_600 = opt_slsqp(h_600, n_600, t4, alpha=500000, maxiter=2000)

        if c_600 < best_c:
            best_c = c_600
            best_h = h_600
            best_n = n_600

    # Phase 3: Try n=900
    if time.time() < deadline - 200:
        n_900 = 900
        h_900 = upsample(best_h, best_n, n_900)
        c_900 = compute_c5(h_900, n_900)

        t5 = min(time.time() + 150, deadline - 160)
        h_900, c_900 = opt_lbfgs(h_900, n_900, t5, min_time=50)

        if time.time() < deadline - 130:
            t6 = min(time.time() + 50, deadline - 130)
            h_900, c_900 = opt_slsqp(h_900, n_900, t6, alpha=300000, maxiter=1500)

        if c_900 < best_c:
            best_c = c_900
            best_h = h_900
            best_n = n_900

    # Phase 4: Momentum gradient descent with restarts
    if time.time() < deadline - 80:
        t7 = min(time.time() + 50, deadline - 80)
        h7, c7 = opt_momentum(best_h, best_n, t7, alpha=100000, max_iter=3000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

        # Try a restart with perturbation
        if time.time() < deadline - 50:
            h_r = random_restart(best_h, best_n, scale=0.03)
            t7b = min(time.time() + 30, deadline - 50)
            h_r, c_r = opt_momentum(h_r, best_n, t7b, alpha=100000, max_iter=1500)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 5: Critical pair search
    if time.time() < deadline - 50:
        t8 = min(time.time() + 50, deadline - 50)
        best_h, best_c = opt_critical_pair(best_h, best_n, t8, top_k=10)

    # Phase 6: Gradient-guided pair
    if time.time() < deadline - 30:
        t9 = min(time.time() + 25, deadline - 30)
        h9, c9 = grad_guided_pair(best_h, best_n, t9)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 12:
        t10 = min(time.time() + 15, deadline - 12)
        h10, c10 = opt_slsqp(best_h, best_n, t10, alpha=2000000, maxiter=2000)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Phase 8: Final critical pair search with tiny steps
    if time.time() < deadline - 2:
        t11 = deadline - 1
        best_h, best_c = opt_critical_pair(best_h, best_n, t11, top_k=6)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
