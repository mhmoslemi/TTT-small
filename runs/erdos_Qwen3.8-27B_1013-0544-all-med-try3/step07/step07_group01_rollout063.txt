from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

    c_shift = alpha * (c - c.max())
    mx = c_shift.max()

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

            h = project_feasible(h, n)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def mid_coord_descent(h, n, deadline, max_sweeps=5000):
    """Coordinate descent trying intermediate values for smoother convergence."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    k_vals = np.arange(n_corr) - (n - 1)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = float(np.max(corr))
        improved = False
        
        for i in range(n):
            old = h[i]
            if old < 0.01:
                candidates = [0.0, 0.25, 0.5]
            elif old > 0.99:
                candidates = [1.0, 0.75, 0.5]
            else:
                candidates = [0.0, 0.5, 1.0]
            
            best_t = old
            best_val = max_corr
            for t in candidates:
                dh = t - old
                if abs(dh) < 1e-15:
                    continue
                ik_fwd = i + k_vals
                fwd_valid = (ik_fwd >= 0) & (ik_fwd < n)
                ik_bwd = i - k_vals
                bwd_valid = (ik_bwd >= 0) & (ik_bwd < n)
                
                new_corr = corr.copy()
                new_corr[fwd_valid] += dh * (1.0 - h[ik_fwd[fwd_valid]])
                new_corr[bwd_valid] -= dh * h[ik_bwd[bwd_valid]]
                new_corr[n-1] -= dh * dh
                
                new_max = float(np.max(new_corr))
                if new_max < best_val - 1e-14:
                    best_val = new_max
                    best_t = t
            
            if best_t != old:
                dh = best_t - old
                h[i] = best_t
                max_corr = best_val
                corr = np.correlate(h, 1 - h, mode='full')
                improved = True
        
        if not improved:
            break
        if sweep % 50 == 0:
            h = project_feasible(h, n)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def pair_search_fine(h, n, deadline, deltas=None, radius=None):
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - (radius or n))
                hi = min(n, i + (radius or n) + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def active_set_optimize(h, n, deadline, top_k=5, rounds=10):
    """Optimize sum of top-k correlations using L-BFGS-B."""
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for rnd in range(rounds):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h, n)
        top_idx = np.argsort(c_vec)[-top_k:]
        top_vals = c_vec[top_idx]
        active_weight = np.zeros(2 * n - 1)
        for idx in top_idx:
            active_weight[idx] = 1.0
        
        def obj_active(x):
            cv = compute_c5_vec(x, n)
            return float(np.sum(cv * active_weight)) + 0.01 * (x.sum() - n/2.0)**2
        
        def grad_active(x):
            cv = compute_c5_vec(x, n)
            g = np.zeros(n)
            dx = 2.0 / n
            for m in top_idx:
                k = m - (n - 1)
                wm = dx
                if k >= 0:
                    nv = n - k
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk:n])
            g += 0.02 * (x.sum() - n/2.0)
            return g
        
        try:
            res = minimize(obj_active, h, jac=grad_active, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': 300, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def binarize_and_optimize(h, n, deadline, rng):
    """Push toward binary, re-optimize."""
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Adjust to correct sum
    target = n / 2.0
    cur_sum = h_bin.sum()
    if cur_sum > target:
        # Remove some 1s
        ones = np.where(h_bin == 1.0)[0]
        excess = int(cur_sum - target)
        # Remove from positions with lowest "importance"
        for _ in range(min(excess, len(ones))):
            idx = rng.choice(ones)
            h_bin[idx] = 0.0
            ones = np.where(h_bin == 1.0)[0]
    elif cur_sum < target:
        zeros = np.where(h_bin == 0.0)[0]
        deficit = int(target - cur_sum)
        for _ in range(min(deficit, len(zeros))):
            idx = rng.choice(zeros)
            h_bin[idx] = 1.0
            zeros = np.where(h_bin == 0.0)[0]
    
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    
    # Also try soft binarization
    h_soft = np.where(h > 0.7, 1.0, np.where(h < 0.3, 0.0, 0.5))
    h_soft = project_feasible(h_soft, n)
    c_soft = compute_c5(h_soft, n)
    
    if c_soft < c_bin:
        h_bin = h_soft
        c_bin = c_soft
    
    # Optimize from binary start
    alloc = min(60, deadline - time.time() - 30)
    if alloc > 20:
        h_opt, c_opt = lbfgs_phase(h_bin, n, time.time() + alloc * 0.6, max_iters=100)
        if time.time() < deadline - 10:
            h_opt, c_opt = fast_coord_descent(h_opt, n, time.time() + min(30, deadline - time.time() - 5), max_sweeps=2000)
        return h_opt, c_opt
    return h_bin, c_bin

def multi_start_optimize(h, n, deadline, rng):
    """Try random perturbations and re-optimize."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for trial in range(5):
        if time.time() > deadline - 30:
            break
        # Perturb
        h_pert = h.copy()
        noise = rng.randn(n) * 0.05
        h_pert = np.clip(h_pert + noise, 0, 1)
        h_pert = project_feasible(h_pert, n)
        
        alloc = min(80, (deadline - 30 - time.time()) / (5 - trial))
        if alloc < 20:
            break
        
        h_opt, c_opt = lbfgs_phase(h_pert, n, time.time() + alloc * 0.5, max_iters=80)
        if time.time() < deadline - 10:
            h_opt, c_opt = fast_coord_descent(h_opt, n, time.time() + min(20, deadline - time.time() - 5), max_sweeps=1000)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(250, 0.5)
        n0 = 250
    
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: L-BFGS-B with aggressive alpha schedule
    t1 = min(t0 + 180, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n, t1, max_iters=150)
    
    # Phase 2: Fast coordinate descent (the key innovation)
    if time.time() < deadline - 350:
        t2 = min(t0 + 400, deadline - 300)
        best_h, best_c = fast_coord_descent(best_h, n, t2, max_sweeps=15000)
    
    # Phase 3: Mid-value coordinate descent
    if time.time() < deadline - 250:
        t3 = min(t0 + 500, deadline - 200)
        best_h, best_c = mid_coord_descent(best_h, n, t3, max_sweeps=3000)
    
    # Phase 4: Active-set optimization
    if time.time() < deadline - 200:
        t4 = min(t0 + 600, deadline - 180)
        best_h, best_c = active_set_optimize(best_h, n, t4, top_k=5, rounds=8)
    
    # Phase 5: Fine pair search
    if time.time() < deadline - 150:
        t5 = min(t0 + 750, deadline - 130)
        best_h, best_c = pair_search_fine(best_h, n, t5,
            deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
            radius=50)
    
    # Phase 6: Binarization attempt
    if time.time() < deadline - 100:
        t6 = min(t0 + 850, deadline - 80)
        h_b, c_b = binarize_and_optimize(best_h, n, t6, rng)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b
    
    # Phase 7: Multi-start perturbation
    if time.time() < deadline - 80:
        t7 = min(t0 + 920, deadline - 60)
        best_h, best_c = multi_start_optimize(best_h, n, t7, rng)
    
    # Phase 8: Final coordinate descent + pair search
    if time.time() < deadline - 15:
        t8a = min(time.time() + 25, deadline - 10)
        best_h, best_c = fast_coord_descent(best_h, n, t8a, max_sweeps=5000)
    
    if time.time() < deadline - 5:
        t8b = min(time.time() + 10, deadline - 2)
        best_h, best_c = pair_search_fine(best_h, n, t8b,
            deltas=[0.001, 0.0005, 0.0002, 0.0001],
            radius=30)
    
    # Phase 9: Try different resolutions
    if time.time() < deadline - 30:
        for n_try in [200, 250, 400]:
            if time.time() > deadline - 15:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0/(2*len(best_h))
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0/(2*n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            
            alloc = min(25, deadline - time.time() - 10)
            if alloc < 10:
                break
            
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5, max_iters=80)
            if time.time() < deadline - 5:
                h_t, c_t = fast_coord_descent(h_t, n_try, time.time() + min(10, deadline - time.time() - 3), max_sweeps=2000)
            
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
    
    # Final cleanup
    nf = len(best_h)
    best_h = project_feasible(best_h, nf)
    best_c = compute_c5(best_h, nf)
    
    return best_h, best_c, nf
