def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_full, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _lag_gradient(h, n, m):
    """Gradient of c_m with respect to h, where m is the actual lag (can be negative)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    ap = abs(m)
    if ap >= n:
        return grad
    if ap == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
        return grad
    if m > 0:
        grad[:n - ap] += (1.0 - h[ap:]) * dx
        grad[ap:] -= h[:n - ap] * dx
    else:
        grad[ap:] += (1.0 - h[:-ap]) * dx
        grad[:n - ap] -= h[ap:] * dx
    return grad

def active_set_projected_gradient(h, n, deadline, top_m=5, max_steps=50000):
    """Active-set projected gradient with momentum."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    momentum = 0.0
    v = np.zeros(n)
    lr = 0.01
    step = 0
    patience = 0
    
    while time.time() < deadline and step < max_steps:
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        
        # Compute smooth max gradient over top lags
        a = c[top_idx]
        alpha = 500
        weights = np.exp(alpha * (a - a.max()))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            m = int(idx) - (n - 1)
            grad += w * _lag_gradient(h_curr, n, m)
        
        # Momentum SGD step
        v = 0.9 * v - lr * grad
        h_new = project_feasible(h_curr + v, n)
        
        # Backtracking line search
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            # Shrink step
            lr = max(lr * 0.9, 1e-6)
            v *= 0.5
        
        h_curr = h_new
        step += 1
        
        # Periodically recompute active set and adjust lr
        if step % 100 == 0:
            top_m = min(10, 3 + step // 1000)
            if patience > 200:
                lr = 0.01
                v = np.zeros(n)
                patience = 0
        
        if patience > 500:

            rng = np.random.RandomState(step)
            h_curr = project_feasible(h_curr + 0.01 * rng.randn(n), n)
            lr = 0.01
            v = np.zeros(n)
            patience = 0
    
    return best_h, best_c

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    v = cp.Variable(n)
    t = cp.Variable()
    cons = [cp.sum(v) == n / 2.0, v >= 0, v <= 1,
            cp.abs(v - h_curr) <= trust]
    for k in range(2 * n - 1):
        m = k - (n - 1)
        g = _lag_gradient(h_curr, n, m) / dx  # normalize to get gradient of the sum, not the integral
        # c[k] + dx * (g @ (v - h_curr)) <= t  →  c[k] + (g @ (v - h_curr)) <= t
        cons.append(c[k] + g @ (v - h_curr) <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    if v.value is None:
        return h_curr.copy()
    return project_feasible(v.value, n)

def local_search(h, n, deadline):
    h = h.copy()

    # Pairwise swaps focused on worst lag
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005]:

                        h[j] -= delta
                    else:
