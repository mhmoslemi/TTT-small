def resample_h(h, new_n):
   h=np.asarray(h,dtype=float)
   old_n=h.size
   if old_n==new_n: return h.copy()
   x_old=(np.arange(old_n)+0.5)/float(old_n)
   x_new=(np.arange(new_n)+0.5)/float(new_n)
   out=np.interp(x_new, x_old, h, left=h[0], right=h[-1])
   return out

def compute_c5(h, n):

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    target = float(target)
    if x.sum() == target:

        else:

def enforce_sum(h, target):

    out = np.interp(x_new, x_old, h, left=h[0], right=h[-1])

def build_P_Q(h, zeros, ones):
    n = h.size
    L = 2 * n - 1
    P = np.zeros((len(zeros), L), dtype=np.int32)
    Q = np.zeros((len(ones), L), dtype=np.int32)
    h32 = np.asarray(h, dtype=np.int32)
    comp_rev = (1 - h32)[::-1].copy()

        P[r, a:a + n] += comp_rev

    for r, b in enumerate(ones):
        Q[r, n - 1 - b:n - 1 - b + n] += h32
        Q[r, b:b + n] -= comp_rev

    return P, Q


def binary_delta(h, a, b):
    n = h.size
    L = 2 * n - 1
    delta = np.zeros(L, dtype=np.int32)
    h32 = np.asarray(h, dtype=np.int32)
    comp_rev = (1 - h32)[::-1].copy()

    delta[n - 1 - a:n - 1 - a + n] -= h32
    delta[a:a + n] += comp_rev

    delta[n - 1 - b:n - 1 - b + n] += h32
    delta[b:b + n] -= comp_rev

    delta[n - 1] -= 2
    delta[n - 1 + a - b] += 1
    delta[n - 1 + b - a] += 1
    return delta

            np.add(Q, base, out=T)
            T[:, 0] = -INF
            cand = T.max(axis=1)

            v = base[0] - 2 + Q0
            np.maximum(cand, v, out=cand)

            idx1 = (n - 1 + a - ones).astype(np.intp)
            v = base[idx1] + Q[row_ids, idx1] + 1
            np.maximum(cand, v, out=cand)

            idx2 = (n - 1 + ones - a).astype(np.intp)
            v = base[idx2] + Q[row_ids, idx2] + 1
            np.maximum(cand, v, out=cand)

            mc = int(cand.min())
            if mc < best_cand:
                best_cand = mc
                best_zr = zr
                best_bo = int(np.argmin(cand))

        if best_zr >= 0 and best_cand < cur:
            a = int(zeros[best_zr])
            b = int(ones[best_bo])
            h[a] = 1
            h[b] = 0
            C = np.correlate(h, 1 - h, mode="full").astype(np.int32)
            actual = int(C.max())

            if actual < cur:
                cur = actual
                best_c = actual
                best_h = h.copy()
                it += 1
            else:
                h[a] = 0
                h[b] = 1
                break
        else:
            break

    return best_h, best_c


def binary_anneal(h0, n, deadline, rng):
    if rng is None:
        rng = np.random.RandomState(0)

    h = round_to_binary(np.asarray(h0, dtype=float), n // 2)
    C = np.correlate(h, 1 - h, mode="full").astype(np.int32)
    cur = int(C.max())

    best_h = h.copy()
    best_c = cur
    stale = 0

    while time.time() < deadline:
        rem = deadline - time.time()
        if rem < 1.0:
            break

        desc_dl = time.time() + min(rem, max(2.0, 0.16 * rem))
        h2, _ = binary_descent(h, n, desc_dl)
        h = np.asarray(h2, dtype=np.int32)
        C = np.correlate(h, 1 - h, mode="full").astype(np.int32)
        cur = int(C.max())

        if cur < best_c:
            best_c = cur
            best_h = h.copy()
            stale = 0
        else:
            stale += 1

        if time.time() >= deadline - 1.0:
            break

        max_worse = 1
        if rem > 80:
            max_worse = 3
        if stale > 1:
            max_worse = min(5, max_worse + 1)

        if cur == best_c:
            k = int(rng.randint(1, 4))
        else:
            k = int(rng.randint(2, 5))
        k += min(6, 2 * stale)

        for _ in range(k):
            rem = deadline - time.time()
            if rem < 0.5:
                break
            zeros = np.nonzero(h == 0)[0]
            ones = np.nonzero(h == 1)[0]
            if zeros.size == 0 or ones.size == 0:
                break

            a = int(zeros[int(rng.randint(zeros.size))])
            b = int(ones[int(rng.randint(ones.size))])
            delta = binary_delta(h, a, b)
            Cn = C + delta
            nc = int(Cn.max())

            if nc <= cur + max_worse:
                h[a] = 1
                h[b] = 0
                C = Cn
                cur = nc
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                    stale = 0

        if cur > best_c + 5:
            h = best_h.copy()
            C = np.correlate(h, 1 - h, mode="full").astype(np.int32)
            cur = int(C.max())

    return best_h, best_c


def binary_optimize_n(n, parent, n_parent, deadline, seed):
    n = int(n)
    if n < 4 or n % 2 != 0:
        n = max(4, ((n + 1) // 2) * 2)

    m = n // 2
    rng = np.random.RandomState(seed)

    starts = []
    weights = []

    if parent is not None:
        if parent.size == n:
            pf = np.array(parent, dtype=float).copy()
        else:
            pf = resample_h(parent, n)
        starts.append(round_to_binary(pf, m))
        weights.append(3.0)

        for eps in (0.01, 0.04):
            starts.append(round_to_binary(pf + eps * rng.randn(n), m))
            weights.append(1.5)

    starts.append(make_random_h(n, rng))
    weights.append(1.0)
    starts.append(make_wave_h(n, rng))
    weights.append(1.0)

    best_h = None
    best_count = 10**9
    total_w = float(sum(weights))

    for idx, st in enumerate(starts):
        rem = deadline - time.time()
        if rem < 4.0:
            break

        w = max(0.1, weights[idx])
        alloc = rem * (w / total_w)
        if alloc < 5.0:
            alloc = min(rem, max(5.0, 0.2 * rem))
        if alloc > rem:
            alloc = rem

        hh, cc = binary_anneal(st, n, time.time() + alloc, rng)
        if cc < best_count:
            best_count = int(cc)
            best_h = np.asarray(hh, dtype=np.int32).copy()
        total_w -= w

    return best_h, best_count


def smooth_aux_q(x):
    return np.correlate(x, 1.0 - x, mode="full")


def smooth_objective(x):
    q = smooth_aux_q(x)
    m = float(q.max())
    a = float(_SM_ALPHA)
    if a <= 1.0:
        return m
    z = a * (q - m)
    z = np.minimum(z, 0.0)
    mx = float(z.max())
    e = np.exp(z - mx)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return m
    return float(m + (mx + math.log(s)) / a)


def smooth_gradient(x):
    n = int(_SM_N)
    q = smooth_aux_q(x)
    m = float(q.max())
    a = float(_SM_ALPHA)
    if a <= 1.0:
        p = np.full(q.size, 1.0 / float(q.size), dtype=float)
    else:
        z = a * (q - m)
        z = np.minimum(z, 0.0)
        mx = float(z.max())
        e = np.exp(z - mx)
        s = float(e.sum())
        if s <= 0.0 or not math.isfinite(s):
            p = np.zeros(q.size, dtype=float)
            p[int(np.argmax(q))] = 1.0
        else:
            p = e / s

    c1 = np.convolve(p, 1.0 - x, mode="full")
    c2 = np.convolve(x, p[::-1], mode="full")
    return (c1[n - 1:n - 1 + n] - c2[n - 1:n - 1 + n]).astype(float)


def smooth_constraint(x):
    return float(x.sum()) - float(_SM_T)


def smooth_constraint_jac(x):
    return np.ones(_SM_N, dtype=float)


def slsqp_smooth(h, n, deadline, alpha_schedule=None, maxiter_schedule=None):
    global _SM_N, _SM_T, _SM_ALPHA

    h = np.asarray(h, dtype=float).copy()
    target = float(n) / 2.0
    h = project_box_sum(h, target)
    h = enforce_sum(h, target)

    _SM_N = int(n)
    _SM_T = target

    if alpha_schedule is None:
        alpha_schedule = [100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0, 1000000.0]
    if maxiter_schedule is None:
        maxiter_schedule = [120, 150, 180, 220, 260, 300, 300]

    bounds = [(0.0, 1.0)] * int(n)
    cons = [{"type": "eq", "fun": smooth_constraint, "jac": smooth_constraint_jac}]

    for i, al in enumerate(alpha_schedule):
        if time.time() > deadline:
            break
        it = maxiter_schedule[i] if i < len(maxiter_schedule) else 180
        _SM_ALPHA = float(al)
        try:
            res = minimize(
                smooth_objective,
                h,
                jac=smooth_gradient,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(it), "ftol": 1e-15, "disp": 0},
            )
            x = res.x
        except Exception:
            x = h
        x = np.asarray(x, dtype=float)
        x = project_box_sum(x, target)
        x = enforce_sum(x, target)
        h = x
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 5.0
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    n_parent = 0
    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
        n_parent = int(parent.size)
        if n_parent > 0:
            best_h = parent.copy()
            best_n = n_parent
            best_c = compute_c5(best_h, best_n)
    except Exception:
        parent = None
        n_parent = 0

    if best_h is None:
        best_n = 250
        best_h = np.full(best_n, 0.5, dtype=float)
        best_h = enforce_sum(project_box_sum(best_h, best_n / 2.0), best_n / 2.0)
        best_c = compute_c5(best_h, best_n)

    cont_budget = min(240.0, max(70.0, 0.22 * (budget - 10.0)))
    bin_deadline = max(t0 + 30.0, deadline - cont_budget - 8.0)

    nlist = []
    parent_even = 0
    if n_parent > 0:
        parent_even = n_parent if n_parent % 2 == 0 else n_parent + 1
        if 4 <= parent_even <= 700:
            nlist.append(parent_even)

    for nn in (300, 400, 500, 350, 450, 320, 250, 360, 550, 600):
        if nn not in nlist:
            nlist.append(nn)

    weights = []
    first = parent_even if parent_even > 0 else nlist[0]
    for nn in nlist:
        if nn == first:
            w = 1.0
        elif nn <= 320:
            w = 1.25
        elif nn <= 450:
            w = 1.0
        else:
            w = 0.65
        weights.append(w)

    total_w = float(sum(weights))
    for idx, nn in enumerate(nlist):
        rem = bin_deadline - time.time()
        if rem < 20.0:
            break

        alloc = rem * (weights[idx] / total_w)
        cap = 95.0 if nn < 400 else 75.0
        alloc = min(alloc, cap)
        if alloc < 12.0:
            alloc = min(rem, max(12.0, 0.15 * rem))
        if alloc > rem:
            alloc = rem

        hh, cc = binary_optimize_n(nn, parent, n_parent, time.time() + alloc, int(seed) + 31 * nn)
        if hh is not None and cc < 10**8:
            c5 = float(int(cc)) * (2.0 / float(nn))
            if c5 < best_c:
                best_c = c5
                best_h = np.asarray(hh, dtype=np.int32).copy()
                best_n = int(nn)
        total_w -= weights[idx]

    rem = deadline - time.time()
    if rem > 45.0 and best_h is not None:
        if best_n % 2 == 0 and 100 <= best_n <= 320:
            n_cont = int(best_n)
        elif n_parent > 0 and n_parent <= 320 and n_parent % 2 == 0:
            n_cont = int(n_parent)
        else:
            n_cont = 250

        if best_n == n_cont:
            hf = np.asarray(best_h, dtype=float).copy()
        else:
            hf = resample_h(best_h, n_cont)
        target = n_cont / 2.0
        hf = project_box_sum(hf, target)
        hf = enforce_sum(hf, target)

        cont_dl = deadline - 4.0
        if cont_dl > time.time() + 40.0:
            alphas = [
                100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0, 1000000.0
            ]
            its = [100, 120, 160, 220, 260, 320, 320]
        elif cont_dl > time.time() + 20.0:
            alphas = [500.0, 5000.0, 100000.0, 1000000.0]
            its = [120, 180, 240, 240]
        else:
            alphas = [2000.0, 50000.0]
            its = [160, 160]

        hf = slsqp_smooth(hf, n_cont, cont_dl, alphas, its)
        c2 = compute_c5(hf, n_cont)
        if c2 < best_c:
            best_c = c2
            best_h = np.asarray(hf, dtype=float).copy()
            best_n = int(n_cont)

        rem = deadline - time.time()
        if rem > 25.0:
            hb = round_to_binary(hf, n_cont // 2)
            hb, cc = binary_anneal(hb, n_cont, deadline - 2.0, rng)
            c3 = float(int(cc)) * (2.0 / float(n_cont))
            if c3 < best_c:
                best_c = c3
                best_h = np.asarray(hb, dtype=np.int32).copy()
                best_n = int(n_cont)

    rem = deadline - time.time()
    if rem > 10.0 and best_n % 2 == 0 and best_n >= 4:
        hb = round_to_binary(np.asarray(best_h, dtype=float), best_n // 2)
        hb, cc = binary_anneal(hb, best_n, deadline - 1.0, rng)
        c4 = float(int(cc)) * (2.0 / float(best_n))
        if c4 < best_c:
            best_c = c4
            best_h = np.asarray(hb, dtype=np.int32).copy()
            best_n = int(best_n)

    best_h = np.asarray(best_h, dtype=float).copy()
    best_h = enforce_sum(best_h, best_n / 2.0)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
