import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_overlaps(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(h)
        if diff > 0:
            for i in idx[::-1]:
                if h[i] < 1.0:
                    add = min(1.0 - h[i], diff)
                    h[i] += add
                    diff -= add
                    if abs(diff) < 1e-15:
                        break
        else:
            for i in idx:
                if h[i] > 0.0:
                    sub = min(h[i], -diff)
                    h[i] -= sub
                    diff += sub
                    if abs(diff) < 1e-15:
                        break
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, deadline, step_schedule=None, verbose=False):
    """Single-coordinate descent with projection. Much faster than pair search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    n_stall = 0
    max_stall = 50
    
    if step_schedule is None:
        step_schedule = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
    
    s_idx = 0
    for step in step_schedule:
        if time.time() > deadline:
            break
        # Bisection to shrink step if no improvement
        improved = False
        for _pass in range(500):
            if time.time() > deadline:
                return h, best_c
            pass_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Try increase
                for sign in [1, -1]:
                    delta = sign * step
                    hi = h[i] + delta
                    if 0.0 <= hi <= 1.0:
                        h[i] = hi
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            pass_improved = True
                            improved = True
                        else:

                            h = project_feasible(h, n)
            if not pass_improved:

    return h, best_c

def coord_descent_adaptive(h, n, deadline, base_step=0.01):
    """Adaptive coordinate descent that shrinks step when stalled."""
    h = h.copy()
    best_c = compute_c5(h, n)
    step = base_step
    stall_count = 0
    max_stalls = 3
    
    while time.time() < deadline and step > 1e-7:
        pass_improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for sign in [1, -1]:
                delta = sign * step
                hi = h[i] + delta
                if 0.0 <= hi <= 1.0:
                    h[i] = hi
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        pass_improved = True
                        # Slightly grow step on success
                        step = min(step * 1.5, base_step)
                    else:
                        h[i] = h[i] - delta
                        h = project_feasible(h, n)
        
        if not pass_improved:
            stall_count += 1
            if stall_count >= max_stalls:
                step *= 0.5
                stall_count = 0
        else:
            stall_count = 0
    
    return h, best_c

def projected_subgradient(h, n, deadline, lr=0.001, T=0.01, n_avg=None):
    """Projected subgradient with multiplicative weights for minimax."""
    h = h.copy()
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    n_iter = 0
    max_iter = 5000
    
    # Maintain exponential weights for lags
    c = compute_overlaps(h, n)
    w = np.exp(-c / T)
    w /= w.sum()
    
    h_avg = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        n_iter += 1
        
        c = compute_overlaps(h, n)
        k_star = int(np.argmax(c))
        best_at_iter = float(c[k_star])
        
        if best_at_iter < best_c - 1e-15:
            best_c = best_at_iter
            best_h = h.copy()
        
        # Compute subgradient at active lag
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            for j in range(lag, n):
                grad[j] += (1.0 - h[j - lag])
            for j in range(0, n - lag):
                grad[j] -= h[j + lag]
        else:
            m = -lag
            for j in range(m, n):
                grad[j] += (1.0 - h[j - m])
            for j in range(0, n - m):
                grad[j] -= h[j + m]
        
        # Gradient step with projection
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Update weights (multiplicative weights)
        c = compute_overlaps(h_new, n)
        w = w * np.exp(c / T)
        w *= 1.0 / w.sum()
        
        # Track average (Champly-style)
        h_avg = (h_avg * (n_iter - 1) + h_new) / n_iter
        
        h = h_new
        
        # Decay learning rate
        lr = max(0.001 / (1 + 0.01 * it), 1e-7)
        # Anneal temperature
        if it % 500 == 0 and it > 0:
            T *= 0.9
    
    # Check average solution
    h_avg = project_feasible(h_avg, n)
    c_avg = compute_c5(h_avg, n)
    if c_avg < best_c:
        best_c = c_avg
        best_h = h_avg
    
    return project_feasible(best_h, n), best_c

def multiplicative_weights_qp(h, n, deadline, eta=0.5, n_iter_max=200):
    """Multiplicative weights with QP inner loop (using projected gradient for QP)."""
    h = h.copy()
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    T = 1.0
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        
        c = compute_overlaps(h, n)
        if c.max() < best_c - 1e-15:
            best_c = float(c.max())
            best_h = h.copy()
        
        # Update weights: w_k ∝ exp(c_k / T)
        w = np.exp(c / T)
        w /= w.sum()
        
        # Inner: min_h sum_k w_k * c_k(h) using projected gradient
        # c_k(h) = sum_i h[i+k](1-h[i]) -- this is quadratic in h
        # Gradient of sum_k w_k c_k(h):
        h_inner = h.copy()
        lr_inner = 0.01
        
        for pg_iter in range(50):
            if time.time() > deadline:
                break
            grad = np.zeros(n)
            for k_full in range(len(c)):
                if w[k_full] < 1e-8:
                    continue
                lag = k_full - (n - 1)
                if lag >= 0:
                    for j in range(lag, n):
                        grad[j] += w[k_full] * (1.0 - h_inner[j - lag])
                    for j in range(0, n - lag):
                        grad[j] -= w[k_full] * h_inner[j + lag]
                else:
                    m = -lag
                    for j in range(m, n):
                        grad[j] += w[k_full] * (1.0 - h_inner[j - m])
                    for j in range(0, n - m):
                        grad[j] -= w[k_full] * h_inner[j + m]
            h_inner = h_inner - lr_inner * grad
            h_inner = project_feasible(h_inner, n)
        
        h_inner = project_feasible(h_inner, n)
        c_inner = compute_c5(h_inner, n)
        if c_inner < best_c:
            best_c = c_inner
            best_h = h_inner.copy()
            h = h_inner.copy()
        
        T *= 1.05  # slightly increase temperature for exploration
    
    return project_feasible(best_h, n), best_c

            hp = best_h + rng.normal(0, scale, n_current)
            hp = project_feasible(hp, n_current)
            alloc_r = min(25, (deadline - time.time() - 50) * 0.4)
            if alloc_r < 15:
                break
            ht, ct = coord_descent_adaptive(hp, n_current, time.time() + alloc_r, base_step=0.008)
            if ct < best_c:
                best_c = ct
                best_h = ht
    
    # Phase 6: Final fine coordinate descent (remaining time)
    n_final = len(best_h)
    if time.time() < deadline - 2:
        best_h, best_c = coord_descent_adaptive(best_h, n_final, deadline - 2, base_step=0.003)
    
    # Final tiny steps
    if time.time() < deadline - 1:
        steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        for s in steps:
            if time.time() > deadline - 1:
                break
            for i in range(n_final):
                if time.time() > deadline - 1:
                    break
                for sign in [1, -1]:
                    delta = sign * s
                    hi = best_h[i] + delta
                    if 0.0 <= hi <= 1.0:
                        best_h[i] = hi
                        best_h = project_feasible(best_h, n_final)
                        c_new = compute_c5(best_h, n_final)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                        else:
                            best_h[i] = best_h[i] - delta
                            best_h = project_feasible(best_h, n_final)
    
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    n_final = len(best_h)
    return best_h, best_c, n_final
