def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:

            tot = room.sum()

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx

f_α(h) = max_k c_k + (1/α) log(sum_k exp(α(c_k - max_k c_k)))

As α → ∞, f_α → max_k c_k. For finite α, f_α is a smooth upper bound on max_k c_k.

The gradient of f_α is:
∇f_α = (1/α) * sum_k exp(α(c_k - max)) / sum_j exp(α(c_j - max)) * ∇c_k

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha, active_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    if active_idx is not None:
        c_use = c[active_idx]
        idx_map = active_idx
    else:
        c_use = c
        idx_map = np.arange(len(c))
    m = c_use.max()
    shifted = alpha * (c_use - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones_like(c_use) / len(c_use)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for i in range(len(idx_map)):
        wk = w[i]
        if wk < 1e-11:
            continue
        k_idx = int(idx_map[i])
        lag = k_idx - (n - 1)
        m_abs = abs(lag)
        if m_abs == 0:
            grad += wk * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[lag:] += wk * dx * (1.0 - x[:n - lag])
            grad[:n - lag] -= wk * dx * x[lag:]
        else:
            grad[:n - m_abs] += wk * dx * (1.0 - x[m_abs:])
            grad[m_abs:] -= wk * dx * x[:n - m_abs]
    return float(val), grad

def slsqp_opt(h, n, alpha, active_idx=None, maxiter=5000, ftol=1e-16):
    def fun(x):
        return smooth_obj_and_grad(x, n, alpha, active_idx)[0]
    def jac(x):
        return smooth_obj_and_grad(x, n, alpha, active_idx)[1]
    def con(x):
        return x.sum() - n / 2.0
    def jcon(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': jcon}]
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_optimize(h, n, deadline, top_m=5, max_rounds=30):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 5:
                break
            h_new = slsqp_opt(h_curr, n, alpha, active_idx=top_idx, maxiter=4000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                h_curr = h_new
    return best_h, best_c
