def smooth_max_all(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_active(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def power_max_active(x, n, top_idx, p):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    return float(np.sum(active ** p) ** (1.0 / p))

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n or m == 0:
        if m == 0:
            grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    if top_idx is not None:

    else:
        obj = lambda x: smooth_max_all(x, n, alpha)
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

        return h.copy()

def cp_external_approx(h_curr, n, trust, deadline):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    top_idx = np.argsort(c)[-15:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if k in top_idx or k == n - 1:
            grad = lag_gradient(h_curr, n, lag)
            cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
        else:
            cons.append(c[k] + dx * (norm_lag_grad_bound(n, lag) * trust) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def norm_lag_grad_bound(n, lag):
    m = abs(lag)
    if m >= n:

    return 1.0

        oo_end = time.time() + min(40, (deadline - time.time()) * 0.3)
        trust = 0.2
        for _ in range(5):
            if time.time() > oo_end:
                break
            h_new = cp_external_approx(best_h, n, trust, oo_end)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new
            trust *= 0.7
            if trust < 0.01:
                break

    if time.time() < deadline - 20:
        as_end = time.time() + min(60, (deadline - time.time()) * 0.4)
        h_new, c_new = active_set_iterate(best_h, n, as_end)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 15:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-8:]
        gd_end = time.time() + min(20, (deadline - time.time()) * 0.3)
        h_new, c_new = gradient_descent_smooth(best_h, n, 50000, top_idx,
                                               lr=0.01, n_steps=500, deadline=gd_end)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-5:]
        for p in [100, 500]:
            if time.time() > deadline - 10:
                break
            h_new = slsqp_power(best_h, n, p, top_idx, maxiter=3000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    if time.time() < deadline - 5:
        h_new = slsqp_smooth(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 2:
        best_h, best_c = local_search_focused(best_h, n, deadline - 1, top_lags=4, max_rounds=300)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        n_target = 400
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_target)
        h_interp = np.interp(x_new, x_old, h_init)
        h_interp = project_feasible(h_interp, n_target)
        alloc = min(350, (deadline - t0) * 0.55)
        h_opt, c_opt = optimize_n(h_interp, n_target, t0 + alloc)
        best_h, best_c = h_opt, c_opt

    if best_h is not None and time.time() < deadline - 80:
        n2 = 500
        x_best = np.linspace(0, 2, len(best_h))
        x_500 = np.linspace(0, 2, n2)
        h_500 = np.interp(x_500, x_best, best_h)
        h_500 = project_feasible(h_500, n2)
        alloc2 = min(200, (deadline - time.time()) * 0.45)
        if alloc2 > 60:
            h_opt2, c_opt2 = optimize_n(h_500, n2, time.time() + alloc2)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2

    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 25) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_smooth(best_h, n, 1000000, maxiter=6000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_search_focused(best_h, n, deadline - 1, top_lags=3, max_rounds=500)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
