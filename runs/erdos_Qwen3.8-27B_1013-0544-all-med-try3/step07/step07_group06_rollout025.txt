```python
def compute_swap_delta(h, a, b, n):
    """Compute the change in full correlation when swapping h[a] 0->1 and h[b] 1->0."""
    lags = np.arange(-(n-1), n)  # length 2n-1
    delta = np.zeros(2*n-1)
    
    # a+lag valid: a+lag in [0, n-1]
    mask_a_plus = (a + lags >= 0) & (a + lags < n)
    delta[mask_a_plus] += 1.0 - h[a + lags[mask_a_plus]]
    
    # a-lag valid: a-lag in [0, n-1]
    mask_a_minus = (a - lags >= 0) & (a - lags < n)
    delta[mask_a_minus] -= h[a - lags[mask_a_minus]]
    
    # b+lag valid: b+lag in [0, n-1]
    mask_b_plus = (b + lags >= 0) & (b + lags < n)
    delta[mask_b_plus] -= 1.0 - h[b + lags[mask_b_plus]]
    
    # b-lag valid: b-lag in [0, n-1]
    mask_b_minus = (b - lags >= 0) & (b - lags < n)
    delta[mask_b_minus] += h[b - lags[mask_b_minus]]
    
    return delta
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def objective(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

    target = n // 2

    best_c = np.inf
    best_h = None

        else:

            n_flips = rng.randint(3, min(15, target))
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def objective(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_sa_search(h_init, n, deadline, n_restarts=8, seed=42):

    best_h = None

        t_deadline = t_start + time_per_restart
        if time.time() > deadline:
            break
        
        # Initialize

            h = h_init.copy()
        else:
            h = best_h.copy()
            n_flips = min(target, 5 + restart * 3)
            for _ in range(n_flips):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    a = rng.choice(ones)
                    b = rng.choice(zeros)
                    h[a], h[b] = 0.0, 1.0

        if n_ones != target:

        cur_c = corr.max() * dx

        total_sweeps = 50
        cooling = (T_end / T_start) ** (1.0 / total_sweeps)
        T = T_start
        
        for sweep in range(total_sweeps):
            if time.time() > t_deadline:
                break
            T *= cooling

            zeros = np.where(h == 0)[0]
            n_ones = len(ones)
            n_zeros = len(zeros)

            sweep_improved = False

                m1 = (i_pos + lags_arr >= 0) & (i_pos + lags_arr < n)

                m2 = (i_pos - lags_arr >= 0) & (i_pos - lags_arr < n)
                delta[m2] -= h[i_pos - lags_arr[m2]]
                
                # j going 1 -> 0
                m3 = (j_pos + lags_arr >= 0) & (j_pos + lags_arr < n)
                delta[m3] -= 1.0 - h[j_pos + lags_arr[m3]]
                m4 = (j_pos - lags_arr >= 0) & (j_pos - lags_arr < n)
                delta[m4] += h[j_pos - lags_arr[m4]]
                
                # Corrections for interaction
                diff_ij = j_pos - i_pos
                if 0 <= (n - 1 + diff_ij) < 2 * n - 1:
                    delta[n - 1 + diff_ij] += 1.0
                diff_ji = i_pos - j_pos
                if 0 <= (n - 1 + diff_ji) < 2 * n - 1:
                    delta[n - 1 + diff_ji] += 1.0
                
                new_corr = corr + delta
                nc_counts = new_corr.max()
                nc = nc_counts * dx
                
                if nc_counts < corr.max():
                    # Improvement
                    h[i_pos] = 1.0
                    h[j_pos] = 0.0
                    corr = new_corr
                    # Update ones and zeros
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    sweep_improved = True
                elif T > 0.05 and rng.random() < np.exp(-(nc_counts - corr.max()) / T):
                    # Accept worse move
                    h[i_pos] = 1.0
                    h[j_pos] = 0.0
                    corr = new_corr
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                # else: reject

        if actual_c < best_c:
```

The key issue is: for each zero position i, we try all one positions j. That's O(n) per i, and there are O(n) zeros, so O(n²) per sweep.

**Optimization: Only try promising swaps**

For a given zero position i, the best j to swap with is the one that minimizes (corr + delta_ij).max(). We can approximate this by:
- The delta at the position where corr is maximum should be negative
- So we want delta at the argmax of corr to be as negative as possible

This gives a quick filter: for a given i, compute the "contribution" of position i to the max lag, and only try j that would reduce it.

Instead of trying all pairs, randomly sample K pairs per sweep and accept the best improvement found. This makes each sweep O(K*n) instead of O(n³).

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def slsqp_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def objective(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_swap_delta(h, i, j, n, lags_arr):
    """Compute delta in correlation when h[i] goes 0->1 and h[j] goes 1->0."""
    delta = np.zeros(2 * n - 1)
    
    m1 = (i + lags_arr >= 0) & (i + lags_arr < n)
    delta[m1] += 1.0 - h[i + lags_arr[m1]]
    m2 = (i - lags_arr >= 0) & (i - lags_arr < n)
    delta[m2] -= h[i - lags_arr[m2]]
    
    m3 = (j + lags_arr >= 0) & (j + lags_arr < n)
    delta[m3] -= 1.0 - h[j + lags_arr[m3]]
    m4 = (j - lags_arr >= 0) & (j - lags_arr < n)
    delta[m4] += h[j - lags_arr[m4]]
    
    # Interaction corrections
    d1 = j - i
    idx1 = n - 1 + d1
    if 0 <= idx1 < 2 * n - 1:
        delta[idx1] += 1.0
    d2 = i - j
    idx2 = n - 1 + d2
    if 0 <= idx2 < 2 * n - 1:
        delta[idx2] += 1.0
    
    return delta

    """Random-sampling simulated annealing on binary configs."""
    dx = 2.0 / n
    target = n // 2
    lags_arr = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    
    best_c = np.inf
    best_h = None
    
    # Initial
    h = h_init.copy()
    n_ones = int(np.round(h.sum()))
    if n_ones != target:
        h = project_binary(h, n)
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = corr.max()
    
    # SA schedule: total attempts based on time
    # Estimate attempts per second
    t0 = time.time()
    
    # Phase-based temperature
    T = 2.0  # in "count" units
    
    while time.time() < deadline:
        # Gradually cool
        elapsed = time.time() - t0
        total_time = deadline - t0
        frac = elapsed / total_time
        T = 2.0 * (1.0 - frac) + 0.05 * frac
        
        # Do a batch of attempts
        batch_size = 10000
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        improved_in_batch = 0
        for _ in range(batch_size):
            if time.time() > deadline:
                break
            i_idx = rng.randint(len(zeros))
            j_idx = rng.randint(len(ones))
            i_pos = zeros[i_idx]
            j_pos = ones[j_idx]
            
            delta = compute_swap_delta(h, i_pos, j_pos, n, lags_arr)
            new_max = (corr + delta).max()
            
            if new_max < cur_max:
                h[i_pos] = 1.0
                h[j_pos] = 0.0
                corr = corr + delta
                cur_max = new_max
                improved_in_batch += 1
                # Update arrays
                if improved_in_batch >= 10:
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    improved_in_batch = 0
            elif T > 0.05 and rng.random() < np.exp(-(new_max - cur_max) / T):
```

In 'full' mode, the output has length len(a) + len(v) - 1 = 2n-1.
