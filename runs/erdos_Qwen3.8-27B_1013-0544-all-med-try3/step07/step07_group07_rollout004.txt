def structured_init(n, pattern='baldwin'):
    """Create structured initial binary patterns."""
    h = np.zeros(n)
    if pattern == 'baldwin':
        # Baldwin-like: distribute 1s as evenly as possible with slight perturbation
        phi = (1 + np.sqrt(5)) / 2
        for i in range(n // 2):
            pos = int((i * phi) % n)
            h[pos] = 1.0
    elif pattern == 'random':
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    elif pattern == 'blocks':
        # Alternating blocks of varying size
        block_sizes = []
        remaining = n
        sizes = [3, 5, 7, 4, 6, 8, 5, 4, 7, 6]
        for s in sizes:
            if remaining <= 0:
                break
            b = min(s, remaining)
            block_sizes.append(b)
            remaining -= b
        if remaining > 0:
            block_sizes.append(remaining)
        pos = 0
        for i, b in enumerate(block_sizes):
            if i % 2 == 0:
                h[pos:pos+b] = 1.0
            pos += b
    elif pattern == 'strips':
        # Two strips
        h[:n//4] = 1.0
        h[3*n//4:] = 1.0
    return h

def multi_resolution_optimize(h_start, n_start, deadline, resolutions):
    """Optimize at multiple resolutions, promoting the best solution."""
    best_h = h_start.copy()
    best_c = compute_c5(best_h, n_start)
    n = n_start
    
    for nr in resolutions:
        if time.time() > deadline - 30:
            break
        # Upsample
        if nr > n:
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, nr)
            h_new = np.interp(x_new, x_old, best_h)
        else:
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, nr)
            h_new = np.interp(x_new, x_old, best_h)
        
        h_new = project_to_constraint(h_new, nr)
        
        # Continuous optimization
        phase_dl = min(time.time() + 40, deadline - 20)
        for alpha in [50, 200, 1000, 5000, 20000]:
            if time.time() > phase_dl - 5:
                break
            h_new, c_new = pgd_optimize(h_new, nr, alpha, phase_dl, max_iter=3000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        
        # Binary refinement
        if time.time() < deadline - 15:
            hb = to_binary(best_h, nr)
            phase_dl2 = min(time.time() + 25, deadline - 10)
            hb, cb = binary_local_search(hb, nr, phase_dl2, max_sweeps=100, trial_count=300)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

def anneal_binary_fast(hb, n, deadline, seed=42, n_iters=500000):
    """Fast simulated annealing for binary solutions."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = hb.copy()
    corr = binary_corr(h, n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    ones_list = list(ones_set)
    zeros_list = list(zeros_set)
    T = 0.05
    cool = (1e-8 / T) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = zeros_list[rng.randint(len(zeros_list))]
        j = ones_list[rng.randint(len(ones_list))]
        
        # Compute delta efficiently
        # h[i]: 0→1, h[j]: 1→0
        # Change in c[k] for each k:
        # Adding +1 at i: c[k] += (1-h[i+k]) for valid i+k [k = m-(n-1)]
        # Removing 1 at j: c[k] += h[j+k] for valid j+k (was -h[j+k]*(1-h[i+k])... )
        
        # It's simpler to just recompute for the affected positions
        h[i], h[j] = 1.0, 0.0
        new_corr = binary_corr(h, n)
        nc = new_corr.max() * dx
        
        accept = False
        if nc <= cur_c:
            accept = True
        elif T > 1e-10:
            if rng.random() < np.exp(-(nc - cur_c) / T * n * 0.1):
                accept = True
        
        if accept:
            corr = new_corr
            cur_c = nc
            ones_list.remove(j)
            zeros_list.remove(i)
            ones_list.append(i)
            zeros_list.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        
        T *= cool
    
    return best_h, best_c
