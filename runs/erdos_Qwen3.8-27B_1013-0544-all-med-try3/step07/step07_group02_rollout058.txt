# --- Core computation functions ---

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[0:n-m] += (1.0 - h[lag:n])
        grad[m:n] += -h[0:n-m]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_and_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        e = np.exp(shifted - mx)
    else:
        e = np.exp(shifted)
    s = e.sum()
    val = m + (mx + np.log(s)) / alpha
    w = e / s
    return val, w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    val, w = smooth_max_and_weights(c, alpha)
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        if w[ki] < 1e-10:

            grad[gm:n] += w[ki] * (-h[0:n-gm])
        else:
            grad[0:n-gm] += w[ki] * (1.0 - h[gm:n])
            grad[gm:n] += w[ki] * (-h[0:n-gm])
    return grad * dx

    c = compute_all_lags(h, n)
    sm, _ = smooth_max_and_weights(c, alpha)
    sv = np.sum(h) - n / 2.0
    return sm + pen * sv * sv

def _lbfgs_g(h, n, alpha, pen):
    g = grad_smooth_max(h, n, alpha)
    sv = np.sum(h) - n / 2.0
    g += 2.0 * pen * sv
    return g

def lbfgsb_optimize(h, n, alpha, deadline, max_rounds=3):
    h_curr = h.copy()
    bounds = [(0.0, 1.0)] * n

            h_curr = project_feasible(res.x, n)

            break
