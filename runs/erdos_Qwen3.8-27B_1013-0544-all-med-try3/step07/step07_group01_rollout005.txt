    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 5:
            break
        t_alloc = min(max_time_per_level, remaining - 2)
        
        mu = 10.0 * alpha  # penalty for constraint violation
        
        def obj(x):
            val = smooth_max_obj(x, n, alpha)
            val += mu * (x.sum() - n/2.0)**2
            return val
        
        def jac(x):
            g = smooth_max_grad(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n/2.0)
            return g
        
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters//n_levels, 
                                   'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': max_iters*5//n_levels,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def targeted_optimize(h, n, deadline, top_k=5):
    """Optimize using top-k correlation sum as objective"""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    # Use increasing k (start small, grow to include all)
    k_list = [2, 3, 5, 8, 15, 30, 50, len(h)*2-1]
    
    time_per = (deadline - time.time()) / len(k_list)
    time_per = max(3.0, min(30.0, time_per))
    
    for k in k_list:
        if time.time() > deadline - 2:
            break
        k = min(k, 2*n-1)
        t_alloc = min(time_per, deadline - time.time() - 2)
        
        def obj_topk(x):
            c = compute_c5_full(x, n)
            kk = min(k, len(c))
            topk = np.partition(c, -kk)[-kk:]
            val = topk.sum()
            val += 100.0 * (x.sum() - n/2.0)**2
            return val
        
        def jac_topk(x):
            c = compute_c5_full(x, n)
            kk = min(k, len(c))
            idx = np.argsort(c)[-kk:]
            dx = 2.0 / n
            g = np.zeros(n)
            for m in idx:
                lag = m - (n-1)
                if lag >= 0:
                    nv = n - lag
                    g[0:nv] += dx * (1.0 - x[lag:n])
                    g[lag:n] += dx * (-x[0:nv])
                else:
                    la = -lag
                    nv = n - la
                    g[la:n] += dx * (1.0 - x[0:nv])
                    g[0:nv] += dx * (-x[la:n])
            g += 200.0 * (x.sum() - n/2.0)
            return g
        
        try:
            res = minimize(obj_topk, h, jac=jac_topk, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-16,
                                   'maxfun': 1000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def pair_search_local(h, n, deadline, radius=30, max_rounds=20):
    """Local pair search with adaptive step sizes"""
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        improved_any = False
        for _ in range(max_rounds):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Try moving mass from i to j
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
            if not improved:
                break
        if not improved_any:
            pass
    
    return h, best_c

def slsqp_refine(h, n, deadline, alpha=100000):
    """SLSQP refinement"""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    def con_fun(x):
        return x.sum() - n/2.0
    
    def con_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                      options={'maxiter': 500, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

Actually, let me reconsider the problem. The key metric is:
C₅ = max_k correlate(h, 1-h, k) * dx

For the "center" lag (k=0 in the physical sense, which is index n-1 in the correlate output):
correlate[n-1] = sum_i h[i] * (1-h[i]) = sum h - sum h^2

For this to be small, we need sum h^2 to be close to sum h = n/2.
This means h should be close to binary (0 or 1).

For off-center lags, the situation is different. The linear correlation at lag k only has n-|k| terms.

I think the approach should be:
1. Make h as close to binary as possible (to minimize the center lag)
2. Among binary (or near-binary) configurations, find the one that minimizes off-center correlations

Let me try a different angle: **formulate the binary optimization as a combinatorial problem**.

For a binary sequence b with n/2 ones:
overlap[k] = number of pairs (i, i+k) where b[i+k]=1 and b[i]=0 (for k>0, i from 0 to n-1-k)

We want to minimize max_k overlap[k].

This is equivalent to: for a 0-1 matrix where we mark positions of 1s, minimize the maximum number of "1-adjacent-to-0" pairs at any shift.

One approach: simulated annealing on the binary sequence.
- Start with a random or Thue-Morse binary sequence
- Propose moves: swap a 0 and a 1, or flip (if it maintains balance)
- Accept based on energy = max_k overlap[k] (or sum of top-k)

Hmm, this is getting complex. Let me think of a simpler but effective approach.

Actually, the most efficient way to compute the overlap for a binary sequence is:
- For each 1-position p, add 1 to all lags where there's a 0 at the corresponding position
- This is equivalent to: for each 1-position p, for each 0-position q, add 1 to lag (p-q)

But that's O(n_ones * n_zeros) = O(n²/4) which is fine for n=300.

Actually, I can do it faster:
overlap[lag] = sum_i h[i] * (1-h[i+lag]) (for positive lag)
= (number of 1s in positions 0..n-1-lag) - (number of positions where both h[i]=1 and h[i+lag]=1)

Let me just use FFT-based correlation for efficiency.

Actually, for the simulated annealing, let me just recompute the full correlate after each swap. For n=300, np.correlate takes maybe 0.1ms. So we can do 10000+ swaps in 1 second.

Let me redesign the approach:

1. **Binary simulated annealing** (fast, discrete)
2. **Continuous L-BFGS-B** (refine the binary solution)
3. **Local pair search** (final polish)

The key insight is: the continuous optimum is likely very close to a binary solution (with slight smoothing at transitions). So optimizing the binary pattern first, then doing continuous refinement, should be effective.
