def free_var_optimize(h, n, alpha, deadline):
    """Optimize only the free variables (those not at 0 or 1)."""
    free_mask = (h > 0.05) & (h < 0.95)
    free_idx = np.where(free_mask)[0]
    n_free = len(free_idx)
    
    if n_free == 0:
        return h, compute_c5(h, n)
    
    if n_free > 50:
        # Too many free variables, just do global optimization
        return h, compute_c5(h, n)
    
    fixed = h.copy()
    
    def obj(x):
        h_trial = fixed.copy()
        h_trial[free_idx] = x
        h_trial = np.clip(h_trial, 0, 1)

        target_sum = n / 2.0

        m = c.max()
        shifted = alpha * (c - m)
        return float(m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha)
    
    x0 = h[free_idx]

    h_bin = to_binary(h, n)

    best_c = compute_c5(h_bin, n)
