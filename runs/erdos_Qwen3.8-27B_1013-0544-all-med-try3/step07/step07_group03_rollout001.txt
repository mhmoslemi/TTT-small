```python
        c_top_pert = c_pert[top_k_indices]
        obj_pert = np.log(np.sum(np.exp(alpha * (c_top_pert - m)))) / alpha  # approximate
        # Actually this is getting messy. Let me just do forward differences.
        grad[i] = (obj_pert - obj) / 1e-7

```

This is getting too complicated for the analytical gradient. Let me just use numerical gradients but be smart about it.

Actually, for the projected gradient method, I can compute the gradient of the smooth max more efficiently:

The smooth max is: f(h) = (1/α) log(sum_k exp(α * c_k(h)))
df/dh_i = (1/sum_k exp(α*c_k)) * sum_k exp(α*c_k) * dc_k/dh_i
         = sum_k w_k * dc_k/dh_i

where w_k = exp(α*c_k) / sum_j exp(α*c_j)

And dc_k/dh_i can be computed as:
c_k(h) = sum_j h[j] * (1 - h[j + lag(k,j)]) (over valid j)

dc_k/dh_i = (1 - h[i+lag]) * [i is in "first" position] + (some other terms from the quadratic part)

This is doable but tedious. For n=250, numerical gradient is fine (250 * O(n²) = 250 * 62500 ≈ 15M ops per gradient).

Let me simplify my approach:

1. Use SLSQP with log-sum-exp (as before but with better parameters)
2. Add a "top-lag" focused optimization phase
3. Use a faster local search (only try pairs where one element is near 0 or 1)
4. Try multiple resolutions
5. Be very careful with time management

Let me write the final code:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            frac = diff / tot
            h = np.clip(h + frac * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            frac = diff / tot
            h = np.clip(h * (1 + frac), 0.0, 1.0)
            # Alternative: proportional reduction
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    # Final bisection adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(100):
            if diff > 0:
                mask = h < 1.0
                if not mask.any():
                    break
                h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
            else:
                mask = h > 0.0
                if not mask.any():
                    break
                h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) < 1e-14:
                break
    return h

def slsqp_smooth(h, n, alpha, maxiter=1000, ftol=1e-13):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_toplevels(h, n, alpha, top_m, maxiter=1000):
    """Optimize using top-M lags only."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        c_x = np.correlate(x, 1 - x, mode='full') * dx
        c_top = c_x[top_idx]
        m = c_top.max()
        sh = alpha * (c_top - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline, deltas, radius):
    """Greedy pair search with local radius."""
    best_c = compute_c5(h, n)
    current = h.copy()
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return current, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    # Try moving mass from i to j
                    if current[i] - d >= 0.0 and current[j] + d <= 1.0:
                        current[i] -= d
                        current[j] += d
                        c = compute_c5(current, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            current[i] += d
                            current[j] -= d
                    # Try moving mass from j to i
                    if current[j] - d >= 0.0 and current[i] + d <= 1.0:
                        current[j] -= d
                        current[i] += d
                        c = compute_c5(current, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            current[j] += d
                            current[i] -= d
    return current, best_c

def stochastic_local_search(h, n, deadline, rng, n_iters=5000):
    """Random pair swap search."""
    best_c = compute_c5(h, n)
    current = h.copy()
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    while time.time() < deadline:
        batch = min(200, n_iters)
        for _ in range(batch):
            if time.time() > deadline:
                break
            i, j = rng.choice(n, 2, replace=False)
            d = deltas[rng.randint(len(deltas))]
            sign = rng.choice([1, -1])
            di = sign * d
            dj = -sign * d
            if 0 <= current[i] + di <= 1 and 0 <= current[j] + dj <= 1:
                current[i] += di
                current[j] += dj
                c = compute_c5(current, n)
                if c < best_c - 1e-15:
                    best_c = c
                else:
                    current[i] -= di
                    current[j] -= dj
    return current, best_c

    # Phase 1: SLSQP with increasing alpha
    alphas = [200, 500, 2000, 10000, 50000, 200000]
