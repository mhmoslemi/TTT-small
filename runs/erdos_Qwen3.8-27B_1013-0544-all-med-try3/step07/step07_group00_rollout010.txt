import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    # Ensure exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            idx = np.where(room > 0)[0]
            if len(idx) > 0:
                per = diff / len(idx)
                h[idx] = np.minimum(h[idx] + per, 1.0)
        else:
            idx = np.where(h > 0)[0]
            if len(idx) > 0:
                per = diff / len(idx)
                h[idx] = np.maximum(h[idx] + per, 0.0)
    return h

def grad_at_lag(h, n, lag_shift):
    """Gradient of c_k = sum h[s]*(1-h[s-k]) * dx w.r.t. h, for shift k = lag_shift."""
    dx = 2.0 / n
    g = np.zeros(n)
    k = lag_shift
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        # c_k = dx * sum_{s=k}^{n-1} h[s]*(1-h[s-k])
        # d(c_k)/d(h[s]) = dx*(1-h[s-k]) for s=k..n-1
        # d(c_k)/d(h[s]) = -dx*h[s+k] for s=0..n-1-k
        g[k:] += (1.0 - h[:m]) * dx
        g[:m] += (-h[k:]) * dx
    else:
        kk = -k
        m = n - kk
        if m <= 0:
            return g
        # c_{-kk} = dx * sum_{s=kk}^{n-1} h[s-kk]*(1-h[s])
        # d(c)/d(h[s]) = dx*(1-h[s]) for s=kk..n-1  (as first factor)
        # d(c)/d(h[s]) = -dx*h[s+kk] for s=0..n-1-kk  (as second factor)
        g[kk:] += (1.0 - h[:m]) * dx
        g[:m] += (-h[kk:]) * dx
    return g

def find_active_lags(c_vec, tol_frac=0.005):
    """Find lags within tol_frac fraction of the max."""
    m = c_vec.max()
    threshold = m - max(1e-10, tol_frac * m)
    return np.where(c_vec >= threshold)[0]

def smooth_obj(h, n, active_lags, alpha, mu=0.0):
    """Smoothed max objective over active lags."""
    c_vec = compute_c5_vec(h, n)
    vals = c_vec[active_lags]
    m = vals.max()

        obj = m
    else:
        sh = alpha * (vals - m)
        smx = sh.max()
        if smx > 50:
            obj = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
        else:
            obj = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:

    return obj

def smooth_grad(h, n, active_lags, alpha, mu=0.0):
    """Gradient of smoothed max objective."""
    c_vec = compute_c5_vec(h, n)
    vals = c_vec[active_lags]
    m = vals.max()
    if len(active_lags) == 1:
        w = np.array([1.0])
    else:
        sh = alpha * (vals - m)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
    g = np.zeros(n)

        g += wi * grad_at_lag(h, n, k)

        h_up, c_up = run_lbfgs_rounds(h_up, n_new, min(time.time() + 80, deadline - 20), n_rounds=4, mu=0.002)
        # Quick local search
        if time.time() < deadline - 20:
            h_up, c_up = local_search(h_up, n_new, deadline - 5, K=20)
        return h_up, c_up, n_new
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    np.random.seed(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n = 400
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Quick L-BFGS to improve (60s)
    t1 = min(t0 + 60, deadline - 500)
    if time.time() < t1:
        best_h, best_c = run_lbfgs_rounds(best_h, n, t1, n_rounds=4, mu=0.01)
    
    # Phase 2: More L-BFGS rounds with tighter active set (80s)
    t2 = min(time.time() + 80, deadline - 400)
    if time.time() < t2:
        best_h, best_c = run_lbfgs_rounds(best_h, n, t2, n_rounds=4, mu=0.001)
    
    # Phase 3: Main local search - the key phase (500s)
    t3 = min(time.time() + 500, deadline - 100)
    if time.time() < t3:
        best_h, best_c = local_search(best_h, n, t3, K=40)
    
    # Phase 4: Fine pair search for final polish (100s)
    t4 = min(time.time() + 100, deadline - 50)
    if time.time() < t4:
        best_h, best_c = fine_pair_search(best_h, n, t4, radius=40)
    
    # Phase 5: SLSQP polish (30s)
    t5 = min(time.time() + 30, deadline - 20)
    if time.time() < t5:
        h5, c5 = run_slqp(best_h, n, t5, alpha=500000, maxiter=2000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Another local search round (remaining time)
    if time.time() < deadline - 5:
        best_h, best_c = local_search(best_h, n, deadline - 2, K=25)
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
