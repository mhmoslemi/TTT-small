import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_pair_descent(h, n, deadline, radius=30, max_passes=500):
    """Coordinate descent on pairs within a local neighborhood"""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            j_lo = max(0, i - radius)
            j_hi = min(n, i + radius + 1)
            for j in range(j_lo, j_hi):
                if i == j:
                    continue
                h_new, c_new = pair_exact_optimize(h, n, i, j)
                if c_new < best_c - 1e-14:
                    h = h_new
                    best_c = c_new
                    improved = True
        if not improved:
            break
    
    return h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):

    h = h.copy()

            if tot < 1e-15:

            h += (diff / tot) * h

def smooth_obj(x, n, alpha):

Let me also think about whether the ternary search is the best approach. Since f(t) = max_k (base_k + t * slope_k) is convex and piecewise linear, the minimum is at a "kink" where two lines cross at the top of the envelope. 

An alternative: use the fact that at the optimum, the subgradient contains 0. The subgradient of f(t) is [min{slope_k : c[k](t) = f(t)}, max{slope_k : c[k](t) = f(t)}]. So 0 ∈ subgradient means there exist active k values with slope ≥ 0 and slope ≤ 0.

A faster approach: 
1. At t=0, find the active set A = {k : c[k] = max(c)}
2. If all slopes in A are > 0, the minimum is at t < 0
3. If all slopes in A are < 0, the minimum is at t > 0

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:

def _pair_optimize_t(c_curr, slopes, h_i, h_j, n_corr, n):
    """Find optimal t for pair (i,j) given precomputed coefficients. Returns (t_best, c_candidate)."""
    t_lo = max(-h_i, h_j - 1.0)
    t_hi = min(1.0 - h_i, h_j)
    if t_lo >= t_hi - 1e-15:
        return 0.0, float(c_curr.max())
    lo, hi = t_lo, t_hi
    for _ in range(14):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c_curr + m1 * slopes)
        f2 = np.max(c_curr + m2 * slopes)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t_best = (lo + hi) / 2.0
    c_candidate = float(np.max(c_curr + t_best * slopes))
    return t_best, c_candidate

def fast_local_pair_descent(h, n, deadline, radius=None, max_passes=300):
    """Fast coordinate descent on nearby pairs using exact 1D optimization."""
    if radius is None:

    h = h.copy()

        pass_improved = False

            l_i = i - shift_offsets

            ci = np.zeros(n_corr)
            ci[vm] = 1.0 - h[m_i[vm]]

            for j in range(j_lo, j_hi):

                slopes = ci - cj
                t_best, c_candidate = _pair_optimize_t(c_curr, slopes, h[i], h[j], n_corr, n)

                if abs(t_best) < 1e-14:
                    continue

                    h[j] -= t_best
                    np.clip(h, 0.0, 1.0, out=h)

                    t2 = t_best * t_best
                    k0 = n - 1
                    k1 = i - j + (n - 1)
                    k2 = j - i + (n - 1)
                    c_curr[k0] += -2.0 * t2
                    c_curr[k1] += t2
                    c_curr[k2] += t2

                    ci[vl] = ci[vl] + (-h[l_i[vl]])

        if not pass_improved:
            break

    final_c = compute_c5(h, n)
    return h, final_c
