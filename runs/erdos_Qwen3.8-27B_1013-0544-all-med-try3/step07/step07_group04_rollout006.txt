import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n - m] -= h[m:]
            grad[m:] += (1.0 - h[:n - m])
        else:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def slsqp_pass(h, n, alpha, maxiter=4000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-10:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha, top_idx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lp_outer_step(h_curr, n, trust, n_lags=15):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    top_k = np.argsort(c)[-n_lags:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in top_k:
        lag = k - (n - 1)
        grad = _lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCIP, verbose=False)
            except Exception:
                return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def lp_outer_loop(h, n, deadline, max_iters=15):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = lp_outer_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.01, trust * 0.85)
    return best_h, best_c

def local_search_ext(h, n, deadline, min_step=0.0001):
    h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, min_step]
    for step in steps:
        if time.time() > deadline:
            break
        stall = 0
        while stall < 3:
            if time.time() > deadline:
                break
            c_all = compute_all_lags(h, n)
            k_max = int(np.argmax(c_all))
            lag = k_max - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            rn = n - m
            best_improve = 0.0
            best_i = -1
            best_dir = 0
            base_cw = c_all[k_max]
            for i in range(rn):
                j = i + m
                diff = h[i] - h[j]
                if abs(diff) < 1e-9:
                    continue
                d = min(step, abs(diff) * 0.45)
                if d < 1e-10:
                    continue
                if diff > 0:
                    h[i] -= d
                    h[j] += d
                    dir_sign = 1
                else:
                    h[i] += d
                    h[j] -= d
                    dir_sign = -1
                cw = np.dot(h[:rn], 1.0 - h[m:]) * (2.0 / n)
                impr = base_cw - cw
                if impr > best_improve:
                    best_improve = impr
                    best_i = i
                    best_dir = dir_sign
                if dir_sign == 1:
                    h[i] += d
                    h[j] -= d
                else:
                    h[i] -= d
                    h[j] += d
            if best_i >= 0 and best_improve > 1e-13:
                j = best_i + m
                d = min(step, abs(h[best_i] - h[j]) * 0.45)
                if best_dir == 1:
                    h[best_i] -= d
                    h[j] += d
                else:
                    h[best_i] += d
                    h[j] -= d
                h = project_feasible(h, n)
                new_c = compute_c5(h, n)
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    stall = 0
                else:
                    stall += 1
            else:
                stall = 99
    return h, best_c

def subgrad_pass(h, n, deadline, n_iter=50000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = 0.05
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        grad = _lag_grad(h_curr, n, lag)
        h_new = h_curr - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
        if it % 2000 == 0:
            lr *= 0.92
    return best_h, best_c

def optimize_combined(h_start, n, deadline):
