                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            best_new_max = curr_max + 1.0
            best_i, best_j = -1, -1
            tried = 0
            max_tries = n - m

            for i in range(max_tries):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] == h[j]:
                    continue
                tried += 1
                h[i], h[j] = h[j], h[i]
                new_lags = np.correlate(h, 1 - h, mode='full') * dx
                new_max = float(new_lags.max())
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_i, best_j = i, j
                h[i], h[j] = h[j], h[i]

            if best_i >= 0 and best_new_max < curr_max - 1e-14:
                h[best_i], h[best_j] = h[best_j], h[best_i]
                curr_max = best_new_max
                found = True

        if found:
            if curr_max < best_c - 1e-14:
                best_c = curr_max
                best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 5:
                break

    return best_h, best_c

def directed_simulated_annealing(h, n, deadline, rng):
    h = h.copy()
    dx = 2.0 / n
    curr_c = compute_c5(h, n)
    best_c = curr_c
    best_h = h.copy()
    T = 0.02
    t_end = 1e-9
    cool = 0.99995
    lag_cache = np.correlate(h, 1 - h, mode='full') * dx

    while time.time() < deadline:
        if rng.randint(10) == 0:
            lag_cache = np.correlate(h, 1 - h, mode='full') * dx
            k_star = int(np.argmax(lag_cache))
            lag = k_star - (n - 1)
            if lag != 0:
                m = abs(lag)
                i = rng.randint(n - m)
                j = i + m
                if h[i] == h[j]:
                    i, j = rng.randint(n, size=2)
                    if i == j or h[i] == h[j]:
                        continue
            else:
                i, j = rng.randint(n, size=2)
                if i == j or h[i] == h[j]:
                    continue
        else:
            i, j = rng.randint(n, size=2)
            if i == j or h[i] == h[j]:
                continue

        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h, n)
        delta = new_c - curr_c

        if delta < 0 or (T > t_end and rng.random() < np.exp(-delta / T)):
            curr_c = new_c
            if curr_c < best_c - 1e-14:
                best_c = curr_c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

        T = max(t_end, T * cool)

    return best_h, best_c

def multi_lag_sa(h, n, deadline, rng, top_k=5):
    h = h.copy()
    dx = 2.0 / n
    curr_c = compute_c5(h, n)
    best_c = curr_c
    best_h = h.copy()
    T = 0.015
    t_end = 1e-9
    cool = 0.99997

    while time.time() < deadline:
        lag_cache = np.correlate(h, 1 - h, mode='full') * dx
        top_lags = np.argsort(lag_cache)[-top_k:]
        
        k_idx = top_lags[rng.randint(top_k)]
        lag = k_idx - (n - 1)
        if lag == 0:
            i, j = rng.randint(n, size=2)
            if i == j or h[i] == h[j]:
                continue
        else:
            m = abs(lag)
            if m >= n:
                continue
            i = rng.randint(n - m)
            j = i + m
            if h[i] == h[j]:
                i2, j2 = rng.randint(n, size=2)
                if i2 == j2 or h[i2] == h[j2]:
                    continue
                i, j = i2, j2

        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h, n)
        delta = new_c - curr_c

        if delta < 0 or (T > t_end and rng.random() < np.exp(-delta / T)):
            curr_c = new_c
            if curr_c < best_c - 1e-14:
                best_c = curr_c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

        T = max(t_end, T * cool)

    return best_h, best_c

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _top_k_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 50:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _full_smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_polish(h, n, alpha, top_idx=None, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        if top_idx is not None:
            res = minimize(_top_k_obj, h.copy(), args=(n, top_idx, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        else:
            res = minimize(_full_smooth_obj, h.copy(), args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_descent(h, n, deadline, alpha=50000, lr=0.5, momentum=0.9):
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    patience = 0

    def _lag_grad(h, n, lag):
        grad = np.zeros(n)
        m = abs(lag)
        if m >= n:
            return grad
        if lag == 0:
            grad[:] = 1.0 - 2.0 * h
            return grad
        if lag > 0:
            grad[m:n] += (1.0 - h[0:n-m])
            grad[0:n-m] += -h[m:n]
        else:
            grad[0:n-m] += (1.0 - h[m:n])
            grad[m:n] += -h[0:n-m]
        return grad

    for it in range(100000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        m_val = c.max()
        shifted = alpha * (c - m_val)
        mx_s = shifted.max()
        if mx_s > 50:
            w = np.exp(shifted - mx_s)
        else:
            w = np.exp(shifted)
        w = w / w.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[k] * _lag_grad(h, n, lag)
        grad *= dx

        v = momentum * v + lr * grad
        h_new = project_feasible(h - v, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1

        h = h_new
        if patience > 500:
            lr *= 0.7
            v *= 0.5
            patience = 0
        if it % 5000 == 4999:
            lr = max(0.05, lr * 0.85)

    return best_h, best_c

def pairwise_improvement(h, n, deadline, rng):
    """Try random pairs and keep improvement if found."""
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()

    while time.time() < deadline:
        i, j = rng.randint(n, size=2)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-14:
            best_c = new_c
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Binary projection + greedy search
    h_bin = to_binary(h_init, n_init)
    h_bin_c = compute_c5(h_bin, n_init)
    if h_bin_c < best_c:
        best_h = h_bin.copy()
        best_c = h_bin_c

    alloc1 = min(180, (deadline - t0) * 0.2)
    dl1 = t0 + alloc1
    h_new, c_new = greedy_worst_lag_search(best_h, n_init, dl1, top_k=3)
    if c_new < best_c:
        best_h = h_new
        best_c = c_new

    # Phase 2: Directed SA
    alloc2 = min(250, (deadline - t0) * 0.25)
    dl2 = time.time() + alloc2
    h_new, c_new = directed_simulated_annealing(best_h, n_init, dl2, rng)
    if c_new < best_c:
        best_h = h_new
        best_c = c_new

    # Phase 3: Multi-lag SA with restarts
    if time.time() < deadline - 100:
        alloc3 = min(200, (deadline - time.time()) * 0.25)
        dl3 = time.time() + alloc3
        h_new, c_new = multi_lag_sa(best_h, n_init, dl3, rng, top_k=5)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new

    # Phase 4: SLSQP polish on top lags
    if time.time() < deadline - 50:
        c_all = compute_all_lags(best_h, n_init)
        top_idx = np.argsort(c_all)[-5:]
        h_new = slsqp_polish(best_h, n_init, 100000, top_idx=top_idx, maxiter=2000)
        c_new = compute_c5(h_new, n_init)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new

    # Phase 5: Gradient descent for fine-tuning
    if time.time() < deadline - 40:
        alloc5 = min(60, (deadline - time.time()) * 0.3)
        dl5 = time.time() + alloc5
        h_new, c_new = gradient_descent(best_h, n_init, dl5, alpha=100000, lr=1.0, momentum=0.85)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new

    # Phase 6: Try different resolutions
    n_candidates = []
    if n_init < 600:
        n_candidates = [n_init + 50, n_init + 100]
    else:
        n_candidates = [450, n_init + 50]
    n_candidates = [x for x in n_candidates if x != n_init and x % 2 == 0 and 300 <= x <= 800]

    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.3)
        if alloc < 40:
            break

        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)

        # Greedy search at new resolution
        h_new_r, c_new_r = greedy_worst_lag_search(h_interp, n_try, time.time() + alloc * 0.5, top_k=3)
        if c_new_r < best_c:
            best_c = c_new_r
            best_h = h_new_r

        # SA at new resolution
        if time.time() < deadline - 50:
            h_sa, c_sa = directed_simulated_annealing(best_h, len(best_h), time.time() + min(40, (deadline - time.time()) * 0.3), rng)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa

    # Phase 7: Final pairwise improvement
    if time.time() < deadline - 15:
        n_final = len(best_h)
        alloc7 = min(30, (deadline - time.time()) * 0.5)
        h_new, c_new = pairwise_improvement(best_h, n_final, time.time() + alloc7, rng)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new

    # Phase 8: Final SLSQP + gradient
    if time.time() < deadline - 10:
        n_final = len(best_h)
        h_new = slsqp_polish(best_h, n_final, 200000, maxiter=1500)
        c_new = compute_c5(h_new, n_final)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new

    # Final: greedy one more time
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_new, c_new = greedy_worst_lag_search(best_h, n_final, deadline - 1, top_k=2, max_rounds=2000)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
