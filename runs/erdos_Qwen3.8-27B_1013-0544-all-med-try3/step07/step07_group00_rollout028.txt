            if tot < 1e-13:

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha

            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    
    # Add sum constraint penalty
    s = x.sum() - n / 2.0
    val += 1e-6 * s * s
    grad += 2e-6 * s * np.ones(n)
    
    return val, grad

def lbfgs_step(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        val, _ = obj_grad(x, n, alpha)
        return val
    
    def g(x):
        _, grad = obj_grad(x, n, alpha)
        return grad
    
    res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-25,
                            'gtol': 1e-16, 'maxfun': maxiter * 5, 'maxls': 20})
    return project_feasible(res.x, n)

def active_newton(h, n, deadline, max_iter=100):
    """Newton method on the active set of shifts."""
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        # Active set: shifts within 0.01% of max
        threshold = m - max(1e-9, 1e-4 * m)
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Compute the average subgradient
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx_val
                grad[k:n] += (-h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx_val
                grad[0:nv] += (-h[kk:n]) * dx_val
        grad /= len(active)
        
        # Projected gradient step with backtracking
        step = 0.01
        improved = False
        for _ in range(30):
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                best_c = c_trial
                h = h_trial
                improved = True
                break
            step *= 0.5
            if step < 1e-12:
                break
        
        if not improved:
            break
    
    return h, best_c

def coordinate_polish(h, n, deadline):
    """1D coordinate descent with line search."""
    best_c = compute_c5(h, n)
    
    for outer in range(10):
        if time.time() > deadline:
            break
        
        # Get gradient to prioritize coordinates
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()

                nv = n - k

            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx_val
                grad[0:nv] += (-h[kk:n]) * dx_val
        grad /= len(active)
        
        # Try coordinates in order of |gradient|
        coords = np.argsort(-np.abs(grad))
        
        any_improved = False
        for i in coords:
            if time.time() > deadline:
                break
            # Line search in coordinate i
            # Try a few step sizes
            for base_step in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
                for direction in [1, -1]:
                    new_val = h[i] + direction * base_step
                    if 0 < new_val < 1:
                        h[i] = new_val
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                            # Try to go further in same direction
                            for further in [2, 3, 5, 10]:
                                nv2 = h[i] + direction * base_step * (further - 1)
                                if 0 < nv2 < 1:
                                    h[i] = nv2
                                    c2 = compute_c5(h, n)
                                    if c2 < best_c - 1e-15:
                                        best_c = c2
                                    else:
                                        h[i] -= direction * base_step * (further - 1)
                                else:
                                    break
                        else:
                            h[i] -= direction * base_step
        if not any_improved:
            break
    
    return h, best_c

Let me try a different approach: **bisection on t with a QP feasibility check**

For a given t, we want to know if there exists h such that:
- c_k ≤ t for all k
- sum(h) = n/2
- 0 ≤ h ≤ 1

This is a quadratic constraint feasibility problem. We can solve it by minimizing:
minimize Σ_k max(0, c_k - t)² + λ * (sum(h) - n/2)²
subject to 0 ≤ h ≤ 1

If the minimum is 0 (or very close), then t is feasible. We can bisection on t to find the minimum feasible t.

But this is expensive because each QP solve requires many iterations.

2. Run L-BFGS for 5000 iterations
3. Then do a custom "active set" refinement where I:
   a. Find the active set A
   b. Solve a small Newton system on A
   c. Project back to feasible set

Here's the Newton refinement idea:
- Let A be the active set of shifts
- We want to minimize max_{k in A} c_k
- At the optimum, all c_k for k in A should be equal
- So we solve: find h' = h + d such that c_{k1}(h') = c_{k2}(h') = ... for all k in A, and sum(d) = 0
- This is a linear system (approximately, for small d)

The linearization: c_k(h + d) ≈ c_k(h) + ∇c_k(h) · d
Setting c_{k_i}(h + d) ≈ c_{k_i}(h) + ∇c_{k_i}(h) · d = t for all i
And sum(d) = 0

This is a system of (|A| + 1) equations in n unknowns. We can solve it using least squares.

        # Plus: sum(d) = 0

        # This means: c_k(h) + ∇c_k(h)·d = t
        # So: ∇c_k(h)·d = t - c_k(h)
        # With the constraint sum(d) = 0

        # Or we can set t = m - ε for some small ε

        # And sum(d) = 0

                grads[i, 0:nv] = (-h[kk:n]) * dx_val

        P = np.eye(n) - np.ones((n, n)) / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def correlated_grad(h, n, shift_idx):
    """Gradient of c_k with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smoothed_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smoothed_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wi = w[mm]
        if wi < 1e-14:

        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_lbfgs(h, n, alpha, maxiter):

    def func(x):
        return smoothed_max_obj(x, n, alpha)

        return smoothed_max_grad(x, n, alpha)
    
    res = minimize(func, h, jac=grad, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-25,
                            'gtol': 1e-16, 'maxfun': maxiter * 5, 'maxls': 20})
    return project_feasible(res.x, n)

    best_c = compute_c5(h, n)
    best_delta = 0

    # Need 0 ≤ h[i]+d ≤ 1 and 0 ≤ h[j]-d
    lo = max(-(h[i]), -(1.0 - h[j]))  # d ≥ -(h[i]) and d ≥ h[j] - 1
    hi = min(h[i], 1.0 - h[j])  # d ≤ h[i] and d ≤ 1 - h[j]
    # Wait, let me reconsider
    # h[i] + d ∈ [0,1] → d ∈ [-h[i], 1-h[i]]
    # h[j] - d ∈ [0,1] → d ∈ [h[j]-1, h[j]]
    lo = max(-h[i], h[j] - 1.0)

    lo = max(lo, -max_step)
    hi = min(hi, max_step)
