def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

        m = g.max()

        return None  # Let SLSQP use numeric gradient

def simulated_annealing(h_init, n, t_start, t_end, n_steps, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    target = n // 2
    
    # Ensure h is binary with exactly target ones
    h = np.zeros(n)
    if h_init is not None:
        sorted_idx = np.argsort(h_init)[::-1]
        h[sorted_idx[:target]] = 1.0
    else:
        h[rng.choice(n, target, replace=False)] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_energy = corr.max()
    best_energy = cur_energy
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        # Temperature
        frac = step / max(n_steps - 1, 1)
        T = t_start * (t_end / t_start) ** frac
        
        # Pick random pair: one from ones, one from zeros
        a = ones[rng.randint(len(ones))]  # will become 0
        b = zeros[rng.randint(len(zeros))]  # will become 1
        
        # Compute delta
        delta = np.zeros(2 * n - 1)
        im = b + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        
        imn = b - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        
        am = a + m_range
        msk = (am >= 0) & (am < n)
        delta[msk] -= 1.0 - h[am[msk]]
        
        амn = a - m_range
        msk = (амn >= 0) & (амn < n)
        delta[msk] += h[амn[msk]]
        
        # Cross terms
        delta[n-1] -= 2.0
        k1 = n - 1 + (b - a)
        delta[k1] += 1.0
        k2 = n - 1 + (a - b)
        delta[k2] += 1.0
        
        new_corr = corr + delta
        new_energy = new_corr.max()
        
        # Acceptance
        dE = new_energy - cur_energy
        if dE <= 0 or (T > 0 and rng.random() < np.exp(-dE / T)):
            h[a], h[b] = 0.0, 1.0
            corr = new_corr
            cur_energy = new_energy
            ones.remove(a)
            ones.append(b)
            zeros.remove(b)
            zeros.append(a)
            
            if cur_energy < best_energy:
                best_energy = cur_energy
                best_h = h.copy()
    
    return best_h, best_energy * dx

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        s = shifted.max()
        return m + (s + np.log(np.exp(shifted - s).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_simulated_annealing(n, h_init, t_start, t_end, n_steps, deadline, seed=42, max_flat=5000):
    """Run SA on binary sequences of length n with n/2 ones."""
    rng = np.random.RandomState(seed)
    target = n // 2
    m_range = np.arange(-(n-1), n)
    
    # Initialize
    h = np.zeros(n)
    if h_init is not None:
        sorted_idx = np.argsort(h_init)[::-1]
        h[sorted_idx[:target]] = 1.0
    else:
        h[rng.choice(n, target, replace=False)] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_e = corr.max()
    best_e = cur_e
    best_h = h.copy()
    flat_count = 0
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if flat_count > max_flat:
            break
        
        frac = step / max(n_steps - 1, 1)
        T = t_start * (t_end / t_start) ** frac
        
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        delta = np.zeros(2*n-1)
        im = b + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = b - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        am = a + m_range
        msk = (am >= 0) & (am < n)
        delta[msk] -= 1.0 - h[am[msk]]
        аmn = a - m_range
        msk = (амn >= 0) & (амn < n)
        delta[msk] += h[амn[msk]]
        delta[n-1] -= 2.0
        kb = n - 1 + (b - a)
        if 0 <= kb < 2*n-1:
            delta[kb] += 1.0
        ka = n - 1 + (a - b)
        if 0 <= ka < 2*n-1:
            delta[ka] += 1.0
        
        new_corr = corr + delta
        new_e = new_corr.max()
        dE = new_e - cur_e
        
        if dE <= 0 or (T > 1e-10 and rng.random() < np.exp(-dE / T)):
            h[a], h[b] = 0.0, 1.0
            corr = new_corr
            cur_e = new_e
            # Update lists (swap-remove for O(1))
            ia = ones.index(a)
            ones[ia] = ones[-1]
            ones.pop()
            ones.append(b)
            ib = zeros.index(b)
            zeros[ib] = zeros[-1]
            zeros.pop()
            zeros.append(a)
            
            if cur_e < best_e:
                best_e = cur_e
                best_h = h.copy()
                flat_count = 0
            else:
                flat_count += 1
        else:
            flat_count += 1
    
    return best_h, best_e
```
