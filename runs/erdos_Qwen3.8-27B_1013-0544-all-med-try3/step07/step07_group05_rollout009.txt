import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = np.maximum(0.0, 1.0 - h)
            mask = room > 1e-14
            tot = room[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * room[mask]
        else:
            mask = h > 1e-14
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=10000, ftol=1e-17):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,

    best_h = h.copy()

            break

        c = compute_c5_vec(h, n)
        m = c.max()

            best_c = m
            best_h = h.copy()
        elif m < best_c - 1e-15:
            best_c = m
            best_h = h.copy()
        
        active = np.where(c > m - eps)[0]
        
        grad = np.zeros(n)
        for k in active:
            lo1 = max(0, n - 1 - k)
            hi1 = min(n - 1, 2 * n - 2 - k)
            if lo1 <= hi1:
                j_start = lo1 + k - (n - 1)
                j_end = hi1 + k - (n - 1)
                grad[lo1:hi1+1] += 1.0 - h[j_start:j_end+1]
