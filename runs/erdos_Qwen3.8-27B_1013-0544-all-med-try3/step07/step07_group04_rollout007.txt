    c = np.correlate(h, 1-h, mode='full') * dx
    shifted = alpha * c
    mx = shifted.max()

    w = exp_c / exp_c.sum()  # weights for each lag
    obj = (mx + np.log(exp_c.sum())) / alpha

    for idx in range(2*n - 1):

            grad[m:] += w[idx] * dx * (1 - h[:n-m])

            mp = -m
            grad[:n-mp] += w[idx] * dx * (1 - h[mp:])
