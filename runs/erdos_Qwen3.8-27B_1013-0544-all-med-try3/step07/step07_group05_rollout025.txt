def cutting_plane_step(h, n, S, dx):
    """Solve LP: min t s.t. linearized f_s(h) + grad_s^T (h_new - h) <= t"""
    # Build the LP
    # Variables: h_new (n vars), t (1 var)
    # Constraints: for each s in S: grad_s^T h_new <= t - f_s(h) + grad_s^T h
    #              0 <= h_new <= 1
    #              sum(h_new) = n/2
    
    # This is an LP, solve with cvxpy
    import cvxpy as cp
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    for s in S:
        grad = compute_grad_f_s(h, s, n)
        f_val = compute_f_s(h, s, n)
        constraints.append(grad @ h_var <= t_var - f_val + grad @ h)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)  # or GLOP, GLPK
    
    return h_var.value, t_var.value

import cvxpy as cp
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_shifts(h, n):
    """Compute f_s for all shifts s"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx  # This gives f_s * dx for each shift position

def project_feasible(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-14:
                break
            h = h + diff / room.sum() * room
        else:
            if h.sum() < 1e-14:
                break
            h = h + diff / h.sum() * h
    return np.clip(h, 0, 1)

def get_gradient_f_s(h, s, n):
    """Compute gradient of f_s with respect to h, where s is the shift index in the correlation output.
    
    The correlation output index m corresponds to shift s = m - (n-1).
    For s >= 0: f_s = sum_{i=0}^{n-1-s} h[i](1-h[i+s])
    For s < 0: f_s = sum_{i=0}^{n-1-|s|} h[i+|s|](1-h[i])
    """
    grad = np.zeros(n)
    if s == 0:
        grad = 1 - 2*h
    elif s > 0:
        # f_s = sum_{i=0}^{n-1-s} h[i](1-h[i+s])

            grad[i+s] += -h[i]

        for i in range(n - t):

            grad[i] += -h[i+t]

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        # Distribute the remaining difference
        if diff > 0:
            idx = np.argsort(1.0 - h)[-int(abs(diff)):]
            for i in idx:
                h[i] = min(1.0, h[i] + diff / len(idx))
        else:
            idx = np.argsort(h)[-int(abs(diff)):]
            for i in idx:
                h[i] = max(0.0, h[i] + diff / len(idx))
    return h

    """Gradient of f_s (shift s) with respect to h.
    s is the actual shift (can be negative).
    """
    grad = np.zeros(n)
    if s == 0:
        return 1.0 - 2.0 * h
    elif s > 0:

        #          - h[j-s] for j in [s, n-1] (as second factor)
        grad[0:n-s] += (1.0 - h[s:n])  # first factor contribution
        grad[s:n] += -h[0:n-s]  # second factor contribution
    else:

        # f_{-t} = sum_{i=0}^{n-1-t} h[i+t](1-h[i])

        grad[t:n] += (1.0 - h[0:n-t])  # first factor
        grad[0:n-t] += -h[t:n]  # second factor

                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
                    h_new = np.clip(h_var.value, 0, 1)
                    h_new = project_feasible(h_new, n)
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            improved = False
            for t_frac in [0.7, 0.5, 0.3, 0.2, 0.1]:
                h_try = (1 - t_frac) * h + t_frac * h_new
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-12:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_c

def smooth_obj_fn(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 400:
        return float(m + (mx + np.log(np.max(np.exp(shifted - mx)) + np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj_fn, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_gradient(h, n, deadline, alpha=50000, max_iter=5000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    step = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1-h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        weights = np.exp(shifted - mx) / np.sum(np.exp(shifted - mx))
        
        grad = np.zeros(n)
        for k in range(len(c)):
            if weights[k] > 1e-8:
                s = k - (n - 1)
                grad += weights[k] * get_gradient_f_s(h, s, n)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            step = min(step * 1.05, 0.1)
        else:
            h = h_new.copy()
            step *= 0.7
            if step < 1e-7:
                step = 1e-7
    
    return best_h, best_c

def fast_local_search(h, n, deadline, deltas=None, radius=100):
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        old_i, old_j = h[i], h[j]
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = old_i, old_j
            if not improved:
                break
    
    return h, best_c

def fourier_optimize(h_start, n, K, deadline):
    x = np.arange(n) * 2.0 / n
    h_ac = h_start - 0.5
    params0 = np.zeros(2 * K)
    for k in range(1, K + 1):
        params0[2*(k-1)] = 2.0/n * np.sum(h_ac * np.cos(2*np.pi*k*x/n))
        params0[2*(k-1)+1] = 2.0/n * np.sum(h_ac * np.sin(2*np.pi*k*x/n))
    
    best_params = params0.copy()
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    
    def obj(params):
        h = np.full(n, 0.5)
        for k in range(1, K+1):
            h += params[2*(k-1)] * np.cos(2*np.pi*k*x/n)
            h += params[2*(k-1)+1] * np.sin(2*np.pi*k*x/n)
        h = project_feasible(h, n)
        return compute_c5(h, n)
    
    rng = np.random.RandomState(42)
    for restart in range(3):
        if time.time() > deadline:
            break
        if restart == 0:
            p0 = params0.copy()
        else:
            p0 = params0 + rng.normal(0, 0.02, 2*K)
        
        try:
            res = minimize(obj, p0, method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-10})
            h_new = np.full(n, 0.5)
            for k in range(1, K+1):
                h_new += res.x[2*(k-1)] * np.cos(2*np.pi*k*x/n)
                h_new += res.x[2*(k-1)+1] * np.sin(2*np.pi*k*x/n)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                best_params = res.x.copy()
        except Exception:
            continue
    
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # LP subroutine first
    alloc_lp = min(60, (deadline - time.time()) * 0.4)
    if alloc_lp > 5:
        h_lp, c_lp = lp_svn(best_h, n, top_k=499, max_iters=300, deadline=time.time()+alloc_lp)
        if c_lp < best_c:
            best_c = c_lp
            best_h = h_lp
    
    # SLSQP with increasing alpha
    if time.time() < deadline - 20:
        for alpha in [1000, 10000, 100000, 1000000]:
            if time.time() > deadline - 15:
                break
            alloc = min(20, deadline - time.time() - 15)
            if alloc < 3:
                break
            ht = slsqp_optimize(best_h, n, alpha, maxiter=2000)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht
    
    # Projected gradient
    if time.time() < deadline - 10:
        pg_deadline = min(deadline - 8, time.time() + 30)
        hp, cp = projected_gradient(best_h, n, pg_deadline, alpha=50000, max_iter=3000)
        if cp < best_c:
            best_c = cp
            best_h = hp
    
    # Local search
    if time.time() < deadline - 3:
        if n <= 300:
            best_h, best_c = fast_local_search(best_h, n, deadline - 1, radius=min(150, n//2))
        else:
            best_h, best_c = fast_local_search(best_h, n, deadline - 1, radius=80)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Optimize at initial resolution
    alloc1 = (deadline - t0) * 0.35
    h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    
    # Phase 2: Try different resolutions
    x_init = np.linspace(0, 2, n_init, endpoint=False)
    for n_try in [200, 300, 350]:
        if time.time() > deadline - 60:
            break
        remaining = deadline - time.time()
        alloc = min(100, remaining * 0.25)
        if alloc < 30:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        if n_try == n_init:
            h_interp = best_h.copy()
        else:
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_at_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Fourier-based global search at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 40:
        for K in [10, 20, 40]:
            if time.time() > deadline - 20:
                break
            f_deadline = min(deadline - 18, time.time() + 30)
            h_f, c_f = fourier_optimize(best_h, n_best, K, f_deadline)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
    
    # Phase 4: Random restarts
    if time.time() < deadline - 40:
        n_r = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(1, int(remaining / 40))
        scales = [0.01, 0.03, 0.05]
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_r)
            hp = project_feasible(hp, n_r)
            alloc = min(35, deadline - time.time() - 20)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_at_n(hp, n_r, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Final polishing
    if time.time() < deadline - 2:
        n_f = len(best_h)
        if n_f <= 300:
            best_h, best_c = fast_local_search(best_h, n_f, deadline - 1,
                                               deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                               radius=min(200, n_f//2))
        else:
            best_h, best_c = fast_local_search(best_h, n_f, deadline - 1,
                                               deltas=[0.001, 0.0005, 0.0002, 0.0001],
                                               radius=100)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
