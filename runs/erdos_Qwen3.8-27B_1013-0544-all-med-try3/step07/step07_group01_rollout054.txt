                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        h[i] += move
                        h[j] -= move
        if not improved:
            step *= 0.5
            alpha *= 2.0
            if step < 1e-12:
                break
    return h, best_c

def active_constraint_opt(h, n, deadline, alpha=10000, top_k=5, maxiter=500):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for _ in range(5):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        idx = np.argsort(c)[::-1][:top_k]
        def obj(x):
            cx = compute_c5_vec(x, n)
            m = cx.max()
            val = np.mean([cx[i] for i in idx])
            sh = alpha * (cx - m)
            mx = sh.max()
            if mx > 50:
                val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
            else:
                val = m + np.log(np.sum(np.exp(sh))) / alpha
            return float(val)
        cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_jac, 'args': (n,)}]
        t_rem = deadline - time.time()
        mi = max(10, min(maxiter, int(t_rem * 10)))
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': mi, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fourier_init(n, n_harmonics, rng):
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    h += 0.5
    for k in range(1, n_harmonics + 1):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    h = np.clip(h, 0, 1)
    h = project_feasible(h, n)
    return h

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B on parent resolution with corrected gradient
    t1_end = min(t0 + 180, deadline - 500)
    if t1_end > t0 + 30:
        h1, c1 = lbfgs_phase(best_h, best_n, t1_end, max_iters=300)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 400:
        for a in [200000, 1000000, 5000000]:
            if time.time() > deadline - 400:
                break
            t_rem = min(80, deadline - time.time() - 400)
            if t_rem < 20:
                break
            h2, c2 = slsqp_phase(best_h, best_n, time.time() + t_rem, alpha=a, maxiter=2000)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Gradient descent with corrected gradient
    if time.time() < deadline - 300:
        t3 = min(time.time() + 60, deadline - 300)
        h3, c3 = gradient_descent_correct(best_h, best_n, t3, lr_init=0.003, alpha_init=500.0)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Local gradient search
    if time.time() < deadline - 200:
        t4 = min(time.time() + 60, deadline - 200)
        h4, c4 = local_gradient_search(best_h, best_n, t4, step_init=0.002)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try higher resolutions
    for n_try in [400, 500, 600]:
        if time.time() > deadline - 150:
            break
        t_rem = deadline - time.time() - 50
        if t_rem < 80:
            break
        h_up = upsample(best_h, best_n, n_try)
        alloc = min(100, t_rem * 0.5)
        if alloc < 40:
            break
        h_t, c_t = lbfgs_phase(h_up, n_try, time.time() + alloc * 0.6, max_iters=200)
        if time.time() < deadline - 40:
            t_s = min(40, deadline - time.time() - 30)
            if t_s > 20:
                h_t, c_t = slsqp_phase(h_t, n_try, time.time() + t_s, alpha=2000000, maxiter=1500)
        if c_t < best_c - 1e-10:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 6: Active constraint optimization on best
    if time.time() < deadline - 80:
        t6 = min(time.time() + 60, deadline - 80)
        h6, c6 = active_constraint_opt(best_h, best_n, t6, alpha=10000, top_k=5, maxiter=300)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final L-BFGS + pair search
    if time.time() < deadline - 30:
        t7 = min(time.time() + 50, deadline - 30)
        h7, c7 = lbfgs_phase(best_h, best_n, t7, max_iters=200)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine pair search
    if time.time() < deadline - 10:
        t8 = deadline - 5
        h8, c8 = pair_search(best_h, best_n, t8,
                            [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                            radius=80)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Ultra-fine local search
    if time.time() < deadline - 3:
        h9, c9 = pair_search(best_h, best_n, deadline - 2,
                            [0.00005, 0.00002, 0.00001],
                            radius=50)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
