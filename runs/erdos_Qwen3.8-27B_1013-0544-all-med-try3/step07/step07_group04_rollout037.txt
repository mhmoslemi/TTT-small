```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_subgradient(h, n, active_lags):
    """Compute subgradient of max_k g_k(h) at h, using active lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in active_lags:
        m = abs(lag)
        if m == 0:
            grad += (1 - 2*h)
        elif lag > 0:
            # g_k = dx * sum_{i=0}^{n-1-k} h[i](1-h[i+k])
            # d/dh[j] = dx*(1-h[j+k]) for j < n-k
            #         - dx*h[j-k] for j >= k
            if m < n:
                grad[:n-m] += (1 - h[m:])
                grad[m:] += -h[:n-m]
        else:
            # lag < 0, m = -lag
            # g_k = dx * sum_{i=m}^{n-1} h[i](1-h[i-m])
            # d/dh[j] = dx*(1-h[j-m]) for j >= m
            #         -dx*h[j+m] for j <= n-1-m
            if m < n:
                grad[m:] += (1 - h[:n-m])
                grad[:n-m] += -h[m:]
    grad /= len(active_lags)  # Average over active lags
    return dx * grad

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def projected_subgradient(h, n, deadline, 
                          step_decay=0.999, 
                          base_step=0.02,
                          momentum=0.9,
                          tol=1e-10):
    """Projected subgradient method with momentum for the min-max problem."""
    h_curr = h.copy()
    v = np.zeros(n)  # Momentum
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    step = base_step
    for it in range(100000):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h_curr, n)
        cur_c = c.max()
        
        if cur_c < best_c - 1e-14:
            best_c = cur_c
            best_h = h_curr.copy()
        
        # Find active lags (within tolerance of max)
        tol_active = 1e-6
        active_mask = c >= cur_c - tol_active
        active_lags = np.where(active_mask)[0] - (n - 1)
        
        # Compute subgradient
        grad = compute_subgradient(h_curr, n, active_lags)
        
        # Update with momentum
        v = momentum * v - step * grad
        h_new = h_curr + v
        h_new = project_feasible(h_new, n)
        
        # Check for convergence
        if np.max(np.abs(h_new - h_curr)) < tol:
            break
        
        h_curr = h_new
        step *= step_decay
    
    return best_h, best_c
```

Actually, let me try a completely different approach: **bisection on the target value with LP relaxation**.

For a target t, we want to know if there exists h such that:
- 0 ≤ h ≤ 1
- sum(h) = n/2
- g_k(h) ≤ t for all k

The constraints g_k(h) ≤ t are quadratic (bilinear). But we can use a cutting plane approach:
1. Start with h₀
2. For the current h, the violated constraints (g_k(h) > t) give us information
3. Linearize g_k at h: g_k(h) ≈ g_k(h) + ∇g_k(h)·(h - h)
4. Add this as an LP constraint
5. Solve the LP
6. If the LP solution h_new satisfies all g_k(h_new) ≤ t, we're done
7. Otherwise, go to step 3 with h_new

This is the "outer approximation" method, but used for feasibility checking within a bisection.

The advantage: the LP is small (n variables, growing number of constraints) and can be solved quickly.

Let me implement this:

```python
def check_feasibility_lp(h_start, n, t, max_iters=2000):
    """Check if target t is achievable using LP outer approximation.
    Returns (h, c5) of the best solution found."""
    dx = 2.0 / n
    h_curr = h_start.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    # Maintain a list of linearization points
    linearizations = []  # List of (h_point, lag)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h_curr, n)
        cur_c = c.max()
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_curr.copy()
        
        if cur_c <= t + 1e-10:
            break  # Feasible!
        
        # Add linearization for the worst lag
        k_star = int(np.argmax(c))
        lag_star = k_star - (n - 1)
        grad = compute_lag_gradient(h_curr, n, lag_star)
        linearizations.append((h_curr.copy(), lag_star, c[k_star], grad))
        
        # Solve LP
        h_var = cp.Variable(n)
        cons = [cp.sum(h_var) == n/2.0, h_var >= 0, h_var <= 1]
        for (h_pt, lag, c_val, grad_pt) in linearizations:
            cons.append(c_val + grad_pt @ (h_var - h_pt) <= t)
        
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var - h_curr)), cons)
        try:
            prob.solve(solver=cp.CBC)
        except:
            prob.solve(solver=cp.GLOP)
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_curr = project_feasible(h_new, n)
    
    return best_h, best_c
```

```python
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] += -h[:n - m]
    else:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] += -h[m:]
    return grad

def compute_weighted_gradient(h, n, top_idx, weights):
    """Compute weighted gradient using only top lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        grad += w * compute_lag_gradient(h, n, lag)
    return dx * grad

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

                          base_lr=0.05, momentum=0.85, decay=0.9995):

        if np.max(np.abs(h_new - h_curr)) < 1e-13:
            break
        h_curr = h_new
        lr *= decay
        if it % 500 == 0:
            lr = max(base_lr * 0.1, lr)
    
    return best_h, best_c

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_obj_topk(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    return smooth_logsumexp(active, alpha)

def slsqp_optimize(h, n, alpha, top_k, maxiter=3000, ftol=1e-15):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_k:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_topk, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lp_outer_approx(h, n, deadline, max_iters=500, trust=None):
    """LP outer approximation with growing constraint pool."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    
    # Pool of linearizations: (point, lag, value, gradient)
    pool = []
    
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        
        c = compute_all_lags(h_curr, n)
        cur_max = c.max()
        if cur_max < best_c - 1e-13:
            best_c = cur_max
            best_h = h_curr.copy()
        
        # Add top-3 lags to pool
        top_idx = np.argsort(c)[-3:]
        for k_idx in top_idx:
            lag = k_idx - (n - 1)
            grad = compute_lag_gradient(h_curr, n, lag)
            pool.append((h_curr.copy(), c[k_idx], grad))
        
        # Keep only the most recent 200 + worst lags
        if len(pool) > 200:
            # Keep constraints that are close to binding
            active_pool = []
            for (pt, val, gr) in pool:
                # Check if this constraint is still relevant
                h_diff = h_curr - pt
                predicted = val + dx * (gr @ h_diff)
                actual = compute_lag_value(h_curr, n, (gr.sum() * 0 + 0))  # placeholder
                active_pool.append((pt, val, gr))
            pool = active_pool[-200:]
        
        # Solve LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        
        # Add linearization constraints (limit to last 100 for speed)
        recent = pool[-100:]
        for (pt, val, gr) in recent:
            cons.append(val + dx * (gr @ (h_var - pt)) <= t_var)
        
        # Add trust region
        if trust is not None:
            cons.append(cp.norm(h_var - h_curr, 1) <= trust)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
    
    return best_h, best_c

def compute_lag_value(h, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return 0.0
    if lag > 0:
        return float(np.sum(h[:n-m] * (1 - h[m:])) * dx)
    elif lag < 0:
        return float(np.sum(h[m:] * (1 - h[:n-m])) * dx)
    else:
        return float(np.sum(h * (1 - h)) * dx)

def binary_hill_climb(h_init, n, deadline, max_iters=200000):
    """Hill climbing on binary (0-1) solutions."""
    # Round to binary
    h = np.where(h_init > 0.5, 1.0, 0.0)
    # Fix to have exactly n/2 ones
    num_ones = int(h.sum())
    target_ones = n // 2
    if num_ones > target_ones:
        ones = np.where(h == 1.0)[0]
        to_remove = rng_module.choice(ones, size=num_ones - target_ones, replace=False)
        h[to_remove] = 0.0
    elif num_ones < target_ones:
        zeros = np.where(h == 0.0)[0]
        to_add = rng_module.choice(zeros, size=target_ones - num_ones, replace=False)
        h[to_add] = 1.0
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Pick random swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng_module.randint(len(ones))]
        b = zeros[rng_module.randint(len(zeros))]
        
        h[a] = 0.0
        h[b] = 1.0
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
    
    return best_h, best_c

def continuous_local_search(h, n, deadline):
    """Swap-based local search for continuous h."""
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                break
            
            # Try moving mass from high to low at the worst lag
            if lag > 0:
                for i in range(n - m):
                    j = i + m
                    if h[i] > h[j] + 1e-6:
                        delta = min(d, (h[i] - h[j]) * 0.5)
                        h[i] -= delta
                        h[j] += delta
                        if compute_c5(h, n) < best_c - 1e-14:
                            best_c = compute_c5(h, n)
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-6:
                        delta = min(d, (h[j] - h[i]) * 0.5)
                        h[i] += delta
                        h[j] -= delta
                        if compute_c5(h, n) < best_c - 1e-14:
                            best_c = compute_c5(h, n)
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
            else:
                for i in range(n - m):
                    j = i + m
                    if h[j] > h[i] + 1e-6:
                        delta = min(d, (h[j] - h[i]) * 0.5)
                        h[j] -= delta
                        h[i] += delta
                        if compute_c5(h, n) < best_c - 1e-14:
                            best_c = compute_c5(h, n)
                            improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
                    elif h[i] > h[j] + 1e-6:
                        delta = min(d, (h[i] - h[j]) * 0.5)
                        h[j] += delta
                        h[i] -= delta
                        if compute_c5(h, n) < best_c - 1e-14:
                            best_c = compute_c5(h, n)
                            improved = True
                        else:
                            h[j] -= delta
                            h[i] += delta
    return h, best_c

rng_module = np.random.RandomState(42)

def optimize_at_n(h_start, n, deadline, rng):
    """Full optimization pipeline at a given resolution."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Projected subgradient (fast, many iterations)
    if time.time() < deadline - 20:
        t1_end = min(time.time() + 60, deadline - 30)
        h_sg, c_sg = projected_subgradient(best_h, n, t1_end, alpha=500, top_k=20,
                                            base_lr=0.03, momentum=0.85)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    
    # Phase 2: LP outer approximation
    if time.time() < deadline - 25:
        t2_end = min(time.time() + 60, deadline - 25)
        h_lp, c_lp = lp_outer_approx(best_h, n, t2_end, max_iters=300)
        if c_lp < best_c:
            best_c = c_lp
            best_h = h_lp
    
    # Phase 3: SLSQP polish
    if time.time() < deadline - 15:
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_optimize(best_h, n, alpha, top_k=8, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 4: More subgradient with higher alpha
    if time.time() < deadline - 15:
        t4_end = min(time.time() + 30, deadline - 15)
        h_sg2, c_sg2 = projected_subgradient(best_h, n, t4_end, alpha=5000, top_k=8,
                                              base_lr=0.01, momentum=0.9)
        if c_sg2 < best_c:
            best_c = c_sg2
            best_h = h_sg2
    
    # Phase 5: Local search
    if time.time() < deadline - 5:
        h_ls, c_ls = continuous_local_search(best_h, n, min(deadline - 2, time.time() + 20))
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def try_binary(h_start, n, deadline, rng):
    """Try binary solution and then relax."""
    global rng_module
    rng_module = rng
    h_bin, c_bin = binary_hill_climb(h_start, n, deadline, max_iters=100000)
    # Now use the binary solution as starting point for continuous optimization
    if time.time() < deadline - 10:
        h_cont, c_cont = optimize_at_n(h_bin, n, min(deadline - 2, time.time() + 40), rng)
        if c_cont < c_bin:
            return h_cont, c_cont
    return h_bin, c_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    global rng_module
    rng_module = rng
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Phase A: Optimize at initial resolution
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Phase B: Try other resolutions
    n_candidates = [200, 250, 400, 500, 600]
    if n_init in n_candidates:
        n_candidates.remove(n_init)
    
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(80, remaining * 0.2)
        
        if best_h is not None and len(best_h) >= n_try:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase C: Try binary approach at best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        h_bin, c_bin = binary_hill_climb(best_h, n, time.time() + 30, max_iters=50000)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        # Also try starting from random binary
        h_rand = np.concatenate([np.ones(n//2), np.zeros(n - n//2)])
        rng.shuffle(h_rand)
        h_bin2, c_bin2 = binary_hill_climb(h_rand, n, time.time() + 20, max_iters=30000)
        if c_bin2 < best_c:
            best_c = c_bin2
            best_h = h_bin2
    
    # Phase D: Multiple restarts with perturbation
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(35, (deadline - time.time() - 25) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase E: Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 500000, top_k=5, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Final subgradient
        if time.time() < deadline - 5:
            h_sg, c_sg = projected_subgradient(best_h, n, deadline - 3,
                                                alpha=10000, top_k=5,
                                                base_lr=0.005, momentum=0.95)
            if c_sg < best_c:
                best_c = c_sg
                best_h = h_sg
        # Final local search
        if time.time() < deadline - 2:
            best_h, best_c = continuous_local_search(best_h, n, deadline - 1)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
```
