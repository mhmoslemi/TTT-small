import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-17:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh - mx))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    if mu > 0:
        g = g + 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh - mx))) / alpha
    return float(val)

def _slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_multi_alpha(h, n, deadline, alphas, max_iters_each=150, mu_base=1.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        mu = mu_base * (alpha / 1000.0)
        if mu > 100.0:
            mu = 100.0
        try:
            t_start = time.time()
            res = minimize(_obj, h, jac=_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters_each, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters_each * 12,
                                   'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=1000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    except Exception:
        pass
    return best_h, best_c

def projected_gradient_descent(h, n, deadline, alpha=50000, lr_init=0.002, max_iters=5000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g = project_feasible(h - lr * g, n) - h
        # Check if the projected step is meaningful
        step_norm = np.linalg.norm(g)
        if step_norm < 1e-16:
            break
        # Backtracking
        improved = False
        for _ in range(30):
            h_new = project_feasible(h + lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-17:
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new.copy()
                    stall_count = 0
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            stall_count += 1
            if stall_count > 50 or lr < 1e-15:
                break
        else:
            if it % 50 == 0:
                lr = min(lr * 1.1, 0.01)
    return best_h, best_c

def pair_search_opt(h, n, deadline, steps, radius=60):
    best_c = compute_c5(h, n)
    for d in steps:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def gradient_guided_pairs(h, n, deadline, alpha=5000, step=0.003, max_rounds=500):
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        # Find pairs where moving mass from high-grad to low-grad helps
        # d(obj) ≈ d*(g[i] - g[j]) for h[i]+=d, h[j]-=d
        # Want g[i] < g[j] (decrease obj)
        order = np.argsort(g)
        improved = False
        # Try a few of the best pairs
        for idx in range(min(20, n)):
            i = order[idx]
            j = order[~order.np.argmax(g[order]) if False else ~order.np.argmax(g[order]) if False else 0]
        # Simpler: try pairs (low_grad, high_grad)
        n_try = min(30, n // 2)
        for a in range(n_try):
            if time.time() > deadline:
                return h, best_c
            i = order[a]
            for b in range(n - n_try, n):
                if b - i < 2:
                    continue
                j = order[b]
                move = step
                if h[i] + move <= 1.0 and h[j] - move >= 0.0:
                    h[i] += move; h[j] -= move
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        a = 0  # restart scan
                        break
                    else:
                        h[i] -= move; h[j] += move
                break  # try one high-grad partner per low-grad index
        if not improved:
            step *= 0.7
            if step < 1e-6:
                break
    return h, best_c
