from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Project to binary: top n/2 values become 1."""
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n//2]
    h_bin[idx] = 1.0
    return h_bin

def binary_greedy(h, n, deadline, max_rounds=200):

    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_all = compute_all_lags(h, n)
        # Get top lags (most violated)
        top_lags = np.argsort(c_all)[-10:]
        
        improved = False
        for k in top_lags:
            if time.time() > deadline:
                break
            lag = k - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            # Find (1,0) pairs at this lag
            if lag > 0:
                # h[i]=1, h[i+lag]=0, i from 0 to n-1-lag
                mask = (h[:n-m] == 1) & (h[m:] == 0)
            else:

                # So we want h[i]=1, h[i-m]=0, i from m to n-1

                # In terms of the correlation: the pair is (j-m, j) with h[j-m]=0, h[j]=1

                else:
                    j = i - m

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h * (1 + diff / tot), 0, 1)
    return h

    idx = np.argsort(h)[::-1][:n//2]
    h_bin[idx] = 1.0
    return h_bin

def get_candidates_at_lag(h, n, lag):
    """Get indices i such that h[i]=1 and h[i+lag]=0 (for lag>0)
    or h[i]=1 and h[i-lag]=0 (for lag<0)."""
    m = abs(lag)
    if m >= n:
        return np.array([], dtype=int)
    if lag > 0:
        mask = (h[:n-m] == 1) & (h[m:] == 0)
        return np.where(mask)[0]
    else:
        # h[i]=1, h[i+lag]=0 where lag<0, so h[i-m]=0
        # i ranges from m to n-1
        mask = (h[m:] == 1) & (h[:n-m] == 0)
        return np.where(mask)[0] + m

def greedy_binary(h, n, deadline, verbose=False):
    """Greedy binary local search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    no_improve = 0
    
    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        # Get top 5 lags
        top_ks = np.argsort(c_all)[-5:]
        
        improved = False
        for k in top_ks:
            if time.time() > deadline:
                break
            lag = k - (n - 1)
            if lag == 0:
                continue

                improved = True
                no_improve = 0

    return best_h, best_c

    current_c = compute_c5(h, n)
    best_c = current_c
    best_h = h.copy()

                best_h = h.copy()

def gradient_polish(h, n, deadline, alpha=500):
    """Gradient-based polishing for near-binary solutions."""
    dx = 2.0 / n
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(best_h, n)
    v = np.zeros(n)
    lr = 0.5
    momentum = 0.9

        if time.time() > deadline:
            break
        
        # Compute smooth max gradient
        c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-10:
                continue
            lag = k - (n-1)
            mm = abs(lag)
            if mm >= n:
                continue

                grad[:] += w[k] * (1.0 - 2.0 * h_curr)
            elif lag > 0:
                grad[mm:n] += w[k] * (1.0 - h_curr[:n-mm])
                grad[:n-mm] += w[k] * (-h_curr[mm:])
            else:
                grad[:n-mm] += w[k] * (1.0 - h_curr[mm:])

        v = momentum * v + lr * grad

        c_new = compute_c5(h_curr, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_curr.copy()
        
        # Adaptive learning rate
        if it % 1000 == 0:
            lr = max(0.01, lr * 0.8)
    
    return best_h, best_c

                continue
            cands = candidates[:60]
            best_delta = 1e-15
            best_i = None
            for i in cands:
                j = i + lag
                if j < 0 or j >= n:
                    continue
                h[i], h[j] = h[j], h[i]
                new_c = compute_c5(h, n)
                h[i], h[j] = h[j], h[i]
                delta = best_c - new_c
                if delta > best_delta:
                    best_delta = delta
                    best_i = i
            if best_i is not None:
                j = best_i + lag
                h[best_i], h[j] = h[j], h[best_i]
                best_c = compute_c5(h, n)
                best_h = h.copy()
                improved = True
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    resolutions = sorted(set([n_init] + [350, 400, 450, 500, 550, 600]))

    for n in resolutions:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break

        if h_init is not None and n == n_init:
            h = h_init.copy()
        elif best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_old, best_h)
            h = project_feasible(h, n)
        else:
            h = np.random.rand(n) * 0.3 + 0.35
            h = project_feasible(h, n)

        # Phase A: brief gradient to shape
        if remaining > 120:
            gdl = time.time() + min(30, remaining * 0.1)
            h, _ = gradient_polish(h, n, gdl, alpha=200, lr_start=0.8)

        # Phase B: binary + greedy
        h_bin = to_binary(h, n)
        c_bin = compute_c5(h_bin, n)
        if remaining > 80:
            gdl2 = time.time() + min(50, remaining * 0.25)
            h_bin, c_bin = multi_lag_greedy(h_bin, n, gdl2)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

        # Phase C: simulated annealing at this resolution
        remaining = deadline - time.time() - 20
        if remaining > 60:
            sadl = time.time() + min(100, remaining * 0.4)
            h_sa, c_sa = sa_binary(h_bin, n, sadl, rng, T_start=0.02, T_end=0.0005)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.copy()

        # Phase D: greedy on SA result
        remaining = deadline - time.time() - 15
        if remaining > 30:
            gdl3 = time.time() + min(40, remaining * 0.5)
            h_g2, c_g2 = multi_lag_greedy(h_sa, n, gdl3)
            if c_g2 < best_c:
                best_c = c_g2
                best_h = h_g2.copy()

        # Phase E: iterated local search
        remaining = deadline - time.time() - 10
        if remaining > 40:
            ildl = time.time() + min(80, remaining * 0.6)
            h_il, c_il = iterated_ls(h_bin, n, ildl, rng, k_pert=10)
            if c_il < best_c:
                best_c = c_il
                best_h = h_il.copy()

    # Final phase: work hard at the best resolution
    n_best = len(best_h)
    if time.time() < deadline - 40:
        remaining = deadline - time.time() - 10
        # More SA at best resolution
        if remaining > 50:
            sadl = time.time() + min(remaining * 0.5, 200)
            h_sa2, c_sa2 = sa_binary(best_h, n_best, sadl, rng, T_start=0.01, T_end=0.0002)
            if c_sa2 < best_c:
                best_c = c_sa2
                best_h = h_sa2.copy()

        # Greedy polish
        if time.time() < deadline - 15:
            h_g3, c_g3 = multi_lag_greedy(best_h, n_best, deadline - 5)
            if c_g3 < best_c:
                best_c = c_g3
                best_h = h_g3.copy()

        # Gradient polish on final
        if time.time() < deadline - 10:
            h_pol, c_pol = gradient_polish(best_h, n_best, deadline - 2, alpha=2000, lr_start=0.3)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol.copy()

        # Final greedy
        if time.time() < deadline - 3:
            h_fin = to_binary(best_h, n_best)
            h_fin, c_fin = greedy_binary(h_fin, n_best, deadline - 1, max_no_improve=20)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
