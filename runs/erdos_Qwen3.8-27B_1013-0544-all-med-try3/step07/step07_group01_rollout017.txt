def _sm_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def _sm_obj_pen(x, n, alpha, mu):
    return _sm_obj(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _sm_grad_pen(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _eq_con(x, n):
    return x.sum() - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_obj_pen, h, jac=_sm_grad_pen, args=(n, alpha, 5.0),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                               'maxfun': maxiter * 25, 'maxls': 100})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        h = project_feasible(h, n)
        return h, compute_c5(h, n)

def slsqp_optimize(h, n, deadline, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_sm_obj, h, jac=grad_smooth_max, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 100})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        h = project_feasible(h, n)
        return h, compute_c5(h, n)

def proj_grad_optimize(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 30000, 100000, 300000, 1000000]
    lr = 0.004
    mu = 0.5
    v = np.zeros(n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        for _ in range(5000):
            if time.time() > deadline:
                break
            g = grad_smooth_max(h, n, alpha)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-16:
                break
            v = mu * v - lr * g
            improved = False
            for scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
                h_new = project_feasible(h + scale * v, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    h = h_new
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                mu = max(0.3, mu - 0.03)
                if lr < 1e-12:
                    break
            else:
                lr = min(lr * 1.03, 0.015)
                mu = min(mu + 0.01, 0.85)
    return best_h, best_c

def subgrad_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.003
    for _ in range(50000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        k_max = int(np.argmax(c))
        k = k_max - (n - 1)
        dx = 2.0 / n
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[0:nv] += dx * (1.0 - h[k:n])
            grad[k:n] += dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += dx * (1.0 - h[0:nv])
            grad[0:nv] += dx * (-h[kk:n])
        improved = False
        for scale in [lr, lr * 0.5, lr * 0.1]:
            h_new = project_feasible(h - scale * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-17:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
        if not improved:
            lr *= 0.9
            if lr < 1e-12:
                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = 0 if radius is None else max(0, i - radius)
                hi = n if radius is None else min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-9 and h[j] - d >= -1e-9:
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-9 and h[j] + d <= 1.0 + 1e-9:
                        h[i] -= d
                        h[j] += d
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return best_h, best_c

def gradient_guided_pair(h, n, deadline, step=0.001):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in [200, 1000, 5000, 20000]:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    move = step * (g[j] - g[i])
                    if abs(move) < 1e-13:
                        continue
                    if h[i] + move >= -1e-9 and h[i] + move <= 1.0 + 1e-9 and \
                       h[j] - move >= -1e-9 and h[j] - move <= 1.0 + 1e-9:
                        h[i] += move
                        h[j] -= move
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return best_h, best_c

def multi_alpha_slsqp(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [20000, 100000, 500000, 2000000, 10000000]
    for a in alphas:
        if time.time() > deadline:
            break
        alloc = min(60, deadline - time.time() - 5)
        if alloc < 10:
            break
        h_n, c_n = slsqp_optimize(best_h, n, time.time() + alloc, alpha=a, maxiter=2000)
        if c_n < best_c:
            best_c = c_n
            best_h = h_n
    return best_h, best_c

def try_resolution(h_best, best_c, n_best, deadline, n_try, alloc):
    x_old = np.linspace(0, 2, n_best)
    x_new = np.linspace(0, 2, n_try)
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_feasible(h_interp, n_try)
    # Quick L-BFGS
    t_sub = time.time() + alloc * 0.4
    h_t, c_t = lbfgs_optimize(h_interp, n_try, t_sub, alpha=50000, maxiter=500)
    if c_t < best_c:
        best_c = c_t
        h_best = h_t
        n_best = n_try
    # SLSQP if time
    if time.time() < time.time() + alloc * 0.5 and alloc * 0.5 > 15:
        t_sub2 = t_sub + alloc * 0.3
        h_t2, c_t2 = slsqp_optimize(h_best, n_best, t_sub2, alpha=200000, maxiter=1500)
        if c_t2 < best_c:
            best_c = c_t2
            h_best = h_t2
    return h_best, best_c, n_best

def random_restart(h, n, seed, deadline, proj_scale=0.02):
    rng = np.random.RandomState(seed)
    h_pert = h + rng.randn(n) * proj_scale
    h_pert = project_feasible(h_pert, n)
    best_c = compute_c5(h, n)
    h_n, c_n = lbfgs_optimize(h_pert, n, min(time.time() + 80, deadline), alpha=50000, maxiter=1000)
    if c_n < best_c:
        return h_n, c_n
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 300
        h0 = np.full(n, 0.5)

    h = project_feasible(h0, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n

    # Phase 1: L-BFGS-B with increasing alpha (fast full-space optimization)
    alpha_schedule = [(100, 200, 20000.0), (500, 500, 20000.0), (2000, 1000, 20000.0),
                      (10000, 2000, 20000.0), (50000, 3000, 20000.0)]
    for a, mi, t_alloc in alpha_schedule:
        if time.time() > deadline - 200:
            break
        t_end = min(time.time() + t_alloc, deadline - 200)
        h_n, c_n = lbfgs_optimize(best_h, best_n, t_end, alpha=a, maxiter=mi)
        if c_n < best_c - 1e-16:
            best_c = c_n
            best_h = h_n

    # Phase 2: SLSQP with multiple alphas (exact constraint)
    if time.time() < deadline - 150:
        best_h, best_c = multi_alpha_slsqp(best_h, best_n, deadline - 150)

    # Phase 3: Projected gradient with momentum
    if time.time() < deadline - 120:
        t3 = min(time.time() + 100, deadline - 120)
        h3, c3 = proj_grad_optimize(best_h, best_n, t3)
        if c3 < best_c - 1e-16:
            best_c = c3
            best_h = h3

    # Phase 4: Subgradient refinement (exact max)
    if time.time() < deadline - 80:
        t4 = min(time.time() + 40, deadline - 80)
        h4, c4 = subgrad_refine(best_h, best_n, t4)
        if c4 < best_c - 1e-17:
            best_c = c4
            best_h = h4

    # Phase 5: Gradient-guided pair search
    if time.time() < deadline - 60:
        t5 = min(time.time() + 60, deadline - 60)
        h5, c5 = gradient_guided_pair(best_h, best_n, t5, step=0.001)
        if c5 < best_c - 1e-17:
            best_c = c5
            best_h = h5

    # Phase 6: Local pair search (moderate radius)
    if time.time() < deadline - 40:
        t6 = min(time.time() + 50, deadline - 40)
        h6, c6 = pair_search(best_h, best_n, t6,
                             deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                             radius=80)
        if c6 < best_c - 1e-17:
            best_c = c6
            best_h = h6

    # Phase 7: Try different resolutions
    if time.time() < deadline - 30:
        n_cur = best_n
        for n_try in [200, 400, 250, 350]:
            if n_try == n_cur:
                continue
            if time.time() > deadline - 25:
                break
            alloc = min(40, deadline - time.time() - 10)
            if alloc < 20:
                break
            best_h, best_c, best_n = try_resolution(best_h, best_c, best_n,
                                                     deadline, n_try, alloc)

    # Phase 8: Final polish - L-BFGS + SLSQP + pair search on best
    if time.time() < deadline - 5:
        n_f = best_n
        if time.time() < deadline - 4:
            h_f, c_f = lbfgs_optimize(best_h, n_f, deadline - 4, alpha=200000, maxiter=500)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
        if time.time() < deadline - 2:
            h_f2, c_f2 = pair_search(best_h, n_f, deadline - 1,
                                     deltas=[0.0001, 0.00005, 0.00002, 0.00001],
                                     radius=50)
            if c_f2 < best_c:
                best_c = c_f2
                best_h = h_f2

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
