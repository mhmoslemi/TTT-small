    dx = 2.0 / n
    G = np.zeros((n, 2*n-1))
    for s_idx in range(2*n-1):
        shift = s_idx - (n-1)
        if shift >= 0:
            # j - shift >= 0 means j >= shift
            # j + shift < n means j < n - shift
            # term 1: (1-h[j-shift]) for j in [shift, n-1]
            # term 2: -h[j+shift] for j in [0, n-1-shift]
            G[shift:, s_idx] += (1.0 - h[:n-shift]) * dx  # term 1
            G[:n-shift, s_idx] -= h[shift:] * dx  # term 2

            # j - shift < n means j < n - sh
            # term 1: (1-h[j+sh]) for j in [0, n-1-sh]
            # j + shift = j - sh < n always (for j < n)
            # j + shift >= 0 means j >= sh
            # term 2: -h[j-sh] for j in [sh, n-1]
            G[:n-sh, s_idx] += (1.0 - h[sh:]) * dx  # term 1
            G[sh:, s_idx] -= h[:n-sh] * dx  # term 2

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, n):
    dx = 2.0 / n
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        s = m - (n - 1)
        if s == 0:
            G[:, m] = (1.0 - 2.0 * h) * dx
        elif s > 0:
            G[s:, m] += (1.0 - h[:n - s]) * dx
            G[:n - s + 1, m] -= h[s:] * dx
        else:
            sh = -s
            G[:n - sh + 1, m] += (1.0 - h[sh:]) * dx
            G[sh:, m] -= h[:n - sh] * dx
    return G

def line_search_ternary(f, b, dmax):
    if dmax < 1e-14:
        return 0.0, f.max()
    a, c = 0.0, dmax
    f_max = f.max()
    for _ in range(35):
        m1 = a + (c - a) / 3.0
        m2 = c - (c - a) / 3.0
        v1 = np.max(f + m1 * b)
        v2 = np.max(f + m2 * b)
        if v1 < v2:
            c = m2
        else:
            a = m1
    delta = (a + c) / 2.0
    f_new = np.max(f + delta * b)
    return delta, f_new

def slsqp_smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    val = np.sum(np.exp(shifted))
    if val > 0:
        return float(m + np.log(val) / alpha)
    return float(m)

def slsqp_sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def slsqp_sum_con_jac(x, n):
    return np.ones(n)

def run_slsqp(h, n, deadline, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_sum_con, 'jac': slsqp_sum_con_jac, 'args': (n,)}]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(slsqp_smooth_obj, best_h, args=(n, a), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            ht = project_feasible(res.x, n)
            c = compute_c5(ht, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = ht
        except Exception:
            pass
    return best_h, best_c

def one_sweep(h, n, deadline, rng, radius=None):
    f = compute_f(h, n)
    G = compute_grad(h, n)
    c5 = f.max()
    n_imp = 0
    i_arr, j_arr = np.triu_indices(n, k=1)
    if radius is not None:
        mask = np.abs(i_arr.astype(float) - j_arr.astype(float)) <= radius
        i_arr = i_arr[mask]
        j_arr = j_arr[mask]
    perm = rng.permutation(len(i_arr))
    i_sh = i_arr[perm]
    j_sh = j_arr[perm]
    update_count = 0
    for idx in range(len(i_sh)):
        if time.time() > deadline:
            break
        i = i_sh[idx]
        j = j_sh[idx]
        for direction in (1, -1):
            if direction == 1:
                dmax = min(h[i], 1.0 - h[j])
            else:
                dmax = min(h[j], 1.0 - h[i])
            if dmax < 1e-14:
                continue
            b = direction * (G[i, :] - G[j, :])
            f_cur = f.max()
            delta, f_new = line_search_ternary(f, b, dmax)
            if delta > 1e-12 and f_new < f_cur - 1e-14:
                h[i] += direction * delta
                h[j] -= direction * delta
                f = f + delta * b
                n_imp += 1
                update_count += 1
                if update_count % 80 == 0:
                    f = compute_f(h, n)
                    G = compute_grad(h, n)
    f = compute_f(h, n)
    c5 = f.max()
    return h, f, c5, n_imp

def subgradient_refine(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    f = compute_f(best_h, n)
    G = compute_grad(best_h, n)
    lr = 0.001
    n_iter = 0
    n_total = 5000
    while n_iter < n_total and time.time() < deadline:
        n_iter += 1
        lr = 0.001 * (0.999 ** n_iter)
        top_k = np.argsort(f)[-3:]
        grad = G[:, top_k].mean(axis=1)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        h_new = project_feasible(best_h - lr * grad, n)
        f_new = compute_f(h_new, n)
        c_new = f_new.max()
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            f = f_new
            G = compute_grad(best_h, n)
    return best_h, best_c

def near_pair_search(h, n, deadline, rng, radius=60):
    f = compute_f(h, n)
    G = compute_grad(h, n)
    c5 = f.max()
    n_total_imp = 0
    for _sweep in range(500):
        if time.time() > deadline:
            break
        i_arr, j_arr = np.triu_indices(n, k=1)
        mask = np.abs(i_arr.astype(float) - j_arr.astype(float)) <= radius
        i_arr = i_arr[mask]
        j_arr = j_arr[mask]
        if len(i_arr) == 0:
            continue
        perm = rng.permutation(len(i_arr))
        i_sh = i_arr[perm]
        j_sh = j_arr[perm]
        update_count = 0
        n_imp = 0
        for idx in range(len(i_sh)):
            if time.time() > deadline:
                return h, c5, n_total_imp
            i = i_sh[idx]
            j = j_sh[idx]
            for direction in (1, -1):
                if direction == 1:
                    dmax = min(h[i], 1.0 - h[j])
                else:
                    dmax = min(h[j], 1.0 - h[i])
                if dmax < 1e-14:
                    continue
                b = direction * (G[i, :] - G[j, :])
                f_cur = f.max()
                delta, f_new = line_search_ternary(f, b, dmax)
                if delta > 1e-12 and f_new < f_cur - 1e-14:
                    h[i] += direction * delta
                    h[j] -= direction * delta
                    f = f + delta * b
                    n_imp += 1
                    n_total_imp += 1
                    update_count += 1
                    if update_count % 100 == 0:
                        f = compute_f(h, n)
                        G = compute_grad(h, n)
        f = compute_f(h, n)
        c5 = f.max()
        if n_imp == 0:
            break
    return h, c5, n_total_imp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    n = 250
    h = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        if n_init != n:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_old, h_init)
        else:
            h = h_init.copy()
        h = project_feasible(h, n)
    except Exception:
        h = np.full(n, 0.5)
        h = project_feasible(h, n)

    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warm-up (15% of time)
    t_phase1 = t0 + (deadline - t0) * 0.15
    if time.time() < t_phase1:
        h_opt, c_opt = run_slsqp(best_h, n, t_phase1, maxiter=3000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Full pair search sweeps (50% of time)
    t_phase2 = t0 + (deadline - t0) * 0.62
    n_sweeps = 0
    while time.time() < t_phase2 - 5:
        h_curr, f_curr, c_curr, n_imp = one_sweep(best_h, n, t_phase2 - 5, rng, radius=None)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
        n_sweeps += 1
        if n_imp == 0:
            break
        if n_sweeps % 10 == 0:
            h_sls, c_sls = run_slsqp(best_h, n, min(time.time() + 10, t_phase2 - 10), maxiter=2000)
            if c_sls < best_c:
                best_c = c_sls
                best_h = h_sls

    # Phase 3: Near-pair search for fine tuning (20% of time)
    t_phase3 = t0 + (deadline - t0) * 0.85
    if time.time() < t_phase3:
        h_curr, c_curr, _ = near_pair_search(best_h, n, t_phase3, rng, radius=80)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr

    # Phase 4: Subgradient + restart (7% of time)
    t_phase4 = t0 + (deadline - t0) * 0.92
    if time.time() < t_phase4:
        for restart in range(5):
            if time.time() > t_phase4 - 15:
                break
            scale = 0.02 * (1 + restart * 0.5)
            h_pert = best_h + rng.normal(0, scale, n)
            h_pert = project_feasible(h_pert, n)
            h_sub, c_sub = subgradient_refine(h_pert, n, min(time.time() + 8, t_phase4 - 5), rng)
            if c_sub < best_c:
                best_c = c_sub
                best_h = h_sub
            h_sls, c_sls = run_slsqp(h_sub, n, min(time.time() + 5, t_phase4 - 3), maxiter=3000)
            if c_sls < best_c:
                best_c = c_sls
                best_h = h_sls

    # Phase 5: Final very fine pair search
    if time.time() < deadline - 2:
        h_curr, f_curr, c_curr, _ = one_sweep(best_h, n, deadline - 2, rng, radius=100)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr
        h_curr, c_curr2, _ = near_pair_search(best_h, n, deadline - 1, rng, radius=40)
        if c_curr2 < best_c:
            best_c = c_curr2
            best_h = h_curr

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
