def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):

def delta_for_swap(h, i, j, d, n):
    """Compute delta in correlation at lag d when swapping h[i]:0->1, h[j]:1->0. O(1)."""
    delta = 0.0
    # x = i: h changes 0->1
    t = i + d
    if 0 <= t < n:
        if t == j:
            delta += 1.0
        else:
            delta += 1.0 - h[t]
    # x = j: h changes 1->0
    t = j + d
    if 0 <= t < n:
        if t == i:
            delta += -1.0
        else:
            delta += h[t] - 1.0
    # x = i-d: target changes 0->1 (at position i)
    t = i - d
    if 0 <= t < n and t != i and t != j:
        delta += -h[t]
    # x = j-d: target changes 1->0 (at position j)
    t = j - d
    if 0 <= t < n and t != i and t != j:
        delta += h[t]
    return delta

def fast_binary_local_search(hb, n, deadline, max_sweeps=2000, K=20):
    """Fast binary local search using active lag tracking. O(K) per swap evaluation."""
    dx = 2.0 / n
    h = hb.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_list = np.where(h == 0)[0].tolist()
    n_corr = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if not zeros_list:
            break
        # Find top K lags
        top_lags = np.argsort(corr)[::-1][:K]
        top_vals = corr[top_lags].copy()
        improved = False
        new_zeros = []
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in list(ones_set):
                # Compute delta for top K lags
                est_max = 0.0
                for idx_k, lag in enumerate(top_lags):
                    d = lag - (n - 1)  # convert index to actual lag
                    dk = delta_for_swap(h, i, j, d, n)
                    new_val = top_vals[idx_k] + dk
                    if new_val > est_max:
                        est_max = new_val
                # Also estimate: the lag not in top K could be max
                # Use a conservative bound: second max outside top K
                if K < n_corr:
                    rest_max = corr[np.setdiff1d(np.arange(n_corr), top_lags)].max() if n_corr > K else 0
                else:
                    rest_max = 0
                nc_val = max(est_max, rest_max) * dx
                if nc_val < best_nc - 1e-15:
                    best_nc = nc_val
                    best_j = j
            if best_j >= 0:
                # Perform swap
                h[i] = 1.0
                h[best_j] = 0.0
                ones_set.remove(best_j)
                # Update correlation
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                new_zeros.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        zeros_list = new_zeros if improved else zeros_list
        # Remove already-assigned zeros
        if improved:
            zeros_list = [z for z in zeros_list if h[z] == 0]
        if not improved:
            break
    return best_h, best_c

def anneal_binary_fast(hb, n, deadline, seed=42, max_iters=3000000, T0=0.01, cool=0.99998):
    """Simulated annealing with O(n) delta computation per step."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = hb.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = T0
    m_range = np.arange(-(n - 1), n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 1000 == 0:
            T *= cool
        else:
            T *= 1.0  # keep T constant for small steps to save floating point
            T = T0 * (cool ** (it // 1000))
        
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        
        # Compute full delta using O(n) vectorized approach
        # delta[m] for all lags m
        # Use the formula: for lag d, delta = contributions from positions i, j, i-d, j-d
        # Vectorized: for all d in m_range
        delta = np.zeros(2 * n - 1)
        
        # Contribution from x=i: 1 - h[i+d] for valid i+d
        im = i + m_range
        msk = (im >= 0) & (im < n)
        vals = 1.0 - h[im[msk]]
        vals[im[msk] == j] = 1.0  # special case: i+d == j
        delta[msk] += vals
        
        # Contribution from x=j: h[j+d] - 1 for valid j+d
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        vals2 = h[jm[msk]] - 1.0
        vals2[jm[msk] == i] = -1.0  # special case: j+d == i (h[i]=0, so 0-1=-1, same)
        delta[msk] += vals2
        
        # Contribution from x=i-d: -h[i-d] for valid i-d, i-d not in {i,j}
        imn = i - m_range
        msk = (imn >= 0) & (imn < n) & (imn != i) & (imn != j)
        delta[msk] += -h[imn[msk]]
        
        # Contribution from x=j-d: h[j-d] for valid j-d, j-d not in {i,j}
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n) & (jmn != i) & (jmn != j)
        delta[msk] += h[jmn[msk]]
        
        new_corr = corr + delta
        nc = new_corr.max() * dx
        
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(-3000.0 * (nc - cur_c) / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c

def pgd_continuous(h, n, alphas, deadline, max_iter=3000):
    """Projected gradient descent with softmax approximation of max."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c, _ = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        cur_c = best_c
        step = min(float(n) / (4.0 * alpha), 0.3)
        
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            # Compute log-sum-exp gradient
            c = np.correlate(h_cur, 1 - h_cur, mode='full')
            a = alpha * c
            sm = a.max()
            exp_c = np.exp(a - sm)
            exp_c_sum = exp_c.sum()
            if exp_c_sum < 1e-30:
                break
            p = exp_c / exp_c_sum
            gnorm2 = np.sum(p ** 2)
            if gnorm2 < 1e-20:
                break
            
            # Gradient: d/dc [sum p[m] * c[m]] where c[m] = sum_x h[x]*(1-h[x+d_m])
            # = sum_m p[m] * d c[m]/d h
            # dc[m]/dh[x] = (1-h[x+d_m]) for the h[x] term - h[x-d_m] for the h in target
            # Simplified: grad[x] = sum_m p[m] * [(1-h[x+dm]) - h[x-dm]]
            grad = np.zeros(n)
            m_range = np.arange(-(n - 1), n)
            for idx_k in range(len(p)):
                if p[idx_k] < 1e-14:
                    continue
                d = idx_k - (n - 1)
                pk = p[idx_k]
                # h[x] * (1-h[x+d]): derivative wrt h[x] is (1-h[x+d])
                im = np.arange(max(0, -d), min(n, n - d))
                tgt = im + d
                grad[im] += pk * (1.0 - h_cur[tgt])
                # h[x-d] * (1-h[x]): derivative wrt h[x] is -h[x-d]
                im2 = np.arange(max(0, d), min(n, n + d))
                src = im2 - d
                grad[im2] -= pk * h_cur[src]
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            direction = -grad / gnorm
            
            accepted = False
            for ls in range(30):
                h_new = project(h_cur + step * (0.5 ** ls) * direction, n)
                c_new, _ = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    h_cur = h_new
                    cur_c = c_new
                    step *= (0.5 ** ls)
                    accepted = True
                    break
            if not accepted:
                # Try smaller step
                step *= 0.5
                if step < 1e-6:
                    break
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_cur.copy()
    
    return best_h, best_c

def make_block_pattern(n, seed):
    rng = np.random.RandomState(seed)
    hb = np.zeros(n)
    pos = 0
    while pos < n:
        if pos + n // 4 <= n:
            blen = rng.randint(n // 8, n // 4 + 1)
            blen = min(blen, n - pos)
            if rng.random() < 0.5:
                hb[pos:pos + blen] = 1.0
            pos += blen
        else:
            break
    # Adjust to have exactly n/2 ones
    n_ones = int(hb.sum())
    if n_ones < n // 2:
        zeros_idx = np.where(hb == 0)[0]
        if len(zeros_idx) > 0:
            need = n // 2 - n_ones
            pick = rng.choice(len(zeros_idx), size=min(need, len(zeros_idx)), replace=False)
            hb[zeros_idx[pick]] = 1.0
    elif n_ones > n // 2:
        ones_idx = np.where(hb == 1)[0]
        need = n_ones - n // 2
        pick = rng.choice(len(ones_idx), size=min(need, len(ones_idx)), replace=False)
        hb[ones_idx[pick]] = 0.0
    return hb

        best_h = hb2.copy()
        best_n = n
    
    # Phase 3: Annealing at n=400 (250s)
    p3 = t0 + 400
    ha3 = to_binary(best_h, n)
    ha3, ca3 = anneal_binary_fast(ha3, n, p3, seed=seed, max_iters=3000000, T0=0.015, cool=0.99997)
    if ca3 < best_c:
        best_c = ca3
        best_h = ha3.copy()
        best_n = n
    
    # Local search after annealing
    p3b = t0 + 450
    hb3b = to_binary(best_h, n)
    hb3b, cb3b = fast_binary_local_search(hb3b, n, p3b, max_sweeps=3000, K=30)
    if cb3b < best_c:
        best_c = cb3b
        best_h = hb3b.copy()
        best_n = n
    
    # Phase 4: Try n=512 (350s)
    p4_start = t0 + 460
    p4_end = t0 + 810
    if time.time() < p4_end - 60:
        n512 = 512
        # Interpolate from best solution
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, n512)
        hh512 = np.interp(xn, xo, best_h)
        hh512 = project(hh512, n512)
        
        # PGD at 512
        p4a = min(p4_end - 200, t0 + 560)
        hh512, ch512 = pgd_continuous(hh512, n512, [100, 500, 2000], p4a, max_iter=3000)
        
        # Binary + local search
        hb512 = to_binary(hh512, n512)
        p4b = min(p4_end - 100, t0 + 660)
        hb512, cb512 = fast_binary_local_search(hb512, n512, p4b, max_sweeps=3000, K=30)
        
        # Annealing
        p4c = min(p4_end - 10, t0 + 780)
        ha512, ca512 = anneal_binary_fast(hb512, n512, p4c, seed=seed + 1, max_iters=2000000, T0=0.01, cool=0.99998)
        
        # Local search after annealing
        hb512b = to_binary(ha512, n512)
        hb512b, cb512b = fast_binary_local_search(hb512b, n512, p4_end - 5, max_sweeps=1000, K=25)
        
        c512 = min(cb512, ca512, cb512b)
        h512 = hb512b if cb512b == c512 else (ha512 if ca512 == c512 else hb512)
        if c512 < best_c:
            best_c = c512
            best_h = h512.copy()
            best_n = n512
        else:
            # Try random start at 512
            if time.time() < p4_end - 30:
                rng2 = np.random.RandomState(seed + 100)
                hb_rand = make_random_binary(n512, rng2)
                hb_rand, cb_rand = fast_binary_local_search(hb_rand, n512, min(p4_end - 10, time.time() + 40), max_sweeps=2000, K=25)
                if cb_rand < best_c:
                    best_c = cb_rand
                    best_h = hb_rand.copy()
                    best_n = n512
    
    # Phase 5: Try n=600 (200s)
    p5_end = t0 + 1010
    if time.time() < p5_end - 40:
        n600 = 600
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, n600)
        hh600 = np.interp(xn, xo, best_h)
        hh600 = project(hh600, n600)
        
        p5a = min(p5_end - 60, time.time() + 60)
        hh600, ch600 = pgd_continuous(hh600, n600, [200, 1000], p5a, max_iter=2000)
        
        hb600 = to_binary(hh600, n600)
        p5b = min(p5_end - 20, time.time() + 80)
        hb600, cb600 = fast_binary_local_search(hb600, n600, p5b, max_sweeps=3000, K=30)
        
        p5c = min(p5_end - 5, time.time() + 60)
        ha600, ca600 = anneal_binary_fast(hb600, n600, p5c, seed=seed + 2, max_iters=1500000, T0=0.012, cool=0.99998)
        
        hb600b = to_binary(ha600, n600)
        hb600b, cb600b = fast_binary_local_search(hb600b, n600, p5_end - 2, max_sweeps=1000, K=25)
        
        c600 = min(cb600, ca600, cb600b)
        h600 = hb600b if cb600b == c600 else (ha600 if ca600 == c600 else hb600)
        if c600 < best_c:
            best_c = c600
            best_h = h600.copy()
            best_n = n600
    
    # Phase 6: Final PGD polish (remaining time)
    if time.time() < deadline - 5:
        best_h, best_c = pgd_continuous(best_h, best_n, [5000, 20000, 100000], deadline - 3, max_iter=2000)
        # Re-check binary
        if best_n == int(best_n):
            hb_final = to_binary(best_h, best_n)
            hb_final, cb_final = fast_binary_local_search(hb_final, best_n, deadline - 1, max_sweeps=500, K=25)
            if cb_final < best_c:
                best_c = cb_final
                best_h = hb_final.copy()
    
    # Final evaluation
    best_h = project(best_h, best_n)
    best_c, _ = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
