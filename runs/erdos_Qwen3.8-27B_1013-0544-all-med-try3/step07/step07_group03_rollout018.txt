def compute_c5_full(h, n):
    """Returns (c5_value, correlation_array)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_feasible(h, n):
    """Project to {0<=h<=1, sum(h)=n/2} via bisection"""
    target = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Bisection on lambda: h_i = clip(h_i - lambda, 0, 1), find lambda s.t. sum = target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(h - lam, 0.0, 1.0)
    # Tiny residual fixup
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, n, beta, step_size):
    """One subgradient step with projection"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    
    # Exponential weights
    w = np.exp(beta * (corr - cmax))
    sw = w.sum()
    if sw < 1e-30:
        return h
    w /= sw
    
    # Compute weighted subgradient
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wv = w[k_idx]
        if wv < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            # c_k = sum_{i=0}^{n-1-k} h[i](1-h[i+k]) * dx
            # d/dh[j]: (1-h[j+k]) if j<n-k; -h[j-k] if j>=k
            grad[:n-k] += wv * dx * (1.0 - h[k:n])
            grad[k:] += wv * dx * (-h[:n-k])
        else:
            kk = -k
            # c_{-kk} = sum_{i=0}^{n-1-kk} h[i+kk](1-h[i]) * dx
            # d/dh[j]: (1-h[j-kk]) if j>=kk; -h[j] if j<n-kk
            grad[kk:] += wv * dx * (1.0 - h[:n-kk])
            grad[:n-kk] += wv * dx * (-h[:n-kk])
    
    # Gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

                best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                                  [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=60)
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                                  [0.005, 0.002, 0.001, 0.0005], radius=40)

    # Phase 5: Ultra-fine polish
    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        ht = slsqp_refine(best_h, n, 2000000, maxiter=5000)
        c = compute_c5_full(ht, n)[0]
        if c < best_c:
            best_c = c
            best_h = ht
        if n <= 250 and time.time() < deadline - 0.5:
            best_h, best_c = pair_search(best_h, n, deadline,
                                         [0.0002, 0.0001, 0.00005, 0.00002])
        elif time.time() < deadline - 0.5:
            best_h, best_c = near_pair_search(best_h, n, deadline,
                                              [0.0002, 0.0001], radius=30)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c, _ = compute_c5_full(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c, _ = compute_c5_full(best_h, len(best_h))
    return best_h, best_c, len(best_h)
