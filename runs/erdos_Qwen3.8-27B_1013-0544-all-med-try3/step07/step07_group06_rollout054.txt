def simulated_annealing(h_init, n, deadline, seed=42):
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    
    best_c = cur_c
    best_h = h.copy()
    best_corr = corr.copy()
    
    T0 = 0.1  # initial temperature
    T_end = 0.001
    steps = 500000
    
    for step in range(steps):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (step / steps)
        
        # Propose random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(zeros)
        j = rng.choice(ones)
        
        # Compute delta
        im = i + m_range
        imn = i - m_range
        jm = j + m_range
        jmn = j - m_range
        delta = np.zeros(2 * n - 1)
        msk = (im >= 0) & (im < n)

        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        
        new_corr = corr + delta
        nc = new_corr.max() * dx
        dT = nc - cur_c

def project_feasible(h, n):

def corr_delta(h, i, j, n, m_range):
    im = i + m_range
    imn = i - m_range
