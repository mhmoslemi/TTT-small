            if tot < 1e-16:

            if tot < 1e-16:

            h += (diff / tot) * h

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    scaled = alpha * (c - m)
    max_s = scaled.max()

    log_sum = max_s + np.log(exp_s.sum())
    obj = float(m + log_sum / alpha)
    
    # Gradient computation
    # w has length 2n-1, indexed 0..2n-2
    # For s=0: contribution w[n-1]*dx*(1-2*h[j])
    # For s>0: contribution (w[n-1+s]+w[n-1-s])*dx*[...]
    
    grad = np.zeros(n)
    
    # s=0
    w0 = w[n-1] * dx
    grad += w0 * (1.0 - 2.0 * h)
    
    # s>0
    for s in range(1, n):
        wp = (w[n-1+s] + (w[n-1-s] if n-1-s >= 0 else 0)) * dx
        if wp < 1e-18:
            continue
        # (1-h[j+s]) for j=0..n-1-s
        if n - s > 0:
            grad[0:n-s] += wp * (1.0 - h[s:n])
            # -h[j-s] for j=s..n-1
            grad[s:n] -= wp * h[0:n-s]
    
    return obj, grad

                h_new = h_y - lr * grad

Hmm, the LP constraint for each k is:
c_k(h_new) ≤ C

Linearized: c_k(h) + dc_k/dh|_h · (h_new - h) ≤ C

So: (dc_k/dh|_h) · h_new ≤ C + c_k(h) - (dc_k/dh|_h) · h

The LHS is a dot product of the gradient vector with h_new. This is linear in h_new. Good.

But computing all 2n-1 gradients separately is O(n²(2n)) which is expensive. Let me use the structure:

For each k, dc_k/dh[j] depends only on |k-(n-1)| (the absolute lag). So I only need to compute n distinct gradient vectors (one per lag s=0,...,n-1), and each can be computed in O(n).

Total: O(n²) to compute all gradients. For n=250, that's ~62,500 operations. Fine.

Actually, I already have the formula:
For lag s (s>0): dc/dh[j] = dx * [(1-h[j+s])*I(0≤j≤n-1-s) + (-h[j-s])*I(s≤j≤n-1)]
For s=0: dc/dh[j] = dx * (1-2h[j])

So I can precompute s_grad[s] for s=0..n-1, each of length n.

Then the LP constraint for output index p (lag k = p-(n-1)) is:
s = |k|
s_grad[s] · h_new ≤ C + c[p] - s_grad[s] · h

But wait, for a given s, there are two output indices (p = n-1-s and p = n-1+s) that share the same gradient structure but different c values. So I get 2 constraints per s (except s=0 which has 1).

Total LP constraints: (2n-1) inequality + n box + 1 equality = O(n). 

Each constraint is a row of the form: s_grad[s] · h_cv + (-1)*C_cv ≤ c[p] - s_grad[s]·h

In cvxpy, this is:
s_grad[s] @ h_cv - C_cv <= c_val - s_grad[s] @ h

Or equivalently: s_grad[s] @ h_cv <= C_cv + c_val - s_grad[s] @ h

Let me implement this. But I'm worried about CVXPY overhead for 500 constraints. Let me use a direct LP solver (scipy.optimize.linprog or scipy.optimize.milp).

Actually, scipy.optimize.linprog might be faster for this size. Let me use it.

Variables: [h[0], ..., h[n-1], C] → n+1 variables
Objective: minimize C → c = [0, ..., 0, 1]

Constraints (inequality, A_ub @ x <= b_ub):
- For each p: s_grad[s_p] @ h - C <= c_p - s_grad[s_p] @ h_current
  → [s_grad[s_p], -1] @ x <= c_p - s_grad[s_p] @ h_current
- -h[i] <= 0 → [-e_i, 0] @ x <= 0 (n constraints)
- h[i] <= 1 → [e_i, 0] @ x <= 1 (n constraints)

Equality: sum(h) = n/2
  → [ones(n), 0] @ x = n/2

This is a standard LP with (2n-1 + 2n) inequality constraints and 1 equality constraint, with n+1 variables.

For n=250: 749 inequality constraints, 1 equality, 251 variables. Very manageable for linprog!

Let me use scipy.optimize.linprog with method='highs' (should be available in recent scipy).

import numpy as np
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    scaled = alpha * (c - m)
    mx = scaled.max()
    exp_s = np.exp(scaled - mx)
    w = exp_s / exp_s.sum()
    obj = float(m + (mx + np.log(exp_s.sum())) / alpha)
    grad = np.zeros(n)
    n1 = n - 1
    w0 = w[n1] * dx
    if w0 > 1e-25:
        grad += w0 * (1.0 - 2.0 * h)
    for s in range(1, n):
        ws = w[n1 + s]
        if n1 - s >= 0:
            ws += w[n1 - s]
        ws *= dx
        if ws < 1e-25:
            continue
        ns = n - s
        grad[0:ns] += ws * (1.0 - h[s:n])
        grad[s:n] -= ws * h[0:ns]
    return obj, grad

def proj_grad_solve(h, n, alpha, deadline, max_iter=8000, lr=None):
    if lr is None:
        lr = 1.0 / (alpha ** 0.2)
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    step = lr
    obj_prev = smooth_max_grad(h, n, alpha)[0]
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, grad = smooth_max_grad(h, n, alpha)
        s = step
        for _bt in range(30):
            h_new = project_feasible(h - s * grad, n)
            obj_new, _ = smooth_max_grad(h_new, n, alpha)
            if obj_new < obj + 1e-18:
                break
            s *= 0.5
        else:
            break
        h = h_new
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-16:
            best_c5 = c5
            best_h = h.copy()
        if abs(obj - obj_prev) < 1e-16:
            break
        obj_prev = obj
    return best_h, best_c5

def lp_refine(h, n, deadline, max_iter=80):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vals = np.correlate(h, 1 - h, mode='full') * dx
        # Compute gradient vectors for each lag s
        n1 = n - 1
        # Build A_ub, b_ub for linprog
        # Variables: [h(0),...,h(n-1), C]
        # Minimize C: c_obj = [0,...,0,1]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        rows_a = []
        rows_b = []
        # For each output index p, add: s_grad[s] @ h_var - C <= c_p - s_grad[s] @ h
        for p in range(2 * n - 1):
            s = abs(p - n1)
            g = np.zeros(n)
            if s == 0:
                g[:] = dx * (1.0 - 2.0 * h)
            else:
                ns = n - s
                g[0:ns] += dx * (1.0 - h[s:n])
                g[s:n] -= dx * h[0:ns]
            row = np.zeros(n + 1)
            row[0:n] = g
            row[n] = -1.0
            rhs = c_vals[p] - g @ h
            rows_a.append(row)
            rows_b.append(rhs)
        # Box constraints: -h[i] <= 0
        for i in range(n):
            row = np.zeros(n + 1)
            row[i] = -1.0
            rows_a.append(row)
            rows_b.append(0.0)
        # h[i] <= 1
        for i in range(n):
            row = np.zeros(n + 1)
            row[i] = 1.0
            rows_a.append(row)
            rows_b.append(1.0)
        A_ub = np.array(rows_a)
        b_ub = np.array(rows_b)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 0:n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=[(0, 1)] * n + [(0, None)], method='highs')
        except Exception:
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=[(0, 1)] * n + [(0, None)], method='revised simplex')
            except Exception:
                break
        if res.status != 0:
            break
        h_new = res.x[0:n]
        c_lp = res.x[n]
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # No improvement, try slightly tighter C
            break
    return best_h, best_c5

def pair_search_local(h, n, deadline, deltas, max_passes=100):

                        if c < best_c - 1e-15:
