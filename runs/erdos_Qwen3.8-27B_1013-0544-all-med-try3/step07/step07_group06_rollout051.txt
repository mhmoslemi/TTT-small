            n_cand = min(len(ones_list), top_k + 4)
            top_idx = np.argsort(score)[:top_k]
            rand_idx = rng.choice(len(ones_list), min(4, len(ones_list)), replace=False)
            cand_set = set(np.concatenate([top_idx, rand_idx]).tolist())
            if i in ones_list:
                cand_set.discard(int(np.where(ones_list == i)[0][0]) if i in ones_list else -1)
            candidates = list(cand_set)
            rng.shuffle(candidates)

            for ci in candidates:
                j = int(ones_list[ci])
                if j == i:
                    continue

                jp2 = j + m_offsets
                jm2 = j - m_offsets
                valid_jp2 = (jp2 >= 0) & (jp2 < n)
                valid_jm2 = (jm2 >= 0) & (jm2 < n)

                full_delta = base_delta.copy()
                full_delta[valid_jp2] += h[jp2[valid_jp2]] - 1.0
                full_delta[valid_jm2] += h[jm2[valid_jm2]]

                # Cross-term corrections
                full_delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < total_bins:
                    full_delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < total_bins:
                    full_delta[k2] += 1.0

                new_corr = corr + full_delta
                new_max = float(new_corr.max())

                accept = False
                if new_max < cur_max - 1e-14:
                    accept = True
                elif sa_temp > 0 and new_max > cur_max:
                    delta_obj = (new_max - cur_max) * dx
                    if rng.random() < np.exp(-delta_obj / max(sa_temp, 1e-10)):
                        accept = True

                if accept:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = new_corr
                    cur_max = new_max
                    ones_mask[i] = True
                    ones_mask[j] = False
                    ones_list = np.where(ones_mask)[0]
                    zeros_list = np.where(~ones_mask)[0]
                    if cur_max * dx < best_c:
                        best_c = cur_max * dx
                        best_h = h.copy()
                    found = True
                    no_improve_count = 0
                    break

        sweep += 1
        if not found:
            # Try increasing SA temperature occasionally
            if sa_temp == 0 and sweep > 100:
                pass  # handled by no_improve_count perturbation above

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_budget = budget_s - 5
    deadline = t0 + total_budget
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n_best = n0

    # Phase 1: SLSQP smoothing (15% time)
    p1 = t0 + 0.15 * total_budget
    h = best_h.copy()
    for alpha in [500, 5000, 50000, 500000, 5000000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n0, alpha, 1500, p1)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Primary binary local search (35% time)
    p2 = t0 + 0.50 * total_budget
    hb, cb = binary_local_search(best_h, n0, p2, max_sweeps=10000, seed=seed, top_k=10)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        n_best = n0

    # Phase 3: SLSQP polish + re-binarize (10% time)
    p3 = t0 + 0.60 * total_budget
    if time.time() < p3:
        hp = project_feasible(best_h + 0.05 * rng.randn(n0), n0)
        for alpha in [100000, 1000000, 10000000]:
            if time.time() > p3:
                break
            hp = slsqp_optimize(hp, n0, alpha, 2000, p3)
            c = compute_c5(hp, n0)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
        if time.time() < p3 + 5:
            hb, cb = binary_local_search(best_h, n0, p3 + 5, max_sweeps=1000, seed=seed + 1)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                n_best = n0

    # Phase 4: Multi-n exploration (15% time)
    p4 = t0 + 0.75 * total_budget
    for nh in [200, 220, 240, 250, 260, 280, 300, 320]:
        if time.time() > p4 - 15:
            break
        if nh == n_best:
            continue
        xo = np.linspace(0, 2, n_best, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hh = slsqp_optimize(hh, nh, 100000, 500, p4 - 12)
        ch = compute_c5(hh, nh)
        if time.time() < p4 - 8:
            hb2, cb2 = binary_local_search(hh, nh, p4 - 5, max_sweeps=800, seed=seed + nh, top_k=8)
            if cb2 < ch:
                ch = cb2
                hh = hb2.copy()
        if ch < best_c:
            best_c = ch
            best_h = nh * np.ones(nh)  # placeholder to track
            best_h = hh.copy()
            n_best = nh

    # Phase 5: Extended binary search with SA (30% time)
    p5 = deadline - 15
    # Reset with current best
    h_cur = best_h.copy()
    if len(h_cur) != n_best:
        h_cur = best_h.copy()
    hb, cb = binary_local_search(h_cur, n_best, p5, max_sweeps=20000, seed=seed + 50, top_k=12)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        n_best = n_best

    # Phase 6: Final polish (remaining time)
    if time.time() < deadline - 3:
        h_fin = slsqp_optimize(best_h.copy(), n_best, 50000000, 2000, deadline - 2)
        c = compute_c5(h_fin, n_best)
        if c < best_c:
            best_c = c
            best_h = h_fin.copy()
        if time.time() < deadline - 1:
            hb, cb = binary_local_search(best_h, n_best, deadline, max_sweeps=500, seed=seed + 99, top_k=10)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

    # Final: ensure feasibility and recompute
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
