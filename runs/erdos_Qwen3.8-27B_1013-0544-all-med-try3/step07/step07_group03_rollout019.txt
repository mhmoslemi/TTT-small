import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    a = alpha * (c - m)
    a = np.clip(a, -500, 500)
    mx = a.max()
    e = np.exp(a - mx)
    s = e.sum()
    return float(m + (mx + np.log(s)) / alpha)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    a = alpha * (c - m)
    a = np.clip(a, -500, 500)
    mx = a.max()
    e = np.exp(a - mx)
    s = e.sum()
    w = e / s
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wk = w[p]
        if wk < 1e-15:
            continue
        k = p - (n - 1)
        j_lo = max(0, -k)
        j_hi = min(n - 1, n - 1 - k)
        if j_hi >= j_lo:
            grad[j_lo:j_hi+1] += wk * (1.0 - x[j_lo-k:j_hi-k+1])
            grad[j_lo:j_hi+1] -= wk * x[j_lo+k:j_hi+k+1]
    grad *= dx
    return grad

def slsqp_optimize(h0, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj_func(x):
        return smooth_max_obj(x, n, alpha)
    def obj_jac(x):
        return smooth_max_grad(x, n, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_func, h0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.fun < smooth_max_obj(h0, n, alpha):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0.copy()

def min_max_linear_1d(base, coeff):
    f0 = float(np.max(base))
    f1 = float(np.max(base + coeff))
    active0 = base >= f0 - 1e-13
    if float(np.min(coeff[active0])) >= -1e-11:
        return 0.0
    active1 = (base + coeff) >= f1 - 1e-13
    if float(np.max(coeff[active1])) <= 1e-11:
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = base + mid * coeff
        fmid = float(np.max(vals))
        active = vals >= fmid - 1e-12
        cmin = float(np.min(coeff[active]))
        cmax = float(np.max(coeff[active]))
        if cmin <= 1e-11 and cmax >= -1e-11:
            return mid
        elif cmin > 1e-11:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def coord_descent(h, n, max_iter=300, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            hj = h[j]
            coeff = np.zeros(2 * n - 1)
            coeff[j:j+n] += (1.0 - h)[::-1]
            coeff[n-1-j:2*n-1-j] -= h
            base = c - hj * coeff
            t_opt = min_max_linear_1d(base, coeff)
            if abs(t_opt - hj) > 1e-13:
                delta = t_opt - hj
                h[j] = t_opt
                c += delta * coeff
                c5_new = float(np.max(c) * 2.0 / n)
                if c5_new < best_c:
                    best_c = c5_new
                    improved = True
        if it % 10 == 9:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            bc = compute_c5(h, n)
            if bc < best_c:
                best_c = bc
            else:
                h = h.copy()
        if not improved and it > 2:
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 - 1e-15 and h[j] - d >= 1e-15:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def swap_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    while time.time() < deadline:
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for val in [0.0, 1.0]:
                adj = val - h[j]
                if abs(adj) < 1e-10:
                    continue
                h[j] = val
                h2 = project_feasible(h, n)
                c = compute_c5(h2, n)
                if c < best_c - 1e-14:
                    best_c = c
                    h = h2.copy()
                    improved = True
                else:
                    h[j] = val - adj
        if not improved:
            break
    return h, best_c

def optimize_full(h0, n, deadline):
    h = project_feasible(h0.copy(), n)
    best_c = compute_c5(h, n)

    alphas = [300, 1000, 5000, 20000, 80000, 300000]
    for a in alphas:
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = min(2000, max(300, int(remaining * 2)))
        ht = slsqp_optimize(h, n, a, maxit)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    remaining = deadline - time.time()
    if remaining > 10:
        h, best_c = coord_descent(h, n, max_iter=200, deadline=time.time() + remaining * 0.4)

    remaining = deadline - time.time()
    if remaining > 5:
        if n <= 150:
            h, best_c = pair_search(h, n, time.time() + remaining * 0.5,
                                    [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            h, best_c = pair_search(h, n, time.time() + remaining * 0.5,
                                    [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                    radius=60)

    remaining = deadline - time.time()
    if remaining > 8:
        ht = slsqp_optimize(h, n, 500000, min(3000, max(500, int(remaining))))
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    remaining = deadline - time.time()
    if remaining > 5:
        h, best_c = coord_descent(h, n, max_iter=100, deadline=time.time() + remaining * 0.5)

    remaining = deadline - time.time()
    if remaining > 3:
        if n <= 150:
            h, best_c = pair_search(h, n, time.time() + remaining * 0.7,
                                    [0.0005, 0.0002, 0.0001, 0.00005])
        else:
            h, best_c = pair_search(h, n, time.time() + remaining * 0.7,
                                    [0.0005, 0.0002, 0.0001],
                                    radius=40)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize the initial construction
    alloc1 = min(350, (deadline - t0) * 0.35)
    if alloc1 > 30:
        h_opt, c_opt = optimize_full(h_init, n_init, t0 + alloc1)
        best_h, best_c = h_opt, c_opt

    # Phase 2: Try different n values
    x_init = np.linspace(0, 2, n_init)
    for n_try in sorted(set([120, 160, 200, 250, 300, 350])):
        if time.time() > deadline - 60:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        if n_try <= n_init:
            h_interp = np.interp(x_new, x_init, h_init)
        else:
            h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        remaining = deadline - time.time() - 30
        alloc = min(150, remaining * 0.15)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_full(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_init = n_try
            x_init = np.linspace(0, 2, n_try)

    # Phase 3: Multi-restart with perturbations around best
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 50))
        n_restarts = min(n_restarts, 12)
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scales = [0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6]
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 15) * 0.3)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_full(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Extended local search on best
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        if n <= 150:
            best_h, best_c = pair_search(best_h, n, deadline - 8,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005,
                                          0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 8,
                                         [0.005, 0.002, 0.001, 0.0005,
                                          0.0002, 0.0001],
                                         radius=80)

    # Phase 5: Final SLSQP + coordinate descent polish
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        ht = slsqp_optimize(best_h, n, 1000000, 3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 5:
            best_h, best_c = coord_descent(best_h, n, max_iter=200,
                                           deadline=deadline - 3)
        if time.time() < deadline - 2:
            if n <= 150:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.0001, 0.00005], radius=30)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
