import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] *= (target / h[mask].sum())
    return np.clip(h, 0.0, 1.0)

def correlation_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    k = shift_idx - center
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def softmax_obj_and_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    threshold = 1e-12
    active = np.where(w > threshold)[0]
    for wi, idx in zip(w[active], active):
        k = idx - center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return val, grad

def topk_obj_and_grad(h, n, active_idx, alpha):
    c = compute_c5_vec(h, n)
    vals = c[active_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for wi, idx in zip(w, active_idx):
        if wi < 1e-14:
            continue
        k = idx - center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return val, grad

def adam_optimize(h, n, deadline, alpha_init=50000, alpha_end=500000, lr_init=0.005, top_k=5, tolerance=0.005):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    lr = lr_init
    target = n / 2.0

    while time.time() < deadline - 1:
        t += 1
        frac = min(1.0, t / 5000.0)
        curr_alpha = alpha_init + (alpha_end - alpha_init) * frac
        tol = tolerance * (1.0 - 0.5 * frac)

        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-8, tol * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > max(top_k, 8):
            top_vals = c_vec[active]
            order = np.argsort(-top_vals)[:top_k]
            active = active[order]
        if len(active) < 1:
            active = np.array([np.argmax(c_vec)])

        _, g = topk_obj_and_grad(h, n, active.astype(int), curr_alpha)

        b1, b2, eps = 0.9, 0.999, 1e-8
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        m_hat = m_v / (1 - b1**t)
        v_hat = v_v / (1 - b2**t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_feasible(h, n)

        c = compute_c5(h, n)
        if c < best_c - 1e-16:
            best_c = c
            best_h = h.copy()

        if t % 500 == 0:
            lr *= 0.97

    return best_h, best_c

def lbfgs_smooth(h, n, deadline, alpha, mu=1.0):
    def obj(x):
        v, _ = softmax_obj_and_grad(x, n, alpha)
        return v + mu * (x.sum() - n / 2.0)**2

    def grad(x):
        _, g = softmax_obj_and_grad(x, n, alpha)
        g = g + 2.0 * mu * (x.sum() - n / 2.0)
        return g

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': 3000, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': 20000, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def subgradient_active(h, n, deadline, n_iters=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.005
    t = 0
    while time.time() < deadline - 1 and t < n_iters:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - 1e-6
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        if len(active) > 3:
            active = active[:3]

        g = np.zeros(n)
        for idx in active:
            g += correlation_grad_single(h, n, int(idx))
        g /= len(active)

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = lr / (gn + 1e-15) * g
        h_new = h - step
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new

        t += 1
        if t % 2000 == 0:
            lr *= 0.995
    return best_h, best_c

def local_pair_search(h, n, deadline, n_rounds=50):
    best_c = compute_c5(h, n)
    step = 0.0005
    rng = np.random.RandomState(0)
    for r in range(n_rounds):
        if time.time() > deadline - 1:
            break
        improved = False
        n_trials = min(3000, n * n // 80)
        for _ in range(n_trials):
            if time.time() > deadline - 1:
                return h, best_c
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            for d in (step, -step):
                hi, hj = h[i] + d, h[j] - d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d
        if not improved:
            step *= 0.6
            if step < 1e-8:
                break
        else:
            step = min(step * 1.1, 0.002)
    return h, best_c

def gradient_line_search(h, n, deadline, max_iters=2000):
    best_c = compute_c5(h, n)
    base_step = 0.002
    t = 0
    while time.time() < deadline - 1 and t < max_iters:
        t += 1
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-7, 0.003 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > 5:
            active = active[:5]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        g = np.zeros(n)
        for idx in active:
            g += correlation_grad_single(h, n, int(idx))
        g /= len(active)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Find best descent direction via 2-point moves
        g_sorted = np.argsort(-np.abs(g))[:40]
        best_improvement = 0
        best_move = None

        for ii in range(len(g_sorted)):
            if time.time() > deadline - 1:
                break
            i = g_sorted[ii]
            for jj in range(len(g_sorted)):
                if ii == jj:
                    continue
                j = g_sorted[jj]
                if g[i] > g[j]:
                    di, dj = -1.0, 1.0
                else:
                    di, dj = 1.0, -1.0

                for mult in (1.0, 0.5, 0.25, 0.1, 0.05):
                    s = base_step * mult
                    hi = h[i] + s * di
                    hj = h[j] + s * dj
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            imp = best_c - c
                            if imp > best_improvement:
                                best_improvement = imp
                                best_move = (i, j, s*di, s*dj)
                                best_c = c
                        else:
                            h[i], h[j] = h[i] - s*di, h[j] - s*dj

        if best_move is None:
            base_step *= 0.5
            if base_step < 1e-7:
                break
    return h, best_c
