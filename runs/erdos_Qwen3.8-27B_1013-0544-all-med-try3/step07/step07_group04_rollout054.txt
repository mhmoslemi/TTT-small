            best_c = c5
            best_h = h.copy()

        top_idx = np.argsort(c)[-top_count:]
        threshold = c5 * 0.85
        active = top_idx[c[top_idx] > threshold]
        if len(active) == 0:
            active = top_idx[-1:]

        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            w = c[k] - threshold
            grad += w * lag_grad(h, n, lag)

        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break

        h_new = h - lr * grad / gn
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new <= c5 + 1e-12:
            h = h_new
        else:
            lr *= 0.85
        lr = max(lr, 1e-6)

    return best_h, best_c

def local_refine(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)

    deltas = [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        rounds = 0
        while improved and rounds < 200:
            if time.time() > deadline:
                break
            improved = False
            rounds += 1
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-6:]

            for k_st in top_idx:
                if time.time() > deadline:
                    break
                lag = k_st - (n - 1)
                if lag == 0:
                    mid = np.where((h > 0.08) & (h < 0.92))[0]
                    if len(mid) >= 2:
                        a = mid[np.argmin(h[mid])]
                        b = mid[np.argmax(h[mid])]
                        delta = min(d, (0.5 - h[a]) * 0.4, (h[b] - 0.5) * 0.4)
                        if delta > 1e-10:
                            h[a] += delta
                            h[b] -= delta
                            cn = compute_c5(h, n)
                            if cn < best_c - 1e-14:
                                best_c = cn
                                improved = True
                            else:
                                h[a] -= delta
                                h[b] += delta
                    continue

                m = abs(lag)
                if m >= n:
                    continue

                for i in range(n - m):
                    if time.time() > deadline or not improved and rounds > 50:
                        if time.time() > deadline:
                            break
                    j = i + m
                    if lag > 0:
                        if h[i] > h[j] + 1e-5:
                            delta = min(d, (h[i] - h[j]) * 0.35, h[i] * 0.7, (1.0 - h[j]) * 0.7)
                            if delta < 1e-10:
                                continue
                            h[i] -= delta
                            h[j] += delta
                            if compute_c5(h, n) < best_c - 1e-14:
                                best_c = compute_c5(h, n)
                                improved = True
                            else:
                                h[i] += delta
                                h[j] -= delta
                    else:
                        if h[j] > h[i] + 1e-5:
                            delta = min(d, (h[j] - h[i]) * 0.35, h[j] * 0.7, (1.0 - h[i]) * 0.7)
                            if delta < 1e-10:
                                continue
                            h[j] -= delta
                            h[i] += delta
                            if compute_c5(h, n) < best_c - 1e-14:
                                best_c = compute_c5(h, n)
                                improved = True
                            else:
                                h[j] += delta
                                h[i] -= delta

    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLSQP full, increasing alpha
    if time.time() < deadline - 20:
        for alpha in [5000, 30000, 150000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_solve(best_h, n, alpha, maxiter=5000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 2: Active-set with adaptive top lags
    if time.time() < deadline - 20:
        for top_m in [4, 6, 10, 15]:
            if time.time() > deadline - 12:
                break
            c = compute_all_lags(best_h, n)
            for alpha in [50000, 200000]:
                if time.time() > deadline - 10:
                    break
                top_idx = np.argsort(c)[-top_m:]
                h_new = slsqp_solve(best_h, n, alpha, maxiter=4000, ftol=1e-15, top_idx=top_idx)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                c = compute_all_lags(best_h, n)
                top_idx = np.argsort(c)[-top_m:]

    # Phase 3: Multi-lag gradient descent
    if time.time() < deadline - 15:
        gd_end = time.time() + min(50, (deadline - time.time()) * 0.3)
        h_new, c_new = multi_lag_descent(best_h, n, gd_end)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: SLSQP high alpha
    if time.time() < deadline - 10:
        h_new = slsqp_solve(best_h, n, 500000, maxiter=6000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 5: Local refinement
    if time.time() < deadline - 3:
        ls_end = min(deadline - 1, time.time() + 35)
        best_h, best_c = local_refine(best_h, n, ls_end)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize at parent resolution
    if h_init is not None:
        alloc = min(350, (deadline - t0) * 0.38)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt

    # Phase 2: Try alternative resolutions
    n_candidates = [200, 250, 350, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 90:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 40
        if remaining < 55:
            break
        alloc = min(110, remaining * 0.22)

        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)

        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Random restarts near best
    if best_h is not None and time.time() < deadline - 45:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(1, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.012 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 20) * 0.38)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final intensive refinement on best
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        # Very high alpha SLSQP
        rem = deadline - time.time() - 20
        if rem > 30:
            h_new = slsqp_solve(best_h, n, 1000000, maxiter=8000, ftol=1e-17)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Active set final
        if time.time() < deadline - 25:
            c = compute_all_lags(best_h, n)
            for top_m in [3, 5, 8]:
                if time.time() > deadline - 18:
                    break
                top_idx = np.argsort(c)[-top_m:]
                h_new = slsqp_solve(best_h, n, 300000, maxiter=5000, ftol=1e-16, top_idx=top_idx)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                c = compute_all_lags(best_h, n)
                top_idx = np.argsort(c)[-top_m:]
        # Gradient descent
        if time.time() < deadline - 15:
            h_new, c_new = multi_lag_descent(best_h, n, deadline - 8, top_count=10, lr_init=0.002)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Local search
        if time.time() < deadline - 3:
            best_h, best_c = local_refine(best_h, n, deadline - 1)

    # Final projection and evaluation
    if best_h is not None:
        best_h = project_feasible(best_h, int(len(best_h)))
        best_c = compute_c5(best_h, int(len(best_h)))

    if best_h is None:
        n_fallback = 300
        best_h = project_feasible(np.full(n_fallback, 0.5), n_fallback)
        best_c = compute_c5(best_h, n_fallback)

    return best_h, best_c, len(best_h)
