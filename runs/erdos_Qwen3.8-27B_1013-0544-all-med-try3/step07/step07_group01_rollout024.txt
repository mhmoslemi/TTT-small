import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    mx = c.max()
    sh = alpha * (c - mx)
    mx2 = sh.max()
    if mx2 > 500:
        return mx + (mx2 + np.log(np.sum(np.exp(sh - mx2)))) / alpha
    return mx + np.log(np.sum(np.exp(sh))) / alpha

def compute_grad_active(h, n, top_indices, dx):
    grad = np.zeros(n)
    for m in top_indices:
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[k:] += dx * (1.0 - h[0:nv])
            grad[0:nv] += dx * (-h[k:])
        else:
            kk = -k
            nv = n - kk
            grad[0:nv] += dx * (1.0 - h[kk:])
            grad[kk:] += dx * (-h[0:nv])
    return grad

def compute_grad_full(h, n, alpha):

    mx = c.max()
    sh = alpha * (c - mx)
    mx2 = sh.max()
    w = np.exp(sh - mx2)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):

        k = m - (n - 1)

            nv = n - kk

    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        def _obj(x):
            c = compute_c5_vec(x, n)
            return smooth_max_val(c, alpha)
        def _grad(x):
            return compute_grad_full(x, n, alpha)
        try:
            t_rem = max(5, deadline - time.time())
            max_it = max(50, int(300 * t_rem / 200))
            res = minimize(_obj, best_h, jac=_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': min(max_it, max_iters_per_alpha),
                                   'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxfun': min(max_it * 10, 5000)})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def _obj(x):
        c = compute_c5_vec(x, n)
        return smooth_max_val(c, alpha)
    def _grad(x):
        return compute_grad_full(x, n, alpha)
    def _con(x):
        return x.sum() - target
    def _jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _con, 'jac': _jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-top_k:]
        grad = compute_grad_active(h, n, top_idx, dx)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        g = grad / g_norm
        lr = 0.005
        improved = False
        for _ in range(30):

                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return best_h, best_c
