def coordinate_descent(h, n, deadline, max_rounds=50):
    best_c = compute_c5(h, n)
    h = h.copy()
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for step in steps:
        if time.time() > deadline - 3:
            break
        for rnd in range(max_rounds):
            if time.time() > deadline - 3:
                break
            improved = False
            rng = np.random.default_rng(42 + rnd)
            idx = rng.choice(n, size=min(n, 500), replace=False)
            for i in idx:
                if time.time() > deadline - 3:
                    return h, best_c
                for di in [-1, 1]:
                    new_val = h[i] + di * step
                    if new_val < 0 or new_val > 1:
                        continue
                    if h[i] + di * step + (best_c < 0):
                        pass
                    # Adjust sum by modifying another element
                    # Try modifying a random other element
                    mask = np.arange(n) != i
                    # Find element that can absorb the change
                    if di > 0:
                        # h[i] increases, need to decrease something
                        candidates = np.where(h > step + 1e-10)[0]
                    else:
                        candidates = np.where(h < 1 - step - 1e-10)[0]
                    if len(candidates) == 0:
                        continue
                    j = rng.choice(candidates)
                    h[i] += di * step
                    h[j] -= di * step
                    if h[j] < 0 or h[j] > 1:
                        h[i] -= di * step
                        h[j] += di * step
                        continue
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= di * step
                        h[j] += di * step
            if not improved:
                break
    return h, best_c

                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def multi_start_refine(h, n, deadline, n_starts=3, perturb_scale=0.02):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(123)
    
    time_per_start = max(10, (deadline - time.time()) / (n_starts + 1))
    
    for s in range(n_starts):
        if time.time() > deadline - 10:
            break
        h_pert = best_h.copy()
        noise = rng.normal(0, perturb_scale, n)
        h_pert += noise
        h_pert = project_feasible(h_pert, n)
        
        sub_deadline = time.time() + min(time_per_start, deadline - time.time() - 5)
        h_opt, c_opt = adaptive_lbfgs(h_pert, n, sub_deadline,
                                       top_k=20, start_alpha=500, alpha_mult=2.5,
                                       max_alphas=8, mu=0.01)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Quick polish at initial resolution (30s)
    t_phase1 = min(t0 + 30, deadline - 600)
    if t_phase1 > t0 + 5:
        best_h, best_c = adaptive_lbfgs(best_h, n_init, t_phase1,
                                         top_k=25, start_alpha=500, alpha_mult=3.0,
                                         max_alphas=5, mu=0.01)
    
    # Phase 2: Upsample to n=1000
    n = 1000
    h_n = upsample_h(best_h, n_init, n)
    h_n = project_feasible(h_n, n)
    c_n = compute_c5(h_n, n)
    if c_n < best_c:
        best_c = c_n
        best_h = h_n
    
    # Phase 3: Main optimization at n=1000
    t_phase3_end = min(deadline - 300, t0 + 500)
    h_n, c_n = adaptive_lbfgs(h_n, n, t_phase3_end,
                               top_k=30, start_alpha=200, alpha_mult=2.5,
                               max_alphas=15, mu=0.005, ftol=1e-16, gtol=1e-13)
    if c_n < best_c:
        best_c = c_n
        best_h = h_n
    
    # Phase 4: Multi-start perturbation
    if time.time() < deadline - 150:
        t_phase4 = min(time.time() + 120, deadline - 150)
        h_ms, c_ms = multi_start_refine(best_h, n, t_phase4,
                                         n_starts=3, perturb_scale=0.015)
        if c_ms < best_c:
            best_c = c_ms
            best_h = h_ms
    
    # Phase 5: Focused pair descent
    if time.time() < deadline - 80:
        t_phase5 = min(time.time() + 60, deadline - 80)
        h_pd, c_pd = pair_descent_focused(best_h, n, t_phase5,
                                           top_k=20,
                                           delta_list=[0.003, 0.001, 0.0005,
                                                       0.0002, 0.0001, 0.00005])
        if c_pd < best_c:
            best_c = c_pd
            best_h = h_pd
    
    # Phase 6: Final adaptive L-BFGS with high alpha
    if time.time() < deadline - 40:
        t_phase6 = min(time.time() + 35, deadline - 40)
        h_fin, c_fin = adaptive_lbfgs(best_h, n, t_phase6,
                                       top_k=15, start_alpha=50000, alpha_mult=2.0,
                                       max_alphas=5, mu=0.001, ftol=1e-16, gtol=1e-14)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    
    # Phase 7: Ultra-fine coordinate descent
    if time.time() < deadline - 10:
        t_phase7 = min(time.time() + 8, deadline - 10)
        h_cd, c_cd = pair_descent_focused(best_h, n, t_phase7,
                                           delta_list=[0.0001, 0.00005, 0.00002])
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    # Also try at original resolution in case it's better
    if n_init != n:
        h_orig_res = upsample_h(best_h, n, n_init)
        h_orig_res = project_feasible(h_orig_res, n_init)
        c_orig = compute_c5(h_orig_res, n_init)
        if c_orig < best_c:
            best_h = h_orig_res
            best_c = c_orig
            n = n_init
    
    return best_h, best_c, len(best_h)
