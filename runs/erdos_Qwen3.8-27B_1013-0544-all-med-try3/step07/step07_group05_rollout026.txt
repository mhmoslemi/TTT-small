def projected_gradient_descent(h, n, alpha, step_size, n_iter, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        g = grad_smooth_max(h, n, alpha)
        # Subtract mean to preserve sum constraint
        g -= np.mean(g)
        
        # Line search
        lr = step_size
        for _ in range(20):
            h_trial = h - lr * g
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < compute_c5(h, n) - 1e-12:
                break
            lr *= 0.5
        
        h = project_feasible(h - lr * g, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
