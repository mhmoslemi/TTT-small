def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            scale = target_sum / tot
            h = h * scale
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_logsumexp(c, alpha):

        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def smooth_max_weights(c, alpha):

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = h * (target_sum / tot)
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def convex_subproblem(h0, n, top_k, solver='CLARABEL' if 'CLARABEL' in [cp.installed_solvers()] else 'ECOS'):
    """Solve the linearized convex subproblem"""
    c_all = compute_all_lags(h0, n)
    top_idx = np.argsort(c_all)[-top_k:]
    dx = 2.0 / n
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [x >= 0, x <= 1, cp.sum(x) == n/2]
    
    for k_idx in top_idx:
        lag = k_idx - (n-1)
        J0 = c_all[k_idx]
        g = _lag_gradient(h0, n, lag) * dx
        constraints.append(J0 + g @ (x - h0) <= t)
    
    # Also add constraint t >= max of all current lags (to not decrease it)
    # Actually, this is implicit in the top_k constraints
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver)
        if x.value is not None and not np.any(np.isnan(x.value)):
            x_val = np.clip(x.value, 0, 1)
            return x_val
    except Exception:
        pass
    
    return h0.copy()

def solve_lp_subproblem(h0, n, top_k):
    """Faster LP solver using scipy"""
    c_all = compute_all_lags(h0, n)
    top_idx = np.argsort(c_all)[-top_k:]
    dx = 2.0 / n
    m_lags = len(top_idx)
    
    # Variables: x (n), t (1) -> total n+1
    # Minimize: t -> c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints:
    # Equality: sum(x) = n/2 -> 1 * x + 0 * t = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Inequalities: J0_k + g_k @ (x - h0) <= t for each top lag
    # -> J0_k + g_k @ x - g_k @ h0 <= t
    # -> g_k @ x - t <= g_k @ h0 - J0_k
    m_cons = m_lags
    A_ub = np.zeros((m_cons, n + 1))
    b_ub = np.zeros(m_cons)
    
    for i, k_idx in enumerate(top_idx):
        lag = k_idx - (n-1)
        J0 = c_all[k_idx]
        g = _lag_gradient(h0, n, lag) * dx
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = g @ h0 - J0
    
    # Bounds: 0 <= x_i <= 1, -inf <= t <= inf
    bounds = [(0, 1)] * n + [(None, None)]
    
    from scipy.optimize import linprog
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            x_val = np.clip(res.x[:n], 0, 1)
            return x_val
    except Exception:
        pass
    
    return h0.copy()

def linprog_subproblem(h0, n, top_k):
    """Solve the LP subproblem using scipy linprog (faster than cvxpy)"""
    c_all = compute_all_lags(h0, n)
    top_idx = np.argsort(c_all)[-top_k:]
    dx = 2.0 / n
    m_lags = len(top_idx)
    
    # Minimize t subject to:
    #   g_k . x - t <= g_k . h0 - J0_k  for each top lag k
    #   sum(x) = n/2
    #   0 <= x_i <= 1
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([n / 2.0])
    
    A_ub = np.zeros((m_lags, n + 1))
    b_ub = np.zeros(m_lags)
    
    for i, k_idx in enumerate(top_idx):
        lag = k_idx - (n - 1)
        J0 = c_all[k_idx]
        g = _lag_gradient(h0, n, lag) * dx
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = g @ h0 - J0
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return np.clip(res.x[:n], 0.0, 1.0)
    return h0.copy()

def linprog_iteration(h, n, deadline, top_k=30, max_iters=5000, verbose=False):
    """Iterative linearization method"""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    h_curr = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Solve LP subproblem
        h_new = linprog_subproblem(h_curr, n, top_k)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
        else:
            # Try with more lags
            if it % 10 == 0 and it < max_iters - 100:
                h_new2 = linprog_subproblem(h_curr, n, min(top_k + 10, 2*n-1))
                h_new2 = project_feasible(h_new2, n)
                c_new2 = compute_c5(h_new2, n)
                if c_new2 < best_c - 1e-13:
                    best_c = c_new2
                    best_h = h_new2.copy()
                    h_curr = h_new2
                    top_k = min(top_k + 5, 2*n-1)
        
        # Periodically re-evaluate
        c_curr = compute_c5(h_curr, n)
        if abs(c_curr - c_new) < 1e-12:
            if it > 100:
                break
    
    return best_h, best_c

def grad_momentum(h, n, alpha, deadline, lr=0.5, momentum=0.90, max_iter=20000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    patience = 0
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
        else:
            w = w / w_sum
        
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-10:
                continue
            lag = k - (n-1)
            g += w[k] * _lag_gradient(h_curr, n, lag)
        g *= dx
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        v = momentum * v + lr * g / (g_norm + 1e-15)
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        
        if patience > 300:
            lr *= 0.5
            v *= 0.3
            patience = 0
    
    return best_h, best_c

def fine_local_search(h, n, deadline, max_rounds=200):
    """Very fine local search targeting the worst few lags"""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        sorted_idx = np.argsort(c_all)[::-1]
        
        improved_in_round = False
        for k_idx in sorted_idx[:8]:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            
            for delta in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
                if time.time() > deadline:
                    break
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-6:
                        amt = min(delta, (h[i] - h[j]) * 0.5)
                        if amt < 1e-10:
                            continue
                        h[i] -= amt
                        h[j] += amt
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved_in_round = True
                        else:
                            h[i] += amt
                            h[j] -= amt
                    elif h[j] > h[i] + 1e-6:
                        amt = min(delta, (h[j] - h[i]) * 0.5)
                        if amt < 1e-10:
                            continue
                        h[j] -= amt
                        h[i] += amt
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved_in_round = True
                        else:
                            h[j] += amt
                            h[i] -= amt
        if not improved_in_round:
            break
    
    return best_h, best_c
