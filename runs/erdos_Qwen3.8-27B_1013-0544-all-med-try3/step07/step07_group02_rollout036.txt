def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n//2]
    h_bin[idx] = 1.0
    return h_bin

def compute_delta_vec(h, i, j, n):
    """Compute delta vector: change in C per unit delta moved from i to j."""
    L = np.arange(2*n-1) - (n-1)
    dv = np.zeros(2*n-1)
    idx1 = i - L
    m1 = (idx1 >= 0) & (idx1 < n)
    dv[m1] += h[idx1[m1]]
    idx2 = i + L
    m2 = (idx2 >= 0) & (idx2 < n)
    dv[m2] += h[idx2[m2]]
    idx3 = j - L
    m3 = (idx3 >= 0) & (idx3 < n)
    dv[m3] -= h[idx3[m3]]
    idx4 = j + L
    m4 = (idx4 >= 0) & (idx4 < n)
    dv[m4] -= h[idx4[m4]]
    return dv

def efficient_local_search(h, n, deadline, rng, max_iter=5000000):
    """Local search using the delta vector formula for fast evaluation."""
    h = h.copy()
    h = project_feasible(h, n)
    c_all = compute_all_lags(h, n)
    best_c = c_all.max()
    best_h = h.copy()
    
    patience = 0
    max_patience = 50000
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Find worst lag
        k_star = int(np.argmax(c_all))
        
        # Generate candidate swaps
        found = False
        n_attempts = 50
        
        for _ in range(n_attempts):
            if time.time() > deadline:
                break
            
            # Choose i (to decrease) and j (to increase)
            # Bias towards positions that would help the worst lag
            i = rng.randint(n)
            j = rng.randint(n)
            
            if i == j:
                continue
            if h[i] <= h[j] + 1e-6:
                continue
            if h[i] < 1e-8:
                continue
            
            # Compute delta vector
            dv = compute_delta_vec(h, i, j, n)
            
            # Quick check: does the worst lag improve?
            if dv[k_star] >= -1e-12:
                continue
            
            # Line search: find max allowable delta
            curr_max = c_all.max()
            pos_mask = dv > 1e-12
            if pos_mask.any():
                ratios = (curr_max - c_all[pos_mask]) / dv[pos_mask]
                delta_max = ratios.min()
            else:
                delta_max = 1e10
            delta_max = min(delta_max, h[i], 1.0 - h[j])
            
            if delta_max < 1e-10:
                continue
            
            # Try the full step and a fraction
            best_d = 0
            best_new_max = curr_max
            for frac in [1.0, 0.8, 0.5, 0.3]:
                d = delta_max * frac
                if d < 1e-10:
                    continue
                new_max = (c_all + d * dv).max()
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_d = d
            
            if best_d > 1e-10 and best_new_max < best_c - 1e-13:
                h[i] -= best_d
                h[j] += best_d
                c_all = c_all + best_d * dv
                best_c = c_all.max()
                best_h = h.copy()
                patience = 0
                found = True
                break
        
        if not found:
            patience += 1
            if patience > max_patience:
                # Simulated annealing: try a random perturbation
                if rng.random() < 0.3:
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i != j and h[i] > 0.01 and h[j] < 0.99:
                        d = min(0.05, h[i] * 0.5, (1-h[j]) * 0.5)
                        h[i] -= d
                        h[j] += d
                        c_all = compute_all_lags(h, n)
                        new_max = c_all.max()
                        # Accept with probability based on improvement
                        if new_max < best_c + 0.001:
                            best_h = h.copy()
                            patience = 0
                else:
                    patience = 0
    
    h = project_feasible(best_h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def targeted_local_search(h, n, deadline, rng, max_iter=3000000):
    """Targeted search focusing on the worst lag structure."""
    h = h.copy()
    h = project_feasible(h, n)
    c_all = compute_all_lags(h, n)
    best_c = c_all.max()
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        
        if lag == 0:
            break
        
        m = abs(lag)
        found = False
        
        # For the worst lag, identify "problem" positions
        if lag > 0:
            # C(lag) = sum_{i=0}^{n-m-1} h[i+m]*(1-h[i])
            # Problem: h[i+m]=1 and h[i]=0
            problem_i = []
            for i in range(n - m):
                if h[i+m] > 0.8 and h[i] < 0.2:
                    problem_i.append(i)
            # We want to increase h[i] (for problem i) or decrease h[i+m]
            # Try: increase h[i] for a problem position i, decrease h[j] for some j
            candidates_i = problem_i[:20] if problem_i else []
        else:
            m2 = -lag
            # C(-m2) = sum_{i=m2}^{n-1} h[i]*(1-h[i-m2])
            # = sum_{i=0}^{n-m2-1} h[i+m2]*(1-h[i])
            problem_i = []
            for i in range(n - m2):
                if h[i+m2] > 0.8 and h[i] < 0.2:
                    problem_i.append(i)
            candidates_i = problem_i[:20] if problem_i else []
        
        if candidates_i:
            i_target = candidates_i[rng.randint(len(candidates_i))]
            # Find j to decrease (to fund the increase at i_target)
            # Prefer j where h[j] is high and doesn't hurt other lags
            j_candidates = np.where(h > 0.5)[0]
            if len(j_candidates) > 0:
                j = j_candidates[rng.randint(len(j_candidates))]
                if j != i_target:
                    dv = compute_delta_vec(h, j, i_target, n)
                    curr_max = c_all.max()
                    pos_mask = dv > 1e-12
                    if pos_mask.any():
                        ratios = (curr_max - c_all[pos_mask]) / dv[pos_mask]
                        delta_max = ratios.min()
                    else:
                        delta_max = 1e10
                    delta_max = min(delta_max, h[j], 1.0 - h[i_target])
                    
                    if delta_max > 1e-10:
                        best_d = 0
                        best_new_max = curr_max
                        for frac in [1.0, 0.7, 0.5]:
                            d = delta_max * frac
                            new_max = (c_all + d * dv).max()
                            if new_max < best_new_max:
                                best_new_max = new_max
                                best_d = d
                        
                        if best_d > 1e-10 and best_new_max < best_c - 1e-13:
                            h[j] -= best_d
                            h[i_target] += best_d
                            c_all = c_all + best_d * dv
                            best_c = c_all.max()
                            best_h = h.copy()
                            found = True
        
        if not found:
            # Fallback: random swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j and h[i] > h[j] + 0.05 and h[i] > 0.01:
                dv = compute_delta_vec(h, i, j, n)
                curr_max = c_all.max()
                if dv[int(np.argmax(c_all))] < -1e-6:
                    pos_mask = dv > 1e-12
                    if pos_mask.any():
                        ratios = (curr_max - c_all[pos_mask]) / dv[pos_mask]
                        delta_max = min(ratios.min(), h[i], 1.0 - h[j])
                    else:
                        delta_max = min(h[i], 1.0 - h[j])
                    if delta_max > 1e-10:
                        # Try small step
                        d = min(delta_max, 0.02)
                        new_max = (c_all + d * dv).max()
                        if new_max < best_c - 1e-13:
                            h[i] -= d
                            h[j] += d
                            c_all = c_all + d * dv
                            best_c = c_all.max()
                            best_h = h.copy()
    
    h = project_feasible(best_h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def gradient_descent_smooth(h, n, deadline, alpha=100000, lr=1.0, momentum=0.9):
    """Gradient descent on the smooth max objective."""
    h = h.copy()
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    
    for it in range(1000000):
        if time.time() > deadline:
            break
        
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        
        # Gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-10:
                continue
            lag = k - (n-1)
            mm = abs(lag)
            if mm >= n:
                continue
            if lag > 0:
                grad[mm:n] += w[k] * (1.0 - h[0:n-mm])
                grad[0:n-mm] += w[k] * (-h[mm:n])
            elif lag < 0:
                grad[0:n-mm] += w[k] * (1.0 - h[mm:n])
                grad[mm:n] += w[k] * (-h[0:n-mm])
            else:
                grad[:] = w[k] * (1.0 - 2.0 * h)
        
        v = momentum * v + lr * grad
        h_new = project_feasible(h - lr * v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        norm_v = np.linalg.norm(v)
        if norm_v > 100:
            v *= 100.0 / norm_v
        
        if it % 10000 == 9999:
            lr *= 0.9
            v *= 0.8
        
        h = h_new
    
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000):
    """SLSQP polishing with top-m active lags."""
    dx = 2.0 / n
    c_all = compute_all_lags(h, n)
    top_m = min(5, 2*n-1)
    top_idx = np.argsort(c_all)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

def full_local_refine(h, n, deadline, rng):
    """Combined local refinement with multiple strategies."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase A: Efficient local search
    alloc_a = (deadline - time.time()) * 0.4
    if alloc_a > 30:
        h_new, c_new = efficient_local_search(h, n, time.time() + alloc_a, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase B: Targeted local search
    alloc_b = (deadline - time.time()) * 0.3
    if alloc_b > 20:
        h_new, c_new = targeted_local_search(best_h, n, time.time() + alloc_b, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase C: Gradient polishing
    alloc_c = min(30, (deadline - time.time()) * 0.2)
    if alloc_c > 10:
        h_new, c_new = gradient_descent_smooth(best_h, n, time.time() + alloc_c,
                                               alpha=50000, lr=0.5, momentum=0.85)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase D: SLSQP
    if time.time() < deadline - 15:
        h_new = slsqp_polish(best_h, n, alpha=100000, maxiter=1500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase E: Final local search
    if time.time() < deadline - 5:
        h_new, c_new = efficient_local_search(best_h, n, deadline - 1, rng, max_iter=1000000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    # Main optimization at initial resolution
    n = len(best_h)
    
    # Phase 1: Efficient local search (most promising)
    alloc1 = min(400, (deadline - time.time()) * 0.35)
    if time.time() < deadline - 100:
        h_new, c_new = efficient_local_search(best_h, n, time.time() + alloc1, rng,
                                             max_iter=8000000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 2: Targeted search
    if time.time() < deadline - 100:
        alloc2 = min(200, (deadline - time.time()) * 0.2)
        h_new, c_new = targeted_local_search(best_h, n, time.time() + alloc2, rng,
                                            max_iter=5000000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 3: Gradient smoothing
    if time.time() < deadline - 80:
        alloc3 = min(60, (deadline - time.time()) * 0.15)
        h_new, c_new = gradient_descent_smooth(best_h, n, time.time() + alloc3,
                                             alpha=50000, lr=0.3, momentum=0.85)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        else:
            # Try different hyperparameters
            h_new, c_new = gradient_descent_smooth(best_h, n, time.time() + 20,
                                                 alpha=200000, lr=0.8, momentum=0.9)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 4: SLSQP polish
    if time.time() < deadline - 20:
        h_new = slsqp_polish(best_h, len(best_h), alpha=200000, maxiter=2000)
        c_new = compute_c5(h_new, len(best_h))
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 5: Try different resolutions
    n_candidates = [450, 550, 600]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(150, remaining * 0.3)
        if alloc < 30:
            break
        
        # Interpolate
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        
        h_opt, c_opt = full_local_refine(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 6: Final refinement at best resolution
    if time.time() < deadline - 20:
        n_final = len(best_h)
        alloc_final = min(100, deadline - time.time() - 5)
        if alloc_final > 15:
            h_new, c_new = efficient_local_search(best_h, n_final,
                                                 time.time() + alloc_final, rng,
                                                 max_iter=5000000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 7: Ultra-fine final polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_new, c_new = targeted_local_search(best_h, n_final, deadline - 1, rng,
                                            max_iter=2000000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
