def project(h,n):
    h=np.asarray(h,dtype=float)
    target=n/2.0
    lo,hi=-1.0,1.0
    for _ in range(60):
       mid=(lo+hi)*0.5
       hp=np.clip(h-mid,0.0,1.0)
       s=hp.sum()
       if s>target: lo=mid
       else: hi=mid
    hp=np.clip(h-(lo+hi)*0.5,0.0,1.0)
    diff=target-hp.sum()
    if diff>1e-12:
       room=1.0-hp
       s=room.sum()
       if s>1e-15:
          hp += (diff/s)*room
    elif diff<-1e-12:
       s=hp.sum()
       if s>1e-15:
          hp += (diff/s)*hp
    hp=np.clip(hp,0.0,1.0)
    # final exact adjust
    diff=target-hp.sum()
    if abs(diff)>1e-12:
       if diff>0:
          idxs=np.where(hp<1.0)[0]
       else:
          idxs=np.where(hp>0.0)[0]
       for idx in idxs:
           cap=1.0-hp[idx] if diff>0 else -hp[idx]
           stepv=min(cap, abs(diff))* (1.0 if diff>0 else -1.0)
           hp[idx]+=stepv
           diff=-stepv
           if abs(diff)<1e-15: break
    return hp

import numpy as np
import time
import math


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def binary_count(h, n):
    hi = np.asarray(h, dtype=np.int32)
    c = np.correlate(hi, 1 - hi, mode="full")
    return int(np.max(c))


def project(h, n):
    h = np.asarray(h, dtype=float)
    target = n / 2.0

    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        s = float(hp.sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    hp = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(hp.sum())

    if diff > 1e-12:
        room = 1.0 - hp
        s = float(room.sum())
        if s > 1e-15:
            hp += (diff / s) * room
    elif diff < -1e-12:
        s = float(hp.sum())
        if s > 1e-15:
            hp += (diff / s) * hp

    hp = np.clip(hp, 0.0, 1.0)

    diff = target - float(hp.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idxs = np.where(hp < 1.0)[0]
        else:
            idxs = np.where(hp > 0.0)[0]
        for idx in idxs:
            if abs(diff) < 1e-15:
                break
            if diff > 0.0:
                stepv = min(1.0 - hp[idx], diff)
                hp[idx] += stepv
                diff -= stepv
            else:
                stepv = min(hp[idx], -diff)
                hp[idx] -= stepv
                diff += stepv

    return hp


def to_binary(h, n):
    h = np.asarray(h, dtype=float)
    m = n // 2
    hb = np.zeros(n, dtype=np.uint8)
    if m <= 0:
        return hb
    idx = np.argsort(h)[::-1]
    hb[idx[:m]] = 1
    return hb


def complement_binary(h):
    return (1 - np.asarray(h, dtype=np.int8)).astype(np.uint8)


def round_to_n(src, n):
    if src is None:
        return np.full(n, 0.5, dtype=float)
    src = np.asarray(src, dtype=float)
    if len(src) == n:
        return project(src, n)
    old = len(src)
    if old <= 0:
        return np.full(n, 0.5, dtype=float)
    xo = (np.arange(old) + 0.5) * (2.0 / old)
    xn = (np.arange(n) + 0.5) * (2.0 / n)
    vals = np.interp(xn, xo, src)
    return project(vals, n)


def random_balanced(n, rng):
    h = np.zeros(n, dtype=np.uint8)
    m = n // 2
    if m <= 0:
        return h
    idx = rng.choice(n, size=m, replace=False)
    h[idx] = 1
    return h


def random_runs(n, rng, min_len=3, max_len=9):
    h = np.zeros(n, dtype=np.uint8)
    idx = 0
    state = int(rng.randint(2))
    if max_len < min_len:
        max_len = min_len
    while idx < n:
        length = int(rng.randint(min_len, max_len + 1))
        end = min(idx + length, n)
        h[idx:end] = state
        state = 1 - state
        idx = end
    return to_binary(h, n)


def add_binary_start(lst, h, n):

    for s in lst:
        if s.shape == h.shape and np.array_equal(s, h):
            return
    lst.append(h)


def make_binary_starts(hfrac, n, rng, count=8):
    starts = []

    add_binary_start(starts, base, n)

    shifts = [1, max(1, n // 10), max(1, n // 4), int(rng.randint(1, n - 1))]
    for sh in shifts:
        add_binary_start(starts, np.roll(base, sh), n)

    comp = complement_binary(base)

    add_binary_start(starts, rev, n)

    add_binary_start(starts, random_runs(n, rng), n)
    add_binary_start(starts, random_runs(n, rng, 2, 6), n)

def smooth_value_from_c(c, alpha):
    a = alpha * np.asarray(c, dtype=float)
    m = float(np.max(a))
    e = np.exp(a - m)
    s = float(np.sum(e))
    if s <= 0.0:
        return m / alpha
    return (m + math.log(s)) / alpha


def smooth_gradient(h, n, w):
    off = n - 1
    grad = np.zeros(n, dtype=float)
    thresh = 1e-12
    for idx in range(len(w)):
        wk = float(w[idx])
        if wk <= thresh:
            continue
        k = idx - off
        if k >= 0:
            L = n - k
            if L > 0:
                grad[k:k + L] += wk * (1.0 - h[:L])
                grad[:L] -= wk * h[k:k + L]
        else:
            d = -k
            L = n - d
            if L > 0:
                grad[:L] += wk * (1.0 - h[d:d + L])
                grad[d:d + L] -= wk * h[:L]
    return grad

def build_AB(h, n):
    off = n - 1
    L = 2 * n - 1
    A = np.zeros((n, L), dtype=np.int16)
    B = np.zeros((n, L), dtype=np.int16)
    h16 = np.asarray(h, dtype=np.int16)
    ones16 = 1 - h16
    ones_rev = ones16[::-1]

    for p in range(n):
        rowA = A[p]
        rowB = B[p]

        s1 = off + p - (n - 1)
        rowA[s1:s1 + n] += ones_rev
        rowB[s1:s1 + n] -= ones_rev

        s2 = off - p
        rowA[s2:s2 + n] -= h16
        rowB[s2:s2 + n] += h16

        rowA[off] -= 1
        rowB[off] -= 1

    return A, B


def update_AB(A, B, x, delta, n):
    off = n - 1
    sl = off - x
    A[:, sl:sl + n] -= delta
    B[:, sl:sl + n] += delta

    sr = off + x
    A[:, sr:sr - n:-1] -= delta
    B[:, sr:sr - n:-1] += delta


def build_state(h, n):
    hb = to_binary(h, n).astype(np.uint8)
    hi = hb.astype(np.int32)
    c = np.correlate(hi, 1 - hi, mode="full").astype(np.int32)
    A, B = build_AB(hb, n)

    zero = []
    one = []
    for i in range(n):
        if hb[i]:
            one.append(i)
        else:
            zero.append(i)

    cur = int(c.max())
    return (hb, c, A, B, zero, one, cur)


def find_improve_for_i(c, A, B, ones_list, i, off, cur_c):
    base = c + A[i]
    CH = 128
    Lc = len(ones_list)

    for st in range(0, Lc, CH):
        end = min(st + CH, Lc)
        js = np.asarray(ones_list[st:end], dtype=np.intp)
        if js.size == 0:
            continue

        Bch = B[js]
        vals = base[None, :] + Bch
        m = vals.max(axis=1)

        d = i - js
        idxp = off + d
        idxm = off - d

        cand = m
        cand = np.maximum(cand, base[idxp] + Bch[:, idxp] + 1)
        cand = np.maximum(cand, base[idxm] + Bch[:, idxm] + 1)

        k = int(np.argmin(cand))
        if cand[k] < cur_c:
            return int(js[k]), int(cand[k])

    return None


def local_search_first(state, deadline):

            delta[idxm] += 1

            c += delta
            cur = int(c.max())

            h[i] = 1
            h[j] = 0
            update_AB(A, B, i, 1, n)
            update_AB(A, B, j, -1, n)

            if i in zero:
                zero.remove(i)
            else:
                zero = [x for x in zero if x != i]
                state_zero_tmp = zero
            if j in one:
                one.remove(j)
            else:
                one = [x for x in one if x != j]

            zero.append(j)
            one.append(i)
            progress = True
            break

        if not progress:
            break

    return (h, c, A, B, zero, one, cur)


def perturb_binary(h, n, k, rng):
    hk = np.asarray(h, dtype=np.uint8).copy()
    ones = list(np.where(hk == 1)[0])
    zeros = list(np.where(hk == 0)[0])

    for _ in range(k):
        if not ones or not zeros:
            break
        ii = int(rng.randint(len(zeros)))
        jj = int(rng.randint(len(ones)))
        i = zeros[ii]
        j = ones[jj]
        hk[i] = 1
        hk[j] = 0
        zeros[ii] = j
        ones[jj] = i

    return hk


def binary_optimize(h0, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    start = to_binary(h0, n).astype(np.uint8)

    best_h = start.copy()
    best_cnt = binary_count(best_h, n)

    state = build_state(start, n)
    dl1 = min(deadline - 20, time.time() + 120.0)
    state = local_search_first(state, dl1)
    cnt = state[6]
    if cnt < best_cnt:
        best_cnt = cnt
        best_h = state[0].copy()

    while time.time() < deadline - 20:
        base = to_binary(best_h, n).astype(np.uint8)
        r = float(rng.random_sample())

        if r < 0.55:
            k = int(rng.randint(3, 8 + max(1, n // 100)))
            hk = perturb_binary(base, n, k, rng)
        elif r < 0.75:
            shift = int(rng.randint(1, n))
            hk = np.roll(base, shift).copy()
            if rng.random_sample() < 0.5:
                hk = complement_binary(hk)
        elif r < 0.88:
            hk = base[::-1].copy()
            if rng.random_sample() < 0.5:
                hk = complement_binary(hk)
        else:
            hk = random_balanced(n, rng)

        st = build_state(hk, n)

        rem = deadline - time.time()
        lt = min(45.0, max(10.0, rem / 4.0))
        st = local_search_first(st, time.time() + lt)
        cnt = st[6]

        if cnt < best_cnt:
            best_cnt = cnt
            best_h = st[0].copy()

    best_h = to_binary(best_h, n).astype(np.uint8)
    return best_h, compute_c5(best_h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(60.0, float(budget_s)) - 10.0
    if deadline < t0 + 20.0:
        deadline = t0 + max(20.0, float(budget_s))

    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=float).copy()
        n = int(len(hpar))
        has_parent = n > 4
    except Exception:
        n = 400
        hpar = np.full(n, 0.5, dtype=float)
        has_parent = False

    if not has_parent or n < 16:
        n = 400
        hpar = np.full(n, 0.5, dtype=float)

    if n % 2 == 1:
        n = n - 1
        hpar = round_to_n(hpar, n)

    hpar = project(hpar, n)
    parent_float = hpar.copy()

    best_h = hpar.copy()
    best_c = compute_c5(best_h, n)
    best_n = n

    if deadline - time.time() > 70:
        budget1 = min(190.0, 0.18 * (deadline - t0))
        h1, c1 = smooth_pgd(hpar, n, budget1, [10, 50, 200, 1000], seed=seed + 1)
        if c1 < best_c - 1e-12:
            best_c = c1
            best_n = n
            best_h = h1.copy()
            parent_float = h1.copy()

    flat_best = project(best_h, best_n)
    starts = make_binary_starts(flat_best, best_n, rng, count=10)
    top_starts = top_starts_by_count(starts, best_n, max_keep=5)

    binary_deadline = min(deadline - 260.0, t0 + 0.52 * (deadline - t0))
    allocs = [210.0, 110.0, 70.0, 50.0, 35.0]

    for si, st in enumerate(top_starts):
        if time.time() >= binary_deadline:
            break
        alloc = min(allocs[si] if si < len(allocs) else 25.0, binary_deadline - time.time())
        if alloc < 15.0:
            break
        hb, cb = binary_optimize(st, best_n, time.time() + alloc, seed=seed + 1000 + si)
        cb = compute_c5(hb, best_n)
        if cb < best_c - 1e-12:
            best_c = cb
            best_n = best_n
            best_h = hb.copy()

    if deadline - time.time() > 70:
        budget2 = min(180.0, 0.25 * (deadline - time.time()))
        h2, c2 = smooth_pgd(project(best_h, best_n), best_n, budget2, [50, 200, 1000, 5000], seed=seed + 500)
        if c2 < best_c - 1e-12:
            best_c = c2
            best_n = best_n
            best_h = h2.copy()

    if deadline - time.time() > 130:
        alloc3 = min(220.0, deadline - time.time() - 80.0)
        hb3, cb3 = binary_optimize(project(best_h, best_n), best_n, time.time() + alloc3, seed=seed + 900)
        cb3 = compute_c5(hb3, best_n)
        if cb3 < best_c - 1e-12:
            best_c = cb3
            best_n = best_n
            best_h = hb3.copy()

    base_extra = n
    extra_ns = []
    for factor in (1.25, 1.5, 2.0):
        nh = int(round(base_extra * factor / 2.0) * 2)
        if nh > best_n and nh <= 1200:
            extra_ns.append(nh)

    for nh in extra_ns:
        if time.time() > deadline - 140:
            break

        budget_extra = min(150.0, 0.38 * (deadline - time.time()))
        if budget_extra < 45.0:
            break

        h0 = round_to_n(best_h, best_n, nh)
        hf, cf = smooth_pgd(h0, nh, budget_extra, [20, 100, 500, 2000], seed=seed + 5000 + nh)
        if cf < best_c - 1e-12:
            best_c = cf
            best_n = nh
            best_h = hf.copy()

        if deadline - time.time() > 110 and nh <= 800:
            alloc_extra = min(90.0, deadline - time.time() - 70.0)
            hb = to_binary(project(best_h, best_n), nh).astype(np.uint8)
            hb, cb = binary_optimize(hb, nh, time.time() + alloc_extra, seed=seed + 6000 + nh)
            cb = compute_c5(hb, nh)
            if cb < best_c - 1e-12:
                best_c = cb
                best_n = nh
                best_h = hb.copy()

    if deadline - time.time() > 35:
        small_budget = min(35.0, deadline - time.time() - 5.0)
        hf, cf = smooth_pgd(project(best_h, best_n), best_n, small_budget, [1000, 5000], seed=seed + 9999)
        if cf < best_c - 1e-12:
            best_c = cf
            best_n = best_n
            best_h = hf.copy()

    best_h = project(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
