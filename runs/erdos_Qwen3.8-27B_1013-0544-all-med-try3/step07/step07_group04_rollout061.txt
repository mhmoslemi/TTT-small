    """Find optimal h[i] given all other coordinates fixed."""
    dx = 2.0 / n
    # Compute the coefficient of h[i] in each c[p]
    # c[p] = C[p] + h[i] * coef[p]
    # where coef[p] = dx * (1 - h[i+p-(n-1)]) - dx * h[i-p+(n-1)]
    # (with out-of-bounds terms = 0, and we also have the -h[i]*dx*h[...] from
    #  the 1-h[...] expansion)
    
    # Actually, let me just compute C[p] directly:
    # c[p] with h[i] removed
    # This is the full correlation minus the contributions of h[i]
    
    # Full correlation:
    c_full = compute_all_lags(h, n)  # O(n²)
    
    # Contribution of h[i] to c[p]:
    # Term 1: dx * h[i] * (1-h[i+p-(n-1)]) for valid p
    # Term 2: -dx * h[i-p+(n-1)] * h[i] for valid p
    # Total: dx * h[i] * (1 - h[i+p-(n-1)] - h[i-p+(n-1)])
    #        (with out-of-bounds h = 0)
    
    coef = np.zeros(2*n-1)
    # Term 1 valid p range:
    p1_start = max(0, n-1-i)
    p1_end = min(2*n-2, 2*n-2-i)
    for p in range(p1_start, p1_end+1):
        j = i + p - (n-1)
        if 0 <= j <= n-1:
            coef[p] += dx * (1 - h[j])
    
    # Term 2 valid p range:
    p2_start = max(0, i)
    p2_end = min(2*n-2, i + n - 1)
    for p in range(p2_start, p2_end+1):
        j = i - p + (n-1)
        if 0 <= j <= n-1:
            coef[p] -= dx * h[j]
    
    # C[p] = c_full[p] - h[i] * coef[p]
    C = c_full - h[i] * coef
    
    # Now minimize max_p (C[p] + coef[p] * x) over x ∈ [0, 1]
    # This is a 1D convex optimization
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(lambda x: np.max(C + x * coef), bounds=(0, 1), method='bounded',

import numpy as np
from scipy.optimize import minimize
import time

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_all_lags(h, n)))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        mx = shifted.max()
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    c = compute_all_lags(x, n)
    return _smooth_max(c, alpha)
