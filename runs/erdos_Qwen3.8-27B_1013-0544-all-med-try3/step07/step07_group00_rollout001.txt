    thresh = mx - max(1e-8, rel_tol * mx)
    active = np.where(c >= thresh)[0]
    if len(active) > top_k:
        order = np.argsort(-c[active])
        active = active[order[:top_k]]
    return active, c, mx

def coordinate_descent(h, n, deadline, max_passes=None):
    best_c = compute_c5(h, n)
    h = h.copy()
    n_passes = 0
    delta_base = 0.5
    no_improve = 0
    
    for delta in [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline - 5:
            break
        for i_x in range(n):
            if time.time() > deadline - 3:
                return h, best_c
            for i in range(n):
                if time.time() > deadline - 2:
                    return h, best_c
                h_i = h[i]
                for d_sign in [1, -1]:
                    new_val = h_i + d_sign * delta
                    if new_val < -1e-15 or new_val > 1 + 1e-15:
                        continue
                    h[i] = new_val
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        no_improve = 0
                    else:
                        h[i] = h_i
                        no_improve += 1
                # Try moving to 0 or 1
                for target_val in [0.0, 1.0]:
                    if abs(h_i - target_val) < 1e-10:
                        continue
                    h[i] = target_val
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        no_improve = 0
                    else:
                        h[i] = h_i
                        no_improve += 1
        n_passes += 1
        if no_improve > 3 * n:
            no_improve = 0
        if max_passes and n_passes >= max_passes:
            break
    
    # Final: try 2-point moves with small deltas
    if time.time() < deadline - 5:
        for delta2 in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            if time.time() > deadline - 3:
                break
            improved = True
            while improved and time.time() < deadline - 2:
                improved = False
                rng = np.random.default_rng(42 + delta2 * 1e9)
                for _ in range(200):
                    if time.time() > deadline - 1:
                        break
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    if i == j:
                        continue
                    h_i, h_j = h[i], h[j]
                    for s in [delta2, -delta2]:
                        ni = h_i + s
                        nj = h_j - s
                        if ni < -1e-14 or ni > 1 + 1e-14 or nj < -1e-14 or nj > 1 + 1e-14:
                            continue
                        h[i], h[j] = ni, nj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = h_i, h_j
    return h, best_c

def top_only_lbfgs(h, n, deadline, alpha_list, num_active=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alpha_list:
        if time.time() > deadline - 5:
            break
        active, c_vec, mx = get_active_shifts(best_h, n, top_k=num_active, rel_tol=0.0005)
        if len(active) == 0:
            break
        for mu in [0.1, 0.01, 0.001]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(focus_objective, best_h, jac=focus_gradient,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 8000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 50000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def multi_restart_optimize(h, n, deadline, n_restarts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(123)
    
    for r in range(n_restarts):
        if time.time() > deadline - 60:
            break
        # Perturb
        perturbation_scale = 0.02 / (r + 1)
        h_pert = best_h + rng.uniform(-perturbation_scale, perturbation_scale, n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Quick optimize from perturbed start
        t_sub = min(80, deadline - time.time() - 30)
        if t_sub < 10:
            break
        h_opt, c_opt = top_only_lbfgs(h_pert, n, time.time() + t_sub,
                                       alpha_list=[10000, 100000], num_active=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def full_lbfgs(h, n, deadline, configs):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(full_smooth_objective, best_h, jac=full_smooth_gradient,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 40000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample_and_optimize(h_fine_orig, n_fine, n_coarse, deadline):
    """Downsample to n_coarse, optimize, upsample back."""
    x_fine = np.linspace(0, 2, n_fine)
    x_coarse = np.linspace(0, 2, n_coarse)
    h_c = np.interp(x_coarse, x_fine, h_fine_orig)
    h_c = project_feasible(h_c, n_coarse)
    
    t_opt = min(40, (deadline - time.time()) * 0.4)
    if t_opt > 10:
        h_c, c_c = top_only_lbfgs(h_c, n_coarse, time.time() + t_opt,
                                   alpha_list=[5000, 50000, 200000], num_active=4)
    
    h_up = np.interp(x_fine, x_coarse, h_c)
    h_up = project_feasible(h_up, n_fine)
    c_up = compute_c5(h_up, n_fine)
    return h_up, c_up

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    
    # Phase 1: Warm start with moderate alpha (60s)
    t1 = min(t0 + 60, deadline - 600)
    if time.time() < t1:
        h, best_c = full_lbfgs(h, n, t1, configs=[(200, 10.0), (1000, 100.0)])
    
    # Phase 2: Increasing alpha full optimization (120s)
    t2 = min(time.time() + 120, deadline - 450)
    if time.time() < t2:
        h, best_c = full_lbfgs(h, n, t2, configs=[(5000, 500.0), (20000, 2000.0), (50000, 5000.0)])
    
    # Phase 3: Top-shift focused optimization with large alpha (150s)
    t3 = min(time.time() + 150, deadline - 300)
    if time.time() < t3:
        h, best_c = top_only_lbfgs(h, n, t3, alpha_list=[50000, 200000, 1000000], num_active=3)
    
    # Phase 4: Even sharper top-shift (100s)
    t4 = min(time.time() + 100, deadline - 200)
    if time.time() < t4:
        h, best_c = top_only_lbfgs(h, n, t4, alpha_list=[1000000, 5000000, 10000000], num_active=3)
    
    # Phase 5: Re-identify active shifts and optimize (80s)
    t5 = min(time.time() + 80, deadline - 150)
    if time.time() < t5:
        h, best_c = top_only_lbfgs(h, n, t5, alpha_list=[2000000, 10000000], num_active=4)
    
    # Phase 6: Coordinate descent (200s)
    t6 = min(time.time() + 200, deadline - 60)
    if time.time() < t6:
        h, best_c = coordinate_descent(h, n, t6)
    
    # Phase 7: Random restarts (60s)
    t7 = min(time.time() + 60, deadline - 30)
    if time.time() < t7:
        h, best_c = multi_restart_optimize(h, n, t7, n_restarts=2)
    
    # Phase 8: Final coordinate descent on best (40s)
    t8 = min(time.time() + 40, deadline - 5)
    if time.time() < t8:
        h, best_c = coordinate_descent(h, n, t8)
    
    # Phase 9: Final top-shift polish
    if time.time() < deadline - 3:
        h, best_c = top_only_lbfgs(h, n, deadline - 2, alpha_list=[5000000], num_active=2)
    
    # Final projection and evaluation
    h = project_feasible(h, n)
    final_c = compute_c5(h, n)
    if final_c < best_c:
        best_c = final_c
    
    return h, best_c, n
