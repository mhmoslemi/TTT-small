        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        n_iter += 1
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h = h_new
        
        # Adaptive scheduling
        frac = min(1.0, n_iter / 5000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        target_lr = lr_start * (0.3 ** frac)
        lr = max(0.02, target_lr)
        
        if patience >= 500:
            lr *= 0.7
            v *= 0.4
            patience = 0
        
        if n_iter % 2000 == 1999:
            # Re-check and if stuck, perturb
            if patience > 2000:
                h = project_feasible(best_h + 0.01 * np.random.randn(n), n)
                v = np.zeros(n)
                lr = 0.5
                patience = 0
                alpha = max(10, alpha * 0.3)
    
    return best_h, best_c

def top_lag_local_search(h, n, deadline, top_k=5, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        improved = False
        
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = int(k_idx) - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            
            if lag > 0:
                # Want to reduce h[i]*(1-h[i+m])
                # Transfer from high h[i] to high h[i+m]
                scores = h[0:n-m] * (1.0 - h[m:n])
                # Try pairs with highest score
                top_pairs = np.argsort(scores)[-20:]
            else:
                # Want to reduce h[i-m]*(1-h[i])
                m = -lag
                scores = h[m:n] * (1.0 - h[0:n-m])
                top_pairs = np.argsort(scores)[-20:]
            
            for p in top_pairs:
                if time.time() > deadline:
                    break
                if lag > 0:
                    i, j = p, p + m
                    if h[i] > h[j]:
                        for delta in [0.001, 0.005, 0.01, 0.02]:
                            if h[i] - delta < 0 or h[j] + delta > 1:
                                continue
                            h[i] -= delta
                            h[j] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i] += delta
                                h[j] -= delta
                                break
                    break
                else:
                    j, i = p, p + m
                    if h[j] > h[i]:
                        for delta in [0.001, 0.005, 0.01, 0.02]:
                            if h[j] - delta < 0 or h[i] + delta > 1:
                                continue
                            h[j] -= delta
                            h[i] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h.copy()
                                improved = True
                            else:
                                h[j] += delta
                                h[i] -= delta
                                break
                    break
        
        if not improved:
            # Try broader: all top 10 lags with smaller steps
            c_all = compute_all_lags(h, n)
            top_idx = np.argsort(c_all)[-10:]
            still_improved = False
            for k_idx in top_idx:
                if time.time() > deadline:
                    break
                lag = int(k_idx) - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                npair = min(5, n - m)
                for _ in range(npair):
                    i = np.random.randint(0, n - m)
                    j = i + m
                    if h[i] > h[j] + 0.01:
                        delta = min(0.003, (h[i] - h[j]) * 0.3)
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            still_improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 0.01:
                        delta = min(0.003, (h[j] - h[i]) * 0.3)
                        h[j] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            still_improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
            if not still_improved and not improved:
                break
    
    return best_h, best_c

def binary_polish(h, n, deadline, max_rounds=50):
    """Try snapping values toward 0 or 1 while maintaining sum constraint."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_k_idx = np.argmax(c_all)
        lag = top_k_idx - (n - 1)
        m = abs(lag)
        
        if m >= n:
            break
        
        # Find pairs at this lag with values closest to 0.5 (most "wasteful")
        if lag > 0:
            pairs_scored = 1.0 - np.abs(2.0 * h[0:n-m] - 1.0) + np.abs(2.0 * h[m:n] - 1.0)
        else:
            pairs_scored = 1.0 - np.abs(2.0 * h[m:n] - 1.0) + np.abs(2.0 * h[0:n-m] - 1.0)
        
        top_pairs = np.argsort(pairs_scored)[::-1][:10]
        improved = False
        
        for p in top_pairs:
            if time.time() > deadline:
                break
            if lag > 0:
                i, j = p, p + m
            else:
                j, i = p, p + m
            
            # Try pushing h[i] toward 0 or 1 (whichever is closer)
            for target in [0.0, 1.0]:
                diff = target - h[i]
                if abs(diff) < 0.01:
                    continue
                if target > h[i]:
                    # Need to take from somewhere: reduce h[j]
                    need = diff
                    if h[j] < need:
                        continue
                else:
                    # Need to give to somewhere: increase h[j]
                    need = -diff
                    if 1.0 - h[j] < need:
                        continue
                
                h[i] = target
                h[j] += (h[i] - (target - diff))  # maintain sum
                # This is getting complicated, let me just do a simple transfer
                
                # Reset and do simple transfer
                h[i] -= diff
                h[j] += diff
                if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                    h[i] -= -diff  # undo
                    h[j] += -diff
                    continue
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] -= -diff
                    h[j] += -diff
                    h = project_feasible(h, n)
        
        if not improved:
            break
    
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000, top_m=5):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    n_lags = 2 * n - 1
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        return smooth_max_value(active, alpha)
    
    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        w = smooth_max_weights(active, alpha)
        grad = np.zeros(n)
        for idx, ki in enumerate(top_idx):
            if w[idx] < 1e-8:
                continue
            lag = int(ki) - (n - 1)
            grad += w[idx] * _lag_gradient(x, n, lag)
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    
    def sum_eq(x):
        return np.sum(x) - n / 2.0
    
    def sum_eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    best_h = None
    best_c = float('inf')
    best_n = 500
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    if h_init is None:
        h_init = np.full(n_init, 0.5)
    
    # Phase 1: Quick polish at initial resolution (n_init)
    t_phase1_end = t0 + min(80, budget_s * 0.08)
    if time.time() < t_phase1_end:
        h_p1, c_p1 = momentum_optimize(h_init, n_init, t_phase1_end,
                                        alpha_start=50, alpha_end=20000,
                                        lr_start=0.8, momentum=0.93)
        if c_p1 < best_c:
            best_c = c_p1
            best_h = h_p1
            best_n = n_init
    
    # Phase 2: Go to n=1000 resolution
    n_fine = 1000
    t_phase2_end = t0 + min(500, budget_s * 0.50)
    if time.time() < t_phase2_end - 30:
        h_fine = interpolate_h(best_h, len(best_h), n_fine)
        
        # Multi-start at high resolution
        starts = [h_fine.copy()]
        # Add a slightly perturbed version
        starts.append(project_feasible(h_fine + 0.02 * rng.randn(n_fine), n_fine))
        
        for si, h_start in enumerate(starts):
            if time.time() > t_phase2_end - 10:
                break
            alloc = (t_phase2_end - time.time()) * (0.6 if si == 0 else 0.4)
            if si > 0 and alloc < 60:
                break
            h_opt, c_opt = momentum_optimize(h_start, n_fine, time.time() + alloc,
                                              alpha_start=20, alpha_end=80000,
                                              lr_start=0.6, momentum=0.94)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                best_n = n_fine
    
    # Phase 3: SLSQP polish on top lags
    if best_h is not None and time.time() < deadline - 100:
        n_cur = len(best_h)
        h_pol = slsqp_polish(best_h, n_cur, alpha=50000, maxiter=3000, top_m=3)
        c_pol = compute_c5(h_pol, n_cur)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol
            best_n = n_cur
        
        if time.time() < deadline - 60:
            h_pol2 = slsqp_polish(best_h, n_cur, alpha=100000, maxiter=2000, top_m=5)
            c_pol2 = compute_c5(h_pol2, n_cur)
            if c_pol2 < best_c:
                best_c = c_pol2
                best_h = h_pol2
    
    # Phase 4: Top-lag local search
    if best_h is not None and time.time() < deadline - 40:
        ls_deadline = min(deadline - 5, time.time() + (deadline - time.time()) * 0.5)
        h_ls, c_ls = top_lag_local_search(best_h, len(best_h), ls_deadline, top_k=5)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
            best_n = len(best_h)
    
    # Phase 5: Try even higher resolution n=1200
    if best_h is not None and time.time() < deadline - 80:
        n_super = 1200
        h_super = interpolate_h(best_h, len(best_h), n_super)
        alloc = min(120, deadline - 20 - time.time())
        if alloc > 60:
            h_s, c_s = momentum_optimize(h_super, n_super, time.time() + alloc,
                                          alpha_start=30, alpha_end=50000,
                                          lr_start=0.5, momentum=0.94)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
                best_n = n_super
    
    # Phase 6: Final local search polish
    if best_h is not None and time.time() < deadline - 10:
        h_fin, c_fin = top_lag_local_search(best_h, len(best_h), deadline - 2, top_k=3, max_rounds=50)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
            best_n = len(best_h)
    
    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    # Fix sum exactly
    target = n_final / 2.0
    s = best_h.sum()
    if abs(s - target) > 1e-12:
        # Tiny adjustment to the element closest to 0.5
        diff = target - s
        idx = np.argmin(np.abs(best_h - 0.5))
        best_h[idx] += diff
        best_h = np.clip(best_h, 0, 1)
    
    best_c = compute_c5(best_h, n_final)
    
    # Verify feasibility
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    
    return best_h, float(best_c), n_final
