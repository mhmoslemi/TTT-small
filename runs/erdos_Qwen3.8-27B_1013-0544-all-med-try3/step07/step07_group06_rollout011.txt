    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Water filling
        if s > target:
            # Need to decrease: find threshold
            h = np.minimum(h, h - (s - target) / n)  # crude
        else:
            h = np.maximum(h, h + (target - s) / n)

    # If 2i-j is not in [0,n): (1-h[2i-j]) term is 0 (masked out)
    # If 2j-i is not in [0,n): (1-h[2j-i]) term is 0
    # Assuming both in range: = (1-h[2j-i]) - 1 - (1-h[2i-j]) = h[2i-j] - h[2j-i]
    # The true change should be: -1 (from the cross term) + any other legitimate changes
    # Hmm, this is getting too complex to verify analytically.
    
    # Let me just use the correction from the original code and verify numerically.
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2*n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2*n - 1:
        delta[k2] += 1.0
    
    return delta

def simulated_annealing(h, n, t_start, t_end, n_steps, deadline, rng, verbose=False):
    """SA on binary vector"""
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        # Temperature
        frac = step / max(n_steps - 1, 1)
        T = t_start * (t_end / t_start) ** frac
        
        # Random swap
        ip = rng.randint(len(ones))
        jp = rng.randint(len(zeros))
        i = ones[ip]
        j = zeros[jp]
        
        # Compute delta
        m_range = np.arange(-(n-1), n)
        delta = np.zeros(2*n - 1)
        im = i + m_range
        imn = i - m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += h[im[msk]]
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= 1.0 - h[imn[msk]]
        jm = j + m_range
        jmn = j - m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] += 1.0 - h[jm[msk]]
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] -= h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        if 0 <= k1 < 2*n-1: delta[k1] += 1.0
        k2 = n-1 + (i-j)
        if 0 <= k2 < 2*n-1: delta[k2] += 1.0
        
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        dC = new_c - cur_c
        
        if dC <= 0 or (T > 1e-15 and rng.random() < np.exp(-dC / T)):
            # Accept
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_c = new_c
            ones[ip] = j
            zeros[jp] = i
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c
