import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.clip(1.0 - h, 0, None)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

    dx = 2.0 / n
    c = np.correlate(h_in, 1 - h_in, mode='full') * dx
    return smooth_max_val(c, alpha)

    corr = np.correlate(h, 1 - h, mode='full')
    c = corr * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)

def grad_descent(h, n, deadline, alpha_start=50, alpha_end=500, lr=0.5, momentum=0.9):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    it = 0
    while time.time() < deadline:
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / 2000)
        g = c5_gradient(h_curr, n, alpha)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:

        h_curr = h_new

    h = h_bin.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.clip(1.0 - h, 0, None)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def c5_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c = corr * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    nn = 2 * n - 1
    for k in range(nn):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n-lag])
            grad[0:n-lag] += wk * (-h[lag:n])
        else:
            m = -lag
            grad[0:n-m] += wk * (1.0 - h[m:n])
            grad[m:n] += wk * (-h[0:n-m])
    return dx * grad

    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    it = 0
    stagnation = 0

        alpha = alpha0 + (alpha_end - alpha0) * min(1.0, it / 3000)
        g = c5_gradient(h_curr, n, alpha)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stagnation = 0
        else:

        h_curr = h_new

        if stagnation > 500:
            lr *= 0.6
            v *= 0.3
            stagnation = 0
        if it % 2000 == 1999:
            lr = max(0.05, lr * 0.85)
    return best_h, best_c

            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Main optimization at parent resolution
    main_alloc = min(400, (deadline - t0) * 0.4)
    h_opt, c_opt = optimize_at_resolution(best_h, n_init, t0 + main_alloc, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    
    # Try higher resolution
    n_high = min(1000, int(n_init * 1.5))
    if n_high > n_init and time.time() < deadline - 200:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_high)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_high)
        alloc_high = min(250, (deadline - time.time()) * 0.3)
        if alloc_high > 80:
            h_opt2, c_opt2 = optimize_at_resolution(h_interp, n_high, time.time() + alloc_high, rng)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2
    
    # Multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        n_restarts = max(1, int((deadline - time.time() - 40) / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(70, (deadline - time.time() - 30) * 0.35)
            if alloc < 30:
                break
            h_opt3, c_opt3 = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt3 < best_c:
                best_c = c_opt3
                best_h = h_opt3
    
    # Final intensive polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Final SLSQP
        if time.time() < deadline - 10:
            h_new = slsqp_optimize(best_h, n, 5000000, maxiter=6000, top_m=3)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Final local search
        if time.time() < deadline - 2:
            best_h, best_c = local_search_all_pairs(best_h, n, deadline - 1, top_k=5)
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
