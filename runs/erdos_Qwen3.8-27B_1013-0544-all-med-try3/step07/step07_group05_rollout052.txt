            j_hi2 = min(n, n + l)
            if j_hi2 > j_lo2:
                idx = np.arange(j_lo2, j_hi2)
                grad[idx] -= wk * dx * h[idx - l]
        
        # Find top "donors" (high h, positive grad) and "recipients" (low h, negative grad)
        # We want to decrease h where grad > 0 and increase h where grad < 0
        donors = np.where((grad > 0.01) & (h > 0.1))[0]
        recipients = np.where((grad < -0.01) & (h < 0.9))[0]
        
        if len(donors) == 0 or len(recipients) == 0:
            break
        
        # Try a few random pairs
        n_tries = min(20, len(donors) * len(recipients))
        improved = False
        for _ in range(n_tries):
            i = donors[np.random.randint(len(donors))]
            j = recipients[np.random.randint(len(recipients))]
            
            for d in [0.01, 0.005, 0.002, 0.001]:
                hi = h[i] - d
                hj = h[j] + d
                if 0 <= hi and hj <= 1:
                    h[i], h[j] = hi, hj
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i], h[j] = hi + d, hj - d
                        break
        if not improved:
            # Try bidirectional
            for d in [0.005, 0.002]:
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j: continue
                for sign in [1, -1]:
                    hi = h[i] + sign * d
                    hj = h[j] - sign * d
                    if 0 <= hi and hj <= 1:
                        h[i], h[j] = hi, hj
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - sign * d, hj + sign * d
            if not improved:
                break
    
    return best_h, best_c

def smart_pair_search(h, n, deadline, deltas, radius=100):
    """Focused pair search on nearby positions"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    # Move d from i to j
                    if h[i] >= d and h[j] <= 1 - d:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    # Move d from j to i
                    elif j > i and h[j] >= d and h[i] <= 1 - d:
                        h[j] -= d
                        h[i] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
            if not improved:
                break
    return best_h, best_c
