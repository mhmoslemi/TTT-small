```python
def project_onto_simplex_box(h, target_sum):
    # Project h onto {x: 0<=x[i]<=1, sum(x) = target_sum}
    # Using bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        lam = (lo + hi) / 2
        h_proj = np.clip(h - lam, 0, 1)
        s = h_proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(h - (lo+hi)/2, 0, 1)
```

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def project_simplex_box(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def compute_gradient_full(h, n, alpha, top_k=50):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(c.max())
    shifted = (c - m) * alpha
    shifted = np.clip(shifted, -100.0, 100.0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / wsum
    if top_k < len(w):
        idx = np.argsort(-w)[:top_k]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        w2sum = w2.sum()
        if w2sum > 1e-300:
            w = w2 / w2sum
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - n1
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
        else:
            sp = -s
            grad[:n - sp] += wk * (1.0 - h[sp:])
            grad[sp:] -= wk * h[:n - sp]
    return grad, m
