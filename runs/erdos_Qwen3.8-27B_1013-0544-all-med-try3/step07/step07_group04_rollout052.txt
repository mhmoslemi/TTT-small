def compute_c5(h, n):

def compute_all_lags(h, n):

def project_feasible(h, n):

    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()

def coordinate_polish(h_start, n, deadline, n_top_lags=8):

def slsqp_polish(h_start, n, alpha, maxiter=3000):
    def smooth_obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = cc.max()
        shifted = alpha * (cc - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

             'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h_start.copy()

def outer_lp_step(h_curr, n, trust):
    dx = 2.0 / n
    c = np.correlate(h_curr, 1.0 - h_curr, mode='full') * dx
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if m == 0:
            grad = 1.0 - 2.0 * h_curr
        elif lag > 0:
            grad = np.zeros(n)
            grad[:n - m] = 1.0 - h_curr[m:]
            grad[n - m:] = -h_curr[:n - m]
        else:
            grad = np.zeros(n)
            grad[:n - m] = -h_curr[m:]
            grad[n - m:] = 1.0 - h_curr[:n - m]
        linearized = c[k] + dx * (grad @ (h_var - h_curr))
        cons.append(linearized <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = h_var.value
                if h_new is not None:
                    return project_feasible(h_new, n)
            break
        except Exception:
            continue
    return h_curr.copy()

        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 2:
        h_pol, c_pol = coordinate_polish(best_h, n, min(deadline - 1, time.time() + 30), n_top_lags=10)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        c_init = compute_c5(h_init, n_init)
        best_h = h_init.copy()
        best_c = c_init

        alloc = min(350, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    n_candidates = [200, 250, 300, 350, 400, 500]
    if n_init not in n_candidates:
        n_candidates.append(n_init)

    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 30
        if remaining < 40:
            break
        alloc = min(80, remaining * 0.22)

        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)

        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(min(n_restarts, 8)):
            if time.time() > deadline - 40:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 30) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        pg_end = min(deadline - 25, time.time() + min(100, remaining * 0.5))
        h_pg, c_pg, _ = projected_gradient(best_h, n, pg_end,
                                           alpha_init=10000.0, alpha_final=500000.0,
                                           lr_init=0.01, momentum=0.92, top_m=15)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg

    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        h_new = slsqp_polish(best_h, n, alpha=500000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_pol, c_pol = coordinate_polish(best_h, n, deadline - 1, n_top_lags=12)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
