def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Simple projection: add/subtract to meet target
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s > 0:
            h += (diff / s) * room
    else:
        s = h.sum()
        if s > 0:
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

- C(i,j) is the cross-term correction

delta[i,j,lag] = [contribution from changing h[i] from 1 to 0] + [contribution from changing h[j] from 0 to 1] + [cross-correction]

Contribution from h[i]: 1→0:
- Terms where h[i] appears as h[n+lag]: changes from h[i] to 0. For n = i-lag: contribution changes from h[i]*(1-h[i-lag]) to 0. Delta: -h[i]*(1-h[i-lag]) = -(1-h[i-lag]) [since h[i]=1]
  This is valid when i-lag ∈ [0,n-1], i.e., lag ∈ [i-(n-1), i].
  Delta_i[lag] = -(1-h[i-lag]) for lag ∈ [i-(n-1), i]
  
- Terms where h[i] appears as (1-h[n]): changes from (1-h[i])=0 to 1. For n = i: contribution changes from h[i+lag]*(1-h[i])=0 to h[i+lag]*1 = h[i+lag].
  Valid when i+lag ∈ [0,n-1], i.e., lag ∈ [-(i), n-1-i].

  Valid when j+lag ∈ [0,n-1], i.e., lag ∈ [-j, n-1-j].
  Delta_j[lag] -= h[j+lag] for lag ∈ [-j, n-1-j]

Now, the cross-correction: when the same term is affected by both changes.
This happens when {n, n+lag} = {i, j}, i.e.:

        h_int = np.interp(xn, xo, best_h)
        h_int = project_feasible(h_int, nh)
        # SLSQP polish
        h_int = slsqp_optimize(h_int, nh, 50000, 300, p4_end - 100)
        c = compute_c5(h_int, nh)
        # Binary search
        hb = make_binary(h_int, nh)
        hb, cb = local_search(hb, nh, p4_end - 50, max_sweeps=5000, seed=seed+nh)
        cb = compute_c5(hb, nh)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            best_n = nh
        # Try golden ratio init at this resolution
        hg = golden_ratio_init(nh)
        hg, cg = local_search(hg, nh, min(p4_end - 20, t0 + 700), max_sweeps=3000, seed=seed+100+nh)
        cg = compute_c5(hg, nh)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()
            best_n = nh

    # Phase 5: Multi-restart at best resolution
    p5_end = t0 + 950
    n_cur = best_n
    for restart in range(10):
        if time.time() > p5_end:
            break
        # Perturb current best
        hp = best_h.copy()
        n_ones = int(np.sum(hp == 1))
        n_zeros = int(np.sum(hp == 0))
        n_flips = max(2, n_cur // 25)
        for _ in range(n_flips):
            oi = np.where(hp == 1)[0]
            zi = np.where(hp == 0)[0]
            if len(oi) > 0 and len(zi) > 0:
                a = rng.choice(oi)
                b = rng.choice(zi)
                hp[a], hp[b] = 0.0, 1.0
        hp, cp = local_search(hp, n_cur, min(p5_end, t0 + 950), max_sweeps=3000, seed=seed+200+restart)
        cp = compute_c5(hp, n_cur)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
            best_n = n_cur

    # Phase 6: SLSQP + binary polish on best
    if time.time() < dl - 30:
        h_pol = project_feasible(best_h + 0.03 * rng.randn(best_n), best_n)
        for alpha in [100000, 1000000]:
            if time.time() > dl - 15:
                break
            h_pol = slsqp_optimize(h_pol, best_n, alpha, 500, dl - 10)
        c_pol = compute_c5(h_pol, best_n)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol.copy()
        # Binary from polished
        hb = make_binary(h_pol, best_n)
        if time.time() < dl - 5:
            hb, cb = local_search(hb, best_n, dl, max_sweeps=5000, seed=seed+500)
            cb = compute_c5(hb, best_n)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                best_n = best_n

    # Phase 7: Try annealing from best
    if time.time() < dl - 10:
        hb = make_binary(best_h, best_n)
        hb, cb = local_search(hb, best_n, dl, max_sweeps=10000, seed=seed+999,
                             use_annealing=True, initial_temp=0.5)
        cb = compute_c5(hb, best_n)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            best_n = best_n

    # Final
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
