def project(h, n):
    """Project h to satisfy 0≤h≤1 and sum(h)=n/2."""
    target = n / 2.0
    h = h.copy()
    # Bisection on threshold
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual

    if abs(diff) > 1e-10:

            if s > 1e-15:

        else:
            s = hp.sum()
            if s > 1e-15:
