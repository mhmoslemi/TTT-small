∂c_m/∂h[j] = (j ≥ m ? (1-h[j-m]) : 0) - (j ≤ n-1-m ? h[j+m] : 0)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj_all(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    shifted = alpha * (active - m)

def _lag_gradient(h, n, lag):
    """Compute gradient of c_k w.r.t. h for a specific lag."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m >= n:
        return grad
    if lag > 0:
        # c_m = sum_{i=0}^{n-1-m} h[i+m](1-h[i])
        for i in range(n - m):
            grad[i + m] += (1.0 - h[i])
            grad[i] -= h[i + m]

        for i in range(m, n):
            grad[i - m] += (1.0 - h[i])
            grad[i] -= h[i - m]
    return grad
