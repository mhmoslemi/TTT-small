                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue

                    if abs(move) < 1e-12:
                        continue
                    ni = h[i] + move
                    nj = h[j] - move
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i] = ni; h[j] = nj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            progressed = True
                        else:
                            h[i] -= move; h[j] += move
    return h, best_c

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_penalty(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx2 = sh.max()
    if mx2 > 500:
        val = m + (mx2 + np.log(np.sum(np.exp(sh - mx2)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_penalty(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_slsqp(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx2 = sh.max()
    if mx2 > 500:
        val = m + (mx2 + np.log(np.sum(np.exp(sh - mx2)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def cons_fun(x, n):
    return float(x.sum() - n / 2.0)

def cons_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 1.0), (500, 5.0), (1000, 10.0), (5000, 50.0),
        (10000, 100.0), (50000, 500.0), (100000, 1000.0),
        (500000, 5000.0), (1000000, 10000.0), (5000000, 50000.0),
    ]
    for alpha, mu in configs:
        remaining = deadline - time.time()
        if remaining < 5:
            break
        try:
            res = minimize(obj_penalty, h, jac=grad_penalty,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 15,
                                   'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}]
    remaining = deadline - time.time()

        if c_new < best_c:

            h = h_new
    except Exception:
        pass
    return h, best_c

def newton_steps(h, n, deadline, alpha=100000, max_steps=500):
    """Newton-like steps using approximate Hessian."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Simple diagonal scaling: step = -g / (diag_approx)
        # The "curvature" at each point is roughly related to the local second-order effects
        # Use a simple scaling
        diag_approx = 1.0 + np.abs(h) * (1.0 - h) * 4  # rough
        step = -g / np.maximum(diag_approx, 0.1)
        step_norm = np.linalg.norm(step)
        if step_norm < 1e-15:
            break
        
        # Line search
        improved = False
        for trial in range(40):
            lr = 0.1 / (trial + 1)
            h_new = h + lr * step
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            # Try direct gradient descent step
            step_g = -g / g_norm
            for trial in range(30):
                lr = 0.05 / (trial + 1)
                h_new = project_feasible(h + lr * step_g, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    h = h_new
                    best_c = c_new
                    improved = True
                    break
            if not improved:
                break
    
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    for d in deltas:
        progressed = True
        while progressed and time.time() < deadline:
            progressed = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - (radius if radius else n))
                hi = min(n, i + (radius if radius else n) + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            progressed = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            progressed = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
