    # Thue-Morse: t(i) = number of 1s in binary representation of i, mod 2
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # We need exactly n/2 ones
    # Count current ones
    ones_count = tm.sum()
    target = n / 2
    if ones_count == target:
        return tm
    # Adjust: flip some bits
    # If too many ones, flip some 1s to 0s (starting from the end)
    result = tm.copy()
    if ones_count > target:

        ones_idx = np.where(result == 1)[0]

        for idx in ones_idx[-excess:]:
            result[idx] = 0
    else:
        deficit = int(target - ones_count)
        zeros_idx = np.where(result == 0)[0]
        for idx in zeros_idx[:deficit]:
            result[idx] = 1

import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()

        if abs(diff) < 1e-15:
            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    target = int(n / 2)

    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

    h[idx] = 1.0
    return h

def generate_periodic_binary(n, period, rng):
    """Generate periodic binary pattern"""
    half = (period + 1) // 2
    # Create a period with 'half' ones
    pat = np.zeros(period)
    ones_pos = rng.choice(period, half, replace=False)
    pat[ones_pos] = 1.0
    result = np.tile(pat, n // period + 1)[:n]
    # Adjust count
    ones_count = int(result.sum())
    target = int(n // 2)
    if ones_count > target:
        idx = np.where(result == 1)[0]
        to_flip = idx[rng.permutation(len(idx))[:ones_count - target]]
        result[to_flip] = 0
    elif ones_count < target:
        idx = np.where(result == 0)[0]
        to_flip = idx[rng.permutation(len(idx))[:target - ones_count]]
        result[to_flip] = 1
    return result

def binary_local_search(h_bin, n, deadline, rng, max_iters=5000000):
    """Local search: randomly swap a 1 and a 0"""
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Random swap
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue  # No change
        
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            current_c = c
        else:
            h[i], h[j] = h[j], h[i]  # Undo
    
    return h, best_c

def binary_simulated_annealing(h_bin, n, deadline, rng, T0=0.5, T_end=0.001):
    """Simulated annealing on binary space"""
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    current_c = best_c
    
    total_iters = 10000000  # Will be limited by time
    it = 0
    while time.time() < deadline:
        T = T0 * (T_end / T0) ** (it / total_iters)
        it += 1
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h, n)

            if c < best_c:

def slsqp_optimize(h, n, alpha, maxiter=3000):

        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)

    try:

                       options={'maxiter': maxiter, 'ftol': 1e-16})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = int(n / 2)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def generate_random_binary(n, rng):
    target = int(n / 2)
    idx = rng.permutation(n)[:target]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def generate_periodic_binary(n, period, rng):
    half = (period + 1) // 2
    pat = np.zeros(period)
    ones_pos = rng.choice(period, half, replace=False)
    pat[ones_pos] = 1.0
    result = np.tile(pat, n // period + 1)[:n]
    ones_count = int(result.sum())
    target = int(n // 2)
    if ones_count > target:
        idx = np.where(result == 1)[0]
        to_flip = idx[rng.permutation(len(idx))[:ones_count - target]]
        result[to_flip] = 0
    elif ones_count < target:
        idx = np.where(result == 0)[0]
        to_flip = idx[rng.permutation(len(idx))[:target - ones_count]]
        result[to_flip] = 1
    return result

def binary_local_search(h_bin, n, deadline, rng, max_iters=10000000):
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
        else:
            h[i], h[j] = h[j], h[i]
    return h, best_c

def binary_simulated_annealing(h_bin, n, deadline, rng, T0=2.0, T_end=0.0001):
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    current_c = best_c
    total_iters = 20000000
    it = 0
    while time.time() < deadline:
        T = T0 * (T_end / T0) ** (min(it / total_iters, 1.0))
        it += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h, n)
        delta = c - current_c
        if delta < 0 or (T > 1e-7 and np.log(rng.rand() + 1e-300) < -delta / T):
            current_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def continuous_fine_tune(h, n, deadline, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        if time.time() > deadline:
            break
        for _ in range(30000):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = step * (1.0 if rng.rand() < 0.5 else -1.0)
            hi = h[i] + d
            hj = h[j] - d
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                h[i], h[j] = hi, hj
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                else:
                    h[i], h[j] = hi - d, hj + d
    return project_feasible(h, n), best_c

def systematic_pair_search(h, n, deadline, steps):

    for step in steps:

                            improved = True

                        else:

def slsqp_optimize(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def _con(x):
        return float(np.sum(x) - n / 2.0)
    def _con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def multi_restart_binary_search(base_h, n, deadline, rng, num_restarts=5, per_restart_s=20):
    best_h = base_h.copy()
    best_c = compute_c5(base_h, n)
    for r in range(num_restarts):
        if time.time() > deadline - per_restart_s:
            break
        # Perturb: flip a random subset of bits
        h_pert = base_h.copy()
        n_flips = rng.randint(3, 15)
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(n_flips):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i], h_pert[j] = 0.0, 1.0
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
        alloc = min(per_restart_s, deadline - time.time() - 5)
        if alloc < 3:
            break
        h_opt, c_opt = binary_local_search(h_pert, n, time.time() + alloc, rng, max_iters=5000000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c
