One more consideration: the "top-lag targeting" algorithm. Here's a more sophisticated version:

**Algorithm: Weighted Lag Reduction (WLR)**

Given current h with overlap vector c:
1. Compute weights w_k = max(0, c_k - C_target)² for each lag k, where C_target is slightly below the current max
2. The "reduction direction" is: d = -∑_k w_k * ∇_h c_k
3. Take a step: h_new = project(h + lr * d)
4. Update C_target = max(0.95 * current_max, C_target - 0.001)
5. Repeat

This is different from smooth-max because:
- It explicitly targets lags that exceed a target level
- The quadratic weighting makes it focus more on the worst lags
- The target level decreases over time, creating a "progressive tightening"

This is essentially a penalty method with a decreasing penalty parameter.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def grad_smooth_max(h, n, alpha):

    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        lag = k - (n-1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def grad_penalty(h, n, target, p=2):
    """Gradient of sum of (max(0, c_k - target))^p over all lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    excess = np.maximum(0.0, c - target)
    if excess.max() < 1e-15:
        return np.zeros(n)
    w = p * excess ** (p - 1)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        lag = k - (n-1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def momentum_optimize(h, n, deadline, alpha_schedule=None, lr=1.5, momentum=0.93):

    h_curr = h.copy()

    v = np.zeros(n)

def peak_shaving(h, n, deadline, num_phases=10):
    """Iteratively target and reduce the worst lags."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for phase in range(num_phases):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        current_max = c_all.max()
        # Target the top few lags
        top_k = np.argsort(c_all)[-min(5, 2*n-1):]
        # Build a weighted gradient focusing on top lags
        w = np.zeros(len(c_all))
        for idx in top_k:
            w[idx] = (c_all[idx] - 0.95 * current_max) ** 2
        if w.sum() < 1e-15:
            w = np.zeros(len(c_all))
            w[int(np.argmax(c_all))] = 1.0
        w = w / w.sum()
        
        grad = np.zeros(n)
        for k in range(len(c_all)):

                break

def slsqp_top_m(h, n, top_m, alpha, maxiter=3000):
    """SLSQP optimization on smooth max of top m lags."""
    dx = 2.0 / n
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        grad = np.zeros(n)
        for ii, k in enumerate(top_idx):
            if w[ii] < 1e-10:
                continue
            lag = k - (n-1)
            grad += w[ii] * _lag_gradient(x, n, lag)
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
