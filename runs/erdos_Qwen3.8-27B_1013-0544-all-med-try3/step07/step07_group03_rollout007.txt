    """Compute subgradient of max_k C(k) w.r.t. h.
    c_vals has length 2n-1, indexed 0..2n-2.

    Active lags are those where c_vals >= c_max - eps.

    mx = sh.max()

    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    sh = alpha * (ct - m)
    mx = sh.max()
    if mx > 500:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Load parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)

    # Phase A: Optimize at parent resolution
    alloc_a = min(300, (deadline - t0) * 0.3)
    h_a, c_a = optimize_at_n(h_init, n_init, t0 + alloc_a)
    if c_a < best_c:
        best_c = c_a
        best_h = h_a

    # Phase B: Try different resolutions
    for n_try in [200, 300]:
        if time.time() > deadline - 150:
            break
        if n_try == n_init:
            continue
        # Interpolate parent to new resolution
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(150, (deadline - time.time()) * 0.25)
        if alloc < 50:
            break
        h_b, c_b = optimize_at_n(h_interp, n_try, time.time() + alloc)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b

    # Phase C: Try periodic/smooth initializations
    for init_type in ['periodic', 'smooth']:
        if time.time() > deadline - 100:
            break
        for n_try in [200, n_init]:
            if time.time() > deadline - 60:
                break
            if init_type == 'periodic':
                h_try = make_periodic_init(n_try, period=1.0, duty=0.5)
            else:
                h_try = make_smooth_init(n_try, freq=1.0, phase=0.5)
            alloc = min(80, (deadline - time.time()) * 0.2)
            if alloc < 30:
                continue
            h_c, c_c = optimize_at_n(h_try, n_try, time.time() + alloc)
            if c_c < best_c:
                best_c = c_c
                best_h = h_c

    # Phase D: Multi-restart from best solution
    n_best = len(best_h)
    remaining = deadline - time.time()
    n_restarts = max(2, int(remaining / 80))
    for r in range(n_restarts):
        if time.time() > deadline - 40:
            break
        scale = 0.02 + 0.04 * (r / max(1, n_restarts - 1))
        if r % 2 == 1:
            h_p = 1.0 - best_h + rng.normal(0, scale, n_best)
        else:
            h_p = best_h + rng.normal(0, scale, n_best)
        h_p = project_feasible(h_p, n_best)
        alloc = min(60, (deadline - time.time() - 20) * 0.4)
        if alloc < 20:
            break
        h_r, c_r = optimize_at_n(h_p, n_best, time.time() + alloc)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r

    # Phase E: Extended final refinement on best solution
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        # Gradient step
        if time.time() < deadline - 20:
            h_e, c_e = gradient_step_optimize(best_h, n, deadline - 20,
                                               n_iter=50000, init_step=0.003)
            if c_e < best_c:
                best_c = c_e
                best_h = h_e

        # SLSQP final
        if time.time() < deadline - 10:
            ht = slsqp_full(best_h, n, 100000, maxiter=1000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht

        # Fine pair search
        if time.time() < deadline - 3:
            if n <= 250:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                    [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
            else:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                    [0.005, 0.002, 0.001, 0.0005], radius=50)

    final_h = project_feasible(best_h, len(best_h))
    final_c = compute_c5(final_h, len(final_h))
    return final_h, final_c, len(final_h)
