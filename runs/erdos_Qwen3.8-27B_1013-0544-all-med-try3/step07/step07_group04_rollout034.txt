import numpy as np
from scipy.optimize import minimize, brentq
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0, 1)
    return proj

def compute_gradient(h, n, p):
    """Compute gradient of sum_k p[k] * C_k(h)
    p is an array of length 2n-1, indexed by r (correlate index)
    Lag k = r - (n-1)
    """
    grad = np.zeros(n)
    # Lag 0
    if p[n-1] > 0:
        grad[:] += p[n-1] * (1 - 2*h)
    # Positive lags k = 1, ..., n-1 (r = n-1+k)
    for k in range(1, n):
        pk = p[n-1+k]
        if pk == 0:
            continue
        # (1-h[j-k]) for j >= k
        grad[k:] += pk * (1 - h[:n-k])
        # -h[j+k] for j <= n-1-k

            best_h = h_new.copy()

            step *= 0.8
            if stall > 20:
                break
        
        prev_grad = grad
        prev_h = h
        h = h_new
    
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, n):
    target = n / 2.0
    # Bisection for lambda
    # f(lambda) = sum(clip(x - lambda, 0, 1)) is decreasing in lambda
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient(h, n, lag_weights):
    """
    Compute gradient of sum_k lag_weights[k] * C_k(h)
    lag_weights: array of length 2n-1 (same indexing as correlate output)
    Returns: gradient vector of length n
    """
    grad = np.zeros(n)
    n_minus_1 = n - 1
    
    # Lag 0: C_0 = sum h[i]*(1-h[i]), dC_0/dh[j] = 1 - 2h[j]
    w0 = lag_weights[n_minus_1]
    if w0 != 0.0:
        grad[:] += w0 * (1.0 - 2.0 * h)
    
    # Positive lags k = 1, ..., n-1 (index in correlate: n_minus_1 + k)
    for k in range(1, n):
        w = lag_weights[n_minus_1 + k]
        if w == 0.0:
            continue
        # dC_k/dh[j] = (1-h[j-k]) for j >= k, and -h[j+k] for j <= n-1-k
        grad[k:] += w * (1.0 - h[:n-k])
        grad[:n-k] -= w * h[k:]
    
    # Negative lags: lag = -k for k = 1, ..., n-1 (index in correlate: n_minus_1 - k)
    for k in range(1, n):
        w = lag_weights[n_minus_1 - k]
        if w == 0.0:
            continue
        # dC_{-k}/dh[j] = (1-h[j+k]) for j <= n-1-k, and -h[j-k] for j >= k
        grad[:n-k] += w * (1.0 - h[k:])
        grad[k:] -= w * h[:n-k]
    
    return grad

    s = exp_c.sum()

    return exp_c / s
