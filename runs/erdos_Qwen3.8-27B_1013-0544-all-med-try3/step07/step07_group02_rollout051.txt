    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def smooth_max(c, alpha):

    mx = shifted.max()
    if mx > 500:

def slsqp_active_lags(h, n, top_m, alpha, maxiter, ftol):
    """SLSQP on top-m lags with smooth max."""
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]

        dx = 2.0 / n
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        return smooth_max(active, alpha)
    
    def obj_grad(x):

        active = cc[top_idx]
        w = smooth_max_grad_weights(active, alpha)
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[i] * lag_gradient(x, n, lag)
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:

                      bounds=bounds, constraints=cons,

        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def multi_resolution_optimize(h_base, n_base, deadline, rng, n_list):

        h_start = np.interp(x_new, x_old, h_base)
        h_start = project_feasible(h_start, n)
        
        # Also try from best found so far
        x_best = np.linspace(0, 2, len(best_h))

            break

        sj = rng.choice(len(zeros))

        oz = zeros[sj]

        oz_end = min(oz + blk - 1, n - 1)

        z_block = list(range(oz, oz_end + 1))

                    curr_h[x] = 0.0

                    old_vals.append(curr_h[x])
                    curr_h[x] = 1.0

                    curr_h[:] = np.where(curr_h > 0.5, 1.0, 0.0)

        T = max(T * cool, T_min)

    return best_h, best_c

def slsqp_top_lags(h, n, top_m, alpha, maxiter=3000, ftol=1e-14):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]

    def obj(x):
        dx = 2.0 / n
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        return smooth_max_val(active, alpha)

    def obj_grad(x):
        dx = 2.0 / n
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        w = smooth_max_weights(active, alpha)
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[i] * lag_gradient(x, n, lag)
        return grad * dx

    def sum_fun(x):
        return float(np.sum(x) - n / 2.0)

    def sum_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]

    try:
        res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def thue_morse(n):
    t = np.array([int(np.binary_repr(i & -i).count('1')) % 2 for i in range(n)], dtype=float)
    # Actually Thue-Morse is popcount mod 2
    t = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    return t

def interpolate_to(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from)
    x_new = np.linspace(0, 2, n_to)
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    if h_init is None:
        h_init = np.full(n_init, 0.5)

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Gradient descent from initial
    p1_end = min(t0 + 120, deadline - 400)
    if p1_end > t0 + 10:
        h1, c1 = gradient_line_search(best_h, n_init, 150, p1_end, lr0=0.3)
        if c1 < best_c:
            best_c, best_h = c1, h1

    # Phase 2: SLSQP on top lags
    if time.time() < deadline - 300:
        s_end = min(deadline - 250, time.time() + 100)
        for top_m in [3, 5, 10]:
            if time.time() > s_end:
                break
            h2 = slsqp_top_lags(best_h, n_init, top_m, 500, maxiter=2500, ftol=1e-14)
            c2 = compute_c5(h2, n_init)
            if c2 < best_c:
                best_c, best_h = c2, h2

    # Phase 3: Binary swap local search (extended)
    if time.time() < deadline - 200:
        bs_end = min(deadline - 150, time.time() + 150)
        h3, c3 = top_lag_swaps(best_h, n_init, bs_end, max_rounds=2000)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Fine continuous search
    if time.time() < deadline - 100:
        fc_end = min(deadline - 60, time.time() + 80)
        h4, c4 = fine_continuous_search(best_h, n_init, fc_end, max_rounds=500)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Block swap SA
    if time.time() < deadline - 80:
        sa_end = min(deadline - 40, time.time() + 50)
        h5, c5 = block_swap_sa(best_h, n_init, sa_end, rng, max_iter=300000)
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: Multi-resolution exploration
    if time.time() < deadline - 60:
        n_candidates = [400, 450, 550, 600]
        for n_try in n_candidates:
            if time.time() > deadline - 40:
                break
            h_interp = interpolate_to(best_h, len(best_h), n_try)
            c_interp = compute_c5(h_interp, n_try)
            # Quick local improvement
            quick_end = min(deadline - 30, time.time() + 40)
            h_mr, c_mr = top_lag_swaps(h_interp, n_try, quick_end, max_rounds=500)
            if c_mr < best_c:
                best_c, best_h = c_mr, h_mr

    # Phase 7: Final combined polish at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 20:
        # Gradient
        g_end = min(deadline - 10, time.time() + 40)
        h_g, c_g = gradient_line_search(best_h, n_best, 200, g_end, lr0=0.2)
        if c_g < best_c:
            best_c, best_h = c_g, h_g

        # Binary swaps
        if time.time() < deadline - 10:
            b_end = min(deadline - 5, time.time() + 30)
            h_b, c_b = top_lag_swaps(best_h, n_best, b_end, max_rounds=1000)
            if c_b < best_c:
                best_c, best_h = c_b, h_b

        # Fine continuous
        if time.time() < deadline - 5:
            f_end = min(deadline - 1, time.time() + 20)
            h_f, c_f = fine_continuous_search(best_h, n_best, f_end, max_rounds=200)
            if c_f < best_c:
                best_c, best_h = c_f, h_f

    # Phase 8: Multistart if time remains
    if time.time() < deadline - 25:
        n_best = len(best_h)
        remaining = deadline - time.time() - 10
        n_restarts = max(1, int(remaining / 25))
        for r in range(n_restarts):
            if time.time() > deadline - 10:
                break
            scale = 0.008 * (1.0 + 0.4 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            rd = min(deadline - 3, time.time() + 20)
            hr, cr = gradient_line_search(hp, n_best, 200, rd, lr0=0.3)
            if time.time() < rd:
                hr2, cr2 = top_lag_swaps(hr, n_best, min(deadline - 2, time.time() + 5), max_rounds=200)
                if cr2 < cr:
                    hr, cr = hr2, cr2
            if cr < best_c:
                best_c, best_h = cr, hr
                best_h = project_feasible(best_h, n_best)
                best_c = compute_c5(best_h, n_best)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
