        if abs(diff) < 1e-14:

            if tot < 1e-15:

            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

    """Find t in [0,1] minimizing max_k (A[k] + B[k]*t)."""
    pos = B > 1e-14
    neg = B < -1e-14
    zro = ~(pos | neg)
    
    # Check zro: need A[zro] <= V for all
    if np.any(zro):

    else:
        V_min_zero = 0.0

    V_hi = max(np.max(A), np.max(A + B))

            break
        V_mid = (V_lo + V_hi) / 2.0
        
        # Feasibility check: exists t in [0,1] with A[k] + B[k]*t <= V_mid for all k
        t_ub = 1.0
        if np.any(pos):
            t_ub = min(1.0, np.min((V_mid - A[pos]) / B[pos]))
        
        t_lb = 0.0
        if np.any(neg):
            t_lb = max(0.0, np.max((V_mid - A[neg]) / B[neg]))
        
        if t_lb <= t_ub:
            V_hi = V_mid
        else:
            V_lo = V_mid
    
    V_opt = V_hi
    
    # Find t
    t_ub = 1.0
    if np.any(pos):
        t_ub = min(1.0, np.min((V_opt - A[pos]) / B[pos]))
    t_lb = 0.0
    if np.any(neg):

    t_opt = np.clip((t_lb + t_ub) / 2.0, 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(A, B):
    """Find t in [0,1] minimizing max_k (A[k] + B[k]*t)."""
    m = len(A)
    pos = B > 1e-14
    neg = B < -1e-14
    zro = ~(pos | neg)

    if np.any(zro):
        V_lo = max(0.0, float(np.max(A[zro])))
    else:
        V_lo = 0.0
    V_hi = max(float(np.max(A)), float(np.max(A + B)))

    if V_hi - V_lo < 1e-15:
        return 0.0, V_hi

    for _ in range(55):
        if V_hi - V_lo < 1e-16:
            break
        V_mid = 0.5 * (V_lo + V_hi)

        if np.any(pos):
            t_ub = min(1.0, float(np.min((V_mid - A[pos]) / B[pos])))
        t_lb = 0.0
        if np.any(neg):
            t_lb = max(0.0, float(np.max((V_mid - A[neg]) / B[neg])))
        if t_lb <= t_ub:
            V_hi = V_mid
        else:
            V_lo = V_mid

    t_ub = 1.0

    h = h_init.copy()
    c = compute_c5_vec(h, n)
    best_c = float(np.max(c))
    best_h = h.copy()
    dx = 2.0 / n
    m = 2 * n - 1
    n1 = n - 1
    k_arr = np.arange(-n1, n, dtype=np.int64)

    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if use_rand:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            old_val = h[i]
            ki = i + k_arr
            kj = i - k_arr
            vi = (ki >= 0) & (ki < n)
            vj = (kj >= 0) & (kj < n)
            alpha = np.zeros(m)
            if np.any(vi):
                alpha[vi] = (1.0 - h[ki[vi]]) * dx
            beta = np.zeros(m)

            B = alpha - beta
            A = c - old_val * B
            t_opt, _ = solve_1d_minimax(A, B)

            new_val = float(np.clip(new_val, 0.0, 1.0))

            c += (new_val - old_val) * B

def compute_grad_smooth(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        w = np.exp(c_shift - mx)
    else:
        w = np.exp(c_shift)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nn = 2 * n - 1
    for m_idx in range(nn):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad
