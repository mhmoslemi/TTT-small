import time

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, lags, mu):
    c = compute_c5_vec(x, n)
    if lags is not None:
        vals = c[lags]
    else:
        vals = c
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:

    else:
        lse = np.log(np.sum(np.exp(sh)))
    val = m + lse / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, lags, mu):
    c = compute_c5_vec(x, n)
    if lags is not None:
        vals = c[lags]
        lags_iter = lags
    else:
        vals = c
        lags_iter = np.arange(len(vals))
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, lags_iter):
        if wi < 1e-16:

            grad[:nv] += wm * (-x[kk:])

    bounds = [(0.0, 1.0)] * n

                      args=(n, alpha, lags, mu), method='L-BFGS-B',

        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

                    return best_h, best_c
                j_start = max(0, i - radius)
                j_end = min(n, i + radius + 1)
                for j in range(j_start, j_end):
                    if i == j:
                        continue
                    # Try moving from i to j
                    if best_h[i] - d >= 0 and best_h[j] + d <= 1:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    h_start = project_feasible(h_init, n_init)
    c_start = compute_c5(h_start, n_init)
    
    # Decide working resolution
    # Start at n=500 equivalent, then go to n=1000
    n_work = 1000
    
    # Upsample to working resolution
    if n_init != n_work:
        old_x = np.linspace(0, 2, n_init, endpoint=False)
        new_x = np.linspace(0, 2, n_work, endpoint=False)
        h_work = np.interp(new_x, old_x, h_init)
    else:
        h_work = h_start.copy()
    h_work = project_feasible(h_work, n_work)
    c_work = compute_c5(h_work, n_work)
    
    best_h = h_work.copy()
    best_c = min(c_start, c_work)
    if c_start < c_work:
        best_h = h_start.copy()
    
    # Phase 1: Optimize at n=500 level (quick)
    if n_init <= 600 and time.time() < deadline - 400:
        h_coarse = project_feasible(h_init, n_init)
        for alpha in [500, 2000, 5000]:
            if time.time() > deadline - 400:
                break
            h_c, c_c = run_lbfgs(h_coarse, n_init, deadline - 400, alpha, None, 1.0, 1500)
            if c_c < c_start:
                c_start = c_c
                h_coarse = h_c
        # Upsample improved coarse solution
        old_x = np.linspace(0, 2, n_init, endpoint=False)
        new_x = np.linspace(0, 2, n_work, endpoint=False)
        h_up = np.interp(new_x, old_x, h_coarse)
        h_up = project_feasible(h_up, n_work)
        c_up = compute_c5(h_up, n_work)
        if c_up < best_c:
            best_h = h_up
            best_c = c_up
    
    # Phase 2: Active-set optimization at n=1000
    if time.time() < deadline - 60:
        t2 = min(time.time() + 450, deadline - 60)
        best_h, best_c = active_set_optimize(best_h, n_work, t2)
    
    # Phase 3: Targeted push on the max lag
    if time.time() < deadline - 30:
        t3 = min(time.time() + 60, deadline - 30)
        h_push, c_push = targeted_push(best_h, n_work, t3)
        if c_push < best_c:
            best_c = c_push
            best_h = h_push.copy()
    
    # Phase 4: SLSQP polish
    if time.time() < deadline - 25:
        t4 = min(time.time() + 40, deadline - 25)
        h_slq, c_slq = slqp_polish(best_h, n_work, t4, 500000, 1500)
        if c_slq < best_c:
            best_c = c_slq
            best_h = h_slq.copy()
    
    # Phase 5: Fine pair refinement
    if time.time() < deadline - 10:
        t5 = min(time.time() + 30, deadline - 10)
        h_pair, c_pair = pair_refine(best_h, n_work, t5, radius=15)
        if c_pair < best_c:
            best_c = c_pair
            best_h = h_pair.copy()
    
    # Phase 6: One more targeted push + SLSQP if time
    if time.time() < deadline - 8:
        h_push2, c_push2 = targeted_push(best_h, n_work, min(time.time() + 20, deadline - 8))
        if c_push2 < best_c:
            best_c = c_push2
            best_h = h_push2.copy()
        if time.time() < deadline - 5:
            h_slq2, c_slq2 = slqp_polish(best_h, n_work, deadline - 3, 2000000, 1000)
            if c_slq2 < best_c:
                best_c = c_slq2
                best_h = h_slq2.copy()
    
    # Final projection and verification
    best_h = project_feasible(best_h, n_work)
    best_c = compute_c5(best_h, n_work)
    
    # Also check at original resolution if it's different
    if n_init != n_work and n_init <= 600:
        old_x = np.linspace(0, 2, n_init, endpoint=False)
        h_orig_res = np.interp(old_x, np.linspace(0, 2, n_work, endpoint=False), best_h)
        h_orig_res = project_feasible(h_orig_res, n_init)
        c_orig = compute_c5(h_orig_res, n_init)
        if c_orig < best_c:
            return h_orig_res, c_orig, n_init
    
    return best_h, best_c, n_work
