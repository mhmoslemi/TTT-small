            h_new = slsqp_smooth(best_h, n, alpha, maxiter=3000, ftol=1e-15, top_idx=top_idx)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
    
    # Phase 4: Another round of outer approximation
    if time.time() < deadline - 20:
        o2_deadline = min(deadline - 15, time.time() + 40)
        h_oo2, c_oo2 = outer_approx_no_trust(best_h, n, max_iters=25, deadline=o2_deadline)
        if c_oo2 < best_c:
            best_c = c_oo2
            best_h = h_oo2
    
    # Phase 5: Pair swap local search
    if time.time() < deadline - 5:
        ls_deadline = min(deadline - 1, time.time() + 30)
        best_h, best_c = pair_swap_local_search(best_h, n, ls_deadline, lags_to_try=5, max_rounds=300)
    
    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 2:
        h_new = slsqp_smooth(best_h, n, 500000, maxiter=2000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def structured_init(n, rng):
    """Generate structured initial candidates."""
    candidates = []
    # Binary-like random
    h1 = (rng.random(n) > 0.5).astype(float)
    h1 = project_feasible(h1, n)
    candidates.append(h1)
    # Smooth cosine with random phase
    x = np.linspace(0, 2 * np.pi * 3, n)
    h2 = 0.5 + 0.3 * np.cos(x + rng.uniform(0, 2 * np.pi))
    h2 = project_feasible(h2, n)
    candidates.append(h2)
    # Anti-correlated random walk
    h3 = 0.5 + 0.15 * rng.randn(n)
    # Apply moving average to smooth
    kernel = np.ones(5) / 5
    h3 = np.convolve(h3, kernel, mode='same')
    h3 = project_feasible(h3, n)
    candidates.append(h3)
    # Step function: alternating blocks
    block = max(1, n // 20)
    h4 = np.zeros(n)
    val = 1.0
    for i in range(n):
        h4[i] = val
        if (i // block + 1) % 2 == 0:
            val = 1.0 - val
    h4 = project_feasible(h4, n)
    candidates.append(h4)
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Load parent if available
    h_init = None
    n_init = 300
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
    
    best_h = None
    best_c = float('inf')
    
    # If parent available, optimize it first
    if h_init is not None:
        alloc = min(300, (deadline - t0) * 0.4)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Try other resolutions
    n_candidates = []
    if h_init is not None:
        n_candidates = [200, 250, 350, 400, 500]
    else:
        n_candidates = [200, 300, 400, 500]
    
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        if alloc < 50:
            break
        
        # Interpolate from best or generate structured init
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
            h_start = project_feasible(h_start, n_try)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
            h_start = project_feasible(h_start, n_try)
        else:
            h_start = np.full(n_try, 0.5)
        
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Multi-start refinement on best resolution
    if best_h is not None and time.time() < deadline - 80:
        n_best = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            # Different perturbation strategies
            if r % 3 == 0:
                scale = 0.03
                perturb = rng.normal(0, scale, n_best)
            elif r % 3 == 1:
                # Swap blocks
                perturb = np.zeros(n_best)
                bs = max(5, n_best // 30)
                for _ in range(10):
                    a = rng.randint(0, n_best - bs)
                    b = rng.randint(0, n_best - bs)
                    delta = 0.05 * (1 if rng.random() > 0.5 else -1)
                    perturb[a:a+bs] += delta
                    perturb[b:b+bs] -= delta
            else:
                # Threshold-based perturbation
                threshold = 0.3 + 0.4 * rng.random()
                perturb = np.where(best_h > threshold, -0.02, 0.02)
            
            hp = project_feasible(best_h + perturb, n_best)
            alloc = min(65, (deadline - time.time() - 30) * 0.35)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final intensive refinement
    if best_h is not None and time.time() < deadline - 10:
        n_final = len(best_h)
        # Extra outer approx
        if time.time() < deadline - 12:
            h_oo, c_oo = outer_approx_no_trust(best_h, n_final, max_iters=15, deadline=deadline - 12)
            if c_oo < best_c:
                best_c = c_oo
                best_h = h_oo
        # Final local search with finer steps
        best_h, best_c = pair_swap_local_search(best_h, n_final, deadline - 2, lags_to_try=8, max_rounds=500)
        best_h = project_feasible(best_h, n_final)
        best_c = compute_c5(best_h, n_final)
    
    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)
    
    return best_h, best_c, len(best_h)
