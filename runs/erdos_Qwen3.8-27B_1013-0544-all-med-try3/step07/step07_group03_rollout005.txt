def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sa_optimize(h, n, t_start, t_end, n_iter, seed):
    """Simulated annealing on the exact C5 objective."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5, _ = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = c5
    T = t_start
    cool = (t_end / t_start) ** (1.0 / max(n_iter, 1))
    
    for it in range(n_iter):
        T *= cool
        # Generate a move
        n_move = rng.randint(2, 5)  # Move 2-4 coordinates
        idx = rng.choice(n, size=n_move, replace=False)
        delta = rng.uniform(-0.02, 0.02, size=n_move)
        h_new = h.copy()
        h_new[idx] += delta
        h_new = project_feasible(h_new, n)
        
        c5_new, _ = compute_c5(h_new, n)
        delta_e = c5_new - c5
        
        if delta_e < 0 or rng.random() < np.exp(-delta_e / max(T, 1e-12)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def pair_search_enhanced(h, n, deadline, deltas):
    """Enhanced pair search with 2 and 3 element moves."""
    best_c = compute_c5(h, n)[0]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            # 2-element moves
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    for sign in [1, -1]:
                        h_new = h.copy()
                        h_new[i] += sign * d
                        h_new[j] -= sign * d
                        if h_new[i] < -1e-12 or h_new[i] > 1+1e-12 or h_new[j] < -1e-12 or h_new[j] > 1+1e-12:

                        h_new = np.clip(h_new, 0, 1)
                        c = compute_c5(h_new, n)[0]
                        if c < best_c - 1e-14:
                            best_c = c
                            h = h_new
                            improved = True
            if not improved:

    return h, best_c

For ℓ < 0: ∂c(ℓ)/∂h[j] = (1-h[j+ℓ]) * 1[j ≥ -ℓ] - h[j-ℓ] * 1[j ≤ n-1+ℓ]

Great, now I can compute the subgradient efficiently.

Then update: h[j] -= lr * subgrad[j], project to [0,1] and sum=n/2.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_subgrad(h, n, eps=1e-6):
    """Compute subgradient of max_k c(k)*dx."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    # Active set: lags within eps of max
    active = np.where(c >= c_max - eps)[0]
    n_active = len(active)
    
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        # d c(lag) / d h[j] = (1-h[j+lag]) * 1[j+lag in [0,n-1]] - h[j-lag] * 1[j-lag in [0,n-1]]
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            # h[j+lag] valid for j in [0, n-1-lag]
            grad[:n-lag] += (1.0 - h[lag:n])
            # h[j-lag] valid for j in [lag, n-1]
            grad[lag:] -= h[:n-lag]
        else:
            l = -lag
            # h[j-l] valid for j in [l, n-1]
            grad[l:] += (1.0 - h[:n-l])
            # h[j+l] valid for j in [0, n-1-l]
            grad[:n-l] -= h[l:]
    grad *= dx / n_active
    return grad, c_max

def subgradient_descent(h, n, deadline, lr_init=0.05, lr_decay=0.999, max_iter=50000, eps_active=1e-4):
    """Subgradient descent with projection."""
    best_h = h.copy()
    best_c = compute_c5(h, n)[0]
    h = h.copy()
    lr = lr_init
    c5 = best_c
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 1000 == 0 and lr < 1e-6:
            break
        
        grad, c5 = compute_subgrad(h, n, eps_active)
        
        # Check for improvement
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        
        # Update
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check if we made progress
        c5_new, _ = compute_c5(h_new, n)
        if c5_new < c5 + 1e-12:
            h = h_new
            c5 = c5_new
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
        # Else: keep h, decrease lr more aggressively
        else:
            lr *= 0.99
        
        lr *= lr_decay
        lr = max(lr, 1e-8)
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    """Standard pair search with in-place modification."""
    best_c = compute_c5(h, n)[0]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)[0]
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)[0]
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def triple_search(h, n, deadline, d, radius=30):
    """3-element moves in a local neighborhood."""
    best_c = compute_c5(h, n)[0]
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(max(0, i-radius), min(n, i+radius)):
                if i == j:
                    continue
                for k in range(max(0, j-radius), min(n, j+radius)):
                    if k == i or k == j:
                        continue
                    # Try: +d at i, -d/2 at j, -d/2 at k
                    if h[i] + d <= 1.0 and h[j] - d/2 >= 0.0 and h[k] - d/2 >= 0.0:
                        h[i] += d; h[j] -= d/2; h[k] -= d/2
                        c = compute_c5(h, n)[0]
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d/2; h[k] += d/2
                    # Try: -d at i, +d/2 at j, +d/2 at k
                    if h[i] - d >= 0.0 and h[j] + d/2 <= 1.0 and h[k] + d/2 <= 1.0:
                        h[i] -= d; h[j] += d/2; h[k] += d/2
                        c = compute_c5(h, n)[0]
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d/2; h[k] -= d/2
    return h, best_c

def gradient_guided_search(h, n, deadline, max_iter=10000):
    """Use subgradient to guide pair moves."""
    best_c = compute_c5(h, n)[0]
    h = h.copy()
    c5 = best_c
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad, c5 = compute_subgrad(h, n, eps=1e-3)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        
        # Find best 2-coordinate move in gradient direction
        # Increase h where grad is most negative, decrease where most positive
        # But we need to preserve the sum, so: h[i] += d, h[j] -= d
        # Best: maximize d * (grad[j] - grad[i]) for grad[j] > grad[i]
        # i.e., pick i with min grad, j with max grad
        
        i_min = np.argmin(grad)
        j_max = np.argmax(grad)
        
        if grad[j_max] - grad[i_min] < 1e-12:
            break
        
        # Try a few step sizes
        for d in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = h.copy()
            h_new[i_min] += d
            h_new[j_max] -= d
            if h_new[i_min] > 1.0 or h_new[j_max] < 0.0:
                # Try reverse
                h_new = h.copy()
                h_new[i_min] -= d
                h_new[j_max] += d
                if h_new[i_min] < 0.0 or h_new[j_max] > 1.0:

            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)[0]
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                break
    
    return h, best_c
