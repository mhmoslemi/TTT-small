                            best_c = c
                            improved = True
                        else:
                            best_h[i] = old_i
                            best_h[j] = old_j
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

The overlap at lag 0 is c_0 = sum h[i](1-h[i]) * dx = (n/2 - sum h[i]²) * dx. To minimize this, we want sum h[i]² to be as large as possible, which means h should be as binary as possible (0s and 1s).

For a binary h with 125 ones and 125 zeros:
- c_0 = 0
- c_k = (number of 1s at positions where the position k away is a 0) × dx

  = (125 - R_k) × dx where R_k = number of 1s with a 1 at distance k

So minimizing max c_k is equivalent to making R_k as uniform as possible (close to 125 for all k), i.e., the autocorrelation of the binary sequence should be as flat as possible.

This is the well-known "minimum autocorrelation" problem for binary sequences! The optimal sequences are related to m-sequences, Barker sequences, and Golay sequences.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s = np.minimum(s, 0.0)
    s = np.maximum(s, -500.0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hx = project_feasible(res.x, n)
        if compute_c5(hx, n) < compute_c5(h, n):
            return hx
    except:

    return h
