    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    cm = c.max()
    a = alpha * (c - cm)
    logsum = np.log(np.sum(np.exp(a)))
    val = cm + logsum / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    cm = c.max()
    a = alpha * (c - cm)
    e = np.exp(a)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    vm = vals.max()
    a = alpha * (vals - vm)
    logsum = np.log(np.sum(np.exp(a)))
    return float(vm + logsum / alpha)

def active_grad(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    vm = vals.max()
    a = alpha * (vals - vm)
    e = np.exp(a)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-13:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    cm = c.max()
    a = alpha * (c - cm)
    logsum = np.log(np.sum(np.exp(a)))
    return float(cm + logsum / alpha)

def slqp_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    cm = c.max()
    a = alpha * (c - cm)
    e = np.exp(a)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

    return x.sum() - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, alphas, mu):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(soft_obj, h, jac=soft_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_active_set(h, n, deadline, top_k=4):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for _ in range(15):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-5, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            break
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])[:top_k]]
        for alpha in [2000, 20000, 200000]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(active_obj, h, jac=active_grad,
                              args=(n, active, alpha), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': 30000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def run_slqp(h, n, deadline, alpha=100000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, project_feasible(h, n),
                      jac=slqp_grad, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'mmax': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def fine_local_search(h, n, deadline, max_radius=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    d0 = h[i]
                    d1 = h[j]
                    if d0 + d <= 1.0 and d1 - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] = d0
                            h[j] = d1
                    if d0 - d >= 0.0 and d1 + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] = d0
                            h[j] = d1
    return h, best_c

def random_restart_opt(h, n, deadline, n_restarts=3, noise_std=0.005):
    rng = np.random.RandomState(123)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    time_per_restart = (deadline - time.time()) / max(1, n_restarts) - 5
    for r in range(n_restarts):
        if time.time() > deadline - 5:
            break
        h_pert = h + rng.randn(n) * noise_std
        h_pert = project_feasible(h_pert, n)
        t_sub = time.time() + max(5, time_per_restart)
        for alpha in [5000, 50000]:
            if time.time() > t_sub - 2:
                break
            try:
                res = minimize(soft_obj, h_pert, jac=soft_grad,
                              args=(n, alpha, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 20000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
            except Exception:
                pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def downsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)

    return project_feasible(h_new, n_new)
