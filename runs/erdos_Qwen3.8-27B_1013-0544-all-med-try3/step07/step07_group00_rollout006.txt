def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h *= (target / tot)
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def get_active_shifts(h, n, tol_frac=0.005, max_k=8):
    c = compute_c5_vec(h, n)
    mx = c.max()
    thresh = mx - max(1e-8, tol_frac * mx)
    active = np.where(c >= thresh)[0]
    if len(active) > max_k:
        order = np.argsort(-c[active])
        active = active[order[:max_k]]
    return active, mx

def smooth_max_obj(x, n, alpha, active):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        return float(mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(mx + np.log(np.sum(np.exp(sh))) / alpha)

    vals = c[active]
    mx = vals.max()

Actually, np.correlate for two arrays of length n is O(n²) in general. But there might be a way to update it incrementally when only 2 entries change.

If we change h[i] by δ and h[j] by -δ, how does the correlation change?

c_k = ∑_{x in overlap} h(x)*(1-h(x+k))

The change in c_k due to h[i] += δ:
- Terms where h[i] appears as the first factor: (1-h[i+k])*δ (if i+k is valid)
- Terms where h[i] appears as the second factor (1-h[i-k]): -h[i-k]*δ (if i-k is valid)

Similarly for h[j] -= δ:
- Terms where h[j] appears as first factor: -(1-h[j+k])*δ
- Terms where h[j] appears as second factor: +h[j-k]*δ

So the change in c_k is:
Δc_k = δ * [(1-h[i+k]) - h[i-k]] - δ * [(1-h[j+k]) - h[j-k]]
     = δ * [(1-h[i+k]) - h[i-k] - 1 + h[j+k] + h[j-k]]
     = δ * [h[j+k] + h[j-k] - h[i+k] - h[i-k]]

(with appropriate bounds checking)

This means we can compute the change in ALL c_k values in O(n) time (one pass over k), rather than O(n²).

Even better: we only need to check if the MAX changes. So we can:
1. Compute the new max from the old max and the changes
2. Only recompute the full correlation if the max might have changed

This is a huge speedup for local search!

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-14:
                break
            h *= (target / s)
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def grad_c_all(h, n, i):
    """Gradient of all c_k w.r.t. h[i], shape (2n-1,)."""
    out = np.zeros(2 * n - 1)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        val = 0.0
        if 0 <= i + k < n:
            val += (1.0 - h[i + k])
        if 0 <= i - k < n:
            val -= h[i - k]
        out[k_idx] = val
    return out

def smooth_max_obj(x, n, alpha, active):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        return float(mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(mx + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(x, n, alpha, active):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n

        wm = wi * dx
        if k >= 0:

        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def get_active(h, n, tol_frac=0.005, max_k=8):

    thresh = mx - max(1e-8, tol_frac * mx)
    active = np.where(c >= thresh)[0]
    if len(active) > max_k:
        order = np.argsort(-c[active])
        active = active[order[:max_k]]
    return active, mx

            if time.time() > deadline - 130:
                break
            h = lbfgs_on_active(h, n, alpha, maxiter=1000)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Phase 5: Fine local search (150s)
    if time.time() < deadline - 30:
        t5 = min(time.time() + 120, deadline - 30)
        h, best_c = fast_local_search_simple(h, n, t5)
        best_h = h.copy()
    
    # Phase 6: Very fine L-BFGS
    if time.time() < deadline - 10:
        h = lbfgs_on_active(h, n, 200000, maxiter=500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Try flipped solution
    h_flip = project_feasible(1.0 - best_h, n)
    c_flip = compute_c5(h_flip, n)
    if c_flip < best_c:
        best_c = c_flip
        best_h = h_flip
    
    # Final verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
