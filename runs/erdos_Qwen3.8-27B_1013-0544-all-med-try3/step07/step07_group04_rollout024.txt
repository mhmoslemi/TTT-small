The smooth approximation: f_smooth(h) = (1/α) * log(sum_k exp(α * q_k(h)))

The gradient: ∇f_smooth(h) = (1/sum_k exp(α*q_k)) * sum_k exp(α*q_k) * ∇q_k(h)

This is a weighted average of the gradients, with weights proportional to exp(α*q_k). For large α, this concentrates on the maximum lag(s).

        lag = k - (n - 1)
        grad = lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=50):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.005, trust * 0.85)
    return best_h, best_c

def local_search_fine(h, n, deadline, min_delta=1e-6):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()

    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        if time.time() > deadline:
            break
        if d < min_delta:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h_curr, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                m = 1
            else:
                m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h_curr[i] > h_curr[j] + 1e-10:
                    delta = min(d, (h_curr[i] - h_curr[j]) * 0.5, h_curr[i], 1.0 - h_curr[j])
                    if delta < 1e-10:
                        continue
                    h_curr[i] -= delta
                    h_curr[j] += delta
                    c_new = compute_c5(h_curr, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_curr.copy()
                        improved = True
                    else:
                        h_curr[i] += delta
                        h_curr[j] -= delta
                elif h_curr[j] > h_curr[i] + 1e-10:
                    delta = min(d, (h_curr[j] - h_curr[i]) * 0.5, h_curr[j], 1.0 - h_curr[i])
                    if delta < 1e-10:
                        continue
                    h_curr[i] += delta
                    h_curr[j] -= delta
                    c_new = compute_c5(h_curr, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_curr.copy()
                        improved = True
                    else:
                        h_curr[i] -= delta
                        h_curr[j] += delta
    return best_h, best_c

def local_search_multi(h, n, deadline, top_m=3):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002]

    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h_curr, n)
            top_idx = np.argsort(c)[-top_m:]
            for k_star in top_idx:
                if time.time() > deadline:
                    break
                lag = k_star - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h_curr[i] > h_curr[j] + 1e-10:
                        delta = min(d, (h_curr[i] - h_curr[j]) * 0.5, h_curr[i], 1.0 - h_curr[j])
                        if delta < 1e-10:
                            continue
                        h_curr[i] -= delta
                        h_curr[j] += delta
                        c_new = compute_c5(h_curr, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h_curr.copy()
                            improved = True
                        else:
                            h_curr[i] += delta
                            h_curr[j] -= delta
                    elif h_curr[j] > h_curr[i] + 1e-10:
                        delta = min(d, (h_curr[j] - h_curr[i]) * 0.5, h_curr[j], 1.0 - h_curr[i])
                        if delta < 1e-10:
                            continue
                        h_curr[i] += delta
                        h_curr[j] -= delta
                        c_new = compute_c5(h_curr, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h_curr.copy()
                            improved = True
                        else:
                            h_curr[i] -= delta
                            h_curr[j] += delta
    return best_h, best_c

def optimize_multistage(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Stage 1: Outer approximation (LP-based)
    if time.time() < deadline - 120:
        oo_end = time.time() + min(60, (deadline - time.time()) * 0.3)
        h_oo, c_oo = outer_approx(best_h, n, oo_end)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo

    # Stage 2: Projected gradient
    if time.time() < deadline - 60:
        pg_end = time.time() + min(80, (deadline - time.time()) * 0.4)
        h_pg, c_pg = projected_gradient(best_h, n, pg_end, alpha=100.0, top_k=20, max_iters=10000)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg

    # Stage 3: SLSQP with multiple alpha values
    for alpha in [5000, 50000, 200000]:
        if time.time() > deadline - 15:
            break
        h_new = slsqp_optimize(best_h, n, alpha, maxiter=2000, ftol=1e-15, top_k=30)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Stage 4: Local search (single lag)
    if time.time() < deadline - 20:
        ls_end = min(deadline - 5, time.time() + 40)
        h_new, c_new = local_search_fine(best_h, n, ls_end)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Stage 5: Multi-lag local search
    if time.time() < deadline - 10:
        ls_end = min(deadline - 3, time.time() + 30)
        h_new, c_new = local_search_multi(best_h, n, ls_end, top_m=3)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def sigmoid_structure(n, n_blocks, hard=True):
    x = np.linspace(0, 2, n)
    block_w = 2.0 / (2 * n_blocks)
    h = np.zeros(n)
    for i in range(n_blocks):
        start = i * 2 * block_w
        end = start + block_w
        mask = (x >= start) & (x <= end)
        h[mask] = 1.0
    if not hard:
        for i in range(n_blocks):
            start = i * 2 * block_w
            end = start + block_w
            for j in range(n):
                if abs(x[j] - start) < 0.05:
                    h[j] = 0.5 * (1 + np.tanh((x[j] - start) / 0.02))
                elif abs(x[j] - end) < 0.05:
                    h[j] = 0.5 * (1 - np.tanh((x[j] - end) / 0.02))
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize parent solution
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.28)
        h_opt, c_opt = optimize_multistage(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Phase 2: Try structured initializations at different resolutions
    structured_starts = []
    for n_try in [200, 300, 400, 500]:
        for nb in [2, 3, 4, 5, 6, 7, 8]:
            structured_starts.append((n_try, sigmoid_structure(n_try, nb)))

    # Phase 3: Try other resolutions from best
    n_candidates = [200, 250, 400, 500, 600]
    for n_try in n_candidates:
        if time.time() > deadline - 200:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 100
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.25)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_multistage(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Random restarts at best resolution
    if best_h is not None:
        n_best = len(best_h)
        restart_budget = deadline - time.time() - 80
        n_restarts = max(1, int(restart_budget / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 70:
                break
            scale = 0.008 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, deadline - time.time() - 40)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_multistage(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Try structured starts if time allows
    if best_h is not None and time.time() < deadline - 100:
        n_best = len(best_h)
        for nb in [3, 4, 5, 6]:
            if time.time() > deadline - 60:
                break
            h_struct = sigmoid_structure(n_best, nb)
            # Blend with best
            blend = 0.3 * h_struct + 0.7 * best_h
            blend = project_feasible(blend, n_best)
            alloc = min(40, deadline - time.time() - 30)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_multistage(blend, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 6: Final intensive polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Very fine SLSQP
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=3000, ftol=1e-16, top_k=50)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Final local search
        ls_end = min(deadline - 1, time.time() + 20)
        best_h, best_c = local_search_fine(best_h, n, ls_end, min_delta=5e-6)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        if h_init is not None:
            best_h = project_feasible(h_init, n_init)
        else:
            best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
