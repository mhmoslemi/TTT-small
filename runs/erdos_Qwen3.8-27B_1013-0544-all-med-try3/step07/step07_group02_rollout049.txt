    """Minimize smooth max of specific lags using SLSQP"""
    dx = 2.0 / n
    active_indices = [lag + (n-1) for lag in active_lags]

        c = compute_all_lags(x, n)
        c_sub = c[active_indices]
        return smooth_max(c_sub, alpha)

def lag_gradient(h, n, lag):

    if m == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    grad[0:n-m] += (1.0 - h[m:n])
    grad[m:n] += -h[0:n-m]
    return grad

def run_gradient_phase(h, n, deadline, rng):
    """Adam optimizer on smooth max of top K lags"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_lags = 2 * n - 1
    
    # Adam state
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    lr_base = 0.5
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    phase_configs = [
        (20, 100),   # top 20 lags, alpha=100
        (10, 200),   # top 10 lags, alpha=200
        (5, 500),    # top 5 lags, alpha=500
        (3, 1000),   # top 3 lags, alpha=1000
    ]
    
    total_iters_per_phase = 500
    
    for top_k, alpha in phase_configs:
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_k:]
        
        for it in range(total_iters_per_phase):
            if time.time() > deadline:
                break
            t += 1
            # Compute gradient
            c = compute_all_lags(h, n)
            c_sub = c[top_idx]
            m_val = c_sub.max()
            shifted = alpha * (c_sub - m_val)
            shifted = np.clip(shifted, -100, 500)
            w = np.exp(shifted)
            w = w / w.sum()
            
            grad = np.zeros(n)
            dx = 2.0 / n
            for i_k, k in enumerate(top_idx):
                if w[i_k] > 1e-6:
                    lag = k - (n - 1)
                    grad += w[i_k] * lag_gradient(h, n, lag)
            grad *= dx
            
            # Adam update
            m_t = beta1 * m_t + (1 - beta1) * grad
            v_t = beta2 * v_t + (1 - beta2) * grad**2
            m_hat = m_t / (1 - beta1**t)
            v_hat = v_t / (1 - beta2**t)
            
            lr = lr_base / np.sqrt(t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_feasible(h_new, n)
            
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
            
            # Update top index every 100 iters
            if it % 100 == 99:
                c = compute_all_lags(h, n)
                top_idx = np.argsort(c)[-top_k:]
    
    return best_h, best_c

        cx = compute_all_lags(x, n)
        cx_sub = cx[top_idx]
        m = cx_sub.max()
        shifted = alpha * (cx_sub - m)
        shifted = np.clip(shifted, -100, 500)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

            best_h = h_b

    # Phase 5: SLSQP polish on top 2-3 lags
    if time.time() < deadline - 30:
        for top_m in [2, 3]:
            if time.time() > deadline - 20:
                break
            h_s = slsqp_active_set(best_h, n, top_m, 10000, maxiter=3000)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 6: Try n=600
    if time.time() < deadline - 60:
        n600 = 600
        h_600 = upsample(best_h, n, n600)
        # Quick optimization at n=600
        h_600, c_600 = gradient_phase(h_600, n600, min(time.time() + 30, deadline - 30))
        if time.time() < deadline - 20:
            h_600, c_600 = fine_local_search(h_600, n600, min(time.time() + 20, deadline - 10), n_lags_focus=3)
        if c_600 < best_c:
            best_c = c_600
            best_h = h_600

    # Phase 7: Final ultra-fine local search
    if time.time() < deadline - 5:
        h_f, c_f = fine_local_search(best_h, len(best_h), deadline - 2, n_lags_focus=2)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
