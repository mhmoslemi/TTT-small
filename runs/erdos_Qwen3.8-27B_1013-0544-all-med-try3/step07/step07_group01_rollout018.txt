Actually, for a fixed h[0..i-1, i+1..n-1], the objective as a function of h[i] is:
C₅(t) = max_k c_k(t) where c_k(t) = (terms not involving h[i]) + (terms involving h[i] and t)

The terms involving h[i]: for each lag k, the contribution to c_k from position i is:
- If i+k is valid: t * (1-h[i+k]) * dx
- If i-k is valid: h[i-k] * (1-t) * dx

So c_k(t) = A_k + t * B_k where A_k and B_k don't depend on t.
c_k(t) is linear in t! So C₅(t) = max_k (A_k + t*B_k) is the max of linear functions, which is convex in t.

So the 1D subproblem is: minimize a convex piecewise-linear function of t over [0,1] (with the constraint that the sum stays at n/2, which means if we change h[i] from v to t, we need to compensate elsewhere... unless we just do a trial change and re-project).

Actually, if we fix the other n-1 values and vary only h[i], the constraint sum(h) = n/2 is violated unless we don't change the sum. So the 1D optimization should be: fix all h[j] for j≠i, vary h[i] = t, and also adjust one other value h[j*] to maintain the sum. Or, we can just do the coordinate change and re-project.

Simpler: just try different values of t for h[i], re-project, and check C₅. This is like a golden section search but with re-projection.

Even simpler: for each i, try a few values of h[i] (current, current±ε, 0, 1, n/2/(n-1) if others are zero) and pick the best.

import time

    corr = np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    """Smooth max of array c with parameter alpha"""
    cm = c.max()
    d = alpha * (c - cm)
    if d.max() > 500:
        return cm + (d.max() + np.log(np.sum(np.exp(d - d.max())))) / alpha
    return cm + np.log(np.sum(np.exp(d))) / alpha

def smooth_max_weights(c, alpha):
    """Weights for smooth max (softmax of alpha * (c - c.max()))"""
    cm = c.max()
    d = alpha * (c - cm)
    w = np.exp(d)
    w /= w.sum()
    return w

def grad_corr(h, n, w):
    """
    Gradient of weighted sum of correlations.
    w has 2n-1 elements, w[m] is the weight for lag k=m-(n-1).
    Returns gradient w.r.t. h (length n).
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-16:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n-k]
        else:
            kk = -k
            grad[:n-kk] += wm * (1.0 - h[kk:])
            grad[kk:] -= wm * h[:n-kk]
    return grad

def _smo(x, n, alpha):
    hp = np.clip(x, 0.0, 1.0)
    c = compute_corr(hp, n)
    return smooth_max_val(c, alpha)

def _smg(x, n, alpha):
    hp = np.clip(x, 0.0, 1.0)
    c = compute_corr(hp, n)
    w = smooth_max_weights(c, alpha)
    return grad_corr(hp, n, w)

def _pen_obj(x, n, alpha, mu):
    hp = np.clip(x, 0.0, 1.0)
    c = compute_corr(hp, n)
    val = smooth_max_val(c, alpha)
    val += mu * (hp.sum() - n / 2.0) ** 2
    return val

def _pen_grad(x, n, alpha, mu):
    hp = np.clip(x, 0.0, 1.0)
    c = compute_corr(hp, n)
    w = smooth_max_weights(c, alpha)
    g = grad_corr(hp, n, w)
    g += 2.0 * mu * (hp.sum() - n / 2.0)
    return g

    curr = project_feasible(h, n)

                                             'gtol': 1e-13, 'maxls': 30})

def active_set_opt(h, n, deadline, m_top=10, n_iters=500):
    """Focus on top-m correlations and minimize their sum"""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        # Get top m indices
        top_idx = np.argsort(c)[-m_top:]
        top_c = c[top_idx]
        
        # Compute subgradient: average gradient of top-m correlations
        w = np.zeros(2*n-1)
        w[top_idx] = 1.0 / m_top
        g = grad_corr(h, n, w)
        
        # Gradient descent step with backtracking
        lr = 0.01 / m_top
        improved = False
        for _ in range(30):

                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return best_h, best_c

def coord_descent(h, n, deadline):
    """Coordinate descent: for each i, try to optimize h[i]"""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    while time.time() < deadline:

                return best_h, best_c

                improved = True

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smooth_max_val(c, alpha):
    cm = c.max()
    d = alpha * (c - cm)
    if d.max() > 500:
        dm = d.max()
        return cm + (dm + np.log(np.sum(np.exp(d - dm)))) / alpha
    return cm + np.log(np.sum(np.exp(d))) / alpha


def smooth_max_weights(c, alpha):
    cm = c.max()
    d = alpha * (c - cm)
    w = np.exp(d)
    ws = w.sum()
    if ws < 1e-300:
        w[:] = 0.0
        w[np.argmax(c)] = 1.0
        return w
    w /= ws
    return w


def grad_corr(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-16:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            grad[:n - k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n - k]
        else:
            kk = -k
            grad[:n - kk] += wm * (1.0 - h[kk:])
            grad[kk:] -= wm * h[:n - kk]
    return grad


def _smo(x, n, alpha):
    hp = np.clip(x, 0.0, 1.0)
    c = compute_corr(hp, n)
    return smooth_max_val(c, alpha)


def _smg(x, n, alpha):
    hp = np.clip(x, 0.0, 1.0)
    c = compute_corr(hp, n)
    w = smooth_max_weights(c, alpha)
    return grad_corr(hp, n, w)


def _pen_obj(x, n, alpha, mu):
    hp = np.clip(x, 0.0, 1.0)
    c = compute_corr(hp, n)
    val = smooth_max_val(c, alpha)
    val += mu * (hp.sum() - n / 2.0) ** 2
    return val


def _pen_grad(x, n, alpha, mu):
    hp = np.clip(x, 0.0, 1.0)
    c = compute_corr(hp, n)
    w = smooth_max_weights(c, alpha)
    g = grad_corr(hp, n, w)
    g += 2.0 * mu * (hp.sum() - n / 2.0)
    return g


def lbfgs_round(h, n, alpha, deadline, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    curr = project_feasible(h, n)
    try:
        res = minimize(_smo, curr, jac=_smg, args=(n, alpha), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16,
                                               'gtol': 1e-13, 'maxls': 30})
        curr = project_feasible(res.x, n)
    except Exception:
        pass
    return curr


def lbfgs_round_pen(h, n, alpha, mu, deadline, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    curr = project_feasible(h, n)
    try:
        res = minimize(_pen_obj, curr, jac=_pen_grad, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxls': 30})
        curr = project_feasible(res.x, n)
    except Exception:
        pass
    return curr


def active_set_opt(h, n, deadline, m_top=5, max_rounds=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.02
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        top_idx = np.argsort(c)[-m_top:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / float(m_top)
        g = grad_corr(h, n, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        step = lr
        for _ in range(40):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.7
            m_top = min(m_top + 1, 20)
            if lr < 1e-10:
                break
    return best_h, best_c


def coord_descent(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    while time.time() < deadline:
        improved = False
        for d in steps:
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                old = h[i]
                for delta in [d, -d]:
                    t = old + delta
                    if t < 0.0 or t > 1.0:
                        continue
                    h[i] = t
                    hp = project_feasible(h, n)
                    cc = compute_c5(hp, n)
                    if cc < best_c - 1e-14:
                        h = hp
                        best_c = cc
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] = old
        if not improved:
            break
    return best_h, best_h_c(best_h, n) if False else (best_h, best_c)


def best_h_c(h, n):
    return compute_c5(h, n)


def pair_search_fast(h, n, deadline, radius=30):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c


def targeted_mass_transfer(h, n, deadline):
    """Targeted: find max correlation lag, move mass to reduce it."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(500):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        m_star = int(np.argmax(c))
        k_star = m_star - (n - 1)
        # Find position contributing most to c_{k_star}
        if k_star >= 0:
            contrib = h[:n - k_star] * (1.0 - h[k_star:])
        else:
            kk = -k_star
            contrib = h[kk:] * (1.0 - h[:n - kk])
        i_star = int(np.argmax(contrib))
        # Move mass from i_star toward i_star + k_star (or nearby)
        target_pos = i_star + k_star
        if target_pos < 0:
            target_pos = i_star - k_star
        if target_pos >= n:
            target_pos = n - 1 - i_star
        d = min(0.01, h[i_star] * 0.1)
        if d < 1e-6:
            break
        h[i_star] -= d
        h[target_pos] += d
        hp = project_feasible(h, n)
        cc = compute_c5(hp, n)
        if cc < best_c - 1e-14:
            h = hp
            best_c = cc
            best_h = h.copy()
        else:
            h[i_star] += d
            h[target_pos] -= d
    return best_h, best_c


def random_restart_opt(h, n, deadline, n_trials=3, perturb_scale=0.05):
    rng = np.random.RandomState(12345)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_pert = h + rng.randn(n) * perturb_scale
        h_pert = project_feasible(h_pert, n)
        t_sub = min(time.time() + 60, deadline)
        for alpha in [2000, 20000, 200000]:
            if time.time() > t_sub:
                break
            h_pert = lbfgs_round(h_pert, n, alpha, t_sub, maxiter=100)
        c_p = compute_c5(h_pert, n)
        if c_p < best_c:
            best_c = c_p
            best_h = h_pert.copy()
    return best_h, best_c
