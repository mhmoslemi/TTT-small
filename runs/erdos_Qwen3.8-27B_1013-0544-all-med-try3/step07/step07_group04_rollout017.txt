import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target_sum - h.sum()
    if abs(diff) > 0:
        # Distribute among non-boundary elements
        interior = (h > 1e-10) & (h < 1 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0, 1)
    return h

def lag_gradient(h, n, lag):
    """Gradient of overlap at given lag with respect to h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m >= n:
        grad[:] = 0.0
    else:
        if lag > 0:
            # overlap = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
            # d/dh[j]: if j >= m and j+m < n: -h[j-m] (from h[j]*(1-h[j-m]))
            #          if j < n-m: (1-h[j+m]) (from h[j+m]*(1-h[j]))
            grad[m:] -= h[:n-m]  # from h[j] term where j = i+m
            grad[:n-m] += (1.0 - h[m:])  # from (1-h[i]) term
        else:
            # lag < 0, m = -lag
            # overlap = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            grad[:n-m] -= h[m:]  # from h[i] term
            grad[m:] += (1.0 - h[:n-m])  # from (1-h[i+m]) term
    return grad

def lp_active_step(h_curr, n, active_indices, trust):
    """Solve LP: min t s.t. linearized active lags <= t, box, sum, trust."""
    c_all = compute_all_lags(h_curr, n)
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    cons = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
    ]
    
    if trust < 1.0:
        cons.append(h_var - h_curr <= trust)
        cons.append(h_curr - h_var <= trust)
    
    for k in active_indices:
        lag = int(k - (n - 1))
        grad = lag_gradient(h_curr, n, lag)
        linearized = c_all[k] + dx * (grad @ (h_var - h_curr))
        cons.append(linearized <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                return h_curr.copy()
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    
    return project_feasible(h_new, n)

def slp_optimize(h, n, deadline, max_iters=2000):
    """Sequential Linear Programming optimization."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    trust = 0.3
    top_k = 4
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_all = compute_all_lags(h_curr, n)
        active = np.argsort(c_all)[-top_k:]
        
        # Also include any lag that's close to the max
        threshold = c_all.max() - 1e-4
        near_active = np.where(c_all >= threshold)[0]
        active = np.unique(np.concatenate([active, near_active]))
        
        h_new = lp_active_step(h_curr, n, active, trust)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        
        step = np.max(np.abs(h_new - h_curr))
        h_curr = h_new
        
        if step < 1e-10:
            break
        if step < 1e-8:
            trust = max(0.001, trust * 0.7)
        else:
            trust = max(0.001, trust * 0.85)
        
        # Adapt top_k
        if it % 20 == 0:
            top_k = min(10, 3 + it // 20)
    
    return best_h, best_c

def smooth_max_obj(x, n, top_idx, alpha):
    """Smooth max of top_idx lag overlaps."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, deadline, top_m=3, alpha=50000):
    """SLSQP polishing with few active constraints."""
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    
    try:
        res = minimize(smooth_max_obj, h, args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 5000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 0:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)

        grad[:] = 1.0 - 2.0 * h

        pass
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def lp_active_step(h_curr, n, active_indices, trust):
    c_all = compute_all_lags(h_curr, n)
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    if trust < 1.0:
        cons.append(h_var - h_curr <= trust)
        cons.append(h_curr - h_var <= trust)
    for k in active_indices:
        lag = int(k - (n - 1))
        grad = lag_gradient(h_curr, n, lag)
        linearized = c_all[k] + dx * (grad @ (h_var - h_curr))
        cons.append(linearized <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

        active = np.unique(np.concatenate([active, near_active]))
        h_new = lp_active_step(h_curr, n, active, trust)

    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def make_binary_start(n, n_blocks, rng):
    h = np.zeros(n)
    ones = n // 2
    block_size = n // n_blocks
    pattern = []
    remaining_ones = ones
    for b in range(n_blocks):
        bs = block_size if b < n_blocks - 1 else (n - block_size * (n_blocks - 1))
        if rng.random() < 0.5:
            fill = min(bs, remaining_ones)
            pattern.extend([1.0] * fill + [0.0] * (bs - fill))
            remaining_ones -= fill
        else:
            fill = min(bs, remaining_ones)
            pattern.extend([0.0] * (bs - fill) + [1.0] * fill)
            remaining_ones -= fill
    if remaining_ones > 0:
        idx = np.where(np.array(pattern) == 0.0)[0]
        if len(idx) > 0:
            pattern[np.random.choice(idx, remaining_ones, replace=False)] = 1.0
    h = np.array(pattern[:n])
    return project_feasible(h, n)
