import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def logsumexp_max(c, alpha):
    """Smooth approximation to max(c) using log-sum-exp."""
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_objective(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = logsumexp_max(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_gradient(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Only compute for significant weights
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

        k = int(idx - (n - 1))
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alpha, mu, maxiter, label=""):

                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxiter * 10, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        h_new = np.clip(res.x, 0, 1)
        c_raw = compute_c5(h_new, n)
        if c_raw < c_new:
            h_new, c_new = h_new, c_raw
        return h_new, c_new
    except Exception as e:

        return h0, compute_c5(h0, n)

def run_active_set(h, n, deadline, alpha, mu, maxiter, label=""):

def run_slqp(h, n, deadline, alpha, maxiter):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = compute_c5_vec(x, n)
        m = np.max(c)
        sh = alpha * (c - m)
        smx = np.max(sh)
        if smx > 500:
            return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def cons_eq(x):
        return x.sum() - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = np.clip(res.x, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h0, compute_c5(h0, n)

def pair_descent(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            # Focus on coordinates that affect the max shifts
            active_shifts = np.where(c_vec >= max_val * 0.999)[0]
            
            # For each active shift, find the coordinates that matter most
            # and try to reduce the max
            for _ in range(100):  # max iterations per step size
                if time.time() > deadline:
                    return best_h, best_c
                if not improved:
                    break
                    
                # Try random pairs
                for _attempt in range(50):
                    if time.time() > deadline:
                        return best_h, best_c
                    i = int(np.random.randint(n))
                    j = int(np.random.randint(n))
                    if i == j:
                        continue
                    
                    for direction in [1, -1]:
                        if direction == 1:
                            if best_h[i] < step and best_h[j] > 1 - step:
                                continue
                        else:
                            if best_h[i] > 1 - step and best_h[j] < step:
                                continue
                        
                        best_h[i] += direction * step
                        best_h[j] -= direction * step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            break
                        else:
                            best_h[i] -= direction * step
                            best_h[j] += direction * step
                
                if improved:
                    best_h = project_feasible(best_h, n)
                    best_c = compute_c5(best_h, n)
    
    return best_h, best_c

def coordinate_descent(h, n, deadline):
    """Coordinate descent focusing on the most impactful coordinates."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(42)
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        # Find the shift that achieves max
        max_shifts = np.where(c_vec >= max_val - 1e-10)[0]
        
        # Compute gradient at the max shift
        grad = np.zeros(n)
        for idx in max_shifts[:10]:
            k = int(idx - (n - 1))
            dx = 2.0 / n
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[:nv] += (1.0 - best_h[k:]) * dx
                    grad[k:] += (-best_h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:] += (1.0 - best_h[:nv]) * dx
                    grad[:nv] += (-best_h[kk:]) * dx
        
        # Try moving in negative gradient direction for top coordinates
        top_coords = np.argsort(-np.abs(grad))[:20]
        found_improvement = False
        
        for i in top_coords:
            if time.time() > deadline:
                break
            for step in [0.05, 0.02, 0.01, 0.005, 0.001]:
                for direction in [-1, 1]:
                    new_val = best_h[i] + direction * step
                    if 0 <= new_val <= 1:
                        # Need to maintain sum constraint
                        # Adjust another coordinate
                        j = int(rng.randint(n))
                        if j == i:
                            j = (j + 1) % n
                        adj = direction * (-step)
                        if 0 <= best_h[j] + adj <= 1:
                            best_h[i] = new_val
                            best_h[j] += adj
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-16:
                                best_c = c
                                found_improvement = True
                                break
                            else:
                                best_h[i] = new_val - direction * step
                                best_h[j] -= adj
                if found_improvement:
                    break
            if found_improvement:
                break
        
        if not found_improvement:
            # Try a random perturbation to escape
            for _ in range(10):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                step = rng.uniform(-0.01, 0.01)
                if 0 <= best_h[i] + step <= 1 and 0 <= best_h[j] - step <= 1:
                    old_i, old_j = best_h[i], best_h[j]
                    best_h[i] += step
                    best_h[j] -= step
                    c = compute_c5(best_h, n)
                    if c < best_c:
                        best_c = c
                        found_improvement = True
                        break
                    else:
                        best_h[i] = old_i
                        best_h[j] = old_j
    
    return best_h, best_c

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_val(c, alpha):
    m = np.max(c)
    sh = alpha * (c - m)
    smx = np.max(sh)
    if smx > 500:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return m, w

                    continue
                for d in [1, -1]:
                    ni = best_h[i] + d * step
                    nj = best_h[j] - d * step
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        best_h[i] = ni
                        best_h[j] = nj
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            progress = True
                            break
                        else:
                            best_h[i] -= d * step
                            best_h[j] += d * step
    return best_h, best_c

def random_perturb(h, n, rng, scale=0.05):
    h_p = h.copy()
    h_p += rng.normal(0, scale, size=n)
    h_p = project_feasible(h_p, n)
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = np.full(n_init, 0.5)
    
    n = 500
    x_old = np.linspace(0, 2, n_init)
    x_new = np.linspace(0, 2, n)
    h_work = np.interp(x_new, x_old, h_init)
    h_work = project_feasible(h_work, n)
    
    best_h = h_work.copy()
    best_c = compute_c5(best_h, n)
    print(f"Start: n={n}, C5={best_c:.8f}")
    
    # Phase 1: Progressive L-BFGS (moderate alpha)
    t1_end = t0 + 150
    configs1 = [(100, 20.0, 1500), (500, 100.0, 1500), (2000, 500.0, 2000),
                (5000, 2000.0, 2000), (10000, 5000.0, 2000)]
    for alpha, mu, iters in configs1:
        if time.time() > t1_end - 10:
            break
        t_sub = min(time.time() + 50, t1_end - 5)
        h_r, c_r = run_lbfgs(best_h, n, t_sub, alpha, mu, iters)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    print(f"Phase 1: C5={best_c:.8f}")
    
    # Phase 2: Subset (active-set) optimization with high alpha
    t2_end = t0 + 400
    configs2 = [(20000, 5000.0, 2000), (50000, 10000.0, 2000),
                (100000, 20000.0, 2500), (300000, 50000.0, 2500)]
    for alpha, mu, iters in configs2:
        if time.time() > t2_end - 10:
            break
        t_sub = min(time.time() + 60, t2_end - 5)
        h_r, c_r = run_subset_lbfgs(best_h, n, t_sub, alpha, mu, iters)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    print(f"Phase 2: C5={best_c:.8f}")
    
    # Phase 3: Another round of full L-BFGS with very high alpha
    t3_end = t0 + 550
    configs3 = [(50000, 10000.0, 2000), (200000, 50000.0, 2000),
                (500000, 100000.0, 2000)]
    for alpha, mu, iters in configs3:
        if time.time() > t3_end - 10:
            break
        t_sub = min(time.time() + 50, t3_end - 5)
        h_r, c_r = run_lbfgs(best_h, n, t_sub, alpha, mu, iters)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    print(f"Phase 3: C5={best_c:.8f}")
    
    # Phase 4: SLSQP with equality constraint
    t4_end = t0 + 680
    for alpha in [50000, 200000, 500000]:
        if time.time() > t4_end - 10:
            break
        t_sub = min(time.time() + 45, t4_end - 5)
        h_r, c_r = run_slqp(best_h, n, t_sub, alpha, 2000)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    print(f"Phase 4: C5={best_c:.8f}")
    
    # Phase 5: Fine pair descent
    t5_end = t0 + 830
    h_r, c_r = fine_tune(best_h, n, t5_end, rng)
    if c_r < best_c:
        best_c = c_r
        best_h = h_r
    print(f"Phase 5: C5={best_c:.8f}")
    
    # Phase 6: Perturbation + re-optimization (escape local opt)
    if time.time() < deadline - 200:
        t6_end = min(time.time() + 150, deadline - 200)
        for pert_scale in [0.1, 0.05, 0.02]:
            if time.time() > t6_end - 30:
                break
            h_p = random_perturb(best_h, n, rng, pert_scale)
            c_p = compute_c5(h_p, n)
            # Quick optimization from perturbed point
            t_sub = min(time.time() + 40, t6_end - 20)
            h_r, c_r = run_lbfgs(h_p, n, t_sub, 50000, 10000.0, 1500)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
                print(f"  Perturbation {pert_scale} improved! C5={c_r:.8f}")
    print(f"Phase 6: C5={best_c:.8f}")
    
    # Phase 7: Subset optimization again (refined active set)
    if time.time() < deadline - 100:
        t7_end = min(time.time() + 60, deadline - 100)
        for alpha in [100000, 500000, 1000000]:
            if time.time() > t7_end - 10:
                break
            t_sub = min(time.time() + 25, t7_end - 5)
            h_r, c_r = run_subset_lbfgs(best_h, n, t_sub, alpha, 20000.0, 1500)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    print(f"Phase 7: C5={best_c:.8f}")
    
    # Phase 8: Final fine tune
    if time.time() < deadline - 30:
        t8_end = min(time.time() + 25, deadline - 10)
        h_r, c_r = fine_tune(best_h, n, t8_end, rng)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    print(f"Phase 8: C5={best_c:.8f}")
    
    # Phase 9: Try upsample to n=1000 with very brief optimization
    if time.time() < deadline - 40:
        n_hi = 1000
        x_lo = np.linspace(0, 2, n)
        x_hi = np.linspace(0, 2, n_hi)
        h_hi = np.interp(x_hi, x_lo, best_h)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        if c_hi < best_c:
            # Very brief optimization at high res
            t_sub = min(time.time() + 20, deadline - 10)
            h_hi, c_hi = run_lbfgs(h_hi, n_hi, t_sub, 50000, 10000.0, 1000)
            if c_hi < best_c:
                best_c = c_hi
                # Downsample back to n=500 for consistent return
                h_lo = np.interp(x_lo, x_hi, h_hi)
                h_lo = project_feasible(h_lo, n)
                c_lo = compute_c5(h_lo, n)
                if c_lo < best_c:
                    best_c = c_lo
                    best_h = h_lo
                else:
                    best_h = h_lo
                    best_c = c_lo
                print(f"Phase 9 (n=1000 helped): C5={best_c:.8f}")
    
    # Final cleanup
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    print(f"FINAL: n={n}, C5={best_c:.8f}")
    
    return best_h, best_c, n
