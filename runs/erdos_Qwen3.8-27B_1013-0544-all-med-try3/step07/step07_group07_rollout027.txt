            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def make_delta(h, n, i, j, m_range, corr_len):
    delta = np.zeros(corr_len)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < corr_len:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < corr_len:
        delta[k2] += 1.0
    return delta

def local_search_binary(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    corr_len = 2 * n - 1
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_arr):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            best_d = None
            for j in ones_arr:
                d = make_delta(h, n, i, j, m_range, corr_len)
                nc = (corr + d).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
                    best_d = d
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                corr += best_d
                cur_c = best_nc
                ii = np.where(zeros_arr == i)[0]
                if len(ii) > 0:
                    zeros_arr[ii[0]] = best_j
                ij = np.where(ones_arr == best_j)[0]
                if len(ij) > 0:
                    ones_arr[ij[0]] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def anneal_binary(h, n, deadline, seed=42, T0=0.03, T1=0.0005, beta=800.0):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr_len = 2 * n - 1
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    T = T0
    cool = 0.999995
    while time.time() < deadline:
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        d = make_delta(h, n, i, j, m_range, corr_len)
        nc = (corr + d).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(-beta * (nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    corr_len = 2 * n - 1
    for k_idx in range(corr_len):
        pk = p[k_idx]
        if pk < 1e-16:
            continue
        k = k_idx - (n - 1)
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

def pgd_solve(h, n, alphas, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        cur_c = best_c
        base_step = 0.3
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            grad = lse_grad(h, n, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            direction = -grad / gnorm
            accepted = False
            for ls in range(35):
                step = base_step * (0.5 ** ls)
                h_new = project(h + step * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    h = h_new
                    cur_c = c_new
                    base_step = step * 1.5
                    accepted = True
                    break
            if accepted:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def perturb_binary(h, n, rng, frac=0.05):
    h = h.copy()
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    n_flip = max(1, int(frac * n // 2))
    for _ in range(n_flip):
        if len(ones_idx) > 0 and len(zeros_idx) > 0:
            oi = ones_idx[rng.randint(len(ones_idx))]
            zi = zeros_idx[rng.randint(len(zeros_idx))]
            h[oi], h[zi] = 0.0, 1.0
            ones_idx = np.delete(ones_idx, np.where(ones_idx == oi)[0][0])
            zeros_idx = np.delete(zeros_idx, np.where(zeros_idx == zi)[0][0])
            ones_idx = np.append(ones_idx, zi)
            zeros_idx = np.append(zeros_idx, oi)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    
    n = n0
    h0 = project(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Quick binary local search (10% time)
    p1 = t0 + 0.10 * total
    hb = to_binary(best_h, n)
    hb, cb = local_search_binary(hb, n, p1, max_sweeps=100)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 2: Multi-start annealing on binary (30% time)
    p2 = t0 + 0.40 * total
    n_starts = 6
    per_start = p2 - time.time()
    for s in range(n_starts):
        start_dl = t0 + (0.10 + 0.30 * (s + 1) / n_starts) * total
        if time.time() > start_dl - 2:
            continue
        if s == 0:
            ha = best_h.copy()
        else:
            ha = perturb_binary(best_h, n, rng, frac=0.03 + 0.02 * s)
        ha, ca = anneal_binary(ha, n, start_dl, seed=seed + s * 1000, T0=0.04 - 0.003 * s, T1=0.0003, beta=600.0)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()
        # Quick local search polish
        if time.time() < start_dl - 5:
            hl, cl = local_search_binary(best_h, n, start_dl - 2, max_sweeps=30)
            if cl < best_c:
                best_c = cl
                best_h = hl.copy()
    
    # Phase 3: PGD continuous optimization (20% time)
    p3 = t0 + 0.60 * total
    hc = best_h.copy()
    if np.all((hc <= 1e-10) | (hc >= 1 - 1e-10)):
        hc[rng.choice(n, size=n // 10, replace=False)] = 0.5
    hc = project(hc, n)
    alphas3 = [100, 500, 2000, 8000, 30000, 100000, 300000]
    hc, cc = pgd_solve(hc, n, alphas3, p3, max_iter=3000)
    if cc < best_c:
        best_c = cc
        best_h = hc.copy()
    
    # Phase 4: Binary round + local search + PGD cycle (15% time)
    p4 = t0 + 0.75 * total
    for cycle in range(3):
        cycle_dl = t0 + (0.60 + 0.15 * (cycle + 1) / 3.0) * total
        if time.time() > cycle_dl - 10:
            break
        hb2 = to_binary(best_h, n)
        hb2, cb2 = local_search_binary(hb2, n, cycle_dl - 20, max_sweeps=50)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
        ha2, ca2 = anneal_binary(best_h, n, cycle_dl - 10, seed=seed + 5000 + cycle, T0=0.02, beta=500.0)
        if ca2 < best_c:
            best_c = ca2
            best_h = ha2.copy()
        hc2 = project(best_h + rng.random(n) * 0.1 - 0.05, n)
        alphas4 = [500, 3000, 15000, 100000]
        hc2, cc2 = pgd_solve(hc2, n, alphas4, cycle_dl - 2, max_iter=2000)
        if cc2 < best_c:
            best_c = cc2
            best_h = hc2.copy()
    
    # Phase 5: Higher resolution (25% time)
    p5 = dl - 3
    for nh in [500, 640, 800]:
        if time.time() > p5 - 40:
            break
        remaining = p5 - time.time()
        if remaining < 30:
            break
        # Interpolate from current best
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project(hh, nh)
        # Quick local search at new resolution
        hb3 = to_binary(hh, nh)
        loc_dl = time.time() + min(remaining * 0.25, 60)
        hb3, cb3 = local_search_binary(hb3, nh, loc_dl, max_sweeps=40)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()
            n = nh
        # PGD at new resolution
        hc3 = project(hb3 + rng.random(nh) * 0.05, nh)
        pgd_dl = time.time() + min(remaining * 0.45, 90)
        alphas5 = [100, 500, 3000, 20000, 100000, 500000]
        hc3, cc3 = pgd_solve(hc3, nh, alphas5, pgd_dl, max_iter=2000)
        if cc3 < best_c:
            best_c = cc3
            best_h = hc3.copy()
            n = nh
        # Short annealing at new resolution
        if time.time() < p5 - 15:
            hb4 = to_binary(best_h, n)
            an_dl = time.time() + min(remaining * 0.2, 40)
            hb4, cb4 = anneal_binary(hb4, n, an_dl, seed=seed + 9000, T0=0.02, beta=400.0)
            if cb4 < best_c:
                best_c = cb4
                best_h = hb4.copy()
    
    # Final: one more PGD polish with the best found
    if time.time() < dl - 5:
        hc_final = project(best_h + rng.random(n) * 0.02, n)
        alphas_f = [2000, 20000, 200000, 2000000]
        hc_final, cc_final = pgd_solve(hc_final, n, alphas_f, dl - 2, max_iter=3000)
        if cc_final < best_c:
            best_c = cc_final
            best_h = hc_final.copy()
    
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
