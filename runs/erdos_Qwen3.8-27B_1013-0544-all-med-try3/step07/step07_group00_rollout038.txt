import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def compute_coeff_matrix(h, n):
    dx = 2.0 / n
    coeff = np.zeros((n, 2 * n - 1))
    for k in range(n):
        m = k + (n - 1)
        nv = n - k
        coeff[0:nv, m] += (1.0 - h[k:n])
        coeff[k:n, m] += (-h[0:nv])
    for kk in range(1, n):
        m = (n - 1) - kk
        nv = n - kk
        coeff[kk:n, m] += (1.0 - h[0:nv])
        coeff[0:nv, m] += (-h[kk:n])
    coeff *= dx
    return coeff

def lbfgs_phase(h, n, deadline, configs):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def top_k_optimize(h, n, deadline, top_k=5):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def tk_obj(x, top_idx, alpha, mu):
        c = compute_c5_vec(x, n)
        vals = c[top_idx]
        mx = vals.max()
        sh = alpha * (vals - mx)
        smx = sh.max()
        if smx > 500:
            val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
        else:
            val = mx + np.log(np.sum(np.exp(sh))) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    
    def tk_grad(x, top_idx, alpha, mu):
        c = compute_c5_vec(x, n)
        vals = c[top_idx]
        mx = vals.max()
        sh = alpha * (vals - mx)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for wi, mm in zip(w, top_idx):
            k = mm - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
        grad += 2.0 * mu * (x.sum() - n / 2.0)
        return grad
    
    for iteration in range(20):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])][:top_k]
        if len(active) == 0:
            break
        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(tk_obj, h, jac=tk_grad,
                              args=(active, alpha, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 20000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def exact_pair_polish(h, n, deadline, n_candidates=60, n_grid=40):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    m0 = n - 1
    coeff = compute_coeff_matrix(best_h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    no_improve_count = 0
    
    for iteration in range(50000):
        if time.time() > deadline - 5:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        threshold = max_val - max(1e-8, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            break
        
        alpha = 10000
        a_vals = c_vec[active]
        sh = alpha * (a_vals - a_vals.max())
        w = np.exp(sh)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, mm in zip(w, active):
            k = mm - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - best_h[k:n])
                grad[k:n] += wm * (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - best_h[0:nv])
                grad[0:nv] += wm * (-best_h[kk:n])
        
        g_sorted = np.argsort(-np.abs(grad))[:n_candidates]
        
        best_improvement = 1e-14
        best_move = None
        best_f_min = max_val
        
        for ii in range(len(g_sorted)):
            i = g_sorted[ii]
            for jj in range(len(g_sorted)):
                if ii == jj:
                    continue
                j = g_sorted[jj]
                
                delta_max = min(1.0 - best_h[i], best_h[j])
                if delta_max < 1e-12:
                    continue
                slope = coeff[i] - coeff[j]
                deltas = np.linspace(0, delta_max, n_grid)
                c_new = c_vec[None, :] + deltas[:, None] * slope[None, :]
                c_new[:, m0] -= 2.0 * deltas**2 * dx
                f_vals = c_new.max(axis=1)
                bi = f_vals.argmin()
                if f_vals[bi] < best_f_min - 1e-15:
                    best_f_min = f_vals[bi]
                    best_improvement = max_val - best_f_min
                    best_move = (i, j, 1, deltas[bi])
                
                delta_max2 = min(best_h[i], 1.0 - best_h[j])
                if delta_max2 < 1e-12:
                    continue
                slope2 = coeff[j] - coeff[i]
                deltas2 = np.linspace(0, delta_max2, n_grid)
                c_new2 = c_vec[None, :] + deltas2[:, None] * slope2[None, :]
                c_new2[:, m0] -= 2.0 * deltas2**2 * dx
                f_vals2 = c_new2.max(axis=1)
                bi2 = f_vals2.argmin()
                if f_vals2[bi2] < best_f_min - 1e-15:
                    best_f_min = f_vals2[bi2]
                    best_improvement = max_val - best_f_min
                    best_move = (i, j, -1, deltas2[bi2])
        
        if best_move is None:
            no_improve_count += 1
            if no_improve_count > 10:
                break
            continue
        
        no_improve_count = 0
        i, j, direction, delta = best_move
        if direction == 1:
            best_h[i] += delta
            best_h[j] -= delta
        else:
            best_h[i] -= delta
            best_h[j] += delta
        
        coeff = compute_coeff_matrix(best_h, n)
        c_vec = compute_c5_vec(best_h, n)
        
        # Golden section refinement
        for _refine in range(15):
            if time.time() > deadline - 3:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            
            if direction == 1:
                d_lo = 0.0
                d_hi = min(1.0 - best_h[i], best_h[j])
            else:
                d_lo = 0.0
                d_hi = min(best_h[i], 1.0 - best_h[j])
            
            if d_hi < 1e-12:
                break
            
            gr = (np.sqrt(5) - 1) / 2
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            
            best_h[i] += direction * d1
            best_h[j] -= direction * d1
            c1 = compute_c5(best_h, n)
            
            best_h[i] += direction * (d2 - d1)
            best_h[j] -= direction * (d2 - d1)
            c2 = compute_c5(best_h, n)
            
            if c1 < c2:
                best_h[i] -= direction * (d2 - d1)
                best_h[j] += direction * (d2 - d1)
                d_hi = d2
            else:
                d_lo = d1
        
        best_h = clip_feasible_fast(best_h, n)
        c_final = compute_c5(best_h, n)
        if c_final < best_c:
            best_c = c_final
        else:
            best_h = project_feasible(h, n)
            coeff = compute_coeff_matrix(best_h, n)
    
    return best_h, best_c

def clip_feasible_fast(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-15:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-15:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def perturbation_search(h, n, deadline, n_perturb=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(123)
    
    for i in range(n_perturb):
        if time.time() > deadline - 10:
            break
        h_perturbed = best_h.copy()
        n_changes = rng.integers(3, 10)
        indices = rng.choice(n, size=n_changes, replace=False)
        perturbations = rng.uniform(-0.02, 0.02, size=n_changes)
        for idx, pert in zip(indices, perturbations):
            h_perturbed[idx] = np.clip(h_perturbed[idx] + pert, 0.0, 1.0)
        h_perturbed = project_feasible(h_perturbed, n)
        
        # Quick L-BFGS on perturbed
        t_sub = min(30, (deadline - time.time()) / (n_perturb - i))
        if t_sub < 5:
            continue
        h_opt, c_opt = lbfgs_phase(h_perturbed, n, time.time() + t_sub,
                                     configs=[(5000, 1000.0), (50000, 50000.0)])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS with varying alpha
    t_phase1 = min(t0 + 200, deadline - 600)
    best_h, best_c = lbfgs_phase(best_h, n, t_phase1,
        configs=[(500, 50.0), (2000, 500.0), (5000, 5000.0),
                 (10000, 50000.0), (50000, 500000.0)])
    
    # Phase 2: Top-k optimization
    if time.time() < deadline - 450:
        t_phase2 = min(time.time() + 120, deadline - 450)
        best_h, best_c = top_k_optimize(best_h, n, t_phase2, top_k=5)
    
    # Phase 3: Exact pair polishing (main engine)
    if time.time() < deadline - 200:
        t_phase3 = min(time.time() + 400, deadline - 200)
        best_h, best_c = exact_pair_polish(best_h, n, t_phase3,
                                            n_candidates=60, n_grid=40)
    
    # Phase 4: L-BFGS refinement on the polished solution
    if time.time() < deadline - 100:
        t_phase4 = min(time.time() + 60, deadline - 100)
        h_ref, c_ref = lbfgs_phase(best_h, n, t_phase4,
                                     configs=[(50000, 50000.0), (200000, 200000.0)])
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref
    
    # Phase 5: Second round of exact pair polishing
    if time.time() < deadline - 40:
        t_phase5 = min(time.time() + 50, deadline - 40)
        best_h, best_c = exact_pair_polish(best_h, n, t_phase5,
                                            n_candidates=80, n_grid=50)
    
    # Phase 6: Perturbation search for escaping local minima
    if time.time() < deadline - 20:
        best_h, best_c = perturbation_search(best_h, n, deadline - 20)
    
    # Final: one more pair polish with best time
    if time.time() < deadline - 3:
        best_h, best_c = exact_pair_polish(best_h, n, deadline - 2,
                                            n_candidates=50, n_grid=30)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
