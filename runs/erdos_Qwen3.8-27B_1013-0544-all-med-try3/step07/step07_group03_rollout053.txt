∂c[idx]/∂x[i] = dx * [ (1-x[i+s])·1[0≤i+s<n] + sum_j x[j]·(-δ_{j+s,i})·1[0≤j<n] ]
= dx * [ (1-x[i+s])·1[0≤i+s<n] - x[i-s]·1[0≤i-s<n] ]

So:
- If i+s is in [0,n): add dx*(1-x[i+s])
- If i-s is in [0,n): subtract dx*x[i-s]

For the softmax-weighted average:
grad[i] = sum_idx weights[idx] * ∂c[idx]/∂x[i]

Let me implement this cleanly:

```python
def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    s_val = shifted.max()
    weights = np.exp(shifted - s_val)
    total = weights.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights /= total
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        s = idx - (n - 1)
        if s >= 0:
            # (1-x[i+s]) for i in [0, n-s)
            grad[:n-s] += w * dx * (1.0 - x[s:s+n-s])
            # -x[i-s] for i in [s, n)
            grad[s:] -= w * dx * x[:n-s]
        else:
            ms = -s
            # (1-x[i+s]) = (1-x[i-ms]) for i in [ms, n)
            grad[ms:] += w * dx * (1.0 - x[:n-ms])
            # -x[i-s] = -x[i+ms] for i in [0, n-ms)
            grad[:n-ms] -= w * dx * x[ms:]
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(len(x))

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    tw = w.sum()
    if tw < 1e-300:
        return np.zeros(n)
    w /= tw
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-10:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[:n - s] += wi * dx * (1.0 - x[s:s + n - s])
            grad[s:] -= wi * dx * x[:n - s]
        else:
            ms = -s
            grad[ms:] += wi * dx * (1.0 - x[:n - ms])
            grad[:n - ms] -= wi * dx * x[ms:]
    return grad

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), jac=lambda x, *a: smooth_max_grad(x, *a),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search_fast(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for _ in range(3):
                if time.time() > deadline:
                    return h, best_c
                for i in range(n - 1):
                    if time.time() > deadline:
                        return h, best_c
                    j = (i + n // 3) % n
                    if i == j:
                        j = (i + 1) % n
                    for (di, dj) in [(d, -d), (-d, d)]:
                        if 0 <= h[i] + di <= 1 and 0 <= h[j] + dj <= 1:
                            h[i] += di
                            h[j] += dj

                                best_c = c
                                improved = True
                            else:
                                h[i] -= di
                                h[j] -= dj
                if not improved:

def periodic_optimize(h_full, n, deadline):
    if n % 2 != 0:
        return h_full, compute_c5(h_full, n)
    half = n // 2
    y = h_full[:half].copy()
    # Build h from y (1-periodic)
    h = np.tile(y, 2)
    h = project_feasible(h, n)
    y = h[:half].copy()
    y = project_feasible(np.tile(y, 2), n)[:half]

    def periodic_obj(yp):
        hp = np.tile(yp, 2)
        return smooth_max_obj(hp, n, 5000)

    def periodic_jac(yp):
        hp = np.tile(yp, 2)
        gf = smooth_max_grad(hp, n, 5000)
        return gf[:half] + gf[half:]

    bounds_p = [(0.0, 1.0)] * half

    def p_sum_fun(yp, nn):
        return 2.0 * yp.sum() - nn / 2.0

    def p_sum_jac(yp, nn):
        return 2.0 * np.ones(half)

    cons_p = [{'type': 'eq', 'fun': p_sum_fun, 'jac': p_sum_jac, 'args': (n,)}]
    try:
        res = minimize(periodic_obj, y, args=(),
                       jac=lambda yp: periodic_jac(yp),
                       method='SLSQP', bounds=bounds_p, constraints=cons_p,
                       options={'maxiter': 2000, 'ftol': 1e-14})
        y_opt = project_feasible(res.x, half) if res.x is not None else y
        h_opt = np.tile(project_feasible(res.x, half), 2)
        h_opt = project_feasible(h_opt, n)
        c_opt = compute_c5(h_opt, n)
        c_cur = compute_c5(h_full, n)
        if c_opt < c_cur:
            return h_opt, c_opt
    except Exception:
        pass
    return h_full, compute_c5(h_full, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)

    # Phase 1: Optimize at parent resolution (40% of time)
    if time.time() < deadline:
        t1 = time.time() + min(400, (deadline - t0) * 0.40)
        h_opt, c_opt = multi_scale_optimize(h_init, n_init, t1, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Try different resolutions (30% of time)
    n_start = len(best_h)
    for n_try in [200, 300, 400]:
        if time.time() > deadline - 80:
            break
        if n_try == n_start:
            continue
        # Interpolate to new resolution
        x_old = np.linspace(0, 2, n_start)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)

        alloc = min(150, max(40, (deadline - time.time()) * 0.25))
        if alloc < 40:
            break
        h_opt, c_opt = multi_scale_optimize(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart at best resolution (20% of time)
    n_best = len(best_h)
    remaining = deadline - time.time() - 30
    if remaining > 60:
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 20) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = multi_scale_optimize(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Periodic constraint attempt (10% of time)
    if time.time() < deadline - 20:
        h_per, c_per = periodic_optimize(best_h, len(best_h), time.time() + 80)
        if c_per < best_c:
            best_c = c_per
            best_h = h_per

    # Phase 5: Final intensive polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Projected gradient with small steps
        if time.time() < deadline - 5:
            best_h, best_c = proj_grad_refine(best_h, n, deadline - 3, max_iter=100000)

        # Very fine pair search
        if time.time() < deadline - 2:
            deltas_final = [0.001, 0.0005, 0.0002, 0.0001]
            if n <= 300:
                best_h, best_c = pair_search_fast(best_h, n, deadline - 1, deltas_final)

        # Final SLSQP
        if time.time() < deadline - 1:
            ht = slsqp_optimize(best_h, n, 2000000, maxiter=2000, deadline=deadline - 0.5)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
