                g[:] += w[k] * (1.0 - 2.0 * h_curr)
            elif lag > 0:
                g[0:n-mm] += w[k] * (1.0 - h_curr[mm:n])
                g[mm:n] += w[k] * (-h_curr[0:n-mm])
            else:
                g[0:n-mm] += w[k] * (-h_curr[mm:n])
                g[mm:n] += w[k] * (1.0 - h_curr[0:n-mm])
        g *= dx
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h_curr = h_new
        if no_improve > 500:
            lr *= 0.6
            v *= 0.2
            no_improve = 0
            if lr < 0.005:
                break
    return best_h, best_c

def directed_binary_search(h, n, deadline, rng, max_rounds=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    stuck_rounds = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        top_idx = np.argsort(c_all)[-8:]
        for k2 in top_idx:
            if time.time() > deadline:
                break
            lag = k2 - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            for trial in range(30):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n - m)
                j = i + m
                if h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    c_all = compute_all_lags(h, n)
                    stuck_rounds = 0
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
            if improved:
                break
        if not improved:
            # Random swap attempts
            for trial in range(80):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i == j or h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    c_all = compute_all_lags(h, n)
                    stuck_rounds = 0
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            # Kick: 3 random swaps to perturb
            stuck_rounds += 1
            for _ in range(3):
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i != j and h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]
            h = to_binary(h, n)
            c_all = compute_all_lags(h, n)
            if stuck_rounds > 15:
                stuck_rounds = 0
                best_h = to_binary(best_h, n)
                h = best_h.copy()
                c_all = compute_all_lags(h, n)
        h = best_h.copy() if improved else h
    return best_h, best_c

def binary_sa(h, n, deadline, rng, n_iter=500000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    current_c = best_c
    T = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it > 0 and it % 20000 == 0:
            T *= 0.65
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h, n)
        delta = new_c - current_c
        if delta < -1e-14 or (delta < 0.005 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            current_c = new_c
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def two_opt_search(h, n, deadline, rng, max_rounds=100):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for trial in range(20):
            if time.time() > deadline:
                break
            i1 = rng.randint(0, n)
            j1 = rng.randint(0, n)
            i2 = rng.randint(0, n)
            j2 = rng.randint(0, n)
            if i1 == j1 or i1 == i2 or i1 == j2 or j1 == i2 or j1 == j2:
                continue
            if h[i1] == h[j1] or h[i2] == h[j2]:
                continue
            if h[i1] == h[i2]:
                continue
            h[i1], h[j1] = h[j1], h[i1]
            h[i2], h[j2] = h[j2], h[i2]
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            else:
                h[i1], h[j1] = h[j1], h[i1]
                h[i2], h[j2] = h[j2], h[i2]
        if not improved:
            # Try 1-swap again
            for trial in range(40):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i == j or h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return best_h, best_c

def slsqq_top_lags(h, n, alpha, top_m, maxiter, deadline):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_m:]

    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    def grad_fn(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        w = np.exp(shifted)
        w /= w.sum()
        g = np.zeros(n)
        for kk in range(len(top_idx)):
            if w[kk] < 1e-10:
                continue
            k_global = top_idx[kk]
            lag = k_global - (n - 1)
            mm = abs(lag)
            if lag == 0:
                g[:] += w[kk] * (1.0 - 2.0 * x)
            elif lag > 0:
                g[0:n-mm] += w[kk] * (1.0 - x[mm:n])
                g[mm:n] += w[kk] * (-x[0:n-mm])
            else:
                g[0:n-mm] += w[kk] * (-x[mm:n])
                g[mm:n] += w[kk] * (1.0 - x[0:n-mm])
        return g * dx

    def sum_con(x):
        return float(np.sum(x) - n / 2.0)

    def sum_con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, jac=grad_fn, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Quick continuous optimization
    p1_deadline = t0 + 120
    if time.time() < p1_deadline:
        for alpha in [80, 200]:
            if time.time() > p1_deadline - 20:
                break
            h_new, c_new = momentum_opt(h, n, alpha, p1_deadline - 10, lr=1.0, momentum=0.9)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new

    # Phase 2: Binary directed search (main)
    h_bin = to_binary(best_h, n)
    p2_deadline = t0 + 450
    h_bin, c_bin = directed_binary_search(h_bin, n, p2_deadline, rng, max_rounds=500)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()

    # Phase 3: Two-opt search
    if time.time() < p2_deadline + 150:
        p3_deadline = min(deadline - 200, t0 + 600)
        h_bin, c_bin = two_opt_search(to_binary(best_h, n), n, p3_deadline, rng, max_rounds=200)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    # Phase 4: Binary SA
    if time.time() < deadline - 150:
        p4_deadline = min(deadline - 150, t0 + 800)
        h_bin, c_bin = binary_sa(to_binary(best_h, n), n, p4_deadline, rng, n_iter=500000)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    # Phase 5: More directed search
    if time.time() < deadline - 80:
        p5_deadline = min(deadline - 50, t0 + 950)
        h_bin, c_bin = directed_binary_search(to_binary(best_h, n), n, p5_deadline, rng, max_rounds=300)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    # Phase 6: SLSQP polish on top lags
    if time.time() < deadline - 30:
        p6_deadline = min(deadline - 20, t0 + 1050)
        h_cont = to_binary(best_h, n).astype(float)
        h_new = slsqq_top_lags(h_cont, n, 500000, top_m=5, maxiter=2000, deadline=p6_deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_bin = to_binary(best_h, n)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin

    # Phase 7: Final directed search
    if time.time() < deadline - 5:
        h_bin = to_binary(best_h, n)
        h_bin, c_bin = directed_binary_search(h_bin, n, deadline - 2, rng, max_rounds=100)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
