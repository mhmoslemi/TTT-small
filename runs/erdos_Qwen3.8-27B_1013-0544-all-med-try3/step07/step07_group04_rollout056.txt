    grad = np.zeros(n)
    m = abs(lag)

    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha


def _smooth_max_grad(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c_sub = c[top_idx]
    else:
        c_sub = c
    m = c_sub.max()
    shifted = alpha * (c_sub - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    grad = np.zeros(n)
    if top_idx is not None:
        for i in range(len(top_idx)):
            if w[i] > 1e-12:
                lag = int(top_idx[i]) - (n - 1)
                grad += w[i] * lag_gradient(x, n, lag)
    else:
        for k in range(len(c)):
            if w[k] > 1e-12:
                lag = k - (n - 1)
                grad += w[k] * lag_gradient(x, n, lag)
    return dx * grad


def _sum_fun(x, n):
    return float(np.sum(x) - n / 2.0)


def _sum_jac(x, n):
    return np.ones(n)


def _top_lag_obj(x, n, top_idx):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    return float(np.mean(cc[top_idx]))


def _top_lag_grad(x, n, top_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in top_idx:
        lag = int(k) - (n - 1)
        grad += lag_gradient(x, n, lag)
    return dx * grad / len(top_idx)


def slsqp_smooth(h, n, alpha, maxiter, deadline, top_k=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    top_idx = None
    if top_k is not None:
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_k:]
    try:
        res = minimize(_smooth_max_val, h, args=(n, alpha, top_idx),
                       jac=lambda x, *a: _smooth_max_grad(x, *a),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h.copy(), n)


def slsqp_topleft(h, n, top_k, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_k:]
    try:
        res = minimize(_top_lag_obj, h, args=(n, top_idx),
                       jac=_top_lag_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h.copy(), n)


def top_lag_rounds(h, n, deadline, top_k=5, rounds=12):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for r in range(rounds):
        if time.time() > deadline:
            break
        h_new = slsqp_topleft(h, n, top_k, 1500, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            break
    return best_h, best_c


def pair_local_search(h, n, deadline, max_total=50000):
    rng = np.random.RandomState(777)
    h = h.copy()
    best_c = compute_c5(h, n)
    total_trials = 0
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    stuck = 0
    for d in deltas:
        if time.time() > deadline:
            break
        improved_in_pass = True
        while improved_in_pass and time.time() < deadline and total_trials < max_total:
            improved_in_pass = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                break
            if lag > 0:
                contrib = h[:n - m] * (1.0 - h[m:])
            else:
                contrib = h[m:] * (1.0 - h[:n - m])
            i_star = int(np.argmax(contrib))
            if lag > 0:
                i, j = i_star, i_star + m
            else:
                i, j = i_star + m, i_star
            if h[i] > h[j]:
                delta = min(d, 0.45 * (h[i] - h[j]))
            else:
                delta = min(d, 0.45 * (h[j] - h[i]))
            if delta < 1e-12:
                break
            if h[i] >= h[j]:
                h[i] -= delta
                h[j] += delta
            else:
                h[i] += delta
                h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            total_trials += 1
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                improved_in_pass = True
                stuck = 0
            else:
                if h[i] >= h[j]:
                    h[i] += delta
                    h[j] -= delta
                else:
                    h[i] -= delta
                    h[j] += delta
                stuck += 1
                if stuck > 30:
                    improved_in_pass = False
    return h, best_c


def randomized_push(h, n, deadline, n_pushes=500):
    """Randomly push pairs apart to reduce worst lag."""
    rng = np.random.RandomState(42)
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(n_pushes):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        m = abs(lag)
        if m == 0 or m >= n:
            break
        if lag > 0:
            contrib = h[:n - m] * (1.0 - h[m:])
        else:
            contrib = h[m:] * (1.0 - h[:n - m])
        # Pick from top 5 contributors
        top_contrib = np.argsort(contrib)[-5:]
        i_pick = top_contrib[rng.randint(0, len(top_contrib))]
        if lag > 0:
            i, j = i_pick, i_pick + m
        else:
            i, j = i_pick + m, i_pick
        if h[i] >= h[j]:
            delta = min(0.05, 0.5 * h[i], 1.0 - h[j])
            h[i] -= delta
            h[j] += delta
        else:
            delta = min(0.05, 0.5 * h[j], 1.0 - h[i])
            h[i] += delta
            h[j] -= delta
        c_new = compute_c5(h, n)
        if c_new >= best_c:
            if h[i] >= h[j]:
                h[i] += delta
                h[j] -= delta
            else:
                h[i] -= delta
                h[j] += delta
    return h, best_c


def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Round 1: Top-lag SLSQP
    if time.time() < deadline - 25:
        alloc = min(50, (deadline - time.time()) * 0.25)
        h1, c1 = top_lag_rounds(best_h, n, time.time() + alloc, top_k=5, rounds=10)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Round 2: Full smooth-max SLSQP
    if time.time() < deadline - 20:
        alloc = min(35, (deadline - time.time()) * 0.2)
        h2 = slsqp_smooth(best_h, n, 30000, 2000, time.time() + alloc, top_k=None)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Round 3: Another top-lag pass with more lags
    if time.time() < deadline - 15:
        alloc = min(35, (deadline - time.time()) * 0.2)
        h3, c3 = top_lag_rounds(best_h, n, time.time() + alloc, top_k=8, rounds=8)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Round 4: Smooth-max with top-K
    if time.time() < deadline - 12:
        alloc = min(25, (deadline - time.time()) * 0.15)
        h4 = slsqp_smooth(best_h, n, 80000, 2000, time.time() + alloc, top_k=10)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Round 5: Local pair search
    if time.time() < deadline - 3:
        h5, c5 = pair_local_search(best_h, n, min(deadline - 1, time.time() + 20), max_total=30000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize at parent resolution
    if h_init is not None:
        n = n_init
        h_start = project_feasible(h_init, n)
        alloc = min(90, (deadline - t0) * 0.10)
        best_h, best_c = optimize_at_n(h_start, n, t0 + alloc)

    # Phase 2: Try other resolutions
    n_candidates = []
    if h_init is not None:
        n_candidates = [x for x in [200, 250, 350, 400, 500, 600] if abs(x - n_init) > 15]
    else:
        n_candidates = [200, 300, 400, 500]

    for n_try in n_candidates:
        if time.time() > deadline - 55:
            break
        remaining = deadline - time.time() - 35
        if remaining < 45:
            break
        alloc = min(75, remaining * 0.22)
        if alloc < 35:
            break

        if best_h is not None:
            x_src = np.linspace(0, 2, len(best_h))
            x_dst = np.linspace(0, 2, n_try)
            h_start = np.interp(x_dst, x_src, best_h)
        elif h_init is not None:
            x_src = np.linspace(0, 2, len(h_init))
            x_dst = np.linspace(0, 2, n_try)
            h_start = np.interp(x_dst, x_src, h_init)
        else:
            h_start = np.full(n_try, 0.5)

        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-start refinement at best resolution
    if best_h is not None and time.time() < deadline - 45:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_starts = max(2, int(remaining / 35))
        scales = [0.04, 0.06, 0.08, 0.12, 0.15, 0.2]
        for s in range(min(n_starts, len(scales))):
            if time.time() > deadline - 18:
                break
            scale = scales[s]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(30, (deadline - time.time() - 18) * 0.4)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Binary-ish restart (push values toward 0/1)
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        thresh = 0.3
        h_bin = np.where(best_h > thresh, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n)
        # Blend with original
        for blend in [0.0, 0.3, 0.5, 0.7]:
            if time.time() > deadline - 20:
                break
            h_blend = (1.0 - blend) * best_h + blend * h_bin
            h_blend = project_feasible(h_blend, n)
            alloc = min(25, (deadline - time.time() - 15) * 0.3)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_at_n(h_blend, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final intensive polish
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        # SLSQP with very high alpha
        if time.time() < deadline - 12:
            h2 = slsqp_smooth(best_h, n, 200000, 4000, deadline - 8, top_k=6)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        # Top-lag final
        if time.time() < deadline - 6:
            h3, c3 = top_lag_rounds(best_h, n, deadline - 4, top_k=4, rounds=5)
            if c3 < best_c:
                best_c = c3
                best_h = h3
        # Pair search final
        if time.time() < deadline - 2:
            best_h, best_c = pair_local_search(best_h, n, deadline - 0.5, max_total=40000)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = 300
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
