```
def project_feasible(x, n=None):
    if n is None:
        n = len(x)
    x = np.asarray(x, dtype=float)
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - x
            tot = float(room.sum())
            if tot < 1e-15:
                break
            x = x + (diff / tot) * room
        else:
            tot = float(x.sum())
            if tot < 1e-15:
                break
            x = x + (diff / tot) * x
    x = np.clip(x, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if diff > 1e-13:
        idx = int(np.argmax(1.0 - x))
        add = min(diff, 1.0 - x[idx])
        if add > 0:
            x[idx] = float(x[idx]) + add
    elif diff < -1e-13:
        idx = int(np.argmin(x))
        sub = min(-diff, x[idx])
        if sub > 0:
            x[idx] = float(x[idx]) - sub
    return x
```

```python
import numpy as np
import time
import math
from scipy.optimize import minimize


def compute_c_nat(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")[::-1].copy()


def exact_c5(h):
    n = int(len(h))
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def evaluate_candidate(h):
    n = int(len(h))
    h = project_feasible(h, n)
    c = compute_c_nat(h)
    b = float(c.max() * 2.0 / n)
    return h, c, b


def project_feasible(x, n=None):
    if n is None:
        n = int(len(x))
    x = np.asarray(x, dtype=float)
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0

    for _ in range(30):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - x
            tot = float(np.sum(room))
            if tot < 1e-15:
                break
            x = x + (diff / tot) * room
        else:
            tot = float(np.sum(x))
            if tot < 1e-15:
                break
            x = x + (diff / tot) * x

    x = np.clip(x, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if diff > 1e-13:
        idx = int(np.argmax(1.0 - x))
        add = min(diff, 1.0 - x[idx])
        if add > 0.0:
            x[idx] = float(x[idx]) + add
    elif diff < -1e-13:
        idx = int(np.argmin(x))
        sub = min(-diff, x[idx])
        if sub > 0.0:
            x[idx] = float(x[idx]) - sub
    return x


def update_c_for_delta(arr, h, p, delta):
    if delta == 0.0:
        return
    n = int(len(h))
    o = n - 1

    # Contribution where p is the first factor h[p] * (1 - h[p+d]).
    lo = max(-p, -(n - 1))
    hi = min(n - 1 - p, n - 1)
    if lo <= hi:
        start = o + lo
        end = o + hi + 1
        arr[start:end] += delta * (1.0 - h[p + lo:p + hi + 1])

    # Contribution where p is the second factor: -h[p-d].
    lo = max(p - (n - 1), -(n - 1))
    hi = min(p, n - 1)
    if lo <= hi:
        start = o + lo
        end = o + hi + 1
        arr[start:end] -= delta * h[p - hi:p - lo + 1][::-1]


def candidate_transfer(c, h, src, dst, t):
    n = int(len(h))
    o = n - 1
    temp = c.copy()
    update_c_for_delta(temp, h, src, -t)
    update_c_for_delta(temp, h, dst, t)
    d = dst - src
    if d != 0:
        cross = t * t
        temp[o + d] += cross
        temp[o - d] += cross
    return float(temp.max()), temp


def attempt_transfer(c, h, src, dst, current, move_mode, active_idx=None):
    max_t = min(float(h[src]), 1.0 - float(h[dst]))
    if max_t <= 1e-12:
        return False, current, 0

    if move_mode == "full":
        ts = (max_t,)
    elif move_mode == "coarse":
        if max_t > 0.5:
            ts = (min(0.05, max_t), min(0.25, max_t))
        elif max_t > 0.1:
            ts = (min(0.02, max_t), max_t)
        else:
            ts = (max_t,)
    else:
        if max_t > 0.5:
            ts = (min(0.05, max_t), min(0.2, max_t), min(0.5, max_t), max_t)
        elif max_t > 0.15:
            ts = (min(0.02, max_t), min(0.1, max_t), max_t)
        else:
            ts = (min(0.01, max_t), max_t)

    unique_t = []
    for t in ts:
        t = float(t)
        if t > max_t + 1e-12:
            t = max_t
        if t <= 1e-12:
            continue
        dup = False
        for u in unique_t:
            if abs(u - t) <= 1e-13:
                dup = True
                break
        if not dup:
            unique_t.append(t)

    if not unique_t:
        return False, current, 0

    best_count = float(current)
    best_t = 0.0
    best_temp = None
    strict_found = False

    use_neutral = active_idx is not None and len(active_idx) > 0
    current_energy = float(np.sum(c[active_idx])) if use_neutral else 0.0
    best_energy = current_energy

    count_tol = 1e-10
    neutral_tol = 1e-9

    for t in unique_t:
        new_count, temp = candidate_transfer(c, h, src, dst, t)
        if new_count < best_count - count_tol:
            best_count = new_count
            best_t = t
            best_temp = temp
            strict_found = True
        elif (not strict_found) and use_neutral and new_count <= float(current) + neutral_tol:
            energy = float(np.sum(temp[active_idx]))
            if energy < best_energy - 1e-8:
                best_t = t
                best_temp = temp
                best_count = new_count
                best_energy = energy

    if best_temp is None:
        return False, current, 0

    h[src] -= best_t
    h[dst] += best_t
    if h[src] < 0.0:
        h[src] = 0.0
    elif h[src] > 1.0:
        h[src] = 1.0
    if h[dst] < 0.0:
        h[dst] = 0.0
    elif h[dst] > 1.0:
        h[dst] = 1.0
    c[:] = best_temp

    if strict_found:
        return True, best_count, 1
    return True, best_count, 2


def local_search_oriented(
    h,
    c,
    deadline,
    top_lags=10,
    move_mode="continuous",
    max_rounds=100000,
    neutral_limit=10,
    max_pairs=None,
    max_accepts_per_round=50,
):
    n = int(len(h))
    o = n - 1
    h = np.asarray(h, dtype=float)
    c = np.asarray(c, dtype=float)

    best_h = h.copy()
    best_c = c.copy()
    best_count = float(c.max())
    current = best_count

    for rnd in range(max_rounds):
        if deadline is not None and time.time() >= deadline:
            break

        current = float(c.max())
        if current < best_count - 1e-12:
            best_count = current
            best_h = h.copy()
            best_c = c.copy()

        maxc = current
        order = np.argsort(c)[::-1]

        active_idx_list = []
        for idx in order:
            if c[idx] < maxc - 1.5:
                break
            if len(active_idx_list) < top_lags and (int(idx) - o) != 0:
                active_idx_list.append(int(idx))

        for idx in order:
            if len(active_idx_list) >= top_lags:
                break
            if (int(idx) - o) != 0 and int(idx) not in active_idx_list:
                active_idx_list.append(int(idx))

        if not active_idx_list:
            break

        active_idx = np.asarray(active_idx_list, dtype=int)
        current_energy = float(np.sum(c[active_idx]))
        strict = 0
        neutral = 0
        accepts = 0

        for idx in active_idx_list:
            if deadline is not None and time.time() >= deadline:
                break
            if accepts >= max_accepts_per_round:
                break

            d = int(idx) - o
            if d > 0:
                start = 0
                end = n - d
            else:
                start = -d
                end = n

            if max_pairs is None:
                for x in range(start, end):
                    if accepts >= max_accepts_per_round:
                        break
                    if (x % 256 == 0) and deadline is not None and time.time() >= deadline:
                        break
                    src = x
                    dst = x + d
                    if h[src] <= 1e-10 or h[dst] >= 1.0 - 1e-10:
                        continue
                    allow_neutral = (strict == 0 and neutral < neutral_limit)
                    ok, newc, typ = attempt_transfer(
                        c, h, src, dst, current, move_mode,
                        active_idx if allow_neutral else None
                    )
                    if ok:
                        current = newc
                        accepts += 1
                        if typ == 1:
                            strict += 1
                            if current < best_count - 1e-12:
                                best_count = current
                                best_h = h.copy()
                                best_c = c.copy()
                        else:
                            neutral += 1
            else:
                srcs = np.arange(start, end, dtype=int)
                dsts = srcs + d
                pot = np.minimum(h[srcs], 1.0 - h[dsts])
                mask = pot > 1e-5
                if np.any(mask):
                    positions_all = np.nonzero(mask)[0]
                    if max_pairs is not None and len(positions_all) > max_pairs:
                        p = pot[positions_all]
                        if len(positions_all) > 4 * max_pairs:
                            part = np.argpartition(p, -max_pairs)[-max_pairs:]
                            part = part[np.argsort(p[part])[::-1]]
                        else:
                            part = np.argsort(p)[::-1][:max_pairs]
                        positions = positions_all[part]
                    else:
                        positions = positions_all

                    for pos in positions:
                        if accepts >= max_accepts_per_round:
                            break
                        if (int(pos) % 256 == 0) and deadline is not None and time.time() >= deadline:
                            break
                        src = int(srcs[pos])
                        dst = int(dsts[pos])
                        if h[src] <= 1e-10 or h[dst] >= 1.0 - 1e-10:
                            continue
                        allow_neutral = (strict == 0 and neutral < neutral_limit)
                        ok, newc, typ = attempt_transfer(
                            c, h, src, dst, current, move_mode,
                            active_idx if allow_neutral else None
                        )
                        if ok:
                            current = newc
                            accepts += 1
                            if typ == 1:
                                strict += 1
                                if current < best_count - 1e-12:
                                    best_count = current
                                    best_h = h.copy()
                                    best_c = c.copy()
                            else:
                                neutral += 1

            current = float(c.max())
            if current < best_count - 1e-12:
                best_count = current
                best_h = h.copy()
                best_c = c.copy()

        # Reconcretize to remove numerical drift.
        c_exact = compute_c_nat(h)
        c[:] = c_exact
        current = float(c.max())
        if current < best_count - 1e-12:
            best_count = current
            best_h = h.copy()
            best_c = c.copy()

        if strict == 0 and neutral == 0:
            break

    return best_h, best_c, best_count


def gradient_active(h, lag_idx, weights):
    n = int(len(h))
    o = n - 1
    g = np.zeros(n, dtype=float)
    for idx, w in zip(lag_idx, weights):
        if abs(w) < 1e-14:
            continue
        d = int(idx) - o
        if d == 0:
            g[:] += w * (1.0 - 2.0 * h)
        elif d > 0:
            g[:n - d] += w * (1.0 - h[d:])
            g[d:] += -w * h[:n - d]
        else:
            m = -d
            g[m:] += w * (1.0 - h[:n - m])
            g[:n - m] += -w * h[m:]
    return g


def gradient_phase(h, deadline, active_k=10, alpha=100.0, lr_start=0.035, max_iter=10000):
    n = int(len(h))
    h = np.asarray(h, dtype=float).copy()
    c = compute_c_nat(h)
    current = float(c.max())
    lr = float(lr_start)
    no_improve = 0

    for it in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break
        current = float(c.max())
        k = min(active_k, len(c))
        if k <= 0:
            break
        kth = min(k - 1, len(c) - 1)
        idx = np.argpartition(c, kth)[-k:]
        vals = c[idx]
        m = float(vals.max())
        sh = alpha * (vals - m)
        w = np.exp(sh)
        s = float(w.sum())
        if s <= 0:
            w = np.ones_like(vals) / float(len(vals))
        else:
            w = w / s

        g = gradient_active(h, idx, w)
        g -= float(np.mean(g))
        gn = float(np.max(np.abs(g)))
        if gn < 1e-14:
            break
        g = g / gn

        improved = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05):
            y = project_feasible(h - lr * mult * g, n)
            c_trial = compute_c_nat(y)
            ct = float(c_trial.max())
            if ct < current - 1e-11:
                h = y
                c = c_trial
                current = ct
                improved = True
                break

        if not improved:
            no_improve += 1
            lr *= 0.8
            if no_improve > 35 or lr < 1e-5:
                break
        else:
            no_improve = 0
            lr = min(0.08, lr * 1.03)

    return h, c, float(c.max())


def slsqp_sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def slsqp_sum_jac(x, n):
    return np.ones(int(len(x)), dtype=float)


def slsqp_smooth_active_val(x, n, idx, alpha):
    c = compute_c_nat(x)
    vals = c[idx]
    if len(vals) == 0:
        return 0.0
    m = float(vals.max())
    sh = alpha * (vals - m)
    s = float(np.sum(np.exp(sh)))
    if s <= 0.0:
        return m
    return float(m + math.log(s) / alpha)


def slsqp_smooth_active_grad(x, n, idx, alpha):
    c = compute_c_nat(x)
    vals = c[idx]
    if len(vals) == 0:
        return np.zeros(int(len(x)), dtype=float)
    m = float(vals.max())
    sh = alpha * (vals - m)
    w = np.exp(sh)
    s = float(w.sum())
    if s <= 0.0:
        w = np.ones_like(vals) / float(len(vals))
    else:
        w = w / s
    return gradient_active(x, idx, w)


def select_active_indices(c, top_k, window=1.0):
    c = np.asarray(c, dtype=float)
    maxc = float(c.max())
    order = np.argsort(c)[::-1]
    idx_list = []
    limit = max(int(top_k), 6)
    for idx in order:
        if c[idx] < maxc - window:
            break
        idx_list.append(int(idx))
        if len(idx_list) >= limit:
            break
    for idx in order:
        if len(idx_list) >= top_k:
            break
        if int(idx) not in idx_list:
            idx_list.append(int(idx))
    cap = max(int(top_k), 6)
    return np.asarray(idx_list[:cap], dtype=int)


def slsqp_active_polish(
    h,
    c,
    deadline,
    rng=None,
    top_k_list=(3, 5),
    noise_list=(0.0, 0.02),
    alpha=500.0,
    max_solves=4,
    maxiter=90,
):
    n = int(len(h))
    best_h = np.asarray(h, dtype=float).copy()
    best_c = compute_c_nat(best_h)
    best_count = float(best_c.max())

    bounds = [(0.0, 1.0)] * n
    cons = {
        "type": "eq",
        "fun": slsqp_sum_con,
        "jac": slsqp_sum_jac,
        "args": (n,),
    }
    solves = 0

    for noise in noise_list:
        for top_k in top_k_list:
            if time.time() >= deadline or solves >= max_solves:
                break
            x0 = best_h.copy()
            if noise > 0.0:
                if rng is not None:
                    x0 = project_feasible(x0 + noise * rng.randn(n), n)
                else:
                    tt = np.linspace(0.0, 1.0, n)
                    x0 = project_feasible(x0 + noise * np.sin(5.0 * math.pi * tt), n)
            c0 = compute_c_nat(x0)
            active = select_active_indices(c0, top_k, window=1.0)
            if len(active) == 0:
                continue
            try:
                res = minimize(
                    slsqp_smooth_active_val,
                    x0,
                    args=(n, active, float(alpha)),
                    method="SLSQP",
                    jac=slsqp_smooth_active_grad,
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": int(maxiter), "ftol": 1e-14},
                )
                cand = project_feasible(np.asarray(res.x, dtype=float), n)
            except Exception:
                cand = project_feasible(x0, n)

            cc = compute_c_nat(cand)
            cb = float(cc.max())
            if cb < best_count - 1e-11:
                best_count = cb
                best_h = cand
                best_c = cc
            solves += 1

    return best_h, best_c, best_count


def annealed_oriented(
    h,
    c,
    deadline,
    rng,
    T0=0.20,
    Tmin=0.01,
    decay=0.9995,
    top_lags=6,
    full_flips=False,
    max_iter=120000,
):
    n = int(len(h))
    o = n - 1
    h = np.asarray(h, dtype=float).copy()
    c = np.asarray(c, dtype=float).copy()

    best_h = h.copy()
    best_c = c.copy()
    best_count = float(c.max())
    current = best_count
    T = float(T0)
    it = 0
    active = []

    while it < max_iter:
        if deadline is not None and time.time() >= deadline:
            break
        it += 1

        if not active or it % 1000 == 0:
            order = np.argsort(c)[::-1]
            maxc = float(c[order[0]])
            active = []
            for idx in order:
                if c[idx] < maxc - 1.0:
                    break
                active.append(int(idx))
                if len(active) >= 2 * top_lags:
                    break
            for idx in order:
                if len(active) >= top_lags:
                    break
                if int(idx) not in active:
                    active.append(int(idx))

        if active and rng.random() < 0.7:
            idx = active[int(rng.randint(len(active)))]
        else:
            idx = int(rng.randint(0, len(c)))

        d = int(idx) - o
        if d == 0:
            continue

        if d > 0:
            x = int(rng.randint(0, n - d))
        else:
            x = int(rng.randint(-d, n))

        src = x
        dst = x + d
        max_t = min(float(h[src]), 1.0 - float(h[dst]))
        if max_t <= 1e-10:
            continue

        if full_flips or max_t > 0.75:
            t = max_t if rng.random() < 0.85 else 0.5 * max_t
        else:
            u = rng.random()
            if u < 0.1:
                t = max_t
            elif u < 0.35:
                t = 0.25 * max_t
            else:
                t = (0.05 + 0.95 * rng.random()) * max_t

        new_count, temp = candidate_transfer(c, h, src, dst, t)
        delta = new_count - current

        if delta < -1e-11:
            accept = True
        else:
            if delta > 0.0:
                if delta < 50.0 * T:
                    accept = rng.random() < math.exp(-delta / T)
                else:
                    accept = False
            else:
                accept = True

        if accept:
            h[src] -= t
            h[dst] += t
            if h[src] < 0.0:
                h[src] = 0.0
            if h[dst] > 1.0:
                h[dst] = 1.0
            c[:] = temp
            current = new_count

            if current < best_count - 1e-11:
                best_count = current
                best_h = h.copy()
                best_c = c.copy()

            if current > best_count + 3.0:
                h = best_h.copy()
                c = best_c.copy()
                current = best_count
                T = max(T, 0.15)

        if T > Tmin:
            T = max(Tmin, T * decay)

    return best_h, best_c, best_count


def upsample_h(h, m):
    n = int(len(h))
    m = int(m)
    if m == n:
        return project_feasible(h, n)
    if m % n == 0:
        return project_feasible(np.repeat(h, m // n), m)
    x_old = np.linspace(0.0, 2.0, n)
    x_new = np.linspace(0.0, 2.0, m)
    return project_feasible(np.interp(x_new, x_old, h), m)


def make_binary_h(h):
    h = np.asarray(h, dtype=float)
    n = int(len(h))
    hb = np.zeros(n, dtype=float)
    order = np.argsort(h)[::-1]
    count_full = n // 2
    if count_full > 0:
        hb[order[:count_full]] = 1.0
    if n % 2 == 1 and n > count_full:
        hb[int(order[count_full])] = 0.5
    return project_feasible(hb, n)


def random_binary_h(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    size = n // 2
    idx = np.array([], dtype=int)
    if size > 0:
        idx = rng.choice(n, size=size, replace=False)
        h[idx] = 1.0
    if n % 2 == 1:
        remaining = np.setdiff1d(np.arange(n), idx)
        if len(remaining) > 0:
            h[int(remaining[0])] = 0.5
    return project_feasible(h, n)

            if cb < best_bound - 1e-12:
                best_h, best_c, best_bound = ht, ct, cb
                n_best = int(len(ht))

    # Analytic-gradient SLSQP polish on a small number of active lags.
    if n_best <= 700 and time.time() < deadline - 80.0:
        alloc = min(120.0, 0.12 * (deadline - time.time()))
        if alloc > 35.0:
            hs, cs, cnt = slsqp_active_polish(
                best_h,
                best_c,
                time.time() + alloc,
                rng=rng,
                top_k_list=(3, 5),
                noise_list=(0.0, 0.02),
                alpha=500.0,
                max_solves=4,
                maxiter=90,
            )
            cb = float(cnt * 2.0 / len(hs))
            if cb < best_bound - 1e-12:
                best_h, best_c, best_bound = hs, cs, cb
                n_best = int(len(hs))

    # Full-transfer local search on continuous / near-binary structures.
    if time.time() < deadline - 50.0:
        alloc = min(90.0, 0.08 * (deadline - time.time()))
        if alloc > 20.0:
            max_pairs = None if n_best <= 700 else min(600, n_best // 2)
            ht, ct, cnt = local_search_oriented(
                best_h.copy(),
                best_c.copy(),
                time.time() + alloc,
                top_lags=10,
                move_mode="full",
                neutral_limit=0,
                max_pairs=max_pairs,
                max_accepts_per_round=60,
            )
            cb = float(cnt * 2.0 / len(ht))
            if cb < best_bound - 1e-12:
                best_h, best_c, best_bound = ht, ct, cb
                n_best = int(len(ht))

    # Binary / near-binary basins.
    if time.time() < deadline - 90.0:
        bases = [best_h, 1.0 - best_h]
        if n_best <= 1000:
            bases.append(random_binary_h(n_best, rng))

        for base in bases:
            if time.time() > deadline - 60.0:
                break
            hb, cb0 = evaluate_candidate(make_binary_h(base))
            cc0 = compute_c_nat(hb)
            if cb0 < best_bound - 1e-12:
                best_h, best_c, best_bound = hb, cc0, cb0
                n_best = int(len(hb))

            if time.time() < deadline - 45.0:
                alloc = min(65.0, 0.08 * (deadline - time.time()))
                if alloc > 15.0:
                    max_pairs = None if hb.size <= 700 else min(500, hb.size // 2)
                    hl, cl, cnt = local_search_oriented(
                        hb.copy(),
                        cc0.copy(),
                        time.time() + alloc,
                        top_lags=10,
                        move_mode="full",
                        neutral_limit=0,
                        max_pairs=max_pairs,
                        max_accepts_per_round=60,
                    )
                    cb = float(cnt * 2.0 / len(hl))
                    if cb < best_bound - 1e-12:
                        best_h, best_c, best_bound = hl, cl, cb
                        n_best = int(len(hl))
                        hb, cc0 = hl, cl

            if time.time() < deadline - 25.0:
                alloc = min(25.0, 0.04 * (deadline - time.time()))
                if alloc > 8.0:
                    ha, ca, cnt = annealed_oriented(
                        hb.copy(),
                        cc0.copy(),
                        time.time() + alloc,
                        rng=rng,
                        T0=0.15,
                        Tmin=0.005,
                        decay=0.9997,
                        top_lags=6,
                        full_flips=True,
                    )
                    cb = float(cnt * 2.0 / len(ha))
                    if cb < best_bound - 1e-12:
                        best_h, best_c, best_bound = ha, ca, cb
                        n_best = int(len(ha))

    # Higher-resolution refinements.
    for m in (1000, 1500, 2000):
        if len(best_h) >= m:
            continue
        if time.time() > deadline - 100.0:
            break
        rem = deadline - time.time() - 30.0
        if rem < 90.0:
            break

        hh, cc, bc = evaluate_candidate(upsample_h(best_h, m))
        if bc > best_bound + 0.035 and rem < 260.0:
            continue

        alloc = min(180.0, 0.36 * rem)
        if alloc <= 40.0:
            continue

        if m <= 1000:
            mode = "coarse"
            top = 8
            paircap = min(600, m // 2)
            accepts = 45
        else:
            mode = "coarse"
            top = 6
            paircap = min(400, m // 3)
            accepts = 30

        hl, cl, cnt = local_search_oriented(
            hh.copy(),
            cc.copy(),
            time.time() + alloc,
            top_lags=top,
            move_mode=mode,
            neutral_limit=0,
            max_pairs=paircap,
            max_accepts_per_round=accepts,
        )
        hh, cc = hl, cl
        cb = float(cnt * 2.0 / m)
        if cb < best_bound - 1e-12:
            best_h, best_c, best_bound = hh, cc, cb
            n_best = int(len(hh))

        if time.time() < deadline - 40.0:
            alloc2 = min(60.0, 0.22 * (deadline - time.time()))
            if alloc2 > 20.0:
                hf, cf, cnt2 = local_search_oriented(
                    hh.copy(),
                    cc.copy(),
                    time.time() + alloc2,
                    top_lags=6,
                    move_mode="full",
                    neutral_limit=0,
                    max_pairs=min(500, m // 2),
                    max_accepts_per_round=35,
                )
                hh, cc = hf, cf
                cb2 = float(cnt2 * 2.0 / m)
                if cb2 < best_bound - 1e-12:
                    best_h, best_c, best_bound = hh, cc, cb2
                    n_best = int(len(hh))

        if m <= 1200 and time.time() < deadline - 30.0:
            alloc3 = min(30.0, 0.08 * (deadline - time.time()))
            if alloc3 > 10.0:
                ha, ca, cnt3 = annealed_oriented(
                    hh.copy(),
                    cc.copy(),
                    time.time() + alloc3,
                    rng=rng,
                    T0=0.12,
                    Tmin=0.005,
                    decay=0.9997,
                    top_lags=6,
                    full_flips=False,
                )
                cb3 = float(cnt3 * 2.0 / m)
                if cb3 < best_bound - 1e-12:
                    best_h, best_c, best_bound = ha, ca, cb3
                    n_best = int(len(ha))

    # Perturbed restarts from the current best.
    if time.time() < deadline - 60.0:
        for scale in (0.02, 0.05):
            if time.time() > deadline - 40.0:
                break
            rem = deadline - time.time() - 20.0
            if rem < 35.0:
                break
            alloc = min(45.0, 0.25 * rem)
            if alloc < 15.0:
                break
            hp = best_h + scale * rng.randn(len(best_h))
            hp, cp, _ = evaluate_candidate(hp)
            hm, cm, cnt = local_search_oriented(
                hp.copy(),
                cp.copy(),
                time.time() + alloc,
                top_lags=10,
                move_mode="continuous" if len(hp) <= 900 else "coarse",
                neutral_limit=4,
                max_pairs=None if len(hp) <= 700 else min(500, len(hp) // 2),
                max_accepts_per_round=40,
            )
            cb = float(cnt * 2.0 / len(hm))
            if cb < best_bound - 1e-12:
                best_h, best_c, best_bound = hm, cm, cb
                n_best = int(len(hm))

    # Final short polish.
    if time.time() < deadline - 2.0:
        alloc = min(20.0, deadline - time.time() - 1.0)
        if alloc > 4.0:
            hf, cf, cnt = local_search_oriented(
                best_h.copy(),
                best_c.copy(),
                time.time() + alloc,
                top_lags=8,
                move_mode="continuous" if len(best_h) <= 900 else "coarse",
                neutral_limit=3,
                max_pairs=None if len(best_h) <= 700 else min(400, len(best_h) // 2),
                max_accepts_per_round=30,
            )
            cb = float(cnt * 2.0 / len(hf))
            if cb < best_bound - 1e-12:
                best_h, best_c, best_bound = hf, cf, cb
                n_best = int(len(hf))

    best_h = project_feasible(best_h, len(best_h))
    cf = compute_c_nat(best_h)
    best_bound = float(cf.max() * 2.0 / len(best_h))
    return best_h, best_bound, int(len(best_h))
```
