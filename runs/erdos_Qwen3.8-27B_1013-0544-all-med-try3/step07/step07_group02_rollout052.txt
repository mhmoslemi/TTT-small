    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def _lag_grad(h, n, lag):
    """Gradient of correlation at given lag w.r.t. h"""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]

    # For positive lag: c[lag] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
    # For negative lag: c[lag] = sum_{i=|lag|}^{n-1} h[i]*(1-h[i+lag])

    # result[n-1+lag] = sum_{i} h[i] * (1-h[i+lag]) for valid i

    m = abs(lag)
    if lag >= 0:
        # i ranges from 0 to n-1-lag
        lo, hi = 0, n-1-lag
    else:
        # i ranges from -lag to n-1
        lo, hi = -lag, n-1
    
    delta = 0.0
    for i in [a, b, a - lag, b - lag]:

            if 0 <= j < n:

The delta for swap (a: 1→0, b: 0→1):
δ = [new contributions from a, b, a-k, b-k] - [old contributions]

For position i=a (h[a]: 1→0):
- Old: 1*(1-h[a+k])
- New: 0*(1-h'[a+k])
- If a+k == b: new = 0*(1-1) = 0, old = 1*(1-0) = 1, δ_a = -1
- If a+k ≠ b: new = 0, old = 1*(1-h[a+k]), δ_a = -(1-h[a+k])

For position i=b (h[b]: 0→1):
- Old: 0*(1-h[b+k]) = 0
- New: 1*(1-h'[b+k])
- If b+k == a: new = 1*(1-0) = 1, old = 0, δ_b = +1
- If b+k ≠ a: new = 1*(1-h[b+k]), old = 0, δ_b = (1-h[b+k])

For position i=a-k (the (1-h[i+k]) term where i+k=a):
- Old: h[a-k]*(1-1) = 0
- New: h'[a-k]*(1-0)
- If a-k == b: h'[b]=1, so new = 1, old = 0, δ_{a-k} = +1
- If a-k ≠ b: h'[a-k]=h[a-k], new = h[a-k]*1, old = 0, δ_{a-k} = h[a-k]

For position i=b-k (the (1-h[i+k]) term where i+k=b):
- Old: h[b-k]*(1-0) = h[b-k]
- New: h'[b-k]*(1-1) = 0
- If b-k == a: h'[a]=0, new = 0*0 = 0, old = 1*1 = 1, δ_{b-k} = -1
- If b-k ≠ a: h'[b-k]=h[b-k], new = h[b-k]*0 = 0, old = h[b-k]*1, δ_{b-k} = -h[b-k]

So for a given lag k and swap (a,b), the total delta is:
δ = δ_a + δ_b + δ_{a-k} + δ_{b-k} (for valid positions)

Where each δ is O(1) to compute.

        else:

            return float(m + np.log(np.sum(np.exp(shifted - shifted.max()))) / alpha)

    bounds = [(0.0, 1.0)] * n

        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        # Also try binarizing the fractional solution
        h_bin2 = binarize(h, n)
        c_bin2 = compute_c5(h_bin2, n)
        if c_bin2 < best_c:
            best_c = c_bin2
            best_h = h_bin2.copy()
    
    # Phase 2: Nesterov gradient on active lags
    if time.time() < deadline - 30:
        for alpha in [200, 500, 1000]:
            if time.time() > deadline - 20:
                break
            h_g, c_g, _ = nesterov_optimize(best_h, n, time.time() + 25, alpha, lr0=0.5, top_k=5)
            if c_g < best_c:
                best_c = c_g
                best_h = h_g
    
    # Phase 3: Fractional local search
    if time.time() < deadline - 15:
        h_fs, c_fs = fractional_local_search(best_h, n, time.time() + 20, max_rounds=40, step=0.02)
        if c_fs < best_c:
            best_c = c_fs
            best_h = h_fs
    
    # Phase 4: SLSQP on top-5 lags
    if time.time() < deadline - 10:
        h_s = slsqp_optimize(best_h, n, alpha=5000, maxiter=2000, ftol=1e-14, top_m=5)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
    
    # Phase 5: Final fractional polish
    if time.time() < deadline - 3:
        h_fp, c_fp = fractional_local_search(best_h, n, deadline - 1, max_rounds=30, step=0.005)
        if c_fp < best_c:
            best_c = c_fp
            best_h = h_fp
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Load parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    n = n_init
    
    # Main optimization at parent resolution
    alloc1 = min(400, (deadline - t0) * 0.4)
    h_opt, c_opt = optimize_from(best_h, n, t0 + alloc1)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    
    # Multi-start: perturb and re-optimize
    n_starts = 5
    for s in range(n_starts):
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 40
        if remaining < 50:
            break
        alloc = min(80, remaining / (n_starts - s) * 0.8)
        
        # Different perturbation strategies
        if s == 0:
            hp = best_h + rng.normal(0, 0.03, n)
        elif s == 1:
            hp = best_h.copy()
            # Swap 3 random pairs
            ones = np.where(hp > 0.7)[0]
            zeros = np.where(hp < 0.3)[0]
            for _ in range(3):
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    hp[i], hp[j] = hp[j], hp[i]
        elif s == 2:
            # Binary + random swap
            hp = binarize(best_h, n)
            a = rng.randint(n)
            b = rng.randint(n)
            if hp[a] != hp[b]:
                hp[a], hp[b] = hp[b], hp[a]
        elif s == 3:
            # Smooth perturbation
            hp = best_h.copy()
            kernel = np.exp(-0.5 * (np.arange(-3, 4) / 1.5) ** 2)
            hp = np.convolve(hp + rng.normal(0, 0.05, n), kernel, mode='same') / kernel.sum()
        else:
            # Larger perturbation
            hp = best_h + rng.normal(0, 0.08, n)
        
        hp = project_feasible(hp, n)
        h_opt, c_opt = optimize_from(hp, n, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Try slightly different resolutions
    for n_try in [n + 2, n + 4, n - 2]:
        if n_try < 100:
            continue
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 50:
            break
        alloc = min(60, remaining * 0.5)
        
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        h_opt, c_opt = optimize_from(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_try
    
    # Final intensive polish at best resolution
    n_final = len(best_h)
    if time.time() < deadline - 20:
        # SLSQP with high alpha on top 3 lags
        h_s = slsqp_optimize(best_h, n_final, alpha=10000, maxiter=3000, ftol=1e-15, top_m=3)
        c_s = compute_c5(h_s, n_final)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
        
        # Nesterov with very sharp focus
        if time.time() < deadline - 10:
            h_g, c_g, _ = nesterov_optimize(best_h, n_final, deadline - 8, alpha=2000, lr0=0.3, top_k=3)
            if c_g < best_c:
                best_c = c_g
                best_h = h_g
        
        # Fractional search with tiny steps
        if time.time() < deadline - 3:
            h_fp, c_fp = fractional_local_search(best_h, n_final, deadline - 1, max_rounds=50, step=0.003)
            if c_fp < best_c:
                best_c = c_fp
                best_h = h_fp
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
