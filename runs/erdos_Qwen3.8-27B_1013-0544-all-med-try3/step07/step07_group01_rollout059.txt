import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def topk_shift_grad(h, n, k_count=3):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    # Find top k shifts
    top_idx = np.argsort(c)[-k_count:]
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in top_idx:
        k = m_idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += dx * (1.0 - h[k:n])
            grad[k:n] += dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += dx * (1.0 - h[0:nv])
            grad[0:nv] += dx * (-h[kk:n])
    return grad, c_max

def fast_swap_delta(h, n, a, b, d):
    delta = np.zeros(2 * n - 1)
    # Positive shifts k=0..n-1 at indices n-1..2n-2
    # dO(k) = d*(1-h[a+k])*1[k<=n-1-a] - d*h[a-k]*1[k<=a]
    #         - d*(1-h[b+k])*1[k<=n-1-b] + d*h[b-k]*1[k<=b]
    
    la = min(n - 1 - a, n - 1)
    if la >= 0:
        delta[n-1 : n-1+la+1] += d * (1.0 - h[a : a+la+1])
    
    lk = min(a, n - 1)
    if lk >= 0:
        delta[n-1 : n-1+lk+1] -= d * h[a-lk : a+1][::-1]
    
    lb = min(n - 1 - b, n - 1)
    if lb >= 0:
        delta[n-1 : n-1+lb+1] -= d * (1.0 - h[b : b+lb+1])
    
    lbk = min(b, n - 1)
    if lbk >= 0:
        delta[n-1 : n-1+lbk+1] += d * h[b-lbk : b+1][::-1]
    
    # Negative shifts: ka=1..n-1 at indices n-2..0
    ka = np.arange(1, n)
    val = np.zeros(ka.shape[0])
    
    m1 = ka <= a
    if m1.any():
        val[m1] += d * (1.0 - h[a - ka[m1]])
    
    m2 = (a + ka) < n
    if m2.any():
        val[m2] -= d * h[a + ka[m2]]
    
    m3 = ka <= b
    if m3.any():
        val[m3] -= d * (1.0 - h[b - ka[m3]])
    
    m4 = (b + ka) < n
    if m4.any():
        val[m4] += d * h[b + ka[m4]]
    
    delta[0 : n-1] += val[::-1]  # indices 0..n-2 correspond to ka=n-1..1
    # Wait, ka=1 maps to index n-2, ka=n-1 maps to index 0
    # So delta[n-1-ka] gets val[m]. Let me redo.
    
    # Reset and redo properly
    delta = np.zeros(2 * n - 1)
    la = min(n - 1 - a, n - 1)
    if la >= 0:
        delta[n-1 : n-1+la+1] += d * (1.0 - h[a : a+la+1])
    lk = min(a, n - 1)
    if lk >= 0:
        delta[n-1 : n-1+lk+1] -= d * h[a-lk : a+1][::-1]
    lb = min(n - 1 - b, n - 1)
    if lb >= 0:
        delta[n-1 : n-1+lb+1] -= d * (1.0 - h[b : b+lb+1])
    lbk = min(b, n - 1)
    if lbk >= 0:
        delta[n-1 : n-1+lbk+1] += d * h[b-lbk : b+1][::-1]
    
    # Negative shifts: ka = 1..n-1, delta index = n-1-ka
    ka = np.arange(1, n)
    val_neg = np.zeros(ka.shape[0])
    m1 = ka <= a
    if m1.any():
        val_neg[m1] += d * (1.0 - h[a - ka[m1]])
    m2 = (a + ka) < n
    if m2.any():
        val_neg[m2] -= d * h[a + ka[m2]]
    m3 = ka <= b
    if m3.any():
        val_neg[m3] -= d * (1.0 - h[b - ka[m3]])
    m4 = (b + ka) < n
    if m4.any():
        val_neg[m4] += d * h[b + ka[m4]]
    
    idx_neg = n - 1 - ka  # for ka=1: n-2, ka=n-1: 0
    delta[idx_neg] += val_neg
    
    return delta

def pair_search_fast(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        delta = fast_swap_delta(h, n, i, j, d)
                        new_max = (c_vec + delta).max()
                        if new_max < best_c - 1e-14:
                            best_c = float(new_max)
                            h[i] += d
                            h[j] -= d
                            c_vec = c_vec + delta
                            improved = True
                            c_vec = compute_c5_vec(h, n)  # refresh
                        else:
                            pass
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        delta = fast_swap_delta(h, n, i, j, -d)
                        new_max = (c_vec + delta).max()
                        if new_max < best_c - 1e-14:
                            best_c = float(new_max)
                            h[i] -= d
                            h[j] += d
                            c_vec = c_vec + delta
                            improved = True
                            c_vec = compute_c5_vec(h, n)
            if not improved:
                break
    return h, best_c

def momentum_gradient_descent(h, n, deadline, lr_init=0.005, momentum=0.9, alpha_init=500.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    alpha = alpha_init
    
    for it in range(50000):
        if time.time() > deadline:
            break
        # Use top-k shift gradient for focus
        c_vec = compute_c5_vec(h, n)
        c_max = c_max_actual = c_vec.max()
        # Smooth max gradient with current alpha
        c_shift = alpha * (c_vec - c_max)
        mx = c_shift.max()
        w = np.exp(c_shift - mx)
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for m_idx in range(2 * n - 1):
            if w[m_idx] < 1e-14:
                continue
            k = m_idx - (n - 1)
            wm = w[m_idx] * dx
            if k >= 0:
                nv = n - k
                g[0:nv] += wm * (1.0 - h[k:n])
                g[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wm * (1.0 - h[0:nv])
                g[0:nv] += wm * (-h[kk:n])
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        # Backtracking line search with momentum
        v = momentum * v + (1 - momentum) * g
        improved = False
        step_lr = lr
        for _ in range(50):
            h_new = project_feasible(h - step_lr * v, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step_lr *= 0.5
        
        if not improved:
            lr *= 0.7
            v *= 0.5
            alpha *= 1.8
            if lr < 1e-12:
                break
        else:
            alpha = max(50.0, alpha * 1.02)
    
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, n, 50000)
    
    def obj_grad(x):
        return smooth_max_grad(x, n, 50000)
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, bounds=bounds,
                      constraints=cons, method='SLSQP',
                      options={'maxiter': max_iters, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    
    # L-BFGS-B with penalty
    def pen_obj(x):
        return smooth_max_obj(x, n, 50000) + 10000.0 * (x.sum() - n/2.0)**2
    
    def pen_grad(x):
        return smooth_max_grad(x, n, 50000) + 20000.0 * (x.sum() - n/2.0)
    
    try:
        res = minimize(pen_obj, h, jac=pen_grad, bounds=bounds,
                      method='L-BFGS-B',
                      options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    
    return h, best_c
