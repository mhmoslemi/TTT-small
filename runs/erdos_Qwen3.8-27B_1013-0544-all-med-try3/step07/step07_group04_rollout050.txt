import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of c[lag] w.r.t. h, where lag is the actual lag (k - (n-1))"""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    return grad

def subgradient_descent(h, n, deadline, top_m=30, max_iters=5000):
    """Subgradient method with active set tracking"""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    # Warm start step size
    step = 0.01
    total_lags = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h_curr, n)
        if c.max() * (2.0/n) < best_c - 1e-14:
            best_c = compute_c5(h_curr, n)
            best_h = h_curr.copy()
        
        # Find top lags
        top_idx = np.argsort(c)[-top_m:]
        
        # Sum of gradients over active lags
        grad = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            # Only add if c[idx] is close to max (within tolerance)
            if c[idx] >= c.max() - 1e-8:
                grad += lag_gradient(h_curr, n, lag)
        
        # Also always include the argmax
        k_star = int(np.argmax(c))
        lag_star = k_star - (n - 1)
        grad += lag_gradient(h_curr, n, lag_star)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Barzilai-Borwein or fixed step
        h_new = h_curr - step * grad / grad_norm
        h_new = project_feasible(h_new, n)
        
        # Adjust step size based on progress
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            step = min(step * 1.1, 0.05)
        else:
            step *= 0.7
        
        # Check convergence
        if np.linalg.norm(h_new - h_curr) < 1e-10:
            break
        h_curr = h_new
    
    return best_h, best_c

from scipy.optimize import minimize
import time
import cvxpy as cp

    c = np.correlate(h, 1 - h, mode='full')

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_max(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _active_smooth(x, n, alpha, active_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[active_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_run(h, n, alpha, maxiter=3000, ftol=1e-15, active_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if active_idx is not None:
        obj = _active_smooth
        args = (n, alpha, active_idx)
    else:
        obj = _smooth_max
        args = (n, alpha)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def cvx_outer_step(h, n, trust):

    dx = 2.0 / n
    c = compute_all_lags(h, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if m == 0:
            # c[0] = sum h[i]*(1-h[i]), gradient: 1-2h[i]

        elif lag > 0:
            grad = np.zeros(n)
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
        else:
            grad = np.zeros(n)
            grad[:n-m] -= h[m:]
            grad[m:] += (1.0 - h[:n-m])
        cons.append(c[k] + dx * (grad @ (h_var - h)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)

            break

        return h.copy()
    h_new = h_var.value
    if h_new is None:

    return project_feasible(h_new, n)

For j = 0 (lag -(n-1)): result = h[n-1] * (1-h[0])  (just one term)
For j = 1 (lag -(n-2)): result = h[n-2]*(1-h[0]) + h[n-1]*(1-h[1])  (two terms)

So the lags near the edges have very few terms and thus small values. The "interesting" lags are in the middle (|lag| << n).

This means the max is likely achieved at a moderate lag (not 0, not n-1). For h with good mixing, the maximum should be at some intermediate lag.

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _active_smooth_obj(x, n, alpha, active_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[active_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_run(h, n, alpha, maxiter=2000, ftol=1e-14, active_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if active_idx is not None:
        obj = _active_smooth_obj
        args = (n, alpha, active_idx)
    else:
        obj = _smooth_max_obj
        args = (n, alpha)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    else:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    return grad

def cvx_outer_step(h, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        grad = _lag_grad(h, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.PDLP]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = h_var.value
                if h_new is not None:
                    return project_feasible(h_new, n)
                break
        except Exception:
            continue
    return h.copy()

def gradient_descent_step(h, n, top_m, step_size, deadline):

        alloc = min(280, (deadline - t0) * 0.30)
        h_opt, c_opt = optimize_focused(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    # Try other resolutions (interpolate from best)
    n_candidates = [400, 500, 350, 450]
    n_candidates = [x for x in n_candidates if x != n_init]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 50
        if remaining < 80:
            break
        alloc = min(120, remaining * 0.25)
        if alloc < 50:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_old, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_focused(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            if r % 4 == 0:
                hp = best_h + rng.normal(0, 0.02, n)
            elif r % 4 == 1:
                mask = (best_h > 0.15) & (best_h < 0.85)
                hp = best_h.copy()
                hp[mask] += rng.normal(0, 0.06, int(mask.sum()))
            elif r % 4 == 2:
                hp = best_h.copy()
                blk = max(1, n // 5)
                st = rng.randint(0, n - blk)
                hp[st:st + blk] += rng.uniform(-0.12, 0.12, blk)
            else:
                # Binary-ish perturbation
                hp = best_h.copy()
                idx = rng.choice(n, size=n // 5, replace=False)
                hp[idx] = np.where(hp[idx] > 0.5, 1.0, 0.0)

            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 35) * 0.45)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_focused(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish at best resolution
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_run(best_h, n, 800000, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 5:
            best_h, best_c = fine_local_search(best_h, n, deadline - 2)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)

    return best_h, best_c, len(best_h)
