def compute_effect_vectors(h, n, d):
    """For each index i, compute the vector of changes to c when h[i] += d.
    Returns array of shape (n, 2n-1)."""
    M = 2 * n - 1
    effects = np.zeros((n, M))
    for i in range(n):
        # Part 1: i as left index in c[m] = sum_k h[k](1-h[k+m-n+1])
        # m range: i+m-n+1 in [0,n-1] → m in [n-1-i, 2n-2-i]

            h_idx = i + m_range - n + 1
            effects[i, m_range] += d * (1.0 - h[h_idx])
        # Part 2: i as right index (k+m-n+1=i → k=i-m+n-1)
        # m range: i-m+n-1 in [0,n-1] → m in [i, i+n-1]
        m_lo2 = max(0, i)
        m_hi2 = min(M - 1, i + n - 1)
        if m_lo2 <= m_hi2:
            m_range2 = np.arange(m_lo2, m_hi2 + 1)
            h_idx2 = i - m_range2 + n - 1
            effects[i, m_range2] += -d * h[h_idx2]

    dx = 2.0 / n
    best_c = float(np.max(c) * dx)
    M = 2 * n - 1
    
    for d in deltas:

        effects = compute_effect_vectors(h, n, d)  # shape (n, M)
        
        for _pass in range(300):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] + d > 1.0 + 1e-12:
                    continue
                for j in range(i + 1, n):
                    if h[j] - d < -1e-12:
                        continue
                    if h[i] + d > 1.0 + 1e-12:
                        continue
                    # delta_c = effects[i] - effects[j]
                    new_max = float(np.max(c + effects[i] - effects[j]))
                    if new_max * dx < best_c - 1e-13:
                        h[i] += d
                        h[j] -= d

                        best_c = float(np.max(c) * dx)
                        improved = True
                        # Recompute effects (since h changed)
                        effects = compute_effect_vectors(h, n, d)
            
            if not improved:
                break
    
    return h, best_c
