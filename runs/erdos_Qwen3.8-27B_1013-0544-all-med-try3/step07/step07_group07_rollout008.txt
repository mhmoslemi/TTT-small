def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

def swap_delta(h, i, j, corr, n, m_range):
    """Compute delta for swapping h[i]:0→1 and h[j]:1→0"""
    delta = np.zeros(2 * n - 1, dtype=np.float64)

    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]

    msk2 = (imn >= 0) & (imn < n)

    jm = j + m_range

    jmn = j - m_range

    delta[n - 1] -= 2.0

    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    s_exp = exp_c.sum()
    if s_exp < 1e-300:
        idx = np.argmax(a)
        p = np.zeros(2 * n - 1)
        p[idx] = 1.0
    else:
        p = exp_c / s_exp
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-14:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def swap_delta(h, i, j, n, m_range):
    delta = np.zeros(2 * n - 1, dtype=np.float64)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk2 = (imn >= 0) & (imn < n)
    delta[msk2] -= h[imn[msk2]]
    jm = j + m_range
    msk3 = (jm >= 0) & (jm < n)
    delta[msk3] -= 1.0 - h[jm[msk3]]
    jmn = j - m_range
    msk4 = (jmn >= 0) & (jmn < n)
    delta[msk4] += h[jmn[msk4]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_local_search(hb, n, deadline, max_sweeps=100):
    dx = 2.0 / n
    h = hb.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list[:]:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = swap_delta(h, i, j, n, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def anneal_binary(hb, n, deadline, seed=42, n_iters=5000000, t_start=0.03):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = hb.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    T = t_start
    cool = 0.999985
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = swap_delta(h, i, j, n, m_range)
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(-50.0 * (nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def subgrad_polish(h, n, deadline, max_iters=500):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    corr = np.correlate(h, 1 - h, mode='full')
    top_k = 3
    step = 1.0 / n
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        c_vals = corr * (2.0 / n)
        idx = np.argsort(c_vals)[::-1][:top_k]
        max_val = c_vals[idx[0]]
        active = [k for k in idx if c_vals[k] > max_val - 1e-8]
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)
            i_min = max(0, k)
            i_max = min(n - 1, n - 1 + k)
            if i_min <= i_max:
                grad[i_min:i_max + 1] += (1.0 - h[i_min - k:i_max - k + 1])
            i_min2 = max(0, -k)
            i_max2 = min(n - 1, n - 1 - k)
            if i_min2 <= i_max2:
                grad[i_min2:i_max2 + 1] -= h[i_min2 + k:i_max2 + k + 1]
        grad /= len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        accepted = False
        for ls in range(40):
            h_new = project(best_h + step * (0.5 ** ls) * direction, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new
                step = min(step * (0.5 ** ls) * 1.5, 0.5)
                accepted = True
                break
        if not accepted:
            step *= 0.5
            if step < 1e-10:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    n = n0
    h_work = project(h0, n)
    best_h = h_work.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Initial PGD (15s)
    p1 = t0 + 15
    best_h, best_c = pgd_solve(best_h, n, [20, 200, 2000, 20000], p1, max_iter=3000)

    # Phase 2: Binary optimization (120s)
    p2 = t0 + 135
    hb = to_binary(best_h, n)
    hb, cb = binary_local_search(hb, n, p2 - 70, max_sweeps=50)
    # Anneal from binary
    ha, ca = anneal_binary(hb, n, p2, seed=seed, n_iters=4000000, t_start=0.04)
    if ca < cb:
        hb = ha.copy()
        cb = ca
    # Quick local search after anneal
    hb, cb = binary_local_search(hb, n, min(p2 + 5, p2), max_sweeps=10)

    # Phase 3: PGD from multiple starts (350s)
    p3 = t0 + 485
    h_starts = [best_h.copy(), hb.copy(), best_h * 0.8 + hb * 0.2]
    for hs in h_starts:
        hs[::7] += 0.01 * np.sin(np.arange(len(hs)) * 0.3)
        h_starts[-1] = project(hs, n)
        break
    alphas3 = [200, 2000, 20000, 200000, 2000000]
    best_h, best_c = pgd_multi_start([best_h, hb, project(hb * 0.5 + best_h * 0.5, n)],
                                       n, alphas3, p3, max_iter=3000)

    # Phase 4: Higher resolution (250s)
    p4 = t0 + 735
    resolutions = [500, 640, 800]
    for nh in resolutions:
        if time.time() > p4 - 50:
            break
        sub_dl = p4 - 40
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project(hh, nh)
        # PGD at new resolution
        hh, ch = pgd_solve(hh, nh, [500, 5000, 50000, 500000], sub_dl - 80, max_iter=2500)
        # Binary refinement
        hb_nh = to_binary(hh, nh)
        hb_nh, cb_nh = binary_local_search(hb_nh, nh, sub_dl - 40, max_sweeps=20)
        if time.time() < sub_dl - 20:
            ha_nh, ca_nh = anneal_binary(hb_nh, nh, sub_dl - 15, seed=seed + nh,
                                         n_iters=800000, t_start=0.03)
            if ca_nh < cb_nh:
                hb_nh, cb_nh = ha_nh, ca_nh
        # PGD from binary
        hh2 = project(hb_nh + 0.01 * np.random.RandomState(seed + nh + 1).randn(nh) * 0.1, nh)
        hh2, ch2 = pgd_solve(hh2, nh, [1000, 10000, 100000, 1000000], sub_dl, max_iter=2000)
        if ch2 < best_c:
            best_c = ch2
            best_h = hh2.copy()
            n = nh

    # Phase 5: Subgradient polish at best resolution (100s)
    p5 = dl - 8
    if time.time() < p5 - 20:
        best_h, best_c = subgrad_polish(best_h, n, p5 - 30, max_iters=1000)

    # Phase 6: Final PGD with very high alpha
    if time.time() < dl - 5:
        alphas_final = [50000, 500000, 5000000]
        best_h, best_c = pgd_solve(best_h, n, alphas_final, dl - 2, max_iter=2000)

    # Final evaluation
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
