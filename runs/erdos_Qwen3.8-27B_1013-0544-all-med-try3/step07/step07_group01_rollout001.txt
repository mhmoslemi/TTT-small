        if time.time() > deadline:
            break
        progress = min(1.0, (time.time() - t0) / total_time)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        y = h + beta * v
        y = np.clip(y, 0.0, 1.0)
        g = grad_smooth_max(y, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        lr = lr_init / (1.0 + 0.5 * g_norm)
        
        v_new = beta * v - lr * g
        h_new = h + v_new
        h_new = project_feasible(h_new, n)
        v = v_new
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if time.time() > deadline:
                        return h, best_c
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def local_pair_fine(h, n, deadline, center, radius=20, deltas=None):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        while time.time() < deadline:
            improved = False
            lo = max(0, center - radius)
            hi = min(n, center + radius + 1)
            for i in range(lo, hi):
                if time.time() > deadline:
                    break
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                if time.time() > deadline:
                    break
            if not improved:
                break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def try_resolution(h_best, n_best, c_best, n_try, deadline, rng):
    x_old = np.linspace(0, 2, n_best)
    x_new = np.linspace(0, 2, n_try)
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_feasible(h_interp, n_try)
    
    t_allow = deadline - time.time() - 10
    if t_allow < 30:
        return h_best, c_best, n_best
    
    t1 = time.time() + t_allow * 0.4
    h_t, c_t = lbfgs_optimize(h_interp, n_try, min(t1, deadline - 60), max_iters=200)
    
    if time.time() < deadline - 80:
        t2 = time.time() + min(60, deadline - time.time() - 80)
        h_t, c_t = slsqp_optimize(h_t, n_try, t2, alpha=500000, maxiter=1000)
    
    if time.time() < deadline - 40:
        t3 = deadline - 20
        h_t, c_t = fine_pair_search(h_t, n_try, t3,
                                     [0.005, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=50)
    
    if c_t < c_best:
        return h_t, c_t, n_try
    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    
    # Phase 1: Aggressive L-BFGS with multiple alpha values
    t1 = min(t0 + 200, deadline - 400)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_iters=250)
    
    # Phase 2: SLSQP with multiple alphas
    if time.time() < deadline - 300:
        for a in [200000, 1000000, 10000000]:
            if time.time() > deadline - 280:
                break
            t2 = min(time.time() + 40, deadline - 260)
            h2, c2 = slsqp_optimize(best_h, best_n, t2, alpha=a, maxiter=1500)
            if c2 < best_c:
                best_c = c2
                best_h = h2
    
    # Phase 3: Nesterov momentum polish
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 180)
        h3, c3 = nesterov_polish(best_h, best_n, t3, lr_init=0.01, beta=0.8,
                                  alpha_start=10000, alpha_end=50000000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Fine pair search (local around peaks)
    if time.time() < deadline - 120:
        c_vec = compute_c5_vec(best_h, best_n)
        top_idx = np.argsort(c_vec)[-5:]
        for ti in top_idx:
            if time.time() > deadline - 100:
                break
            center = ti - (best_n - 1)
            c_idx = max(0, min(best_n - 1, center))
            t4 = min(time.time() + 20, deadline - 80)
            best_h, best_c = local_pair_fine(best_h, best_n, t4, c_idx, radius=30)
    
    # Phase 5: Global fine pair search
    if time.time() < deadline - 60:
        t5 = min(time.time() + 80, deadline - 40)
        best_h, best_c = fine_pair_search(best_h, best_n, t5,
                                           [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                           radius=80)
    
    # Phase 6: Try other resolutions
    for n_try in [200, 250, 350, 400, 500]:
        if time.time() > deadline - 50:
            break
        if n_try == best_n:
            continue
        best_h, best_c, best_n = try_resolution(
            best_h, best_n, best_c, n_try, deadline - 30, rng)
    
    # Phase 7: Final L-BFGS polish
    if time.time() < deadline - 15:
        t7 = deadline - 10
        best_h, best_c = lbfgs_optimize(best_h, best_n, t7, max_iters=100)
    
    # Phase 8: Ultra-fine pair search
    if time.time() < deadline - 3:
        t8 = deadline - 1
        best_h, best_c = fine_pair_search(best_h, best_n, t8,
                                           [0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                           radius=40)
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
