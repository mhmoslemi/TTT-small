def find_optimal_delta(corr, d, lo, hi):
    """Find δ in [lo, hi] that minimizes max(corr + δ*d)."""
    # Evaluate at boundaries
    f_lo = np.max(corr + lo * d)
    f_hi = np.max(corr + hi * d)
    
    # Find crossing of top-2 terms
    # Sort by corr value (descending)
    # The top terms are most likely to be active
    top_k = 10  # consider top-k terms
    idx = np.argsort(corr)[::-1][:top_k]
    
    best_delta = 0.0
    best_val = np.max(corr)  # at δ=0
    
    # Check δ=0
    if f_lo < best_val:
        best_val = f_lo; best_delta = lo
    if f_hi < best_val:
        best_val = f_hi; best_delta = hi
    
    # Check crossing points of top terms
    for a in range(len(idx)):
        for b in range(a+1, len(idx)):
            ia, ib = idx[a], idx[b]
            if abs(d[ia] - d[ib]) < 1e-15:
                continue
            delta = (corr[ia] - corr[ib]) / (d[ib] - d[ia])
            if lo <= delta <= hi:
                val = np.max(corr + delta * d)
                if val < best_val:
                    best_val = val
                    best_delta = delta
    
    return best_delta, best_val
