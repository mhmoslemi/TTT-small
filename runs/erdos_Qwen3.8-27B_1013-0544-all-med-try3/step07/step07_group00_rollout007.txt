def build_lp(h, n, top_k=None):
    """Build LP matrices for the linearized min-max problem."""
    dx = 2.0 / n
    c_vec = np.correlate(h, 1 - h, mode='full') * dx
    
    if top_k is not None:
        # Only include top_k shifts
        active = np.argsort(-c_vec)[:top_k]
    else:
        active = range(2 * n - 1)
    
    n_active = len(active)
    n_vars = n + 1  # h_0..h_{n-1}, t
    
    A = np.zeros((n_active, n_vars))
    b = np.zeros(n_active)
    
    for idx in active:
        k = idx - (n - 1)
        # Build gradient of c_k at h
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
        
        A[idx, :n] = grad
        A[idx, n] = -1.0
        b[idx] = c_vec[idx] - np.dot(grad, h)  # = dx * R_k
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0  # minimize t
    
    bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
    
    return c_obj, A, b, A_eq, b_eq, bounds

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_grad_matrix(h, n):
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        k = j - (n-1)
        if k >= 0:
            nv = n - k
            G[j, 0:nv] = (1.0 - h[k:n]) * dx
            G[j, k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            G[j, kk:n] = (1.0 - h[0:nv]) * dx
            G[j, 0:nv] += (-h[kk:n]) * dx
    return G

def lp_step(h, n, num_shifts=None, time_limit=5.0):

    t_start = time.time()
    c_vec = compute_c5_vec(h, n)
    G = build_grad_matrix(h, n)
    b = c_vec - G @ h  # should equal dx * autocorr
    
    if num_shifts is not None and num_shifts < 2*n-1:
        active = np.argsort(-c_vec)[:num_shifts]
        G_sel = G[active]
        b_sel = b[active]
    else:
        G_sel = G
        b_sel = b
    
    n_a = len(b_sel)
    n_vars = n + 1
    
    A = np.hstack([G_sel, -np.ones((n_a, 1))])
    c_obj = np.zeros(n_vars)

    A_eq[0, :n] = 1.0

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    G = build_grad_matrix(x, n)
    grad = G.T @ w  # (n,)
    grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 10.0), (500, 100.0), (2000, 1000.0),
                   (5000, 5000.0), (10000, 50000.0), (50000, 500000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
