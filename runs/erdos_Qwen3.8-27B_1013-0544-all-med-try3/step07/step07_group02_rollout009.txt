    for n_try in [800, 1000]:
        if time.time() > phase3_deadline - 80:
            break
        remaining = phase3_deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.4)
        if alloc < 40:
            break
        
        h_interp = interpolate_solution(best_h, len(best_h), n_try)
        # Binary SA at higher resolution
        h_bin = binarize(h_interp, n_try)
        T0 = 0.005
        h_sa, c_sa = binary_sa(h_bin, n_try, T0, time.time() + alloc * 0.4, rng, max_iters=30000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
        
        # Continuous at higher resolution
        if time.time() < phase3_deadline - 30:
            h_cont = project_feasible(h_sa * 0.99 + 0.005, n_try)
            h_pg, c_pg = projected_gradient(h_cont, n_try, 200, time.time() + alloc * 0.3, lr_init=0.3, momentum=0.85)
            if c_pg < best_c:
                best_c = c_pg
                best_h = h_pg
    
    # Phase 4: Final polish (remaining time)
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        
        # Final SLSQP
        if time.time() < deadline - 20:
            h_new = slsqp_smooth(best_h, n, 10000, maxiter=2000, top_m=5)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Final local search
        if time.time() < deadline - 3:
            best_h, best_c = multi_pair_local_search(best_h, n, deadline - 1, rng, max_rounds=30)
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_base, 0.5), n_base)
        best_c = compute_c5(best_h, n_base)
    
    return best_h, best_c, len(best_h)
