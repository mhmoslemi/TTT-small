    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')  # length 2n-1
    a = c * dx  # objective values at each shift
    
    # Softmax weights
    a_shifted = alpha * a
    a_max = a_shifted.max()
    e = np.exp(a_shifted - a_max)
    p = e / e.sum()  # softmax weights, length 2n-1
    
    # Gradient of smooth max: sum_k p_k * grad(a_k)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if p[k] < 1e-12:
            continue
        shift = k - (n - 1)  # the actual shift
        # a_k = dx * sum_i h[i] * (1-h[i+shift])
        # d a_k / d h[j] = dx * [(1-h[j+shift]) - h[j-shift]] (with bounds)

                val += (1.0 - h[j + shift])
            if 0 <= j - shift < n:
                val -= h[j - shift]
            grad[j] += p[k] * dx * val
    
    obj = (a_max / alpha + np.log(e.sum()) / alpha)  # smooth max
    return obj, grad

    a_max = np.max(a_scaled)

def project_feasible(h, n):

def smooth_max_grad(h, n, alpha):
    """Compute smooth max of correlation values and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    a = c * dx  # values we're taking max of
    
    a_scaled = alpha * a
    amax = a_scaled.max()
    exp_a = np.exp(a_scaled - amax)
    Z = exp_a.sum()
    p = exp_a / Z
    
    obj = (amax + np.log(Z)) / alpha
    
    # Gradient
    grad = np.zeros(n)
    # Only process significant entries

        if pk < thresh:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-15
            total_room = room[mask].sum()
            if total_room < 1e-14:
                break
            h[mask] += diff * room[mask] / total_room
        else:
            mask = h > 1e-15
            total_h = h[mask].sum()
            if total_h < 1e-14:
                break
            h[mask] += diff * h[mask] / total_h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    a = c * dx
    a_s = alpha * a
    amax = a_s.max()
    e = np.exp(a_s - amax)
    Z = e.sum()
    p = e / Z
    obj = (amax + np.log(Z)) / alpha
    grad = np.zeros(n)
    active = np.where(p > 1e-7)[0]
    for k in active:
        pk = p[k]
        s = k - (n - 1)
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += pk * dx * (1.0 - h[s:])
                grad[s:] -= pk * dx * h[:ns]
        else:
            ss = -s
            ns = n - ss
            if ns > 0:
                grad[ss:] += pk * dx * (1.0 - h[:ns])
                grad[:ns] -= pk * dx * h[ss:]
    return obj, grad

def _fun_lbfgs(x, n, alpha, mu):
    obj, g = smooth_max_val_grad(x, n, alpha)
    sd = x.sum() - n / 2.0
    obj += mu * sd * sd
    g = g + 2.0 * mu * sd
    return obj

def _jac_lbfgs(x, n, alpha, mu):
    _, g = smooth_max_val_grad(x, n, alpha)
    sd = x.sum() - n / 2.0
    g = g + 2.0 * mu * sd
    return g

def run_lbfgs(h, n, alpha, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_fun_lbfgs, h, jac=_jac_lbfgs, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'iprint': -1, 'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coord_descent(h, n, deadline, max_passes=500):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h, best_c
            c = np.correlate(h, 1.0 - h, mode='full')
            B = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                s = k - (n - 1)
                b = 0.0
                if 0 <= j + s < n:
                    b += (1.0 - h[j + s])
                if 0 <= j - s < n:
                    b -= h[j - s]
                B[k] = b * dx
            A = c - B * h[j]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(A + B * m1) <= np.max(A + B * m2):
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) * 0.5
            old = h[j]
            h[j] = t_opt
            new_c = compute_c5(h, n)
            if new_c < best_c - 1e-16:
                best_c = new_c
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h, best_c

def pair_refine(h, n, deadline, deltas, max_passes=100, radius=None):
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n
    for d in deltas:
        if time.time() > deadline:
            break
        for _p in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if j <= i:
                        continue
                    hi_val = h[i] + d
                    hj_val = h[j] - d
                    if 0.0 <= hi_val <= 1.0 and 0.0 <= hj_val <= 1.0:
                        h[i], h[j] = hi_val, hj_val
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = h[i] - d, h[j] + d
                    hi_val2 = h[i] - d
                    hj_val2 = h[j] + d
                    if 0.0 <= hi_val2 <= 1.0 and 0.0 <= hj_val2 <= 1.0:
                        h[i], h[j] = hi_val2, hj_val2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = h[i] + d, h[j] - d
            if not improved:
                break
    return h, best_c

def optimize_at_n(h, n, deadline, mu):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)

    alphas = [200, 500, 1500, 5000, 15000, 50000, 150000]
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        remaining = deadline - time.time() - 15
        alloc_iter = min(3000, int(remaining * 30))
        if alloc_iter < 100:
            break
        ht = run_lbfgs(best_h, n, alpha, mu, maxiter=alloc_iter)
        ct = compute_c5(ht, n)
        if ct < best_c - 1e-15:
            best_c = ct
            best_h = ht.copy()

    if time.time() < deadline - 10:
        best_h, best_c = coord_descent(best_h, n, deadline - 5, max_passes=300)

    return best_h, best_c
