    else:

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    
    try:
        res = minimize(obj_func, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search_refine(h, n, deadline, max_rounds=200):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = 0
        
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(0, n - m):
                if time.time() > deadline:
                    break
                j = i + m if lag > 0 else i - m
                if j < 0 or j >= n:
                    continue
                
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.5)
                else:
                    delta = -min(d, (h[j] - h[i]) * 0.3, h[j] * 0.5)
                
                if abs(delta) < 1e-12:
                    continue
                if h[i] + delta < -1e-12 or h[i] + delta > 1 + 1e-12:
                    continue
                if h[j] - delta < -1e-12 or h[j] - delta > 1 + 1e-12:
                    continue
                
                h[i] += delta
                h[j] -= delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved += 1
                    c_all = compute_all_lags(h, n)
                    k_star = int(np.argmax(c_all))
                    lag = k_star - (n - 1)
                    if lag == 0:
                        return best_h, best_c
                    m = abs(lag)
                    if improved >= 30:
                        break
                else:
                    h[i] -= delta
                    h[j] += delta
        
        if improved == 0:
            # Try broader: all top lags
            found_any = False
            for k2 in np.argsort(c_all)[-8:]:
                if time.time() > deadline:
                    break
                lag2 = int(k2) - (n - 1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for trial_d in [0.003, 0.001]:
                    if time.time() > deadline:
                        break
                    for i in range(0, n - m2):
                        if time.time() > deadline:
                            break
                        j2 = i + m2
                        if h[i] > h[j2]:
                            delta = min(trial_d, (h[i]-h[j2])*0.2, h[i]*0.4)
                        else:
                            delta = -min(trial_d, (h[j2]-h[i])*0.2, h[j2]*0.4)
                        if abs(delta) < 1e-12:
                            continue
                        h[i] += delta
                        h[j2] -= delta
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            found_any = True
                            c_all = compute_all_lags(h, n)
                        else:
                            h[i] -= delta
                            h[j2] += delta
                    if found_any:
                        break
            if not found_any:
                break
    
    return best_h, best_c

def multi_delta_local(h, n, deadline):
    """Try various transfer amounts between pairs at the worst lag."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(100000):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        
        # Find the pair with most "room for improvement"
        if lag > 0:
            contrib = h[0:n-m] * (1 - h[m:n])
        else:
            contrib = h[m:n] * (1 - h[0:n-m])
        
        # Transfer from high h to low h at the worst lag
        if lag > 0:
            for i in range(n - m):
                j = i + m
                if h[i] > 0.3 and h[j] < 0.7:
                    delta = min(h[i] - h[j], 0.02) * 0.5
                    if delta > 1e-6:
                        h[i] -= delta
                        h[j] += delta
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                        else:
                            h[i] += delta
                            h[j] -= delta
        
        if it % 1000 == 999:
            h = best_h.copy()
    
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    
    # Load initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    # Phase 1: Binary SA at n=500 (2 min)
    sa_deadline = t0 + min(120, budget_s * 0.15)
    h_bin, c_bin = binary_sa(500, sa_deadline, seed=seed)
    
    # Also try to get a good solution from the initial
    if h_init is not None:
        h_init_proj = project_feasible(h_init, n_init)
        c_init = compute_c5(h_init_proj, n_init)
    else:
        c_init = float('inf')
    
    # Phase 2: Continuous optimization from best binary (5 min)
    # and from initial (in parallel, sequentially)
    
    candidates = []
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    candidates.append(('bin', h_bin.copy(), 500))
    
    if h_init is not None:
        candidates.append(('init', h_init_proj.copy(), n_init))
    
    for name, h_start, n_start in candidates:
        if time.time() > deadline - 30:
            break
        phase_dl = min(deadline - 20, time.time() + 180)
        h_opt, c_opt = adaptive_gradient_opt(h_start, n_start, phase_dl)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))
    
    # Phase 3: SLSQP polishing on best solution (3 min)
    if best_h is not None and time.time() < deadline - 30:
        n_best = len(best_h)
        slsqp_dl = min(deadline - 20, time.time() + 120)
        for top_m in [3, 5]:
            if time.time() > slsqp_dl - 10:
                break
            h_new = slsqp_polish(best_h, n_best, slsqp_dl, top_m=top_m, alpha=300000, maxiter=1500)
            c_new = compute_c5(h_new, n_best)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 4: Fine local search (5 min)
    if best_h is not None and time.time() < deadline - 20:
        n_best = len(best_h)
        ls_dl = min(deadline - 10, time.time() + 240)
        h_opt, c_opt = local_search_refine(best_h, n_best, ls_dl)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 5: Multi-delta local search
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        md_dl = min(deadline - 5, time.time() + 60)
        h_opt, c_opt = multi_delta_local(best_h, n_best, md_dl)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 6: Try different resolutions
    if best_h is not None and time.time() < deadline - 40:
        n_cur = len(best_h)
        for n_try in [400, 600, 512]:
            if time.time() > deadline - 30:
                break
            if n_try == n_cur:
                continue
            h_up = upsample(best_h, n_cur, n_try)
            c_up = compute_c5(h_up, n_try)
            phase_dl = min(deadline - 20, time.time() + 90)
            h_opt, c_opt = adaptive_gradient_opt(h_up, n_try, phase_dl)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            if time.time() < deadline - 15:
                h_opt2, c_opt2 = local_search_refine(best_h, len(best_h), min(deadline - 5, time.time() + 30), max_rounds=50)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2
    
    # Phase 7: Final polish
    if best_h is not None:
        n_best = len(best_h)
        if time.time() < deadline - 15:
            h_new = slsqp_polish(best_h, n_best, min(deadline - 5, time.time() + 40), top_m=3, alpha=1000000, maxiter=1000)
            c_new = compute_c5(h_new, n_best)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        if time.time() < deadline - 3:
            h_opt, c_opt = local_search_refine(best_h, n_best, deadline - 1, max_rounds=50)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))
    
    if best_h is None:
        best_h = np.full(500, 0.5)
        best_c = compute_c5(best_h, 500)
    
    return best_h, best_c, len(best_h)
