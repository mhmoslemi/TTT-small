import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h to satisfy 0 <= h <= 1 and sum(h) = n/2."""
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Compute gradient of c[lag] w.r.t. h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # c[lag] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # d/dh[i] for i < n-lag: (1 - h[i+lag])
        # d/dh[i] for i >= lag: -h[i-lag]
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += -h[:n-m]
    else:
        # lag < 0, m = -lag

        h_new = project_feasible(h_new, n)

                    break

                if lr < 0.01:
                    break

                no_improve_count += 1
                if no_improve_count > 500:
                    break
                continue
            i_best = inc_candidates[np.argmax(h[curr[inc_candidates] - 0.5])]  # closest to 1
            j_best = dec_candidates[np.argmin(h_curr[dec_candidates] - 0.5)]  # closest to 0
            
            # Hmm this is getting complicated. Let me just handle lag 0 by pushing
            # toward binary.
            # Find i with h most in (0.5, 1) and j with h most in (0, 0.5)
            i_best = int(np.argmax(h_curr * (h_curr > 0.5) * (h_curr < 1 - 1e-10)))
            j_best = int(np.argmin(h_curr * (h_curr < 0.5) * (h_curr > 1e-10)))
            
            if h_curr[i_best] <= 0.5 + 1e-10 or h_curr[j_best] >= 0.5 - 1e-10:
                no_improve_count += 1
                if no_improve_count > 500:
                    break
                continue
            
            delta = min(step, (1 - h_curr[i_best]) * 0.5, h_curr[j_best] * 0.5,
                       h_curr[i_best] - 0.5, 0.5 - h_curr[j_best])
            if delta < 1e-12:
                step = max(step * 0.7, min_step)
                continue
            
            h_curr[i_best] += delta
            h_curr[j_best] -= delta
            h_curr = project_feasible(h_curr, n)
            
            c_check = compute_c5(h_curr, n)
            if c_check < best_c - 1e-15:
                best_c = c_check
                best_h = h_curr.copy()
                no_improve_count = 0
            else:
                h_curr[i_best] -= delta
                h_curr[j_best] += delta
                h_curr = project_feasible(h_curr, n)
                no_improve_count += 1
                step *= 0.85
                if step < min_step:
                    break
            continue
        
        # For non-zero lag: compute gradient and find best pair
        g = lag_gradient(h_curr, n, lag_star)
        
        # Find i (where we want to increase h) and j (where we want to decrease h)
        # Increasing h[i] reduces c[lag] if g[i] < 0
        # Decreasing h[j] reduces c[lag] if g[j] > 0
        # So: i = argmin g (most negative, where increasing h helps)
        #     j = argmax g (most positive, where decreasing h helps)
        valid_i = g.copy()
        valid_i[h_curr >= 1 - 1e-10] = np.inf  # Can't increase
        valid_j = g.copy()
        valid_j[h_curr <= 1e-10] = -np.inf  # Can't decrease
        
        i_best = int(np.argmin(valid_i))  # Most negative gradient → increase h
        j_best = int(np.argmax(valid_j))  # Most positive gradient → decrease h
