def pair_search(h, n, deadline, deltas, radius=None):

    best_c = compute_c5(h, n)
    h = h.copy()

        while time.time() < deadline:
            improved = False
            n_iter = 0

                else:

                    hi = min(n, i + radius + 1)
                    j_range = [j for j in range(lo, hi) if j != i]
                
                for j in j_range:
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room

            h = h * (target / s)

            break
        h_new = slsqp_pow(h, n, time.time() + 40, p=p)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    
    if time.time() < deadline - 10:
        h_new = slsqp_lse(h, n, time.time() + 30, alpha=10000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    
    if n <= 250 and time.time() < deadline - 5:
        h, best_c = pair_search(h, n, deadline - 2,
                                [0.01, 0.005, 0.002, 0.001, 0.0005])
    elif time.time() < deadline - 5:
        h, best_c = pair_search(h, n, deadline - 2,
                                [0.01, 0.005, 0.002, 0.001], radius=40)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    
    n = n_init
    best_h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Optimize from parent (200s)
    p1_end = t0 + min(200, budget_s * 0.22)
    best_h, best_c = optimize_from(best_h, n, p1_end, rng)
    
    # Phase 2: Simulated annealing (100s)
    if time.time() < deadline - 150:
        sa_end = min(time.time() + 100, deadline - 150)
        h_sa, c_sa = simulated_anneal(best_h, n, sa_end, t_start=0.03, t_end=0.0005, n_steps=80000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
    
    # Phase 3: Fine pair search (80s)
    if time.time() < deadline - 100:
        ps_end = min(time.time() + 80, deadline - 100)
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, ps_end,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = pair_search(best_h, n, ps_end,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=50)
    
    # Phase 4: Alternative starting points (200s)
    for alt_seed in [seed + 100, seed + 200, seed + 300]:
        if time.time() > deadline - 120:
            break
        
        # Binary spread
        h_bin = gen_binary_spread(n, alt_seed)
        kernel = np.ones(7) / 7.0
        h_smooth = np.convolve(h_bin, kernel, mode='same')
        h_smooth = project_feasible(h_smooth, n)
        
        alloc_end = min(time.time() + 55, deadline - 80)
        h_alt, c_alt = optimize_from(h_smooth, n, alloc_end, rng)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt
    
    # Phase 5: Different resolutions (150s)
    for n_try in [100, 150, 200, 300, 400]:
        if time.time() > deadline - 60:
            break
        if n_try == n:
            continue
        
        if n_try < n:
            idx = np.linspace(0, n, n_try).astype(int)
            h_try = best_h[idx].copy()
        else:
            h_try = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n), best_h)
        h_try = project_feasible(h_try, n_try)
        
        alloc_end = min(time.time() + 40, deadline - 40)
        h_alt, c_alt = optimize_from(h_try, n_try, alloc_end, rng)
        
        if n_try <= 200 and time.time() < deadline - 30:
            h_alt, c_alt = pair_search(h_alt, n_try, min(time.time() + 15, deadline - 25),
                                       [0.005, 0.002, 0.001, 0.0005])
        
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt
            n = n_try
    
    # Phase 6: SLSQP polish on best (60s)
    if time.time() < deadline - 30:
        polish_end = min(time.time() + 50, deadline - 25)
        for p in [200, 500]:
            if time.time() > deadline - 25:
                break
            h_new = slsqp_pow(best_h, n, time.time() + 25, p=p)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 7: Ultra-fine pair search (30s)
    if time.time() < deadline - 5:
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                                         [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                                         [0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=40)
    
    # Phase 8: Final L-BFGS polish
    if time.time() < deadline - 3:
        h_new = l_bfgs_pen(best_h, n, deadline - 1, p=200, penalty=10000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 9: Final ultra-fine
    if time.time() < deadline - 1:
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                         [0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                         [0.0005, 0.0002, 0.0001], radius=30)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
