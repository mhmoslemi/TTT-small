def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (abs(diff) / tot) * h
    return np.clip(h, 0.0, 1.0)


def slsqp_topk(h, n, alpha, k_top, maxiter=5000, ftol=1e-16):
    dx = 2.0 / n
    target = n / 2.0

    def objective(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if k_top < len(c):
            idx = np.argpartition(-c, k_top)[:k_top]
            top = c[idx]
        else:
            top = c
        m = top.max()
        shifted = alpha * (top - m)
        mx = shifted.max()
        if mx > 300:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    def con_fun(x):
        return (x.sum() - target,)

    def con_jac(x):
        return np.ones(1, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
