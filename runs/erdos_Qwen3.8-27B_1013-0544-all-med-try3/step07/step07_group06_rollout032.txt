2. Solve: min t s.t. corr(k) <= t for active lags k, 0<=h<=1, sum(h)=n/2

- The linearized constraint is: corr(k, h*) + ∇corr(k) * (h - h*) <= t
- This gives a linear constraint in h and t.
- So the linearized problem is a LP!

**Iterative Linearization (Cutting Plane) Approach:**
1. Start with h0.
2. Compute active lags.
3. Solve LP: min t s.t. corr(k, h*) + ∇corr(k, h*)·(h-h*) <= t for active k, 0<=h<=1, sum(h)=n/2.
4. Update h.
5. Recompute active lags.
6. Repeat until convergence.

This is essentially the ellipsoid method or cutting plane method for the non-convex problem. It should converge to a local optimum.

The LP can be solved efficiently with scipy or cvxpy. For n=250, it's a 251-variable LP with a few hundred constraints, which should be very fast.

```python
def cutting_plane(h, n, deadline, max_iter=1000):
    """Iterative linearization for the min-max problem."""
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_max = c.max()
        
        if cur_max < best_c:
            best_c = cur_max
            best_h = h.copy()
        
        # Find active constraints (within tolerance of max)
        tol = 1e-6
        active = np.where(c >= cur_max - tol)[0]
        
        # Compute gradients of correlation at active lags
        # d/dh[a] of corr_full[p] where p is the full index
        # = dx * [(1-h[a+k]) * [a+k in [0,n)] - h[a-k] * [a-k in [0,n)]]
        # where k = p - (n-1)
        
        # Build constraint matrix A and vector b for the LP
        # min t
        # s.t. for each active lag p: sum_a A[p,a] * h[a] <= b[p] + t
        #      where A[p,a] = dx * [(1-h[a+k]) * [valid] - h[a-k] * [valid]]
        #      b[p] = c[p] - sum_a A[p,a] * h[a]  (the linearization offset)
        #      0 <= h <= 1
        #      sum(h) = n/2
        
        # Actually, let's set up the LP as:
        # min t
        # s.t. A @ h - t <= b  (for each active constraint)
        #      A_h @ h = n/2  (sum constraint)
        #      0 <= h <= 1
        
        n_constraints = len(active)
        A = np.zeros((n_constraints, n))
        b = np.zeros(n_constraints)
        
        for idx, p in enumerate(active):
            k = p - (n - 1)
            grad = np.zeros(n)
            for a in range(n):
                # (1-h[a+k]) if a+k in [0,n)
                if k >= 0:
                    if a + k < n:
                        grad[a] += (1 - h[a+k])
                    if a - k >= 0:
                        grad[a] -= h[a-k]
                else:
                    if a + k >= 0:
                        grad[a] += (1 - h[a+k])
                    if a - k < n:
                        grad[a] -= h[a-k]
            A[idx] = dx * grad
            b[idx] = c[idx] - dx * np.dot(grad, h)
            # Verify: corr_linearized = c[idx] + dx * dot(grad, h - h) = c[idx] ✓
        
        # Solve LP
        from scipy.optimize import linprog
        # Variables: [h[0], ..., h[n-1], t]
        # Objective: minimize t -> c = [0, ..., 0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A @ h - t <= b
        # In matrix form: [A | -I] @ [h; t] <= b
        A_ub = np.hstack([A, -np.eye(n_constraints)])
        b_ub = b
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds_h = [(0, 1)] * n + [(cur_max - 0.1, None)]  # t >= some lower bound
        # Actually t >= 0 is enough
        bounds_h = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_h, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            t_new = res.x[-1]
            # Check actual objective
            c_new = np.correlate(h_new, 1 - h_new, mode='full').max() * dx
            if c_new < cur_max + 1e-10:
                h = h_new
            else:
                # The LP solution doesn't actually improve (due to non-convexity)
                # Take a step towards it
                h = 0.5 * h + 0.5 * h_new
                h = project_feasible_fast(h, n)
        else:
            break
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n, p):
    dx = 2.0 / n
    k = p - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        idx1 = np.arange(0, n - k)
        grad[idx1] += (1.0 - h[idx1 + k])
        idx2 = np.arange(k, n)
        grad[idx2] -= h[idx2 - k]
    else:
        ka = -k
        idx1 = np.arange(ka, n)
        grad[idx1] += (1.0 - h[idx1 + k])
        idx2 = np.arange(0, n - ka)
        grad[idx2] -= h[idx2 - k]
    return dx * grad

def cutting_plane(h, n, deadline, max_iter=100000, max_constraints=300):
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    constraint_lags = []
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c = compute_corr(h, n)
        cur_max = c.max()
        
        if cur_max < best_c - 1e-13:
            best_c = cur_max
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 200:
            # Perturb and reset constraints
            h = project_feasible(h + 0.02 * np.random.randn(n), n)
            constraint_lags = []
            stagnant = 0
            continue
        
        # Find active lags
        tol = max(1e-9, 0.005 * cur_max)
        active = np.where(c >= cur_max - tol)[0]
        
        # Accumulate constraints
        existing = set(constraint_lags)
        for p in active:
            existing.add(p)
        constraint_lags = list(existing)
        
        if len(constraint_lags) > max_constraints:
            scores = np.array([c[p] for p in constraint_lags])
            keep = np.argsort(scores)[-max_constraints:]
            constraint_lags = [constraint_lags[i] for i in keep]
        
        n_con = len(constraint_lags)
        if n_con == 0:
            continue
        
        # Build LP
        A_mat = np.zeros((n_con, n))
        b_vec = np.zeros(n_con)
        
        for idx, p in enumerate(constraint_lags):
            g = compute_grad(h, n, p)
            A_mat[idx] = g
            b_vec[idx] = c[p] - np.dot(g, h)
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([A_mat, -np.eye(n_con)])
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_vec, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs',
                          options={'presolve': True})
        except Exception:
            break
        
        if not res.success:
            break
        
        h_lp = res.x[:n]
        
        # Check if LP solution actually improves
        c_lp = np.max(np.correlate(h_lp, 1 - h_lp, mode='full')) * dx
        if c_lp <= cur_max + 1e-12:
            h = h_lp
        else:
            # Damped step
            improved = False
            for alpha in [0.5, 0.3, 0.15, 0.05]:
                h_try = (1 - alpha) * h + alpha * h_lp
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < cur_max - 1e-13:
                    h = h_try
                    improved = True
                    break
            if not improved:
                # Small random perturbation to escape
                h = project_feasible(h + 0.005 * np.random.randn(n), n)
                constraint_lags = []
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h_init, n, deadline, max_sweeps=2000, n_restarts=5, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    rng = np.random.RandomState(seed)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            n_swaps = rng.randint(3, 8)
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            if len(oi) > 0 and len(zi) > 0:
                for _ in range(n_swaps):
                    a = oi[rng.randint(len(oi))]
                    b = zi[rng.randint(len(zi))]
                    h[a], h[b] = 0.0, 1.0
                    oi = np.where(h == 1)[0]
                    zi = np.where(h == 0)[0]
            corr = np.correlate(h, 1 - h, mode='full')
        
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            rng.shuffle(zeros)
            
            for i in zeros:
                if improved:
                    break
                for j in ones:
                    if i == j:
                        continue
                    # Compute delta correlation
                    delta = np.zeros(2 * n - 1)
                    # i goes 0->1
                    pi1 = max(n - 1 - i, 0)
                    pi2 = min(2 * n - 2 - i, 2 * n - 2)
                    if pi1 <= pi2:
                        idx1 = np.arange(pi1, pi2 + 1)
                        delta[idx1] += (1.0 - h[i + idx1 - (n - 1)])
                    pi3 = max(i, 0)
                    pi4 = min(i + n - 1, 2 * n - 2)
                    if pi3 <= pi4:
                        idx3 = np.arange(pi3, pi4 + 1)
                        delta[idx3] -= h[i - idx3 + (n - 1)]
                    # j goes 1->0
                    pj1 = max(n - 1 - j, 0)
                    pj2 = min(2 * n - 2 - j, 2 * n - 2)
                    if pj1 <= pj2:
                        idx1 = np.arange(pj1, pj2 + 1)
                        delta[idx1] -= (1.0 - h[j + idx1 - (n - 1)])
                    pj3 = max(j, 0)
                    pj4 = min(j + n - 1, 2 * n - 2)
                    if pj3 <= pj4:
                        idx3 = np.arange(pj3, pj4 + 1)
                        delta[idx3] += h[j - idx3 + (n - 1)]
                    
                    new_max = (corr + delta).max() * dx
                    if new_max < cur_c - 1e-15:
                        h[i] = 1.0
                        h[j] = 0.0
                        corr += delta
                        cur_c = new_max
                        improved = True
                        break
            
            if not improved:
                break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
```
