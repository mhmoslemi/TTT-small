```python
def compute_gradient_smooth(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    c = corr * dx
    # Compute softmax weights
    a = alpha * c
    a_max = a.max()
    exp_a = np.exp(a - a_max)
    p = exp_a / exp_a.sum()
    # p has length 2n-1, corresponding to lags -(n-1) to +(n-1)
    grad = np.zeros(n)
    for k in range(2*n-1):
        m = k - (n-1)  # lag
        if p[k] < 1e-15:
            continue
        if m == 0:
            grad += p[k] * dx * (1 - 2*h)
        elif m > 0:
            # corr_m = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            # d/dh[j] = (1-h[j+m]) if j < n-m, minus h[j-m] if j >= m
            if m < n:
                grad[:n-m] += p[k] * dx * (1 - h[m:])
                grad[m:] -= p[k] * dx * h[:n-m]
        else:
            m = -m  # make positive
            # For negative lag -m: corr = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
            # = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
            # d/dh[j] = (1-h[j-m]) if j >= m, minus h[j+m] if j < n-m
            if m < n:
                grad[m:] += p[k] * dx * (1 - h[:n-m])
                grad[:n-m] -= p[k] * dx * h[m:]
    return grad
```

        if deadline and time.time() > deadline:
            break
        c = compute_all_overlaps(h, n)
        c_max = float(c.max())
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        k_star = int(np.argmax(c))
        m = k_star - (n - 1)
        g = np.zeros(n)
        if m == 0:
            g[:] = dx * (1.0 - 2.0 * h)
        elif m > 0:
            g[:n - m] += dx * (1.0 - h[m:])
            g[m:] -= dx * h[:n - m]
        else:
            mp = -m
            g[mp:] += dx * (1.0 - h[:n - mp])
            g[:n - mp] -= dx * h[mp:]
        lr = 0.02 / (1.0 + 0.05 * it)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c + 0.005:
            h = h_new
        else:
            h_new2 = h - lr * 0.2 * g
            h_new2 = project_feasible(h_new2, n)
            if compute_c5(h_new2, n) < best_c + 0.002:
                h = h_new2
    return best_h, best_c

def coord_refine(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_overlaps(h, n)
        k_star = int(np.argmax(c))
        m = k_star - (n - 1)
        if m == 0:
            break
        mp = abs(m)
        best_imp = 0.0
        best_j = -1
        best_dir = 0
        for j in range(n):
            if m > 0:
                coeff = 0.0
                if j < n - mp:
                    coeff += (1.0 - h[j + mp])
                if j >= mp:
                    coeff -= h[j - mp]
            else:
                coeff = 0.0
                if j >= mp:
                    coeff += (1.0 - h[j - mp])
                if j < n - mp:
                    coeff -= h[j + mp]
            if coeff > 0:
                imp = coeff * h[j] * dx
            elif coeff < 0:
                imp = (-coeff) * (1.0 - h[j]) * dx
            else:
                continue
            if imp > best_imp:
                best_imp = imp
                best_j = j
                best_dir = -1 if coeff > 0 else 1
        if best_j < 0 or best_imp < 1e-8:
            break
        if best_dir == -1:
            h[best_j] = max(0.0, h[best_j] - min(h[best_j], best_imp / (dx * best_imp + 1e-15) * 0.5))
            h[best_j] = 0.0
        else:
            h[best_j] = 1.0
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
        else:
            break
    return h, best_c

def multi_stage_optimize(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [200, 500, 1500, 5000, 20000, 80000]
    penalty = 1e6
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        maxit = 4000 if alpha > 5000 else 3000
        h_new = lbfgsb_step(h, n, alpha, penalty, maxiter=maxit)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new

    if time.time() < deadline - 8:
        h_new, c_new = projected_subgrad(best_h, n, max_iter=2000, deadline=deadline - 3)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 2:
        best_h, best_c = coord_refine(best_h, n, deadline - 1, max_rounds=100)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init.copy(), n_init)
        best_c = compute_c5(best_h, n_init)

    # Stage 1: Optimize at initial resolution
    if best_h is not None:
        n = len(best_h)
        alloc = min(350, (deadline - t0) * 0.35)
        h_opt, c_opt = multi_stage_optimize(best_h, n, t0 + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Stage 2: Try multiple resolutions
    resolutions = sorted(set([
        int(n_init * 0.75), int(n_init * 0.9), n_init,
        int(n_init * 1.25), int(n_init * 1.5),
        int(n_init * 1.8), int(n_init * 2.0), int(n_init * 2.5)
    ]))
    resolutions = [r for r in resolutions if 80 <= r <= 1200]

    for n_try in resolutions:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        elif h_init is not None:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = multi_stage_optimize(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Stage 3: Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.3)
            if alloc < 20:
                break
            h_opt, c_opt = multi_stage_optimize(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Stage 4: Final high-precision refinement
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # One more L-BFGS-B with very high alpha
        h_new = lbfgsb_step(best_h, n, 200000, 1e7, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Projected subgradient polish
        if time.time() < deadline - 3:
            h_new, c_new = projected_subgrad(best_h, n, max_iter=3000, deadline=deadline - 1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
