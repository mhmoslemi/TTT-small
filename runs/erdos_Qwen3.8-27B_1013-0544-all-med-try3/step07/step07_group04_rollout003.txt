    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    """Compute weighted gradient of sum_k w_k * lag_k."""
    dx = 2.0 / n
    grad = np.zeros(n)

            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def _sum_con_jac(x, n):
    return np.ones(n)


def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def _active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def compute_weighted_gradient(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] += w * (-h[:n - m])
        else:
            m = -lag
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] += w * (-h[m:])
    return dx * grad


def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def active_set_slsqp(h, n, top_idx, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_active_smooth_obj, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def projected_gradient(h_init, n, K, alpha_smooth, max_iter, deadline):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        lags = compute_all_lags(h, n)
        top_idx = np.argsort(lags)[-K:]
        vals = lags[top_idx]
        v_max = vals.max()
        exps = np.exp(alpha_smooth * (vals - v_max))
        s = exps.sum()
        if s <= 0:
            break
        weights_full = np.zeros(2 * n - 1)
        weights_full[top_idx] = exps / s
        grad = compute_weighted_gradient(h, n, weights_full)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        found = False
        step = 0.5 / (g_norm + 1e-10)
        for _ in range(40):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                found = True
                break
            step *= 0.5
        if not found:
            step = 1.0 / (g_norm + 1e-10)
            for _ in range(20):
                h_new = project_feasible(h - step * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    h = h_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
                    found = True
                    break
                step *= 0.5
            if not found:
                break
    return best_h, best_c


def local_search(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            n_pairs = n - m
            for i in range(n_pairs):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4, h[i], 1.0 - h[j])
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.4, h[j], 1.0 - h[i])
                    if delta < 1e-10:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c


def multi_lag_local_search(h, n, deadline, top_k=3):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_k:]
            for k_star in top_idx:
                if time.time() > deadline:
                    break
                lag = k_star - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                n_pairs = n - m
                for i in range(0, n_pairs, max(1, n_pairs // 200)):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.3, h[i], 1.0 - h[j])
                        if delta < 1e-10:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.3, h[j], 1.0 - h[i])
                        if delta < 1e-10:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
    return h, best_c


def full_pipeline_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase A: Projected gradient (fast, good for high n)
    if time.time() < deadline - 20:
        pg_end = time.time() + min(30, (deadline - time.time()) * 0.2)
        h_pg, c_pg = projected_gradient(best_h, n, K=10, alpha_smooth=50,
                                        max_iter=500, deadline=pg_end)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg

    # Phase B: Active-set SLSQP
    if time.time() < deadline - 15:
        c_lags = compute_all_lags(best_h, n)
        top_idx = np.argsort(c_lags)[-8:]
        for alpha_try in [5000, 20000, 50000]:
            if time.time() > deadline - 15:
                break
            h_new = active_set_slsqp(best_h, n, top_idx, alpha_try, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new

    # Phase C: Full SLSQP
    if time.time() < deadline - 12:
        for alpha_try in [10000, 50000]:
            if time.time() > deadline - 12:
                break
            h_new = slsqp_optimize(best_h, n, alpha_try, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new

    # Phase D: More projected gradient
    if time.time() < deadline - 10:
        pg_end2 = time.time() + min(20, (deadline - time.time()) * 0.3)
        h_pg2, c_pg2 = projected_gradient(best_h, n, K=15, alpha_smooth=200,
                                          max_iter=500, deadline=pg_end2)
        if c_pg2 < best_c:
            best_c = c_pg2
            best_h = h_pg2

    # Phase E: Local search
    if time.time() < deadline - 3:
        best_h, best_c = local_search(best_h, n, min(deadline - 1, time.time() + 30))

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize the parent
    if h_init is not None:
        alloc1 = min(100, (deadline - t0) * 0.12)
        h_opt, c_opt = full_pipeline_at_n(h_init, n_init, t0 + alloc1, rng)
        best_h = h_opt
        best_c = c_opt

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    # Phase 2: Try multiple resolutions
    n_candidates = [200, 250, 300, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.22)
        if alloc < 40:
            alloc = 40

        # Interpolate from best
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)

        h_opt, c_opt = full_pipeline_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Try a few perturbed restarts at the best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.01 + 0.008 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 30) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = full_pipeline_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Try one higher resolution if time allows
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        n_high = min(n + 100, 600)
        if n_high > n:
            x_best = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_high)
            h_start = np.interp(x_new, x_best, best_h)
            h_start = project_feasible(h_start, n_high)
            alloc = min(60, (deadline - time.time() - 30) * 0.5)
            if alloc > 25:
                h_opt, c_opt = full_pipeline_at_n(h_start, n_high, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 5: Final intensive polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)

        # Very high alpha SLSQP on top-3 lags
        if time.time() < deadline - 8:
            c_lags = compute_all_lags(best_h, n)
            for k_top in [3, 5]:
                top_idx = np.argsort(c_lags)[-k_top:]
                h_new = active_set_slsqp(best_h, n, top_idx, 200000, maxiter=4000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new

        # Projected gradient with fine parameters
        if time.time() < deadline - 5:
            h_pg, c_pg = projected_gradient(best_h, n, K=5, alpha_smooth=500,
                                            max_iter=300, deadline=deadline - 4)
            if c_pg < best_c:
                best_c = c_pg
                best_h = h_pg

        # Final local search (most important for squeezing out last bits)
        if time.time() < deadline - 1:
            best_h, best_c = local_search(best_h, n, deadline - 0.5)

        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
