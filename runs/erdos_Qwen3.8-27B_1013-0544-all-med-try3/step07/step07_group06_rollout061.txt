            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h *= (1.0 - diff / s)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(x, alpha, n):
    """Objective and gradient for (1/alpha)*log(sum(exp(alpha*g[k])))."""
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    ag = alpha * g
    ag_max = ag.max()
    ex = ag - ag_max
    w = np.exp(ex)
    w_sum = w.sum()
    w /= w_sum
    val = ag_max / alpha + np.log(w_sum) / alpha
    
    # Gradient
    s = np.arange(-(n-1), n, dtype=int)  # shift for each k=0..2n-2
    grad = np.zeros(n)
    for i in range(n):
        valid1 = (i + s >= 0) & (i + s < n)
        valid2 = (i - s >= 0) & (i - s < n)
        if valid1.any():
            grad[i] += (1 - x[i + s[valid1]]).dot(w[valid1])
        if valid2.any():
            grad[i] -= x[i - s[valid2]].dot(w[valid2])
    grad *= dx
    return val, grad

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def fun(x):
        return lse_obj_grad(x, alpha, n)[0]
    def jac(x):
        return lse_obj_grad(x, alpha, n)[1]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h.copy(), method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

Hmm, this is getting very complex and the inner loop over j for each i is O(n²) with O(n) work each = O(n³) per sweep. With n=250, that's ~15M operations per sweep. In Python with numpy, maybe 1-2 seconds per sweep. We could do 100-500 sweeps in the time budget.

But wait, there's a much simpler approach: just recompute corr after each swap. np.correlate for n=250 takes maybe 50μs. So one swap + recompute is ~50μs. In a sweep with 250 zeros and 250 ones, we try 62500 swaps, taking 62500*50μs = 3.1 seconds per sweep. We could do ~200 sweeps in 1100 seconds. That might be enough.

But we can do better: for each i, find the BEST j in one pass, then only apply that one swap. This is still O(n²) per sweep but with early termination.

Actually, let me think about this differently. The key insight is:

For a binary h, corr[k] counts the number of (1,0) pairs at distance k (times dx). We want to minimize the max.

A swap of position i (0→1) and j (1→0) changes the correlation by a specific pattern. The max of the new correlation is what matters.

Let me try a different algorithmic approach: **iterative improvement targeting the maximum lag**.

1. Find k* = argmax corr[k]
2. For all pairs (i,j) where the swap would reduce corr[k*], try the swap
3. Among those, pick the one that minimizes the overall max
4. Repeat

This focuses the search on the most impactful swaps.

Actually, you know what, let me just go with a practical approach that I know works:

1. SLSQP with analytic gradient (fast, smooth)
2. Greedy binary search (slower but effective for binary solutions)
3. Try multiple resolutions
4. Use time budget wisely

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h *= (1.0 - diff / s)
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, alpha, n):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    ag = alpha * g
    ag_max = ag.max()
    w = np.exp(ag - ag_max)
    return ag_max / alpha + np.log(w.sum()) / alpha

def lse_gradient(x, alpha, n):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    ag = alpha * g
    ag_max = ag.max()
    w = np.exp(ag - ag_max)
    w_sum = w.sum()
    w /= w_sum
    s = np.arange(-(n - 1), n, dtype=int)
    grad = np.zeros(n)
    for i in range(n):
        v1 = (i + s >= 0) & (i + s < n)
        v2 = (i - s >= 0) & (i - s < n)
        if v1.any():
            grad[i] += (1.0 - x[i + s[v1]]).dot(w[v1])
        if v2.any():
            grad[i] -= x[i - s[v2]].dot(w[v2])
    grad *= dx
    return grad

def slsqp_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    ones_n = np.ones(n)
    def obj(x):
        return lse_objective(x, alpha, n)
    def jac(x):
        return lse_gradient(x, alpha, n)
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return ones_n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def binary_search_fast(h_init, n, deadline, max_sweeps=300, seed=42):
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    s_shift = np.arange(-(n - 1), n, dtype=int)
    full_len = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = list(np.where(h == 0)[0])
        ones = list(np.where(h == 1)[0])
        rng.shuffle(zeros)

        for i in zeros:
            if time.time() > deadline:
                break
            if h[i] != 0:
                continue
            v1 = (i + s_shift >= 0) & (i + s_shift < n)
            A = np.zeros(full_len)
            if v1.any():
                A[v1] = (1.0 - h[i + s_shift[v1]])

            best_j = -1
            best_nc = cur_c

            for j in ones:
                if h[j] != 1:
                    continue
                v2 = (j + s_shift >= 0) & (j + s_shift < n)
                v3 = (j - s_shift >= 0) & (j - s_shift < n)
                B = np.zeros(full_len)
                C = np.zeros(full_len)
                if v2.any():
                    B[v2] = (1.0 - h[j + s_shift[v2]])
                if v3.any():
                    C[v3] = h[j - s_shift[v3]]

                delta = A - B + C
                k_ij = n - 1 + (j - i)
                if 0 <= k_ij < full_len:
                    delta[k_ij] += 2.0
                k_ji = n - 1 + (i - j)
                if 0 <= k_ji < full_len:
                    delta[k_ji] += 1.0
                delta[n - 1] -= 3.0

                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j

            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()

        if not improved:
            break

    return best_h, best_c

def simulated_annealing(h_init, n, deadline, n_iter=50000, seed=42):
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    T0 = 0.05
    T1 = 0.001
    total_iters = n_iter
    zeros_list = np.where(h == 0)[0].tolist()
    ones_list = np.where(h == 1)[0].tolist()

    for it in range(total_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / float(total_iters))
        a = rng.choice(zeros_list)
        b = rng.choice(ones_list)
        if h[a] != 0 or h[b] != 1:
            continue
        h[a] = 1.0
        h[b] = 0.0
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c = new_corr.max() * dx
        delta_e = new_c - cur_c
        if delta_e < 0 or rng.random() < np.exp(-delta_e / T):
            corr = new_corr
            cur_c = new_c
            zeros_list[b] = a
            ones_list[a] = b
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[a] = 0.0
            h[b] = 1.0
            zeros_list[zeros_list.index(a)] = b
            ones_list[ones_list.index(b)] = a

    return best_h, best_c

    deadline = t0 + budget_s - 3
