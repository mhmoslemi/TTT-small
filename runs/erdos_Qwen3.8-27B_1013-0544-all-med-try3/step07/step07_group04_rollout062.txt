import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float((m + mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float((m + np.log(np.sum(np.exp(shifted)))) / alpha)

def _smooth_obj_active(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float((m + mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float((m + np.log(np.sum(np.exp(shifted)))) / alpha)

def slsqp_run(h, n, alpha, top_idx=None, maxiter=5000, ftol=1e-13):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        try:
            res = minimize(_smooth_obj_active, h, args=(n, top_idx, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        except Exception:
            return h.copy()
    else:
        try:
            res = minimize(_smooth_obj_full, h, args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        except Exception:
            return h.copy()
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return h.copy()

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    wsum = np.exp(shifted).sum()
    return float(m + np.log(wsum) / alpha)

def _smooth_obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    wsum = np.exp(shifted).sum()
    return float(m + np.log(wsum) / alpha)

def slsqp_solve(h, n, alpha, top_idx, maxiter=5000, ftol=1e-13):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        try:
            res = minimize(_smooth_obj, h, args=(n, alpha, top_idx),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        except Exception:
            return h.copy()
    else:
        try:
            res = minimize(_smooth_obj_full, h, args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        except Exception:
            return h.copy()
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return h.copy()

def _lag_gradient_basis(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] += dx * (1.0 - h[m:])
        grad[m:] += dx * (-h[:n - m])
    else:
        grad[:n - m] += dx * (1.0 - h[m:])
        grad[m:] -= dx * h[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _lag_gradient_basis(np.ones(n) * 0, n, 0)  # placeholder
        # Linearize: c[k] + sum_j (d c[k] / d h[j])_{h_curr} * (h[j] - h_curr[j]) <= t
        g = _lag_gradient_basis(h_curr, n, lag)
        cons.append(c[k] + g @ (h_var - h_curr) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_loop(h, n, deadline, max_iters=15):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)

        h_curr = h_new
        trust *= 0.85

    return best_h, best_c

            m = abs(lag)

            # Try pairs for this lag
