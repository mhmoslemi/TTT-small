def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_of_lag(h, n, m):

    s = m - (n - 1)  # shift
    g = np.zeros(n)
    # For valid i: 0 <= i < n and 0 <= i+s < n
    if s >= 0:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    # max(shifted) = 0, stable
    e = np.exp(shifted)
    log_sum_exp = np.log(e.sum())
    obj = c_max + log_sum_exp / alpha
    weights = e / e.sum()
    
    # Gradient computation
    g = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        s = m - (n - 1)
        if s >= 0:
            idx = np.arange(n - s)

            t = -s
            idx = np.arange(t, n)
            g[idx] += dx * w * (1.0 - x[idx - t])
            g[idx - t] -= dx * w * x[idx]
    return obj, g

                          method='L-BFGS-B', bounds=bounds,

            h_new = project_feasible(res.x, n)

    return h, best_c

    best_c = compute_c5(h, n)
    best_h = h.copy()

            best_h = h.copy()

                g[idx] += dx * (1.0 - h[idx + s])
                g[idx + s] -= dx * h[idx]
            else:

                g[idx - t] -= dx * h[idx]

        # Step size with backtracking

        if g_norm < 1e-15:

        eta = eta0 / (1.0 + it * 0.01)

                found = True
                break

            if eta0 < 1e-8:
                break
    
    return best_h, best_c

def fine_pair_search(h, n, max_time, deltas, radius=100):
    """Local pair search with very small steps."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:
        if time.time() > max_time:
            break
        for _pass in range(200):
            if time.time() > max_time:
                break
            improved = False
            for i in range(n):
                if time.time() > max_time:
                    break
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    for sign in [1, -1]:
                        d_signed = d * sign
                        hi = h[i] + d_signed
                        hj = h[j] - d_signed
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - d_signed, hj + d_signed
            if not improved:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_lag(h, n, m):
    s = m - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        idx = np.arange(n - s)
        g[idx] += (1.0 - h[idx + s])
        g[idx + s] -= h[idx]
    else:
        t = -s
        idx = np.arange(t, n)
        g[idx] += (1.0 - h[idx - t])
        g[idx - t] -= h[idx]
    return g

def smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    e = np.exp(shifted)
    log_sum_exp = np.log(e.sum())
    obj = float(c_max + log_sum_exp / alpha)
    weights = e / e.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 5e-10:
            continue
        s = m - (n - 1)
        if s >= 0:
            idx = np.arange(n - s)
            g[idx] += dx * w * (1.0 - x[idx + s])
            g[idx + s] -= dx * w * x[idx]
        else:
            t = -s
            idx = np.arange(t, n)
            g[idx] += dx * w * (1.0 - x[idx - t])
            g[idx - t] -= dx * w * x[idx]
    return obj, g

    h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > max_time:
            break
        try:
            res = minimize(smooth_obj_and_grad, h, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds, jac=True,
                          options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-12,
                                   'maxfun': 20000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def peak_shaving(h, n, max_time, tol_frac=0.001):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    eta = 0.05
    stall_count = 0
    for it in range(80000):
        if time.time() > max_time:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c_max = corr.max()
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 3000:
                break
        threshold = c_max - tol_frac * c_max
        active = np.where(corr >= threshold)[0]
        g = np.zeros(n)
        for m in active:
            g += grad_lag(h, n, m)
        g /= len(active)
        g *= dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
