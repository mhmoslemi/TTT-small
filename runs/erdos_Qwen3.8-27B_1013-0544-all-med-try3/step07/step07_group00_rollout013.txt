import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def hinge_obj(x, n, alpha, mu, top_k=10):
    """Hinge loss on top correlations."""
    c = compute_c5_vec(x, n)
    m = c.max()
    # Hinge loss: penalize shifts above (m - margin)
    margin = 0.001
    excess = np.maximum(c - (m - margin), 0)
    val = m + alpha * np.sum(excess**2) + mu * (x.sum() - n/2.0)**2
    return float(val)

def hinge_grad(x, n, alpha, mu, top_k=10):
    c = compute_c5_vec(x, n)
    m = c.max()
    margin = 0.001
    active = c > (m - margin)
    # Gradient of m + alpha * sum(excess^2) w.r.t. x
    # This is complex due to the max, so use the top shifts directly
    dx = 2.0 / n
    # Get indices of active shifts

    for idx in active_idx:
        k = idx - (n - 1)
        wm = 1.0 / len(active_idx) + 2.0 * alpha * max(c[idx] - (m - margin), 0)
        if k >= 0:

            grad[0:nv] += wm * (1.0 - x[k:n])

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    t1 = min(t0 + 100, deadline - 500)
    if t1 > t0 + 10:
        h, best_c = lbfgs_optimize(h, n, t1,
            configs=[(5000, 5.0), (50000, 50.0), (500000, 500.0), (2000000, 5000.0)])
        best_h = h.copy()

    # Phase 2: Paired coordinate descent (~150s)
    t2 = min(t0 + 250, deadline - 400)
    if t2 > time.time() + 10:
        h, c2 = paired_cd(h, n, t2, n_sweeps=300, n_pairs=200)
        if c2 < best_c:
            best_c = c2
            best_h = h.copy()

    # Phase 3: Active-set focused optimization (~80s)
    t3 = min(t0 + 330, deadline - 350)
    if t3 > time.time() + 10:
        h, c3 = active_set_optimize(h, n, t3, top_k=6,
                                     alphas=[20000, 100000, 500000])
        if c3 < best_c:
            best_c = c3
            best_h = h.copy()

    # Phase 4: L-BFGS with very high alpha (~60s)
    t4 = min(t0 + 390, deadline - 300)
    if t4 > time.time() + 10:
        h, c4 = lbfgs_optimize(h, n, t4,
            configs=[(5000000, 5000.0), (20000000, 50000.0)])
        if c4 < best_c:
            best_c = c4
            best_h = h.copy()

    # Phase 5: Multi-resolution search (~100s)
    t5 = min(t0 + 490, deadline - 250)
    if t5 > time.time() + 30:
        h_mr, c5_mr = multi_res_opt(h, n, t5)
        if c5_mr < best_c:
            best_c = c5_mr
            best_h = h_mr.copy()
            n = len(best_h)

    # Phase 6: Directional search (~80s)
    t6 = min(t0 + 570, deadline - 200)
    if t6 > time.time() + 10:
        h, c6 = directional_search(h, n, t6, base_delta=0.004)
        if c6 < best_c:
            best_c = c6
            best_h = h.copy()

    # Phase 7: Another L-BFGS round (~80s)
    t7 = min(t0 + 650, deadline - 150)
    if t7 > time.time() + 10:
        h, c7 = lbfgs_optimize(h, n, t7,
            configs=[(100000, 100.0), (1000000, 1000.0), (5000000, 10000.0)])
        if c7 < best_c:
            best_c = c7
            best_h = h.copy()

    # Phase 8: Active-set again (~60s)
    t8 = min(t0 + 710, deadline - 120)
    if t8 > time.time() + 10:
        h, c8 = active_set_optimize(h, n, t8, top_k=4,
                                     alphas=[50000, 500000, 5000000])
        if c8 < best_c:
            best_c = c8
            best_h = h.copy()

    # Phase 9: Stochastic local search (~100s)
    t9 = min(t0 + 810, deadline - 80)
    if t9 > time.time() + 10:
        h, c9 = stochastic_local_search(h, n, t9, base_step=0.003)
        if c9 < best_c:
            best_c = c9
            best_h = h.copy()

    # Phase 10: Final L-BFGS polish (~50s)
    t10 = min(t0 + 860, deadline - 40)
    if t10 > time.time() + 5:
        h, c10 = lbfgs_optimize(h, n, t10,
            configs=[(10000000, 10000.0)])
        if c10 < best_c:
            best_c = c10
            best_h = h.copy()

    # Phase 11: Final directional + paired CD (~50s)
    t11 = deadline - 3
    if time.time() < t11 - 10:
        h, c11a = directional_search(h, n, min(time.time() + 25, t11 - 5), base_delta=0.002)
        if c11a < best_c:
            best_c = c11a
            best_h = h.copy()
    if time.time() < t11 - 5:
        h, c11b = paired_cd(h, n, t11, n_sweeps=100, n_pairs=100)
        if c11b < best_c:
            best_c = c11b
            best_h = h.copy()

    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
