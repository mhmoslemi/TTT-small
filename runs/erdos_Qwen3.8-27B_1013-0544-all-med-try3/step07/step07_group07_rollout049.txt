def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max+1] += pk * (1.0 - h[i_min-k:i_max-k+1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2+1] -= pk * h[i_min2+k:i_max2+k+1]
    return grad

def pgd_solve(h, n, alphas, deadline, max_iter=5000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        cur_c = best_c
        step = min(float(n) / (4.0 * alpha), 0.5)
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            grad = lse_grad(h, n, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            direction = -grad / gnorm
            accepted = False
            for ls in range(30):
                h_new = project(h + step * (0.5 ** ls) * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    h = h_new
                    cur_c = c_new
                    step = step * (0.5 ** ls)
                    accepted = True
                    break
            if accepted:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max+1] += pk * (1.0 - h[i_min-k:i_max-k+1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2+1] -= pk * h[i_min2+k:i_max2+k+1]
    return grad

def pgd_solve(h, n, alphas, deadline, max_iter=5000):

def swap_delta(h, i, j, n, corr):
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n-1] -= 2.0

        delta[k1] += 1.0
    k2 = n - 1 + (i - j)

    return delta

    # Phase 6: Second annealing pass at different temperature (remaining time)
    p6 = dl - 3
    if time.time() < p6 - 15:
        ha2 = to_binary(best_h, n)
        ha2, ca2 = anneal_binary(ha2, n, p6, seed=seed+100, T_init=0.015, cool_rate=0.99999)
        if ca2 < best_c:
            best_c = ca2
            best_h = ha2.copy()
    
    # Final: try higher resolution if time permits
    if time.time() < dl - 40:
        nh = n * 2
        if nh <= 1600:
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = np.interp(xn, xo, best_h)
            hh = project(hh, nh)
            hh, ch = adaptive_pgd(hh, nh, dl - 20)
            if ch < best_c:
                best_c = ch
                best_h = hh.copy()
                n = nh
            hb4 = to_binary(best_h, n)
            hb4, cb4 = binary_local_search(hb4, n, dl - 5, max_sweeps=100)
            if cb4 < best_c:
                best_c = cb4
                best_h = hb4.copy()
    
    # Final evaluation
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
