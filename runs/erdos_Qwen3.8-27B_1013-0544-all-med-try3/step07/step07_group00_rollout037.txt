import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def corr_gradient(h, n, shift_idx):
    """Gradient of correlation at shift_idx with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj_and_grad(h, n, alpha, active_shifts=None):

    c = compute_c5_vec(h, n)
    if active_shifts is not None:
        m = c.max()
        sh = alpha * (c[active_shifts] - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        val = m + (mx + np.log(w.sum())) / alpha  # log-sum-exp
        # Actually: m + logsumexp(sh)/alpha = m + (mx + log(sum(exp(sh-mx))))/alpha
    else:
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        val = m + (mx + np.log(w.sum())) / alpha

    dx = 2.0 / n
    grad = np.zeros(n)

        for wi, idx in zip(w, active_shifts):
            gm = corr_gradient(h, n, idx)
            grad += wi * gm
    else:
        for wi, idx in zip(w, range(2*n-1)):
            if wi < 1e-14:
                continue
            gm = corr_gradient(h, n, idx)
            grad += wi * gm
    
    return float(val), grad

    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]

    for d in deltas:
        if time.time() > deadline - 10:
            break
        improved = True
        iters = 0
        while improved and iters < max_iters:
            improved = False
            iters += 1
            if iters % 200 == 0 and time.time() > deadline - 10:
                break
            for _ in range(200):
                if time.time() > deadline - 10:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                if h_ds[i] + d <= 1.0 and h_ds[j] - d >= 0.0:
                    h_ds[i] += d
                    h_ds[j] -= d
                    c = compute_c5(h_ds, n)
                    if c < c_ds - 1e-15:
                        c_ds = c
                        improved = True
                    else:
                        h_ds[i] -= d
                        h_ds[j] += d
                if h_ds[i] - d >= 0.0 and h_ds[j] + d <= 1.0:
                    h_ds[i] -= d
                    h_ds[j] += d
                    c = compute_c5(h_ds, n)
                    if c < c_ds - 1e-15:
                        c_ds = c
                        improved = True
                    else:
                        h_ds[i] += d
                        h_ds[j] -= d
    return h_ds, c_ds


def gradient_guided_pair_search(h, n, deadline, rng, base_step=0.002):
    best_c = compute_c5(h, n)
    h_gg = h.copy()
    c_gg = best_c
    step = base_step

    while time.time() < deadline - 5 and step > 1e-7:
        c_vec = compute_c5_vec(h_gg, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]

        grad = np.zeros(n)
        for idx in active:
            grad += corr_gradient(h_gg, n, idx)
        grad /= len(active)

        # Find top gradient magnitude indices
        top_idx = np.argsort(-np.abs(grad))[:80]

        found = False
        for _ in range(3000):
            if time.time() > deadline - 5:
                break
            i = top_idx[rng.randint(len(top_idx))]
            j = top_idx[rng.randint(len(top_idx))]
            if i == j:
                continue
            gi, gj = grad[i], grad[j]
            # Move in direction that reduces max correlation
            if gi > gj and h_gg[i] - step >= 0 and h_gg[j] + step <= 1:
                h_gg[i] -= step
                h_gg[j] += step
                c = compute_c5(h_gg, n)
                if c < c_gg - 1e-15:
                    c_gg = c
                    found = True
                else:
                    h_gg[i] += step
                    h_gg[j] -= step
            elif gj > gi and h_gg[j] - step >= 0 and h_gg[i] + step <= 1:
                h_gg[j] -= step
                h_gg[i] += step
                c = compute_c5(h_gg, n)
                if c < c_gg - 1e-15:
                    c_gg = c
                    found = True
                else:
                    h_gg[j] += step
                    h_gg[i] -= step
        if not found:
            step *= 0.5

    return h_gg, c_gg


def subgradient_polish(h, n, deadline):
    best_c = compute_c5(h, n)
    h_pol = h.copy()
    c_pol = best_c
    step = 0.01
    stall_count = 0

    while time.time() < deadline - 5 and step > 1e-8:
        h_new = gradient_subgrad_step(h_pol, n, step)
        c_new = compute_c5(h_new, n)
        if c_new < c_pol - 1e-14:
            c_pol = c_new
            h_pol = h_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 5:
                step *= 0.5
                stall_count = 0
    return h_pol, c_pol


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)

    # Load initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except (NameError, Exception):
        n_orig = 500
        h_init = np.full(n_orig, 0.5)

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    best_n = n_orig

    # === Phase 1: Progressive L-BFGS with active set at original resolution ===
    # Use a series of increasing alpha values
    alphas_phase1 = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    time_per_alpha = 60  # seconds per alpha level
    elapsed_limit = 350

    for alpha in alphas_phase1:
        if time.time() - t0 > elapsed_limit:
            break
        if time.time() > deadline - 400:
            break

        c_vec = compute_c5_vec(best_h, best_n)
        max_val = c_vec.max()

        # Find active shifts (within 1% of max)
        for eps in [0.005, 0.002, 0.001]:
            active = find_active_shifts(c_vec, max_val, eps)
            if len(active) < 1:
                active = np.array([np.argmax(c_vec)])

            maxiter = max(500, min(5000, int(1500000 / max(1, best_n * len(active)))))
            h_new = run_lbfgs_active(best_h, best_n, alpha, active, maxiter, 1e-16)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                # Update c_vec for next eps iteration
                c_vec = compute_c5_vec(best_h, best_n)
                max_val = c_vec.max()

    # === Phase 2: Very high alpha refinement with tight active set ===
    for _ in range(20):
        if time.time() > deadline - 300:
            break
        c_vec = compute_c5_vec(best_h, best_n)
        max_val = c_vec.max()
        active = find_active_shifts(c_vec, max_val, 0.0005)
        if len(active) < 1:
            active = np.array([np.argmax(c_vec)])

        h_new = run_lbfgs_active(best_h, best_n, 5000000, active, 3000, 1e-18)
        c_new = compute_c5(h_new, best_n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
        else:
            # Try with slightly wider active set
            active_wide = find_active_shifts(c_vec, max_val, 0.003)
            h_new = run_lbfgs_active(best_h, best_n, 1000000, active_wide, 2000, 1e-16)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # === Phase 3: Try higher resolution ===
    n_up = best_n * 2
    if n_up <= 2000 and time.time() < deadline - 250:
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_up)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_up)
        c_up = compute_c5(h_up, n_up)

        # Optimize at higher resolution
        up_alphas = [200, 2000, 20000, 200000, 1000000]
        for alpha in up_alphas:
            if time.time() > deadline - 200:
                break
            c_vec = compute_c5_vec(h_up, n_up)
            max_val = c_vec.max()
            active = find_active_shifts(c_vec, max_val, 0.005)
            if len(active) < 1:
                active = np.array([np.argmax(c_vec)])
            maxiter = max(500, min(3000, int(500000 / max(1, n_up * len(active)))))
            h_up = run_lbfgs_active(h_up, n_up, alpha, active, maxiter, 1e-16)
            c_up = compute_c5(h_up, n_up)

        # Additional refinement at high resolution
        for _ in range(5):
            if time.time() > deadline - 180:
                break
            c_vec = compute_c5_vec(h_up, n_up)
            max_val = c_vec.max()
            active = find_active_shifts(c_vec, max_val, 0.001)
            h_up = run_lbfgs_active(h_up, n_up, 5000000, active, 2000, 1e-18)
            c_up = compute_c5(h_up, n_up)

        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            best_n = n_up

    # === Phase 4: Subgradient polish ===
    if time.time() < deadline - 120:
        h_pol, c_pol = subgradient_polish(best_h, best_n, min(time.time() + 80, deadline - 100))
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    # === Phase 5: Gradient-guided pair search ===
    if time.time() < deadline - 80:
        h_gg, c_gg = gradient_guided_pair_search(best_h, best_n,
                                                  min(time.time() + 60, deadline - 60),
                                                  rng, base_step=0.003)
        if c_gg < best_c:
            best_c = c_gg
            best_h = h_gg

    # === Phase 6: Direct search with random pairs ===
    if time.time() < deadline - 40:
        h_ds, c_ds = direct_search(best_h, best_n,
                                    min(time.time() + 50, deadline - 25),
                                    rng, max_iters=30000)
        if c_ds < best_c:
            best_c = c_ds
            best_h = h_ds

    # === Phase 7: SLSQP final polish with equality constraint ===
    if time.time() < deadline - 10:
        n_f = len(best_h)
        bounds = [(0.0, 1.0)] * n_f

        def slqp_obj_w(x):
            return slqp_smooth_obj(x, n_f, 2000000)

        def slqp_grad_w(x):
            return slqp_smooth_grad(x, n_f, 2000000)

        def sum_fun_w(x):
            return sum_eq_fun(x, n_f)

        def sum_jac_w(x):
            return sum_eq_jac(x, n_f)

        cons = [{'type': 'eq', 'fun': sum_fun_w, 'jac': sum_jac_w}]
        try:
            res = minimize(slqp_obj_w, best_h, jac=slqp_grad_w, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1500, 'ftol': 1e-14})
            h_s = project_feasible(res.x, n_f)
            c_s = compute_c5(h_s, n_f)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
        except Exception:
            pass

    # === Phase 8: Final direct search micro-polish ===
    if time.time() < deadline - 3:
        n_f = len(best_h)
        d = 0.00001
        improved = True
        while improved and time.time() < deadline - 3:
            improved = False
            for _ in range(10000):
                if time.time() > deadline - 3:
                    break
                i = rng.randint(n_f)
                j = rng.randint(n_f)
                if i == j:
                    continue
                if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                    best_h[i] += d
                    best_h[j] -= d
                    c = compute_c5(best_h, n_f)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] -= d
                        best_h[j] += d
                if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                    best_h[i] -= d
                    best_h[j] += d
                    c = compute_c5(best_h, n_f)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d

    # Final: ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
