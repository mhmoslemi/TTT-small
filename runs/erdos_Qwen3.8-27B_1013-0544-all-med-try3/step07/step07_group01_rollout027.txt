                    h[i] = 1.0
                    h[j] = 0.0
            # Rebuild if needed
            if i != ones_sh[ii] or h[i] != 1:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        
        if c_max_new < best_c if 'c_max_new' in dir() else False:
            pass
        if not improved:
            T *= T_decay
            if T < T_min:
                break
        # Update best
        cur_c = compute_c5(h, n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

def binary_hill_climb(h, n, deadline):
    """Fast hill climbing on binary patterns - only accept improvements."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(500):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            if time.time() > deadline:
                return best_h, best_c
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    zeros = np.where(h == 0)[0]
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    
    return best_h, best_c

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, alpha_schedule=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    if alpha_schedule is None:
        alpha_schedule = [(100, 1e-4), (500, 1e-5), (2000, 1e-6),
                         (10000, 1e-7), (50000, 1e-8), (200000, 1e-9)]
    
    for alpha, mu in alpha_schedule:
        if time.time() > deadline - 5:
            break
        t_remaining = deadline - time.time() - 3
        if t_remaining < 10:
            break
        
        def obj(x):
            return smooth_max_obj(x, n, alpha) + mu * (x.sum() - n/2.0)**2
        
        def obj_grad(x):
            return smooth_max_grad(x, n, alpha) + 2*mu*(x.sum() - n/2.0)*np.ones(n)
        
        maxiter = min(500, max(50, int(t_remaining * 2)))
        try:
            res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxfun': maxiter*20,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def slsqp_optimize(h, n, deadline, alpha=100000):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    def constr(x):
        return float(x.sum() - n/2.0)
    
    def constr_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': constr, 'jac': constr_jac}]
    maxiter = 2000
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def bottleneck_optimize(h, n, deadline, n_top=10):
    """Focus optimization on the top n_top correlation lags."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    
    # Find top lags
    top_indices = np.argsort(c)[-n_top:]
    top_set = set(top_indices.tolist())
    
    # Use a subgradient-like approach focused on top lags
    lr = 0.001
    alpha = 500
    for it in range(5000):
        if time.time() > deadline:
            break
        
        # Recompute top lags periodically
        if it % 50 == 0:
            c = compute_c5_vec(h, n)
            top_indices = np.argsort(c)[-n_top:]
            top_set = set(top_indices.tolist())
        
        # Compute gradient of the sum of top correlations
        grad = np.zeros(n)
        for m_idx in top_indices:
            k = m_idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += dx * (1.0 - h[k:n])
                grad[k:n] += dx * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += dx * (1.0 - h[0:nv])
                grad[0:nv] += dx * (-h[kk:n])
        
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        grad /= g_norm
        
        # Line search
        improved = False
        for step in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-12:
                break
    
    return h, best_c

def multi_start_binary(n, deadline, n_starts=5, rng=None):
    """Try random binary starting points and optimize each."""
    if rng is None:
        rng = np.random.RandomState(42)
    best_h = None
    best_c = np.inf
    per_start = (deadline - time.time()) / n_starts - 5
    
    for s in range(n_starts):
        if time.time() > deadline - 10:
            break
        # Random binary with n/2 ones
        positions = rng.choice(n, size=n//2, replace=False)
        h = np.zeros(n)
        h[positions] = 1.0
        t_end = time.time() + per_start
        if t_end > deadline - 5:
            t_end = deadline - 5
        h, c = binary_hill_climb(h, n, t_end)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    if best_h is not None:
        return best_h, best_c
    return None, np.inf

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    # Phase 1: Binary hill climbing from parent (n=300)
    t_phase1 = min(t0 + 250, deadline - 400)
    h_bin = round_to_binary(h_init, n_init)
    h_bin, c_bin = binary_hill_climb(h_bin, n_init, t_phase1)
    best_c = min(c_bin, compute_c5(h_init, n_init))
    best_h = h_bin.copy() if c_bin < compute_c5(h_init, n_init) else project_feasible(h_init, n_init)
    print(f"After Phase 1 (binary HC): C5={best_c:.6f}, t={time.time()-t0:.1f}s")
    
    # Phase 2: Continuous refinement on best
    t_phase2 = min(time.time() + 200, deadline - 250)
    if time.time() < t_phase2:
        h_ref, c_ref = lbfgs_optimize(best_h, n_init, t_phase2)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref
        print(f"After Phase 2 (L-BFGS): C5={best_c:.6f}, t={time.time()-t0:.1f}s")
    
    # Phase 3: Binary annealing for escaping local minima
    t_phase3 = min(time.time() + 150, deadline - 200)
    if time.time() < t_phase3:
        h_anna = round_to_binary(best_h, n_init)
        h_anna, c_anna = binary_local_search(h_anna, n_init, t_phase3, max_rounds=100, use_annealing=True)
        if c_anna < best_c:
            best_c = c_anna
            best_h = h_anna
        # Also try continuous refinement of annealed solution
        if c_anna < best_c + 0.001 and time.time() < deadline - 50:
            h_ar, c_ar = lbfgs_optimize(h_anna, n_init, min(time.time() + 60, deadline - 30))
            if c_ar < best_c:
                best_c = c_ar
                best_h = h_ar
        print(f"After Phase 3 (annealing): C5={best_c:.6f}, t={time.time()-t0:.1f}s")
    
    # Phase 4: Multi-resolution exploration
    t_phase4_end = deadline - 100
    n_options = [150, 200, 250, 400, 500, 750]
    n_options = [n for n in n_options if n != n_init]
    
    # Sort by "closeness" to n_init for efficiency
    n_options.sort(key=lambda x: abs(x - n_init))
    
    per_res = max(30, (t_phase4_end - time.time()) / len(n_options) - 10)
    
    for n_try in n_options:
        if time.time() > t_phase4_end:
            break
        alloc = min(per_res, t_phase4_end - time.time() - 5)
        if alloc < 25:
            break
        
        # Interpolate best solution to new resolution
        x_old = np.linspace(0, 2, n_init) if n_init > 1 else np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        t_sub = time.time() + alloc
        
        # Binary search at new resolution
        h_b = round_to_binary(h_interp, n_try)
        h_b, c_b = binary_hill_climb(h_b, n_try, min(time.time() + alloc*0.4, t_sub - alloc*0.6))
        
        # Continuous refinement
        c_best_here = min(c_b, compute_c5(h_interp, n_try))
        h_best_here = h_b.copy() if c_b < compute_c5(h_interp, n_try) else h_interp.copy()
        
        if time.time() < t_sub - 10:
            h_r, c_r = lbfgs_optimize(h_best_here, n_try, min(time.time() + alloc*0.5, t_sub - 5))
            if c_r < c_best_here:
                c_best_here = c_r
                h_best_here = h_r
        
        if c_best_here < best_c:
            best_c = c_best_here
            best_h = h_best_here
            n_init = n_try
            print(f"  n={n_try}: C5={c_best_here:.6f} (NEW BEST)")
        else:
            print(f"  n={n_try}: C5={c_best_here:.6f}")
    
    print(f"After Phase 4 (multi-res): C5={best_c:.6f}, t={time.time()-t0:.1f}s")
    
    # Phase 5: SLSQP fine-tuning on best
    t_phase5 = min(time.time() + 120, deadline - 30)
    if time.time() < t_phase5:
        nf = len(best_h)
        for alpha in [50000, 200000]:
            if time.time() > t_phase5:
                break
            h_s, c_s = slsqp_optimize(best_h, nf, min(time.time() + 50, t_phase5), alpha=alpha)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
        print(f"After Phase 5 (SLSQP): C5={best_c:.6f}, t={time.time()-t0:.1f}s")
    
    # Phase 6: Final bottleneck optimization + binary polish
    t_phase6 = deadline - 5
    if time.time() < t_phase6 - 20:
        nf = len(best_h)
        h_bn, c_bn = bottleneck_optimize(best_h, nf, min(time.time() + 40, t_phase6 - 20), n_top=5)
        if c_bn < best_c:
            best_c = c_bn
            best_h = h_bn
        
        # Final binary polish
        if time.time() < t_phase6 - 5:
            h_bp = round_to_binary(best_h, nf)
            h_bp, c_bp = binary_hill_climb(h_bp, nf, t_phase6 - 2)
            if c_bp < best_c:
                best_c = c_bp
                best_h = h_bp
    
    # Final projection and verification
    nf = len(best_h)
    best_h = project_feasible(best_h, nf)
    best_c = compute_c5(best_h, nf)
    
    print(f"Final: C5={best_c:.6f}, n={nf}, t={time.time()-t0:.1f}s")
    return best_h, best_c, nf
