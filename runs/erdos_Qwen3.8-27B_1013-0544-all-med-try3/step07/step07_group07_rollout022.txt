import numpy as np
import time

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fft_convolve(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a) + len(b) - 1
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    L = next_pow2(n)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:n]


def corr_counts_fft(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    b = 1.0 - h
    conv = fft_convolve(h, b[::-1])
    c = conv[::-1].copy()
    c[c < 0.0] = 0.0
    return c


def corr_counts(h, use_direct_threshold=1024):
    n = len(h)
    if n <= use_direct_threshold:
        h = np.asarray(h, dtype=np.float64)
        c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    else:
        c = corr_counts_fft(h)
    c = np.asarray(c, dtype=np.float64).copy()
    c[c < 0.0] = 0.0
    return c


def direct_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def fast_c5(h, n):
    if n <= 3200:

    return float(np.max(corr_counts(h)) * (2.0 / float(n)))


def project_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    target = 0.5 * float(n)
    if len(h) == 0:
        return h
    lo = min(-1.0, float(np.min(h)) - 1.0)
    hi = max(1.0, float(np.max(h)) + 1.0)
    if lo == hi:
        lo -= 1.0
        hi += 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        s = float(hp.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    hp = np.clip(h - mid, 0.0, 1.0)
    diff = target - float(hp.sum())
    if diff > 0.0:
        room = 1.0 - hp
        rs = float(room.sum())
        if rs > 1e-14:

    elif diff < 0.0:
        s = float(hp.sum())

        if diff > 0.0:
            i = int(np.argmax(1.0 - hp))
            hp[i] = min(1.0, hp[i] + diff)
        else:
            i = int(np.argmax(hp))
            hp[i] = max(0.0, hp[i] + diff)

def normalize_sum(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    target = 0.5 * float(n)

def softmax_weights(c_l, beta, dx):
    v = np.asarray(c_l, dtype=np.float64) * float(dx)
    if v.size == 0:
        return np.empty(0, dtype=np.float64)
    vmax = float(np.max(v))
    if not np.isfinite(vmax):
        p = np.zeros_like(v)
        p[int(np.nanargmax(v))] = 1.0
        return p
    ex = np.exp(float(beta) * (v - vmax))
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(v)
        p[int(np.argmax(v))] = 1.0
        return p
    return ex / s


def lse_value(c_l, beta, dx):
    v = np.asarray(c_l, dtype=np.float64) * float(dx)
    if v.size == 0:
        return 0.0
    vmax = float(np.max(v))
    if not np.isfinite(vmax):
        return float(beta) * vmax
    ex = np.exp(float(beta) * (v - vmax))
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        return float(beta) * vmax
    return float(math.log(s) + float(beta) * vmax)


def grad_direct_from_weights(h, p):
    n = len(h)
    off = n - 1
    grad = np.zeros(n, dtype=np.float64)
    b = 1.0 - h
    h = np.asarray(h, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    idx = np.nonzero(p > 1e-16)[0]
    for t in idx:
        w = float(p[t])
        if w == 0.0:
            continue
        lag = int(t - off)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            cnt = n - lag
            if cnt > 0:
                grad[:cnt] += w * b[lag:]
                grad[lag:] -= w * h[:cnt]
        else:
            r = -lag
            cnt = n - r
            if cnt > 0:
                grad[r:] += w * b[:cnt]
                grad[:cnt] -= w * h[r:]
    return grad


def grad_from_weights(h, p, cost_threshold=200000):
    h = np.asarray(h, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = len(h)
    idx = np.nonzero(p > 1e-16)[0]
    cost = len(idx) * n
    if cost <= cost_threshold:
        return grad_direct_from_weights(h, p)
    off = n - 1
    b = 1.0 - h
    conv1 = fft_convolve(b, p[::-1])
    conv2 = fft_convolve(h, p)
    return (conv1 - conv2)[off:off + n]


def upsample(h_old, n_old, n_new):
    if n_old == n_new:
        return np.asarray(h_old, dtype=np.float64).copy()
    n_old = max(1, int(n_old))
    n_new = max(1, int(n_new))
    xo = (np.arange(n_old, dtype=np.float64) + 0.5) * (2.0 / float(n_old))
    xn = (np.arange(n_new, dtype=np.float64) + 0.5) * (2.0 / float(n_new))
    h = np.interp(xn, xo, np.asarray(h_old, dtype=np.float64))
    return project_h(h, n_new)


def select_resolutions(n0, max_n=3200, max_count=4):

            res.append(n)
    for extra in (2048, 3200):
        if extra <= max_n and extra not in res and extra > res[0]:

    res = sorted(set(res))

def binarize(h, n, seed=0):
    h = np.asarray(h, dtype=np.float64)
    n = int(n)
    target = n // 2
    if target <= 0:
        return np.zeros(n, dtype=np.int8)
    if target >= n:
        return np.ones(n, dtype=np.int8)
    rng = np.random.RandomState(seed)
    noise = rng.uniform(size=n) * 1e-12 * max(1.0, float(np.max(np.abs(h))))
    keys = h + noise
    order = np.argsort(keys)
    idx = order[n - target:]
    hb = np.zeros(n, dtype=np.int8)
    hb[idx] = 1
    return hb


def build_A(hb, n):
    L = 2 * n - 1
    off = n - 1
    A = np.zeros((n, L), dtype=np.int8)
    h = hb.astype(np.int8)
    b = 1 - h
    for lag in range(1, n):
        cnt = n - lag
        cpos = off + lag
        A[:cnt, cpos] += b[lag:]
        A[lag:, cpos] -= h[:cnt]
        cneg = off - lag
        A[lag:, cneg] += b[:cnt]
        A[:cnt, cneg] -= h[lag:]
    A[:, off] = 1 - 2 * h
    return A


def update_A_flip(A, x, delta_h, n, rows):
    off = n - 1
    sign = -int(delta_h)
    cols1 = x - rows + off
    cols2 = rows - x + off
    A[rows, cols1] += sign
    A[rows, cols2] += sign


def greedy_binary(hb, n, deadline, seed=1):
    hb = np.asarray(hb, dtype=np.int8).copy()
    n = int(n)
    target = n // 2
    if int(hb.astype(np.int64).sum()) != target:
        hb = binarize(hb.astype(np.float64), n, seed=seed)

    off = n - 1
    L = 2 * n - 1
    h32 = hb.astype(np.int32)
    b32 = (1 - hb).astype(np.int32)
    c = np.correlate(h32, b32, mode='full').astype(np.int32)
    cur_max = int(c.max())
    best_c = cur_max
    best_h = hb.copy()

    A = build_A(hb, n)
    c16 = c.astype(np.int16)
    rows = np.arange(n, dtype=np.int64)
    rng = np.random.RandomState(seed)

    sweep_limit = 10000
    sweep = 0
    sample_limit = 0 if n <= 2048 else 160

    while time.time() < deadline and sweep < sweep_limit:
        sweep += 1
        zeros = np.where(hb == 0)[0]
        ones = np.where(hb == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break

        order = rng.permutation(zeros)
        ones_list = [int(x) for x in ones]
        improved = False

        for i in order:
            if time.time() > deadline:
                break
            if hb[i] != 0 or not ones_list:
                continue

            ones_arr = np.asarray(ones_list, dtype=np.int64)
            if n > 2048 and len(ones_arr) > sample_limit:
                sel = rng.choice(len(ones_arr), size=sample_limit, replace=False)
                ones_arr = ones_arr[sel]

            M = len(ones_arr)
            if M == 0:
                break

            base = A[ones_arr].astype(np.int16, copy=True)
            base -= A[i][None, :]
            base += c16[None, :]
            base[:, off] -= 2

            if M:
                idx = np.arange(M)
                base[idx, ones_arr - i + off] += 1
                base[idx, i - ones_arr + off] += 1

            mx = base.max(axis=1)
            k = int(np.argmin(mx))
            if mx[k] < cur_max:
                j = int(ones_arr[k])
                delta = (A[i] - A[j]).astype(np.int16)
                delta[off] -= 2
                delta[j - i + off] += 1
                delta[i - j + off] += 1

                c += delta.astype(np.int32)
                hb[i] = 1
                hb[j] = 0
                update_A_flip(A, i, 1, n, rows)
                update_A_flip(A, j, -1, n, rows)

                c16 = c.astype(np.int16)
                cur_max = int(c.max())

                try:
                    ones_list.remove(j)
                except Exception:
                    ones_list = [x for x in ones_list if x != j]
                ones_list.append(int(i))

                improved = True
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = hb.copy()

        if not improved:
            break

    return best_h.astype(np.float64), float(best_c * (2.0 / float(n)))


def anneal_binary(hb, n, deadline, seed=2):
    hb = np.asarray(hb, dtype=np.int8).copy()
    n = int(n)
    target = n // 2
    if int(hb.astype(np.int64).sum()) != target:
        hb = binarize(hb.astype(np.float64), n, seed=seed)

    off = n - 1
    h32 = hb.astype(np.int32)
    b32 = (1 - hb).astype(np.int32)
    c = np.correlate(h32, b32, mode='full').astype(np.int32)
    cur_max = int(c.max())
    best_c = cur_max
    best_h = hb.copy()

    A = build_A(hb, n)
    c16 = c.astype(np.int16)
    rows = np.arange(n, dtype=np.int64)
    rng = np.random.RandomState(seed)

    zeros = [int(x) for x in np.where(hb == 0)[0]]
    ones = [int(x) for x in np.where(hb == 1)[0]]
    T = 0.5
    cool = 0.99995

    while time.time() < deadline and len(zeros) > 0 and len(ones) > 0:
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]

        delta = (A[i] - A[j]).astype(np.int16)
        delta[off] -= 2
        delta[j - i + off] += 1
        delta[i - j + off] += 1

        nc = int((c16 + delta).max())
        accept = False
        if nc < cur_max:
            accept = True
        else:
            diff = int(nc - cur_max)
            if diff <= 60 and T > 1e-6:
                if rng.random() < math.exp(-float(diff) / T):
                    accept = True

        if accept:
            c += delta.astype(np.int32)
            hb[i] = 1
            hb[j] = 0
            update_A_flip(A, i, 1, n, rows)
            update_A_flip(A, j, -1, n, rows)
            c16 = c.astype(np.int16)
            cur_max = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_max < best_c:
                best_c = cur_max
                best_h = hb.copy()
        T *= cool
        if T < 1e-6 and cur_max >= best_c:
            break

    return best_h.astype(np.float64), float(best_c * (2.0 / float(n)))


def wave_initial(n, seed=0, n_harm=5):
    rng = np.random.RandomState(seed)
    n = int(n)
    t = np.arange(n, dtype=np.float64) / float(n)
    s = np.zeros(n, dtype=np.float64)
    for r in range(1, n_harm + 1):
        amp = (0.35 + 0.65 * rng.random()) / (1.0 + 0.45 * r)
        ph = 2.0 * np.pi * rng.random()
        s += amp * np.cos(2.0 * np.pi * r * t + ph)
    thr = float(np.median(s))
    hb = (s > thr).astype(np.int8)
    target = n // 2
    cur = int(hb.sum())
    if cur > target:
        ones = np.where(hb == 1)[0]
        vals = s[ones]
        bad = np.argsort(vals)
        bad = bad[:cur - target]
        hb[ones[bad]] = 0
    elif cur < target:
        zeros = np.where(hb == 0)[0]
        vals = s[zeros]
        good = np.argsort(vals)[::-1]
        good = good[:target - cur]
        hb[zeros[good]] = 1
    return hb.astype(np.float64)

            h_cur = upsample(best_h, best_n, n)
            if carry_n != n:
                h_carry = upsample(carry_h, carry_n, n)
                c_cur = fast_c5(h_cur, n)
                c_carry = fast_c5(h_carry, n)
                if c_carry < c_cur:
                    h_cur = h_carry
            if best_n == n and best_c < fast_c5(h_cur, n):
                h_cur = best_h.copy()
            else:
                h_cur = project_h(h_cur, n)

        phase_best_h = h_cur.copy()
        phase_best_c = fast_c5(phase_best_h, n)
        if best_n == n and best_c < phase_best_c:
            phase_best_h = best_h.copy()
            phase_best_c = best_c
            h_cur = best_h.copy()

        remaining = max(10.0, phase_end - phase_start)
        if n <= 1600:
            smooth_frac = 0.56
            binary_frac = 0.30
            smooth2_frac = 0.14
        elif n <= 2048:
            smooth_frac = 0.50
            binary_frac = 0.35
            smooth2_frac = 0.15
        else:
            smooth_frac = 0.54
            binary_frac = 0.26
            smooth2_frac = 0.20

        dl_smooth = phase_start + smooth_frac * remaining
        dl_binary = dl_smooth + binary_frac * remaining
        dl_finish = phase_end

        try:
            h_s, c_s = smooth_optimize(h_cur, n, min(dl_smooth, final_dl - 35.0))
        except Exception:
            h_s, c_s = h_cur.copy(), fast_c5(h_cur, n)

        if c_s < phase_best_c:
            phase_best_c = c_s
            phase_best_h = h_s.copy()
        if c_s < best_c:
            best_c = c_s
            best_h = h_s.copy()
            best_n = n

        if n <= 3200:
            cd = direct_c5(phase_best_h)
            if cd < phase_best_c:
                phase_best_c = cd
                phase_best_h = phase_best_h.copy()
            cd_best = direct_c5(best_h)
            if cd_best < best_c:
                best_c = cd_best

        h_for_binary = phase_best_h
        if n <= 3200 and time.time() < min(dl_binary, final_dl - 30.0) - 5.0:
            seedb = int(seed) + 1000 + int(n)
            try:
                hb = binarize(h_for_binary, n, seed=seedb)
                h_b, c_b = greedy_binary(hb, n, min(dl_binary, final_dl - 30.0), seed=seedb + 1)
            except Exception:
                h_b, c_b = h_for_binary.copy(), fast_c5(h_for_binary, n)

            if c_b < phase_best_c:
                phase_best_c = c_b
                phase_best_h = h_b.copy()
            if c_b < best_c:
                best_c = c_b
                best_h = h_b.copy()
                best_n = n

            if time.time() < min(dl_finish, final_dl - 25.0) - 25.0:
                try:
                    h_a, c_a = anneal_binary(
                        h_b.astype(np.int8),
                        n,
                        min(dl_finish, final_dl - 25.0) - 20.0,
                        seed=seedb + 11
                    )
                except Exception:
                    h_a, c_a = h_b.copy(), fast_c5(h_b, n)
                if c_a < phase_best_c:
                    phase_best_c = c_a
                    phase_best_h = h_a.copy()
                if c_a < best_c:
                    best_c = c_a
                    best_h = h_a.copy()
                    best_n = n
                h_for_smooth2 = h_a if c_a < c_b else h_b
            else:
                h_for_smooth2 = h_b

            if time.time() < min(dl_finish, final_dl - 20.0) - 10.0:
                try:
                    h_sb, c_sb = smooth_optimize(
                        h_for_smooth2,
                        n,
                        min(dl_finish, final_dl - 20.0),
                        betas=(50.0, 150.0, 400.0, 150.0, 800.0, 2500.0)
                    )
                except Exception:
                    h_sb, c_sb = h_for_smooth2.copy(), fast_c5(h_for_smooth2, n)

                if c_sb < phase_best_c:
                    phase_best_c = c_sb
                    phase_best_h = h_sb.copy()
                if c_sb < best_c:
                    best_c = c_sb
                    best_h = h_sb.copy()
                    best_n = n

        if n <= 3200 and time.time() < final_dl - 20.0:
            cd = direct_c5(phase_best_h)
            if cd < phase_best_c:
                phase_best_c = cd
            cd_best = direct_c5(best_h)
            if cd_best < best_c:
                best_c = cd_best

        carry_h = phase_best_h.copy()
        carry_n = n

    if time.time() < final_dl - 30.0 and best_n > 0:
        try:
            h_extra, c_extra = smooth_optimize(
                best_h, best_n, min(final_dl - 5.0, time.time() + 20.0),
                betas=(100.0, 400.0, 1200.0, 4000.0)
            )
            if c_extra < best_c:
                best_c = c_extra
                best_h = h_extra.copy()
        except Exception:
            pass

    try:
        best_h = normalize_sum(best_h, best_n)
        best_c = direct_c5(best_h)
    except Exception:
        pass

    return best_h, float(best_c), int(best_n)
