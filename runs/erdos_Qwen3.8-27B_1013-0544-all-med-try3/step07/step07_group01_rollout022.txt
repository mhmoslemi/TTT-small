            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    shift = alpha * (c - c.max())
    mx = shift.max()
    w = np.exp(shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def lbfgs_round(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h, jac=smooth_grad,
                  args=(n, alpha), method='L-BFGS-B',
                  bounds=bounds,
                  options={'maxiter': maxiter, 'ftol': 1e-18,
                           'gtol': 1e-14, 'maxfun': maxiter * 5})
    return project_feasible(res.x, n)

def multi_alpha_optimize(h, n, deadline, alphas=(100, 300, 1000, 3000, 10000, 30000), 
                          perturb=0.02, seed=42):
    """Run L-BFGS with increasing alpha, adding perturbations between rounds."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 5:
            break
        h_cur = best_h.copy()
        if i > 0 and perturb > 0:
            noise = rng.uniform(-perturb, perturb, n)
            h_cur = project_feasible(h_cur + noise, n)
        
        h_new = lbfgs_round(h_cur, n, alpha, maxiter=200)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def generate_binary_start(n, seed):

    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    # Smooth slightly
    from scipy.ndimage import uniform_filter1d
    h = uniform_filter1d(h, size=3, mode='wrap')

def generate_periodic_start(n, period, seed=0):

def pair_search_fast(h, n, deadline, deltas):
    """Fast pair search with limited radius."""
    best_c = compute_c5(h, n)
    radius = 30
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] >= d and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

**Alternative: Direct subgradient method**

The function f(h) = max_k c_k(h) is convex (max of linear functions... wait, c_k(h) is quadratic in h, not linear). So f is not convex. The log-sum-exp is a smooth approximation.

Actually, c_k(h) = sum_i h[i]*(1-h[i+k]) is a quadratic form in h. So f(h) = max_k c_k(h) is the max of quadratic functions, which is convex! (Max of convex functions is convex.)

Wait, is c_k(h) convex? c_k(h) = sum_i h[i] - sum_i h[i]*h[i+k] = (sum of h[i]) - (quadratic form).

The quadratic form sum_i h[i]*h[i+k] is NOT convex in general (it's an indefinite quadratic form). So c_k(h) is NOT convex. Therefore f(h) is NOT convex either.

This means the problem is non-convex, and we can get stuck in local optima. The log-sum-exp + L-BFGS approach is a heuristic that works well in practice.

**Another alternative: CP** (convergent programming) or **S**uccessive **A**pproximation

At each iteration:
1. Evaluate c_k(h) for all k
2. Identify the "active" set A = {k : c_k(h) ≈ max}
3. Solve: min_{h} sum_{k∈A} c_k(h) subject to constraints
   (This is a quadratic program with box constraints)
4. Update A based on the new h
5. Repeat

This is essentially the "active set" method for min-max problems. It might be more efficient because we're solving QPs instead of using L-BFGS on a smoothed objective.

    dx = 2.0 / n

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

            if tot < 1e-14:

            if tot < 1e-14:

            diverse_starts.append(gen_staircase_start(n_best, 8 + s * 4, seed + 300 + s))

        for h_s in diverse_starts:
            if time.time() > deadline - 60:
                break
            t_alloc = min(time.time() + 20, deadline - 60)
            h_s, c_s = multi_alpha_optimize(
                h_s, n_best, t_alloc,
                alphas=(500, 3000, 15000),
                perturb=0.0, seed=seed + 400)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 6: Final high-alpha refinement + SLSQP
    n_f = len(best_h)
    if time.time() < deadline - 40:
        t6 = min(time.time() + 80, deadline - 20)
        best_h, best_c = multi_alpha_optimize(
            best_h, n_f, t6,
            alphas=(30000, 100000, 500000),
            perturb=0.003, seed=seed + 500)
        if time.time() < deadline - 15:
            h_sl = slsqp_round(best_h, n_f, 100000, deadline - 10, maxiter=500)
            c_sl = compute_c5(h_sl, n_f)
            if c_sl < best_c:
                best_c = c_sl
                best_h = h_sl

    # Phase 7: Pair search fine-tuning
    if time.time() < deadline - 2:
        best_h, best_c = pair_search_quick(
            best_h, len(best_h), deadline - 1,
            [0.003, 0.001, 0.0005, 0.0002], radius=20)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
