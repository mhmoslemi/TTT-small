def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_lp(h0, n):
    """Build the LP for SCA iteration."""
    dx = 2.0 / n
    nv = n + 1  # n h-variables + t
    c_obj = np.zeros(nv)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: B_k @ h - t <= -C_k for each shift k
    n_shifts = 2 * n - 1
    A_ub = np.zeros((n_shifts, nv))
    b_ub = np.zeros(n_shifts)
    
    for s in range(n_shifts):
        k = s - (n - 1)  # shift value
        if k >= 0:
            nv2 = n - k
            for j in range(nv2):
                A_ub[s, j] += dx * (1.0 - h0[j + k])
                A_ub[s, j + k] += -dx * h0[j]
            A_ub[s, n] = -1.0
            b_ub[s] = -dx * np.dot(h0[0:nv2], h0[k:n])
        else:
            m = -k
            nv2 = n - m
            for j in range(nv2):
                A_ub[s, j + m] += dx * (1.0 - h0[j])
                A_ub[s, j] += -dx * h0[j + m]
            A_ub[s, n] = -1.0
            b_ub[s] = -dx * np.dot(h0[m:n], h0[0:nv2])
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, 2.0)]
    
    return c_obj, A_ub, b_ub, A_eq, b_eq, bounds

def sca_iteration(h0, n, damping=1.0, max_iter=200, deadline=float('inf')):
    """Run SCA (successive convex approximation) iterations."""
    h = h0.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_obj, A_ub, b_ub, A_eq, b_eq, bounds = build_lp(h, n)
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        h_new = res.x[:n]
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        
        if damping < 1.0:
            h = (1 - damping) * h + damping * h_new
            h = project_feasible(h, n)
        else:
            h = h_new
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        # Check convergence
        if np.max(np.abs(h_new - h0)) < 1e-10:
            break
        h0 = h_new
    
    return best_h, best_c
