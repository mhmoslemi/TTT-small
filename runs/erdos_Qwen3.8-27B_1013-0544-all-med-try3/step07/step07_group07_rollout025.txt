def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def make_swap_delta(i, j, h, n, m_range):
    """Compute delta in corr when h[i]:0->1 and h[j]:1->0."""
    delta = np.zeros(2 * n - 1)
    
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += (1.0 - h[im[msk]])
    
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= (1.0 - h[jm[msk]])
    
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    
    return delta

def anneal(h, n, deadline, rng, T0=0.008, n_steps=200000, kick_size=4, kick_T=0.005):
    """Main annealing loop with periodic kicks."""
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_idx = np.where(h == 1)[0].copy()
    zeros_idx = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    
    T = T0
    cool_per_step = np.exp(np.log(0.1) / n_steps)  # cool to 10% over n_steps
    no_improve = 0
    max_no_improve = 30000
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        # Random swap
        zi = rng.randint(len(zeros_idx))
        oi = rng.randint(len(ones_idx))
        i = zeros_idx[zi]
        j = ones_idx[oi]
        
        delta = make_swap_delta(i, j, h, n, m_range)
        nc = (corr + delta).max() * dx
        dc = nc - cur_c
        
        if dc <= 0 or (T > 1e-8 and rng.random() < np.exp(-dc / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_idx[zi] = j
            ones_idx[oi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cool_per_step
        
        # Kick if stuck
        if no_improve > max_no_improve:
            for _ in range(kick_size):
                zi = rng.randint(len(zeros_idx))
                oi = rng.randint(len(ones_idx))
                i = zeros_idx[zi]
                j = ones_idx[oi]
                delta = make_swap_delta(i, j, h, n, m_range)
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = (corr).max() * dx
                zeros_idx[zi] = j
                ones_idx[oi] = i
            T = kick_T
            no_improve = 0
    
    return best_h, best_c, h, corr, ones_idx, zeros_idx
