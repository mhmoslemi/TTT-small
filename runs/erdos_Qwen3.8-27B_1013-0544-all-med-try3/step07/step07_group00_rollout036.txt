    # at any x where it would be the max

        bi = base_sorted[idx]

            last_b = hull[-1][1]
            last_xl = hull[-1][2]

                # Same slope, keep the one with higher intercept
                if bi > last_b:
                    hull.pop()
                    continue
                else:
                    break

            # ai*x + bi = last_a*x + last_b
            # x = (last_b - bi) / (ai - last_a)
            x_int = (last_b - bi) / (ai - last_a)

                hull.pop()
            else:
                # This line becomes active at x_int
                break

            hull.append((ai, bi, -np.inf))

            bi = base_sorted[idx]
            last_a = hull[-1][0]
            last_b = hull[-1][1]
            if ai == last_a:
                if bi > last_b:
                    hull.pop()
                    if not hull:
                        hull.append((ai, bi, -np.inf))
                    else:
                        # Recompute x_intersect with new last
                        last_a = hull[-1][0]
                        last_b = hull[-1][1]
                        x_int = (last_b - bi) / (ai - last_a)
                        hull.append((ai, bi, x_int))
                # else: skip this line
            else:
                x_int = (last_b - bi) / (ai - last_a)
                hull.append((ai, bi, x_int))

    # The upper envelope is piecewise linear, with slope = a_i for each segment
    # The minimum is where the slope changes from negative to positive
    
    best_x = lo
    best_val = np.max(base + a * lo)

        ai, bi, x_left = line
        # This line is active for x >= x_left (until the next line takes over)
        if x_left > hi:
            break
        # Check the point x = max(lo, x_left)
        x_check = max(lo, x_left)
        val = ai * x_check + bi

        # For the upper envelope, by construction it should be
        if val < best_val - 1e-15:
            best_val = val
            best_x = x_check
        # Also check x = hi if this line is active there
        # The last line on the hull is active up to +inf

    if val_hi < best_val - 1e-15:
        best_val = val_hi
        best_x = hi
    
    return best_x

def upper_envelope_and_min(base, a, lo=0.0, hi=1.0):
    """
    Find min_{x in [lo,hi]} max_k (base[k] + a[k]*x)
    base, a: 1D arrays of same length
    Returns (x_opt, f_opt)
    """
    L = len(base)
    if L == 0:
        return lo, 0.0
    if L == 1:
        if a[0] >= 0:
            return lo, base[0] + a[0]*lo
        else:
            return hi, base[0] + a[0]*hi
    
    # Sort by slope, break ties by intercept (higher first)
    order = np.lexsort((-base, a))  # sort by a ascending, then base descending
    a_s = a[order]
    b_s = base[order]

    # Each entry: (slope, intercept, x_start) where x_start is where this line becomes the max
    hull_slopes = []
    hull_intercepts = []
    hull_starts = []
    
    for i in range(L):
        ai, bi = a_s[i], b_s[i]
        
        # Remove lines from the top of the hull that are dominated
        while len(hull_slopes) > 0:
            # Check if new line dominates the top line
            # Intersection of new line and top line:
            # ai*x + bi = hull_slopes[-1]*x + hull_intercepts[-1]
            # x = (hull_intercepts[-1] - bi) / (ai - hull_slopes[-1])
            if ai == hull_slopes[-1]:
                # Same slope: keep higher intercept
                if bi > hull_intercepts[-1]:
                    hull_slopes.pop()
                    hull_intercepts.pop()
                    hull_starts.pop()
                    continue
                else:
                    break  # New line is dominated
            x_int = (hull_intercepts[-1] - bi) / (ai - hull_slopes[-1])
            if x_int <= hull_starts[-1]:
                # New line becomes better before the current top line is active
                hull_slopes.pop()
                hull_intercepts.pop()
                hull_starts.pop()
            else:
                break
        
        if len(hull_slopes) == 0:
            hull_slopes.append(ai)
            hull_intercepts.append(bi)
            hull_starts.append(-np.inf)
        else:
            x_int = (hull_intercepts[-1] - bi) / (ai - hull_slopes[-1])
            hull_slopes.append(ai)
            hull_intercepts.append(bi)
            hull_starts.append(x_int)
    
    # Now find the minimum over [lo, hi]
    # The upper envelope is: for each segment i, line i is active for x in [hull_starts[i], hull_starts[i+1])
    # (with the last segment active for x in [hull_starts[-1], +inf))
    
    L_hull = len(hull_slopes)
    best_x = lo
    best_f = np.inf
    
    for i in range(L_hull):
        x_start = hull_starts[i]
        x_end = hull_starts[i+1] if i+1 < L_hull else np.inf
        
        # Clamp to [lo, hi]
        x1 = max(lo, x_start)
        x2 = min(hi, x_end)
        
        if x1 >= x2:
            continue
        
        # The line is: f(x) = hull_slopes[i]*x + hull_intercepts[i]
        # Minimum of a linear function on [x1, x2]:
        if hull_slopes[i] > 0:
            x_opt = x1
        elif hull_slopes[i] < 0:
            x_opt = x2
        else:
            x_opt = x1
        
        f_opt = hull_slopes[i]*x_opt + hull_intercepts[i]
        
        if f_opt < best_f:
            best_f = f_opt
            best_x = x_opt
    
    return best_x, best_f
