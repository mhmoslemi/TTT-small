    idx = best_j + k_range - n + 1

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def best_improvement_search(h, n, deadline, max_iters=50000):
    """Best-improvement local search for binary h."""
    dx = 2.0 / n
    L = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    h = h.astype(float).copy()
    
    # Precompute index arrays
    rows = np.arange(n)[:, None]
    cols = np.arange(L)[None, :]
    idx1 = rows + n - 1 - cols  # for g[p+n-1-k]
    idx2 = rows + cols - n + 1  # for h[p+k-n+1]
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        nz, no = len(zeros), len(ones)
        g = 1 - h
        
        # Compute F: F[i, k] = g[i+n-1-k] - h[i+k-n+1] (with bounds)
        F = np.zeros((n, L))
        F[m1] = g[idx1[m1]]
        F[m2] -= h[idx2[m2]]
        
        # Compute G: G[i, k] = corr[k] - g[i+n-1-k] + h[i+k-n+1] (with bounds)
        G = np.zeros((n, L))
        G += corr[None, :]
        G[m1] -= g[idx1[m1]]
        G[m2] += h[idx2[m2]]
        
        # Find best swap
        F_sub = F[zeros, :]  # (nz, L)
        G_sub = G[ones, :]   # (no, L)
        
        # new_max[i,j] = max_k F_sub[i,k] + G_sub[j,k]
        # Vectorized: for each i, compute max over k of F_sub[i] + G_sub[j] for all j
        new_max = np.zeros((nz, no))
        for i_idx in range(nz):
            new_max[i_idx, :] = np.max(F_sub[i_idx, :][None, :] + G_sub, axis=1)
        
        best_flat = np.argmin(new_max)
        best_ii = best_flat // no
        best_jj = best_flat % no
        best_val = new_max[best_ii, best_jj]
        
        if best_val >= cur_max - 1e-10:
            break  # No improvement possible
        
        best_i = zeros[best_ii]
        best_j = ones[best_jj]
        
        # Apply swap
        h[best_i] = 1.0
        h[best_j] = 0.0
        g = 1 - h
        
        # Update correlation
        # Recompute (for correctness, since we need exact values)
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        cur_max = corr.max()
    
    return h, cur_max * dx
