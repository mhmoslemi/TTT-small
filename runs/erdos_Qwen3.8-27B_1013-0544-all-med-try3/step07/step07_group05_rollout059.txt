import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(h_p < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h_p > 1e-14)[0]
        if len(idx) > 0:
            h_p[idx] += diff / len(idx)
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def smooth_max_value(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    emax = shifted.max()
    return m + (emax + np.log(np.sum(np.exp(shifted - emax)))) / alpha

def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    logits = alpha * c
    logits -= logits.max()
    w = np.exp(logits)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    A = 1.0 - h
    B = h
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-18:
            continue
        s = k - (n - 1)
        j1_lo = max(0, -s)
        j1_hi = min(n, n - s)
        if j1_lo < j1_hi:
            grad[j1_lo:j1_hi] += wk * A[j1_lo + s:j1_hi + s]
        j2_lo = max(0, s)
        j2_hi = min(n, n + s)
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] -= wk * B[j2_lo - s:j2_hi - s]
    return grad

def ln_obj_func(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    emax = shifted.max()
    return m + (emax + np.log(np.sum(np.exp(shifted - emax)))) / alpha

def _con_eq(x, n):
    return float(x.sum() - n / 2.0)

def _con_eq_jac(x, n):
    return np.ones(len(x))

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac, 'args': (n,)}]
    try:
        res = minimize(ln_obj_func, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

def fista_opt(h0, n, alpha, deadline, max_iter=100000):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    h_curr = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    step = 0.05
    n_reject_streak = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_gradient(h_prev, n, alpha)
        z = h_prev - step * g
        h_new = project_feasible(z, n)
        f_new = smooth_max_value(h_new, n, alpha)
        f_prev = smooth_max_value(h_prev, n, alpha)
        gnorm2 = np.sum(g * g)

        if f_new <= f_prev + 0.1 * step * gnorm2:
            n_reject_streak = max(0, n_reject_streak - 1)
            if n_reject_streak == 0 and step < 0.3:
                step = min(step * 1.05, 0.3)
            h_curr = h_new
        else:
            step *= 0.5
            n_reject_streak += 1
            if n_reject_streak > 20:
                break
            h_curr = h_prev
            t = 1.0

        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c - 1e-15:
            best_c = c_curr
            best_h = h_curr.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (1.0 - t) / t_new
        h_prev = h_curr + theta * (h_curr - h_prev)
        h_prev = project_feasible(h_prev, n)
        t = t_new

    return best_h, best_c

def local_pair_search(h, n, deadline, deltas, radius, max_passes=500):
    best_c = compute_c5(h, n)
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if j == i:
                        continue
                    hi, hj = h[i] + d, h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    if j > i:
                        hi2, hj2 = h[i] - d, h[j] + d
                        if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                            h[i], h[j] = hi2, hj2
                            c = compute_c5(h, n)
                            if c < best_c - 1e-16:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi2 + d, hj2 - d
            if not improved:
                break
    return h, best_c
