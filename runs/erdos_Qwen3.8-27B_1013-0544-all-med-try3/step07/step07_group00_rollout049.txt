from scipy.optimize import minimize, linprog

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

        if wmm < 1e-15:

        wm = wmm * dx
        if k >= 0:

            grad[k:n] += wm * (-x[0:nv])

    return grad

    else:

    return float(val)

    vals = np.array([c[i] for i in active_idx])
    mx = vals.max()

def run_lbfgs(h, n, deadline, alphas, mu=100.0, maxiter=3000):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _correlation_grad_component(grad, x, n, shift_idx, weight):
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += weight * (1.0 - x[k:n])
        grad[k:n] += weight * (-x[0:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += weight * (1.0 - x[0:nv])
        grad[0:nv] += weight * (-x[kk:n])

def full_smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wmm = w[mm]
        if wmm < 1e-16:
            continue
        _correlation_grad_component(grad, x, n, mm, wmm * dx)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in active], dtype=float)
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in active], dtype=float)
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-16:
            continue
        _correlation_grad_component(grad, x, n, idx, wi * dx)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_round(h, n, deadline, alphas, mu, maxiter, maxfun_per):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(full_smooth_obj, h, jac=full_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxfun_per, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def active_set_round(h, n, deadline, top_k, alphas, mu, maxiter, maxfun_per):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for _ in range(12):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(active_obj, h, jac=active_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxfun_per, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def targeted_local_search(h, n, deadline):
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    top_shifts = np.argsort(-c_vec)[:8]
    active_idx = set()
    for s in top_shifts:
        k = s - (n - 1)
        if k >= 0:
            active_idx.update(range(0, n - k))
            active_idx.update(range(k, n))
        else:
            kk = -k
            active_idx.update(range(kk, n))
            active_idx.update(range(0, n - kk))
    active_list = list(active_idx)
    active_list.sort(key=lambda i: -abs(h[i] - 0.5))
    active_list = active_list[:min(120, len(active_list))]
    
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in active_list:
                if time.time() > deadline:
                    return h, best_c
                hi_val = h[i]
                for j in active_list:
                    if i == j:
                        continue
                    if hi_val + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            hi_val = h[i]
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            hi_val = h[i]
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def perturbed_restart(h, n, rng, sigma, frac):
    h_new = h.copy()
    n_idx = max(1, int(n * frac))
    idx = rng.choice(n, n_idx, replace=False)
    h_new[idx] += rng.normal(0, sigma, size=n_idx)
    h_new = np.clip(h_new, 0, 1)
    return project_feasible(h_new, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
