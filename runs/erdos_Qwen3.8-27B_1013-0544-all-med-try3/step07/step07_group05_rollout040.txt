    gr=(sqrt5-1)/2
    c=b-gr*(b-a); d=a+gr*(b-a)
    fc=f(c); fd=f(d)

```python
import numpy as np
import time
import math
from scipy.optimize import minimize


SM_CACHE = {"x": None, "n": -1, "a": -1.0, "val": 0.0, "g": None}


def _arrays_equal(a, b):
    if a is b:
        return True
    if a is None or b is None:
        return False
    if a.shape != b.shape:
        return False
    try:
        return bool(np.array_equal(a, b))
    except Exception:
        return False


def project_feasible(h, n):
    v = np.asarray(h, dtype=float)
    if v.size != n:
        v = np.resize(v, (n,))
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    target = 0.5 * float(n)
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += diff / float(cnt)
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / float(n)
            x = np.clip(x, 0.0, 1.0)
    return x


def raw_c(h):
    return np.correlate(h, 1.0 - h, mode="full")


def eval_solution(h):
    n = len(h)
    r = raw_c(h)
    return float(r.max()) * (2.0 / float(n))


def sum_con(x, n):
    return float(np.sum(x)) - 0.5 * float(n)


def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)


def smooth_point(x, n, alpha):
    dx = 2.0 / float(n)
    r = raw_c(x)
    T = r * dx
    z = alpha * T
    m = float(np.max(z))
    s = np.exp(np.clip(z - m, -50.0, 0.0))
    ss = float(np.sum(s))
    if ss <= 0.0:
        s = np.ones_like(s)
        ss = float(s.size)
    val = (m + math.log(ss)) / alpha
    pi = s / ss
    off = n - 1
    first = np.convolve(1.0 - x, pi, mode="full")[off:off + n]
    second = np.convolve(pi[::-1], x, mode="full")[off:off + n]
    grad = dx * (first - second)
    return val, grad


def smooth_fun(x, n, alpha):
    c = SM_CACHE
    if c["n"] == n and abs(c["a"] - alpha) < 1e-12 and _arrays_equal(c["x"], x):
        return c["val"]
    val, g = smooth_point(x, n, alpha)
    c["x"] = np.asarray(x, dtype=float).copy()
    c["n"] = int(n)
    c["a"] = float(alpha)
    c["val"] = val
    c["g"] = g
    return val


def smooth_jac(x, n, alpha):
    c = SM_CACHE
    if c["n"] == n and abs(c["a"] - alpha) < 1e-12 and _arrays_equal(c["x"], x):
        return c["g"]
    val, g = smooth_point(x, n, alpha)
    c["x"] = np.asarray(x, dtype=float).copy()
    c["n"] = int(n)
    c["a"] = float(alpha)
    c["val"] = val
    c["g"] = g
    return g


def slsqp_stage(h0, n, deadline, rng):
    h = project_feasible(h0, n)
    r = raw_c(h)
    best_raw = float(r.max())
    best_c = best_raw * (2.0 / float(n))
    best = h.copy()

    if n > 600:
        return best, best_c

    if n <= 260:
        alphas = (200.0, 1200.0, 7000.0)
        maxiters = (220, 160, 100)
    else:
        alphas = (400.0, 2500.0)
        maxiters = (120, 70)

    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    bounds = [(0.0, 1.0) for _ in range(n)]

    for a, it in zip(alphas, maxiters):
        if time.time() > deadline - 3.0:
            break
        SM_CACHE["x"] = None
        SM_CACHE["n"] = -1
        SM_CACHE["a"] = -1.0
        SM_CACHE["val"] = 0.0
        SM_CACHE["g"] = None
        try:
            res = minimize(
                smooth_fun,
                h,
                args=(n, a),
                jac=smooth_jac,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(it), "ftol": 1e-12, "disp": False},
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                ht = project_feasible(res.x, n)
                ct = eval_solution(ht)
                if np.isfinite(ct) and ct < best_c:
                    best_c = ct
                    best = ht
                    h = ht
                elif np.isfinite(ct) and ct <= best_c + 1e-3:
                    h = ht
        except Exception:
            pass

    h = project_feasible(best, n)
    ct = eval_solution(h)
    if ct < best_c:
        best_c = ct
        best = h
    return best, best_c


def gradient_from_pi(h, n, pi):
    off = n - 1
    first = np.convolve(1.0 - h, pi, mode="full")[off:off + n]
    second = np.convolve(pi[::-1], h, mode="full")[off:off + n]
    return first - second


def _pair_f_and_deriv(t, c_aff, L_aff, Q_aff, unaff):
    vals = c_aff + L_aff * t + Q_aff * (t * t)
    vm = float(vals.max())
    f = unaff if unaff > vm else vm
    tol = 1e-11 + 1e-12 * max(1.0, abs(f))
    d = -1e300
    if unaff >= f - tol:
        d = 0.0
    if vm >= f - tol:
        act = vals >= (f - tol)
        if np.any(act):
            dd = float(np.max(L_aff[act] + 2.0 * Q_aff[act] * t))
            if dd > d:
                d = dd
    return f, d


def pair_line_search(h, c, max_raw, i, j, idx, off, K, d0_skip, improve_tol):
    tmax = min(1.0 - h[i], h[j])
    if tmax <= 1e-13:
        return None

    one_minus = 1.0 - h
    L = np.zeros(K, dtype=float)
    L += np.bincount(i - idx + off, weights=one_minus, minlength=K)
    L -= np.bincount(idx - i + off, weights=h, minlength=K)
    L -= np.bincount(j - idx + off, weights=one_minus, minlength=K)
    L += np.bincount(idx - j + off, weights=h, minlength=K)

    Q = np.zeros(K, dtype=float)
    Q[off] -= 2.0
    ji = i - j + off
    ij = j - i + off
    if ji != off:
        Q[ji] += 1.0
    if ij != off:
        Q[ij] += 1.0

    eps = 1e-12
    L[np.abs(L) < eps] = 0.0
    Q[np.abs(Q) < eps] = 0.0

    mask = (L != 0.0) | (Q != 0.0)
    if not np.any(mask):
        return None

    unmask = ~mask
    if np.any(unmask):
        unaff = float(np.max(c[unmask]))
    else:
        unaff = -1e300

    c_aff = c[mask]
    L_aff = L[mask]
    Q_aff = Q[mask]

    act_tol = max(1e-10, 1e-11 * max_raw)
    act0 = c >= (max_raw - act_tol)
    if np.any(act0):
        d0 = float(np.max(L[act0]))
    else:
        d0 = 0.0
    if d0 > d0_skip:
        return None

    f0 = max_raw
    best_f = f0
    best_t = 0.0
    found = False

    for frac in (0.01, 0.05, 0.20, 0.50, 1.00):
        t = frac * tmax
        f, _d = _pair_f_and_deriv(t, c_aff, L_aff, Q_aff, unaff)
        if f + improve_tol < best_f:
            best_f = f
            best_t = t
            found = True

    if not found:
        return None

    f_hi, d_hi = _pair_f_and_deriv(tmax, c_aff, L_aff, Q_aff, unaff)
    if f_hi + 1e-14 < best_f:
        best_f = f_hi
        best_t = tmax

    if d_hi > -1e-12:
        lo = 0.0
        hi = tmax
        for _ in range(35):
            mid = 0.5 * (lo + hi)
            f_mid, d_mid = _pair_f_and_deriv(mid, c_aff, L_aff, Q_aff, unaff)
            if f_mid + 1e-14 < best_f:
                best_f = f_mid
                best_t = mid
            if d_mid < 0.0:
                lo = mid
            else:
                hi = mid
            if hi - lo <= 1e-13 * max(1.0, tmax):
                break
        f_mid, _ = _pair_f_and_deriv(0.5 * (lo + hi), c_aff, L_aff, Q_aff, unaff)
        if f_mid + 1e-14 < best_f:
            best_f = f_mid
            best_t = 0.5 * (lo + hi)

    if best_f < f0 - improve_tol:
        return (float(best_t), L, Q, float(best_f))
    return None


def default_directions(n):
    if n <= 180:
        return 2600
    if n <= 350:
        return 1800
    if n <= 700:
        return 900
    if n <= 1024:
        return 520
    return 350


def select_directions(h, g, n, M_total, rng):
    if M_total <= 0:
        return []
    size = n * n
    M_total = int(min(M_total, size - 1))
    if M_total <= 0:
        return []

    M_act = int(0.55 * M_total)
    M_bin = int(0.20 * M_total)
    M_rand = M_total - M_act - M_bin
    if M_act < 0:
        M_act = 0
    if M_bin < 0:
        M_bin = 0
    if M_rand < 0:
        M_rand = 0

    dirs = []

    if M_act > 0:
        S = g[:, None] - g[None, :]
        np.fill_diagonal(S, np.inf)
        flat = S.ravel()
        if M_act < size - 1:
            idx = np.argpartition(flat, M_act - 1)[:M_act]
        else:
            idx = np.delete(np.arange(size), np.arange(0, size, n + 1))
        ia = idx // n
        ja = idx % n
        dirs.extend(zip(ia.tolist(), ja.tolist()))
        del S

    if M_bin > 0:
        B = h[:, None] - h[None, :]
        np.fill_diagonal(B, -np.inf)
        flat = -B.ravel()
        if M_bin < size - 1:
            idx = np.argpartition(flat, M_bin - 1)[:M_bin]
        else:
            idx = np.delete(np.arange(size), np.arange(0, size, n + 1))
        ia = idx // n
        ja = idx % n
        dirs.extend(zip(ia.tolist(), ja.tolist()))
        del B

    if M_rand > 0:
        for _ in range(M_rand):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
                if i == j:
                    j = (j + 2) % n
            dirs.append((i, j))

    if len(dirs) > 1:
        seen = set()
        uniq = []
        for d in dirs:
            key = int(d[0]) * n + int(d[1])
            if key not in seen:
                seen.add(key)
                uniq.append(d)
        dirs = uniq

    if len(dirs) > M_total:
        order = rng.permutation(len(dirs))
        dirs = [dirs[int(i)] for i in order[:M_total]]
    else:
        rng.shuffle(dirs)

    return dirs


def bit_parity(i):
    p = 0
    while i:
        p ^= (i & 1)
        i >>= 1
    return p


def vdc_value(i):
    i += 1
    f = 0.5
    r = 0.0
    while i:
        if i & 1:
            r += f
        i >>= 1
        f *= 0.5
    return r


def alternative_starts(n, rng, limit=4):
    if n <= 0:
        return []
    cands = []

    h = np.zeros(n, dtype=float)
    idx = np.arange(n)
    rng.shuffle(idx)
    half = n // 2
    h[idx[:half]] = 1.0
    if n % 2 == 1:
        h[idx[half]] = 0.5
    cands.append(h)

    h = np.zeros(n, dtype=float)
    for i in range(n):
        if bit_parity(i) == 0:
            h[i] = 1.0
    cands.append(h)

    vals = [vdc_value(i) for i in range(n)]
    order = np.argsort(vals)
    h = np.zeros(n, dtype=float)
    h[order[:half]] = 1.0
    if n % 2 == 1:
        h[order[half]] = 0.5
    cands.append(h)

    alpha = (math.sqrt(5.0) - 1.0) / 2.0
    h = np.zeros(n, dtype=float)
    for i in range(n):
        y = (i + 0.5) * alpha
        if (y - math.floor(y)) < 0.5:
            h[i] = 1.0
    cands.append(h)

    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    for m in (1, 2, 3, 5):
        h = 0.5 + 0.5 * np.sin(2.0 * math.pi * float(m) * x)
        cands.append(h)

    res = []
    for h in cands:
        h = project_feasible(h, n)
        c = eval_solution(h)
        inserted = False
        for pos in range(len(res)):
            if c < res[pos][0]:
                res.insert(pos, (c, h))
                inserted = True
                break
        if not inserted:
            res.append((c, h))

    if limit and len(res) > limit:
        res = res[:limit]
    return res


def interpolate_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h, n_new)
    if n_old == 1:
        return project_feasible(np.full(n_new, float(h[0])), n_new)
    x_old = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)


def pair_local_optimize(h0, n, deadline, rng, max_passes=70, M_total=None):
    h = project_feasible(h0, n)
    r = raw_c(h)
    max_raw = float(r.max())
    best_raw = max_raw
    best_h = h.copy()

    if M_total is None:
        M_total = default_directions(n)
    M_total = max(50, int(M_total))

    idx = np.arange(n, dtype=int)
    off = n - 1
    K = 2 * n - 1

    d0_base = max(1e-9, 1e-10 * max_raw)
    improve_base = max(1e-10, 1e-10 * max_raw)

    for p in range(max_passes):
        if time.time() > deadline - 2.0:
            break

        r = raw_c(h)
        max_raw = float(r.max())
        if max_raw < best_raw:
            best_raw = max_raw
            best_h = h.copy()

        act_tol = max(1e-8, 1e-6 * max_raw)
        act = np.where(r >= (max_raw - act_tol))[0]
        if act.size == 0:
            act = np.array([int(np.argmax(r))], dtype=int)

        pi = np.zeros(K, dtype=float)
        if (p % 3 == 0) or act.size <= 15:
            z = 900.0 * (r - max_raw)
            w = np.exp(np.clip(z, -50.0, 0.0))
            ws = float(np.sum(w))
            if ws > 0.0:
                pi = w / ws
            else:
                pi[act[0]] = 1.0
        else:
            m = max(1, min(int(act.size), max(5, int(act.size) // 2)))
            chosen = act[rng.choice(act.size, size=m, replace=False)]
            pi[chosen] = 1.0 / float(m)

        g = gradient_from_pi(h, n, pi)

        M_use = M_total
        if p == 0:
            M_use = min(n * (n - 1), int(1.5 * M_total))

        dirs = select_directions(h, g, n, M_use, rng)
        improved = False

        for i, j in dirs:
            if time.time() > deadline - 1.0:
                break
            res = pair_line_search(h, r, max_raw, int(i), int(j), idx, off, K, d0_base, improve_base)
            if res is None:
                continue
            tau, L, Q, pf = res
            if pf + 1e-12 >= max_raw:
                continue

            oi = int(i)
            oj = int(j)
            old_i = h[oi]
            old_j = h[oj]

            h[oi] = old_i + tau
            h[oj] = old_j - tau
            if h[oi] > 1.0:
                h[oi] = 1.0
            if h[oj] < 0.0:
                h[oj] = 0.0

            r += L * tau + Q * (tau * tau)
            new_max = float(r.max())

            if new_max < max_raw - 1e-12:
                improved = True
                max_raw = new_max
                if max_raw < best_raw:
                    best_raw = max_raw
                    best_h = h.copy()
            else:
                h[oi] = old_i
                h[oj] = old_j
                r -= L * tau + Q * (tau * tau)

        if abs(float(h.sum()) - 0.5 * float(n)) > 1e-8:
            hp = project_feasible(h, n)
            rp = raw_c(hp)
            mp = float(rp.max())
            if mp < max_raw + max(1e-12, 1e-10 * max_raw):
                h = hp
                r = rp
                max_raw = mp
                if max_raw < best_raw:
                    best_raw = max_raw
                    best_h = h.copy()

        if not improved:
            break

    h = project_feasible(best_h, n)
    c = eval_solution(h)
    return h, c


def optimize_stage(h0, n, stage_end, rng, M_total=None, do_slsqp=True):
    h = project_feasible(h0, n)
    c0 = eval_solution(h)
    best_h = h.copy()
    best_c = c0

    if time.time() >= stage_end:
        return best_h, best_c

    rem = stage_end - time.time()
    M_use = M_total if M_total is not None else default_directions(n)

    if rem > 30.0:
        if n <= 350:
            p1 = min(rem * 0.45, 170.0)
        elif n <= 700:
            p1 = min(rem * 0.45, 100.0)
        else:
            p1 = min(rem * 0.45, 55.0)
        p1_end = time.time() + p1
        try:
            ht, ct = pair_local_optimize(best_h, n, p1_end, rng, M_total=M_use)
            if np.isfinite(ct) and ct < best_c:
                best_c = ct
                best_h = ht
        except Exception:
            pass

    rem = stage_end - time.time()
    if do_slsqp and n <= 600 and rem > 70.0:
        if n <= 350:
            sl_time = min(rem * 0.35, 120.0)
        else:
            sl_time = min(rem * 0.35, 70.0)
        sl_end = time.time() + sl_time
        try:
            ht, ct = slsqp_stage(best_h, n, sl_end, rng)
            if np.isfinite(ct) and ct < best_c:
                best_c = ct
                best_h = ht
        except Exception:
            pass

    rem = stage_end - time.time()
    if rem > 25.0:
        p2_end = stage_end - 3.0
        try:
            ht, ct = pair_local_optimize(best_h, n, p2_end, rng, M_total=int(0.7 * M_use))
            if np.isfinite(ct) and ct < best_c:
                best_c = ct
                best_h = ht
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c = eval_solution(best_h)
    return best_h, best_c


def next_resolution(n, cap=1024):
    if n >= cap:
        return cap
    nxt = 2 * n
    if nxt > cap:
        nxt = cap
    if nxt % 2 == 1:
        nxt += 1
    if nxt > cap:
        nxt = cap
    return int(nxt)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    deadline = t0 + max(10.0, budget) - 5.0
    rng = np.random.RandomState(int(seed))

    h_parent = None
    try:
        hp = np.asarray(initial_h_values, dtype=float)
        if hp.size > 0:
            h_parent = hp.copy()
    except Exception:
        pass

    if h_parent is None or len(h_parent) == 0:
        n0 = 250
        h_parent = project_feasible(np.full(n0, 0.5, dtype=float), n0)
    else:
        n0 = int(len(h_parent))
        h_parent = project_feasible(h_parent, n0)

    best_h = h_parent.copy()
    best_c = eval_solution(best_h)
    best_n = n0

    # Try a few explicit designs at the parent resolution; if one is clearly better, use it.
    if time.time() < deadline - 40.0:
        try:
            alts = alternative_starts(n0, rng, limit=3)
            for c, h in alts:
                if np.isfinite(c) and c < best_c:
                    best_c = c
                    best_h = h
                    best_n = n0
        except Exception:
            pass

    # Stage 1: optimize at the parent/current best resolution.
    if time.time() < deadline - 60.0:
        total = deadline - time.time()
        alloc1 = min(430.0, 0.52 * total)
        stage_end = time.time() + alloc1
        if stage_end > deadline - 35.0:
            stage_end = deadline - 35.0
        if stage_end > time.time() + 25.0:
            try:
                ht, ct = optimize_stage(best_h, best_n, stage_end, rng, M_total=None, do_slsqp=(best_n <= 600))
                if np.isfinite(ct) and ct < best_c:
                    best_c = ct
                    best_h = ht
                    best_n = best_n
            except Exception:
                pass

    # Stage 2: upscale the best known construction.
    if time.time() < deadline - 130.0 and best_n < 1024:
        n2 = next_resolution(best_n, cap=1024)
        if n2 != best_n:
            try:
                h2 = interpolate_h(best_h, n2)
                c2 = eval_solution(h2)
                if np.isfinite(c2) and c2 < best_c:
                    best_c = c2
                    best_h = h2
                    best_n = n2

                total = deadline - time.time()
                alloc2 = min(300.0, 0.48 * total)
                stage_end = time.time() + alloc2
                if stage_end > deadline - 30.0:
                    stage_end = deadline - 30.0
                if stage_end > time.time() + 35.0:
                    ht, ct = optimize_stage(h2, n2, stage_end, rng, M_total=None, do_slsqp=(n2 <= 600))
                    if np.isfinite(ct) and ct < best_c:
                        best_c = ct
                        best_h = ht
                        best_n = n2
            except Exception:
                pass

    # Stage 3: one more upscale if possible.
    if time.time() < deadline - 110.0 and best_n < 1024:
        n3 = next_resolution(best_n, cap=1024)
        if n3 != best_n:
            try:
                h3 = interpolate_h(best_h, n3)
                c3 = eval_solution(h3)
                if np.isfinite(c3) and c3 < best_c:
                    best_c = c3
                    best_h = h3
                    best_n = n3

                total = deadline - time.time()
                alloc3 = min(190.0, 0.55 * total)
                stage_end = time.time() + alloc3
                if stage_end > deadline - 25.0:
                    stage_end = deadline - 25.0
                if stage_end > time.time() + 30.0:
                    ht, ct = optimize_stage(h3, n3, stage_end, rng, M_total=None, do_slsqp=False)
                    if np.isfinite(ct) and ct < best_c:
                        best_c = ct
                        best_h = ht
                        best_n = n3
            except Exception:
                pass

    # Final polishing at the best resolution.
    if time.time() < deadline - 10.0:
        try:
            final_end = deadline - 2.0
            ht, ct = pair_local_optimize(best_h, best_n, final_end, rng,
                                         max_passes=120,
                                         M_total=default_directions(best_n))
            if np.isfinite(ct) and ct < best_c:
                best_c = ct
                best_h = ht
                best_n = best_n
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    best_c = eval_solution(best_h)

    return best_h, float(best_c), int(best_n)
```
