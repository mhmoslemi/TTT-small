def grad_smooth_fft(weights, h, n):
    n1 = n-1
    L = 1 << (3*n + 5).bit_length()  # >= 3n+5
    W = 2*n -1
    w_asc = weights[::-1].copy()
    aA = np.zeros(L); aA[:W] = w_asc
    bA = np.zeros(L); bA[:n] = 1.0 - h
    corrA = np.fft.irfft(np.fft.rfft(aA) * np.conj(np.fft.rfft(bA)), n=L)
    idxA = (np.arange(n) - n1) % L
    A = corrA[idxA].real
    w_rev = w_asc[::-1]
    aB = np.zeros(L); aB[:W] = w_rev
    bB = np.zeros(L); bB[:n] = h
    corrB = np.fft.irfft(np.fft.rfft(aB) * np.conj(np.fft.rfft(bB)), n=L)
    B = corrB[idxA].real
    return A - B

def adjust_sum_exact(h,n):
    target=0.5*n
    h=np.array(h,dtype=float).copy()
    h=np.clip(h,0,1)
    diff=target - h.sum()
    if abs(diff) > 1e-14:
        free=(h > 1e-10) & (h < 1.0 - 1e-10)
        if free.any():
            h[free] += diff / float(np.count_nonzero(free))
        else:
            if diff > 0:
               idxs=np.where(h < 1.0 - 1e-10)[0]
            else:
               idxs=np.where(h > 1e-10)[0]
            if len(idxs) > 0:
               h[idxs] += diff / len(idxs)
            else:
               idx = int(np.argmin(np.abs(h-0.5)))

        h=np.clip(h,0,1)
    # second small correction on same free
    diff=target-h.sum()
    if abs(diff)>1e-12:
        if diff>0: idxs=np.where(h < 1.0 - 1e-12)[0]
        else: idxs=np.where(h > 1e-12)[0]
        if len(idxs)>0:
            h[idxs] += diff/len(idxs)
            h=np.clip(h,0,1)
    return h

    c = raw * (2.0 / n)

    return exact_c_values(h, n)[1]


def fft_c_raw(h):
    n = len(h)
    L = next_pow2(2 * n - 1)
    a = np.zeros(L)
    q = np.zeros(L)
    a[:n] = h
    q[:n] = (1.0 - h)[::-1]
    raw = np.fft.irfft(np.fft.rfft(a) * np.fft.rfft(q), L)
    return raw[:2 * n - 1]


def lse_value_weights(c, alpha):
    m = float(np.max(c))
    if alpha <= 0.0:
        w = np.full(len(c), 1.0 / max(1, len(c)))
        return m, w
    t = np.clip(alpha * (c - m), -700.0, 0.0)
    e = np.exp(t)
    s = float(e.sum())
    if s <= 0.0:
        w = np.full(len(c), 1.0 / max(1, len(c)))
        return m, w
    return m + np.log(s) / alpha, e / s


def grad_smooth_fft(w, h, n):
    n1 = n - 1
    W = 2 * n - 1
    L = next_pow2(3 * n - 2)

    idx = (np.arange(n) - n1) % L
    w_asc = np.asarray(w, dtype=float)[::-1].copy()

    a = np.zeros(L)
    b = np.zeros(L)

    a[:W] = w_asc
    b[:n] = 1.0 - h
    corr_a = np.fft.irfft(np.fft.rfft(a) * np.conj(np.fft.rfft(b)), L)
    A = corr_a[idx].real

    a[:] = 0.0
    b[:] = 0.0
    a[:W] = w_asc[::-1]
    b[:n] = h
    corr_b = np.fft.irfft(np.fft.rfft(a) * np.conj(np.fft.rfft(b)), L)
    B = corr_b[idx].real

    return A - B


def _lse_obj(x, n, alpha):
    raw = fft_c_raw(x)
    c = raw * (2.0 / n)
    v, _ = lse_value_weights(c, alpha)
    return float(v)


def _lse_jac(x, n, alpha):
    raw = fft_c_raw(x)
    c = raw * (2.0 / n)
    _, w = lse_value_weights(c, alpha)
    g = grad_smooth_fft(w, x, n) * (2.0 / n)
    return g


def _eq_fun(x, n):
    return float(np.sum(x) - 0.5 * n)

    return np.ones(n, dtype=float)


def slsqp_lse(h, n, alpha, maxiter):
    h = project_feasible(h, n)
    mi = int(min(max(0, maxiter), 60))
    if mi <= 0:
        return h, exact_c_value(h)

    bounds = [(0.0, 1.0)] * n
    constraints = [
        {"type": "eq", "fun": _eq_fun, "jac": _eq_jac, "args": (n,)}
    ]
    try:
        res = minimize(
            _lse_obj,
            h,
            args=(n, alpha),
            jac=_lse_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": mi, "ftol": 1e-10},
        )
        if res is None or not hasattr(res, "x"):
            return h, exact_c_value(h)
        h2 = project_feasible(res.x, n)
        c2 = exact_c_value(h2)
        return h2, c2
    except Exception:
        return h, exact_c_value(h)

def active_raw(h, p, n):

    nv = n - k
    if d >= 0:
        a = h[:nv]
        b = h[k:]
    else:
        a = h[k:]
        b = h[:nv]
    return float(np.sum(a) - np.dot(a, b))


def active_grad(h, p, n):

    d = n1 - int(p)
    k = abs(d)
    nv = n - k
    g = np.zeros(n, dtype=float)

        g[k:] += -h[:nv]
    else:

    return g

    return float(x[n])


def _t_obj_jac(x, n):
    g = np.zeros(n + 1, dtype=float)
    g[n] = 1.0
    return g


def _t_eq_fun(x, n):
    return float(np.sum(x[:n]) - 0.5 * n)


def _t_eq_jac(x, n):
    g = np.empty(n + 1, dtype=float)
    g[:n] = 1.0
    g[n] = 0.0
    return g


def _t_vec_ineq_fun(x, n, active):
    active = np.asarray(active, dtype=int)
    if len(active) == 0:
        return np.empty(0, dtype=float)
    h = x[:n]
    t = float(x[n])
    dx = 2.0 / n
    vals = np.empty(len(active), dtype=float)
    for i, p in enumerate(active):
        vals[i] = active_raw(h, int(p), n)
    return t - dx * vals


def _t_vec_ineq_jac(x, n, active):
    active = np.asarray(active, dtype=int)
    K = len(active)
    J = np.empty((K, n + 1), dtype=float)
    J[:, n] = 1.0
    if K == 0:
        return J
    h = x[:n]
    dx = 2.0 / n
    for i, p in enumerate(active):
        g = active_grad(h, int(p), n)
        J[i, :n] = -dx * g
    return J


def _t_ineq_fun(x, n, active, idx):
    h = x[:n]
    p = int(active[idx])
    return float(x[n] - (2.0 / n) * active_raw(h, p, n))


def _t_ineq_jac(x, n, active, idx):
    h = x[:n]
    p = int(active[idx])
    g = np.empty(n + 1, dtype=float)
    g[n] = 1.0
    g[:n] = -(2.0 / n) * active_grad(h, p, n)
    return g


def select_active(c, K, n):
    L = 2 * n - 1
    K = int(max(1, K))
    if L <= 0:
        return np.array([], dtype=int)

    order = np.argsort(c)[::-1]
    half = max(1, min(K, L) // 2)
    base = order[:half].astype(int)
    mirrors = (L - 1 - base).astype(int)
    cand = np.concatenate((base, mirrors, np.array([n - 1], dtype=int)))
    cand = np.unique(cand)
    cand = cand[np.argsort(c[cand])[::-1]]
    return cand[:K].astype(int)


def slsqp_t_active(h, n, active, maxiter, use_vector=True):
    global _VECTOR_OK
    h = project_feasible(h, n)
    active = np.asarray(active, dtype=int)
    active = np.unique(active)
    mi = int(min(max(0, maxiter), 60))

    if mi <= 0 or len(active) == 0:
        return h, exact_c_value(h), None

    dx = 2.0 / n
    cmax = float(np.max(fft_c_raw(h)) * dx)
    maxr = 0.0
    for p in active:
        r = active_raw(h, int(p), n)
        if r > maxr:
            maxr = r
    cmax = max(cmax, maxr * dx)

    t0 = min(1.999999, cmax + 0.01)
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = t0

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    eq_con = {"type": "eq", "fun": _t_eq_fun, "jac": _t_eq_jac, "args": (n,)}
    res = None

    if use_vector and _VECTOR_OK:
        cons = [eq_con]
        cons.append(
            {
                "type": "ineq",
                "fun": _t_vec_ineq_fun,
                "jac": _t_vec_ineq_jac,
                "args": (n, active),
            }
        )
        try:
            res = minimize(
                _t_obj,
                x0,
                args=(n,),
                jac=_t_obj_jac,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": mi, "ftol": 1e-10},
            )
        except Exception:
            _VECTOR_OK = False
            res = None

    if res is None or not hasattr(res, "x"):
        cons = [eq_con]
        for idx in range(len(active)):
            cons.append(
                {
                    "type": "ineq",
                    "fun": _t_ineq_fun,
                    "jac": _t_ineq_jac,
                    "args": (n, active, idx),
                }
            )
        mi_s = min(mi, 40)
        try:
            res = minimize(
                _t_obj,
                x0,
                args=(n,),
                jac=_t_obj_jac,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": mi_s, "ftol": 1e-10},
            )
        except Exception:
            res = None

    if res is None or not hasattr(res, "x"):
        return h, exact_c_value(h), None

    h2 = project_feasible(res.x[:n], n)
    c2 = exact_c_value(h2)
    t2 = float(res.x[n])
    return h2, c2, t2


def optimize_active_t(h, n, deadline, rounds, K, maxiter, use_vector=True, rng=None):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = exact_c_value(h)
    no_imp = 0
    if rng is None:
        rng = np.random.RandomState(1234)

    for _ in range(rounds):
        if time.time() > deadline - 35:
            break

        raw = fft_c_raw(h)
        c = raw * (2.0 / n)
        active = select_active(c, K, n)

        if n <= 400:
            chunk = int(min(maxiter, 60))
        elif n <= 700:
            chunk = int(min(maxiter, 45))
        else:
            chunk = int(min(maxiter, 30))

        h2, c2, _ = slsqp_t_active(h, n, active, chunk, use_vector=use_vector)

        if c2 < best_c - 1e-13:
            best_h = project_feasible(h2, n)
            best_c = float(c2)
            h = best_h.copy()
            no_imp = 0
        else:
            no_imp += 1
            if c2 < best_c + 0.0005:
                h = project_feasible(h2, n)
            else:
                h = best_h.copy()
                if no_imp == 2:
                    noise = rng.normal(size=n)
                    if n > 5:
                        kernel = np.ones(5) / 5.0
                        noise = np.convolve(noise, kernel, mode="same")
                    nn = float(np.linalg.norm(noise))
                    if nn > 0:
                        noise /= nn
                    h = project_feasible(best_h + 0.01 * noise, n)
            if no_imp >= 4:
                break

    return best_h, best_c


def projected_gradient(h, n, deadline, alpha=5000.0, lr=0.015, max_iter=500):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = exact_c_value(h)
    step = float(lr)
    alpha_cur = float(alpha)

    for _ in range(max_iter):
        if time.time() > deadline - 3:
            break

        raw = fft_c_raw(h)
        c = raw * (2.0 / n)
        _, w = lse_value_weights(c, alpha_cur)
        g = grad_smooth_fft(w, h, n) * (2.0 / n)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break

        d = -g
        improved = False
        s = step
        for _trial in range(10):
            h_try = project_feasible(h + s * d, n)
            raw_try = fft_c_raw(h_try)
            c_try_fft = float(np.max(raw_try) * (2.0 / n))
            if c_try_fft <= best_c - 1e-14:
                c_try = exact_c_value(h_try)
                if c_try <= best_c - 1e-14:
                    h = h_try
                    best_h = h.copy()
                    best_c = float(c_try)
                    step = min(0.05, step * 1.1)
                    improved = True
                    break
            s *= 0.5

        if not improved:
            alpha_cur *= 1.4
            step *= 0.6
            if step < 1e-7:
                break

    return best_h, best_c


def interp_h(h_src, n_src, n_dst):
    if n_dst == n_src:
        return project_feasible(h_src.copy(), n_dst)

    h_src = np.asarray(h_src, dtype=float)

    if n_dst > n_src:
        dxs = 2.0 / n_src
        dx = 2.0 / n_dst
        xs = (np.arange(n_src) + 0.5) * dxs
        xd = (np.arange(n_dst) + 0.5) * dx
        h_new = np.interp(xd, xs, h_src, left=h_src[0], right=h_src[-1])
    else:
        if n_src % n_dst == 0:
            ratio = n_src // n_dst
            h_new = h_src.reshape(n_dst, ratio).mean(axis=1)
        else:
            dxs = 2.0 / n_src
            dx = 2.0 / n_dst
            xs = (np.arange(n_src) + 0.5) * dxs
            xd = (np.arange(n_dst) + 0.5) * dx
            h_new = np.interp(xd, xs, h_src, left=h_src[0], right=h_src[-1])

    return project_feasible(h_new, n_dst)


def random_smooth_start(n, rng, amp=0.5):
    if rng is None:
        rng = np.random.RandomState(7)
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    b = np.zeros(n, dtype=float)
    for _ in range(8):
        f = float(int(rng.randint(1, 7)))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a = float(amp * rng.uniform(0.1, 0.4))
        b += a * np.cos(np.pi * f * x + ph)
    mx = float(np.max(np.abs(b)))
    if mx > 1e-12:
        b = b / mx * amp
    h = 0.5 + 0.5 * b
    return project_feasible(h, n)


def make_candidate_ns(n_parent, budget_s):
    nq = int(n_parent)
    out = []
    if nq > 900:
        candidates = [600, 900]
    elif 10 <= nq <= 300:
        candidates = [
            nq,
            min(900, max(nq + 1, 2 * nq)),
            min(900, max(nq + 2, 3 * nq)),
        ]
    elif 301 <= nq <= 600:
        n2 = min(900, max(nq + 1, 2 * nq))
        n3 = min(900, max(n2 + 1, 3 * nq))
        candidates = [nq, n2, n3]
    elif 601 <= nq <= 900:
        candidates = [nq, 600, 900]
    else:
        candidates = [300]

    for n in candidates:
        n = int(n)
        if 10 <= n <= 900 and n not in out:
            out.append(n)

    if budget_s < 600 and len(out) > 1:
        out = out[:1]
    if budget_s < 900 and len(out) > 2:
        out = out[:2]

    if len(out) == 0:
        out = [300]
    return out


def lse_profile(n):
    if n <= 400:
        return ([50.0, 200.0, 1000.0, 5000.0], [80, 100, 120, 150])
    if n <= 700:
        return ([100.0, 500.0, 2000.0], [80, 100, 120])
    return ([500.0, 2000.0], [80, 100])


def active_profile(n):
    if n <= 400:
        return (6, 22, 70)
    if n <= 700:
        return (5, 18, 60)
    return (4, 14, 50)


def quick_refine(h, n, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = exact_c_value(h)

    if time.time() < deadline - 25:
        h2, c2 = slsqp_lse(h, n, 500.0, 25)
        if c2 < best_c:
            best_h = project_feasible(h2, n)
            best_c = float(c2)
        elif c2 < best_c + 0.001:
            h = project_feasible(h2, n)

    if time.time() < deadline - 12:
        raw = fft_c_raw(h)
        c = raw * (2.0 / n)
        active = select_active(c, 12, n)
        h2, c2, _ = slsqp_t_active(h, n, active, 25, use_vector=True)
        if c2 < best_c:
            best_h = project_feasible(h2, n)
            best_c = float(c2)

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(max(60.0, float(budget_s)))
    hard_deadline = t0 + budget_s - 30.0
    rng = np.random.RandomState(int(seed) % (2**31 - 1))

    try:
        hv = initial_h_values
        if hv is None:
            raise ValueError("no parent")
        h_init = np.asarray(hv, dtype=float).copy()
        if h_init.size == 0 or not np.all(np.isfinite(h_init)):
            raise ValueError("bad parent")
    except Exception:
        h_init = np.full(300, 0.5, dtype=float)

    n_init = int(len(h_init))
    h_init = project_feasible(h_init, n_init)

    best_h = h_init.copy()
    best_c = exact_c_value(best_h)
    best_n = n_init

    ns = make_candidate_ns(n_init, budget_s)

    usable = max(60.0, budget_s - 60.0)
    cnt = len(ns)
    if cnt >= 3:
        fracs = [0.40, 0.30, 0.18]
    elif cnt == 2:
        fracs = [0.60, 0.28]
    elif cnt == 1:
        fracs = [0.85]
    else:
        fracs = []

    cum = 0.0
    phase_deadlines = []
    for f in fracs:
        cum += f * usable
        phase_deadlines.append(min(hard_deadline, t0 + cum))

    for idx, n in enumerate(ns):
        if idx >= len(phase_deadlines):
            continue
        phase_dead = phase_deadlines[idx]

        if time.time() > phase_dead - 30:
            break

        if idx == 0 and abs(n - n_init) <= 1 and n_init <= 900:
            h_start = project_feasible(h_init, n_init)
        else:
            h_start = interp_h(best_h, best_n, n)

        alphas, totals = lse_profile(n)
        h1, c1 = optimize_lse(h_start, n, phase_dead, alphas, totals)
        if c1 < best_c:
            best_h = project_feasible(h1, n)
            best_c = float(c1)
            best_n = int(n)

        rounds, Akt, amx = active_profile(n)
        h2, c2 = optimize_active_t(
            h1, n, phase_dead,
            rounds=rounds,
            K=Akt,
            maxiter=amx,
            use_vector=True,
            rng=rng
        )
        if c2 < best_c:
            best_h = project_feasible(h2, n)
            best_c = float(c2)
            best_n = int(n)

        if time.time() < phase_dead - 25:
            h3, c3 = projected_gradient(h2, n, phase_dead, alpha=5000.0, lr=0.015, max_iter=250)
            if c3 < best_c:
                best_h = project_feasible(h3, n)
                best_c = float(c3)
                best_n = int(n)

    if time.time() < hard_deadline - 160:
        probe_n = int(best_n)
        h_comp = project_feasible(1.0 - best_h, probe_n)
        hq, cq = quick_refine(h_comp, probe_n, min(hard_deadline, time.time() + 70.0))
        if cq < best_c:
            best_h = project_feasible(hq, probe_n)
            best_c = float(cq)
            best_n = int(probe_n)

    if time.time() < hard_deadline - 130:
        probe_n = int(best_n)
        h_rand = random_smooth_start(probe_n, rng, amp=0.5)
        hq, cq = quick_refine(h_rand, probe_n, min(hard_deadline, time.time() + 70.0))
        if cq < best_c:
            best_h = project_feasible(hq, probe_n)
            best_c = float(cq)
            best_n = int(probe_n)

    if time.time() < hard_deadline - 40:
        rounds, Akt, amx = active_profile(best_n)
        h_f, c_f = optimize_active_t(
            best_h, best_n, hard_deadline,
            rounds=3,
            K=max(10, min(20, Akt)),
            maxiter=max(35, min(55, amx)),
            use_vector=True,
            rng=rng
        )
        if c_f < best_c:
            best_h = project_feasible(h_f, best_n)
            best_c = float(c_f)

    if time.time() < hard_deadline - 6:
        h_f, c_f = projected_gradient(
            best_h, best_n, hard_deadline,
            alpha=10000.0 if best_n > 500 else 5000.0,
            lr=0.01,
            max_iter=300
        )
        if c_f < best_c:
            best_h = project_feasible(h_f, best_n)
            best_c = float(c_f)

    best_h = project_feasible(best_h, best_n)
    best_h = adjust_sum_exact(best_h, best_n)
    best_c = exact_c_value(best_h, best_n)

    return best_h, float(best_c), int(len(best_h))
