    n_full = 2 * n - 1

        return grad

        return grad

        h_new, c_new = local_search_exchange(h_curr, n, deadline - 5)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    
    if time.time() < deadline - 5:
        h_new, c_new = binary_nudge(best_h, n, deadline - 3)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    
    if time.time() < deadline - 3:
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    best_h = None
    best_c = float('inf')
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        c_init = compute_c5(h_init, n_init)
        best_h, best_c = h_init.copy(), c_init
        
        alloc = min(250, (deadline - t0) * 0.35)
        h_opt, c_opt = full_optimize(h_init, n_init, t0 + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    
    # Try alternative resolutions
    alt_ns = [200, 400, 250, 350, 500]
    for n_try in alt_ns:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 50
        if remaining < 70:
            break
        alloc = min(130, remaining * 0.3)
        if best_h is not None and n_try == len(best_h):
            continue
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
        else:
            x_best = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_try)
        if best_h is not None:
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.interp(x_new, x_best, h_init)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = full_optimize(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    
    # Refine at best resolution
    if best_h is not None and time.time() < deadline - 40:
        n_best = len(best_h)
        alloc = min(100, (deadline - time.time()) * 0.4)
        h_opt, c_opt = full_optimize(best_h, n_best, time.time() + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    
    # Local search + binary nudge
    if best_h is not None and time.time() < deadline - 20:
        n_best = len(best_h)
        h_new, c_new = local_search_exchange(best_h, n_best, deadline - 10)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        h_new, c_new = binary_nudge(best_h, n_best, deadline - 5)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    
    # Final SLSQP polish
    if best_h is not None and time.time() < deadline - 8:
        n_best = len(best_h)
        h_new = slsqp_optimize(best_h, n_best, 2000000, maxiter=3000)
        c_new = compute_c5(h_new, n_best)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    
    # Final local search
    if best_h is not None and time.time() < deadline - 2:
        n_best = len(best_h)
        h_new, c_new = local_search_exchange(best_h, n_best, deadline - 1)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
