```python
    grad = np.zeros(n)
```

```python
def smooth_subgradient_step(h, n, alpha, eta):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    
    # Soft-max weights
    c_shift = c - c.max()
    p = np.exp(alpha * c_shift)
    p /= p.sum()
    
    # Weighted gradient
    grad = np.zeros(n)
    for k_idx, pk in enumerate(p):
        if pk < 1e-10:
            continue
        k = k_idx - (n-1)
        if k >= 0:
            grad[:n-k] += pk * (1 - h[k:])
            grad[k:] -= pk * h[:n-k]
        else:
            k = -k
            grad[k:] += pk * (1 - h[:n-k])
            grad[:n-k] -= pk * h[k:]
    
    return h - eta * grad
```

```python
            if k < n:

                grad[:n-k] -= wk * h[k:]

                    bk2 = b - k

                        best_h = h_b.copy()
                    break

                break
        if not improved:
            break
```

```python
def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Projected subgradient (fast continuous optimization)
    p1 = t0 + 0.15 * (budget_s - 5)
    h = best_h.copy()
    for alpha_init in [100, 500, 2000]:
        phase_dl = min(p1, t0 + (budget_s - 5) * 0.05)
        if time.time() > phase_dl:
            break
        hb, cb = projected_subgradient(h, n0, phase_dl, eta0=0.03,
                                       momentum=0.9, alpha=alpha_init, max_active=50)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            best_n = n0
        h = hb.copy()

    # Phase 2: SLSQP polish on best continuous solution
    p2 = t0 + 0.25 * (budget_s - 5)
    if time.time() < p2:
        for alpha_s in [2000, 10000, 100000]:
            if time.time() > p2 - 5:
                break
            hp = slsqp_polish(best_h, best_n, p2 - 3, alpha_s, 1000)
            c = compute_c5(hp, best_n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
                best_n = best_n

    # Phase 3: Binary local search on the best solution
    p3 = t0 + 0.55 * (budget_s - 5)
    hb, cb = binary_local_search(best_h, best_n, p3, max_sweeps=500,
                                 n_restarts=4, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        best_n = best_n

    # Phase 4: Multi-resolution exploration
    p4 = t0 + 0.72 * (budget_s - 5)
    for nh in [100, 150, 200, 300]:
        if time.time() > p4 - 20:
            break
        if nh == best_n:
            continue
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        # Quick subgradient
        phase_dl = min(p4 - 10, time.time() + 30)
        hh2, ch2 = projected_subgradient(hh, nh, phase_dl, eta0=0.02,
                                         momentum=0.85, alpha=500, max_active=40)
        ch_final = compute_c5(hh2, nh)
        if ch_final < best_c:
            best_c = ch_final
            best_h = hh2.copy()
            best_n = nh
        # Quick binary on this resolution
        phase_dl2 = min(p4 - 5, time.time() + 20)
        hb2, cb2 = fast_binary_swap(hh2, nh, phase_dl2, 500, seed + nh)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            best_n = nh

    # Phase 5: Extended binary search with more iterations
    p5 = t0 + 0.88 * (budget_s - 5)
    hb, cb = binary_local_search(best_h, best_n, p5, max_sweeps=1000,
                                 n_restarts=3, seed=seed + 100)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        best_n = best_n

    # Phase 6: Final SLSQP + binary polish
    p6 = dl - 5
    if time.time() < p6 - 10:
        h_final = best_h.copy()
        c_final = best_c
        # SLSQP on continuous
        h_s = slsqp_polish(h_final, best_n, dl - 15, 50000, 1500)
        c_s = compute_c5(h_s, best_n)
        if c_s < c_final:
            h_final = h_s.copy()
            c_final = c_s
        # Binary polish
        if time.time() < dl - 8:
            hb_f, cb_f = fast_binary_swap(h_final, best_n, dl - 5, 300, seed + 200)
            if cb_f < c_final:
                h_final = hb_f.copy()
                c_final = cb_f
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
