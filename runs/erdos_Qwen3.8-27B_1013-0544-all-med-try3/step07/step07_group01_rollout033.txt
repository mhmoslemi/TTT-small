def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    shifted = alpha * (c - c_max)
    m = shifted.max()
    log_sum = m + np.log(np.sum(np.exp(shifted - m)))
    return c_max + log_sum / alpha

    c_max = c.max()
    shifted = alpha * (c - c_max)
    m = shifted.max()
    w = np.exp(shifted - m)
    w /= w.sum()
    dx = 2.0 / n

Actually, let me be more careful with FISTA. The standard FISTA algorithm is:

```
y_0 = x_0, t_0 = 1
for k = 0, 1, 2, ...:
    x_{k+1} = prox_{1/L * f}(y_k)  # in our case, project(y_k - (1/L)*grad f(y_k))
    t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2
    y_{k+1} = x_{k+1} + (t_k - 1) / t_{k+1} * (x_{k+1} - x_k)
```

Where prox is the proximal operator. For our problem with the projection constraint, the proximal operator of the indicator function of the feasible set is just the projection.

Let me also think about the step size. The gradient of the smooth max function has Lipschitz constant L. For the smooth max:
f(h) = c_max + (1/alpha) * log(Σ exp(alpha * (c_k - c_max)))

The Hessian involves second derivatives of c_k, which are related to the correlation matrix. Roughly, L ≈ alpha * (max |Hessian of c_k|). Since c_k is quadratic, its Hessian is constant (it's the correlation matrix C_k). So L ≈ alpha * max_k ||C_k||.

The correlation matrices C_k have norm 1 (they're shift matrices), so L ≈ alpha. Thus the step size should be 1/L ≈ 1/alpha.

But this is a very rough bound. In practice, we can use backtracking line search.

Under the constraint that we can only do "swaps" (move mass from one coordinate to another while preserving the sum), the best swap in the steepest descent direction is:
- Find i with the largest g[i] (where we most want to decrease h[i])
- Find j with the smallest g[j] (where we most want to increase h[j])
- Move mass from i to j

But a single swap might not be optimal. A better approach is to do a "transport" from high-gradient to low-gradient indices, proportional to the gradient difference. This is essentially a Wasserstein distance minimization step.

1. Sort indices by gradient: g_π(1) ≤ g_π(2) ≤ ... ≤ g_π(n)
2. The "best" direction is to decrease h at the top indices and increase h at the bottom indices
3. Do a batch of swaps in this direction

This is equivalent to a restricted Frank-Wolfe step where we only swap within a neighborhood.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_weights(c_vec, alpha):
    c_max = c_vec.max()
    shifted = alpha * (c_vec - c_max)
    m = shifted.max()
    w = np.exp(shifted - m)
    w /= w.sum()
    return w, c_max

def grad_smooth_max(h, n, alpha, weights=None):
    if weights is None:
        c = compute_c5_vec(h, n)
        weights, _ = compute_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = weights[m_idx]
        if wm < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm_dx = wm * dx
        if k >= 0:
            nv = n - k
            grad[k:] += wm_dx * (1.0 - h[:nv])
            grad[:nv] += wm_dx * (-h[k:])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm_dx * (1.0 - h[kk:])
            grad[kk:] += wm_dx * (-h[:nv])
    return grad

def fista_anneal(h, n, deadline, alpha_start=50, alpha_end=100000, iters_per_alpha=200):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ratio = alpha_end / alpha_start
    n_alphas = int(np.ceil(np.log(ratio) / np.log(1.15)))
    n_alphas = max(n_alphas, 5)
    for ai in range(n_alphas):
        if time.time() > deadline:
            break
        alpha = alpha_start * (ratio ** (ai / max(n_alphas - 1, 1)))
        alpha = min(alpha, alpha_end)
        y = best_h.copy()
        x_prev = best_h.copy()
        t_mom = 1.0
        step = 1.0 / (alpha * 2.0)
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                return best_h, best_c
            c_vec = compute_c5_vec(y, n)
            w, c_max = compute_weights(c_vec, alpha)
            g = grad_smooth_max(y, n, alpha, w)
            x_new = project_feasible(y - step * g, n)
            c_new = compute_c5(x_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = x_new.copy()
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = project_feasible(x_new + ((t_mom - 1.0) / t_new) * (x_new - x_prev), n)
            x_prev = x_new.copy()
            t_mom = t_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c

def lbfgs_smooth(h, n, deadline, alpha=50000, maxiter=300):
    def obj(x, a):
        h_local = project_feasible(x, n)
        c = compute_c5_vec(h_local, n)
        cm = c.max()
        sh = a * (c - cm)
        mx = sh.max()
        val = cm + (mx + np.log(np.sum(np.exp(sh - mx)))) / a
        return float(val)
    def grad(x, a):
        h_local = project_feasible(x, n)
        c = compute_c5_vec(h_local, n)
        w, _ = compute_weights(c, a)
        return grad_smooth_max(h_local, n, a, w)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, args=(alpha,), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_pair_search(h, n, deadline, step_size=0.003, max_alphas=5):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    for alpha in [1000, 5000, 20000, 100000, 500000][:max_alphas]:
        for _ in range(100):
            if time.time() > deadline:
                return h, best_c
            c_vec = compute_c5_vec(h, n)
            w, _ = compute_weights(c_vec, alpha)
            g = grad_smooth_max(h, n, alpha, w)
            improved = False
            order = np.argsort(g)
            for _ in range(30):
                if time.time() > deadline:
                    return h, best_c
                i = order[rng.randint(n // 2, n)]
                j = order[rng.randint(0, n // 2)]
                if i == j:
                    continue
                for eps in [step_size, step_size * 0.5, step_size * 2]:
                    if h[i] - eps >= 0 and h[j] + eps <= 1:
                        h[i] -= eps
                        h[j] += eps
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            break
                        else:
                            h[i] += eps
                            h[j] -= eps
            if not improved:
                break
    return h, best_c

def pair_search_local(h, n, deadline, radius=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def try_other_resolution(h_best, n_best, n_try, deadline, alloc):
    x_old = np.linspace(0, 2, n_best, endpoint=False) + 1.0 / n_best
    x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_feasible(h_interp, n_try)
    t_end = time.time() + alloc
    h_t, c_t = fista_anneal(h_interp, n_try, min(t_end - 10, deadline),
                            alpha_start=100, alpha_end=50000, iters_per_alpha=100)

    if time.time() < deadline - 5:

    return h_t, c_t
