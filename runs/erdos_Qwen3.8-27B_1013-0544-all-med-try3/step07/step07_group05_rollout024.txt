Let me also think about the **exact projection** onto {h ∈ [0,1]^n : ∑h = n/2}:

The Euclidean projection of a point y onto this polytope is:
proj(y)[i] = clip(y[i] - τ, 0, 1) for some τ chosen so that ∑ proj(y)[i] = n/2.

To find τ, we bisect on τ ∈ [min(y) - 1, max(y)]:
- f(τ) = ∑ clip(y[i] - τ, 0, 1) is a decreasing function of τ
- Find τ such that f(τ) = n/2

This is a standard "water-filling" problem and can be solved in O(n log n) by sorting, or O(n) with bisection.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def exact_project(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        tau = 0.5 * (lo + hi)
        h = np.clip(y - tau, 0.0, 1.0)
        if h.sum() > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    s = alpha * (c - c.max())
    p = np.exp(s)
    p /= p.sum()
    n_full = 2 * n - 1
    grad = np.full(n, dx)
    for k in range(n_full):
        pk = p[k]
        if pk < 1e-16:

        dk = k - (n - 1)
        base = pk * dx
        if dk >= 0:
            grad[:n - dk] -= base * h[dk:]
        else:
            grad[-dk:] -= base * h[-dk:]
        if dk > 0:
            grad[dk:] -= base * h[:n - dk]
        elif dk < 0:
            grad[:n + dk] -= base * h[-dk:]
    return grad
