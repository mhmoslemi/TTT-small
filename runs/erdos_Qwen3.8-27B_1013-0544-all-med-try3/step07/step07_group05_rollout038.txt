```
def active_gradient(c, h, n, beta=0.0, ctol_raw=1e-8, max_active=2000, exp_cutoff=60.0):
    L=c.size; off=n-1; m=c.max()
    weights = np.zeros(L, dtype=float)
    if beta > 0:
        z = beta*(c-m)
        # For large negative set -exp_cutoff? We'll just mask z > -exp_cutoff
        mask = z > -exp_cutoff
        if not mask.any():
            mask=np.zeros(L,bool); mask[int(np.argmax(c))]=True
        vals=np.exp(np.clip(z[mask], -exp_cutoff, 0.0))
        total=vals.sum()
        if total <=0: vals[:] =1; total=vals.size
        weights[mask]=vals/total
    else:
        threshold = m - ctol_raw
        mask = c >= threshold
        if not mask.any(): mask[argmax]=True
        weights[mask]=1.0/mask.sum()
    if max_active and int(weights.size) > max_active:
       idxs=np.nonzero(weights)[0]
       if len(idxs) > max_active:
          order=np.argsort(c[idxs])
          keep=order[-max_active:]
          wkeep=weights[idxs[keep]]
          weights[:] = 0.0
          weights[idxs[keep]] = wkeep / wkeep.sum()
    g=np.zeros(n)
    idxs=np.nonzero(weights)[0]
    for idx in idxs:
        w=weights[idx]
        d=idx-off
        start = max(0, -d)
        stop = min(n, n - d)
        # update
        g[start+d:stop+d] += w * (1.0 - h[start:stop])
        g[start:stop]     -= w * h[start+d:stop+d]
    return g
```

```python
def _get_nfft(n):
    val = _NF_CACHE.get(n)
    if val is None:
        need = 2 * int(n) - 1
        val = 1
        while val < need:
            val <<= 1
        _NF_CACHE[int(n)] = val
    return val


def compute_full_corr(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    Nf = _get_nfft(n)
    Fh = np.fft.rfft(h, Nf)
    Fg = np.fft.rfft(1.0 - h, Nf)
    corr = np.fft.irfft(Fh * np.conjugate(Fg), n=Nf)

            start = 0
        if stop > n:
            stop = n
        if stop <= start:
            continue

        g[start + d:stop + d] += w * (1.0 - h[start:stop])
        g[start:stop] -= w * h[start + d:stop + d]

    return g


def smooth_jac(x, n, alpha):
    n = int(n)
    x = np.asarray(x, dtype=float)
    if x.size != n or not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    c = compute_full_corr(x, n)
    return active_gradient(c, x, n, beta=float(alpha), ctol_raw=0.0)


def slsqp_one(h, n, alpha, maxiter, bounds, constraints):
    n = int(n)
    try:
        res = minimize(
            smooth_obj,
            np.asarray(h, dtype=float),
            args=(n, float(alpha)),
            jac=smooth_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": 1e-14},
        )
        x = np.asarray(res.x, dtype=float)
        return project_feasible(x, n), bool(getattr(res, "success", False))
    except Exception:
        return None, False


def slsqp_refine(h, n, deadline, mode="low"):
    n = int(n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5_exact(h, n)
    current_h = h.copy()
    current_c = best_c

    bounds = [(0.0, 1.0)] * n
    constraints = [
        {
            "type": "eq",
            "fun": _sum_con,
            "jac": _sum_con_jac,
            "args": (n,),
        }
    ]

    if mode == "low":
        if n <= 320:
            alphas = [50.0, 200.0, 1000.0, 4000.0,
                      16000.0, 64000.0, 250000.0, 1000000.0]
            base_iter = 70
        elif n <= 650:
            alphas = [100.0, 500.0, 2500.0, 10000.0,
                      40000.0, 160000.0, 640000.0]
            base_iter = 40
        else:
            alphas = [500.0, 2500.0, 12000.0, 50000.0, 220000.0]
            base_iter = 22
    else:
        if n <= 320:
            alphas = [200.0, 1000.0, 5000.0, 20000.0,
                      80000.0, 320000.0, 1200000.0]
            base_iter = 55
        elif n <= 650:
            alphas = [500.0, 2500.0, 10000.0, 40000.0,
                      160000.0, 640000.0]
            base_iter = 32
        else:
            alphas = [2000.0, 10000.0, 50000.0, 220000.0]
            base_iter = 16

    for alpha in alphas:
        if time.time() > deadline - 12.0:
            break

        for _chunk in range(4):
            rem = deadline - time.time()
            if rem <= 12.0:
                break

            # Keep individual SLSQP calls modest so the hard wall-clock budget is respected.
            maxiter = base_iter
            if rem < maxiter:
                maxiter = max(12, int(rem // 2))

            x, ok = slsqp_one(current_h, n, alpha, maxiter, bounds, constraints)
            if x is None:
                break

            c = compute_c5_exact(x, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = x.copy()

            if c <= current_c + 1e-7:
                current_h = x
                current_c = c

            if ok:
                break
            if c > current_c + 1e-7:
                break

        if time.time() < deadline - 12.0:
            hs = try_symmetrize(current_h, n)
            cs = compute_c5_exact(hs, n)
            if cs < current_c - 1e-14:
                current_h = hs
                current_c = cs
            if cs < best_c - 1e-14:
                best_c = cs
                best_h = hs.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5_exact(best_h, n)
    return best_h, best_c


def resample(h, n_old, n_new):
    n_old = int(n_old)
    n_new = int(n_new)
    if n_old == n_new:
        return project_feasible(h, n_new)
    if n_old <= 0:
        return project_feasible(np.full(n_new, 0.5), n_new)

    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return project_feasible(h_new, n_new)


def try_symmetrize(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    hs = 0.5 * (h + h[::-1])
    return project_feasible(hs, n)


def projected_gradient_refine(h, n, deadline, rng=None):
    n = int(n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_raw = float(np.max(compute_full_corr(h, n)))

    betas = [1000.0, 10000.0, 100000.0, 1000000.0]
    steps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.0012, 0.0006]

    for beta in betas:
        if time.time() > deadline:
            break

        stall = 0
        for _it in range(5000):
            if time.time() > deadline:
                break

            c = compute_full_corr(h, n)
            C = float(np.max(c))
            if C < best_raw - 1e-13:
                best_raw = C
                best_h = h.copy()

            g = active_gradient(c, h, n, beta=float(beta), ctol_raw=0.0)
            gn = float(np.linalg.norm(g))
            if gn < 1e-13:
                break

            d = -g / gn
            accepted = False
            for step in steps:
                hp = project_feasible(h + step * d, n)
                Cn = float(np.max(compute_full_corr(hp, n)))
                if Cn + 1e-14 < C:
                    h = hp
                    accepted = True
                    break

            if accepted:
                stall = 0
            else:
                stall += 1
                if rng is not None and stall > 8 and (_it % 8 == 0):
                    h = project_feasible(h + rng.normal(0.0, 0.01, n), n)
                    stall = 0
                if stall > 50:
                    break

        if time.time() < deadline:
            hs = try_symmetrize(h, n)
            Cs = float(np.max(compute_full_corr(hs, n)))
            if Cs < best_raw - 1e-13:
                best_raw = Cs
                best_h = hs.copy()

    h = project_feasible(best_h, n)
    exact = compute_c5_exact(h, n)
    return h, exact


def gradient_pair_refine(h, n, deadline, rng=None):
    n = int(n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_raw = float(np.max(compute_full_corr(h, n)))

    top = 4
    extremes = 2
    deltas = [0.004, 0.002, 0.001, 0.0005,
              0.00025, 0.000125, 0.0000625, 0.00003125]

    for delta in deltas:
        if time.time() > deadline - 2.0:
            break

        h = project_feasible(h, n)
        passes = 0

        while time.time() < deadline - 2.0:
            c = compute_full_corr(h, n)
            C = float(np.max(c))
            if C < best_raw - 1e-13:
                best_raw = C
                best_h = h.copy()

            low_idx = []
            high_idx = []

            ctol = max(1e-9, 5.0e-7 * n)
            g0 = active_gradient(c, h, n, beta=0.0, ctol_raw=ctol)
            g1 = active_gradient(c, h, n, beta=1.0e4, ctol_raw=0.0)
            g2 = active_gradient(c, h, n, beta=1.0e6, ctol_raw=0.0)

            for g in (g0, g1, g2):
                if float(np.linalg.norm(g)) < 1e-15:
                    continue
                order = np.argsort(g)
                low_idx.extend(order[:top].tolist())
                high_idx.extend(order[-top:].tolist())

            order_h = np.argsort(h)
            low_idx.extend(order_h[:extremes].tolist())
            high_idx.extend(order_h[-extremes:].tolist())

            low_idx = np.unique([int(i) for i in low_idx if 0 <= int(i) < n])
            high_idx = np.unique([int(i) for i in high_idx if 0 <= int(i) < n])

            improved = False
            for i in low_idx:
                if h[i] >= 1.0:
                    continue
                for j in high_idx:
                    if i == j or h[j] <= 0.0:
                        continue
                    for d in (delta, -delta):
                        hi = h[i] + d
                        hj = h[j] - d
                        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                            continue

                        h[i] = hi
                        h[j] = hj
                        Cn = float(np.max(compute_full_corr(h, n)))
                        if Cn + 1e-14 < C:
                            C = Cn
                            improved = True
                            if C < best_raw - 1e-13:
                                best_raw = C
                                best_h = h.copy()
                        else:
                            h[i] = hi - d
                            h[j] = hj + d

            if not improved:
                break

            passes += 1
            if passes >= 60:
                break

            if passes % 15 == 0:
                hs = try_symmetrize(h, n)
                Cn = float(np.max(compute_full_corr(hs, n)))
                if Cn + 1e-14 < C:
                    h = hs
                    C = Cn
                    improved = True
                    if C < best_raw - 1e-13:
                        best_raw = C
                        best_h = h.copy()

    h = project_feasible(best_h, n)
    exact = compute_c5_exact(h, n)
    return h, exact


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 8.0)
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(parent))
        parent = project_feasible(parent, n0)
    except Exception:
        n0 = 250
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        parent = 0.5 + 0.25 * np.cos(np.pi * x)
        parent = project_feasible(parent, n0)

    best_h = parent.copy()
    best_c = compute_c5_exact(best_h, n0)

    # Refine at the parent resolution.
    if time.time() < deadline - 150.0:
        alloc = min((deadline - t0) * 0.45, 430.0)
        if alloc > 30.0:
            h, c = slsqp_refine(best_h, int(len(best_h)), t0 + alloc, mode="low")
            if c < best_c:
                best_h = h
                best_c = c

    # Try a higher resolution if the current one is coarse.
    cur_n = int(len(best_h))
    if cur_n < 512:
        n_next = 512
    elif cur_n < 768:
        n_next = 768
    else:
        n_next = cur_n

    if n_next != cur_n and time.time() < deadline - 140.0:
        remaining = deadline - time.time()
        alloc = min(360.0, remaining * 0.4)
        if alloc > 80.0:
            h_interp = resample(best_h, cur_n, n_next)
            h, c = slsqp_refine(h_interp, n_next, time.time() + alloc, mode="high")
            if c < best_c:
                best_h = h
                best_c = c

    # A short projected-gradient polish on the best construction.
    if time.time() < deadline - 90.0:
        alloc = min(90.0, (deadline - time.time()) * 0.3)
        if alloc > 25.0:
            h, c = projected_gradient_refine(best_h, int(len(best_h)), time.time() + alloc, rng)
            if c < best_c:
                best_h = h
                best_c = c

    # Final low-level pair refinement using exact FFT correlations.
    if time.time() < deadline - 10.0:
        h, c = gradient_pair_refine(best_h, int(len(best_h)), deadline - 4.0, rng)
        if c < best_c:
            best_h = h
            best_c = c

    n_final = int(len(best_h))
    best_h = project_feasible(best_h, n_final)
    best_h = force_exact_sum(best_h, n_final)
    best_c = compute_c5_exact(best_h, n_final)

    return best_h, float(best_c), n_final
```
