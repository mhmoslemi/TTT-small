import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n/2.0)**2

def penalty_grad(x, n, alpha, mu):
    return smooth_max_grad(x, n, alpha) + 2.0*mu*(x.sum() - n/2.0)

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 1.0), (300, 3.0), (800, 5.0), (2000, 10.0),
        (5000, 20.0), (15000, 50.0), (50000, 100.0),
        (100000, 300.0), (300000, 1000.0), (500000, 3000.0)
    ]
    for alpha, mu in configs:
        remaining = deadline - time.time()
        if remaining < 5:
            break
        maxit = min(max_iters, int(remaining / 0.5))
        if maxit < 10:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit*3,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, alpha_init=1000, n_active=5):
    """Optimize using only the top-k active constraints"""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = alpha_init
    lr = 0.005
    
    for it in range(5000):
        if time.time() > deadline:
            break
        
        c = compute_c5_vec(h, n)
        # Find active indices (top n_active)
        idx = np.argsort(c)[-n_active:]
        c_max = c.max()
        
        # Weighted gradient from active set only
        w = np.zeros(2*n-1)
        for i in idx:
            w[i] = np.exp(alpha * (c[i] - c_max))
        w /= w.sum()
        
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for m_idx in idx:
            if w[m_idx] < 1e-12:
                continue
            k = m_idx - (n-1)
            wm = w[m_idx] * dx_val
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
        
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        grad /= g_norm
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)

                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break

        if not improved:
            lr *= 0.7
            alpha *= 1.3
            if lr < 1e-13:
                break
        
        # Occasionally add more active constraints
        if it % 50 == 0 and n_active < 15:
            n_active += 1
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                        if c < best_c - 1e-14:

                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

    try:

                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def _pen_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _pen_grad(x, n, alpha, mu):
    return smooth_max_grad(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def _sl_obj(x, n, alpha):
    return smooth_max_val(x, n, alpha)

def _sl_cons(x, n):
    return x.sum() - n / 2.0

def _sl_jac(x, n):
    return np.ones(len(x))

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [
        (50, 0.5), (200, 2.0), (500, 5.0), (1500, 10.0),
        (4000, 25.0), (10000, 50.0), (30000, 100.0),
        (80000, 300.0), (200000, 800.0), (400000, 2000.0),
        (700000, 5000.0)
    ]
    for alpha, mu in configs:
        remaining = deadline - time.time()
        if remaining < 3:
            break
        maxit = min(max_iters, max(10, int(remaining / 0.4)))
        try:
            res = minimize(_pen_obj, h, jac=_pen_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': maxit * 4,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, alpha_init=500, n_active_init=5):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = alpha_init
    n_active = n_active_init
    lr = 0.004
    dx_val = 2.0 / n

    for it in range(8000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        top_idx = np.argsort(c)[-n_active:]
        w = np.exp(alpha * (c[top_idx] - c_max))
        w /= w.sum()

        grad = np.zeros(n)
        for ii, m_idx in enumerate(top_idx):
            if w[ii] < 1e-12:
                continue
            k = m_idx - (n - 1)
            wm = w[ii] * dx_val
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        grad /= g_norm

        step = lr
        improved = False
        for _ in range(35):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.6
            alpha *= 1.4
            if lr < 1e-14:
                break
        if it % 80 == 0 and n_active < 20:
            n_active += 1
    return best_h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sl_cons, 'jac': _sl_jac, 'args': (n,)}]
    try:
        res = minimize(_sl_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def fine_pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                            improved_any = True
                        else:

def block_search(h, n, deadline, block_sizes=(2, 3), radius=40):
    best_c = compute_c5(h, n)
    for bs in block_sizes:
        deltas = [0.002, 0.001, 0.0005, 0.0002]
        for d in deltas:
            improved_any = True
            while improved_any and time.time() < deadline:
                improved_any = False
                for i in range(n - bs):
                    if time.time() > deadline:
                        return h, best_c
                    lo = max(0, i - radius)
                    hi = min(n - bs, i + radius + 1)
                    for j in range(lo, hi):
                        if abs(i - j) < bs:
                            continue
                        # Move block [i:i+bs] up, block [j:j+bs] down
                        if np.all(h[i:i+bs] + d <= 1.0) and np.all(h[j:j+bs] - d >= 0.0):
                            h[i:i+bs] += d
                            h[j:j+bs] -= d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved_any = True
                            else:
                                h[i:i+bs] -= d

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
