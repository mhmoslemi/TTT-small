def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')

    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')
    c_scaled = c * dx
    m = c_scaled.max()
    sh = alpha * (c_scaled - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    num_shifts = 2 * n - 1
    A = np.zeros((num_shifts, n))
    one_minus_x = 1.0 - x

        else:
            A[k_idx, -s:] = one_minus_x[:n+s]

    return grad

    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = project_feasible(h_fine, n_fine)
    return h_fine

def fine_pair_search(h, n, deadline, center_radius=20):
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Quick L-BFGS warm-up
    if time.time() < deadline - 100:
        t1 = min(time.time() + 40, deadline - 100)
        h, c = quick_lbfgs(h, n, t1)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Alternating LP (main workhorse)
    remaining = deadline - time.time() - 80
    if remaining > 100:
        h, c = alternating_lp(h, n, time.time() + remaining, max_iters=500)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Try from best_h with another LP pass (different starting point effect)
    remaining = deadline - time.time() - 50
    if remaining > 60:
        h2, c2 = alternating_lp(best_h, n, time.time() + remaining, max_iters=200)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 4: L-BFGS polish
    if time.time() < deadline - 30:
        t4 = min(time.time() + 25, deadline - 10)
        h, c = quick_lbfgs(best_h, n, t4)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Fine pair search
    if time.time() < deadline - 10:
        t5 = min(time.time() + 20, deadline - 5)
        h, c = fine_pair_search(best_h, n, t5, center_radius=15)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Final: ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
