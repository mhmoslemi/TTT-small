            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

def _sum_con(x, n):
    return np.sum(x) - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 100})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, max_passes=500):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            return h, best_c
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def local_pair_search(h, n, deadline, deltas, radius=80, max_passes=300):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            return h, best_c
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def full_optimize(h, n, deadline, rng, do_full_pairs=True):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # SLSQP with increasing alpha
    alphas = [5000, 50000, 500000, 5000000, 50000000]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=4000, ftol=1e-18)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht.copy()
    
    # Pair search
    if time.time() < deadline - 10:
        if do_full_pairs and n <= 350:
            best_h, best_c = pair_search(best_h, n, deadline - 5,
                [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        else:
            best_h, best_c = local_pair_search(best_h, n, deadline - 5,
                [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                radius=80)
    
    return best_h, best_c
