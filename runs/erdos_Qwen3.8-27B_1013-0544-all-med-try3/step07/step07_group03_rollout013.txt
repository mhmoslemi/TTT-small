def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()

    # np.correlate(p, q, 'full') has length (2n-1)+n-1 = 3n-2
    # result[k] = sum_i p[i+k]*q[i] for valid i
    # T1[j] = sum_i p[i + (n-1-j)] * q[i]
    # = result[2n-2 + (n-1-j)] = result[3n-3-j]
    
    corr_pq = np.correlate(p, q, mode='full')  # length 3n-2
    T1 = corr_pq[3*n-3 : 2*n-3 : -1]  # indices 3n-3, 3n-4, ..., 2n-2, length n
    
    # T2: convolution of h with p
    # T2[j] = sum_{m=0}^{j} h[m] * p[n-1+j-m]
    # = np.convolve(h, p, 'full')[n-1+j]
    
    conv_hp = np.convolve(h, p, mode='full')  # length n + (2n-1) - 1 = 3n-2
    T2 = conv_hp[n-1 : 2*n-1]  # indices n-1, n, ..., 2n-2, length n
    
    grad = dx * (T1 - T2)
    return grad, c5_smooth, p

                h_new = project_feasible(h_new, n)

                stall_count = 0

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)

                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d

                            best_c = c
                            improved = True
