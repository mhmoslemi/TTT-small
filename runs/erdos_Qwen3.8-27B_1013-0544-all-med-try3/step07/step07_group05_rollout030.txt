import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_and_topk(h, n, topk=1):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    c5 = float(np.max(c) * dx)
    # Get indices of top k values
    k_idx = np.argsort(c)[-topk:][::-1]
    return c5, c, k_idx

def grad_at_lag(h, n, lag):
    """Compute gradient of correlation at the given lag."""
    grad = np.zeros(n)
    if lag >= 0:
        # (1-h[j-lag]) for j >= lag
        if n - lag > 0:
            grad[lag:] += 1.0 - h[:n-lag]
        # -h[j+lag] for j < n-lag
        if n - lag > 0:
            grad[:n-lag] -= h[lag:]
    else:
        m = -lag
        # (1-h[j+m]) for j < n-m
        if n - m > 0:
            grad[:n-m] += 1.0 - h[m:]
        # -h[j+m] for j >= m
        if n - m > 0:
            grad[m:] -= h[:n-m]
    return grad

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-15:
        # Distribute among non-bound entries
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:

def subgradient_optimize(h, n, deadline, topk=3, momentum=0.9, 
                         eta0=0.01, eta_min=1e-6, patience=200):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    v = np.zeros(n)
    eta = eta0
    no_improve = 0
    n_iter = 0
    
    while time.time() < deadline:

def fw_linear_min(g, n):
    """Frank-Wolfe: find s in {0≤s≤1, sum s = n/2} minimizing g·s."""
    # Set s[i]=1 for the n/2 indices with smallest g
    idx = np.argsort(g)[:n//2]
    s = np.zeros(n)
    s[idx] = 1.0
    return s

def fw_subgradient(h, n, deadline, topk=5, gamma=0.5, momentum=0.3):
    """Frank-Wolfe style subgradient method."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    v = np.zeros(n)
    
    while time.time() < deadline:
        c5, c, k_idx = compute_c5_and_topk(h, n, topk=topk)
        
        # Average gradient over top k's
        g = np.zeros(n)
        for ki in k_idx:
            lag = int(ki) - (n - 1)
            g += grad_at_lag(h, n, lag)
        g /= len(k_idx)
        
        # FW step: find s that minimizes g·s over feasible set
        s = fw_linear_min(g, n)
        
        # Update
        h_new = (1 - gamma) * h + gamma * s
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        gamma *= 0.999  # Slow decay
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas, max_passes=200, radius=None):
    """Local search by pairwise transfers."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_lo, j_hi = i + 1, n
                else:
                    j_lo = max(0, i - radius)
                    j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def random_binary_init(n, n_blocks, rng):
    """Create a binary initialization with n_blocks blocks of 1s."""
    block_len = n // (2 * n_blocks)
    gap_len = block_len  # gap = block for total sum = n/2
    h = np.zeros(n)
    positions = np.sort(rng.choice(n, n_blocks, replace=False))
    for pos in positions:
        start = pos % n
        for k in range(block_len):
            h[(start + k) % n] = 1.0
    # Fix sum
    h = project_feasible(h, n)
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def grad_at_lag(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[lag:] += 1.0 - h[:n - lag]
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if n - m > 0:
            grad[:n - m] += 1.0 - h[m:]
            grad[m:] -= h[:n - m]
    return grad


def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def _sum_con_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def subgradient_core(h, n, deadline, topk=3, eta0=0.05, momentum=0.85,
                     eta_min=1e-7, patience=300):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    v = np.zeros(n)
    eta = eta0
    no_improve = 0
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        c5, c, k_idx = _compute_c5_topk(h, n, topk)

        g = np.zeros(n)
        for ki in k_idx:
            lag = int(ki) - (n - 1)
            g += grad_at_lag(h, n, lag)
        g /= len(k_idx)
        g *= dx

        v = momentum * v - eta * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            eta = min(eta * 1.02, eta0)
        else:
            h = h_new
            no_improve += 1
            if no_improve > patience:
                eta = max(eta * 0.6, eta_min)
                v *= 0.3
                no_improve = 0
                if eta < eta_min:
                    h = best_h.copy()
                    v = np.zeros(n)
                    eta = eta0 * 0.05

    return best_h, best_c


def _compute_c5_topk(h, n, topk):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    c5 = float(np.max(c) * dx)
    if topk >= len(c):
        k_idx = np.arange(len(c) - 1, -1, -1)
    else:
        k_idx = np.argpartition(c, -topk)[-topk:]
        k_idx = k_idx[np.argsort(c[k_idx])[::-1]]
    return c5, c, k_idx


def subgradient_with_restarts(h, n, deadline, rng, topk=3,
                              eta0=0.05, patience=250):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    while time.time() < deadline - 10:

            best_c = c_curr
            best_h = h_curr
        else:
            h_pert = best_h + rng.normal(0, 0.015, n)
            h_pert = project_feasible(h_pert, n)
            c_pert = compute_c5(h_pert, n)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert

    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas, max_passes=200, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_lo = i + 1
                    j_hi = n
                else:
                    j_lo = max(0, i - radius)
                    j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c


def coordinate_descent(h, n, deadline, step=0.01, max_iters=50000):
    """Coordinate descent: adjust one pair at a time with adaptive step."""
    h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step * (1 if rng.rand() < 0.5 else -1)
        hi = h[i] + d
        hj = h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
            else:
                h[i], h[j] = hi - d, hj + d
                step *= 0.99
                if step < 1e-6:
                    step = 0.005
        else:
            step *= 0.95

    return h, best_c

    return h

            h_sine = sine_init(n, freq, amp, rng)
            alloc = min(40, (deadline - time.time() - 30) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_single(h_sine, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Strategy 4: Final polishing of the best
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        if n <= 300:
            best_h, best_c = fine_pair_search(best_h, n, deadline - 2,
                                              [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                              max_passes=250)
        else:
            best_h, best_c = fine_pair_search(best_h, n, deadline - 2,
                                              [0.001, 0.0005, 0.0002],
                                              radius=80, max_passes=150)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    # Strategy 5: One more subgradient pass on best
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht, ct = subgradient_core(best_h, n, deadline - 3, topk=5,
                                  eta0=0.01, momentum=0.9, patience=100)
        if ct < best_c:
            best_c = ct
            best_h = ht
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)

    if best_h is None:
        n = 250
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
