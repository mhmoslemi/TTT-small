    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

                        best_c = c_new
                        improved = True
                        c = compute_all_lags(h, n)
                        break
                    else:
                        if lag > 0:
                            h[i] -= d
                            h[j] += d
                        else:
                            h[i] += d
                            h[j] -= delta
    return h, best_c

def optimize_sequence(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Broad SLSQP
    if time.time() < deadline - (deadline - time.time()) * 0.3:
        h_new = slsqp_full(best_h, n, alpha=2000, maxiter=3000, ftol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 2: Medium SLSQP
    if time.time() < deadline - 15:
        h_new = slsqp_full(best_h, n, alpha=20000, maxiter=5000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 3: Sharp SLSQP
    if time.time() < deadline - 10:
        h_new = slsqp_full(best_h, n, alpha=200000, maxiter=8000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 4: Active-set optimization
    if time.time() < deadline - 15:
        c = compute_all_lags(best_h, n)
        for top_m in [5, 8, 12]:
            if time.time() > deadline - 10:
                break
            top_idx = np.argsort(c)[-top_m:]
            h_new = slsqp_active(best_h, n, top_idx, alpha=100000, maxiter=5000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
                c = compute_all_lags(best_h, n)

    # Phase 5: Very sharp SLSQP
    if time.time() < deadline - 8:
        h_new = slsqp_full(best_h, n, alpha=2000000, maxiter=6000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 6: Local search
    if time.time() < deadline - 3:
        best_h, best_c = paired_local_search(best_h, n, deadline - 1, base_delta=0.002)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Primary: optimize at parent resolution
    if h_init is not None:
        alloc_primary = min(500, (deadline - t0) * 0.55)
        h_opt, c_opt = optimize_sequence(h_init, n_init, t0 + alloc_primary, rng)
        best_h, best_c = h_opt, c_opt

    # Try nearby even resolutions
    nearby_ns = []
    if best_h is not None:
        nearby_ns = [n_init - 2, n_init + 2, n_init + 4, n_init - 4]
    for n_try in nearby_ns:
        if n_try < 50 or n_try > 800:
            continue
        if n_try % 2 != 0:
            continue
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(80, remaining * 0.2)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_sequence(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    # Try higher resolution
    for n_hi in [400, 500]:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 40
        if remaining < 70:
            break
        alloc = min(90, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_hi)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_hi)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_hi, 0.5)
        h_start = project_feasible(h_start, n_hi)
        h_opt, c_opt = optimize_sequence(h_start, n_hi, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 30) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_sequence(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt

    # Binary push experiment
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        for amt in [0.02, 0.05, 0.1]:
            if time.time() > deadline - 40:
                break
            hb = binary_push(best_h, n, amount=amt)
            c_b = compute_c5(hb, n)
            if c_b < best_c + 0.002:  # worth exploring
                alloc = min(40, deadline - time.time() - 30)
                if alloc > 20:
                    h_opt, c_opt = optimize_sequence(hb, n, time.time() + alloc, rng)
                    if c_opt < best_c:
                        best_c, best_h = c_opt, h_opt

    # Final ultra-fine polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Very high alpha SLSQP
        h_new = slsqp_full(best_h, n, alpha=5000000, maxiter=5000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        # Active set with top 20
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-20:]
        h_new = slsqp_active(best_h, n, top_idx, alpha=500000, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        # Tiny delta local search
        if time.time() < deadline - 2:
            best_h, best_c = paired_local_search(best_h, n, deadline - 1, base_delta=0.0005)

    # Final projection
    if best_h is None:
        n_use = n_init if n_init % 2 == 0 else n_init + 1
        best_h = project_feasible(np.full(n_use, 0.5), n_use)
        best_c = compute_c5(best_h, n_use)
    else:
        n = len(best_h)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
