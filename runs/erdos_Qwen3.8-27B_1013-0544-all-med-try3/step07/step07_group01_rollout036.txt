import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    """Compute smooth approximation of max(c)."""
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, n, alpha):
    """Gradient of smooth max of correlation."""
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, alphas=None, max_iters=200):
    """L-BFGS-B optimization with increasing smooth max parameter."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max(compute_c5_vec(x, n), alphas[-1])  # This won't work with changing alpha
    
    # Better: do one minimize call per alpha
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def obj_alpha(x, a=alpha):
            return smooth_max(compute_c5_vec(x, n), a)
        
        def grad_alpha(x, a=alpha):
            return smooth_max_grad(x, n, a)
        
        remaining = deadline - time.time()
        if remaining < 5:
            break
        try:
            res = minimize(obj_alpha, h, jac=grad_alpha, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 5,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(c.max())

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    # Final exact fix: adjust one element
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_max_val(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for lag in range(n_lags):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def coordinate_descent(h, n, deadline, mu=5.0, n_grid=50, max_passes=5000):
    """Coordinate descent with penalty for integral constraint."""
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        order = np.random.permutation(n)
        for i in order:
            if time.time() > deadline:
                return h, best_c
            old_val = h[i]
            if old_val == 0.0 or old_val == 1.0:
                # Still try, might move away from boundary
                pass
            
            # Compute c with h[i] = 0 and h[i] = 1
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = old_val
            
            delta_c = c1 - c0
            S = h.sum() - old_val  # sum of all other components
            
            # Grid of candidate values
            t_grid = np.linspace(0.0, 1.0, n_grid)
            # c(t)[m] = c0[m] + t * delta_c[m]
            # Evaluate max_c at each t:
            # max_vals[t_idx] = max_m (c0[m] + t_grid[t_idx] * delta_c[m])
            # Vectorized:
            # Create matrix: (n_grid, n_lags)
            c_grid = c0[np.newaxis, :] + np.outer(t_grid, delta_c)
            max_c = c_grid.max(axis=1)
            penalty = mu * (S + t_grid - target) ** 2
            total = max_c + penalty
            best_idx = np.argmin(total)
            new_val = t_grid[best_idx]
            
            h[i] = new_val
    
    # Final projection
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:

    """Iterated local search: perturb + local search."""
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for r in range(max_restarts):
        if time.time() > deadline:
            break
        # Perturb
        noise = np.random.randn(n) * noise_std
        h_pert = h + noise
        h_pert = project_feasible(h_pert, n)
        
        # Quick coordinate descent as "local search"
        t_sub = min(time.time() + 30, deadline - 5)
        h_ls, c_ls = coordinate_descent(h_pert, n, t_sub, mu=20.0, n_grid=30, max_passes=2000)
        
        if c_ls < best_c - 1e-12:
            best_c = c_ls
            best_h = h_ls.copy()
            h = h_ls
            # Reduce noise on success
            noise_std *= 0.8
        else:
            # Keep current, gradually increase noise
            noise_std = min(0.15, noise_std * 1.1)
    
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except (NameError, Exception):
        h = np.full(250, 0.5)
        n = 250
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: Coordinate descent (large time allocation)
    t1 = min(t0 + 350, deadline - 450)
    if t1 > t0 + 30:
        h1, c1 = coordinate_descent(h, n, t1, mu=10.0, n_grid=50, max_passes=50000)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
    
    # Phase 2: Second round of coordinate descent with different mu
    if time.time() < deadline - 350:
        t2 = min(time.time() + 200, deadline - 350)
        h2, c2 = coordinate_descent(best_h, n, t2, mu=50.0, n_grid=60, max_passes=50000)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
    
    # Phase 3: L-BFGS-B with smooth max
    if time.time() < deadline - 200:
        t3 = min(time.time() + 150, deadline - 200)
        h3, c3 = lbfgs_optimize(best_h, n, t3, max_iters=400)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    
    # Phase 4: SLSQP
    if time.time() < deadline - 150:
        t4 = min(time.time() + 80, deadline - 150)
        h4, c4 = slsqp_optimize(best_h, n, t4, alpha=500000, maxiter=3000)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
    
    # Phase 5: Iterated local search (different basin exploration)
    if time.time() < deadline - 120:
        t5 = min(time.time() + 100, deadline - 120)
        h5, c5 = iterated_local_search(best_h, n, t5, noise_std=0.03, max_restarts=30)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
    
    # Phase 6: Pair local search (fine-tuning)
    if time.time() < deadline - 80:
        t6 = min(time.time() + 80, deadline - 80)
        h6, c6 = pair_local_search(best_h, n, t6,
                                   deltas=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                   radius=80)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
    
    # Phase 7: Try higher resolution
    if time.time() < deadline - 80:
        for n_try in [400, 500, 600]:
            if time.time() > deadline - 60:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new_pts = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new_pts, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            
            alloc = min(60, deadline - time.time() - 40)
            if alloc < 25:
                break
            
            h_t, c_t = coordinate_descent(h_interp, n_try, time.time() + alloc * 0.5,
                                         mu=10.0, n_grid=40, max_passes=30000)
            if time.time() < deadline - 30:
                h_t, c_t = lbfgs_optimize(h_t, n_try,
                                         time.time() + min(25, deadline - time.time() - 20),
                                         max_iters=150)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t.copy()
    
    # Phase 8: Final coordinate descent on best
    if time.time() < deadline - 20:
        nf = len(best_h)
        t8 = min(time.time() + 30, deadline - 10)
        h8, c8 = coordinate_descent(best_h, nf, t8, mu=50.0, n_grid=60, max_passes=20000)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()
    
    # Phase 9: Final pair search
    if time.time() < deadline - 3:
        nf = len(best_h)
        h9, c9 = pair_local_search(best_h, nf, deadline - 1,
                                   deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                   radius=50)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
