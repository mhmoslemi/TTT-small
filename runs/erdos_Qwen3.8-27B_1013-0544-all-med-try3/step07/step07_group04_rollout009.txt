```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def compute_weighted_gradient(h, n, weights):
    dx = 2.0 / n
    g = np.zeros(n)
    w0 = weights[n - 1]
    if abs(w0) > 1e-15:
        g += w0 * dx * (1.0 - 2.0 * h)
    for m in range(1, n):
        w = weights[n - 1 + m] + weights[n - 1 - m]
        if abs(w) < 1e-15:
            continue
        g[m:] += w * dx * (1.0 - h[:n - m])
        g[:n - m] -= w * dx * h[m:]
    return g


def softmax_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    cm = c.max()
    sh = alpha * (c - cm)

    if mx > 500:
        e = np.exp(sh - mx)
        return float(cm + (mx + np.log(e.sum())) / alpha)
    e = np.exp(sh)
    return float(cm + np.log(e.sum()) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_con_jac(x, n):
    return np.ones(n)


def slsqp_analytic(h, n, alpha, maxiter=4000, ftol=1e-17):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(softmax_obj, h, args=(n, alpha), jac=softmax_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def slsqp_active(h, n, top_idx, alpha, maxiter=4000, ftol=1e-17):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(active_soft_obj, h, args=(n, top_idx, alpha),
                       jac=active_soft_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def lp_outer_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1.0,
            cp.abs(h_var - h_curr) <= trust]
    n_lags = 2 * n - 1
    for k_idx in range(n_lags):
        lag = k_idx - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if m == 0:
            grad = dx * (1.0 - 2.0 * h_curr)
        elif lag > 0:
            grad = np.zeros(n)
            grad[m:] -= h_curr[:n - m]
            grad[m:] += 1.0
            grad[:n - m] += (1.0 - h_curr[m:])
            grad[:n - m] -= 1.0
            grad[m:] += 0
            grad = np.zeros(n)
            # For lag > 0: overlap = dx * sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
            # d/dh[j]: if j in [m, n-1]: dx*(1-h[j-m])
            #          if j in [0, n-1-m]: dx*(-h[j+m])
            grad[m:] = dx * (1.0 - h_curr[:n - m])
            grad[:n - m] -= dx * h_curr[m:]
        else:
            # lag < 0, m = -lag
            # overlap = dx * sum_{i=m}^{n-1} h[i]*(1-h[i-m])
            # d/dh[j]: if j in [m, n-1]: dx*(1-h[j-m])
            #          if j in [0, n-1-m]: dx*(-h[j+m])
            grad = np.zeros(n)
            grad[m:] = dx * (1.0 - h_curr[:n - m])
            grad[:n - m] -= dx * h_curr[m:]
        cons.append(c[k_idx] + grad @ (h_var - h_curr) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return h_curr.copy(), None
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy(), None
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy(), None
    return project_feasible(h_new, n), t_var.value


def outer_approx_lp(h, n, deadline, max_iters=40):

    best_h = h_curr.copy()

    trust = 0.12
    for it in range(max_iters):
        if time.time() > deadline - 5:

        if step < 1e-9:
            break
        h_curr = h_new
        trust *= 0.85
        trust = max(trust, 0.002)
    return best_h, best_c

    best_c = compute_c5(best_h, n)

    # SLSQP with analytical gradient
    for alpha in [5000, 50000, 200000]:
        if time.time() > deadline - 15:
            break
        h_new = slsqp_analytic(best_h, n, alpha, maxiter=4000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new

    # Outer approximation LP
    if time.time() < deadline - 40:
        oo_end = min(deadline - 40, time.time() + 100)
        h_oo, c_oo = outer_approx_lp(best_h, n, oo_end, max_iters=40)
        if c_oo is not None and c_oo < best_c:
            best_c = c_oo
            best_h = h_oo

    # Active set SLSQP
    if time.time() < deadline - 20:
        for top_m in [4, 6, 10]:
            if time.time() > deadline - 20:
                break
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-top_m:]
            for alpha in [20000, 100000]:
                if time.time() > deadline - 20:
                    break
                h_new = slsqp_active(best_h, n, top_idx, alpha, maxiter=3000, ftol=1e-17)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new

    # Final SLSQP
    if time.time() < deadline - 15:
        h_new = slsqp_analytic(best_h, n, 500000, maxiter=5000, ftol=1e-18)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new

    # Local search
    if time.time() < deadline - 5:
        best_h, best_c = local_search_multi(best_h, n, min(deadline - 2, time.time() + 40))

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize at initial resolution
    if h_init is not None:
        alloc_end = t0 + min(200, (deadline - t0) * 0.28)
        best_h, best_c = optimize_at_n(h_init, n_init, alloc_end, rng)

    # Phase 2: Try other resolutions
    n_candidates = [400, 200, 600, 250, 350, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 80:
            break
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        elif h_init is not None:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        alloc = min(130, remaining * 0.35)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(1, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 50) * 0.45)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Very high alpha SLSQP
        if time.time() < deadline - 20:
            h_new = slsqp_analytic(best_h, n, 1000000, maxiter=5000, ftol=1e-18)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        # Final local search
        best_h, best_c = local_search_multi(best_h, n, deadline - 2)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
```
