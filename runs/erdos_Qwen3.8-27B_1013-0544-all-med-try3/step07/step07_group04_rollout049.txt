Actually, the standard definition is:
`correlate(a, v, 'full')[k] = sum_i a[i] * conj(v[i - (k - (len(v)-1))])`

For our case with a=h and v=1-h:
`correlate(h, 1-h, 'full')[k] = sum_i h[i] * (1-h[i-(k-(n-1))])`

So for lag = k - (n-1):

- When lag > 0: sum_{i=lag}^{n-1} h[i] * (1-h[i-lag])
- When lag < 0: sum_{i=0}^{n-1-lag} h[i] * (1-h[i-lag])

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax(vals, alpha):
    m = vals.max()
    s = alpha * (vals - m)
    sm = s.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return smax(c, alpha)

def _obj_top(x, n, idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return smax(c[idx], alpha)

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_run(h0, n, alpha, top_idx=None, maxiter=5000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    if top_idx is not None:
        f = _obj_top
        args = (n, top_idx, alpha)
    else:
        f = _obj_full
        args = (n, alpha)
    try:
        res = minimize(f, h0, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def _lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
    elif m >= n:
        pass
    elif lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += 1.0 - h_curr[:n - m]
    else:
        grad[:n - m] += 1.0 - h_curr[m:]
        grad[m:] -= h_curr[:n - m]
    return grad

def outer_lp_step(h_curr, n, trust):
    dx = 2.0 / n
    c_all = np.correlate(h_curr, 1.0 - h_curr, mode='full') * dx
    c_max = c_all.max()
    active_mask = c_all >= (c_max - 0.02)
    top_idx = np.argsort(c_all)[-30:]
    active_mask[top_idx] = True

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    if trust > 0:
        cons.append(cp.norm(h_var - h_curr, 2) <= trust)

    for k in range(2 * n - 1):
        if not active_mask[k]:
            continue
        lag = k - (n - 1)
        g = _lag_gradient(h_curr, n, lag)
        cons.append(c_all[k] + dx * (g @ (h_var - h_curr)) <= t_var)

    for k in range(2 * n - 1):
        if active_mask[k]:
            continue
        lag = k - (n - 1)
        g = _lag_gradient(h_curr, n, lag)
        cons.append(c_all[k] + dx * (g @ (h_var - h_curr)) <= t_var + 0.008)

    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

            break

        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)

        h_opt, c_opt = refine_full(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(1, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.015 + 0.008 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.45)
            if alloc < 20:
                break
            h_opt, c_opt = refine_full(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        h_new = slsqp_run(best_h, n, 500000, maxiter=6000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new

    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = local_search(best_h, n, deadline - 1, rng=rng)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
