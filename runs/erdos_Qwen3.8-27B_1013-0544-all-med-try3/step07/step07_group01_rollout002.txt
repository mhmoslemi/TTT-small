    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_corr(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for m_idx in range(n_lags):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj_func(x, n, alpha):
    c = compute_corr(x, n)
    return smooth_max_val(c, alpha)

def _jac_func(x, n, alpha):
    return smooth_max_grad(x, n, alpha)

def _con_eq(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def do_lbfgs(h, n, deadline, alphas_list, max_iters_per_alpha=200):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas_list:
        if time.time() > deadline:
            break
        t_alloc = min(40, deadline - time.time())
        if t_alloc < 3:
            break
        try:
            res = minimize(_obj_func, h, jac=_jac_func,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters_per_alpha,
                                   'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def do_slsqp(h, n, deadline, alphas_list, maxiter=3000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac, 'args': (n,)}]
    for alpha in alphas_list:
        if time.time() > deadline:
            break
        t_alloc = min(60, deadline - time.time())
        if t_alloc < 5:
            break
        try:
            res = minimize(_obj_func, h, jac=_jac_func,
                          args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-17})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas, radius):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[j] - d >= 0.0 and h[i] + d <= 1.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def binary_snap(h, n, strength=1.3):
    h2 = h.copy()
    for i in range(n):
        if h2[i] > 0.5:
            h2[i] = 0.5 + (h2[i] - 0.5) * strength
        else:
            h2[i] = 0.5 - (0.5 - h2[i]) * strength
    return project_feasible(h2, n)

def gradient_descent_fine(h, n, deadline, lr_init=0.005, alpha_init=500.0):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    for it in range(5000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.6
            alpha *= 1.3
        if lr < 1e-14:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    n_best = n_init

    # Phase 1: L-BFGS-B at initial resolution (budget: ~150s)
    t_phase1 = min(t0 + 150, deadline - 500)
    h1, c1 = do_lbfgs(best_h, n_init, t_phase1,
                      [500, 5000, 50000, 200000, 1000000],
                      max_iters_per_alpha=150)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: SLSQP at initial resolution (budget: ~200s)
    t_phase2 = min(time.time() + 200, deadline - 300)
    h2, c2 = do_slsqp(best_h, n_init, t_phase2,
                      [500000, 2000000, 10000000],
                      maxiter=3000)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Gradient descent fine-tuning
    if time.time() < deadline - 250:
        t3 = min(time.time() + 60, deadline - 250)
        h3, c3 = gradient_descent_fine(best_h, n_init, t3,
                                       lr_init=0.003, alpha_init=2000.0)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Try higher resolution n=400
    t_phase4_start = time.time()
    if t_phase4_start < deadline - 300:
        n4 = 400
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n4)
        h4 = np.interp(x_new, x_old, best_h)
        h4 = project_feasible(h4, n4)
        t_end4 = min(time.time() + 200, deadline - 200)
        h4, c4 = do_lbfgs(h4, n4, min(t_end4 - 100, time.time() + 80),
                          [5000, 50000, 200000], max_iters_per_alpha=200)
        if time.time() < t_end4:
            h4, c4 = do_slsqp(h4, n4, t_end4,
                              [500000, 2000000], maxiter=2500)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            n_best = n4

    # Phase 5: Try n=500 if time allows
    if time.time() < deadline - 250:
        n5 = 500
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n5)
        h5 = np.interp(x_new, x_old, best_h)
        h5 = project_feasible(h5, n5)
        t_end5 = min(time.time() + 150, deadline - 150)
        h5, c5v = do_lbfgs(h5, n5, min(t_end5 - 60, time.time() + 60),
                           [10000, 100000], max_iters_per_alpha=150)
        if time.time() < t_end5:
            h5, c5v = do_slsqp(h5, n5, t_end5,
                               [1000000], maxiter=2000)
        if c5v < best_c:
            best_c = c5v
            best_h = h5
            n_best = n5

    # Phase 6: Binary snap + re-optimize
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        h_snap = binary_snap(best_h, n_cur, strength=1.2)
        c_snap = compute_c5(h_snap, n_cur)
        if c_snap < best_c:
            best_h = h_snap
            best_c = c_snap
        if time.time() < deadline - 60:
            h_s, c_s = do_slsqp(best_h, n_cur,
                                min(time.time() + 50, deadline - 60),
                                [2000000, 10000000], maxiter=2000)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 7: Pair search - coarse to fine
    n_cur = len(best_h)
    if time.time() < deadline - 50:
        t_ps = min(time.time() + 150, deadline - 50)
        deltas_coarse = [0.01, 0.005, 0.002, 0.001]
        rad = min(60, n_cur // 4)
        best_h, best_c = pair_search(best_h, n_cur, t_ps,
                                     deltas_coarse, radius=rad)

    # Phase 8: Fine pair search
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        t_ps2 = min(time.time() + 100, deadline - 30)
        deltas_fine = [0.0005, 0.0002, 0.0001, 0.00005]
        rad2 = min(40, n_cur // 6)
        best_h, best_c = pair_search(best_h, n_cur, t_ps2,
                                     deltas_fine, radius=rad2)

    # Phase 9: Final SLSQP polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        t_final = min(time.time() + 40, deadline - 8)
        h_f, c_f = do_slsqp(best_h, n_cur, t_final,
                            [10000000, 50000000], maxiter=2000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 10: Tiny pair search
    if time.time() < deadline - 2:
        n_cur = len(best_h)
        best_h, best_c = pair_search(best_h, n_cur, deadline - 1,
                                     [0.00002, 0.00001, 0.000005],
                                     radius=25)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
