    else:

def subgradient_step(h, n, step_size, top_m=3):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Get the top m lags
    top_indices = np.argsort(c)[-top_m:]
    
    # Compute subgradient for the max of top lags
    # c_k = sum_i h[i] * (1 - h[i+k]) * dx (with appropriate index ranges)
    # dc_k/dh[i] = (1 - h[i+k]) * dx - h[i-k] * dx (when both terms exist)
    # (the second term comes from h[i] appearing in h[i-k] position)
    
    grad = np.zeros(n)
    for k_idx in top_indices:
        # k_idx is in [0, 2n-2], corresponding to lag k = k_idx - (n-1)
        lag = k_idx - (n - 1)
        if lag >= 0:
            # c_k = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k]) * dx
            for i in range(n - lag):
                grad[i] += (1 - h[i + lag]) * dx * (1.0 / top_m)
                if i + lag < n:
                    grad[i + lag] += (-h[i]) * dx * (1.0 / top_m)
        else:
            lag_abs = -lag
            # c_k = sum_{i=lag_abs}^{n-1} h[i] * (1-h[i-lag_abs]) * dx
            for i in range(lag_abs, n):
                grad[i] += (1 - h[i - lag_abs]) * dx * (1.0 / top_m)
                if i - lag_abs >= 0:
                    grad[i - lag_abs] += (-h[i]) * dx * (1.0 / top_m)
    
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

            if tot < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corrs(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sum_fun(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(x.shape)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 500:
        val = m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    return float(val)

def slsqp(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(soft_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def pair_search_full(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def equalize_lags(h, n, deadline, n_iter=500, top_m=5):
    """Iteratively reduce the spread of top-m correlations."""
    best_c = compute_c5(h, n)
    step = 0.01
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_corrs(h, n)
        m = c.max()
        if m <= best_c:
            best_c = m
        top_idx = np.argsort(c)[-top_m:]
        top_vals = c[top_idx]
        spread = top_vals.max() - top_vals.min()
        if spread < 1e-8:
            break
        # Compute subgradient of the max (average of top-m normals)
        grad = np.zeros(n)
        for k_idx in top_idx:
            lag = k_idx - (n - 1)
            if lag >= 0:
                for i in range(n - lag):
                    grad[i] += (1.0 - h[i + lag]) / top_m
                    if i + lag < n:
                        grad[i + lag] += (-h[i]) / top_m
            else:
                la = -lag
                for i in range(la, n):
                    grad[i] += (1.0 - h[i - la]) / top_m
                    if i - la >= 0:
                        grad[i - la] += (-h[i]) / top_m
        # Scale gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            step *= 0.7
            if step < 1e-8:
                break
    return h, best_c

    deadline = t0 + budget_s - 5

            h = h_new
