```python
    """Simulated annealing on binary solutions"""
    h = h_binary.copy()
    best_h = h.copy()
    best_overlap = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    
    total_iters = 100000
    current_iter = 0
    
    while time.time() < deadline:
        current_iter += 1
        T = T0 * (T_end / T0) ** (current_iter / total_iters)

        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)

            lag = rng.randint(1, n//2)

            continue

        i = rng.randint(0, n - m)

            c_all = compute_all_lags(h, n)
        else:
            # Undo
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_overlap
```

```python
                  bounds=bounds, options={'maxiter': 10000, 'ftol': 1e-20, 'gtol': 1e-10})
    return project_feasible(res.x, n)
```

This should be much faster than SLSQP for n=500.

But wait, with a penalty of 1e6, the effective objective might be dominated by the penalty, making the optimization focus on satisfying the constraint rather than minimizing the actual objective. I need to balance this.

A better approach: use an adaptive penalty (start small, increase over time). Or use an augmented Lagrangian.

Actually, the simplest approach that works well in practice:
1. Start with h that satisfies the constraint
2. At each L-BFGS-B step, the bound constraints are handled, but the sum constraint might be violated
3. After each batch of iterations, project back to the feasible set

Or even simpler: just use the projected gradient method with a good step size. For n=500, each iteration is O(n^2) (for the correlate), which is about 250,000 operations. In numpy, this takes about 50-100μs per iteration. So we can do about 10,000-20,000 iterations per second. With 300 seconds, that's 3-6 million iterations. More than enough.

Let me go with the projected gradient approach but with better step size selection and momentum.

OK, I think I've spent enough time planning. Let me write the code.

Here's my final design:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def vec_lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag > 0:
        grad[0:n-m] += (1.0 - h[lag:n])
        grad[lag:n] += (-h[0:n-m])
    elif lag < 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += (-h[m:n])
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def smooth_max_with_grad(h, n, alpha, top_k=None):
    """Compute smooth max value, weights, and gradient. Optionally limit to top_k lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    
    if top_k is not None and top_k < len(c):
        # Only consider top K lags
        top_idx = np.argsort(c)[-top_k:]
        c_active = c[top_idx]
        m = c_active.max()
        shifted = alpha * (c_active - m)
        mx = shifted.max()
        if mx > 500:
            w_active = np.exp(shifted - mx)
        else:
            w_active = np.exp(shifted)
        w_active = w_active / w_active.sum()
        w = np.zeros(len(c))
        w[top_idx] = w_active
        val = m + np.log(w_active.sum()) / alpha
    else:
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        val = m + np.log(w.sum()) / alpha
    
    val = float(val)
    
    # Gradient: only compute for lags with significant weight
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-8:
            continue
        lag = k - (n-1)
        grad += w[k] * vec_lag_gradient(h, n, lag)
    grad *= dx
    return val, w, grad

def projected_grad_optimize(h, n, alpha_start, alpha_end, deadline, max_iter=500000):
    """Projected gradient with momentum and adaptive alpha"""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    lr = 0.05
    momentum = 0.9
    alpha = alpha_start
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Adaptive alpha: increase over time
        progress = min(1.0, it / (max_iter * 0.8))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        val, w, grad = smooth_max_with_grad(h_curr, n, alpha)
        
        # Momentum update
        v = momentum * v + lr * grad
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            lr = min(lr * 1.02, 0.1)
        else:
            # Step back
            h_curr = h_new  # still move forward but reduce step
            lr *= 0.9
            v *= 0.5
            if lr < 0.001:
                break
    
    return best_h, best_c

def binary_simulated_annealing(h_binary, n, deadline, rng):
    """SA on binary solutions"""
    h = (h_binary > 0.5).astype(float)
    # Ensure exactly n/2 ones
    idx = np.argsort(h_binary)[::-1][:n//2]
    h = np.zeros(n)
    h[idx] = 1.0
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    
    it = 0
    T = 0.01
    T_min = 1e-6
    
    while time.time() < deadline:
        it += 1
        T = max(T_min, T * 0.99999)
        
        # Focus on worst lag
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            lag = rng.randint(1, n//2)
        
        m = abs(lag)
        i = rng.randint(0, n - m)
        j = i + m
        
        if h[i] == h[j]:
            continue
        
        # Swap
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        
        if c_new < best_c or (c_new > best_c and rng.random() < np.exp(-(c_new - best_c) / T)):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            c_all = compute_all_lags(h, n)
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c

def top_k_local_search(h, n, deadline, rng, top_k=5):
    """Local search considering top K lags"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    stagnant = 0
    
    while time.time() < deadline:
        stagnant += 1
        if stagnant > 5000:
            break
        
        # Get top K lags
        top_idx = np.argsort(c_all)[-top_k:]
        
        improved = False
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            
            # Try several candidate moves at this lag
            for _ in range(50):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n - m)
                j = i + m
                
                if h[i] > h[j] + 1e-6:
                    delta = min(rng.uniform(0.002, 0.02), (h[i]-h[j])*0.3)
                elif h[j] > h[i] + 1e-6:
                    delta = -min(rng.uniform(0.002, 0.02), (h[j]-h[i])*0.3)
                else:
                    continue
                
                if abs(delta) < 1e-8:
                    continue
                
                h[i] -= delta
                h[j] += delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    c_all = compute_all_lags(h, n)
                    improved = True
                    stagnant = 0
                else:
                    h[i] += delta
                    h[j] -= delta
                    h = project_feasible(h, n)
        
        if not improved:
            stagnant += 100
    
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new points"""
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
