def smooth_max(c, alpha):
    """Compute smooth approximation of max via log-sum-exp."""
    m = c.max()
    shifted = alpha * (c - m)
    # Use log-sum-exp trick
    max_s = shifted.max()
    val = m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha
    weights = np.exp(shifted - max_s)
    weights /= weights.sum()
    return float(val), weights

    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    weights = np.exp(shifted - max_s)
    weights /= weights.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)

    # The correlation at lag k is sum_i h[i] * (1-h[i+k]) * dx
    # (with appropriate bounds)

        if weights[lag_idx] < 1e-15:
            continue
        k = lag_idx - (n - 1)
        w = weights[lag_idx] * dx

def lbfgs_with_constraint(h, n, deadline, max_iters=300):
    """L-BFGS-B with soft constraint on sum."""
    best_c = compute_c5(h, n)
    best_h = h.copy()

            break

            val = smooth_max_val(x, n, alpha)
            val += mu * (x.sum() - n / 2.0) ** 2
            return val

            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def grad_c5(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    weights = np.exp(shifted - max_s)
    weights /= weights.sum()

    dx = 2.0 / n
    grad = np.zeros(n)

    for lag_idx in range(2 * n - 1):
        if weights[lag_idx] < 1e-14:
            continue
        k = lag_idx - (n - 1)
        w = weights[lag_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += w * (1.0 - h[k:n])
            grad[k:n] += w * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += w * (1.0 - h[0:nv])
            grad[0:nv] += w * (-h[kk:n])

    return grad


def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    return float(m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha)

    val = smooth_max_val(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    return smooth_max_val(x, n, alpha)


def slsqp_grad(x, n, alpha):
    return grad_c5(x, n, alpha)


def slsqp_con_fun(x, n):
    return float(x.sum() - n / 2.0)


def slsqp_con_jac(x, n):

    bounds = [(0.0, 1.0)] * n
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]

                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

            improved = False

            order_up = np.argsort(-g)
            order_down = np.argsort(g)
            for i in order_up[:n // 2]:

                    c = compute_c5(best_h, n)
