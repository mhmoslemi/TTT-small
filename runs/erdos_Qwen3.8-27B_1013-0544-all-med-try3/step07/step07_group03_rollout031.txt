        else:
            e = np.exp(sh)
        w = e / e.sum()

                if n - 1 - t >= 0:
                    grad[0:n-t] -= wi * dx * h[t:n]

        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm
        
        # Momentum update
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stuck_count = 0
            # Slowly decrease lr
            lr *= 0.999
        else:
            stuck_count += 1
            if stuck_count > 50:
                lr *= 0.5
                v = np.zeros(n)
                stuck_count = 0
                # Increase alpha to sharpen
                alpha = min(alpha * 2, 1000000)
        
        if lr < 1e-8:
            break
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

                        if time.time() > deadline - 2:
                            break
                        lo = max(0, i - radius)
                        hi = min(len(h_copy), i + radius + 1)
                        for j in range(lo, hi):
                            if i == j:
                                continue
                            if h_copy[i] + d <= 1.0 and h_copy[j] - d >= 0.0:
                                h_copy[i] += d
                                h_copy[j] -= d
                                c = compute_c5(h_copy, len(h_copy))
                                if c < best_c - 1e-15:
                                    best_c = c
                                    best_h = h_copy.copy()
                                    improved = True
                                else:
                                    h_copy[i] -= d
                                    h_copy[j] += d
                            if h_copy[i] - d >= 0.0 and h_copy[j] + d <= 1.0:
                                h_copy[i] -= d
                                h_copy[j] += d
                                c = compute_c5(h_copy, len(h_copy))
                                if c < best_c - 1e-15:
                                    best_c = c
                                    best_h = h_copy.copy()
                                    improved = True
                                else:
                                    h_copy[i] += d
                                    h_copy[j] -= d
            h_copy = best_h.copy()
    
    # Phase 5: Final polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 5000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 1:
            # Tiny pair moves
            h_copy = best_h.copy()
            for d in [0.0001, 0.00005]:
                for i in range(n):
                    if time.time() > deadline - 0.5:
                        break
                    for j in range(i+1, n):
                        if h_copy[i] + d <= 1.0 and h_copy[j] - d >= 0.0:
                            h_copy[i] += d
                            h_copy[j] -= d
                            c = compute_c5(h_copy, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h_copy.copy()
                            else:
                                h_copy[i] -= d
                                h_copy[j] += d
                        if h_copy[i] - d >= 0.0 and h_copy[j] + d <= 1.0:
                            h_copy[i] -= d
                            h_copy[j] += d
                            c = compute_c5(h_copy, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h_copy.copy()
                            else:
                                h_copy[i] += d
                                h_copy[j] -= d
    
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
