    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    ms = shifted.max()
    es = np.exp(shifted - ms)
    w = es / es.sum()
    val = m + (ms + np.log(es.sum())) / alpha
    
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        k = lag + (n - 1)
        wk = w[k] * dx
        if abs(wk) < 1e-13:
            continue
        if lag >= 0:
            grad[:n-lag] += wk * (1.0 - h[lag:])
        else:
            grad[-lag:] += wk * (1.0 - h[:n+lag])
        if lag > 0:
            grad[lag:] -= wk * h[:n-lag]
        elif lag < 0:
            grad[:n+lag] -= wk * h[-lag:]
    
    return val, grad

def projected_gradient(h, n, deadline, alphas=None, max_iter=50000):

        alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_cur = h.copy()
        c_cur = compute_c5(h_cur, n)
        lr = 1.0 / alpha  # initial learning rate
        no_improve = 0
        
        for it in range(max_iter):
            if time.time() > deadline:
                break
            val, grad = smooth_obj_grad(h_cur, n, alpha)
            # Backtracking line search
            for _ in range(30):
                h_new = np.clip(h_cur - lr * grad, 0.0, 1.0)
                h_new = project_sum(h_new, target)
                c_new = compute_c5(h_new, n)
                if c_new < c_cur - 1e-12:
                    h_cur = h_new
                    c_cur = c_new
                    no_improve = 0
                    break
                lr *= 0.5
            else:
                # No improvement possible with this step
                lr = max(lr, 1e-10)
                no_improve += 1
                if no_improve > 50:
                    break
            if no_improve > 200:
                break
        
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
        h = best_h.copy()
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=2000, n_restarts=3, seed=42):

    best_c = np.inf
    best_h = np.zeros(n)
    rng = np.random.RandomState(seed)

        if time.time() > deadline:
            break
        if restart == 0:
            # Start from sorted (top n/2 are 1)
            sorted_idx = np.argsort(h)[::-1]
            h_cur = np.zeros(n)
            h_cur[sorted_idx[:target]] = 1.0
        else:
            # Perturb from best
            h_cur = best_h.copy()
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            for _ in range(rng.randint(3, 10)):
                if len(ones) > 0 and len(zeros) > 0:
                    a = rng.choice(ones)
                    b = rng.choice(zeros)
                    h_cur[a], h_cur[b] = 0.0, 1.0
        
        corr = np.correlate(h_cur, 1 - h_cur, mode='full')
        cur_c = corr.max() * dx
        ones_list = list(np.where(h_cur == 1)[0])
        zeros_list = list(np.where(h_cur == 0)[0])
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            # Shuffle for diversity
            rng2 = np.random.RandomState(seed + restart * 1000 + sweep)
            rng2.shuffle(zeros_list)
            rng2.shuffle(ones_list)
            
            for i in zeros_list:
                if time.time() > deadline:
                    break
                for j in ones_list:
                    if abs(i - j) > 2 * n:  # skip (always true, just for structure)
                        continue
                    # Compute new correlation
                    new_h = h_cur.copy()
                    new_h[i] = 1.0
                    new_h[j] = 0.0
                    new_corr = np.correlate(new_h, 1 - new_h, mode='full')
                    nc = new_corr.max() * dx
                    if nc < cur_c - 1e-15:
                        h_cur = new_h
                        corr = new_corr
                        cur_c = nc
                        ones_list.remove(j)
                        zeros_list.remove(i)
                        ones_list.append(i)
                        zeros_list.append(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_cur.copy()
    
    return best_h, best_c
