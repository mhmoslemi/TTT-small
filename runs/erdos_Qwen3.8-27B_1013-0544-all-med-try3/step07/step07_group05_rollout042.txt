        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_c_k(h, n, k):
    """Compute gradient of c[k] w.r.t. h, where c = correlate(h, 1-h, 'full')."""
    idx = np.arange(n)
    g = np.zeros(n)
    # Term 1: (1 - h[i + k - (n-1)]) if valid
    idx1 = idx + k - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    g[valid1] = 1.0 - h[idx1[valid1]]
    # Term 2: -h[i - k + (n-1)] if valid
    idx2 = idx - k + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    g[valid2] -= h[idx2[valid2]]
    return g

def subgradient_optimize(h, n, deadline, lr0=0.02, decay=0.999, momentum=0.3, tol_frac=0.05):
    """Subgradient method with momentum for minimizing max correlation."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    step = lr0
    
    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full')
        c_max = c.max()
        
        # Active set: within tol_frac of max
        active_mask = c >= (c_max - c_max * tol_frac)
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask[np.argmax(c)] = True
            n_active = 1
        
        # Average gradient over active set
        g = np.zeros(n)
        for k in np.where(active_mask)[0]:
            g += grad_c_k(h, n, int(k))
        g /= n_active
        
        # Momentum update
        v = momentum * v - step * g
        h = h + v
        h = project_feasible(h, n)
        
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        
        step *= decay
    
    return best_h, best_c

    # Phase 4: Final SLSQP refinement
    if time.time() < deadline - 3:

    return best_h, best_c
