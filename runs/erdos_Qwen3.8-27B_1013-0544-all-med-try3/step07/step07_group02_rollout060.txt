def compute_c5(h, n):
    dx = 2.0 / n

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
        return float(m + (mx + np.log(np.sum(w))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()

        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)

    if s < 1e-300:

        return w

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] += -h[m:]
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] += -h[:n - m]
    return grad

        lag = k - (n - 1)
        grad += w[k] * lag_gradient(h, n, lag)
    return grad * dx

    return np.ones(len(x))

        h4a = project_feasible(h4a, n_cur)
        alloc4 = min(90, (deadline - time.time()) * 0.15)
        h4, c4 = gradient_descent(h4a, n_cur, time.time() + alloc4,
                                  alpha=50000, lr=0.08, momentum=0.95)
        if c4 < best_c:
            best_c = c4
            best_h = h4
        best_h_global, best_c_global = best_h, best_c

        # Small noise restart
        if time.time() < deadline - 120:
            h4b = best_h + rng.normal(0, 0.06, n_cur)
            h4b = project_feasible(h4b, n_cur)
            alloc4b = min(60, (deadline - time.time()) * 0.1)
            h4b_opt, c4b = gradient_descent(h4b, n_cur, time.time() + alloc4b,
                                            alpha=30000, lr=0.1, momentum=0.94)
            if c4b < best_c:
                best_c = c4b
                best_h = h4b_opt
            best_h_global, best_c_global = best_h, best_c

    # Phase 5: Multi-resolution exploration
    if time.time() < deadline - 120:
        multi_resolution_optimize(best_h, len(best_h), deadline, rng)
        if best_h_global is not None and best_c_global < best_c:
            best_c = best_c_global
            best_h = best_h_global

    # Phase 6: SLSQP on best
    if time.time() < deadline - 40:
        n_fin = len(best_h)
        h6 = slsqp_optimize(best_h, n_fin, alpha=150000, maxiter=2000)
        c6 = compute_c5(h6, n_fin)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final binary search pass (short)
    if time.time() < deadline - 30 and len(best_h) <= 600:
        h7b, c7b = binary_swap_search(best_h, len(best_h), deadline - 15,
                                      max_iter=3000, seed=seed + 99)
        if c7b < best_c:
            best_c = c7b
            best_h = h7b

    # Phase 8: Final local search
    if time.time() < deadline - 3:
        n_fin = len(best_h)
        h8, c8 = fine_local_search(best_h, n_fin, deadline - 1)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
