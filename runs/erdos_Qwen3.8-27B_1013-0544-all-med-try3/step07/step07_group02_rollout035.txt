import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h *= (1.0 + diff / t)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1][:n // 2]
    b = np.zeros(n)
    b[idx] = 1.0

def lag_slope(h, n, lag):
    """Compute s(p) for the given lag. Positive lag means C = sum h[i]*(1-h[i+lag]).
    For the swap to improve C, we need s(i) < s(j) where i is being removed (was 1) and j added (was 0).
    """
    s = np.zeros(n)
    if lag > 0:
        s[:n - lag] -= (1.0 - h[lag:])
        s[lag:] += h[:n - lag]
    elif lag < 0:
        m = -lag
        s[:n - m] += h[m:]
        s[m:] -= (1.0 - h[:n - m])
    return s

def binary_greedy_search(h, n, deadline, max_rounds=200000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    stall = 0
    center = n - 1

    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        c_vals = np.correlate(h, 1 - h, mode='full')
        k_star = int(np.argmax(c_vals))
        lag = k_star - center
        if c_vals[k_star] == 0 or lag == 0:
            break

        s = lag_slope(h, n, lag)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        s_ones = s[ones]
        s_zeros = s[zeros]

        o_order = np.argsort(s_ones)
        z_order = np.argsort(s_zeros)[::-1]

        found = False
        for oi in range(min(30, len(o_order))):
            ii = ones[o_order[oi]]
            if found:
                break

                    found = True

def binary_annealing(h, n, deadline, T_start=0.005, T_end=0.0001, max_iters=500000, rng=None):

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_obj_jac(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    grad = np.zeros(n)
    center = n - 1
    n_total = 2 * n - 1
    for k_idx in range(n_total):

        if lag == 0:
            grad += w[k_idx] * (1.0 - 2.0 * x)

            grad[:n - m2] += w[k_idx] * (1.0 - x[m2:])

    return grad * dx

def slsqp_polish(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_obj(x, n, alpha)

    def jac(x):
        return smooth_max_obj_jac(x, n, alpha)

    def con_fun(x):
        return x.sum() - n / 2.0

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

        w = np.exp(shifted - mx)

            if w[k_idx] < 1e-12:

            if lag == 0:
