import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, idx, dx):
    n_mid = n - 1
    k = idx - n_mid
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(h, n, alpha, top_k):
    c = compute_c5_vec(h, n)
    if top_k < len(c):
        idx = np.argsort(-c)[:top_k]
        c_use = c[idx]
    else:
        idx = np.arange(len(c))
        c_use = c
    m = c_use.max()
    sh = alpha * (c_use - m)
    smx = sh.max()
    if smx > 500:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha, top_k):
    c = compute_c5_vec(h, n)
    if top_k < len(c):
        idx = np.argsort(-c)[:top_k]
        c_use = c[idx]
    else:
        idx = np.arange(len(c))
        c_use = c
    m = c_use.max()
    sh = alpha * (c_use - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx_val in zip(w, idx):
        if wi < 1e-15:
            continue
        grad += wi * shift_grad(h, n, idx_val, dx)
    return grad

def variance_obj(h, n, top_k, alpha):
    """Objective that penalizes variance of top-k shifts + their mean."""
    c = compute_c5_vec(h, n)
    idx = np.argsort(-c)[:top_k]
    c_top = c[idx]
    mean_c = c_top.mean()
    var_c = c_top.var()
    sh = alpha * (c_top - mean_c)
    smx = sh.max()
    soft_max = mean_c + np.log(np.sum(np.exp(sh - smx)) / len(c_top)) + smx / alpha
    return soft_max + 0.1 * var_c

def variance_grad(h, n, top_k, alpha):
    c = compute_c5_vec(h, n)
    idx = np.argsort(-c)[:top_k]
    c_top = c[idx]
    mean_c = c_top.mean()
    dx = 2.0 / n
    sh = alpha * (c_top - mean_c)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, idx_val in zip(w, idx):
        if wi < 1e-15:
            continue
        g = shift_grad(h, n, idx_val, dx)
        grad += (wi + 1.0 / top_k) * g
        grad -= (1.0 / top_k) * g
    return grad

def run_lbfgs(h, n, deadline, alpha_list=None, top_k_list=None):

    v = np.zeros(n)
    y = h.copy()

        momentum_t = (1 + np.sqrt(1 + 4 * momentum_t**2)) / 2

        y = h + mu * (h - y)
        grad = smooth_max_grad(y, n, alpha, top_k)
        v = momentum * v + grad
        h_new = y - lr * v
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
