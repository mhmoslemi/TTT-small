import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def anneal_run(h_init, n, deadline, seed, T_init=0.15, T_min=0.00005):

    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1.0)[0].copy()
    zeros = np.where(h == 0.0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    N = 2 * n - 1
    delta = np.zeros(N)
    nz = n // 2

    T = T_init
    cool_rate = 0.99995
    stalling = 0
    no_improve = 0
    reheat_count = 0

    while time.time() < deadline:
        # Process in mini-batches to check time and do periodic recomputation
        batch_size = 5000
        for _ in range(batch_size):
            if time.time() >= deadline:
                break
            ai = rng.randint(nz)
            bi = rng.randint(nz)
            i = int(zeros[ai])
            j = int(ones[bi])
            if i == j:

            delta[:] = 0.0
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] = 1.0 - h[im[msk]]
            imn = i - m_range
            msk2 = (imn >= 0) & (imn < n)

            jm = j + m_range

            delta[n - 1] -= 2.0

                delta[k2] += 1.0

def iterative_ls(h_init, n, deadline, seed, n_restarts=5):
    """Perturb best solution, then local search."""
    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    time_per_restart = (deadline - time.time()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.time() > deadline:
            break
        if time.time() > deadline - 3:
            break
        # Perturb: flip k random pairs
        h_pert = best_h.copy()
        n_flip = rng.randint(3, min(12, n // 10))
        for _ in range(n_flip):
            ones_idx = np.where(h_pert == 1.0)[0]
            zeros_idx = np.where(h_pert == 0.0)[0]
            if len(ones_idx) < 1 or len(zeros_idx) < 1:
                break
            oi = rng.randint(len(ones_idx))
            zi = rng.randint(len(zeros_idx))
            h_pert[ones_idx[oi]] = 0.0
            h_pert[zeros_idx[zi]] = 1.0
        # Local search on perturbed
        dl_sr = min(time.time() + time_per_restart * 0.7, deadline - 2)
        h_ls, c_ls = local_search(h_pert, n, dl_sr, max_sweeps=100)
        # Also run brief anneal
        dl_an = min(time.time() + time_per_restart * 0.25, deadline - 1)
        if time.time() < dl_an - 2:

        if c_ls < best_c:

def interpolate_and_optimize(h_best, n_old, n_new, deadline, seed):
    """Interpolate to higher resolution and optimize."""
    if n_new <= n_old:
        return h_best, compute_c5(h_best, n_old), n_old
    xo = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    xn = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_interp = np.interp(xn, xo, h_best, period=2.0)
    h_interp = project(h_interp, n_new)

    h_an, c_an = anneal_run(h_ls, n_new, dl2, seed=seed, T_init=0.1, T_min=0.0001)
    return h_an, c_an, n_new
