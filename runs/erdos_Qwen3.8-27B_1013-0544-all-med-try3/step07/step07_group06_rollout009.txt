def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

        if time.time() > deadline:
            break

            idx = np.argsort(h)[::-1]

            n_swaps = rng.randint(1, max(2, n // 50))

                    h_cur[a], h_cur[b] = 0.0, 1.0
                    ones = np.where(h_cur == 1)[0]

    """Minimize max_k corr_k * dx via projected subgradient."""
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n-1), n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1-h, mode='full')
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        k_star = corr.argmax() - (n-1)  # the max lag
        # Compute gradient of corr_{k_star}
        grad = np.zeros(n)
        for i in range(n):
            g = 0.0
            if 0 <= i + k_star < n:

            if 0 <= i - k_star < n:

            grad[i] = g
        grad *= dx
        
        # Projected gradient step
        h = h - lr * grad
        h = project_feasible(h, n)
        
        # Adaptive learning rate
        if it % 1000 == 0:
            lr *= 0.95
    
    return best_h, best_c

-h[i+k*] - h[i-k*] + 1 = λ
But h[i+k*] and h[i-k*] might also be free...
