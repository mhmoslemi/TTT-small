def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project(h, n):
    # Project to {0<=h<=1, sum(h)=n/2}
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    if abs(diff) > 1e-12:

            s = room.sum()
            if s > 1e-15:

        else:
            s = hp.sum()
            if s > 1e-15:

    cur_c = corr.max() * dx

            break
        improved = False
        for i in list(zeros_list):

                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:

                imn = i - m_range

                jm = j + m_range

                jmn = j - m_range

                nc = (corr + delta).max() * dx

                    best_j = j
            if best_j >= 0:

                    best_h = h.copy()
