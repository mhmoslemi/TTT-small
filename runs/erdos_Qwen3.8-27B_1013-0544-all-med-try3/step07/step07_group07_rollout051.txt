                #   if j appears as m: term -h[m+k] = -h[j+k], valid if j+k in [0,n-1]

                step = min(step * 1.1, 1.0)
            else:
                step *= 0.7
                if step < 1e-8:
                    break
    
    return best_h, best_c

            h_new = project_to_sum1(h_cur - step * grad, n)
            c_new = compute_c5(h_new, n)

            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()

            if c_new <= cur_c + 1e-14:
                h_cur = h_new
                cur_c = c_new
                step = min(step * 1.05, 1.0 / alpha)
            else:
                step *= 0.8
                if step < 1e-10:
                    break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    h_init = to_binary(h0, n)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Initial local search (20% time)
    dl1 = t0 + 0.20 * total
    h1, c1 = local_search(h_init, n, dl1, max_sweeps=300)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Multiple annealing rounds with kicks (45% time)
    dl2 = t0 + 0.65 * total
    n_rounds = 6
    round_time = (dl2 - time.time()) / max(n_rounds, 1)
    h_curr = best_h.copy()

    for r in range(n_rounds):
        if time.time() > dl2:
            break
        # Anneal
        a_dl = min(dl2 - 20, time.time() + round_time * 0.65)
        ha, ca = anneal_binary(h_curr, n, a_dl, seed=seed + 1000 * r)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()

        # Local search after anneal
        ls_dl = min(dl2 - 10, time.time() + 30)
        hls, cls = local_search(ha, n, ls_dl, max_sweeps=80)
        if cls < best_c:
            best_c = cls
            best_h = hls.copy()
        h_curr = hls.copy()

        # Kick: perturb several positions to escape local optimum
        if time.time() < dl2 - 15:
            rng = np.random.RandomState(seed + 50000 + 7 * r)
            h_kick = h_curr.copy()
            n_kick = rng.randint(3, 7)
            ones_k = list(np.where(h_kick == 1)[0])
            zeros_k = list(np.where(h_kick == 0)[0])
            for _ in range(n_kick):
                ii = rng.randint(len(zeros_k))
                jj = rng.randint(len(ones_k))
                h_kick[zeros_k[ii]], h_kick[ones_k[jj]] = 1.0, 0.0
                zeros_k[ii] = ones_k[jj]
                ones_k[jj] = zeros_k[ii]
            # Local search from kicked state
            ks_dl = min(dl2 - 5, time.time() + round_time * 0.3)
            hks, cks = local_search(h_kick, n, ks_dl, max_sweeps=50)
            if cks < best_c:
                best_c = cks
                best_h = hks.copy()
            if cks <= cls:
                h_curr = hks.copy()

    # Phase 3: Continuous PGD polish (20% time)
    dl3 = t0 + 0.88 * total
    h_cont = best_h.astype(float).copy()
    h_cont, c_cont = pgd_continuous(h_cont, n, dl3, alphas=[100, 500, 2000, 10000, 50000, 200000])
    if c_cont < best_c:
        best_c = c_cont
        best_h = h_cont.copy()

    # Phase 4: Binary from continuous, then local search
    if time.time() < dl - 15:
        h_bin = to_binary(best_h, n)
        bin_dl = min(dl - 5, time.time() + 40)
        h_bin, c_bin = local_search(h_bin, n, bin_dl, max_sweeps=200)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

        # Final continuous polish from best binary
        if time.time() < dl - 3:
            h_fin = best_h.astype(float).copy()
            h_fin, c_fin = pgd_continuous(h_fin, n, dl - 2, alphas=[5000, 20000, 100000])
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin.copy()

    # Final projection and evaluation
    best_h = project_to_sum1(best_h, n)
    best_c = compute_c5(best_h, n)

    # Verify constraints
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1 + 1e-10)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
