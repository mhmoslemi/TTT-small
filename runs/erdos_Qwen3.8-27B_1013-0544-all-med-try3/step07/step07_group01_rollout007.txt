def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, k):
    """Gradient of correlation at lag k (k = m - (n-1)) w.r.t. h."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[0:nv] += (1.0 - h[k:n])
        g[k:n] += (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        g[kk:n] += (1.0 - h[0:nv])
        g[0:nv] += (-h[kk:n])
    return g * dx

def active_set_smooth_max(h, n, alpha, top_M=15):
    """Compute smooth max using only top M correlation values, and its gradient."""
    c = compute_c5_vec(h, n)
    # Get top M indices
    top_idx = np.argsort(c)[-top_M:]
    c_top = c[top_idx]
    # Smooth max of top M
    c_shift = alpha * (c_top - c_top.max())
    mx = c_shift.max()
    if mx > 500:
        val = c_top.max() + (mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha
    else:
        val = c_top.max() + np.log(np.sum(np.exp(c_shift - mx))) / alpha
    w = np.exp(c_shift - mx)
    w /= w.sum()
    # Gradient
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        m = top_idx[idx]
        k = m - (n - 1)

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

def lbfgs_active(h, n, deadline, top_M=15, max_iters=500):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

def coordinate_descent(h, n, deadline):
    """Coordinate descent: for each i, try moving h[i] and adjusting h[j] for constraint."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    
    # For each coordinate i, we want to find better h[i]
    # Constraint: sum(h) = target, so if we change h[i] by d, we need to change something else by -d
    # Strategy: change h[i] and distribute -d to all other coordinates proportionally
    
    iterations = 0
    while time.time() < deadline and iterations < 50:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Try several values of h[i]
            for delta in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002,
                          0.001, -0.001, 0.5, -0.5, 1.0 - h[i] - 0.001, h[i] - 0.001]:
                new_hi = h[i] + delta
                if new_hi < 0 or new_hi > 1:
                    continue
                # Distribute -delta to other coordinates
                h_new = h.copy()
                h_new[i] = new_hi
                remaining = -delta
                others = np.ones(n)
                others[i] = 0
                # Proportional redistribution (respecting bounds)
                # Simple: just add to the coordinate with most room
                # Actually, let's do proportional: h_new[j] += remaining * (h_new[j]) / sum(h_new[other])
                # But that might push out of bounds. Let's just add uniformly.
                other_count = n - 1
                uniform_add = remaining / other_count
                h_new -= uniform_add
                h_new[i] = new_hi
                if np.any(h_new < -1e-12) or np.any(h_new > 1 + 1e-12):
                    continue
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
                    break
        iterations += 1
        if not improved:
            break
    return best_h, best_c

def pair_search_fine(h, n, deadline):
    """Fine pair search with adaptive step sizes."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(i + 1, n):
                    # Move step from i to j
                    if h[i] - step >= 0 and h[j] + step <= 1:
                        h[i] -= step; h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] += step; h[j] -= step
                    # Move step from j to i
                    if h[j] - step >= 0 and h[i] + step <= 1:
                        h[j] -= step; h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
