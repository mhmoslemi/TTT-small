import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c_k(h, k, n):
    dx = 2.0 / n
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m_idx = np.arange(s, n)
        grad[m_idx] += dx * (1.0 - h[m_idx - s])
        m_idx2 = np.arange(0, n - s)
        grad[m_idx2] -= dx * h[m_idx2 + s]
    else:
        t = -s
        m_idx = np.arange(0, n - t)
        grad[m_idx] += dx * (1.0 - h[m_idx + t])
        m_idx2 = np.arange(t, n)
        grad[m_idx2] -= dx * h[m_idx2 - t]
    return grad

def subgradient_optimize(h, n, deadline, top_p=5, max_iter=50000, lr_init=0.01):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c.max()
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h.copy()
        
        # Find top-p lags
        top_k = np.argsort(c)[-top_p:]
        # Compute gradient as weighted sum
        # Use log-sum-exp for smooth weights
        alpha = 500.0
        shifted = alpha * (c - c_max)
        max_s = shifted.max()
        weights = np.exp(shifted - max_s)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for k in top_k:
            if weights[k] < 1e-8:
                continue
            g += weights[k] * grad_c_k(h, k, n)
        
        # Backtracking line search
        t = lr
        for bt in range(30):
            h_new = project_feasible(h - t * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c or c_new < c_max - 1e-12:
                h = h_new
                break
            t *= 0.5
            if t < 1e-10:
                break
        else:
            # No improvement possible with this gradient
            break
        
        # Diminishing step size
        lr *= 0.9995
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_optimize(h, n, deadline, max_passes=10000, rng=None):
    dx = 2.0 / n
    h = project_feasible(h, n)
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = c.max()
    shifts = np.arange(2 * n - 1) - (n - 1)
    pad_len = n - 1
    hp = np.concatenate([np.zeros(pad_len), h, np.zeros(pad_len)])
    rng_local = rng if rng is not None else np.random.RandomState(0)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        if _pass % 5 == 4:
            rng_local.shuffle(order)
        for ii in range(n):
            i = order[ii]
            if time.time() > deadline:
                return h, best_c
            base_i = i + pad_len
            for j in range(i + 1, n):
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                base_j = j + pad_len
                A = dx * (hp[base_j - shifts] + hp[base_j + shifts] - hp[base_i - shifts] - hp[base_i + shifts])
                a, b = d_lo, d_hi
                for _ in range(22):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    f1 = np.max(c + m1 * A)
                    f2 = np.max(c + m2 * A)
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                d_opt = (a + b) / 2.0
                f_opt = np.max(c + d_opt * A)
                if f_opt < best_c - 1e-14:
                    h[i] += d_opt
                    h[j] -= d_opt
                    c += d_opt * A
                    best_c = f_opt
                    hp[base_i] += d_opt

                    improved = True

                    return h, best_c

def sinusoidal_perturb(h, n, rng, scale=0.01):
    h_p = h.copy()
    x = np.arange(n)
    for p in rng.choice([2, 3, 4, 5, 7, 11], size=3, replace=False):
        h_p += scale * rng.uniform(0.5, 1.0) * np.sin(2.0 * np.pi * p * x / n + rng.uniform(0, 2 * np.pi))
    return project_feasible(h_p, n)

def optimize_at_resolution(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    alphas = [500, 2000, 10000, 50000]
    t_check = time.time()
    for a in alphas:
        if time.time() > deadline - 30:
            break
        ht = slsqp_step(best_h, n, a, maxiter=2500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    if time.time() < deadline - 10:
        best_h, best_c = pair_optimize(best_h, n, deadline - 5, rng=rng)

    if time.time() < deadline - 15:
        ht, ct = subgradient_polish(best_h, n, deadline - 10)
        if ct < best_c:
            best_c = ct
            best_h = ht

    if time.time() < deadline - 3:
        best_h, best_c = pair_optimize(best_h, n, deadline - 1, max_passes=5000, rng=rng)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250

    if h_init is None:
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    alloc1 = min(120, (deadline - t0) * 0.2)
    h_opt, c_opt = optimize_at_resolution(best_h, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    remaining = deadline - time.time() - 20
    alloc2 = remaining * 0.45
    h_opt, c_opt = pair_optimize(best_h, n_init, time.time() + alloc2, rng=rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    n_try = 400
    remaining = deadline - time.time() - 40
    if remaining > 120:
        alloc3 = min(180, remaining * 0.45)
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc3, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_final = n_try
        else:
            x_new2 = np.linspace(0, 2, n_init)
            h_interp2 = np.interp(x_new2, x_new, h_opt)
            h_interp2 = project_feasible(h_interp2, n_init)
            c2 = compute_c5(h_interp2, n_init)
            if c2 < best_c:
                best_c = c2
                best_h = h_interp2
            n_final = n_init

    remaining = deadline - time.time() - 10
    if remaining > 30:
        restarts = max(2, int(remaining / 60))
        for r in range(restarts):
            if time.time() > deadline - 15:
                break
            n_cur = len(best_h)
            hp = sinusoidal_perturb(best_h, n_cur, rng, scale=0.008 * (1 + r * 0.3))
            alloc = min(45, (deadline - time.time() - 15) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = pair_optimize(hp, n_cur, time.time() + alloc, max_passes=3000, rng=rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if time.time() < deadline - 5:
        n_cur = len(best_h)
        best_h, best_c = pair_optimize(best_h, n_cur, deadline - 2, max_passes=3000, rng=rng)
        best_h = project_feasible(best_h, n_cur)
        best_c = compute_c5(best_h, n_cur)

    return best_h, best_c, len(best_h)
