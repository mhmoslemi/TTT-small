def compute_smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1 - x, mode='full')
    c = c_raw * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx_s = shifted.max()
    if mx_s > 500:
        e = np.exp(shifted - mx_s)
        log_sum = mx_s + np.log(e.sum())
    else:
        e = np.exp(shifted)
        log_sum = np.log(e.sum())
    p = e / e.sum()  # softmax weights
    obj = m + log_sum / alpha
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        lag = k - (n - 1)
        m_lag = abs(lag)
        if m_lag == 0:
            grad += pk * dx * (1 - 2*x)
        else:

            # grad[j] += pk * dx * [(1-x[j+m_lag]) * I(j < n-m_lag) - x[j-m_lag] * I(j >= m_lag)]
            if n - m_lag > 0:

            if m_lag < n:
                grad[m_lag:] -= pk * dx * x[:n-m_lag]
    return obj, grad

        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-16:
            h = h_try
            best_c = c_try
            stall = 0
            step = min(step * 1.02, 0.01)
        else:
            stall += 1
            step *= 0.85
            if step < 1e-7:
                break
            if stall > 50:
                break
    return h, best_c

def pair_local_search(h, n, deadline, num_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for rnd in range(num_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        m_lag = abs(lag)
        if m_lag == 0 or m_lag >= n:
            break
        # Find worst pair for this lag
        # c_lag = dx * sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        # To reduce: make h[i] close to h[i+m]
        pairs = h[:n - m_lag]  # h[i]
        pairs2 = h[m_lag:]     # h[i+m]
        # Contribution to c_lag: pairs*(1-pairs2)
        contrib = pairs * (1.0 - pairs2)
        # Also check negative contribution: (1-pairs)*pairs2 would reduce c_lag if we increase it
        # Actually we want to REDUCE c_lag = sum h[i]*(1-h[i+m])
        # Best: reduce the largest terms
        idx_sorted = np.argsort(contrib)[-10:]
        for idx in idx_sorted:
            if time.time() > deadline:
                break
            i = idx
            j = idx + m_lag
            if h[i] > h[j]:
                delta = min(0.002, (h[i] - h[j]) * 0.3)
                if delta < 1e-10:
                    continue
                h[i] -= delta
                h[j] += delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                else:
                    h[i] += delta
                    h[j] -= delta
                    h = project_feasible(h, n)
            elif h[j] > h[i]:
                delta = min(0.002, (h[j] - h[i]) * 0.3)
                if delta < 1e-10:
                    continue
                h[i] += delta
                h[j] -= delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                else:
                    h[i] -= delta
                    h[j] += delta
                    h = project_feasible(h, n)
    return h, best_c

def optimize_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # SLSQP pass 1
    if time.time() < deadline - 30:
        h_new = slsqp_optimize(best_h, n, 5000, maxiter=2000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # SLSQP pass 2
    if time.time() < deadline - 30:
        h_new = slsqp_optimize(best_h, n, 20000, maxiter=2000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Gradient local search
    if time.time() < deadline - 15:
        h_new, c_new = gradient_local_search(best_h, n, time.time() + min(60, (deadline - time.time()) * 0.5), num_top=5)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Pair local search
    if time.time() < deadline - 5:
        h_new, c_new = pair_local_search(best_h, n, deadline - 1, num_rounds=300)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def _make_fourier_init(n, freq, amp):
    x = np.arange(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    best_h = None
    best_c = float('inf')

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 300

    # Phase 1: Optimize parent at its resolution
    if h_parent is not None:
        alloc1 = min(120, (deadline - t0) * 0.25)
        h1, c1 = optimize_n(h_parent, n_parent, t0 + alloc1)
        best_h, best_c = h1, c1

    # Phase 2: Try n=400 by interpolation
    n_try = 400
    if time.time() < deadline - 150:
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        alloc2 = min(150, (deadline - time.time()) * 0.3)
        h2, c2 = optimize_n(h_start, n_try, time.time() + alloc2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Try n=500
    n_try3 = 500
    if time.time() < deadline - 150:
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try3)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.full(n_try3, 0.5)
        h_start = project_feasible(h_start, n_try3)
        alloc3 = min(150, (deadline - time.time()) * 0.3)
        h3, c3 = optimize_n(h_start, n_try3, time.time() + alloc3)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Random restarts on best_n
    if best_h is not None and time.time() < deadline - 80:
        n_best = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc_r = min(50, (deadline - time.time() - 30) * 0.5)
            if alloc_r < 20:
                break
            hr, cr = optimize_n(hp, n_best, time.time() + alloc_r)
            if cr < best_c:
                best_c = cr
                best_h = hr

    # Phase 5: Final intensive polish on best
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        # Final SLSQP
        if time.time() < deadline - 20:
            h_f = slsqp_optimize(best_h, n_best, 100000, maxiter=3000, ftol=1e-18)
            c_f = compute_c5(h_f, n_best)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
        # Final gradient LS
        if time.time() < deadline - 5:
            h_f, c_f = gradient_local_search(best_h, n_best, deadline - 2, num_top=7, alpha_ls=1000)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
        # Final pair LS
        if time.time() < deadline - 3:
            h_f, c_f = pair_local_search(best_h, n_best, deadline - 1, num_rounds=500)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f

    if best_h is None:
        best_h = project_feasible(np.full(n_parent, 0.5), n_parent)
        best_c = compute_c5(best_h, n_parent)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
