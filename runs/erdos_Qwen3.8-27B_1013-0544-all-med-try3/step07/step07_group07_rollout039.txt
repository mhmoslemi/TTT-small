import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - hp
            s = room.sum()
            if s > 1e-15:
                hp += (diff / s) * room
        else:
            s = hp.sum()
            if s > 1e-15:
                hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    a = alpha * c
    sm = a.max()
    exp_c = np.exp(a - sm)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-16:
            continue
        i_min = max(0, k)
        i_max = min(n - 1, n - 1 + k)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += pk * (1.0 - h[i_min - k:i_max - k + 1])
        i_min2 = max(0, -k)
        i_max2 = min(n - 1, n - 1 - k)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= pk * h[i_min2 + k:i_max2 + k + 1]
    return grad

def pgd_solve(h, n, alphas, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_work = best_h.copy()
        cur_c = best_c
        step = min(float(n) / (4.0 * alpha), 0.3)
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            grad = lse_grad(h_work, n, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            direction = -grad / gnorm
            accepted = False
            for ls in range(50):
                h_new = project(h_work + step * (0.5 ** ls) * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    h_work = h_new
                    cur_c = c_new
                    step = max(step * (0.5 ** ls), 1e-6)
                    accepted = True
                    break
            if accepted:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_work.copy()
            else:
                step = max(step * 0.5, 1e-8)
                if step < 1e-10:
                    break
        c_final = compute_c5(h_work, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_work.copy()
    return best_h, best_c

def binary_local_search(hb, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h = hb.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    m_range = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_list):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def anneal_binary(hb, n, deadline, seed=42, T_start=0.015, T_end=0.001):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = hb.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n - 1), n)
    n_iters = 5000000
    total_time = max(deadline - time.time(), 0.1)
    n_accepted = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 50000 == 0 and it > 0:
            frac = it / n_iters
            T = T_start * (T_end / T_start) ** frac
        else:
            T = T_start * (T_end / T_start) ** (it / n_iters)
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        nc = float((corr + delta).max() * dx)
        delta_c = nc - cur_c
        if delta_c <= 0 or (T > 1e-10 and rng.random() < np.exp(-delta_c / (T * dx))):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            n_accepted += 1
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def double_swap_search(hb, n, deadline, max_rounds=200):

                continue
            j1 = int(ones_arr[rng.randint(len(ones_arr))])
            j2 = int(ones_arr[rng.randint(len(ones_arr))])
            if j1 == j2:
                continue
            delta = np.zeros(2 * n - 1)
            for i in [i1, i2]:
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
            for j in [j1, j2]:
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 4.0
            k1 = n - 1 + (j1 - i1)
            if 0 <= k1 < 2 * n - 1:
                delta[k1] += 1.0
            k2 = n - 1 + (i1 - j1)
            if 0 <= k2 < 2 * n - 1:
                delta[k2] += 1.0
            k3 = n - 1 + (j2 - i2)
            if 0 <= k3 < 2 * n - 1:
                delta[k3] += 1.0
            k4 = n - 1 + (i2 - j2)
            if 0 <= k4 < 2 * n - 1:
                delta[k4] += 1.0
            k5 = n - 1 + (j1 - i2)
            if 0 <= k5 < 2 * n - 1:
                delta[k5] += 1.0
            k6 = n - 1 + (i2 - j1)
            if 0 <= k6 < 2 * n - 1:
                delta[k6] += 1.0
            k7 = n - 1 + (j2 - i1)
            if 0 <= k7 < 2 * n - 1:
                delta[k7] += 1.0
            k8 = n - 1 + (i1 - j2)
            if 0 <= k8 < 2 * n - 1:
                delta[k8] += 1.0
            nc = float((corr + delta).max() * dx)
            if nc < cur_c - 1e-15:
                h[i1], h[j1] = 1.0, 0.0
                h[i2], h[j2] = 1.0, 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = float(corr.max() * dx)
                np.delete(ones_arr, [np.where(ones_arr == j1)[0][0], np.where(ones_arr == j2)[0][0]])
                np.append(ones_arr, [i1, i2])
                np.delete(zeros_arr, [np.where(zeros_arr == i1)[0][0], np.where(zeros_arr == i2)[0][0]])
                np.append(zeros_arr, [j1, j2])
                ones_arr = np.where(h == 1)[0].copy()
                zeros_arr = np.where(h == 0)[0].copy()
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c
