    # Phase 4: Subgradient polish
    if time.time() < deadline - 10:
        sg_end = time.time() + min(30, (deadline - time.time()) * 0.3)
        h_sg, c_sg = subgradient_polish(best_h, n, sg_end)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    
    # Phase 5: Final SLSQP + local search
    if time.time() < deadline - 5:
        h_new = slsqp_optimize(best_h, n, 200000, 4000, deadline - 2)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 2:
            h_ls2, c_ls2 = intensive_local_search(best_h, n, deadline - 1)
            if c_ls2 < best_c:
                best_c = c_ls2
                best_h = h_ls2
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Start from parent if available
    if h_init is not None:
        n_parent = n_init
        alloc1 = min(350, (deadline - t0) * 0.38)
        h_opt, c_opt = optimize_at_n(h_init, n_parent, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt
    
    # Try other resolutions
    n_candidates = [400, 500, 200, 600]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 80:
            break
        alloc = min(150, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 30) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final intensive polishing
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 500000, 5000, deadline - 3)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 2:
            best_h, best_c = intensive_local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
