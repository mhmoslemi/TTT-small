def fft_correlate(a, b):
    """Compute np.correlate(a, b, mode='full') using FFT."""
    n = len(a) + len(b) - 1
    fa = np.fft.rfft(a, n)
    fb = np.fft.rfft(b, n)
    result = np.fft.irfft(fa * fb.conjugate(), n)[:len(a)+len(b)-1]
    return result

    target = n / 2.0

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()

    # ∂c[m]/∂h[i] = dx * [(1-h[i+k]) * 1_{valid} - h[i-k] * 1_{valid}]

        if k >= 0:

            # (1-h[i+k]) for i=0..nv-1
            grad[0:nv] += wm * (1.0 - h[k:n])
            # -h[i-k] for i=k..n-1, i.e., i-k = 0..nv-1
            grad[k:n] += wm * (-h[0:nv])
        else:

            nv = n - kk
            # (1-h[i+k]) = (1-h[i-kk]) for i=kk..n-1
            grad[kk:n] += wm * (1.0 - h[0:nv])
            # -h[i-k] = -h[i+kk] for i=0..nv-1
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def compute_delta_c(h, n, i, j, delta):
    """Compute the change in each c[m] when h[i]+=delta, h[j]-=delta."""
    dx = 2.0 / n
    delta_c = np.zeros(2*n-1)
    
    # Effect on c[m] from changing h[i] by +delta:
    # c[m] includes h[x]*(1-h[x+k]) for valid x
    # h[i] appears as first factor when i+k is valid: contribution +delta*(1-h[i+k])
    # h[i] appears as second factor when i-k is valid: contribution -delta*h[i-k]
    
    for m in range(2*n-1):
        k = m - (n-1)
        # h[i] as first factor: need 0 <= i < n and 0 <= i+k < n
        contrib = 0.0
        if 0 <= i < n and 0 <= i+k < n:
            contrib += delta * (1.0 - h[i+k])
        # h[i] as second factor: need 0 <= i-k < n and 0 <= i < n
        if 0 <= i-k < n and 0 <= i < n:
            contrib += -delta * h[i-k]
        
        # h[j] as first factor: -delta
        if 0 <= j < n and 0 <= j+k < n:
            contrib += -delta * (1.0 - h[j+k])
        # h[j] as second factor: +delta
        if 0 <= j-k < n and 0 <= j < n:
            contrib += delta * h[j-k]
        
        delta_c[m] = dx * contrib
    
    return delta_c

def precompute_influence(h, n):
    """Precompute influence vectors v[i] where v[i][m] = ∂c[m]/∂h[i] / dx."""
    dx = 2.0 / n
    k = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    v = np.zeros((n, 2*n-1))
    for i in range(n):
        # (1-h[i+k]) for valid i+k
        ik = i + k
        mask1 = (ik >= 0) & (ik < n)
        v[i, mask1] = 1.0 - h[ik[mask1]]
        # -h[i-k] for valid i-k
        ik2 = i - k
        mask2 = (ik2 >= 0) & (ik2 < n)
        v[i, mask2] -= h[ik2[mask2]]
    return v

        t4 = min(time.time() + 0.08 * budget_s, deadline - 0.25 * budget_s)
        h, c = gradient_pair_search(h, n, t4, step_size=0.001)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    h = best_h.copy()
    
    # Phase 5: Fine refinement with very small steps (~20% budget)
    t5 = t0 + 0.82 * budget_s
    h, best_c = refine_with_unequal_steps(h, n, t5)
    best_h = h.copy()
    
    # Phase 6: Try n=360 resolution (~10% budget)
    if time.time() < deadline - 60:
        n2 = 360
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_feasible(h2, n2)
        alloc = min(40, deadline - time.time() - 20)
        if alloc > 10:
            h2, c2 = lbfgs_optimize(h2, n2, time.time() + alloc * 0.5, max_iters=80)
            if time.time() < deadline - 15:
                h2, c2 = fast_pair_search(h2, n2, min(time.time() + 15, deadline - 10),
                                           [0.005, 0.002, 0.001, 0.0005], max_sweeps=10)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                n = n2
    
    # Phase 7: Final ultra-fine search (~7% budget)
    if time.time() < deadline - 3:
        nf = len(best_h)
        h_final, c_final = refine_with_unequal_steps(best_h, nf, deadline - 1)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
