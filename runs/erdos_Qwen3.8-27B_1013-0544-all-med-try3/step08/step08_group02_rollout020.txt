    c_shift: the raw correlation (before multiplying by dx), index k corresponds to shift k-(n-1)
    p: position to remove 1 from
    q: position to add 1 to

    """

    # h[p] changes 1→0:
    #   as first factor: shift k where p+k valid → k in [-p, n-1-p]
    #   as second factor: shift k where p-k valid → k in [p-(n-1), p]
    # h[q] changes 0→1:
    #   as first factor: shift k where q+k valid → k in [-q, n-1-q]
    #   as second factor: shift k where q-k valid → k in [q-(n-1), q]

            mask1 = (ks >= -p) & (ks <= n-1-p)
            if mask1.any():
                idx1 = ks[mask1] + p
                delta[mask1] -= (1 - h[idx1])
            
            mask2 = (ks >= -q) & (ks <= n-1-q)
            if mask2.any():
                idx2 = ks[mask2] + q
                delta[mask2] += (1 - h[idx2])
            
            mask3 = (ks >= p-n+1) & (ks <= p)
            if mask3.any():
                idx3 = p - ks[mask3]
                delta[mask3] += h[idx3]
            
            mask4 = (ks >= q-n+1) & (ks <= q)
            if mask4.any():
                idx4 = q - ks[mask4]
                delta[mask4] -= h[idx4]
            
            # Convert to corr index range
            ci_min = (n-1) - k_max
            ci_max = (n-1) - k_min
            ci_min = max(0, ci_min)
            ci_max = min(n_corr-1, ci_max)
            
            if ci_min <= ci_max:
                # Map delta to corr indices
                # corr index = (n-1) - k, so k = (n-1) - corr_index
                # k = k_min + i → corr_index = (n-1) - k_min - i
                # So corr[ci_max : ci_min-1 : -1] corresponds to delta[0], delta[1], ...
                # Or: corr[(n-1)-k_min - i] += delta[i]
                # Let's just compute it directly:
                corr_new = corr[ci_min:ci_max+1].copy()
                # k range for this corr slice: k from (n-1)-ci_max to (n-1)-ci_min
                k_start = (n-1) - ci_max
                # delta[i] corresponds to k = k_min + i
                # corr[ci_min+j] corresponds to k = (n-1) - (ci_min+j)
                # We need: (n-1) - (ci_min+j) = k_min + i → i = (n-1) - ci_min - j - k_min
                # Since ci_min = (n-1) - k_max (possibly clamped), let's just iterate:
                for j in range(ci_max - ci_min + 1):
                    k_val = (n-1) - (ci_min + j)
                    i = k_val - k_min
                    if 0 <= i < len(delta):
                        corr_new[j] = corr[ci_min+j] + delta[i]
                
                new_max_affected = corr_new.max()
                # Max outside affected range
                if ci_min > 0:
                    max_outside = corr[:ci_min].max()
                else:
                    max_outside = 0
                if ci_max < n_corr-1:
                    if max_outside == 0:
                        max_outside = corr[ci_max+1:].max()
                    else:
                        max_outside = max(max_outside, corr[ci_max+1:].max())
                
                new_max = max(new_max_affected, max_outside)
                
                # Accept/reject (simulated annealing)
                if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
                    # Apply swap
                    h[p] = 0
                    h[q] = 1
                    ones_set.remove(p)
                    ones_set.add(q)
                    zeros_list[zeros_list.index(p)] = q  # Update list
                    corr[ci_min:ci_max+1] = corr_new
                    cur_max = new_max
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
                
                T = max(T * cooling, T_min)
                
                if stall > 5000:
                    T = max(T * 1.5, 0.01)  # Reheat
                    stall = 0
    
    return project_feasible(best_h, n)
