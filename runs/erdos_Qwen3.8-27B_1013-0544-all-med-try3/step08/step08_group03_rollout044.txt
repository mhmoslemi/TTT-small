    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    for it in range(50000):
        if time.time() > deadline:
            break
        y = h + v
        g = smooth_max_grad(y, n, alpha)
        # Projected gradient step with Nesterov
        v_new = 0.9 * v - lr * g
        h_new = project_feasible(y + v_new, n)
        c_new = compute_c5(h_new, n)
        if c_new <= best_c + 1e-13:
            v = h_new - y
            h = h_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
        else:
            # Reduce step
            v = 0.9 * v - (lr / 2) * g
            h_new = project_feasible(y + v, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-13:
                h = h_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
            else:
                lr *= 0.5
                v = np.zeros(n)
            if lr < 1e-10:
                lr = 0.001
                v = np.zeros(n)
    return best_h, best_c
