```python
        w = np.exp(sh - sh_max)
        w /= w.sum()
        shift = sh_max / alpha
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in range(n):
            if diff > 0:
                add = min(diff, 1.0 - h[idx[i]])
                h[idx[i]] += add
                diff -= add
            else:
                rem = min(-diff, h[idx[i]])
                h[idx[i]] -= rem
                diff += rem
            if abs(diff) < 1e-14:
                break
    return h

def smooth_max_c5(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    if sh_max > 500:
        return float(m + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    if sh_max > 500:
        w = np.exp(sh - sh_max)
    else:
        w = np.exp(sh)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-15:
            continue
        lag = m_idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wm * dx * (1 - x[lag:])
            grad[lag:] -= wm * dx * x[:n - lag]
        else:
            l = -lag
            grad[l:] += wm * dx * (1 - x[:n - l])
            grad[:n - l] -= wm * dx * x[l:]
    return grad

def obj_penalty(x, n, alpha, mu):
    return smooth_max_c5(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def grad_penalty(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

def lbfgs_optimize(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_penalty, h, jac=grad_penalty, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxls': 50})
        h_new = project_feasible(res.x, n)
        if time.time() > deadline:
            return h
        return h_new
    except Exception:
        return h

def coord_descent(h, n, deadline, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    target = n / 2.0
    for d in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                # Try increasing h[i] and decreasing h[j]
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def near_coord_descent(h, n, deadline, steps, radius=50):
    best_c = compute_c5(h, n)
    for d in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_push(h, n):
    """Threshold and re-feasibilize"""
    h_bin = (h > 0.5).astype(float)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    cnt = h_bin.sum()
    if cnt > target:
        # Need to flip some 1s to 0s
        # Pick the ones where h was closest to 0.5 (most "uncertain")
        ones = np.where(h_bin == 1)[0]
        ones = ones[np.argsort(h[ones])]
        for idx in ones[:int(cnt - target)]:
            h_bin[idx] = 0
    elif cnt < target:
        # Need to flip some 0s to 1s
        zeros = np.where(h_bin == 0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        for idx in zeros[:int(target - cnt)]:
            h_bin[idx] = 1
    return h_bin

def optimize_from(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    # Multiple α and μ values
    alphas = [100, 300, 1000, 3000, 10000]
    mus = [1000, 5000, 20000]
    
    for a in alphas:
        if time.time() > deadline - 15:
            break
        for mu in mus[:2]:
            if time.time() > deadline - 10:
                break
            if n <= 200:
                maxiter = 300
            else:
                maxiter = 200
            ht = lbfgs_optimize(h, n, a, mu, maxiter, deadline)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                h = ht
    
    # Binary push
    if time.time() < deadline - 20:
        h_bp = binary_push(h, n)
        c_bp = compute_c5(h_bp, n)
        if c_bp < best_c:
            best_c = c_bp
            h = h_bp
        # Optimize from binary push
        for a in [1000, 5000]:
            if time.time() > deadline - 15:
                break
            ht = lbfgs_optimize(h, n, a, 20000, 150, deadline)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                h = ht
    
    # Coordinate descent
    if n <= 200:
        if time.time() < deadline - 5:
            h, best_c = coord_descent(h, n, deadline - 5)
    else:
        if time.time() < deadline - 5:
            h, best_c = near_coord_descent(h, n, deadline - 5,
                                           [0.01, 0.005, 0.002, 0.001, 0.0005],
                                           radius=60)
    
    # Final SLSQP-like polish (small perturbation search)
    if time.time() < deadline - 3:
        for a in [50000]:
            ht = lbfgs_optimize(h, n, a, 50000, 200, deadline - 2)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                h = ht
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in range(n):
            if diff > 0:
                add = min(diff, 1.0 - h[idx[i]])
                h[idx[i]] += add
                diff -= add
            else:
                rem = min(-diff, h[idx[i]])
                h[idx[i]] -= rem
                diff += rem
            if abs(diff) < 1e-14:
                break
    return h

    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    grad = np.zeros(n)
    if indices is not None:
        idx_list = indices
    else:
        idx_list = range(2 * n - 1)
    for k_i, m_idx in enumerate(idx_list):
        wm = w[k_i]
        if wm < 1e-15:
            continue
        lag = m_idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wm * dx * (1.0 - x[lag:])
            grad[lag:] -= wm * dx * x[:n - lag]
        else:
            l = -lag
            grad[l:] += wm * dx * (1.0 - x[:n - l])
            grad[:n - l] -= wm * dx * x[l:]
    return grad

def _obj_pen(x, n, alpha, mu, indices=None):
    return _smooth_max_val(x, n, alpha, indices) + mu * (x.sum() - n / 2.0) ** 2

def _grad_pen(x, n, alpha, mu, indices=None):
    g = _smooth_max_grad(x, n, alpha, indices)
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

def lbfgs_call(h, n, alpha, mu, maxiter, deadline, indices=None):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_pen, h, jac=_grad_pen,
                       args=(n, alpha, mu, indices),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 40})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def find_top_k(h, n, k=10):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    return np.argsort(c)[::-1][:k]

def binary_push(h, n):
    h_bin = (h > 0.5).astype(float)
    target = n / 2.0
    cnt = h_bin.sum()
    if cnt > target:
        ones = np.where(h_bin == 1)[0]
        ones = ones[np.argsort(h[ones])]
        for idx in ones[:int(cnt - target)]:
            h_bin[idx] = 0
    elif cnt < target:
        zeros = np.where(h_bin == 0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        for idx in zeros[:int(target - cnt)]:
            h_bin[idx] = 1
    return h_bin

def coord_descent(h, n, deadline, deltas, radius=0):
    best_c = compute_c5(h, n)
    target = n / 2.0
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if radius == 0:
                    j_lo, j_hi = 0, n
                else:
                    j_lo = max(0, i - radius)
                    j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def grad_guided_cd(h, n, deadline, deltas, top_count=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            g = _smooth_max_grad(h, n, 500)
            g_sorted_neg = np.argsort(g)[:top_count]
            g_sorted_pos = np.argsort(g)[::-1][:top_count]
            for i in g_sorted_neg:
                if time.time() > deadline:
                    break
                for j in g_sorted_pos:
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    # Phase A: L-BFGS-B with increasing alpha
    alpha_mu_list = [(100, 5000), (300, 5000), (1000, 10000), (3000, 10000), (10000, 20000)]
    for a, mu in alpha_mu_list:
        if time.time() > deadline - 20:
            break
        maxiter = 200 if n <= 200 else 150
        ht = lbfgs_call(h, n, a, mu, maxiter, deadline)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    # Phase B: Top-K adaptive refinement
    if time.time() < deadline - 30:
        for k in [5, 10, 20]:
            if time.time() > deadline - 20:
                break
            top_k = find_top_k(h, n, k)
            for a in [500, 2000, 8000]:
                if time.time() > deadline - 15:
                    break
                ht = lbfgs_call(h, n, a, 10000, 100, deadline, indices=top_k)
                c = compute_c5(ht, n)
                if c < best_c - 1e-14:
                    best_c = c
                    h = ht
                    top_k = find_top_k(h, n, k)

    # Phase C: Binary push + re-optimize
    if time.time() < deadline - 30:
        h_bp = binary_push(h, n)
        c_bp = compute_c5(h_bp, n)
        if c_bp < best_c:
            best_c = c_bp
            h = h_bp
        for a, mu in [(500, 10000), (2000, 20000)]:
            if time.time() > deadline - 20:
                break
            ht = lbfgs_call(h, n, a, mu, 100, deadline)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                h = ht

    # Phase D: Gradient-guided coordinate descent
    if n <= 250:
        if time.time() < deadline - 10:
            h, best_c = grad_guided_cd(h, n, deadline - 5,
                                       [0.01, 0.005, 0.002, 0.001, 0.0005],
                                       top_count=40)
    else:
        if time.time() < deadline - 10:
            h, best_c = coord_descent(h, n, deadline - 5,
                                      [0.01, 0.005, 0.002, 0.001, 0.0005],
                                      radius=80)

    # Phase E: Final L-BFGS-B polish
    if time.time() < deadline - 5:
        ht = lbfgs_call(h, n, 20000, 50000, 150, deadline - 2)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial construction (largest allocation)
    alloc1 = min(250, (deadline - t0) * 0.25)
    h_opt, c_opt = optimize_full(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try different resolutions
    x_init = np.linspace(0, 2, n_init)
    for n_try in [200, 150, 300]:
        if time.time() > deadline - 120:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, best_h if best_h is not None else h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(120, (deadline - time.time()) * 0.15)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_full(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart from perturbed best
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.02 + 0.08 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 20) * 0.3)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_full(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Extended coordinate descent on best solution
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        if n <= 250:
            best_h, best_c = coord_descent(best_h, n, deadline - 5,
                                           [0.005, 0.002, 0.001, 0.0005,
                                            0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = coord_descent(best_h, n, deadline - 5,
                                           [0.005, 0.002, 0.001, 0.0005,
                                            0.0002, 0.0001],
                                           radius=60)

    # Phase 5: Final L-BFGS-B + tiny coordinate descent
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        for a in [30000, 80000]:
            if time.time() > deadline - 4:
                break
            ht = lbfgs_call(best_h, n, a, 50000, 200, deadline - 3)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
        if time.time() < deadline - 2:
            if n <= 250:
                best_h, best_c = coord_descent(best_h, n, deadline - 1,
                                               [0.0005, 0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = coord_descent(best_h, n, deadline - 1,
                                               [0.0005, 0.0002, 0.0001],
                                               radius=40)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
