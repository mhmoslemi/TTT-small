            best_c = c_trial
            best_h = h_trial.copy()
            h_curr = h_trial
        else:
            step *= 0.8
        if step < 1e-10:
            step = 0.005
            mom *= 0.5
    return best_h, best_c

def pair_search_1d(h, n, deadline, radius=40, n_grid=30):
    best_c = compute_c5(h, n)
    deltas_coarse = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for d_coarse in deltas_coarse:
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if j <= i:
                    continue
                for sign in [1, -1]:
                    d = sign * d_coarse
                    if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                        continue
                    nh = h.copy()
                    nh[i] += d
                    nh[j] -= d
                    c = compute_c5(nh, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        h = nh
                        improved_any = True
                    elif not improved_any:
                        pass
        if not improved_any:
            break
    return h, best_c

def pair_search_fine(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    step_sizes = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for ss in step_sizes:
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if j <= i:
                    continue
                for sign in [1, -1]:
                    d = sign * ss
                    if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                        continue
                    nh = h.copy()
                    nh[i] += d
                    nh[j] -= d
                    c = compute_c5(nh, n)
                    if c < best_c - 1e-15:

                        h = nh
                        improved_any = True
        if not improved_any:
            break
    return h, best_c

def coordinate_search(h, n, deadline):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for ss in steps:
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for sign in [1, -1]:
                d = sign * ss
                if h[i] + d < 0 or h[i] + d > 1:
                    continue
                # Distribute -d to the most "room" coordinate
                nh = h.copy()
                nh[i] += d
                # Find j with most room (if d>0, need to subtract from j)
                if d > 0:
                    j = int(np.argmin(nh[1:])) + 1 if nh[1:].min() < nh[0] else 0
                    if j == i:
                        j = (i + 1) % n
                    if nh[j] - d < 0:
                        continue
                    nh[j] -= d
                else:
                    j = int(np.argmax(nh[1:])) + 1 if nh[1:].max() > nh[0] else 0
                    if j == i:
                        j = (i + 1) % n
                    if nh[j] - d > 1:
                        continue
                    nh[j] -= d
                nh = project_feasible(nh, n)
                c = compute_c5(nh, n)
                if c < best_c - 1e-14:
                    best_c = c
                    h = nh
                    improved = True
        if not improved:
            break
    return h, best_c

def try_resolution(h_src, n_src, n_target, budget_s):
    if n_target == n_src:
        return h_src.copy(), compute_c5(h_src, n_src)
    x_src = np.linspace(0, 2, n_src)
    x_tgt = np.linspace(0, 2, n_target)
    h_interp = np.interp(x_tgt, x_src, h_src)
    h_interp = project_feasible(h_interp, n_target)
    c = compute_c5(h_interp, n_target)
    # Quick SLSQP
    if time.time() < t_global_deadline - 5:
        ht = slsqp_optimize(h_interp, n_target, 10000, maxiter=1000)
        ct = compute_c5(ht, n_target)
        if ct < c:
            c = ct
            h_interp = ht
    # Subgradient
    if time.time() < t_global_deadline - 5:
        ht, ct = subgrad_descent(h_interp, n_target, min(30, t_global_deadline - time.time() - 5))
        if ct < c:
            c = ct
            h_interp = ht
    return h_interp, c

def run(seed=42, budget_s=1100, **kwargs):
    global t_global_deadline
    t0 = time.time()
    t_global_deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    h = project_feasible(h_init, n_init)
    best_h = h.copy()
    best_c = compute_c5(h, n_init)
    n_best = n_init

    # Phase 1: SLSQP polish on initial (30s)
    alloc = min(40, t_global_deadline - t0 - 5)
    if alloc > 10:
        ht = slsqp_optimize(h, n_init, 20000, maxiter=2000)
        ct = compute_c5(ht, n_init)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            h = ht

    # Phase 2: Subgradient descent (60s)
    alloc2 = min(60, t_global_deadline - time.time() - 5)
    if alloc2 > 10:
        ht, ct = subgrad_descent(h, n_init, alloc2)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            h = ht

    # Phase 3: Smooth subgradient (40s)
    alloc3 = min(40, t_global_deadline - time.time() - 5)
    if alloc3 > 10:
        ht, ct = smooth_subgrad_descent(h, n_init, alloc3, alpha=5000)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            h = ht

    # Phase 4: Coarse pair search (80s)
    alloc4 = min(80, t_global_deadline - time.time() - 5)
    if alloc4 > 10:
        h, best_c = pair_search_1d(h, n_init, time.time() + alloc4, radius=50)
        best_h = h.copy()

    # Phase 5: SLSQP again (30s)
    alloc5 = min(35, t_global_deadline - time.time() - 5)
    if alloc5 > 10:
        ht = slsqp_optimize(h, n_init, 50000, maxiter=1500)
        ct = compute_c5(ht, n_init)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            h = ht

    # Phase 6: Fine pair search (100s)
    alloc6 = min(100, t_global_deadline - time.time() - 5)
    if alloc6 > 10:
        h, best_c = pair_search_fine(h, n_init, time.time() + alloc6, radius=40)
        best_h = h.copy()

    # Phase 7: Try other resolutions (60s each, up to 2)
    for n_try in [200, 300]:
        if time.time() > t_global_deadline - 60:
            break
        alloc = min(55, t_global_deadline - time.time() - 10)
        if alloc < 20:
            break
        ht, ct = try_resolution(best_h, n_best, n_try, alloc)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            n_best = n_try

    # Phase 8: Final subgradient on best (30s)
    if time.time() < t_global_deadline - 40:
        n_cur = len(best_h)
        ht, ct = subgrad_descent(best_h, n_cur, min(30, t_global_deadline - time.time() - 35))
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            n_best = n_cur

    # Phase 9: Final fine pair search (40s)
    if time.time() < t_global_deadline - 30:
        n_cur = len(best_h)
        h_fin, c_fin = pair_search_fine(best_h, n_cur, t_global_deadline - 25, radius=30)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()

    # Phase 10: Final SLSQP + pair (25s)
    if time.time() < t_global_deadline - 15:
        n_cur = len(best_h)
        ht = slsqp_optimize(best_h, n_cur, 100000, maxiter=2000)
        ct = compute_c5(ht, n_cur)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
        if time.time() < t_global_deadline - 5:
            h_fin, c_fin = pair_search_fine(best_h, n_cur, t_global_deadline - 3, radius=25)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin.copy()

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    # Verify with evaluate function if available
    try:
        verified_c = evaluate_erdos_solution(best_h)
        if verified_c is not None and verified_c < best_c:
            best_c = float(verified_c)
    except Exception:
        pass

    return best_h, best_c, n_final
