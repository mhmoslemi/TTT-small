    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

    return grad

def lp_polish(h, n, deadline, n_rounds=20, top_k=10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds_h = [(0.0, 1.0)] * n
    target = n / 2.0
    
    for rnd in range(n_rounds):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        # Build LP: min t s.t. c_k(h*) + grad_c_k^T (h - h*) <= t
        # Variables: [h_0, ..., h_{n-1}, t]
        # Objective: minimize t -> c_obj = [0,...,0, 1]
        c_obj = np.zeros(n + 1)

        A_ub = []
        b_ub = []

            # ck_val + gk^T(h - h*) <= t  =>  gk^T h - t <= ck_val - gk^T h*
            row = np.zeros(n + 1)
            row[:n] = gk
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(ck_val - gk @ best_h)
        
        if len(A_ub) == 0:
            break
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lp = bounds_h + [(0.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs', options={'time_limit': 10})
            if res.success:
                h_lp = res.x[:n]
                # Line search along h* -> h_lp
                best_alpha = 0.0

            h_per = periodic_init(n_work, period, 0.5)
            c_per = compute_c5(h_per, n_work)
            hp, cp = lbfgs_phase(h_per, n_work, deadline - 25,
                alphas=[1000, 10000, 100000], mu=0.005,
                maxiter_per=[1000, 1500, 2000])
            if cp < best_c:
                best_c = cp
                best_h = hp
                best_n = n_work
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
