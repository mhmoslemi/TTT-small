import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, target):
    """Project h onto {x: 0≤x_i≤1, sum(x)=target}"""
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_val(c, alpha):
    """Compute smooth max of c with parameter alpha"""
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    val = np.exp(shifted - mx)
    return m + (mx + np.log(val.sum())) / alpha

def smooth_max_grad(h, n, alpha):
    """Gradient of smooth max of correlation values"""
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def objective_penalty(h, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = smooth_max_val(c, alpha)
    val = m + mu * (h.sum() - n / 2.0) ** 2

def objective_penalty_grad(h, n, alpha, mu):
    g = smooth_max_grad(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0) * np.ones(n)
    return g

def fista_optimize(h, n, deadline, alpha=50000, max_iter=5000):
    """FISTA (accelerated proximal gradient) with projection"""
    h = project_box_sum(h.copy(), n / 2.0)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    
    h_prev = h.copy()
    t_k = 1.0
    step = 1.0  # initial step size (1/L)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # FISTA momentum
        t_prev = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = h + ((t_prev - 1.0) / t_k) * (h - h_prev)
        
        # Gradient step
        g = smooth_max_grad(y, n, alpha)
        
        # Backtracking line search for step size
        step_trial = step
        while True:
            h_cand = project_box_sum(y - step_trial * g, target)
            # Simple check: if gradient step reduces objective
            f_y = smooth_max_val(compute_c5_vec(y, n), alpha)
            f_cand = smooth_max_val(compute_c5_vec(h_cand, n), alpha)
            decrease = f_y - f_cand
            expected = 0.5 * step_trial * np.dot(g, y - h_cand)
            if decrease >= expected * 0.5:  # relaxed condition
                break
            step_trial *= 0.5
            if step_trial < 1e-12:
                break
        
        h_prev = h
        h = h_cand
        step = step_trial
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, target):
    """Euclidean projection onto {x: 0<=xi<=1, sum(x)=target}"""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def get_smooth_max_and_weights(c, alpha):
    """Return smooth max value and softmax weights"""
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    val = m + (mx + np.log(w.sum())) / alpha
    return val, w

def gradient_smooth_max(h, n, alpha):
    """Gradient of smooth max of C5 correlation values"""
    c = compute_c5_vec(h, n)
    val, w = get_smooth_max_and_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-15:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

    c = compute_c5_vec(h, n)
    val, _ = get_smooth_max_and_weights(c, alpha)
    val += mu * (h.sum() - n / 2.0) ** 2
    return val

    g = gradient_smooth_max(h, n, alpha)
    s = h.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, target):
    """Euclidean projection onto {x: 0<=xi<=1, sum(x)=target}"""
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_val_and_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        w = np.zeros_like(c)
        w[shifted > mx - 50] = 1.0
        w /= w.sum()
        val = m + (mx + np.log(np.exp(shifted - mx).sum())) / alpha
    else:
        w = np.exp(shifted - mx)
        w /= w.sum()
        val = m + (mx + np.log(w.sum())) / alpha
    return val, w

def gradient_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    _, w = smooth_max_val_and_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += -wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += -wm * h[kk:]
    return grad

def obj_penalty(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    val, _ = smooth_max_val_and_weights(c, alpha)
    s = h.sum() - n / 2.0
    val += mu * s * s
    return val

def obj_penalty_grad(h, n, alpha, mu):
    g = gradient_smooth_max(h, n, alpha)
    s = h.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_optimize(h, n, deadline, alphas, mu=5.0, max_iter=200):
    h = project_box_sum(h.copy(), n / 2.0)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 5:
            break
        max_it = max(30, min(max_iter, int(remaining * 3)))
        try:
            res = minimize(obj_penalty, h, jac=obj_penalty_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_it, 'ftol': 1e-18,
                                  'maxfun': max_it * 12, 'maxls': 30})
            h_new = project_box_sum(res.x, n / 2.0)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

            f_y = smooth_max_val_and_weights(compute_c5_vec(y, n), alpha)[0]
            f_cand = smooth_max_val_and_weights(compute_c5_vec(h_cand, n), alpha)[0]
            if f_cand <= f_y - 0.1 * s * np.dot(g, y - h_cand) + 1e-18:
                h_new = h_cand
                break
            s *= 0.5
        else:
            h_new = project_box_sum(y - s * g, target)
        
        h_prev = h
        h = h_new
        step = max(s, 1e-8)
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def focus_max_optimize(h, n, deadline, max_iter=20000):
    h = project_box_sum(h.copy(), n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    step = 0.02
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        k_max = int(np.argmax(c))
        k = k_max - (n - 1)
        
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] += dx * (1.0 - h[k:])
            g[k:] += -dx * h[:nv]
        else:
            kk = -k
            nv = n - kk
            g[kk:] += dx * (1.0 - h[:nv])
            g[:nv] += -dx * h[kk:]
        
        s = step
        improved = False
        for _ in range(30):
            h_new = project_box_sum(h - s * g, target)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            s *= 0.5
        if not improved:
            step *= 0.7
            if step < 1e-12:
                break
        else:
            step = min(step * 1.1, 0.1)
    
    return best_h, best_c

def local_pair_search(h, n, deadline, base_delta=0.002, radius=40):
    h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    deltas = [base_delta, base_delta * 0.5, base_delta * 0.25, base_delta * 0.1]
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def random_local_search(h, n, deadline, n_trials=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    deltas = [0.003, 0.001, 0.0005, 0.0002]
    
    for d in deltas:
        any_improved
        = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for _ in range(n_trials):
                if time.time() > deadline:
                    break
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d; h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c; any_improved = True
                        break
                    else:
                        h[i] -= d; h[j] += d
                elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    h[i] -= d; h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c; any_improved = True
                        break
                    else:
                        h[i] += d; h[j] -= d
    return h, best_c

def simulated_anneal(h, n, deadline, max_iter=5000, T_init=0.001, T_end=1e-8):
    h = project_box_sum(h.copy(), n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    curr_c = best_c
    T = T_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T_init * (T_end / T_init) ** (it / max(1, max_iter))
        
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = np.random.uniform(-0.01, 0.01)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_box_sum(h_new, target)
        
        c_new = compute_c5(h_new, n)
        delta_c = c_new - curr_c
        
        if delta_c < 0 or np.random.random() < np.exp(-delta_c / max(T, 1e-12)):
            h = h_new
            curr_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 3
    
    # Initialize from parent if available
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    
    best_h = project_box_sum(h_init, n_init / 2.0)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    
    # Phase 1: Quick L-BFGS at original resolution
    if time.time() < deadline - 500:
        t1 = min(time.time() + 70, deadline - 500)
        h1, c1 = lbfgs_optimize(best_h, best_n, t1,
                                alphas=(5000, 50000, 200000, 1000000),
                                mu=5.0, max_iter=150)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Interpolate to n=400 and optimize
    h400_candidate = None
    if time.time() < deadline - 350:
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, 400)
        h_interp = np.interp(x_new, x_old, best_h)
        h400 = project_box_sum(h_interp, 200.0)
        t2 = min(time.time() + 150, deadline - 350)
        h400, c400 = lbfgs_optimize(h400, 400, t2,
                                     alphas=(5000, 50000, 200000, 1000000),
                                     mu=3.0, max_iter=200)
        if c400 < best_c:
            best_c = c400
            best_h = h400.copy()
            best_n = 400
        h400_candidate = (h400, c400)
    
    # Phase 3: Interpolate to n=500 and optimize
    if time.time() < deadline - 250:
        src_h = best_h
        src_n = best_n
        x_old = np.linspace(0, 2, src_n)
        x_new = np.linspace(0, 2, 500)
        h_interp = np.interp(x_new, x_old, src_h)
        h500 = project_box_sum(h_interp, 250.0)
        t3 = min(time.time() + 150, deadline - 250)
        h500, c500 = lbfgs_optimize(h500, 500, t3,
                                     alphas=(5000, 50000, 200000, 1000000),
                                     mu=3.0, max_iter=200)
        if c500 < best_c:
            best_c = c500
            best_h = h500.copy()
            best_n = 500
    
    # Phase 4: FISTA at current best
    if time.time() < deadline - 180:
        t4 = min(time.time() + 120, deadline - 180)
        h_f, cf = fista_optimize(best_h, best_n, t4, alpha=200000, max_iter=8000)
        if cf < best_c:
            best_c = cf
            best_h = h_f
    
    # Phase 5: Focus-max optimization
    if time.time() < deadline - 130:
        t5 = min(time.time() + 80, deadline - 130)
        h5, c5 = focus_max_optimize(best_h, best_n, t5, max_iter=15000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Another round of L-BFGS with very high alpha
    if time.time() < deadline - 100:
        t6 = min(time.time() + 70, deadline - 100)
        h6, c6 = lbfgs_optimize(best_h, best_n, t6,
                                 alphas=(1000000, 5000000, 10000000),
                                 mu=1.0, max_iter=100)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Local pair search
    if time.time() < deadline - 40:
        t7 = min(time.time() + 50, deadline - 40)
        radius = 30 if best_n <= 400 else 25
        h7, c7 = local_pair_search(best_h, best_n, t7, base_delta=0.001, radius=radius)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Simulated annealing for diversification
    if time.time() < deadline - 20:
        t8 = min(time.time() + 25, deadline - 15)
        h8, c8 = simulated_anneal(best_h, best_n, t8, max_iter=3000,
                                   T_init=0.0005, T_end=1e-10)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Final random local search
    if time.time() < deadline - 5:
        t9 = deadline - 3
        h9, c9 = random_local_search(best_h, best_n, t9, n_trials=3000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final cleanup
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
