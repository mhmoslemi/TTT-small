            grad[kk:] += w[mm] * (1.0 - h[:nv]) * dx
            grad[:nv] += w[mm] * (-h[kk:]) * dx
    return grad

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=5000):
    def obj(x):
        return lse_obj(x, n, alpha)
    def obj_grad(x):
        return lse_grad(x, n, alpha)
    def con_fun(x):
        return float(x.sum() - n / 2.0)
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def lbfgs_optimize(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [100, 1000, 10000, 100000, 500000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        def obj(x):
            return lse_obj(x, n, alpha)
        def grad(x):
            g = lse_grad(x, n, alpha)
            # Add constraint gradient (penalty)
            g = g + 10.0 * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 10000, 'ftol': 1e-18,
                                                   'gtol': 1e-15, 'maxfun': 100000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def local_search(h, n, deadline, max_radius=80, steps=None):
    if steps is None:
        steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for step in steps:
        if time.time() > deadline - 2:
            break
        improved = True
        while improved and time.time() < deadline - 2:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            # Compute combined gradient (pointing in direction that reduces active constraints)
            grad = np.zeros(n)
            for idx in active:
                k = int(idx) - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[:nv] += (1.0 - h[k:]) * dx
                    grad[k:] += (-h[:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:] += (1.0 - h[:nv]) * dx
                    grad[:nv] += (-h[kk:]) * dx
            if len(active) > 0:
                grad /= len(active)
            # Try moving mass from high-gradient to low-gradient points
            # to reduce the max
            sort_idx = np.argsort(-np.abs(grad))
            top = sort_idx[:min(max_radius, n // 4)]
            bottom = sort_idx[n - min(max_radius, n // 4):]
            for i in top:
                if time.time() > deadline - 2:
                    break
                for j in bottom:
                    if i == j:
                        continue
                    # Try h[i] += step, h[j] -= step
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Try h[i] -= step, h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

Hmm, but I'm worried this is too similar to the previous approach. Let me think about what's fundamentally different I can do.

**Key idea: Use the bilinear structure more directly.**

The objective is: C₅ = max_k [sum_i h[i] - sum_i h[i]*h[i+k']]*dx

Since sum_i h[i] = n/2 (fixed), this is:
C₅ = max_k [(n/2) - sum_i h[i]*h[i+k']]*dx = max_k [(n/2)*dx - (sum_i h[i]*h[i+k'])*dx]

Since (n/2)*dx = (n/2)*(2/n) = 1, we have:
C₅ = max_k [1 - sum_i h[i]*h[i+k']*dx]

So minimizing C₅ is equivalent to minimizing max_k [1 - Q_k(h)] where Q_k(h) = sum_i h[i]*h[i+k']*dx.

This is equivalent to maximizing min_k Q_k(h).

The Q_k(h) are (weighted) autocorrelations of h. We want to maximize the minimum autocorrelation over all shifts.

This is a classic "maximin" problem. The optimal h should have "flat" autocorrelation spectrum, similar to a Barker code or m-sequence.

**Different approach: Spectral method**

The autocorrelation Q_k = sum_i h[i]*h[i+k'] can be related to the Fourier transform:
Q_k = (1/n) * sum_freq |H(f)|² * exp(2πi*f*k'/n) (approximately, for DFT)

So maximizing min_k Q_k is related to making the power spectrum |H(f)|² as flat as possible (white noise) while maintaining the DC component (mean) at 0.5.

But this is just intuition, not a direct algorithm.

**Different approach: Alternating minimization**

Since Q_k(h) = h^T P_k h (where P_k is a circulant-like matrix), we can try:
1. Fix the "shape" of h (e.g., which bins are 0, 0.5, or 1)
2. Optimize the exact values

Or:

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()
    sh = alpha * (c - m)

            if len(i_candidates) == 0 or len(j_candidates) == 0:
                break
            # Take top 50 of each
            i_top = i_candidates[np.argsort(grad_zero[i_candidates])[:min(50, len(i_candidates))]]
            j_top = j_candidates[np.argsort(-grad_zero[j_candidates])[:min(50, len(j_candidates))]]
            found = False
            for i in i_top:

                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            found = True
                        else:
                            h[i] -= step
                            h[j] += step
    return h, best_c

def slsqp_polish(h, n, deadline, alpha=2000000, maxiter=10000):
    def obj(x):
        return lse_objective(x, n, alpha, 0.0)
    def grad(x):
        return lse_gradient(x, n, alpha, 0.0)
    def con_fun(x):
        return float(x.sum() - n / 2.0)
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except:
        return h.copy(), compute_c5(h, n)

def perturbation_restart(h, n, scale, rng):
    """Add small random perturbation preserving constraints."""
    h_pert = h.copy()
    pert = rng.standard_normal(n) * scale
    # Make it zero-sum
    pert -= pert.mean()
    h_pert += pert
    return project_feasible(h_pert, n)
