And a **coordinate-descent refinement**: for each pair (i, j), find the optimal h[i], h[j] given all others fixed. This is a 2D optimization that can be done analytically.

Actually, for a pair (i, j), the objective as a function of h[i] and h[j] (with all other h fixed) is:
f(h[i], h[j]) = max_k c_k(h[i], h[j])

For each k, c_k is linear in h[i] and h[j] (since it only involves products h[a]*h[b] where at most one of a,b is i or j... wait, no. c_k = sum h[a]*(1-h[a+k]). If both i and j appear in this sum, we get a product h[i]*h[j] term if j = i+k or i = j+k.

So for a given k, c_k as a function of (h[i], h[j]):
- If j = i+k or i = j+k: c_k contains the term h[i]*h[j] (or equivalent), so it's bilinear
- Otherwise: c_k is linear in h[i] and h[j] separately

This makes the 2D subproblem a concave (or bilinear) function, which is hard to optimize exactly. The max over k makes it even harder.

So coordinate descent on pairs is not straightforward. Let me stick with the gradient-based approach + local perturbation.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += min(1.0, diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += min(1.0, diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Positive shifts k = 0..n-1, index in c: n-1+k
    for k in range(n):
        wm = w[n - 1 + k]
        if wm < 1e-14:
            continue
        wm = wm * dx
        nv = n - k
        if k > 0:
            grad[k:n] += wm * (1.0 - h[0:nv])
        grad[0:nv] += wm * (-h[k:n])
    # Negative shifts kk = 1..n-1, index in c: n-1-kk
    for kk in range(1, n):
        wm = w[n - 1 - kk]
        if wm < 1e-14:
            continue
        wm = wm * dx
        nv = n - kk
        grad[kk:n] += wm * (1.0 - h[0:nv])
        grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    val = smooth_max_obj(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_alphas=8):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [10, 50, 200, 1000, 5000, 20000, 100000, 500000]
    mus = [1, 1, 1, 1, 0.1, 0.01, 0.001, 0.0001]
    
    for i in range(max_alphas):
        if time.time() > deadline - 3:
            break
        alpha = alphas[min(i, len(alphas)-1)]
        mu = mus[min(i, len(mus)-1)]
        t_alloc = min(120, deadline - time.time() - 3)
        if t_alloc < 5:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'feps': 1e-14,
                                   'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=1000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return float(smooth_max_obj(x, n, alpha))
    
    # Constraint: sum(x) = n/2
    from scipy.optimize import minimize as min2
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)})
    
    try:
        res = min2(obj, h, method='SLSQP', bounds=bounds,
                  constraints=[cons],
                  options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def momentum_pg(h, n, deadline, alpha=500, lr=0.001, momentum=0.9):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    
    for it in range(50000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        # Project gradient onto feasible manifold (sum constraint)
        g = g - g.mean()
        v = momentum * v + (1 - momentum) * g
        d = -v
        # Line search
        best_step = 0
        for trial in range(30):
            step = lr / (2 ** trial)
            new_h = project_feasible(h + step * d, n)
            c = compute_c5(new_h, n)
            if c < best_c - 1e-15:
                best_c = c
                h = new_h
                best_h = h.copy()
                best_step = step
                break
        if best_step == 0:
            lr *= 0.5
            if lr < 1e-12:
                break
    
    return best_h, best_c

def targeted_local_search(h, n, deadline):
    """Target the worst shifts for local improvement."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            # Focus on indices near the maximum
            threshold = best_c - 0.005
            active = np.where(c_vec > threshold)[0]
            
            for idx in active:
                if time.time() > deadline:
                    return h, best_c
                k = idx - (n - 1)
                # For this shift, try to reduce c_k
                # c_k = sum h[i+k]*(1-h[i]) for valid i (if k>=0)
                # or sum h[i]*(1-h[i-kk]) for valid i (if k<0)

                        # Try moving mass from h[i+k] to h[i] (or vice versa)

                            h[i+k] -= d
                            h[i] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i+k] += d
                                h[i] -= d
                else:
                    kk = -k
                    nv = n - kk
                    for i in range(kk, kk + nv, max(1, nv // 100)):
                        if h[i] > d and h[i-kk] < 1.0 - d:
                            h[i] -= d
                            h[i-kk] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] += d
                                h[i-kk] -= d
            
            if not improved:
                break
    
    return h, best_c

def generic_pair_search(h, n, deadline, deltas, radius=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
