        if s >= 0:
            nv = n - s
            grad[0:nv] += wm * (1.0 - x[s:n])
            grad[s:n] += -wm * x[0:nv]
        else:
            ll = -s
            nv = n - ll
            grad[ll:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += -wm * x[ll:n]
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def top_k_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 300:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def top_k_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-12:
            continue
        s = idx - (n - 1)
        wm = wi * dx
        if s >= 0:
            nv = n - s
            grad[0:nv] += wm * (1.0 - x[s:n])
            grad[s:n] += -wm * x[0:nv]
        else:
            ll = -s
            nv = n - ll
            grad[ll:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += -wm * x[ll:n]
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_set_optimize(h, n, deadline, n_restarts=3):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    for restart in range(n_restarts):
        if time.time() > deadline - 30:
            break
        cur_h = best_h.copy()
        if restart > 0:
            n perturbs = 10 + restart * 5
            idx = np.random.choice(n, n_perturbs, replace=False)
            cur_h[idx] += np.random.uniform(-0.01, 0.01, n_perturbs)
            cur_h = project_feasible(cur_h, n)
        
        for iter_round in range(8):
            if time.time() > deadline - 15:
                break
            c_vec = compute_c5_vec(cur_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-8, 0.005 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > 8:
                order = np.argsort(-c_vec[active])
                active = active[order[:8]]
            
            if len(active) == 0:
                break
            
            for alpha, mu in [(2000, 10.0), (10000, 50.0), (50000, 200.0), (200000, 1000.0)]:
                if time.time() > deadline - 10:
                    break
                try:
                    res = minimize(top_k_obj, cur_h, jac=top_k_grad,
                                  args=(n, active, alpha, mu), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': 3000, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxfun': 20000, 'maxls': 40})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                        cur_h = h_new
                except Exception:
                    pass
    
    return best_h, best_c

def full_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 5.0), (500, 50.0), (2000, 300.0),
                   (5000, 1000.0), (20000, 5000.0), (100000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(soft_max_obj, h, jac=soft_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 4000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    def sq_fun(x):
        return float(x.sum() - n / 2.0)
    def sq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac}]
    try:
        res = minimize(soft_max_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def random_perturb_restart(h, n, deadline, n_tries=5, scale=0.005):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for t in range(n_tries):
        if time.time() > deadline - 20:
            break
        h_try = h.copy()
        idx = np.random.choice(n, min(20, n), replace=False)
        h_try[idx] += np.random.uniform(-scale, scale, len(idx))
        h_try = project_feasible(h_try, n)
        
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_h = h_try
            best_c = c_try
        
        # Quick L-BFGS from perturbed start
        if time.time() > deadline - 10:
            break
        try:
            for alpha, mu in [(5000, 100.0), (50000, 1000.0)]:
                if time.time() > deadline - 8:
                    break
                res = minimize(soft_max_obj, h_try, jac=soft_max_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': 15000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h_try = h_new
        except Exception:
            pass
    
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def two_point_local_search(h, n, deadline, deltas=None):
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        iters = 0
        while improved and iters < 200:
            improved = False
            iters += 1
            # Sample random pairs for efficiency at high n
            n_pairs = min(500, n * 2)
            for _ in range(n_pairs):
                if time.time() > deadline:
                    return h, best_c
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
                if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    h[i] -= d
                    h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 8
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_orig = 200
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick polish at original resolution
    t1 = min(t0 + 40, deadline - 500)
    if t1 > t0 + 10:
        best_h, best_c = full_lbfgs(best_h, n_orig, t1,
            configs=[(200, 20.0), (1000, 200.0), (5000, 2000.0)])
    
    # Phase 2: Multi-resolution refinement
    # Go to n=1000
    n1 = 1000
    h1 = upsample(best_h, n_orig, n1)
    c1 = compute_c5(h1, n1)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    t2 = min(t0 + 200, deadline - 350)
    if t2 > time.time() + 20:
        h1, c1 = full_lbfgs(h1, n1, t2,
            configs=[(200, 50.0), (1000, 500.0), (5000, 3000.0),
                     (20000, 10000.0), (100000, 50000.0)])
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 3: Active-set optimization at n=1000
    t3 = min(t0 + 400, deadline - 250)
    if t3 > time.time() + 15:
        h_act, c_act = active_set_optimize(best_h, n1, t3, n_restarts=3)
        if c_act < best_c:
            best_c = c_act
            best_h = h_act
    
    # Phase 4: Try n=2000
    n2 = 2000
    h2 = upsample(best_h, len(best_h), n2)
    c2 = compute_c5(h2, n2)
    
    t4_min = time.time() + 30
    t4_max = deadline - 120
    if t4_max > t4_min:
        h2, c2 = full_lbfgs(h2, n2, t4_max,
            configs=[(100, 20.0), (500, 200.0), (2000, 1000.0), (10000, 5000.0)])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 5: Active-set at the highest resolution
    n_final = len(best_h)
    t5_min = time.time() + 15
    t5_max = deadline - 80
    if t5_max > t5_min:
        h_act2, c_act2 = active_set_optimize(best_h, n_final, t5_max, n_restarts=2)
        if c_act2 < best_c:
            best_c = c_act2
            best_h = h_act2
    
    # Phase 6: Random perturbations
    t6_max = deadline - 50
    if t6_max > time.time() + 10:
        h_r, c_r = random_perturb_restart(best_h, n_final, t6_max, n_tries=3, scale=0.003)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    
    # Phase 7: SLSQP polish
    t7_max = deadline - 15
    if t7_max > time.time() + 5:
        for alpha in [50000, 200000]:
            if time.time() > t7_max:
                break
            h_s, c_s = slqp_polish(best_h, n_final, t7_max, alpha=alpha, maxiter=1500)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
    
    # Phase 8: Local two-point search
    t8_max = deadline - 3
    if t8_max > time.time() + 3:
        best_h, best_c = two_point_local_search(best_h, n_final, t8_max,
            deltas=[0.0005, 0.0002, 0.0001])
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
