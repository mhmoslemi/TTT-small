    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        
        # Compute subgradient (uniform over active set)
        dx = 2.0 / n
        active = np.where(c_vec >= c_max - 1e-6)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)

def coord_descent_refine(h, n, deadline, max_sweeps=50):
    """Coordinate descent on pairs with gradient-based direction."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        active = np.where(c_vec >= c_max - 1e-5)[0]

        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        
        # Update the most negative gradient component (increase h)
        # and most positive (decrease h)
        # But we need sum(h) = n/2, so we transfer between pairs
        
        # Find best pair (i,j) for transfer
        best_pair_c = best_c
        best_i = -1
        best_j = -1
        best_step = 0.0
        best_transfer = 0.0
        
        # Try top gradient positions
        neg_idx = np.argsort(grad)[:20]  # most negative = want to increase h
        pos_idx = np.argsort(-grad)[:20]  # most positive = want to decrease h
        
        for i in neg_idx:
            if best_h[i] >= 1.0:
                continue
            for j in pos_idx:
                if i == j:
                    continue
                if best_h[j] <= 0.0:
                    continue
                # Transfer delta from j to i
                for delta in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
                    if best_h[i] + delta > 1.0 or best_h[j] - delta < 0.0:
                        continue
                    best_h[i] += delta
                    best_h[j] -= delta
                    c = compute_c5(best_h, n)
                    if c < best_pair_c:
                        best_pair_c = c
                        best_i, best_j = i, j
                        best_step = delta
                        best_transfer = 1
                    best_h[i] -= delta
                    best_h[j] += delta
                # Reverse transfer
                for delta in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
                    if best_h[j] + delta > 1.0 or best_h[i] - delta < 0.0:

                    if c < best_pair_c:
                        best_pair_c = c
                        best_i, best_j = i, j
                        best_step = delta
                        best_transfer = -1
                    best_h[i] += delta
                    best_h[j] -= delta
        
        if best_i >= 0:
            if best_transfer > 0:
                best_h[best_i] += best_step
                best_h[best_j] -= best_step
            else:
                best_h[best_i] -= best_step
                best_h[best_j] += best_step
            best_h = project_feasible(best_h, n)
            best_c = best_pair_c
            improved = True
        
        if not improved:
            break
    
    return best_h, best_c

Let me also think about a key insight: the correlation `np.correlate(h, 1-h, 'full')` has a specific structure. The value at lag 0 (middle of the full correlation) is:
c[n-1] = sum_j h[j] * (1-h[j]) = sum_j h[j] - sum_j h[j]² 

This is maximized when h is as "spread out" as possible (many values near 0.5) and minimized when h is binary (many 0s and 1s).

For the other lags, the structure depends on how the "mass" of h is distributed relative to its shifted version.

The key to minimizing the max is to make all lags equally small. This is reminiscent of a "flat" autocorrelation, which is related to Barker sequences and similar constructions.

For a binary sequence of length n with n/2 ones, the autocorrelation at lag k is:
R(k) = sum_j h[j] * h[j+k]

We want R(k) to be as uniform as possible (ideally n/4 for all k, which is the "chirp" level).

The minimum possible max of (n/2 - R(k)) is achieved when R(k) is as large as possible for all k simultaneously, which is constrained by R(0) = n/2.

This is essentially a "difference set" or "Barker sequence" type problem.

**New idea: KKT-based polishing**

At the optimum, we should have:
- For active correlations k: the directional derivative in any feasible direction is >= 0
- The integral constraint is active: sum(h) = n/2
- For h[i] = 0: the gradient component <= 0 (can't decrease further)
- For h[i] = 1: the gradient component >= 0 (can't increase further)
- For 0 < h[i] < 1: the gradient component should be a constant (related to the Lagrange multiplier for the integral constraint)

So the "ideal" gradient for interior points should be a constant. We can use this to guide the polishing: transfer mass from points with above-average gradient to points with below-average gradient.

    best_h = h.copy()

    dx = 2.0 / n
    eta = 0.005
    
    for it in range(n_iter):
        if time.time() > deadline:

        grad = np.zeros(n)
        for idx in active:

            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:

                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        n_active = len(active)
        if n_active > 0:
            grad /= n_active
        
        # Remove the mean (integral constraint): we want to modify h

        interior = (best_h > 1e-6) & (best_h < 1 - 1e-6)
        grad_proj = grad - np.mean(grad)  # remove component along (1,...,1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    sig = np.where(w > 1e-12)[0]
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in sig:
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgradient(x, n, tol_frac=0.001):

    c_max = c.max()
    tol = max(1e-9, tol_frac * c_max)
    active = np.where(c >= c_max - tol)[0]

            nv = n - kk

            grad[0:nv] += (-x[kk:n]) * dx_val
    n_act = len(active)
    if n_act > 0:
        grad /= n_act

def lbfgs_phase(h, n, deadline, alphas, mu, maxit_list):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

        if rem < 20:

        if i < len(maxit_list):

        else:
            maxit = max(500, int(rem * 3))
        maxit = min(maxit, max(200, int(rem * 4)))
        try:

def proj_grad_momentum(h, n, deadline, n_iter=8000, eta0=0.008, beta=0.7):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    eta = eta0
    momentum = np.zeros(n)
    prev_grad = None
    for it in range(n_iter):
        if time.time() > deadline:
            break
        grad = subgradient(best_h, n)
        if prev_grad is not None:
            momentum = beta * momentum + (1 - beta) * grad
        else:
            momentum = grad.copy()
        prev_grad = grad.copy()
        # Remove mean to preserve sum approximately
        mom_proj = momentum - np.mean(momentum)
        h_new = best_h - eta * mom_proj
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.01, 0.03)
        else:
            # Backtrack
            accepted = False
            for fac in [0.5, 0.2, 0.05]:
                h_try = best_h - eta * fac * mom_proj
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-16:
                    best_c = c_try
                    best_h = h_try
                    eta = min(eta * 1.005, 0.03)
                    accepted = True
                    break
            if not accepted:
                eta = max(eta * 0.6, 1e-5)
                momentum *= 0.5
    return best_h, best_c

def grad_pair_search(h, n, deadline, n_rounds=500):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        grad = subgradient(best_h, n)
        interior = (best_h > 0.001) & (best_h < 0.999)
        if not np.any(interior):
            break
        # Find best donor (grad > 0, want to decrease) and recipient (grad < 0, want to increase)
        g_int = grad[interior]
        if len(g_int) < 2:
            break
        # Top donors and recipients
        idx_int = np.where(interior)[0]
        order = np.argsort(g_int)
        donors = idx_int[order[-10:]]   # highest grad -> want to decrease h
        recipients = idx_int[order[:10]]  # lowest grad -> want to increase h
        improved = False
        for j in donors:
            if improved:
                break
            for i in recipients:
                if i == j:
                    continue
                for delta in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                    if best_h[i] + delta > 1.0 or best_h[j] - delta < 0.0:
                        continue
                    best_h[i] += delta
                    best_h[j] -= delta
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                        break
                    else:
                        best_h[i] -= delta
                        best_h[j] += delta
                        break
                if improved:
                    break
                # Reverse: decrease i, increase j
                for delta in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
                    if best_h[j] + delta > 1.0 or best_h[i] - delta < 0.0:
                        continue
                    best_h[i] -= delta
                    best_h[j] += delta
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                        break
                    else:
                        best_h[i] += delta
                        best_h[j] -= delta
                        break
    return best_h, best_c

                            best_h[j] -= d
    return best_h, best_c

def upsample(h_coarse, n_coarse, n_fine):
    x_c = np.linspace(0, 2, n_coarse, endpoint=False)
    x_f = np.linspace(0, 2, n_fine, endpoint=False)
    h_fine = np.interp(x_f, x_c, np.concatenate([h_coarse, h_coarse[:1]]), period=n_coarse)
    # Actually, np.interp with period:
    h_fine = np.interp(x_f, x_c, h_coarse, period=n_coarse)
    h_fine = project_feasible(h_fine, n_fine)
    return h_fine

def multi_restart(h_best, c_best, n, deadline, n_restarts=3, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        # Structured perturbation
        h_pert = h_best.copy()
        n_pert = rng.randint(5, 20)
        indices = rng.choice(n, size=n_pert, replace=False)
        for idx in indices:
            delta = rng.uniform(-0.02, 0.02)
            h_pert[idx] += delta
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert > c_best + 0.005:
            continue
        # Quick optimization from perturbed start
        rem = deadline - time.time() - 5
        if rem < 50:
            break
        h_opt, c_opt = lbfgs_phase(h_pert, n, time.time() + min(rem * 0.6, 80),
                                    alphas=[1000, 10000, 100000, 500000],
                                    mu=0.01, maxit_list=[500, 1000, 1500, 2000])
        if c_opt < c_best:
            h_best = h_opt
            c_best = c_opt
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    # Load initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: L-BFGS-B at original resolution
    p1_end = t0 + 280
    h1, c1 = lbfgs_phase(best_h, best_n, p1_end,
        alphas=[20, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.01,
        maxit_list=[800, 1000, 1500, 2000, 2500, 3000, 3500, 4000, 5000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set refinement at original resolution
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 80, deadline - 400)
        h2, c2 = active_phase(best_h, best_n, p2_end, top_k=6,
            alphas=[10000, 50000, 200000, 1000000, 5000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Projected gradient with momentum
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 60, deadline - 350)
        h3, c3 = proj_grad_momentum(best_h, best_n, p3_end, n_iter=6000, eta0=0.005)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to higher resolution and optimize
    n_fine = 1000
    if time.time() < deadline - 250 and best_n < n_fine:
        h_fine = upsample(best_h, best_n, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        if c_fine < best_c + 0.005:
            p4_end = min(time.time() + 200, deadline - 150)
            h4, c4 = lbfgs_phase(h_fine, n_fine, p4_end,
                alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
                mu=0.005,
                maxit_list=[500, 800, 1500, 2000, 2500, 3000, 3500, 4000])
            if c4 < best_c:
                best_c = c4
                best_h = h4
                best_n = n_fine

    # Phase 5: Active set at best resolution
    if time.time() < deadline - 120:
        p5_end = min(time.time() + 80, deadline - 100)
        h5, c5 = active_phase(best_h, best_n, p5_end, top_k=5,
            alphas=[50000, 200000, 1000000, 5000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Projected gradient at best resolution
    if time.time() < deadline - 100:
        p6_end = min(time.time() + 50, deadline - 80)
        h6, c6 = proj_grad_momentum(best_h, best_n, p6_end, n_iter=5000, eta0=0.003)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Gradient-guided pair search
    if time.time() < deadline - 70:
        p7_end = min(time.time() + 40, deadline - 50)
        h7, c7 = grad_pair_search(best_h, best_n, p7_end, n_rounds=300)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Multi-restart
    if time.time() < deadline - 60:
        p8_end = min(time.time() + 50, deadline - 40)
        best_h, best_c = multi_restart(best_h, best_c, best_n, p8_end,
                                        n_restarts=3, rng=rng)

    # Phase 9: Final fine sweep
    if time.time() < deadline - 15:
        p9_end = deadline - 5
        h9, c9 = fine_sweep(best_h, best_n, p9_end,
                           deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Final projected gradient (very short)
    if time.time() < deadline - 5:
        p10_end = deadline - 2
        h10, c10 = proj_grad_momentum(best_h, best_n, p10_end, n_iter=2000, eta0=0.002)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
