def adam_optimize(h, n, deadline, lr=0.01, beta1=0.9, beta2=0.999, alpha_start=100, alpha_end=100000):
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        t += 1
        # Anneal alpha
        alpha = alpha_start * (alpha_end/alpha_start) ** (t / max(t, 1000))
        
        c = compute_c5_vec(h, n)
        c_max = c.max()
        # Soft max weights
        sh = alpha * (c - c_max)
        w = np.exp(sh)
        w /= w.sum()
        
        # Gradient of smooth max
        grad = np.zeros(n)
        dx = 2.0 / n
        for mm in range(2*n-1):
            if w[mm] < 1e-12:
                continue
            k = mm - (n-1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
        
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        h_new = project_feasible(h_new, n)

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha

def adam_optimize(h, n, deadline, lr=0.5, alpha_start=50, alpha_end=200000, 
                  beta1=0.9, beta2=0.999, eps=1e-8, mu=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    patience = 500
    no_improve = 0
    cur_alpha = alpha_start
    
    while time.time() < deadline - 5:
        t += 1
        # Geometric annealing
        progress = min(1.0, t / 5000.0)
        cur_alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        c = compute_c5_vec(h, n)
        c_max = c.max()
        
        if c_max < best_c - 1e-15:
            best_c = c_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Reduce learning rate
            lr *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break
        
        # Compute weights
        sh = cur_alpha * (c - c_max)
        mx = sh.max()
        w = np.exp(sh - mx)
        w_total = w.sum()
        if w_total < 1e-300:
            w = np.zeros(2*n-1)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_total
        
        # Gradient
        dx = 2.0 / n
        grad = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-13:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - h[k:])
                grad[k:] += wm * (-h[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - h[:nv])
                grad[:nv] += wm * (-h[kk:])
        
        if mu > 0:
            grad += 2.0 * mu * (h.sum() - n / 2.0)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        h = h_new
    
    return best_h, best_c

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    g = full_grad(x, n, w)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w_local = np.exp(sh - smx)
    w_local /= w_local.sum()
    w_full = np.zeros(2 * n - 1)
    for idx, wi in zip(active, w_local):
        w_full[idx] = wi
    g = full_grad(x, n, w_full)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(remaining * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxit * 10,
                                   'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_h = h_new
                best_c = c_new
        except Exception:
            pass
    return best_h, best_c

def adam_iter(h, n, deadline, lr_init=2.0, alpha_init=50, alpha_max=300000,
              beta1=0.9, beta2=0.999, eps=1e-8, mu=0.005, patience=800):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    lr = lr_init
    no_img = 0

    while time.time() < deadline - 2:
        t += 1
        progress = min(1.0, t / 8000.0)
        alpha = alpha_init * (alpha_max / alpha_init) ** progress

        c = compute_c5_vec(h, n)
        c_max = c.max()
        if c_max < best_c - 1e-15:
            best_c = c_max
            best_h = h.copy()
            no_img = 0
        else:
            no_img += 1
        if no_img > patience:
            lr *= 0.6
            no_img = 0
            if lr < 1e-7:
                break

        sh = alpha * (c - c_max)
        mx = sh.max()
        w = np.exp(sh - mx)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w /= ws

        grad = full_grad(h, n, w)
        if mu > 0:
            grad += 2.0 * mu * (h.sum() - n / 2.0)

        m_v = beta1 * m_v + (1.0 - beta1) * grad
        v_v = beta2 * v_v + (1.0 - beta2) * grad * grad
        m_hat = m_v / (1.0 - beta1 ** t)
        v_hat = v_v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        h = h_new

    return best_h, best_c

def active_lbfgs(h, n, deadline, top_k=8, alphas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000]

    for rnd in range(15):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_v = c_vec.max()
        thr = max_v - max(1e-8, 0.0003 * max_v)
        active = np.where(c_vec >= thr)[0]
        if len(active) > top_k:
            idx_sorted = np.argsort(-c_vec[active])
            active = active[idx_sorted[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(100, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxit * 10,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new <= best_c:
                    best_h = h_new
                    best_c = c_new
            except Exception:
                pass
    return best_h, best_c

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def slqp_polish(h, n, deadline, alpha=200000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def random_pair_search(h, n, deadline, n_trials=80000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    step_choices = np.array([0.002, 0.001, 0.0005, 0.005, 0.008])

    for s in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step_choices[rng.randint(len(step_choices))]
        done = False
        if h[i] + d <= 1.0 and h[j] - d >= 0.0:
            h[i] += d
            h[j] -= d
            c = compute_c5(h, n)
            if c < best_c - 1e-16:
                best_c = c
                best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
            done = True
        if not done and h[i] - d >= 0.0 and h[j] + d <= 1.0:
            h[i] -= d
            h[j] += d
            c = compute_c5(h, n)
            if c < best_c - 1e-16:
                best_c = c
                best_h = h.copy()
            else:
                h[i] += d
                h[j] -= d
    return best_h, best_c

def deterministic_coord(h, n, deadline, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 5:
            improved = False
            for i in range(n):
                if time.time() > deadline - 5:
                    break
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def make_periodic(n, duty=0.5, period_frac=1.0):
    h = np.zeros(n)
    t = np.arange(n) * (2.0 / n)
    period = 2.0 * period_frac
    phase = (t % period)
    h[phase < period * duty] = 1.0
    h = project_feasible(h, n)
    return h

def binary_project(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] += remaining
    return project_feasible(h_bin, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick L-BFGS polish at original n (120s)
    p1_end = t0 + 120
    h1, c1 = lbfgs_phase(h0, n0, p1_end,
                         alphas=[100, 1000, 10000, 100000, 500000],
                         mu=0.01, maxiters=[1500, 2000, 2500, 3000, 3000])
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Upsample to n=1000 and optimize (120s)
    n_main = 1000
    if time.time() < deadline - 600:
        x_fine = np.linspace(0, 2, n_main)
        x_coarse = np.linspace(0, 2, best_n)
        h_up = np.interp(x_fine, x_coarse, best_h)
        h_up = project_feasible(h_up, n_main)
        c_up = compute_c5(h_up, n_main)

        p2_end = min(time.time() + 100, deadline - 600)
        h2, c2 = lbfgs_phase(h_up, n_main, p2_end,
                             alphas=[20, 100, 500, 2000, 10000, 50000],
                             mu=0.005, maxiters=[1000, 1500, 2000, 2500, 2500, 3000])
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, n_main
        elif c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_main

    # Phase 3: Main Adam optimization at n_main (500s)
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 480, deadline - 200)
        h3, c3 = adam_iter(best_h, best_n, p3_end,
                          lr_init=3.0, alpha_init=50, alpha_max=300000,
                          beta1=0.9, beta2=0.999, mu=0.003, patience=1000)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Active-set L-BFGS refinement (120s)
    if time.time() < deadline - 100:
        p4_end = min(time.time() + 100, deadline - 90)
        h4, c4 = active_lbfgs(best_h, best_n, p4_end, top_k=6,
                              alphas=[20000, 100000, 500000, 2000000])
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Second Adam pass with tighter params (80s)
    if time.time() < deadline - 60:
        p5_end = min(time.time() + 60, deadline - 50)
        h5, c5 = adam_iter(best_h, best_n, p5_end,
                          lr_init=0.5, alpha_init=200000, alpha_max=1000000,
                          beta1=0.95, beta2=0.999, mu=0.001, patience=300)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: SLSQP polish (50s)
    if time.time() < deadline - 40:
        p6_end = min(time.time() + 35, deadline - 30)
        h6, c6 = slqp_polish(best_h, best_n, p6_end, alpha=500000, maxiter=800)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Random pair search (remaining time - 15s)
    if time.time() < deadline - 15:
        p7_end = deadline - 3
        n_trials = int((deadline - 3 - time.time()) * 2000)
        n_trials = max(5000, min(n_trials, 200000))
        h7, c7 = random_pair_search(best_h, best_n, p7_end, n_trials=n_trials)
        if c7 < best_c:
            best_h, best_c = h7, c7

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
