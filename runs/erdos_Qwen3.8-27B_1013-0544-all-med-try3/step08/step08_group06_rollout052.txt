def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h

            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha, top_k=5):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    # Use a sharp approximation: focus on top values
    m = c.max()
    # Subtract m and use log-sum-exp on the top few
    sorted_c = np.sort(c)[::-1]

    # Smooth max of top values
    shift = top - m
    max_shift = shift.max()
    log_sum = max_shift + np.log(np.sum(np.exp(shift - max_shift)))
    return float(m + log_sum / alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def obj_toph(x, n, alpha, top_k):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    if top_k >= len(c):
        sh = c - m
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    order = np.argpartition(c, -top_k)[-top_k:]
    top = c[order]
    sh = top - m
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def slsqp_step(h, n, alpha, top_k, maxiter=2000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_toph, h, args=(n, alpha, top_k), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    for n_seg in [2, 3, 4, 5, 6, 8]:
        seg_w = 2.0 / n_seg
        for pattern_idx in range(min(8, 2**n_seg)):
            vals = []
            for s in range(n_seg):
                bit = (pattern_idx >> s) & 1
                vals.append(0.8 if bit else 0.2)
            h = np.repeat(vals, n // n_seg + 1)[:n].astype(float)
            h = project_feasible(np.clip(h, 0, 1), n)
            candidates.append(h)
    # Random smooth
    for _ in range(20):
        freqs = rng.choice([1, 2, 3, 4, 5, 6, 7, 8], size=3, replace=False)
        amps = rng.uniform(0.1, 0.4, 3)
        phases = rng.uniform(0, 2*np.pi, 3)
        h = 0.5
        for f, a, p in zip(freqs, amps, phases):
            h += a * np.cos(np.pi * f * x + p)
        h = project_feasible(np.clip(h, 0, 1), n)
        candidates.append(h)
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 2

    # Load parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200
        h_init = np.full(n_init, 0.5)

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at parent resolution
    alloc1 = min(200, (deadline - t0) * 0.30)
    t1 = t0 + alloc1
    h_opt, c_opt = optimize_core(h_init, n_init, t1, rng, mode="standard")
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Try higher resolution
    x_best = np.linspace(0, 2, len(best_h))
    for n_try in [300, 400]:
        if time.time() > deadline - 150:
            break
        remaining = deadline - time.time()
        alloc = min(120, remaining * 0.35)
        if alloc < 40:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n_try)
        # Add small noise to escape exact interpolation
        h_interp += rng.normal(0, 0.005, n_try)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_core(h_interp, n_try, time.time() + alloc, rng, mode="fast")
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            x_best = x_new

    # Phase 3: Back to original resolution with fresh optimization
    if len(best_h) != n_init:
        if time.time() < deadline - 100:
            x_new = np.linspace(0, 2, n_init)
            h_interp = np.interp(x_new, x_best, best_h)
            h_interp = project_feasible(h_interp, n_init)
            h_interp += rng.normal(0, 0.003, n_init)
            h_interp = project_feasible(h_interp, n_init)
            alloc = min(80, deadline - time.time() - 50)
            if alloc > 30:
                h_opt, c_opt = optimize_core(h_interp, n_init, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 4: Structured restarts
    n_cur = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time()
        cands = structured_candidates(n_cur, rng)
        # Sort by initial quality
        cand_c5 = []
        for h_c in cands:
            cand_c5.append(compute_c5(h_c, n_cur))
        cand_c5 = np.array(cand_c5)
        order = np.argsort(cand_c5)
        n_try_struct = min(6, len(order))
        for idx in order[:n_try_struct]:
            if time.time() > deadline - 40:
                break
            alloc = min(50, (deadline - time.time() - 30) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_core(cands[idx], n_cur, time.time() + alloc, rng, mode="fast")
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Random restarts from best
    if time.time() < deadline - 60:
        for r in range(8):
            if time.time() > deadline - 35:
                break
            scale = 0.02 + 0.1 * rng.random()
            hp = best_h + rng.normal(0, scale, n_cur)
            hp = project_feasible(hp, n_cur)
            alloc = min(55, (deadline - time.time() - 25) * 0.45)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_core(hp, n_cur, time.time() + alloc, rng, mode="fast")
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 6: Final intensive polish
    if time.time() < deadline - 4:
        n_f = len(best_h)
        # Multiple SLSQP passes
        for a in [500000, 1000000, 2000000]:
            if time.time() > deadline - 2:
                break
            ht = slsqp_step(best_h, n_f, a, 20, maxiter=3000, ftol=1e-16)
            c = compute_c5(ht, n_f)
            if c < best_c:
                best_c = c
                best_h = ht

    # Final pair search
    if time.time() < deadline - 0.5:
        n_f = len(best_h)
        if n_f <= 120:
            best_h, best_c = pair_search_full(best_h, n_f, deadline - 0.1,
                                              [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        elif n_f <= 200:
            best_h, best_c = pair_search_near(best_h, n_f, deadline - 0.1,
                                              [0.001, 0.0005, 0.0002, 0.0001],
                                              radius=25)
        else:
            best_h, best_c = pair_search_near(best_h, n_f, deadline - 0.1,
                                              [0.001, 0.0005, 0.0002],
                                              radius=20)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
