def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    if mu > 0:
        m += mu * (x.sum() - n / 2.0) ** 2
    return float(m)

def active_grad(x, n, active, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    at_max = vals >= m - 1e-12
    cnt = max(at_max.sum(), 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(active)):
        if not at_max[i]:
            continue
        wm = dx / cnt
        k = active[i] - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_smooth(h, n, alpha, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 10,
                               'maxls': 40, 'maxcor': 30})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run_lbfgs_active(h, n, active, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(active_obj, h, jac=active_grad, args=(n, active, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 10,
                               'maxls': 40, 'maxcor': 30})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def get_active_set(h, n, top_k=10, rel_tol=0.005):
    c_vec = compute_c5_vec(h, n)
    c_max = c_vec.max()
    threshold = c_max - max(1e-10, rel_tol * c_max)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.argsort(-c_vec)[:top_k]
    return active, c_vec

def subgradient_refine(h, n, deadline, n_iter=30000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) > 20:
            active = np.argsort(-c_vec)[:20]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        h = h - eta * grad
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h.copy()
            eta = min(eta * 1.003, 0.01)
        else:
            eta *= 0.995
    
    return best_h, best_c

def binary_local_search(h, n, deadline, n_iter=200000, seed=42):
    rng = np.random.RandomState(seed)
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        orig_i = h[i]
        orig_j = h[j]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
        else:
            h[i] = orig_i
            h[j] = orig_j
    
    return h, best_c

def make_periodic_start(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(ones_per_period):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    # Get initial point from parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500
    
    h_parent = project_feasible(h_parent, n_parent)
    
    # Choose working resolution: n=2000
    n = 2000
    x_parent = np.linspace(0, 2, n_parent)
    x_new = np.linspace(0, 2, n)
    h = np.interp(x_new, x_parent, h_parent)
    h = project_feasible(h, n)
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: Smooth L-BFGS alpha annealing (n=2000) - ~280s
    phase1_schedule = [
        (100, 25),
        (500, 30),
        (2000, 35),
        (10000, 40),
        (50000, 45),
        (200000, 50),
        (1000000, 55),
    ]
    for alpha, secs in phase1_schedule:
        if time.time() > deadline - 380:
            break
        maxit = max(200, int(secs * 7))
        h, c = run_lbfgs_smooth(h, n, alpha, 0.003, maxit)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Active-set focused L-BFGS - ~180s
    p2_end = min(t0 + 550, deadline - 280)
    active_schedule = [
        (8, 0.002, 40),
        (5, 0.001, 45),
        (10, 0.001, 40),
        (15, 0.001, 40),
        (25, 0.0005, 30),
    ]
    for top_k, mu, secs in active_schedule:
        if time.time() > p2_end - 10:
            break
        active, _ = get_active_set(best_h, n, top_k=top_k, rel_tol=0.01)
        maxit = max(400, int(secs * 5))
        h, c = run_lbfgs_active(best_h, n, active, mu, maxit)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            # Try from current h too
            h, c = run_lbfgs_active(h, n, active, mu, maxit // 2)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Phase 3: Subgradient refinement - ~120s
    p3_end = min(t0 + 680, deadline - 180)
    if time.time() < p3_end - 5:
        h, c = subgradient_refine(best_h, n, p3_end, n_iter=25000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 4: More active-set optimization with refreshed active set - ~100s
    p4_end = min(t0 + 790, deadline - 100)
    for top_k in [5, 8, 12]:
        if time.time() > p4_end - 20:
            break
        active, _ = get_active_set(best_h, n, top_k=top_k, rel_tol=0.003)
        maxit = max(500, int(25 * 5))
        h, c = run_lbfgs_active(best_h, n, active, 0.0005, maxit)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 5: Smooth L-BFGS at very high alpha - ~60s
    p5_end = min(t0 + 860, deadline - 55)
    if time.time() < p5_end - 10:
        for alpha in [500000, 2000000, 5000000]:
            if time.time() > p5_end - 10:
                break
            maxit = max(300, int(15 * 6))
            h, c = run_lbfgs_smooth(best_h, n, alpha, 0.001, maxit)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Phase 6: Binary local search on the current solution - ~40s
    p6_end = min(t0 + 910, deadline - 30)
    if time.time() < p6_end - 10:
        h_bin = project_feasible(best_h.copy(), n)
        # Force toward binary
        for _ in range(5):
            h_bin[h_bin > 0.7] = 1.0
            h_bin[h_bin < 0.3] = 0.0
            h_bin = project_feasible(h_bin, n)
        c_bin = compute_c5(h_bin, n)
        
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        
        h_bs, c_bs = binary_local_search(best_h, n, p6_end, n_iter=100000, seed=seed+1)
        if c_bs < best_c:
            best_c = c_bs
            best_h = h_bs.copy()
    
    # Phase 7: Final active-set polish - ~25s
    p7_end = deadline - 3
    if time.time() < p7_end - 5:
        active, _ = get_active_set(best_h, n, top_k=5, rel_tol=0.001)
        maxit = max(500, int((p7_end - time.time()) * 4))
        h, c = run_lbfgs_active(best_h, n, active, 0.0003, maxit)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Final projection and verification
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
