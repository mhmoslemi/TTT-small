    """Encourage values to be near 0 or 1."""
    # Penalty: strength * sum(min(h, 1-h)^2)
    # This is 0 at h=0 or h=1, maximum at h=0.5

    return penalty

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

        if delta is None:
            rho *= 0.5
            if rho < 1e-11:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            consecutive_no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.35, 0.12)
        else:
            consecutive_no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if consecutive_no_improve > 40:
                break
    return best_h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 50.0), (1000, 500.0), (5000, 5000.0), (20000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': 80000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_con_fun, 'jac': slqp_con_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def subgradient_refine(h, n, deadline, init_step=0.008):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = init_step
    stall = 0
    for it in range(10000):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_idx = int(np.argmax(c_vec))
        grad = shift_grad_single(h, n, max_idx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        h_new = h - step * grad / gnorm
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1
        step *= 0.999
        if stall > 200:
            step *= 0.7
            stall = 0
    return best_h, best_c

def targeted_local_search(h, n, deadline, center_radius=70):
    best_c = compute_c5(h, n)
    steps = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for idx in active[:15]:
                g = shift_grad_single(h, n, int(idx))
                grad += np.abs(g)
            grad /= min(len(active), 15)
            top_pos = np.argsort(-grad)[:min(center_radius * 2, n)]
            for i_val in top_pos:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def shift_swap_search(h, n, deadline, max_rounds=200):
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        top_k = np.argsort(-c_vec)[:8]
        improved_this_round = False
        for k_idx in top_k:
            k = int(k_idx) - (n - 1)
            if k == 0:
                continue
            if k > 0:
                pairs = min(n - k, 200)
                idxs = np.random.choice(n - k, min(pairs, n - k), replace=False)
                for x in idxs:
                    if h[x] > 0.1 and h[x + k] < 0.9:
                        a = min(h[x], 1.0 - h[x + k]) * 0.5
                        if a < 1e-7:
                            continue
                        h[x] -= a
                        h[x + k] += a
                        h_p = project_feasible(h, n)
                        c = compute_c5(h_p, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            h = h_p
                            improved_this_round = True
                        else:
                            h[x] += a
                            h[x + k] -= a
                    elif h[x] < 0.9 and h[x + k] > 0.1:
                        a = min(1.0 - h[x], h[x + k]) * 0.5
                        if a < 1e-7:
                            continue
                        h[x] += a
                        h[x + k] -= a
                        h_p = project_feasible(h, n)
                        c = compute_c5(h_p, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            h = h_p
                            improved_this_round = True
                        else:
                            h[x] -= a
                            h[x + k] += a
            elif k < 0:
                kk = -k
                pairs = min(n - kk, 200)
                idxs = np.random.choice(n - kk, min(pairs, n - kk), replace=False)
                for x in idxs:
                    if h[x] > 0.1 and h[x + kk] < 0.9:
                        a = min(h[x], 1.0 - h[x + kk]) * 0.5
                        if a < 1e-7:
                            continue
                        h[x] -= a
                        h[x + kk] += a
                        h_p = project_feasible(h, n)
                        c = compute_c5(h_p, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            h = h_p
                            improved_this_round = True
                        else:
                            h[x] += a
                            h[x + kk] -= a
                    elif h[x] < 0.9 and h[x + kk] > 0.1:
                        a = min(1.0 - h[x], h[x + kk]) * 0.5
                        if a < 1e-7:
                            continue
                        h[x] += a
                        h[x + kk] -= a
                        h_p = project_feasible(h, n)
                        c = compute_c5(h_p, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            h = h_p
                            improved_this_round = True
                        else:
                            h[x] -= a
                            h[x + kk] += a
        if not improved_this_round and rnd > 30:
            break
    return h, best_c

def try_periodic_init(n, period, duty=0.5, phase=0.0, smooth_sigma=0.0):
    x = np.linspace(0, 2, n)
    t = (x + phase) % period
    h = np.where(t < period * duty, 1.0, 0.0)
    if smooth_sigma > 0:
        from scipy.ndimage import gaussian_filter1d
        h = gaussian_filter1d(h, smooth_sigma)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: L-BFGS warmup
    t1 = min(t0 + 55, deadline - 850)
    if t1 > time.time() + 3:
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            configs=[(200, 50.0), (1000, 500.0), (5000, 5000.0), (20000, 50000.0)])

    # Phase 2: Main SQP
    t2 = min(t0 + 160, deadline - 650)
    if t2 > time.time() + 5:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=600, rho_init=0.05)

    # Phase 3: Subgradient refinement
    t3 = min(t0 + 200, deadline - 550)
    if t3 > time.time() + 3:
        h_sg, c_sg = subgradient_refine(best_h, n_orig, t3, init_step=0.006)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    # Phase 4: SLSQP
    t4 = min(t0 + 270, deadline - 450)
    if t4 > time.time() + 5:
        h4, c4 = slqp_polish(best_h, n_orig, t4, alpha=3000000, maxiter=4000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Targeted local search
    t5 = min(t0 + 360, deadline - 350)
    if t5 > time.time() + 3:
        best_h, best_c = targeted_local_search(best_h, n_orig, t5, center_radius=70)

    # Phase 6: SQP round 2
    t6 = min(t0 + 420, deadline - 300)
    if t6 > time.time() + 5:
        best_h, best_c = sqp_optimize(best_h, n_orig, t6, max_iter=400, rho_init=0.025)

    # Phase 7: Shift-swap search
    t7 = min(t0 + 480, deadline - 250)
    if t7 > time.time() + 3:
        best_h, best_c = shift_swap_search(best_h, n_orig, t7, max_rounds=300)

    # Phase 8: Upsample to n=1000
    n_high = 1000
    if time.time() < deadline - 200:
        x_old = np.linspace(0, 2, n_orig, endpoint=False)
        x_new = np.linspace(0, 2, n_high, endpoint=False)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)

        t8 = min(time.time() + 55, deadline - 120)
        if t8 > time.time() + 5:
            h_high, c_high = run_lbfgs(h_high, n_high, t8,
                configs=[(2000, 500.0), (10000, 5000.0), (40000, 50000.0)])

        t9 = min(time.time() + 50, deadline - 60)
        if t9 > time.time() + 5:
            h_high, c_high = sqp_optimize(h_high, n_high, t9, max_iter=200, rho_init=0.04)

        t10 = min(time.time() + 35, deadline - 25)
        if t10 > time.time() + 3:
            h_high, c_high = targeted_local_search(h_high, n_high, t10, center_radius=40)

        t11 = min(time.time() + 20, deadline - 10)
        if t11 > time.time() + 3:
            h_high, c_high = shift_swap_search(h_high, n_high, t11, max_rounds=100)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_orig = n_high

    # Phase 9: Try a periodic init if time permits
    if time.time() < deadline - 80:
        for period in [0.4, 0.5, 1.0/3.0]:
            if time.time() > deadline - 20:
                break
            h_p = try_periodic_init(n_orig, period, 0.5, 0.1)
            c_p = compute_c5(h_p, n_orig)
            if c_p < best_c + 0.03:
                t_rem = min(time.time() + 25, deadline - 10)
                if t_rem > time.time() + 3:
                    h_p, c_p = run_lbfgs(h_p, n_orig, t_rem,
                        configs=[(5000, 5000.0), (20000, 50000.0)])
                    if c_p < best_c:
                        best_c = c_p
                        best_h = h_p

    # Phase 10: Final SQP + polish on global best
    if time.time() < deadline - 20:
        h_f, c_f = sqp_optimize(best_h, n_orig, min(time.time() + 20, deadline - 10),
                                 max_iter=200, rho_init=0.015)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    if time.time() < deadline - 8:
        h_f, c_f = slqp_polish(best_h, n_orig, deadline - 3, alpha=8000000, maxiter=3000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
