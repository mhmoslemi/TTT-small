            new_c = c_arr[None, :] + A_i[None, :] - A_j
            new_max = new_c.max(axis=1)
            bi = int(new_max.argmin())
            if new_max[bi] < c_arr.max() - 1e-12:
                j = int(j_arr[bi])
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                ones.discard(i)
                ones.add(j)
                zeros.add(i)
                zeros.discard(j)
                c_arr = np.correlate(h_bin, 1 - h_bin, 'full')
                bc = c_arr.max() * (2.0 / n)
                if bc < best_c:
                    best_c = bc
                improved = True
        if not improved:
            break
    return h_bin, best_c

def simulated_annealing(h, n, deadline, max_iter=20000):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    c_arr = np.correlate(h_bin, 1 - h_bin, 'full')
    best_c = c_arr.max() * (2.0 / n)
    best_h = h_bin.copy()
    rng = np.random.RandomState(42)
    t_start = 5.0
    t_end = 0.1
    n_ones = np.where(h_bin == 1)[0].copy()
    n_zeros = np.where(h_bin == 0)[0].copy()
    t0 = time.time()
    total_time = max(1.0, deadline - t0)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = it / max(1, max_iter)
        temp = t_start * (t_end / t_start) ** frac
        i_idx = rng.randint(len(n_ones))
        j_idx = rng.randint(len(n_zeros))
        i = n_ones[i_idx]
        j = n_zeros[j_idx]
        h_bin[i], h_bin[j] = 0.0, 1.0
        c_arr = np.correlate(h_bin, 1 - h_bin, 'full')
        new_max = c_arr.max() * (2.0 / n)
        delta = new_max - compute_c5(h_bin, n)
        if new_max < best_c or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            n_ones[i_idx] = j
            n_zeros[j_idx] = i
            if new_max < best_c:
                best_c = new_max
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = 1.0, 0.0
    return best_h, best_c

def multi_alpha_slsqp(h, n, deadline, alphas, maxiter_each):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for a in alphas:
        if time.time() > deadline:
            break
        hh = slsqp_smooth(best_h, n, a, maxiter_each, deadline)
        c = compute_c5(hh, n)
        if c < best_c:
            best_c = c
            best_h = hh.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    best_n = n
    p1 = t0 + 0.30 * (budget_s - 5)
    h_work = best_h.copy()
    for _ in range(15):
        if time.time() > p1:
            break
        h_work, c = subgrad_optimize(h_work, n, p1, max_iter=30000)
        if c < best_c:
            best_c = c
            best_h = h_work.copy()
    if time.time() < dl - 100:
        for nh in [400, 500]:
            if time.time() > dl - 60:
                break
            xo = np.linspace(0, 2, best_n, endpoint=False)
            xn = np.linspace(0, 2, nh, endpoint=False)
            hi = np.interp(xn, xo, best_h)
            hi = project_feasible(hi, nh)
            h_work2 = hi.copy()
            sub_deadline = min(dl - 40, time.time() + 180)
            h_work2, c2 = subgrad_optimize(h_work2, nh, sub_deadline, max_iter=50000)
            if c2 < best_c:
                best_c = c2
                best_h = h_work2.copy()
                best_n = nh
            h_work2, c2 = multi_alpha_slsqp(h_work2, nh, min(dl - 30, time.time() + 90),
                                            [50, 200, 1000, 5000], 400)
            if c2 < best_c:
                best_c = c2
                best_h = h_work2.copy()
                best_n = nh
    if time.time() < dl - 80:
        h_work3 = best_h.copy()
        h_work3, c3 = subgrad_optimize(h_work3, best_n, min(dl - 50, time.time() + 120),
                                       max_iter=50000)
        if c3 < best_c:
            best_c = c3
            best_h = h_work3.copy()
    if time.time() < dl - 60:
        h_work4 = best_h.copy()
        h_work4, c4 = multi_alpha_slsqp(h_work4, best_n, min(dl - 20, time.time() + 100),
                                        [100, 500, 2000, 10000, 50000], 600)
        if c4 < best_c:
            best_c = c4
            best_h = h_work4.copy()
    if time.time() < dl - 40:
        hb, cb = binary_local_search(best_h, best_n, min(dl - 15, time.time() + 120),
                                     max_sweeps=300)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    if time.time() < dl - 20:
        hp = project_feasible(best_h + 0.15 * rng.randn(best_n), best_n)
        hb2, cb2 = binary_local_search(hp, best_n, min(dl - 10, time.time() + 60),
                                       max_sweeps=100)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    if time.time() < dl - 10:
        h_work5 = best_h.copy()
        h_work5, c5 = subgrad_optimize(h_work5, best_n, dl - 5, max_iter=20000)
        if c5 < best_c:
            best_c = c5
            best_h = h_work5.copy()
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
