```python
def coord_descent(h, n, max_sweeps=100, time_budget=None):
    import time
    best_c = compute_c5(h, n)
    n_changed = 0
    for sweep in range(max_sweeps):
        if time_budget and time.time() > time_budget:
            break
        n_changed = 0
        for j in range(n):
            old_val = h[j]
            new_val = solve_1d_sub(h, n, j)
            if abs(new_val - old_val) > 1e-12:
                h[j] = new_val
                n_changed += 1
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
        if n_changed == 0:
            break
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_coord(h, n, j):
    """Solve 1D subproblem: min_t max_m (A_m + B_m*t) for coordinate j."""
    M = 2 * n - 1
    raw_corr = np.correlate(h, 1 - h, 'full')
    
    # Compute B vector
    B = np.zeros(M)
    
    # For m = n-1+k, k=0,...,n-1:
    # B[n-1+k] = (1-h[j+k]) * [k <= n-1-j] + (-h[j-k]) * [k <= j]
    ks = np.arange(n)
    valid1 = ks <= (n - 1 - j)
    valid2 = ks <= j
    B[n-1:] = np.where(valid1, 1 - h[j + ks], 0) + np.where(valid2, -h[j - ks], 0)
    
    # For m = n-1-kk, kk=1,...,n-1:
    # B[n-1-kk] = (1-h[j-kk]) * [kk <= j] + (-h[j+kk]) * [kk <= n-1-j]
    kks = np.arange(1, n)
    valid3 = kks <= j
    valid4 = kks <= (n - 1 - j)
    B[:n-1] = np.where(valid3, 1 - h[j - kks], 0) + np.where(valid4, -h[j + kks], 0)
    
    A = raw_corr - B * h[j]
    
    # Minimize max(A + B*t) over t in [0,1]
    # Use sampling + refinement
    n_sample = 201
    ts = np.linspace(0, 1, n_sample)
    vals = A[None, :] + B[:, None] * ts[None, :]
    f_vals = vals.max(axis=1)
    min_idx = np.argmin(f_vals)
    t_star = ts[min_idx]
    f_star = f_vals[min_idx]
    
    # Refine: find the exact kink point
    if 0 < min_idx < n_sample - 1:
        # Get lines active within tolerance of the max at the minimum
        tol = 1e-10 * max(1.0, abs(f_star))
        active = np.where(vals[:, min_idx] >= f_star - tol)[0]
        if len(active) >= 2:
            # Check all pairs
            a = A[active]
            b = B[active]
            for i in range(len(active)):
                for j2 in range(i + 1, len(active)):
                    if abs(b[i] - b[j2]) > 1e-15:
                        t_int = (a[j2] - a[i]) / (b[i] - b[j2])
                        if 0 <= t_int <= 1:
                            f_int = a[i] + b[i] * t_int
                            if f_int < f_star - 1e-15:
                                f_star = f_int
                                t_star = t_int
    
    return np.clip(t_star, 0.0, 1.0)

def coord_descent(h, n, deadline, verbose=False):
    """Run coordinate descent until deadline."""
    best_c = compute_c5(h, n)
    h = h.copy()
    
    while time.time() < deadline:
        improved = False
        for j in range(n):
            new_val = solve_coord(h, n, j)
            if abs(new_val - h[j]) > 1e-13:
                h[j] = new_val
                improved = True

        if c < best_c:
            best_c = c
            if verbose:
                print(f"  CD sweep: C5 = {best_c:.8f}")
        elif not improved:
            break
        # Check if we've converged
        if not improved:
            break
    
    return h, best_c
```

The function f(t) = max_m (A_m + B_m*t) is convex and piecewise linear on [0,1]. Its minimum is at:

- The minimum of max of lines is the "lower intersection point" of the upper envelope
- This can be found by: sort lines by A_m (value at t=0). The upper envelope at t=0 is determined by the line with max A_m. As t increases, lines with larger B_m overtake it.
- The minimum is where the "transition" happens.

Efficient algorithm:
1. f(0) = max A_m. Let m0 = argmax A_m.
2. f(1) = max (A_m + B_m). Let m1 = argmax (A_m + B_m).
3. If m0 == m1: f is constant, return h[j].
4. If f'(0+) > 0 (i.e., B of all lines active at t=0 are > 0): min at t=0.
5. If f'(1-) < 0 (i.e., B of all lines active at t=1 are < 0): min at t=1.
6. Otherwise, the min is at the intersection of the upper envelope. The upper envelope goes from m0 (at t=0) to m1 (at t=1). The minimum is at the point where the slope changes from negative to positive.

- The minimum is where 0 ∈ subgradient

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_bisect(A, B):
    """Solve min_t max_m (A[m] + B[m]*t) over t in [0,1] via bisection."""
    f0 = A.max()
    f1 = (A + B).max()
    
    # Check boundaries
    active_0 = np.where(A >= f0 - 1e-12)[0]
    if len(active_0) > 0 and B[active_0].min() > 1e-14:
        return 0.0
    active_1 = np.where(A + B >= f1 - 1e-12)[0]
    if len(active_1) > 0 and B[active_1].max() < -1e-14:
        return 1.0
    
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        vals = A + B * mid
        fmid = vals.max()
        active = np.where(vals >= fmid - 1e-12)[0]
        b_min = B[active].min()
        b_max = B[active].max()
        if b_min > 1e-14:
            lo = mid
        elif b_max < -1e-14:
            hi = mid
        else:
            return mid
    return 0.5 * (lo + hi)

def compute_B(h, n, j):
    """Compute the B vector for coordinate j."""
    M = 2 * n - 1
    B = np.zeros(M)
    
    # For m = n-1+k, k=0,...,n-1:
    ks = np.arange(n)
    B[n-1:] = np.where(ks <= (n-1-j), 1.0 - h[j + ks], 0.0) + \
               np.where(ks <= j, -h[j - ks], 0.0)
    
    # For m = n-1-kk, kk=1,...,n-1:
    kks = np.arange(1, n)
    B[:n-1] = np.where(kks <= j, 1.0 - h[j - kks], 0.0) + \
              np.where(kks <= (n-1-j), -h[j + kks], 0.0)
    
    return B

def coord_solve(h, n, j):
    """Solve the 1D subproblem for coordinate j."""
    raw_corr = np.correlate(h, 1 - h, 'full')
    B = compute_B(h, n, j)
    A = raw_corr - B * h[j]
    t_star = solve_1d_bisect(A, B)
    return t_star

def coord_descent(h, n, deadline, max_sweeps=5000):
    """Coordinate descent with projection."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        changed = 0
        for j in range(n):
            new_val = coord_solve(h, n, j)
            if abs(new_val - h[j]) > 1e-13:
                h[j] = new_val
                changed += 1
        
        # Project to satisfy sum constraint
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
        
        if changed < n * 0.01:  # Very few changes → converged
            # Try one more sweep to be sure
            changed2 = 0
            for j in range(n):
                new_val = coord_solve(h, n, j)
                if abs(new_val - h[j]) > 1e-13:
                    h[j] = new_val
                    changed2 += 1
            if changed2 < 5:
                h = project_feasible(h, n)
                best_c = compute_c5(h, n)
                break
    
    return h, best_c
```

def two_point_moves(h, n, deadline, max_iters=1000):
    """Guided 2-point moves to escape flat regions."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    M = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Active shifts (within 1% of max)
        tol = 0.01 * max_val
        active = np.where(c_vec >= max_val - tol)[0]
        
        # Compute subgradient (average of active shift gradients)
        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n])
                grad[k:n] += (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv])
                grad[0:nv] += (-h[kk:n])

        # Want this to be negative
