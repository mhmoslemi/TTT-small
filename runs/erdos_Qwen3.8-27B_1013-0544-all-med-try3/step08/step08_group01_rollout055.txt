def coordinate_descent(h, n, deadline, max_steps=50000):

    best_h = h.copy()

        if time.time() > deadline:
            break
        
        improved = False
        # Try random coordinates
        for _ in range(50):
            i = np.random.randint(n)
            # Try adjusting h[i] up or down by a small amount
            for delta in [0.001, 0.0005, 0.0001, -0.001, -0.0005, -0.0001]:
                new_val = h[i] + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                # Project to maintain integral
                h_proj = project_feasible(h, n)
                c = compute_c5(h_proj, n)
                h = h_proj
                if c < best_c - 1e-14:

                    best_h = h.copy()
                    improved = True
                    break
                # Revert
                h[i] = best_h[i] if abs(new_val - best_h[i]) > 1e-12 else new_val
        
        if not improved and step % 1000 == 0:
            # Try pair moves
            for _ in range(20):
                i = np.random.randint(n)
                j = np.random.randint(n)

                    if h[j] - delta < 0 or h[j] - delta > 1:

                    h[i] += delta
                    h[j] -= delta
                    h_proj = project_feasible(h, n)
                    c = compute_c5(h_proj, n)
                    h = h_proj
                    if c < best_c - 1e-14:
                        best_c = c

                        improved = True
                        break
    
    return best_h, best_c

The correlation at shift k (in samples) is:
c_k = dx * sum_j h[j] * (1 - h[(j-k) mod n]) ... wait, no, it's not periodic.

Actually, np.correlate(h, 1-h, mode='full') at index m (0 to 2n-2) computes:
sum_j h[j] * (1-h[j - m + (n-1)])

where the product is only taken when both indices are in [0, n-1].

So at m = n-1 (shift 0): sum_j h[j]*(1-h[j]) = sum h[j] - sum h[j]^2
At m = n-1+k (shift k): sum_{j=k}^{n-1} h[j]*(1-h[j-k]) for k ≥ 0

At the optimum, we want all correlate[m] * dx to be as low as possible, with the maximum being minimized.

For the shift-0 term: c_0 = dx * sum h[j]*(1-h[j]) = dx * (n/2 - sum h[j]^2)
To minimize c_0, we want to maximize sum h[j]^2, which means making h as "spiky" as possible (values close to 0 or 1).

For small shifts (|k| = 1, 2, ...): the overlap is large, and we need to arrange h so that h[j] and h[j-1] are "opposed" (one is 0, the other is 1).

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

                    if direction == -1:
                        if best_h[i] - s >= 0 and best_h[j] + s <= 1:
                            best_h[i] -= s
                            best_h[j] += s
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                found = True
                            else:
                                best_h[i] += s
                                best_h[j] -= s
                    else:
                        if best_h[i] + s <= 1 and best_h[j] - s >= 0:
                            best_h[i] += s
                            best_h[j] -= s
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                found = True
                            else:
                                best_h[i] -= s
                                best_h[j] += s

        if not found:
            step *= 0.5
            if step < 1e-7:
                break

    return best_h, best_c

def multiresolution_refine(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for n_coarse in [n // 2, n // 4]:
        if n_coarse < 50 or time.time() > deadline - 10:
            continue
        x_fine = np.linspace(0, 2, n)
        x_coarse = np.linspace(0, 2, n_coarse)
        h_c = np.interp(x_coarse, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)

        t_c = min(40.0, (deadline - time.time()) * 0.3)
        if t_c > 15:
            for alpha in [5000, 50000, 200000]:
                if time.time() > deadline - 10:
                    break
                h_c = lbfgs_full(h_c, n_coarse, alpha, maxiter=3000)

        h_f = np.interp(x_fine, x_coarse, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    return best_h, best_c

def perturb_and_reopt(h, n, deadline, n_perturb=5, scale=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for p in range(n_perturb):
        if time.time() > deadline - 30:
            break
        h_p = best_h.copy()
        mask = np.random.random(n) < 0.3
        h_p[mask] += np.random.randn(mask.sum()) * scale
        h_p = project_feasible(h_p, n)

        h_p = lbfgs_full(h_p, n, 50000, maxiter=3000)
        c_p = compute_c5(h_p, n)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Full L-BFGS with increasing alpha
    if time.time() < deadline - 500:
        for alpha in [2000, 10000, 50000, 200000]:
            if time.time() > deadline - 450:
                break
            best_h = lbfgs_full(best_h, n_orig, alpha, maxiter=5000)
            c = compute_c5(best_h, n_orig)
            if c < best_c:
                best_c = c

    # Phase 2: Active-shift focused with high alpha
    if time.time() < deadline - 350:
        for alpha in [100000, 500000, 2000000, 10000000]:
            if time.time() > deadline - 300:
                break
            active, _ = find_active_shifts(best_h, n_orig, frac=0.005)
            best_h = lbfgs_active(best_h, n_orig, alpha, active, maxiter=5000)
            c = compute_c5(best_h, n_orig)
            if c < best_c:
                best_c = c

    # Phase 3: Narrow active set (top 3)
    if time.time() < deadline - 250:
        c_vec = compute_c5_vec(best_h, n_orig)
        top3 = np.argsort(-c_vec)[:max(3, len(c_vec) // 100)]
        for alpha in [500000, 5000000]:
            if time.time() > deadline - 200:
                break
            best_h = lbfgs_active(best_h, n_orig, alpha, top3, maxiter=5000)
            c = compute_c5(best_h, n_orig)
            if c < best_c:
                best_c = c

    # Phase 4: Subgradient refinement with exact max
    if time.time() < deadline - 180:
        t4 = min(time.time() + 120, deadline - 180)
        best_h, best_c = subgradient_refine(best_h, n_orig, t4, lr_init=0.002, max_iter=15000)

    # Phase 5: Multi-resolution refinement
    if time.time() < deadline - 150:
        t5 = min(time.time() + 80, deadline - 150)
        best_h, best_c = multiresolution_refine(best_h, n_orig, t5)

    # Phase 6: Try n=400
    if time.time() < deadline - 120:
        n_alt = 400
        x_best = np.linspace(0, 2, len(best_h))
        x_alt = np.linspace(0, 2, n_alt)
        h_alt = np.interp(x_alt, x_best, best_h)
        h_alt = project_feasible(h_alt, n_alt)
        c_alt = compute_c5(h_alt, n_alt)

        for alpha in [50000, 200000, 1000000]:
            if time.time() > deadline - 80:
                break
            h_alt = lbfgs_full(h_alt, n_alt, alpha, maxiter=4000)
            c_alt = compute_c5(h_alt, n_alt)

        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt

    # Phase 7: Try n=600
    if time.time() < deadline - 100:
        n_alt2 = 600
        x_b = np.linspace(0, 2, len(best_h))
        x_a2 = np.linspace(0, 2, n_alt2)
        h_a2 = np.interp(x_a2, x_b, best_h)
        h_a2 = project_feasible(h_a2, n_alt2)
        c_a2 = compute_c5(h_a2, n_alt2)

        for alpha in [50000, 200000]:
            if time.time() > deadline - 60:
                break
            h_a2 = lbfgs_full(h_a2, n_alt2, alpha, maxiter=3000)
            c_a2 = compute_c5(h_a2, n_alt2)

        if c_a2 < best_c:
            best_c = c_a2
            best_h = h_a2

    # Phase 8: Perturbation + re-optimization to escape local minima
    if time.time() < deadline - 80:
        n_best = len(best_h)
        t8 = min(time.time() + 50, deadline - 80)
        best_h, best_c = perturb_and_reopt(best_h, n_best, t8, n_perturb=4, scale=0.008)

    # Phase 9: Final active-shift high-alpha polish
    if time.time() < deadline - 50:
        n_best = len(best_h)
        for alpha in [2000000, 10000000, 50000000]:
            if time.time() > deadline - 25:
                break
            active, _ = find_active_shifts(best_h, n_best, frac=0.003)
            best_h = lbfgs_active(best_h, n_best, alpha, active, maxiter=5000)
            c = compute_c5(best_h, n_best)
            if c < best_c:
                best_c = c

    # Phase 10: Subgradient final
    if time.time() < deadline - 30:
        t10 = min(time.time() + 25, deadline - 30)
        best_h, best_c = subgradient_refine(best_h, len(best_h), t10, lr_init=0.001, max_iter=8000)

    # Phase 11: Gradient-coordinate refinement
    if time.time() < deadline - 15:
        t11 = min(time.time() + 12, deadline - 15)
        best_h, best_c = gradient_coordinate(best_h, len(best_h), t11, base_step=0.001)

    # Phase 12: Fine pair search
    if time.time() < deadline - 5:
        t12 = deadline - 3
        best_h, best_c = fine_pair_search(best_h, len(best_h), t12, radius=30)

    # Final
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
