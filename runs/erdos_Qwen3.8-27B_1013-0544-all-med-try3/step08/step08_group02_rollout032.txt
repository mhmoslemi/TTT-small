def project_feasible(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    if target is None:
        target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    # proportional corrections
    for _ in range(60):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = float(np.sum(room))
            if tot <= 1e-15:
                break
            # add in proportion to room
            h += diff * (room / tot)
        else:
            tot = s
            if tot <= 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    # exact point adjustment loop
    for _ in range(10):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-14:
            break
        val = target - (s - h)
        good = (val >= -1e-12) & (val <= 1.0 + 1e-12)
        if np.any(good):
            idx = np.where(good)[0]
            # pick one with smallest adjustment
            changes = np.abs(val[idx] - h[idx])
            i = idx[int(np.argmin(changes))]
            h[i] = float(np.clip(val[i], 0.0, 1.0))
            continue
        # fallback greedy
        rem = diff
        if rem > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                if rem <= 0: break
                add = min(1.0 - h[i], rem)
                h[i] += add
                rem -= add
        else:
            order = np.argsort(h)
            for i in order:
                if rem >= 0: break
                sub = min(h[i], -rem)
                h[i] -= sub
                rem += sub
        h = np.clip(h, 0.0, 1.0)
    return h

class _OptTimeout(Exception):

def _timeout_check():
    if time.time() > TIME_DEADLINE[0]:

def project_feasible(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    n = int(h.size)
    if target is None:
        target = 0.5 * float(n)

    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)

    for _ in range(60):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(np.sum(room))
            if tot <= 1e-15:
                break
            h += diff * (room / tot)
        else:
            tot = s
            if tot <= 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)

    for _ in range(12):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-14:
            break

        val = target - (s - h)
        good = (val >= -1e-12) & (val <= 1.0 + 1e-12)
        if np.any(good):
            idx = np.where(good)[0]
            changes = np.abs(val[idx] - h[idx])
            i = int(idx[int(np.argmin(changes))])
            h[i] = float(np.clip(val[i], 0.0, 1.0))
            continue

        rem = diff
        if rem > 0.0:
            order = np.argsort(1.0 - h)
            for i in order:
                if rem <= 0.0:
                    break
                add = min(1.0 - h[i], rem)
                h[i] += add
                rem -= add
        else:
            order = np.argsort(h)
            for i in order:
                if rem >= 0.0:
                    break
                sub = min(h[i], -rem)
                h[i] -= sub
                rem += sub
        h = np.clip(h, 0.0, 1.0)

    return h


def full_c_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_c5(h, n):
    c = full_c_vec(h, n)
    if not np.all(np.isfinite(c)):
        return 1e100
    return float(np.max(c))


def active_values(x, active, n, dx, off):
    out = np.empty(active.size, dtype=float)
    one_minus = 1.0 - x
    for t in range(active.size):
        idx = int(active[t])
        k = idx - off
        if k >= 0:
            nv = n - k
            out[t] = np.dot(x[:nv], one_minus[k:n])
        else:
            kk = -k
            nv = n - kk
            out[t] = np.dot(x[kk:n], one_minus[:nv])
    return out * dx


def smooth_max_value(vals, alpha):
    vals = np.asarray(vals, dtype=float)
    if vals.size == 0:
        return float('nan')
    if not np.all(np.isfinite(vals)):
        return float('nan')
    if alpha <= 0.0:
        return float(np.mean(vals))
    m = float(np.max(vals))
    sh = alpha * (vals - m)
    s = float(np.sum(np.exp(sh)))
    return float(m + np.log(s) / alpha)


def active_obj(x, active, n, dx, off, target, mu, alpha):
    _timeout_check()
    vals = active_values(x, active, n, dx, off)
    if not np.all(np.isfinite(vals)):
        return 1e100
    val = smooth_max_value(vals, alpha)
    if not np.isfinite(val):
        return 1e100
    if mu > 0.0:
        diff = float(np.sum(x) - target)
        val += mu * diff * diff
    return float(val)


def active_grad(x, active, n, dx, off, target, mu, alpha):
    _timeout_check()
    vals = active_values(x, active, n, dx, off)
    if not np.all(np.isfinite(vals)):
        return np.zeros(n, dtype=float)

    m = float(np.max(vals))
    if alpha <= 0.0:
        w = np.full(vals.size, 1.0 / float(vals.size))
    else:
        sh = alpha * (vals - m)
        w = np.exp(sh)
        ws = float(np.sum(w))
        if ws <= 1e-300 or not np.isfinite(ws):
            w = np.full(vals.size, 1.0 / float(vals.size))
        else:
            w = w / ws

    grad = np.zeros(n, dtype=float)
    for t in range(active.size):
        wi = float(w[t])
        if wi <= 1e-300:
            continue
        idx = int(active[t])
        wm = wi * dx
        k = idx - off
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += -wm * x[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[:nv])
            grad[:nv] += -wm * x[kk:n]

    if mu > 0.0:
        grad += 2.0 * mu * (float(np.sum(x)) - target)
    return grad


def _unique_array(vals):
    out = []
    seen = set()
    for v in vals:
        iv = int(v)
        if iv not in seen:
            seen.add(iv)
            out.append(iv)
    if not out:
        return np.empty(0, dtype=np.int64)
    return np.asarray(out, dtype=np.int64)


def select_active(c, K):
    c = np.asarray(c, dtype=float)
    if c.size == 0:
        return np.empty(0, dtype=np.int64)
    if not np.all(np.isfinite(c)):
        c = c.copy()
        c[~np.isfinite(c)] = 0.0

    m = float(np.max(c))
    tol = max(1e-8, 5e-6 + 5e-5 * max(0.0, m))
    idx = np.where(c >= m - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=np.int64)

    order = np.argsort(c[idx])[::-1]
    top = idx[order[:K]].astype(np.int64, copy=False)

    off = (int(c.size) - 1) // 2
    mirrors = []
    mt = max(3, int(K) // 4)
    for idx2 in top[:mt]:
        mirror = off - (int(idx2) - off)
        if 0 <= mirror < c.size:
            mirrors.append(mirror)

    combined = _unique_array(list(top) + list(mirrors))
    cap = int(int(K) * 1.25) + 1
    if combined.size > cap:
        o = np.argsort(c[combined])[::-1]
        combined = combined[o[:cap]]
    else:
        o = np.argsort(c[combined])[::-1]
        combined = combined[o]
    return combined


def resample_to(h, n):
    h = np.asarray(h, dtype=float)
    m = int(h.size)
    if m == int(n):
        return h.copy()
    if m == 1:
        return np.full(int(n), float(h[0]), dtype=float)

    old_x = np.arange(m, dtype=float)
    new_x = np.linspace(0.0, float(m - 1), int(n))
    return np.interp(new_x, old_x, h)


def make_binary(from_h, n):
    h = resample_to(from_h, int(n))
    h = project_feasible(h, int(n))
    target = 0.5 * float(int(n))
    target_int = int(np.floor(target + 1e-12))

    idx = np.argsort(-h, kind="mergesort")
    b = np.zeros(int(n), dtype=float)
    b[idx[:target_int]] = 1.0

    rem = target - float(np.sum(b))
    pos = target_int
    while rem > 1e-12 and pos < int(n):
        j = int(idx[pos])
        add = min(1.0 - b[j], rem)
        b[j] += add
        rem -= add
        pos += 1

    while rem < -1e-12 and pos >= 0:
        j = int(idx[pos])
        sub = min(b[j], -rem)
        b[j] -= sub
        rem += sub
        pos -= 1

    return project_feasible(b, int(n))


def sort_starts(starts):
    arr = list(starts)
    for i in range(len(arr)):
        best = i
        for j in range(i + 1, len(arr)):
            if arr[j][0] < arr[best][0]:
                best = j
        if best != i:
            arr[i], arr[best] = arr[best], arr[i]
    return arr

def lag_transfer_refine(h_start, n, time_end, K=16, samples_per_lag=8):
    h = project_feasible(h_start, int(n))
    c = exact_c5(h, int(n))
    n = int(n)
    off = n - 1

    if time.time() >= float(time_end) - 2.0 or not np.isfinite(c):
        return h, c

    for _ in range(30):
        if time.time() >= float(time_end) - 2.0:
            break

        cvec = full_c_vec(h, n)
        active = select_active(cvec, K)
        if active.size == 0:
            break

        improved = False
        for idx in active:
            if time.time() >= float(time_end) - 2.0:
                break

            k = int(idx) - off
            if k == 0:
                continue

            if k > 0:
                nv = n - k
                shift = k
            else:
                kk = -k
                nv = n - kk
                shift = kk

            if nv <= 0:
                continue

            step = max(1, nv // int(samples_per_lag))
            for s in range(0, nv, step):
                if time.time() >= float(time_end) - 2.0:
                    break

                if k > 0:
                    a = s
                    b = s + shift
                else:
                    b = s
                    a = s + shift

                diff_val = float(h[a] - h[b])
                if diff_val <= 1e-10:
                    continue

                base_t = min(0.04, 0.5 * diff_val)
                if base_t <= 1e-8:
                    continue

                for frac in (1.0, 0.25, 0.10, 0.04):
                    t = base_t * frac
                    if t <= 1e-8:
                        continue

                    h2 = h.copy()
                    h2[a] -= t
                    h2[b] += t
                    h2 = project_feasible(h2, n)
                    c2 = exact_c5(h2, n)
                    if np.isfinite(c2) and c2 < c - 1e-13:
                        h = h2
                        c = c2
                        improved = True
                        break

                if improved:
                    break

            if improved:
                break

        if not improved:
            break

    h_final = project_feasible(h, n)
    c_final = exact_c5(h_final, n)
    if np.isfinite(c_final) and c_final < c:
        c = c_final
    return h_final, float(c_final if np.isfinite(c_final) else c)

    high_count = 5
    no_accept = 0

    for step in range(8000):
        if time.time() >= float(time_end) - 1.0:
            break

        if step % 25 == 0:
            active = select_active(full_c_vec(h, n), K)
            if active.size == 0:
                break

        vals = active_values(h, active, n, dx, off)
        if not np.all(np.isfinite(vals)):
            break
        base = smooth_max_value(vals, alpha)
        if not np.isfinite(base):
            break

        try:
            g = active_grad(h, active, n, dx, off, target, 0.0, alpha)
        except _OptTimeout:
            break
        except Exception:
            break

        if not np.all(np.isfinite(g)):
            break

        order = np.argsort(g)
        low = order[:min(low_count, n)]
        high = order[-min(high_count, n):][::-1]

        accepted = False
        for j in high:
            if accepted:
                break
            if h[j] <= 1e-12:
                continue
            for i in low:
                if i == j:
                    continue
                room = 1.0 - h[i]
                if room <= 1e-12:
                    continue
                maxt = min(float(h[j]), float(room))
                if maxt <= 1e-10:
                    continue

                for frac in (0.10, 0.02, 0.005, 0.001):
                    t = frac * maxt
                    if t < 1e-9:
                        continue

                    h2 = h.copy()
                    h2[i] += t
                    h2[j] -= t

                    vals2 = active_values(h2, active, n, dx, off)
                    val2 = smooth_max_value(vals2, alpha)
                    if np.isfinite(val2) and val2 < base - 1e-10:
                        h2 = project_feasible(h2, n)
                        c2 = exact_c5(h2, n)
                        if np.isfinite(c2) and c2 < c - 1e-13:
                            h = h2
                            c = c2
                            accepted = True
                            break

                if accepted:
                    break

        if accepted:
            no_accept = 0
        else:
            no_accept += 1
            if no_accept > 100 and no_accept % 20 == 0 and time.time() < float(time_end) - 5.0:
                rngjit = np.random.RandomState(1000 + step)
                h2 = project_feasible(h + 0.003 * rngjit.normal(size=n), n)
                c2 = exact_c5(h2, n)
                if np.isfinite(c2) and c2 < c - 1e-13:
                    h = h2
                    c = c2
                    no_accept = 0
            if no_accept > 250:
                break

    TIME_DEADLINE[0] = float('inf')
    h_final = project_feasible(h, n)
    c_final = exact_c5(h_final, n)
    if np.isfinite(c_final) and c_final < c:
        c = c_final
    return h_final, float(c_final if np.isfinite(c_final) else c)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    b = float(budget_s)
    final_end = t0 + b - 10.0
    if final_end < t0 + 30.0:
        final_end = t0 + max(5.0, b - 2.0)

    rng = np.random.RandomState(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = np.full(500, 0.5, dtype=float)

    if parent.size == 0:
        parent = np.full(500, 0.5, dtype=float)
    parent = parent.copy()
    parent[~np.isfinite(parent)] = 0.5
    parent = np.clip(parent, 0.0, 1.0)

    n_main = int(parent.size)
    if n_main < 250 or n_main > 2200:
        n_main = 500
    if n_main % 2 != 0:
        if n_main + 1 <= 2200:
            n_main += 1
        else:
            n_main -= 1
    n_main = max(2, n_main)

    res_list = []
    add_resolution(res_list, n_main, 250, 2200)
    add_resolution(res_list, 2 * n_main, 300, 2200)
    add_resolution(res_list, 4 * n_main, 300, 2200)
    add_resolution(res_list, 1000, 300, 2200)
    add_resolution(res_list, 2000, 300, 2200)
    add_resolution(res_list, 1500, 300, 2200)
    add_resolution(res_list, 1200, 300, 2200)
    add_resolution(res_list, 1800, 300, 2200)

    if b < 180.0:
        h0 = project_feasible(resample_to(parent, n_main), n_main)
        c0 = exact_c5(h0, n_main)
        if not np.isfinite(c0):
            h0 = project_feasible(np.full(500, 0.5), 500)
            c0 = exact_c5(h0, 500)
            n_main = 500
        return h0, float(c0), int(n_main)

    h0 = project_feasible(resample_to(parent, n_main), n_main)
    c0 = exact_c5(h0, n_main)

    h1 = make_binary(parent, n_main)
    c1 = exact_c5(h1, n_main)

    h2 = make_binary(project_feasible(h0 + 0.20 * rng.normal(size=n_main), n_main), n_main)
    c2 = exact_c5(h2, n_main)

    h3 = make_binary(project_feasible(h0 + 0.50 * rng.normal(size=n_main), n_main), n_main)
    c3 = exact_c5(h3, n_main)

    starts = []
    if np.isfinite(c0):
        starts.append((float(c0), h0, n_main, 0))
    if np.isfinite(c1):
        starts.append((float(c1), h1, n_main, 1))
    if np.isfinite(c2):
        starts.append((float(c2), h2, n_main, 2))
    if np.isfinite(c3):
        starts.append((float(c3), h3, n_main, 3))

    if not starts:
        starts = [(0.5, h0, n_main, 99)]

    starts = sort_starts(starts)

    best_h = starts[0][1].copy()
    best_c = starts[0][0]
    best_n = starts[0][2]

    endA = t0 + min(300.0, max(20.0, 0.27 * b))
    endB = t0 + min(480.0, max(200.0, 0.43 * b))
    end2 = t0 + min(730.0, max(360.0, 0.66 * b))
    end3 = t0 + min(910.0, max(520.0, 0.82 * b))

    endA = min(endA, final_end - 30.0)
    endB = min(endB, final_end - 25.0)
    end2 = min(end2, final_end - 20.0)
    end3 = min(end3, final_end - 10.0)

    if endA < t0 + 5.0:
        endA = t0 + 2.0
    if endB < endA + 5.0:
        endB = min(endA + 30.0, final_end - 15.0)
    if end2 < endB + 5.0:
        end2 = min(endB + 30.0, final_end - 10.0)
    if end3 < end2 + 5.0:
        end3 = min(end2 + 20.0, final_end - 5.0)

    full_alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    short_alphas = [500, 5000, 50000, 200000]
    high_alphas = [100, 1000, 5000, 50000, 200000]

    if time.time() < endA - 8.0 and len(starts) > 0:
        h_u, c_u = optimize_active(
            starts[0][1].copy(),
            int(starts[0][2]),
            endA,
            K=60,
            alphas=full_alphas,
            mu=0.02,
        )
        best_h, best_c, best_n = consider_candidate(
            h_u, c_u, int(starts[0][2]), best_h, best_c, best_n
        )

    for s in starts[1:]:
        if time.time() >= endB - 45.0:
            break
        if s[0] > best_c + 0.035:
            continue
        if candidates_close(s[1], starts[0][1], atol=1e-12):
            continue
        h_u, c_u = optimize_active(
            s[1].copy(),
            int(s[2]),
            min(endB, time.time() + 90.0),
            K=60,
            alphas=short_alphas,
            mu=0.02,
        )
        best_h, best_c, best_n = consider_candidate(
            h_u, c_u, int(s[2]), best_h, best_c, best_n
        )

    high_ns = []
    seen = set()
    for m in res_list:
        mm = int(m)
        if mm > int(best_n) or mm != n_main:
            if mm not in seen:
                if len(high_ns) < 2:
                    high_ns.append(mm)
                    seen.add(mm)

    if len(high_ns) > 0 and time.time() < end2 - 35.0:
        nH = int(high_ns[0])
        hA = project_feasible(resample_to(best_h, nH), nH)
        cA = exact_c5(hA, nH)
        hB = make_binary(best_h, nH)
        cB = exact_c5(hB, nH)

        hstarts = []
        if np.isfinite(cA):
            hstarts.append((float(cA), hA, nH, 0))
        if np.isfinite(cB):
            hstarts.append((float(cB), hB, nH, 1))

        if hstarts:
            hstarts = sort_starts(hstarts)
            now = time.time()
            rem = end2 - now
            first_end = now + min(max(25.0, 0.55 * rem), 160.0)
            first_end = min(first_end, end2 - 15.0)
            if first_end <= now + 3.0:
                first_end = now + max(3.0, min(20.0, rem))

            K_H = 60 if nH <= 1200 else 50
            h_u, c_u = optimize_active(
                hstarts[0][1].copy(),
                nH,
                first_end,
                K=K_H,
                alphas=high_alphas,
                mu=0.02,
            )
            best_h, best_c, best_n = consider_candidate(
                h_u, c_u, nH, best_h, best_c, best_n
            )

            if len(hstarts) > 1 and time.time() < end2 - 30.0:
                if not candidates_close(hstarts[1][1], hstarts[0][1], atol=1e-11):
                    h_u, c_u = optimize_active(
                        hstarts[1][1].copy(),
                        nH,
                        end2,
                        K=K_H,
                        alphas=short_alphas,
                        mu=0.02,
                    )
                    best_h, best_c, best_n = consider_candidate(
                        h_u, c_u, nH, best_h, best_c, best_n
                    )

    if len(high_ns) > 1 and time.time() < end3 - 35.0:
        nH2 = int(high_ns[1])
        hA = project_feasible(resample_to(best_h, nH2), nH2)
        cA = exact_c5(hA, nH2)
        hB = make_binary(best_h, nH2)
        cB = exact_c5(hB, nH2)

        hstarts = []
        if np.isfinite(cA):
            hstarts.append((float(cA), hA, nH2, 0))
        if np.isfinite(cB):
            hstarts.append((float(cB), hB, nH2, 1))

        if hstarts:
            hstarts = sort_starts(hstarts)
            now = time.time()
            rem = end3 - now
            first_end = now + min(max(20.0, 0.60 * rem), 130.0)
            first_end = min(first_end, end3 - 10.0)
            if first_end <= now + 3.0:
                first_end = now + max(3.0, min(18.0, rem))

            K_H = 60 if nH2 <= 1200 else 50
            h_u, c_u = optimize_active(
                hstarts[0][1].copy(),
                nH2,
                first_end,
                K=K_H,
                alphas=high_alphas,
                mu=0.02,
            )
            best_h, best_c, best_n = consider_candidate(
                h_u, c_u, nH2, best_h, best_c, best_n
            )

            if len(hstarts) > 1 and time.time() < end3 - 25.0:
                if not candidates_close(hstarts[1][1], hstarts[0][1], atol=1e-11):
                    h_u, c_u = optimize_active(
                        hstarts[1][1].copy(),
                        nH2,
                        end3,
                        K=K_H,
                        alphas=short_alphas,
                        mu=0.02,
                    )
                    best_h, best_c, best_n = consider_candidate(
                        h_u, c_u, nH2, best_h, best_c, best_n
                    )

    remain = final_end - time.time()

        if int(best_n) <= 1200:
            pol_K = 60
            pol_alphas = [5000, 50000, 200000]
        elif int(best_n) <= 1800:

            pol_alphas = [10000, 100000]
