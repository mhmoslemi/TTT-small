import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def full_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    if alpha < 1e6:
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    else:
        return float(m + np.mean(c - m + 1e-10))  # fallback

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000, deadline=None):

    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]

        return h.copy()

    Using grid search + refinement."""
    xs = np.linspace(0, 1, n_grid)

    best_idx = np.argmin(vals)
    x_best = xs[best_idx]

    if best_idx > 0 and best_idx < n_grid - 1:
        lo = xs[max(0, best_idx - 2)]
        hi = xs[min(n_grid - 1, best_idx + 2)]
        # Golden section on [lo, hi]
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(30):
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = np.max(a + x1 * D) + lam * x1
            f2 = np.max(a + x2 * D) + lam * x2
            if f1 < f2:
                hi = x2
            else:
                lo = x1
        x_best = (lo + hi) / 2
    
    return float(np.clip(x_best, 0, 1))
