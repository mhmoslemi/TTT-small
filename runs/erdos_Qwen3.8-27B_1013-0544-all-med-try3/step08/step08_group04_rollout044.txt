import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_corr_matrix(v, n):
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        offset = k - (n-1)
        # valid m: 0 <= m < n and 0 <= m+offset < n
        m_lo = max(0, -offset)
        m_hi = min(n, n - offset)
        if m_lo < m_hi:
            A[k, m_lo:m_hi] = v[m_lo+offset:m_hi+offset]
    return A

def lp_step(h, n, verbose=False):
    v = 1.0 - h
    dx = 2.0 / n
    A = build_corr_matrix(v, n)
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        dx * (A @ h_var) <= t,
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS)  # or GLOP, CBC
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h.copy()
    
    if h_var.value is None:
        return h.copy()
    
    return np.clip(h_var.value, 0.0, 1.0)

def alternating_optimize(h, n, max_iters=200, beta=0.5, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):

        h_lp = lp_step(h, n)

def lp_step_linprog(h, n):
    v = 1.0 - h
    dx = 2.0 / n
    A = build_corr_matrix(v, n)
    # A is (2n-1) x n
    # Constraints: dx * A @ h - t <= 0  →  dx*A @ [h, t^T] <= 0
    # i.e., [dx*A, -1] @ [h; t] <= 0
    
    m = 2*n - 1
    A_ub_full = np.zeros((m, n+1))
    A_ub_full[:, :n] = dx * A
    A_ub_full[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    bounds = [(0.0, 1.0)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return h.copy()
    
    return np.clip(res.x[:n], 0.0, 1.0)

from scipy.optimize import minimize, linprog

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_corr_matrix(v, n):
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        offset = k - (n - 1)
        m_lo = max(0, -offset)
        m_hi = min(n, n - offset)
        if m_lo < m_hi:
            A[k, m_lo:m_hi] = v[m_lo + offset:m_hi + offset]
    return A

def lp_step(h, n):
    v = 1.0 - h
    dx = 2.0 / n
    A = build_corr_matrix(v, n)
    m = 2 * n - 1
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = dx * A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0.0, 1.0)] * n + [(-np.inf, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
