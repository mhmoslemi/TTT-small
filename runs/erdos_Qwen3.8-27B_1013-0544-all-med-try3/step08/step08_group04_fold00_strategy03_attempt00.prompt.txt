       def __init__(self, base_n, max_n, budget, objective):
           self.base_n = base_n
           self.max_n = max_n

           self.objective = objective
           self.history = []

       def resolve_vector(self, s, target_n):
           # Barycentric interpolation
           current_n = len(s)
           new_h = np.interp(np.linspace(0, target_n-1, current_n), 
                            np.linspace(0, current_n-1, current_n),
                            s)
           return (
               project_feasible(new_h, target_n),
               target_n / current_n * self.objective(new_h, len(new_h))
           )

       def optimize(self, initial=None):

               start_sol = create_feasible_vector(np.random, self.base_n)
           else:
               start_sol = initial

       H = cp.Variable(shape=n)

       # Volume preservation
       volume = n/np.sqrt(sum(np.exp(cp.sum_power_pos(H, 1))))
       constraints.append(cp.sum(H) == volume)
       
       # Smoothness
       if n > 1:
           smoothness = cp.sum(cp.abs(H[1:] - H[:-1])) <= 0.5 * np.sqrt(n)
           constraints.append(smoothness)

       # Objective transformation
       def cyclic_offset(x):
           return np.correlate(x, 1-x, mode='full')
           
       obj = cp.Minimize( cp.sum(cp.log(b + 1 + H)) - 
                         cp.sum(cp.multiply(cyclic_offset(x), x[1:])) )
       
       prob = cp.Problem(obj, constraints)
       prob.solve(solver=cp.MOSEK, verbose=True)
       
       if prob.status in ['optimal', 'optimal-inexact']:

                                     np.linspace(0,2,n), H.value), 0,1)
