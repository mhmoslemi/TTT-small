    # -h[t-shift] term: valid when 0 <= t-shift < n
    lo_m = max(0, shift)
    hi_m = min(n, n + shift)
    if lo_m < hi_m:
        f[lo_m:hi_m] -= h[lo_m-shift:hi_m-shift]
    return f
```

This is O(n) and vectorized!

Then the pair search becomes:
1. Compute f_vec = compute_f_vec(h, n, shift) * dx
2. Sort by f_vec
3. Try pairs (low f, high f) with various d
4. Verify with full C₅ computation

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_overlap_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_f_vec(h, n, shift):
    """
    For the overlap at correlation index (n-1+shift), compute the gradient
    with respect to each h[t], divided by dx.
    shift is in range [-(n-1), n-1]
    """
    f = np.zeros(n)
    # Term: (1 - h[t + shift]) for valid t+shift in [0, n-1]
    lo_p = max(0, -shift)
    hi_p = min(n, n - shift)
    if lo_p < hi_p:
        f[lo_p:hi_p] += (1.0 - h[lo_p + shift:hi_p + shift])
    # Term: -h[t - shift] for valid t-shift in [0, n-1]
    lo_m = max(0, shift)
    hi_m = min(n, n + shift)
    if lo_m < hi_m:
        f[lo_m:hi_m] -= h[lo_m - shift:hi_m - shift]
    return f

def smooth_max_obj(h, n, alpha):
    O = compute_overlap_vec(h, n)
    m = O.max()
    shifted = alpha * (O - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):
    O = compute_overlap_vec(h, n)
    m = O.max()
    shifted = alpha * (O - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
        w /= w.sum()
    else:
        w = np.exp(shifted)
        w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    # For each correlation index k, the contribution to grad is w[k] * dx * f_vec at shift k-(n-1)
    # f_vec(t, shift) = (1-h[t+shift]) - h[t-shift] (with bounds)
    # But we need to be more careful: the gradient of O[k] with respect to h[t] is:
    # dx * [(1-h[t+shift])*1(0<=t+shift<n) - h[t-shift]*1(0<=t-shift<n)]
    # where shift = k - (n-1)
    
    # Vectorized: for each k, add w[k]*dx*f_vec(h, n, k-(n-1)) to grad
    # This is O(n * (2n-1)) = O(n²) which for n=600 is 720,000 ops - fine
    
    for k in range(2 * n - 1):
        wmk = w[k]
        if wmk < 1e-15:
            continue
        shift = k - (n - 1)
        f = compute_f_vec(h, n, shift)
        grad += wmk * dx * f
    
    return grad

def targeted_search(h, n, deadline, max_rounds=50000):
    """
    Targeted pair search:
    1. Find the argmax shift
    2. Compute f(t) at that shift
    3. Greedily move mass from high-f to low-f positions
    4. Verify with full C5 computation
    """
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    round_count = 0
    
    while time.time() < deadline and round_count < max_rounds:
        round_count += 1
        O = compute_overlap_vec(h, n)
        O_max = O.max()
        
        # If we've improved, update best
        if O_max < best_c - 1e-15:
            best_c = O_max
        
        # Find top shifts
        top_count = min(20, 2*n-1)
        top_indices = np.argsort(O)[-top_count:]
        top_indices = np.sort(top_indices)
        
        improved = False
        d_values = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        
        for s_idx in top_indices:
            if improved:
                break
            shift = int(s_idx) - (n - 1)
            f = compute_f_vec(h, n, shift)
            order = np.argsort(f)
            
            # Try moves: add to low-f, remove from high-f
            # Limit to top/bottom 30 positions for efficiency
            n_low = min(30, n)
            n_high = min(30, n)
            lows = order[:n_low]
            highs = order[-n_high:]
            
            for i in lows:
                if improved:
                    break
                for j in highs:
                    if i == j:
                        continue
                    for d in d_values:
                        if h[i] + d > 1.0 or h[j] - d < 0.0:
                            continue
                        h[i] += d
                        h[j] -= d
                        c_new = np.max(compute_overlap_vec(h, n))
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                            break
                        else:
                            h[i] -= d
                            h[j] += d
                    if improved:
                        break
                # Also try the reverse direction
                if not improved:
                    for d in d_values:
                        if h[j] + d > 1.0 or h[i] - d < 0.0:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = np.max(compute_overlap_vec(h, n))
                        if c_new < best_c - 1e-15:
                            best_c = c_new
