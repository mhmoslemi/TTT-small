                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.002
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.01)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] += remaining
    return project_feasible(h_bin, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick L-BFGS polish at original resolution
    p1_end = t0 + 80
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[100, 1000, 10000, 100000],
        mu=0.01, maxiters=[1000, 1500, 2000, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=1000 and optimize
    n_mid = 1000
    if time.time() < deadline - 400:
        h_mid = upsample(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        if c_mid < best_c + 0.001:
            p2_end = min(time.time() + 180, deadline - 350)
            h2, c2 = run_lbfgs(h_mid, n_mid, p2_end,
                alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
                mu=0.005, maxiters=[800, 1000, 1500, 1500, 2000, 2500, 3000, 3500])
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = n_mid

    # Phase 3: Active set optimization at best resolution
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 120, deadline - 200)
        h3, c3 = run_active_opt(best_h, best_n, p3_end, top_k=5,
            alphas=[50000, 200000, 1000000, 5000000, 20000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Single constraint dive
    if time.time() < deadline - 180:
        p4_end = min(time.time() + 80, deadline - 140)
        h4, c4 = run_single_dive(best_h, best_n, p4_end, alpha=5000000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try n=1500 if time allows
    n_high = 1500
    if time.time() < deadline - 200 and best_n < n_high:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.002:
            p5_end = min(time.time() + 130, deadline - 130)
            h5, c5 = run_lbfgs(h_high, n_high, p5_end,
                alphas=[10, 50, 200, 1000, 5000, 20000, 100000],
                mu=0.003, maxiters=[500, 700, 1000, 1200, 1500, 1800, 2000])
            if c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = n_high

    # Phase 6: Active set at final resolution
    if time.time() < deadline - 120:
        p6_end = min(time.time() + 60, deadline - 80)
        h6, c6 = run_active_opt(best_h, best_n, p6_end, top_k=4,
            alphas=[100000, 500000, 2000000, 10000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Binary restart exploration
    if time.time() < deadline - 100:
        h_bin = binary_start(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        p7_end = min(time.time() + 50, deadline - 60)
        h7, c7 = run_lbfgs(h_bin, best_n, p7_end,
            alphas=[50, 500, 5000, 50000, 500000],
            mu=0.01, maxiters=[500, 800, 1000, 1200, 1500])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: SLSQP polish
    if time.time() < deadline - 60:
        for a in [500000, 2000000, 10000000]:
            if time.time() > deadline - 50:
                break
            p8_end = min(time.time() + 20, deadline - 40)
            h8, c8 = run_slqp(best_h, best_n, p8_end, alpha=a, maxiter=1000)
            if c8 < best_c:
                best_c = c8
                best_h = h8

    # Phase 9: Subgradient refinement
    if time.time() < deadline - 30:
        p9_end = min(time.time() + 20, deadline - 15)
        h9, c9 = subgrad_refine(best_h, best_n, p9_end, n_iter=5000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Final pair search
    if time.time() < deadline - 5:
        p10_end = deadline - 2
        h10, c10 = pair_search(best_h, best_n, p10_end, radius=40)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final: one more short active-set push
    if time.time() < deadline - 1:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 10000000, 0.001),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 200, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxfun': 500})
            h_final = project_feasible(res.x, best_n)
            c_final = compute_c5(h_final, best_n)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        except Exception:
            pass

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
