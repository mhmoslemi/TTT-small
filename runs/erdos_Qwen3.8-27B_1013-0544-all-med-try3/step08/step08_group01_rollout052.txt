from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def logsumexp_shift(c, alpha):
    """Stable log-sum-exp of alpha*(c - max(c))."""
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = logsumexp_shift(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nn = len(c)  # 2n-1
    for mm in range(nn):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    val = logsumexp_shift(vals, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2

def top_shifts_grad(x, n, active, alpha, mu):

    m = vals.max()

    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 5000
    mu = 100.0
    y = h.copy()
    t = 1.0

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _logsumexp_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def _grad_single_shift(grad, x, n, idx, wm):
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = _logsumexp_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        _grad_single_shift(grad, x, n, mm, w[mm] * dx)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

        x_new = np.linspace(0, 2, n_alt)
        h_alt = np.interp(x_new, x_old, best_h)
        h_alt = project_feasible(h_alt, n_alt)
        t4 = min(time.time() + 120, deadline - 180)
        h4, c4 = lbfgs_optimize(h_alt, n_alt, t4,
            alpha_list=[500, 5000, 50000, 200000],
            mu=100.0, maxiter=2000)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_alt

    # Phase 5: Shift equalization (80s)
    if time.time() < deadline - 100:
        t5 = min(time.time() + 60, deadline - 100)
        h5, c5 = equalize_shifts(best_h, best_n, t5)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Top-shifts again after equalization (60s)
    if time.time() < deadline - 60:
        t6 = min(time.time() + 45, deadline - 60)
        h6, c6 = top_shifts_optimize(best_h, best_n, t6, top_k=4,
            alpha_list=[50000, 200000, 1000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Nesterov projected gradient (40s)
    if time.time() < deadline - 30:
        t7 = min(time.time() + 25, deadline - 30)
        h7, c7 = nesterov_pgd(best_h, best_n, t7, max_iters=3000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Pair refine (25s)
    if time.time() < deadline - 8:
        t8 = min(time.time() + 18, deadline - 6)
        h8, c8 = pair_refine(best_h, best_n, t8, radius=30)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final L-BFGS polish
    if time.time() < deadline - 2:
        h9, c9 = lbfgs_optimize(best_h, best_n, deadline - 1,
            alpha_list=[200000, 1000000],
            mu=200.0, maxiter=800)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
