    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx

    w = np.exp(sh - sh.max())
    w /= w.sum()

    lags = np.arange(-(n-1), n)

    for k_idx in range(2*n-1):

        if wk < 1e-15:
            continue

        if lag >= 0:
            grad[:n-lag] += dx * wk * (1 - h[lag:] - 0)  # -h[i+lag]
            grad[lag:] += dx * wk * (1 - 0 - h[:n-lag])  # -h[i-lag]
            # Wait, I need to be more careful

            grad[:n-lag] -= dx * wk * h[lag:]  # -h[i+lag] part
            grad[lag:] -= dx * wk * h[:n-lag]  # -h[i-lag] part
            # And +dx*wk*1 for all i where the term exists

        else:

            # i-lag = i+l, valid when i < n-l
            grad[l:] -= dx * wk * h[:n-l]  # -h[i+lag] = -h[i-l]
            grad[:n-l] -= dx * wk * h[l:]  # -h[i-lag] = -h[i+l]

    # Sort by gradient: want to decrease h where g is large, increase where g is small
    order = np.argsort(grad)  # ascending: most negative gradient first
    best_c = m
    improved = False

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        val = m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha
    else:

    s = x.sum() - n / 2.0
    val += pen * s * s
    return float(val)
