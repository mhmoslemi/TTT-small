    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    val = smooth_max_value(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _lbfgs_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

def _slsqp_obj(x, n, alpha):
    return smooth_max_value(x, n, alpha)

def _slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, deadline, max_iters=200, alpha_list=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if alpha_list is None:
        alpha_list = [50, 200, 1000, 5000, 50000, 500000]
    mus = [5.0, 10.0, 50.0, 100.0, 500.0]
    for alpha in alpha_list:
        if time.time() > deadline - 3:
            break
        for mu in mus[:3]:  # Use first 3 mus
            if time.time() > deadline - 3:
                break
            try:
                iters = min(max_iters, int(50000 / alpha) + 50)
                iters = max(iters, 30)
                res = minimize(_lbfgs_obj, best_h, jac=_lbfgs_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': iters, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxfun': iters * 5,
                                       'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def slsqp_opt(h, n, deadline, alpha=200000, maxiter=1500):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_descent_momentum(h, n, deadline, lr_init=0.005, alpha=1000.0, momentum=0.9):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    for _ in range(10000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        # Nesterov-like: use momentum
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new
            h = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            # Revert and reduce step
            v = momentum * v  # Keep velocity for next step
            h = project_feasible(h, n)
            h_new2 = project_feasible(h - lr * g, n)
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c - 1e-16:
                best_c = c_new2
                best_h = h_new2
                h = h_new2
            lr *= 0.5
            if lr < 1e-10:
                lr = 0.005
                v = np.zeros(n)
                alpha *= 2
                if alpha > 2000000:
                    break
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None, max_passes=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_passes if max_passes else 10):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - (radius if radius else n))
                hi = min(n, i + (radius if radius else n) + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return best_h, best_c

def gradient_pair_search(h, n, deadline, step_size=0.003):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [100, 500, 2000, 10000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            g = smooth_max_grad(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(max(0, i-30), min(n, i+31)):
                    if i == j:
                        continue
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-10:
                        continue
                    if 0.0 <= h[i] + move <= 1.0 and 0.0 <= h[j] - move <= 1.0:
                        h[i] += move; h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= move; h[j] += move
            if not improved:
                break
    return best_h, best_c

def upsample(h_old, n_old, n_new, rng, noise_scale=0.02):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old, period=2.0)
    # Add structured noise
    noise = np.zeros(n_new)
    for freq in [3, 7, 13]:
        noise += rng.uniform(0.3, 1.0) * np.sin(2 * np.pi * freq * x_new / 2.0 + rng.uniform(0, 2*np.pi))
    h_interp += noise_scale * noise
    return project_feasible(h_interp, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    c_max = c.max()
    sh = alpha * (c - c_max)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    active = np.where(w > 1e-10)[0]
    for m in active:
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    val = smooth_max_value(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _lbfgs_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    return smooth_max_value(x, n, alpha)

def _slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, deadline, alpha_list=None, iters_per_alpha=40):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if alpha_list is None:
        alpha_list = [50, 500, 5000, 50000, 500000]
    for alpha in alpha_list:
        if time.time() > deadline - 3:
            break
        mus = [10.0, 50.0, 200.0]
        for mu in mus:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(_lbfgs_obj, best_h, jac=_lbfgs_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': iters_per_alpha, 'ftol': 1e-16,
                                       'gtol': 1e-13, 'maxfun': iters_per_alpha * 8,
                                       'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def slsqp_opt(h, n, deadline, alpha=200000, maxiter=1200):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_step_opt(h, n, deadline, alpha=5000, lr_init=0.004, max_iters=5000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        step_lr = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step_lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step_lr *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-12:
                alpha *= 3.0
                lr = 0.004
                if alpha > 5000000:
                    break
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=40):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        passes_without_improvement = 0
        while time.time() < deadline:
            if passes_without_improvement >= 3:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 - 1e-12 and h[j] - d >= 1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 1e-12 and h[j] + d <= 1.0 - 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                passes_without_improvement += 1
            else:
                passes_without_improvement = 0
    return best_h, best_c

def grad_guided_pair(h, n, deadline, step=0.003, radius=25):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [200, 1000, 5000, 20000]
    for alpha in alphas:
        no_improve_count = 0
        while time.time() < deadline:
            if no_improve_count > 5:
                break
            improved = False
            g = smooth_max_grad(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    move = -step * (g[i] - g[j])
                    if abs(move) < 1e-10:
                        continue
                    if 1e-12 <= h[i] + move <= 1.0 - 1e-12 and 1e-12 <= h[j] - move <= 1.0 - 1e-12:
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                no_improve_count += 1
            else:
                no_improve_count = 0
    return best_h, best_c

def upsample_with_noise(h_old, n_old, n_new, rng, noise_scale=0.015):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    h_clipped = np.clip(h_interp, 0.0, 1.0)
    noise = np.zeros(n_new)
    for freq in [2, 5, 11, 17]:
        amp = rng.uniform(0.2, 0.8)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        noise += amp * np.sin(2.0 * np.pi * freq * x_new / 2.0 + phase)
    h_new = h_clipped + noise_scale * noise
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, best_n if 'best_n' in dir() else n_init)
    best_n = n_init

    # Phase 1: Quick L-BFGS-B at original n (low-alpha exploration)
    if time.time() < deadline - 500:
        t1 = min(time.time() + 80, deadline - 500)
        best_h, best_c = lbfgs_opt(best_h, best_n, t1,
                                   alpha_list=[50, 200, 2000, 20000],
                                   iters_per_alpha=30)
        best_n = len(best_h)

    # Phase 2: SLSQP at original n (high-alpha precision)
    if time.time() < deadline - 420:
        t2 = min(time.time() + 80, deadline - 420)
        for a in [100000, 500000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_opt(best_h, best_n, min(time.time() + 40, t2),
                               alpha=a, maxiter=800)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        best_n = len(best_h)

    # Phase 3: Upsample to n=600 and optimize
    if time.time() < deadline - 300:
        n600 = 600
        h600 = upsample_with_noise(best_h, best_n, n600, rng, 0.012)
        t3a = min(time.time() + 70, deadline - 220)
        h600, c600 = lbfgs_opt(h600, n600, t3a,
                               alpha_list=[100, 1000, 10000],
                               iters_per_alpha=25)
        if time.time() < deadline - 150:
            t3b = min(time.time() + 60, deadline - 150)
            h600, c600 = slsqp_opt(h600, n600, t3b, alpha=500000, maxiter=500)
        if c600 < best_c:
            best_c = c600
            best_h = h600
            best_n = n600

    # Phase 4: Try n=450
    if time.time() < deadline - 220:
        n450 = 450
        h450 = upsample_with_noise(best_h, best_n, n450, rng, 0.012)
        t4a = min(time.time() + 50, deadline - 160)
        h450, c450 = lbfgs_opt(h450, n450, t4a,
                               alpha_list=[100, 1000, 10000],
                               iters_per_alpha=25)
        if time.time() < deadline - 100:
            t4b = min(time.time() + 40, deadline - 100)
            h450, c450 = slsqp_opt(h450, n450, t4b, alpha=500000, maxiter=400)
        if c450 < best_c:
            best_c = c450
            best_h = h450
            best_n = n450

    # Phase 5: More high-alpha optimization at best resolution
    if time.time() < deadline - 180:
        t5 = min(time.time() + 60, deadline - 180)
        best_h, best_c = lbfgs_opt(best_h, best_n, t5,
                                   alpha_list=[50000, 500000, 2000000],
                                   iters_per_alpha=50)

    if time.time() < deadline - 130:
        t5b = min(time.time() + 50, deadline - 130)
        best_h, best_c = slsqp_opt(best_h, best_n, t5b, alpha=2000000, maxiter=600)

    # Phase 6: Gradient descent with high alpha
    if time.time() < deadline - 100:
        t6 = min(time.time() + 40, deadline - 100)
        h6, c6 = gradient_step_opt(best_h, best_n, t6, alpha=20000, lr_init=0.003)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Gradient-guided pair search
    if time.time() < deadline - 60:
        t7 = min(time.time() + 50, deadline - 60)
        best_h, best_c = grad_guided_pair(best_h, best_n, t7, step=0.002, radius=20)

    # Phase 8: Standard pair search with fine steps
    if time.time() < deadline - 20:
        t8 = min(time.time() + 80, deadline - 8)
        best_h, best_c = pair_search(best_h, best_n, t8,
                                     deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=30)

    # Phase 9: Final SLSQP polish
    if time.time() < deadline - 3:
        t9 = deadline - 2
        best_h, best_c = slsqp_opt(best_h, best_n, t9, alpha=5000000, maxiter=400)

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
