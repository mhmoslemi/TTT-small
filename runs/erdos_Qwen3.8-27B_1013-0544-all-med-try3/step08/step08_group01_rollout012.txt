import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, active=None):
    c = compute_c5_vec(x, n)
    if active is not None:
        c = np.array([c[i] for i in active])
    m = c.max()
    sh = alpha * (c - m)
    # log-sum-exp
    smx = sh.max()
    logsum = smx + np.log(np.sum(np.exp(sh - smx)))
    return m + logsum / alpha

def smooth_grad(x, n, alpha, active=None):
    c = compute_c5_vec(x, n)
    if active is not None:
        c_sub = np.array([c[i] for i in active])
    else:
        c_sub = c
    m = c_sub.max()
    sh = alpha * (c_sub - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    indices = active if active is not None else np.arange(len(c_sub))
    for wi, idx in zip(w, indices):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,

        c_new = compute_c5(h_new, n)

            if tot < 1e-14:

        # Optimize at higher resolution
        c_vec_up = compute_c5_vec(h_up, n_try)
        active_up = np.argpartition(c_vec_up, -5)[-5:]
        h_up, c_up2 = opt_fixed_active(h_up, n_try, active_up, 50000, 500.0,
                                        3000, time.time() + 40)
        active_up = np.argpartition(compute_c5_vec(h_up, n_try), -3)[-3:]
        h_up, c_up3 = opt_fixed_active(h_up, n_try, active_up, 500000, 500.0,
                                        3000, time.time() + 40)
        if c_up3 < best_c and time.time() < deadline - 60:
            # Do coordinate descent at higher res
            h_up, c_up3 = coord_descent(h_up, n_try, 0.001, 10,
                                         time.time() + 20, max_passes=2)
            if c_up3 < best_c:
                best_c = c_up3
                best_h = h_up
                n = n_try

    # Final: also try one more round at original best
    if time.time() < deadline - 10:
        c_vec = compute_c5_vec(best_h, len(best_h))
        active = np.argpartition(c_vec, -2)[-2:]
        h_final, c_final = opt_fixed_active(best_h, len(best_h), active, 10000000,
                                             500.0, 3000, deadline - 2)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final projection
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
