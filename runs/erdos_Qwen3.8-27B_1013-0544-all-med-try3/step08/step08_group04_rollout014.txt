def compute_c5(h, n):

def compute_corr_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_smooth_obj(x, n, alpha, mu=0.0):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        lse = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        lse = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        lse += mu * (x.sum() - n / 2.0) ** 2
    return float(lse)

def lse_smooth_grad(x, n, alpha, mu=0.0):

        w /= w.sum()
    else:
        w = np.exp(shifted)
        w /= w.sum()

    term1 = np.zeros(n)
    term2 = np.zeros(n)

        m_lag = k - (n - 1)
        w_k = w[k]

            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                h = ht

    if time.time() < deadline - 10:
        if n <= 200:
            h, best_c = pair_search_small(h, n, deadline - 10,
                                          [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            h, best_c = gradient_guided_search(h, n, deadline - 10,
                                                alphas=(500, 2000, 10000))

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B optimization at initial resolution (250s)
    alloc1 = min(250, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Try different resolutions (200s)
    x_init = np.linspace(0, 2, n_init)
    resolutions_to_try = []
    for nr in [200, 300, 150, 400]:
        if nr != n_init:
            resolutions_to_try.append(nr)

    for nr in resolutions_to_try:
        if time.time() > deadline - 150:
            break
        x_new = np.linspace(0, 2, nr)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, nr)
        alloc = min(100, (deadline - time.time()) * 0.25)
        if alloc < 30:
            break
        h_opt, c_opt = optimize_at_resolution(h_interp, nr, time.time() + alloc, rng)
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt
            x_init = np.linspace(0, 2, nr)

    # Phase 3: Multi-restart with structured perturbations (250s)
    if time.time() < deadline - 100:
        n_best = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(4, int(remaining / 50))

        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            if r == 0:
                scale = 0.05
            elif r == 1:
                scale = 0.02
            elif r == 2:
                scale = 0.1
            else:
                scale = 0.03 + 0.05 * np.sin(r * 1.7)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(60, (deadline - time.time() - 40) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c - 1e-14:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Alternative structured initializations
    if time.time() < deadline - 80:
        n_alt = len(best_h)
        alternatives = []
        # Periodic pattern
        alt1 = np.where(np.arange(n_alt) % 2 == 0, 0.8, 0.2)
        alternatives.append(alt1)
        # Triangular wave
        x_t = np.linspace(0, 2 * np.pi, n_alt)
        alt2 = 0.5 + 0.4 * np.sin(x_t)
        alternatives.append(alt2)
        # Two-block pattern
        alt3 = np.zeros(n_alt)
        alt3[:n_alt // 4] = 1.0
        alt3[n_alt // 2:3 * n_alt // 4] = 1.0
        alternatives.append(alt3)

        for alt in alternatives:
            if time.time() > deadline - 60:
                break
            alt = project_feasible(alt, n_alt)
            # Blend with current best
            for frac in [0.3, 0.6]:
                if time.time() > deadline - 50:
                    break
                h_blend = frac * alt + (1 - frac) * best_h
                h_blend = project_feasible(h_blend, n_alt)
                alloc = min(40, (deadline - time.time() - 40) * 0.3)
                if alloc < 15:
                    break
                h_opt, c_opt = optimize_at_resolution(h_blend, n_alt, time.time() + alloc, rng)
                if c_opt < best_c - 1e-14:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 5: Final intensive polish (remaining time)
    if best_h is not None and time.time() < deadline - 5:
        n_best = len(best_h)

        # Final L-BFGS-B with very high alpha
        for a in [50000, 200000, 1000000]:
            if time.time() > deadline - 5:
                break
            ht = lbfgsb_optimize(best_h, n_best, a, mu=500.0, maxiter=300)
            ht = project_feasible(ht, n_best)
            c = compute_c5(ht, n_best)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht

        # Final pair/gradient search
        if n_best <= 200 and time.time() < deadline - 3:
            best_h, best_c = pair_search_small(best_h, n_best, deadline - 2,
                                               [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        elif time.time() < deadline - 3:
            best_h, best_c = gradient_guided_search(best_h, n_best, deadline - 2,
                                                     alphas=(2000, 10000, 50000))

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
