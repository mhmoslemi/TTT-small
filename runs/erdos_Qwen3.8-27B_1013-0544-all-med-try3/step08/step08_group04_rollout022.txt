- I(0 <= j-s < n) for j in [0,n): j-s >= 0 means j >= s. And j-s < n is automatic. So valid j: [s, n)

For shift s < 0, let t = -s > 0:

  - If s < 0: j in [-s, n)
- Second term h[j-s] is nonzero for j in [max(0,s), min(n, n+s))
  - If s >= 0: j in [s, n)
  - If s < 0: j in [0, n+s) [since n+s = n-|s|]

```python
def compute_subgradient(h, n, active_idx):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active_idx:
        s = k - (n - 1)
        if s >= 0:
            # First term: j in [0, n-s), val = 1 - h[j+s]
            if n - s > 0:
                g[:n-s] += (1.0 - h[s:n])
            # Second term: j in [s, n), val = -h[j-s]
            if n - s > 0:
                g[s:] -= h[:n-s]
        else:
            t = -s
            # First term: j in [t, n), val = 1 - h[j-t]
            g[t:] += (1.0 - h[:n-t])
            # Second term: j in [0, n-t), val = -h[j+t]
            if n - t > 0:
                g[:n-t] -= h[t:n]
    g *= dx / len(active_idx)
    return g
```

- For each k, c_k(x) = a_k * x + b_k where:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_subgradient(h, n, active_idx):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active_idx:
        s = k - (n - 1)
        if s >= 0:
            ns = n - s
            if ns > 0:
                g[:ns] += (1.0 - h[s:])
                g[s:] -= (h[:ns])
        else:
            t = -s
            nt = n - t
            g[t:] += (1.0 - h[:nt])
            if nt > 0:
                g[:nt] -= h[t:]
    g *= dx / len(active_idx)
    return g

def subgradient_method(h, n, deadline, active_tol=1e-8, n_active_max=20):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step_size = 0.1
    momentum = np.zeros(n)
    
    for it in range(50000):
        if time.time() > deadline:
            break
        corr = compute_corr(h, n)
        max_c = corr.max()
        
        if max_c < best_c - 1e-14:
            best_c = max_c
            best_h = h.copy()
            step_size = min(step_size * 1.2, 0.5)
        elif it % 100 == 0 and max_c > best_c:
            h = best_h.copy()
            step_size *= 0.7
        
        # Active set
        active = np.where(corr >= max_c - active_tol)[0]
        if len(active) > n_active_max:
            active = np.argsort(corr)[-n_active_max:]
        
        g = compute_subgradient(h, n, active)
        momentum = 0.9 * momentum + 0.1 * g
        h_new = h - step_size * momentum
        h_new = project_feasible(h_new, n)
        
        new_c = compute_c5(h_new, n)
        if new_c < max_c + step_size * 0.1:  # accept if not much worse
            h = h_new
        else:
            step_size *= 0.8
        
        if it > 5000:
            step_size *= 0.999
    
    return h, best_c, best_h

def coordinate_descent(h, n, deadline, n_passes=20):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n_full = 2 * n - 1
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h, best_c
            # Compute a_k and b_k for each k
            # c_k = a_k * h[j] + b_k
            # a_k = dx * [(1-h[j+s_k])*I(0<=j+s_k<n) - h[j-s_k]*I(0<=j-s_k<n)]
            s_vals = np.arange(n_full) - (n - 1)  # all shifts
            a = np.zeros(n_full)
            for idx, s in enumerate(s_vals):
                val = 0.0
                if 0 <= j + s < n:
                    val += (1.0 - h[j + s])
                if 0 <= j - s < n:
                    val -= h[j - s]
                a[idx] = val * dx
            
            # b_k = c_k - a_k * h[j]
            # But we need c_k. Let's compute it differently.
            # Actually, for the purpose of finding min max_k (a_k*x + b_k),
            # we can just evaluate at x=0 and x=1 and x=h[j] to find the max.
            
            # More efficient: evaluate c_k at x=0 and x=1
            # c_k(x=0) = c_k - a_k * h[j] (subtract the current contribution)
            # c_k(x=1) = c_k - a_k * h[j] + a_k
            
            corr_current = np.correlate(h, 1 - h, mode='full') * dx
            
            # Coefficient of h[j] in c_k:
            # Same as a computed above
            
            # c_k at x: a_k * x + (corr_current[k] - a_k * h[j])
            # = a_k * (x - h[j]) + corr_current[k]
            
            c0 = corr_current - a * h[j]  # c_k when h[j] = 0
            c1 = c0 + a  # c_k when h[j] = 1
            c_x = c0 + a * h[j]  # should equal corr_current
            
            # We want min_x max_k (c0[k] + a[k]*x) for x in [0,1]
            # This is a convex piecewise-linear function in x
            # The minimum is where the maximum of the lines is minimized
            
            # For efficiency, consider only top correlations
            current_max = corr_current.max()
            active = np.where(corr_current >= current_max - 0.01)[0]
            if len(active) < 5:
                active = np.argsort(corr_current)[-10:]
            
            # Evaluate max at x=0, x=1, and at intersections
            max_at_0 = c0[active].max()
            max_at_1 = c1[active].max()
            
            # Try a few candidate x values
            candidates = [0.0, 1.0, h[j]]
            # Add intersection points of top lines
            a_act = a[active]
            c0_act = c0[active]
            # Sort by slope
            order = np.argsort(a_act)
            for ii in range(len(order)):
                for jj in range(ii+1, min(ii+5, len(order))):
                    ai = a_act[order[ii]]
                    aj = a_act[order[jj]]
                    if abs(ai - aj) > 1e-14:
                        x_int = (c0_act[order[jj]] - c0_act[order[ii]]) / (ai - aj)
                        if 0 <= x_int <= 1:
                            candidates.append(x_int)
            
            # Find best x
            best_x = h[j]
            best_val = max_at_0 if max_at_0 < max_at_1 else max_at_1
            # Wait, we should evaluate at all candidates
            best_val = float('inf')
            for x_c in candidates:
                x_c = max(0.0, min(1.0, x_c))
                vals = c0[active] + a[active] * x_c
                v = vals.max()
                if v < best_val - 1e-12:
                    best_val = v
                    best_x = x_c
            
            if abs(best_x - h[j]) > 1e-10:
                h[j] = best_x
                h = project_feasible(h, n)  # renormalize
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    improved = True
        
        if not improved and p > 3:
            break
    
    return h, best_c
```

Let me optimize: instead of recomputing the full correlation each time we change h[j], we can update it incrementally.

When we change h[j] from old_val to new_val (delta = new_val - old_val):
- For each k, c_k changes by: delta * [(1-h[j+s_k])*I(valid) - h[j-s_k]*I(valid)]
- = delta * a[k] (where a[k] is the coefficient we computed)

So: corr_new = corr_old + delta * a

This is O(n) per update instead of O(n²)!

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_subgrad(h, n, active_idx):
    dx = 2.0 / n
    g = np.zeros(n)
    n_full = 2 * n - 1
    s_vals = np.arange(n_full) - (n - 1)
    for k in active_idx:
        s = int(s_vals[k])
        if s >= 0:
            ns = n - s
            if ns > 0:
                g[:ns] += (1.0 - h[s:])
                g[s:] -= h[:ns]
        else:
            t = -s
            nt = n - t
            g[t:] += (1.0 - h[:nt])
            if nt > 0:
                g[:nt] -= h[t:]
    g *= dx / len(active_idx)
    return g

def subgradient_method(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step_size = 0.3
    n_full = 2 * n - 1
    s_vals = np.arange(n_full) - (n - 1)
    
    for it in range(200000):
        if time.time() > deadline:
            break
        if it % 5000 == 0:
            step_size = max(0.01, step_size * 0.95)
        
        corr = compute_corr(h, n)
        max_c = corr.max()
        
        if max_c < best_c - 1e-14:
            best_c = max_c
            best_h = h.copy()
        
        active = np.where(corr >= max_c - 0.005)[0]
        if len(active) > 25:
            active = np.argsort(corr)[-25:]
        
        g = compute_subgrad(h, n, active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        step = step_size * max_c / (g_norm * max_c + 1e-10)
        step = min(step, 0.3)
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new <= max_c + step * 0.5:
            h = h_new
        else:
            step_size *= 0.85
    
    return h, best_c, best_h

def coord_descent(h, n, deadline):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    n_full = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full') * dx
    best_c = corr.max()
    best_h = h.copy()
    
    for p in range(200):
        if time.time() > deadline:
            break
        for j in range(n):
            if time.time() > deadline:
                return best_h, best_c
            
            a = np.zeros(n_full)
            idx1 = n - 1 - j
            a[idx1:idx1 + n] += (1.0 - h)
            idx2 = j
            a[idx2:idx2 + n] -= h
            a *= dx
            
            c0 = corr - a * h[j]
            
            cm = corr.max()
            active = np.where(corr >= cm - 0.03)[0]
            if len(active) < 5:
                active = np.argsort(c0)[-15:]
            
            c0a = c0[active]
            aa = a[active]
            
            best_x = h[j]
            best_f = (c0a + aa * h[j]).max()
            
            f0 = c0a.max()
            if f0 < best_f:
                best_f, best_x = f0, 0.0
            f1 = (c0a + aa).max()
            if f1 < best_f:
                best_f, best_x = f1, 1.0
            
            order = np.argsort(aa)
            for ii in range(len(order) - 1):
                i1, i2 = order[ii], order[ii + 1]
                denom = aa[i1] - aa[i2]
                if abs(denom) < 1e-14:
                    continue
                xi = (c0a[i2] - c0a[i1]) / denom
                if -1e-9 <= xi <= 1 + 1e-9:
                    xi = max(0.0, min(1.0, xi))
                    fi = (c0a + aa * xi).max()
                    if fi < best_f - 1e-14:
                        best_f, best_x = fi, xi
            
            if abs(best_x - h[j]) > 1e-12:
                delta = best_x - h[j]
                h[j] = best_x
                corr += delta * a
                h[j] = max(0.0, min(1.0, h[j]))
                delta2 = h[j] - best_x
                if abs(delta2) > 1e-15:
                    corr += delta2 * a
        
        h = project_feasible(h, n)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c_now = corr.max()
        if c_now < best_c - 1e-15:
            best_c = c_now
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000, deadline=None):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_c(x):
        return x.sum() - n / 2.0
    
    def sum_j(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_c, 'jac': sum_j}]
    try:
        res = minimize(obj, h, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _sum_c(x, n):
    return x.sum() - n / 2.0

def _sum_j(x):
    return np.ones(x.shape[0])

def obj_smooth_inner(x, n, alpha):

    m = c.max()
    shifted = alpha * (c - m)

def slsqp_fast(h, n, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})

def optimize_single(h0, n, deadline):

    best_c = compute_c5(h, n)

    # Phase B: Coordinate descent
```
