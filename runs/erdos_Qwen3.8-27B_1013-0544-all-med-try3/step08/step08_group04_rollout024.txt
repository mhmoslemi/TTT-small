def slsqp_smooth(h, n, alpha, maxiter=2000):
    """SLSQP with log-sum-exp smoothing of the max."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def sum_c(x, nn):
        return float(x.sum() - nn / 2.0)
    
    def sum_j(x, nn):
        return np.ones(nn)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_c, 'jac': sum_j, 'args': (n,)}]
    try:
        res = minimize(obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_pair_search(h, n, deadline, deltas, radius=50):
    """Local pair search with limited radius."""
    best_c = compute_c5(h, n)
    K = 2 * n - 1
    C = compute_c5_array(h, n)
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coordinate_descent(h, n, n_passes=50, shuffle_rng=None):
    dx = 2.0 / n
    h = h.copy()
    K = 2 * n - 1
    C = compute_c5_array(h, n)
    current_max = C.max()
    k_shifts = np.arange(K) - (n - 1)
    
    for sweep in range(n_passes):
        improved_any = False
        order = list(range(n))
        if shuffle_rng is not None:
            shuffle_rng.shuffle(order)
        
        for i in order:
            a = np.zeros(K)
            valid1 = (i - k_shifts >= 0) & (i - k_shifts < n)
            a[valid1] += (1.0 - h[i - k_shifts[valid1]]) * dx
            valid2 = (i + k_shifts >= 0) & (i + k_shifts < n)
            a[valid2] -= h[i + k_shifts[valid2]] * dx
            
            # Evaluate at boundaries
            f0 = (C + a * (0.0 - h[i])).max()
            f1 = (C + a * (1.0 - h[i])).max()
            
            best_x = h[i]
            best_f = min(f0, f1, current_max)
            if abs(f0 - best_f) < 1e-15:
                best_x = 0.0
            elif abs(f1 - best_f) < 1e-15:
                best_x = 1.0
            
            # Try crossings of top constraints
            top_idx = np.argsort(C)[-5:]
            for ii in range(len(top_idx)):
                for jj in range(ii + 1, len(top_idx)):
                    s1, s2 = top_idx[ii], top_idx[jj]
                    da = a[s1] - a[s2]
                    if abs(da) < 1e-14:
                        continue
                    x_cross = (C[s2] - C[s1]) / da
                    if 0.0 < x_cross < 1.0:
                        f_test = (C + a * (x_cross - h[i])).max()
                        if f_test < best_f - 1e-15:
                            best_f = f_test
                            best_x = x_cross
            
            if abs(best_x - h[i]) > 1e-10:
                delta = best_x - h[i]
                h[i] = best_x
                C += a * delta
                current_max = C.max()
                improved_any = True
        
        h = project_feasible(h, n)
        C = compute_c5_array(h, n)
        current_max = C.max()
        
        if not improved_any:
            break
    
    return h, current_max

def slsqp_smooth(h, n, alpha, maxiter=1000):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def sum_c(x, nn):
        return float(x.sum() - nn / 2.0)
    
    def sum_j(x, nn):
        return np.ones(nn)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_c, 'jac': sum_j, 'args': (n,)}]
    try:
        res = minimize(obj, h, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_pair_search(h, n, deadline, deltas, radius=60):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
