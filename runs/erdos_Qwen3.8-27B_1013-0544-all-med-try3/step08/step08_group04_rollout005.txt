```python
def compute_subgradient(h, n, tol=1e-10):
    """Compute subgradient of C_5(h) = max_k C_k(h)."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    active = np.where(c >= c_max - tol)[0]  # indices in [0, 2n-2]
    
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)  # lag, ranges from -(n-1) to (n-1)
        for i in range(n):
            # Contribution: (1-h[i+k]) if 0<=i+k<n, minus h[i-k] if 0<=i-k<n
            contrib = 0.0
            ik = i + k
            if 0 <= ik < n:
                contrib += (1.0 - h[ik])
            ik2 = i - k
            if 0 <= ik2 < n:
                contrib -= h[ik2]
            grad[i] += contrib * dx
    
    grad /= len(active)
    return grad, c_max
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jacobian(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        hx = project_feasible(res.x, n)
        if compute_c5(hx, n) <= compute_c5(h, n) + 1e-12:
            return hx
    except Exception:
        pass
    return h.copy()

def compute_subgradient(h, n, tol=1e-8):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    active = np.where(c >= c_max - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])

        for i in range(n):

            ik = i + k
            if 0 <= ik < n:
                contrib += (1.0 - h[ik])
            ik2 = i - k
            if 0 <= ik2 < n:
                contrib -= h[ik2]
            grad[i] += contrib * dx
    grad /= float(len(active))
    return grad, c_max

    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_curr = best_c
    lr = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, c_curr = compute_subgradient(h, n, tol=1e-9)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        lr = 0.02 / (gnorm * 0.5 + 1.0)
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 2.0, lr * 0.1, lr * 0.25]:
            h_try = h - lr_try * grad
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:

                h = h_try.copy()
                improved = True

                h = h_try.copy()
                c_curr = c_try

                                              [0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline,
                                              [0.0005, 0.0002, 0.0001],
                                              radius=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
