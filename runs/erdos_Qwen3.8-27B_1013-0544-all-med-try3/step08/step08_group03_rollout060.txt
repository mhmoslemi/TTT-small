def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-14:
                break
            h += (diff / s) * h

                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_descend(h, n, deadline, alpha_init=500.0, lr_init=0.02):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    for it in range(80000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step = lr
        improved = False
        for _ in range(35):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.4
        if not improved:
            lr *= 0.5
            alpha *= 1.2
            if lr < 1e-14:
                break
    return best_h, best_c

def targeted_pair_search(h, n, deadline, deltas, radius, alpha=5000):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        pass_count = 0
        while improved and time.time() < deadline:
            pass_count += 1
            if pass_count > 500:
                break
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if g[i] < g[j] - 1e-15:
                        if h[j] - d >= 0.0 and h[i] + d <= 1.0:
                            h[i] += d
                            h[j] -= d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= d
                                h[j] += d
                    elif g[i] > g[j] + 1e-15:
                        if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                            h[i] -= d
                            h[j] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] += d
                                h[j] -= d
    return h, best_c

def full_pair_search(h, n, deadline, deltas, radius):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        pass_count = 0
        while improved and time.time() < deadline:
            pass_count += 1
            if pass_count > 300:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def upsample_with_noise(h_old, n_old, n_new, sigma, rng):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    noise = rng.randn(n_new) * sigma
    h_new = h_interp + noise
    return project_feasible(h_new, n_new)

def block_init(n, rng):
    """Create a block-based initialization: alternating 0 and 1 blocks."""
    h = np.zeros(n)
    num_blocks = rng.randint(4, 10)
    block_sizes = rng.multinomial(n, np.ones(num_blocks) / num_blocks)
    pos = 0
    for b in range(num_blocks):
        if b % 2 == 0:
            h[pos:pos + block_sizes[b]] = 1.0
        else:
            h[pos:pos + block_sizes[b]] = 0.0
        pos += block_sizes[b]
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n0 = 300

    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: L-BFGS polish at original resolution
    t1 = min(t0 + 80, deadline - 550)
    best_h, best_c = lbfgs_optimize(best_h, n0, t1, max_iters=100)

    # Phase 2: SLSQP at original resolution
    if time.time() < deadline - 400:
        t2 = min(time.time() + 50, deadline - 400)
        for a in [200000, 1000000, 5000000]:
            if time.time() > t2:
                break
            h_s, c_s = slsqp_optimize(best_h, n0, min(time.time() + 20, t2), alpha=a, maxiter=600)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 3: Gradient descent at original resolution
    if time.time() < deadline - 300:
        t3 = min(time.time() + 40, deadline - 300)
        h_gd, c_gd = gradient_descend(best_h, n0, t3, alpha_init=1000.0, lr_init=0.01)
        if c_gd < best_c:
            best_c = c_gd
            best_h = h_gd

    # Phase 4: Fine pair search at original resolution
    if time.time() < deadline - 250:
        t4 = min(time.time() + 70, deadline - 250)
        best_h, best_c = targeted_pair_search(best_h, n0, t4,
                                              [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                              radius=120, alpha=8000)

    # Phase 5: Main effort - upsample to 2n and optimize
    n_up = 2 * n0
    if 400 <= n_up <= 1500 and time.time() < deadline - 350:
        h_up = upsample(best_h, n0, n_up)
        # L-BFGS at higher res
        t5a = min(time.time() + 120, deadline - 200)
        h_up, c_up = lbfgs_optimize(h_up, n_up, t5a, max_iters=120)
        # SLSQP
        if time.time() < deadline - 140:
            t5b = min(time.time() + 50, deadline - 140)
            for a in [300000, 2000000]:
                if time.time() > t5b:
                    break
                h_up, c_up = slsqp_optimize(h_up, n_up, min(time.time() + 25, t5b), alpha=a, maxiter=800)
        # Gradient descent
        if time.time() < deadline - 80:
            t5c = min(time.time() + 35, deadline - 80)
            h_up, c_up = gradient_descend(h_up, n_up, t5c, alpha_init=2000.0, lr_init=0.008)
        # Targeted pair search
        if time.time() < deadline - 40:
            t5d = min(time.time() + 80, deadline - 35)
            h_up, c_up = targeted_pair_search(h_up, n_up, t5d,
                                              [0.003, 0.001, 0.0005, 0.0002, 0.0001],
                                              radius=180, alpha=8000)
        # Full pair search with tiny steps
        if time.time() < deadline - 25:
            t5e = min(time.time() + 40, deadline - 20)
            h_up, c_up = full_pair_search(h_up, n_up, t5e,
                                          [0.0005, 0.0002, 0.0001, 0.00005],
                                          radius=100)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            n0 = n_up

    # Phase 6: Try a noisy upsample (different landscape)
    n_6 = n_up
    if time.time() < deadline - 200:
        h_6 = upsample_with_noise(best_h, len(best_h), n_6, 0.03, rng)
        t6a = min(time.time() + 80, deadline - 120)
        h_6, c_6 = lbfgs_optimize(h_6, n_6, t6a, alphas=(200, 1000, 5000, 50000, 500000),
                                   mus=(20.0, 100.0, 500.0, 2000.0, 5000.0), max_iters=80)
        if time.time() < deadline - 50:
            h_6, c_6 = targeted_pair_search(h_6, n_6, min(time.time() + 40, deadline - 45),
                                            [0.002, 0.001, 0.0005, 0.0002],
                                            radius=150, alpha=5000)
        if c_6 < best_c:
            best_c = c_6
            best_h = h_6

    # Phase 7: Try 1.5x resolution if different from current
    n_7 = int(round(n_up * 1.5))
    if n_7 != n_up and n_7 <= 2000 and time.time() < deadline - 150:
        h_7 = upsample(best_h, len(best_h), n_7)
        t7a = min(time.time() + 60, deadline - 100)
        h_7, c_7 = lbfgs_optimize(h_7, n_7, t7a, alphas=(500, 5000, 50000, 500000),
                                   mus=(50.0, 500.0, 2000.0, 5000.0), max_iters=60)
        if time.time() < deadline - 40:
            h_7, c_7 = targeted_pair_search(h_7, n_7, min(time.time() + 30, deadline - 35),
                                            [0.002, 0.001, 0.0005],
                                            radius=120, alpha=5000)
        if c_7 < best_c:
            best_c = c_7
            best_h = h_7

    # Phase 8: Final SLSQP polish on best
    nf = len(best_h)
    if time.time() < deadline - 15:
        t8 = min(time.time() + 25, deadline - 12)
        h_f, c_f = slsqp_optimize(best_h, nf, t8, alpha=5000000, maxiter=500)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 9: Final fine local search
    if time.time() < deadline - 8:
        t9 = min(time.time() + 20, deadline - 5)
        best_h, best_c = targeted_pair_search(best_h, nf, t9,
                                              [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                              radius=80, alpha=20000)

    # Final L-BFGS micro-polish
    if time.time() < deadline - 3:
        best_h, best_c = lbfgs_optimize(best_h, nf, deadline - 2,
                                         alphas=(1000000, 5000000),
                                         mus=(5000.0, 20000.0), max_iters=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
