def project_feasible(h, n):
    x = np.asarray(h, dtype=float).copy()
    n = int(n)
    if x.size != n:
        x = np.resize(x, n)
    if not np.all(np.isfinite(x)):
        x = np.full(n, 0.5)

    target = 0.5 * n
    x = np.clip(x, 0.0, 1.0)

    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(x - mid, 0.0, 1.0)
        s = float(p.sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    p = np.clip(x - lam, 0.0, 1.0)
    diff = target - float(p.sum())

    if abs(diff) > 1e-15:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - p))
            add = min(diff, 1.0 - p[idx])
            p[idx] += add
            diff -= add
        else:
            idx = int(np.argmax(p))
            sub = min(-diff, p[idx])
            p[idx] -= sub
            diff += sub

    if abs(diff) > 1e-13:
        p = np.clip(p, 0.0, 1.0)

            d = target - s

def conv_full(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    if max(a.size, b.size) <= FFT_THRESHOLD or fftconvolve is None:
        return np.convolve(a, b, mode="full").astype(float, copy=False)
    try:
        c = fftconvolve(a, b, mode="full")
        return np.real(c).astype(float, copy=False)
    except Exception:
        return np.convolve(a, b, mode="full").astype(float, copy=False)


def conv_first(a, b):
    return conv_full(a, b)[:b.size]


def conv_cross(a, b):
    # out[i] = sum_{s=0}^{b.size-1-i} a[s] * b[i+s]
    la = a.size
    full = conv_full(a[::-1], b)
    start = la - 1
    end = start + b.size
    if end <= full.size:
        return full[start:end]
    out = np.zeros(b.size, dtype=float)
    if start < full.size:
        m = min(b.size, full.size - start)
        out[:m] = full[start:start + m]
    return out


def grad_from_weights(h, w, n, mu=0.0):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)

    wp = w[n - 1:].copy()
    wn = np.zeros(n, dtype=float)
    if n > 1:
        wn[1:n] = w[n - 2::-1]

    one_minus = 1.0 - h
    t1 = conv_cross(wp, one_minus)
    b1 = conv_first(wp, h)
    t3 = conv_first(wn, one_minus)
    b2 = conv_cross(wn, h)

    g = dx * (t1 - b1 + t3 - b2)
    if mu != 0.0:
        g = g + 2.0 * float(mu) * (float(h.sum()) - 0.5 * n)
    return g


def compute_c_vec(h, n):
    # Internal objective uses correlate(1-h, h).  Its maximum is identical to
    # correlate(h, 1-h), while the following gradients are aligned with it.
    dx = 2.0 / float(n)
    if n <= FFT_THRESHOLD or fftconvolve is None:
        return np.correlate(1.0 - h, h, mode="full") * dx
    try:
        c = fftconvolve(1.0 - h, h[::-1], mode="full")
        c = np.real(c[:2 * n - 1]) * dx
        return c.astype(float, copy=False)
    except Exception:
        return np.correlate(1.0 - h, h, mode="full") * dx


def c5_direct_actual(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))


def c5_rev_vec_direct(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    dx = 2.0 / float(n)
    return np.correlate(1.0 - h, h, mode="full") * dx


def logsumexp_stable(z):
    m = float(np.max(z))
    if m > 500.0:
        return m + float(np.log(np.sum(np.exp(z - m))))
    return float(np.log(np.sum(np.exp(z - m))))


def make_weights(c, n, top_k, beta):
    c = np.asarray(c, dtype=float)
    m = c.size
    if top_k is None or top_k <= 0 or int(top_k) >= m:
        idx = np.arange(m)
    else:
        idx = np.argsort(-c)[:int(top_k)]

    vals = c[idx]
    mx = float(vals.max())
    sh = float(beta) * (vals - mx)
    sh = sh - float(np.max(sh))
    w = np.zeros(m, dtype=float)
    w[idx] = np.exp(sh)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.ones(m, dtype=float)
        w /= m
    else:
        w /= s

    # Keep a small persistent weight on lag 0; this encourages h to stay binary.
    central = n - 1
    w[central] = max(float(w[central]), 0.05)
    w /= float(w.sum())
    return w


def weighted_obj(x, n, w, mu):
    c = compute_c_vec(x, n)
    val = float(np.dot(w, c))
    if mu:
        val += float(mu) * (float(x.sum()) - 0.5 * n) ** 2
    return val


def weighted_grad(x, n, w, mu):
    return grad_from_weights(x, w, n, mu)


def smooth_obj(x, n, alpha, mu):
    c = compute_c_vec(x, n)
    m = float(c.max())
    z = float(alpha) * (c - m)
    z = z - float(np.max(z))
    val = m + logsumexp_stable(z) / float(alpha)
    if mu:
        val += float(mu) * (float(x.sum()) - 0.5 * n) ** 2
    return float(val)


def smooth_grad(x, n, alpha, mu):
    c = compute_c_vec(x, n)
    m = float(c.max())
    z = float(alpha) * (c - m)
    z = z - float(np.max(z))
    w = np.exp(z)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(c.size, dtype=float)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return grad_from_weights(x, w, n, mu)

def resample_h(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    old = int(h.size)
    h = project_feasible(h, old if old > 0 else n)
    if old == n:
        return h
    if old == 1:
        return np.full(n, float(h[0]))
    x_old = np.linspace(0.0, 2.0, old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.interp(x_new, x_old, h)
    return project_feasible(y, n)


def lowpass_h(h, n, kfrac=0.25):
    h = resample_h(h, n)
    x = h - 0.5
    f = np.fft.rfft(x)
    m = f.size
    keep = max(2, int(kfrac * m))
    f = f.copy()
    if keep < m:
        f[keep:] = 0.0
    y = np.fft.irfft(f, n=n)
    out = 0.5 + y
    return project_feasible(np.clip(out, 0.0, 1.0), n)


def random_wave_start(n, rng):
    n = int(n)
    nmodes = 5
    r = rng.rand(nmodes)
    amps = (r / max(1e-15, float(r.sum()))) * 0.42
    freqs = rng.uniform(1.0, 10.0, size=nmodes)
    phases = rng.uniform(0.0, 2.0 * np.pi, size=nmodes)
    x = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / float(n)
    for i in range(nmodes):
        x += amps[i] * np.sin(2.0 * np.pi * freqs[i] * t + phases[i])
    h = 0.5 + x
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def adjust_binary_count(h, n, rng=None):
    h = np.asarray(h, dtype=float).copy()
    target = int(n // 2)
    cnt = int(np.rint(h.sum()))
    if cnt > target:
        ones = np.nonzero(h > 0.5)[0]
        m = cnt - target
        if rng is None:
            flip = ones[:m]
        else:
            flip = rng.choice(ones, size=m, replace=False)
        h[flip] = 0.0
    elif cnt < target:
        zeros = np.nonzero(h < 0.5)[0]
        m = target - cnt
        if rng is None:
            flip = zeros[:m]
        else:
            flip = rng.choice(zeros, size=m, replace=False)
        h[flip] = 1.0
    return h


def markov_start(n, rng, stay=0.75):
    n = int(n)
    h = np.zeros(n, dtype=float)
    cur = 1.0 if rng.rand() < 0.5 else 0.0
    for i in range(n):
        if rng.rand() > float(stay):
            cur = 1.0 - cur
        h[i] = cur
    return adjust_binary_count(h, n, rng)

def subgrad_refine(h, n, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_direct_actual(best_h, n)
    eta = 0.03

    while time.time() < float(deadline):

        tol = max(1e-9, 1e-4 * m)

        if active.size == 0:
            active = np.array([int(np.argmax(c))])

        w = np.zeros(c.size, dtype=float)
        w[active] = 1.0 / float(active.size)

                pos = np.nonzero((h[:-d] == 1.0) & (h[d:] == 0.0))[0]
            else:
                pos = np.nonzero((h[:-d] == 0.0) & (h[d:] == 1.0))[0]

            if pos.size == 0:
                continue
            if pos.size > max_candidates:
                pos = rng.choice(pos, size=max_candidates, replace=False)

            for p in pos:
                if time.time() >= float(deadline):
                    break
                q = int(p) + d
                p0 = h[p]
                q0 = h[q]

                if s > 0:
                    h[p] = 0.0
                    h[q] = 1.0
                else:
                    h[p] = 1.0
                    h[q] = 0.0

                cnew_v = c5_rev_vec_direct(h, n)
                cnew = float(cnew_v.max())

                if cnew < ccur - 1e-13:
                    cvec = cnew_v
                    ccur = cnew
                    improved_current = True
                    if cnew < best_fast - 1e-13:
                        best_fast = cnew
                        best_h = h.copy()
                        best_c = c5_direct_actual(best_h, n)
                else:
                    h[p] = p0
                    h[q] = q0

            if improved_current:
                break

        if not improved_current:
            no_improve += 1
            old_ccur = ccur
            rand_improved = False

            for _ in range(random_evals):
                if time.time() >= float(deadline):
                    break
                zeros = np.nonzero(h == 0.0)[0]
                ones = np.nonzero(h == 1.0)[0]
                if zeros.size == 0 or ones.size == 0:
                    break
                p = int(rng.choice(zeros))
                q = int(rng.choice(ones))
                p0 = h[p]
                q0 = h[q]
                h[p] = 1.0
                h[q] = 0.0

                cnew_v = c5_rev_vec_direct(h, n)
                cnew = float(cnew_v.max())

                if cnew < ccur + temp:
                    cvec = cnew_v
                    ccur = cnew
                    if cnew < old_ccur - 1e-13:
                        rand_improved = True
                    if cnew < best_fast - 1e-13:
                        best_fast = cnew
                        best_h = h.copy()
                        best_c = c5_direct_actual(best_h, n)
                else:
                    h[p] = p0
                    h[q] = q0

            temp = max(temp * 0.85, 1e-6)
            if rand_improved:
                no_improve = 0
            if no_improve > 8:
                break
        else:
            temp = max(temp * 0.95, 1e-6)

    return best_h, float(best_c)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(int(seed))

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        n_par = make_even_len(parent.size)
        if n_par != parent.size:
            h_par = resample_h(parent, n_par)
        else:
            h_par = project_feasible(parent, n_par)
    else:
        n_par = 512
        h_par = random_wave_start(n_par, rng)

    h_par = project_feasible(h_par, n_par)
    best_h = h_par.copy()
    best_n = n_par
    best_c = c5_direct_actual(best_h, best_n)

    def update_candidate(h_cand, n_cand, c_cand):
        nonlocal best_h, best_n, best_c
        if h_cand is not None and c_cand < best_c:
            best_h = np.asarray(h_cand, dtype=float).copy()
            best_n = int(n_cand)
            best_c = float(c_cand)

    # Parent-resolved strong phases.
    if deadline - time.time() > 160.0:
        end = time.time() + min(70.0, 0.18 * (deadline - time.time()))
        h1, c1 = optimize_smooth_lbfgs(
            best_h, best_n, end,
            alphas=[10.0, 50.0, 200.0, 1000.0],
            mu=0.05,
        )
        update_candidate(h1, best_n, c1)

    if deadline - time.time() > 220.0:
        end = time.time() + min(190.0, 0.45 * (deadline - time.time()))
        h2, c2 = optimize_weighted_cycles(best_h, best_n, end, mu=0.05)
        update_candidate(h2, best_n, c2)

    if deadline - time.time() > 90.0 and best_n <= 1200:
        end = time.time() + min(45.0, 0.12 * (deadline - time.time()))
        h3, c3 = optimize_slsqp_smooth(best_h, best_n, end, alpha=50000.0)
        update_candidate(h3, best_n, c3)

    # Diversified coarse starts.
    if deadline - time.time() > 230.0:
        per = max(25.0, min(55.0, (deadline - time.time() - 180.0) / 4.0))
        starts = [
            (256, lowpass_h(best_h, 256, 0.25)),
            (256, random_wave_start(256, rng)),
            (512, markov_start(512, rng, 0.75)),
            (512, thue_morse_start(512, rng)),
        ]
        for n_c, h_c in starts:
            if deadline - time.time() < 45.0:
                break
            end = time.time() + per
            hc, cc = quick_optimize(h_c, n_c, end)
            update_candidate(hc, n_c, cc)

    # Higher-resolution refinement.
    if deadline - time.time() > 240.0:
        n_high = 1024
        if best_n > n_high:
            n_high = best_n
        h_high = resample_h(best_h, n_high)
        end = deadline - 160.0
        h_high, c_high = optimize_weighted_cycles(
            h_high, n_high, end,
            mu=0.02,
            chunk=25.0,
        )
        update_candidate(h_high, n_high, c_high)

        if deadline - time.time() > 55.0 and best_n <= 1200:
            h_pol, c_pol = optimize_slsqp_smooth(
                best_h, best_n, deadline - 25.0, alpha=100000.0
            )
            update_candidate(h_pol, best_n, c_pol)

    # Binary / pair exchange polishing.
    if deadline - time.time() > 130.0:
        n_bin = 1024 if deadline - time.time() > 210.0 else 512
        if best_n > n_bin:
            n_bin = min(best_n, 1024)
        h_bin = make_binary(resample_h(best_h, n_bin), n_bin)
        h_bin, c_bin = binary_local_search(h_bin, n_bin, deadline - 35.0, rng)
        update_candidate(h_bin, n_bin, c_bin)

    # Final concise subgradient polish.
    if deadline - time.time() > 35.0:
        h_sub, c_sub = subgrad_refine(best_h, best_n, deadline - 6.0)
        update_candidate(h_sub, best_n, c_sub)

    # Final exact feasibility fix.
    best_h = project_feasible(best_h, best_n)
    s = float(best_h.sum())
    diff = 0.5 * best_n - s
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - best_h))
            best_h[idx] = min(1.0, best_h[idx] + diff)
        else:
            idx = int(np.argmax(best_h))
            best_h[idx] = max(0.0, best_h[idx] + diff)
    best_h = np.clip(best_h, 0.0, 1.0)

    best_c = c5_direct_actual(best_h, best_n)
    return best_h, float(best_c), int(best_n)
