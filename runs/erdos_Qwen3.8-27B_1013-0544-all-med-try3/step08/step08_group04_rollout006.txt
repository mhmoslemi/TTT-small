def project_box_sum(v,n):
    v=np.asarray(v,dtype=float)
    target=float(n)/2.0
    if not np.all(np.isfinite(v)): return np.full(n,0.5)
    if target<=0: return np.zeros(n)
    if target>=n: return np.ones(n)
    lo=float(np.min(v))-1.0; hi=float(np.max(v))+1.0
    for _ in range(70):
       mid=0.5*(lo+hi)
       h=np.clip(v+mid,0.0,1.0)
       s=h.sum()
       if s < target:
          lo=mid
       else:
          hi=mid
    h=np.clip(v+0.5*(lo+hi),0.0,1.0)
    res=target-h.sum()
    if abs(res)>1e-14:
       if res>0:
          idxs=np.argsort(1.0-h)[::-1]
          for idx in idxs:
             if res<=1e-15: break
             add=min(res,1.0-h[idx])
             if add>0:

          for idx in idxs:
             if res>=-1e-15: break
             sub=min(-res,h[idx])
             if sub>0:
                h[idx]-=sub; res+=sub
    return h

import numpy as np
import time
import scipy.optimize as spopt

_CE_N = 1
_CE_X = None
_CE_U = None


def compute_c5(h, n):
    if n <= 0:
        return float("inf")
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx


def project_box_sum(v, n):
    v = np.asarray(v, dtype=float).reshape(-1)
    if v.shape[0] != n:
        if v.shape[0] > 0:
            v = np.interp(np.linspace(0.0, 1.0, n),
                          np.linspace(0.0, 1.0, v.shape[0]), v)
        else:
            v = np.full(n, 0.5)
    if n <= 0:
        return v
    target = float(n) / 2.0
    if not np.all(np.isfinite(v)):
        v = np.full(n, 0.5, dtype=float)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h = np.clip(v + mid, 0.0, 1.0)
        s = float(h.sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-13:
        idxs = np.argsort(1.0 - h)[::-1]
        for idx in idxs:
            if res <= 1e-15:
                break
            add = min(res, 1.0 - h[idx])
            if add > 0.0:
                h[idx] += add
                res -= add
    elif res < -1e-13:
        idxs = np.argsort(h)[::-1]
        for idx in idxs:
            if res >= -1e-15:
                break
            sub = min(-res, h[idx])
            if sub > 0.0:
                h[idx] -= sub
                res += sub
    return np.clip(h, 0.0, 1.0)


def symmetrize(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.shape[0] != n:
        h = np.interp(np.linspace(0.0, 1.0, n),
                      np.linspace(0.0, 1.0, max(1, h.shape[0])), h)
    return project_box_sum(0.5 * (h + h[::-1]), n)


def interpolate_to(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    old = h.shape[0]
    if old == n:
        return project_box_sum(h, n)
    if old <= 1:
        return project_box_sum(np.full(n, 0.5), n)
    x_old = np.linspace(0.0, 2.0, old)
    x_new = np.linspace(0.0, 2.0, n)
    return project_box_sum(np.interp(x_new, x_old, h), n)


def top_half_binary(h, n):
    h = project_box_sum(np.asarray(h, dtype=float).reshape(-1), n)
    target = int(n // 2)
    b = np.zeros(n, dtype=float)
    if target > 0:
        idx = np.argsort(h)[::-1][:target]
        b[idx] = 1.0
    return project_box_sum(b, n)


def weighted_gradient(h, n, alpha):
    dx = 2.0 / float(n)
    c = dx * np.correlate(h, 1.0 - h, mode="full")
    cmax = float(np.max(c))
    z = alpha * (c - cmax)
    z = z - float(np.max(z))
    w = np.exp(z)
    ws = float(w.sum())
    if ws <= 0.0 or not np.isfinite(ws):
        return np.zeros(n, dtype=float), cmax
    w /= ws
    g = np.zeros(n, dtype=float)
    offset = n - 1
    for idx in range(len(w)):
        wi = float(w[idx])
        if wi <= 1e-12:
            continue
        m = idx - offset
        if m == 0:
            g += wi * (1.0 - 2.0 * h)
        elif m > 0:
            nm = n - m
            g[:nm] -= wi * h[m:n]
            g[m:n] += wi * (1.0 - h[:nm])
        else:
            q = -m
            nm = n - q
            g[:nm] += wi * (1.0 - h[q:n])
            g[q:n] -= wi * h[:nm]
    return g, cmax


def gradient_refine(h, n, deadline, alphas=(20.0, 50.0, 100.0, 300.0, 800.0), max_iters=80):
    h = project_box_sum(np.asarray(h, dtype=float).reshape(-1), n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    base_step = 0.05
    for alpha in alphas:
        step = base_step
        no_improve = 0
        for _ in range(max_iters):
            if time.time() > deadline:
                break
            g, ccur = weighted_gradient(h, n, float(alpha))
            if ccur + 1e-15 < best_c:
                best_c = ccur
                best_h = h.copy()
            v = -g
            v -= v.mean()
            norm = float(np.max(np.abs(v)))
            if norm < 1e-14:
                break
            v *= (step / norm)
            best_trial = None
            best_trial_c = best_c
            for beta in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
                trial = project_box_sum(h + beta * v, n)
                cc = compute_c5(trial, n)
                if cc + 1e-15 < best_trial_c:
                    best_trial = trial
                    best_trial_c = cc
            if best_trial is not None:
                h = best_trial
                best_c = best_trial_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
                step *= 0.5
                if no_improve > 8 or step < 0.002:
                    break
    h = project_box_sum(best_h, n)
    c = compute_c5(h, n)
    return h, c


def pair_scan_once(h, n, d, old_c, deadline):
    if n < 2 or d <= 0.0:
        return False, h, old_c
    dx = 2.0 / float(n)
    old_raw = old_c / dx if old_c > 0.0 else float("inf")
    best_raw = old_raw
    improve_tol = max(1e-12, 1e-13 / dx)
    eps = 1e-13
    best_i = -1
    best_j = -1
    best_new1 = 0.0
    best_new2 = 0.0
    best_old1 = 0.0
    best_old2 = 0.0
    r = 1.0 - h
    for i in range(n - 1):
        if time.time() > deadline:
            break
        hi = float(h[i])
        ri = 1.0 - hi
        hi_plus = hi + d
        hi_minus = hi - d
        for j in range(i + 1, n):
            hj = float(h[j])
            rj = 1.0 - hj
            if hi_plus <= 1.0 + eps and hj - d >= -eps:
                newi = hi_plus if hi_plus < 1.0 else 1.0
                newj = hj - d if hj - d > 0.0 else 0.0
                old1 = hi
                old2 = hj
                h[i] = newi
                h[j] = newj
                r[i] = 1.0 - newi
                r[j] = 1.0 - newj
                raw = float(np.max(np.correlate(h, r, mode="full")))
                if raw + improve_tol < best_raw:
                    best_raw = raw
                    best_i = i
                    best_j = j
                    best_new1 = newi
                    best_new2 = newj
                    best_old1 = old1
                    best_old2 = old2
                h[i] = hi
                h[j] = hj
                r[i] = ri
                r[j] = rj
            if hi_minus >= -eps and hj + d <= 1.0 + eps:
                newi = hi_minus if hi_minus > 0.0 else 0.0
                newj = hj + d if hj + d < 1.0 else 1.0
                old1 = hi
                old2 = hj
                h[i] = newi
                h[j] = newj
                r[i] = 1.0 - newi
                r[j] = 1.0 - newj
                raw = float(np.max(np.correlate(h, r, mode="full")))
                if raw + improve_tol < best_raw:
                    best_raw = raw
                    best_i = i
                    best_j = j
                    best_new1 = newi
                    best_new2 = newj
                    best_old1 = old1
                    best_old2 = old2
                h[i] = hi
                h[j] = hj
                r[i] = ri
                r[j] = rj
    if best_i >= 0 and best_raw + improve_tol < old_raw:
        h[best_i] = best_new1
        h[best_j] = best_new2
        h = project_box_sum(h, n)
        new_c = compute_c5(h, n)
        if new_c + 1e-13 < old_c:
            return True, h, new_c
        h[best_i] = best_old1
        h[best_j] = best_old2
        h = project_box_sum(h, n)
        return False, h, compute_c5(h, n)
    return False, h, old_c


def full_pair_refine(h, n, deadline, deltas):
    h = project_box_sum(np.asarray(h, dtype=float).reshape(-1), n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if d <= 0.0:
            continue
        while time.time() < deadline - 1.0:
            c = compute_c5(h, n)
            if c + 1e-15 < best_c:
                best_c = c
                best_h = h.copy()
            changed, h, c2 = pair_scan_once(h, n, float(d), c, deadline)
            if not changed:
                break
            c2 = compute_c5(h, n)
            if c2 + 1e-15 < best_c:
                best_c = c2
                best_h = h.copy()
    best_h = project_box_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def coarse_vector(params):
    if _CE_N is None or _CE_X is None or _CE_U is None:
        return np.full(16, 0.5, dtype=float)
    params = np.asarray(params, dtype=float)
    x = np.interp(_CE_U, _CE_X, params)
    return project_box_sum(x, _CE_N)


def coarse_objective(params):
    if _CE_N is None or _CE_X is None or _CE_U is None:
        return 1.0
    params = np.asarray(params, dtype=float)
    x = np.interp(_CE_U, _CE_X, params)
    h = project_box_sum(x, _CE_N)
    return compute_c5(h, _CE_N)


def make_coarse_start(p, seed, maxiter=30, popsize=12):
    global _CE_N, _CE_X, _CE_U
    if p < 3:
        p = 3
    n = 2 * int(p - 1)
    if n < 4:
        n = 4
    x = np.linspace(0.0, 2.0, n)
    _CE_N = n
    _CE_X = np.linspace(0.0, 1.0, p)
    _CE_U = np.minimum(x, 2.0 - x)
    try:
        res = spopt.differential_evolution(
            coarse_objective,
            [(0.0, 1.0)] * p,
            seed=int(seed),
            maxiter=int(maxiter),
            popsize=int(max(3, popsize)),
            tol=1e-4,
            mutation=(0.5, 1.5),
            recombination=0.9,
            polish=False
        )
        h = coarse_vector(res.x)
        return project_box_sum(h, n)
    except Exception:
        rng = np.random.RandomState(int(seed))
        params = np.clip(rng.rand(p), 0.0, 1.0)
        h = coarse_vector(params)
        return project_box_sum(h, n)


def default_tent(n):
    x = np.linspace(0.0, 2.0, n)
    h = np.minimum(x, 2.0 - x)
    return project_box_sum(h, n)
