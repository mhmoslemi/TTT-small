def penalty_obj(x, n, alpha, mu, K):
    c = compute_c5_vec(x, n)
    if K is not None and K < len(c):
        idx = np.argpartition(c, -K)[-K:]
        c_sub = c[idx]
        m = c_sub.max()

    if K is not None and K < len(c):

        mx = sh.max()
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:

    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def momentum_grad_descent(h, n, deadline, lr=0.005, alpha_init=200, momentum=0.9, K=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_vec = np.zeros(n)
    alpha = alpha_init
    lr_cur = lr
    stall_count = 0
    
    while time.time() < deadline:
        g, c = grad_corr(h, n, alpha, K)
        c_max = c.max()
        
        # Nesterov-like: update momentum
        m_vec = momentum * m_vec + (1 - momentum) * g
        m_norm = np.linalg.norm(m_vec)
        if m_norm < 1e-14:
            break
        m_normed = m_vec / m_norm
        
        # Line search
        improved = False
        step = lr_cur
        for _ in range(30):
            h_new = project_feasible(h - step * m_normed, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr_cur *= 0.5
            alpha *= 1.3
            stall_count += 1

                break
        else:
            stall_count = 0
            # Occasionally increase alpha

    return best_h, best_c

def targeted_pair_search(h, n, deadline, base_deltas=None):

    if base_deltas is None:

            c_max = c.max()
            # Find active correlations (within 5% of max)

            improved = False
            # Try swaps guided by gradient: move mass from high-g to low-g
            indices = np.argsort(g)  # low gradient first (good to increase)
            # Try a subset of swaps
            n_try = min(n, 100)
            for i in range(n_try):
                if time.time() > deadline:
                    return h, best_c
                idx_low = indices[i]  # want to increase h here
                # Find indices where we want to decrease
                for j in range(n - n_try, n):
                    idx_high = indices[j]  # want to decrease h here
                    if idx_low == idx_high:

                    # Try moving d from idx_high to idx_low
                    if h[idx_high] - d >= 0 and h[idx_low] + d <= 1:
                        h[idx_low] += d
                        h[idx_high] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            improved_any = True
                        else:
                            h[idx_low] -= d
                            h[idx_high] += d
                    # Try reverse
                    if h[idx_low] - d >= 0 and h[idx_high] + d <= 1:
                        h[idx_low] -= d
                        h[idx_high] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            improved_any = True
                        else:
                            h[idx_low] += d
                            h[idx_high] -= d
            if not improved:

    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_corr(h, n, alpha, K=None):
    c = compute_c5_vec(h, n)
    nc = len(c)
    if K is not None and K < nc:
        idx = np.argpartition(c, -K)[-K:]
        c_sub = c[idx]
        c_max = c_sub.max()
        sh = alpha * (c_sub - c_max)
        mx = sh.max()
        w_sub = np.exp(sh - mx)
        w_sub /= w_sub.sum()
        w = np.zeros(nc)
        w[idx] = w_sub
    else:
        c_max = c.max()
        sh = alpha * (c - c_max)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(nc):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad, c

def _pen_obj(x, n, alpha, mu, K):
    c = compute_c5_vec(x, n)
    nc = len(c)
    if K is not None and K < nc:
        idx = np.argpartition(c, -K)[-K:]
        c_sub = c[idx]
        m = c_sub.max()
        sh = alpha * (c_sub - m)
    else:
        m = c.max()
        sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _pen_grad(x, n, alpha, mu, K):
    g, _ = grad_corr(x, n, alpha, K)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slq_obj(x, n, alpha, K):
    c = compute_c5_vec(x, n)
    nc = len(c)
    if K is not None and K < nc:
        idx = np.argpartition(c, -K)[-K:]
        c_sub = c[idx]
        m = c_sub.max()
        sh = alpha * (c_sub - m)
    else:
        m = c.max()
        sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _slq_con(x, n):
    return float(x.sum() - n / 2.0)

def _slq_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alpha_schedule=None, K=None):
    if alpha_schedule is None:
        alpha_schedule = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    nc = 2 * n - 1
    for i, alpha in enumerate(alpha_schedule):
        if time.time() > deadline - 5:
            break
        mu = max(0.5, alpha * 0.005)
        Ki = K if (K is not None and i >= 2) else None
        if Ki is None and i >= 4:
            Ki = max(50, nc // 3)
        t_remaining = deadline - time.time()
        n_left = len(alpha_schedule) - i
        t_per = max(15, min(45, t_remaining / max(1, n_left) - 5))
        n_iter = max(20, min(150, int(t_per * 2.5)))
        try:
            res = minimize(_pen_obj, h, jac=_pen_grad,
                          args=(n, alpha, mu, Ki), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': n_iter, 'ftol': 1e-19,
                                   'gtol': 1e-15, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=300000, K=150, maxiter=2000):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con, 'jac': _slq_jac, 'args': (n,)}]
    try:
        res = minimize(_slq_obj, h, args=(n, alpha, K), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def momentum_gd(h, n, deadline, lr_init=0.004, alpha_init=300, mom=0.88, K=100,

    best_c = compute_c5(h, n)
    best_h = h.copy()
    mv = np.zeros(n)
    alpha = float(alpha_init)
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        g, c = grad_corr(h, n, alpha, K)
        cmax = c.max()
        mv = mom * mv + (1.0 - mom) * g
        gn = np.linalg.norm(mv)
        if gn < 1e-15:
            break
        mvd = mv / gn
        improved = False
        step = lr
        for _ in range(35):
            h_new = project_feasible(h - step * mvd, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break

            if alpha < alpha_max and cmax > best_c * 1.005:

            lr *= 0.4
            alpha = min(alpha * 1.25, alpha_max)
            stall += 1
            if stall > max_stall or lr < lr_min:
                break
    return best_h, best_c

        h_hi = interp_step(best_h, best_n, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        t5 = min(time.time() + 120, deadline - 120)
        h5, c5 = lbfgs_optimize(h_hi, n_hi, t5,
            alpha_schedule=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000])
        if c5 < best_c:
            best_c, best_h, best_n = c5, h5, n_hi

    # Phase 6: SLSQP at current best
    if time.time() < deadline - 100:
        t6 = min(time.time() + 55, deadline - 80)
        K6 = 100 if best_n > 400 else 200
        h6, c6 = slsqp_optimize(best_h, best_n, t6, alpha=1000000, K=K6, maxiter=3000)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Momentum GD at best
    if time.time() < deadline - 70:
        t7 = min(time.time() + 40, deadline - 50)
        K7 = 60 if best_n > 400 else 100
        h7, c7 = momentum_gd(best_h, best_n, t7, lr_init=0.002, alpha_init=3000,
                            mom=0.92, K=K7, max_stall=40, alpha_max=2000000)
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Phase 8: Gradient local search at best
    if time.time() < deadline - 40:
        t8 = min(time.time() + 30, deadline - 25)
        nc8 = 25 if best_n > 400 else 30
        h8, c8 = grad_local_search(best_h, best_n, t8, n_cand=nc8,
                                   deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c8 < best_c:
            best_c, best_h = c8, h8

    # Phase 9: Fine pair search (radius-limited for speed)
    if time.time() < deadline - 5:
        t9 = deadline - 2
        rad = 40 if best_n > 400 else 60
        h9, c9 = local_pair_search(best_h, best_n, t9,
                                    [0.0005, 0.0002, 0.0001, 0.00005],
                                    radius=rad)
        if c9 < best_c:
            best_c, best_h = c9, h9

    # Final
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
