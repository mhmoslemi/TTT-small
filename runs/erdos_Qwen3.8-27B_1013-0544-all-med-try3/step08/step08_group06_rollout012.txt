            if tot < 1e-14:

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    return float(max_c + np.log(exp_c.sum()) / alpha)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k] * dx
        if pk < 1e-18:
            continue
        m = k - (n - 1)
        # First term: (1-x[j-m]) for valid j
        j_lo = max(0, m)
        j_hi = min(n - 1, m + n - 1)
        if j_lo <= j_hi:

        j2_hi = min(n - 1, n - 1 - m)

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    s = exp_c.sum()
    if s < 1e-300:
        return float(max_c)
    return float(max_c + np.log(s) / alpha)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    s = exp_c.sum()
    if s < 1e-300:
        return np.zeros(n)
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k] * dx
        if pk < 1e-20:
            continue
        m = k - (n - 1)
        j_lo = max(0, m)
        j_hi = min(n - 1, m + n - 1)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += pk * (1.0 - x[j_lo - m: j_hi + 1 - m])
        j2_lo = max(0, -m)
        j2_hi = min(n - 1, n - 1 - m)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += pk * (-x[j2_lo + m: j2_hi + 1 + m])
    return grad

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jacobian(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):

def pair_search(h, n, deadline, deltas, max_passes=80):

    best_h = h.copy()

                return best_h, best_c

                        improved = True
                    else:

                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi2 + d, hj2 - d
            if not improved:
                break
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht, c = slsqp_optimize(best_h, n, a, maxiter=2000, ftol=1e-16)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    if n <= 250:
        if time.time() < deadline - 10:
            best_h, best_c = pair_search(best_h, n, min(deadline - 5, time.time() + 60),
                                         [0.01, 0.005, 0.002, 0.001], max_passes=40)
    else:
        if time.time() < deadline - 10:
            best_h, best_c = near_pair_search(best_h, n, min(deadline - 5, time.time() + 50),
                                              [0.01, 0.005, 0.002], radius=50, max_passes=40)

    if time.time() < deadline - 5:
        ht, c = slsqp_optimize(best_h, n, 500000, maxiter=3000, ftol=1e-17)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    if n <= 250:
        if time.time() < deadline - 2:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.005, 0.002, 0.001, 0.0005], max_passes=50)
    else:
        if time.time() < deadline - 2:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.005, 0.002, 0.001], radius=50, max_passes=50)

    if time.time() < deadline - 2:
        ht, c = slsqp_optimize(best_h, n, 1000000, maxiter=4000, ftol=1e-17)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    if h_init is None:
        h_init = np.full(n_init, 0.5)
        h_init = project_feasible(h_init, n_init)

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at initial resolution with analytical gradient SLSQP
    alloc1 = min(300, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Try other resolutions
    x_init = np.linspace(0, 2, n_init)
    other_ns = [100, 150, 300, 400]
    for n_try in other_ns:
        if n_try == n_init:
            continue
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time()
        alloc = min(80, remaining * 0.15)
        if alloc < 20:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c - 1e-8:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Random restarts from best solution
    if time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.03 + 0.1 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 15) * 0.35)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final intensive pair search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.002, 0.001, 0.0005, 0.0002], max_passes=60)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.002, 0.001, 0.0005], radius=40, max_passes=60)

    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 3:
        n = len(best_h)
        ht, c = slsqp_optimize(best_h, n, 2000000, maxiter=3000, ftol=1e-18)
        if c < best_c:
            best_c = c
            best_h = ht

    # Final projection and evaluation
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
