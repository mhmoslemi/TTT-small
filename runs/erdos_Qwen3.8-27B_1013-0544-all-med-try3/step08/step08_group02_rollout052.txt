def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
            h = np.clip(h, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, active, mu):
    corr = compute_corr(x)
    vals = corr[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, active, mu):
    corr = compute_corr(x)
    vals = corr[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def hinge_obj(x, n, target_c):
    corr = compute_corr(x) * (2.0 / n)
    excess = np.maximum(0.0, corr - target_c)
    return float(np.sum(excess ** 2))

def hinge_grad(x, n, target_c):
    corr = compute_corr(x) * (2.0 / n)
    excess = np.maximum(0.0, corr - target_c)
    active_idx = np.where(excess > 1e-15)[0]
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in active_idx:
        k = ii - (n - 1)
        wm = 2.0 * excess[ii] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_lbfgs(h, n, deadline, alphas, mu, maxiters):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        corr = compute_corr(best_h)
        max_val = corr.max()
        thresh = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(corr >= thresh)[0]
        if len(active) > 30:
            order = np.argsort(-corr[active])[:30]
            active = active[order]
        maxit = maxiters[i] if i < len(maxiters) else 3000
        maxit = min(maxit, max(500, int(remaining * 3)))
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad,
                          args=(n, alpha, active, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 6,
                                   'maxls': 40, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_hinge_bisection(h, n, deadline, c_lo=None, c_hi=None):
    if c_lo is None:
        c_lo = 0.3
    if c_hi is None:
        c_hi = compute_c5(h, n)
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for iteration in range(12):
        remaining = deadline - time.time()
        if remaining < 8:
            break
        mid_c = (c_lo + c_hi) / 2.0
        if mid_c >= best_c:
            c_lo = min(c_hi, best_c)
            c_hi = max(c_lo + 0.0001, best_c + 0.001)
            continue
        maxit = max(300, int(remaining * 1.5))
        try:
            res = minimize(hinge_obj, best_h, jac=hinge_grad,
                          args=(n, mid_c), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-24,
                                   'gtol': 1e-18, 'maxfun': maxit * 5,
                                   'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            if c_new <= mid_c + 1e-9:
                c_hi = mid_c
            else:
                c_lo = mid_c
        except Exception:
            c_lo = mid_c
    return best_h, best_c

def coordinate_descent(h, n, deadline, n_inner=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if best_h[i] < 0.02 or best_h[i] > 0.98:
                    continue
                for j in range(n):
                    if i == j:
                        continue
                    if time.time() > deadline:
                        break
                    if best_h[j] < 0.02 or best_h[j] > 0.98:
                        continue
                    h2 = best_h.copy()
                    h2[i] += d
                    h2[j] -= d
                    if h2[i] > 1.0 or h2[j] < 0.0:
                        continue
                    c2 = compute_c5(h2, n)
                    if c2 < best_c - 1e-14:
                        best_c = c2
                        best_h = h2
                        improved = True
                        break
                    h2[i] -= 2 * d
                    h2[j] += 2 * d
                    if h2[i] < 0.0 or h2[j] > 1.0:
                        continue
                    c2 = compute_c5(h2, n)
                    if c2 < best_c - 1e-14:
                        best_c = c2
                        best_h = h2
                        improved = True
                        break
    return best_h, best_c

def subgrad_step(h, n, eta, active_frac=0.002):
    corr = compute_corr(h)
    c_max = corr.max()
    tol = max(1e-10, active_frac * c_max)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    grad /= len(active)
    return h - eta * grad

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    stalled = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        h2 = subgrad_step(h, n, eta)
        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < best_c - 1e-13:
            best_c = c2
            best_h = h2.copy()
            h = h2
            eta = min(eta * 1.003, 0.015)
            stalled = 0
        else:
            h = h2
            eta *= 0.97
            stalled += 1
            if stalled > 200:
                eta = max(eta, 0.001)
                stalled = 0
    return best_h, best_c

def perturb(h, n, rng, strength=0.02):
    h2 = h.copy()
    mask = (h2 > 0.05) & (h2 < 0.95)
    if mask.sum() < n // 4:
        mask = (h2 > 0.01) & (h2 < 0.99)
    idx = np.where(mask)[0]
    if len(idx) == 0:
        return h2
    k = min(len(idx), max(5, n // 20))
    chosen = rng.choice(idx, size=k, replace=False)
    for i in chosen:
        h2[i] += rng.uniform(-strength, strength)
    return project_feasible(h2, n)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=1000):
    def sl_obj(x):
        corr = compute_corr(x)
        m = corr.max()
        sh = alpha * (corr - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sl_jac(x):
        corr = compute_corr(x)
        m = corr.max()
        sh = alpha * (corr - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w = w / w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for wi, idx in zip(w, np.arange(2 * n - 1)):
            if wi < 1e-15:
                continue
            k = idx - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        rem = int(deadline - time.time()) - 2
        maxit = min(maxiter, max(200, rem * 40))
        res = minimize(sl_obj, h, jac=sl_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: L-BFGS annealing at original resolution (250s)
    p1_end = min(t0 + 250, deadline - 400)
    alphas1 = [20, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    iters1 = [2000, 2500, 3000, 3000, 3500, 3500, 4000, 4000, 5000]
    h1, c1 = run_lbfgs(best_h, best_n, p1_end, alphas1, 0.005, iters1)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Hinge bisection at original resolution (100s)
    if time.time() < deadline - 300:
        p2_end = min(time.time() + 100, deadline - 300)
        h2, c2 = run_hinge_bisection(best_h, best_n, p2_end, c_hi=best_c)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: Subgradient refinement (80s)
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 80, deadline - 250)
        h3, c3 = subgrad_refine(best_h, best_n, p3_end, n_iter=8000)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: Try n=750 upsampling (150s)
    n_up = 750
    if time.time() < deadline - 200:
        x_fine = np.linspace(0, 2, n_up)
        x_coarse = np.linspace(0, 2, best_n)
        h_up = np.interp(x_fine, x_coarse, best_h)
        h_up = project_feasible(h_up, n_up)
        c_up = compute_c5(h_up, n_up)
        p4_end = min(time.time() + 150, deadline - 200)
        alphas4 = [50, 500, 5000, 50000, 500000]
        iters4 = [2000, 2500, 3000, 3000, 4000]
        h4, c4 = run_lbfgs(h_up, n_up, p4_end, alphas4, 0.003, iters4)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
            best_n = n_up

    # Phase 5: Multiple perturbations + short L-BFGS (200s)
    if time.time() < deadline - 180:
        n_trials = 6
        p5_end = deadline - 180
        per_trial = (p5_end - time.time()) / n_trials
        for trial in range(n_trials):
            if time.time() > p5_end:
                break
            strength = 0.01 + 0.01 * (trial % 3)
            h_pert = perturb(best_h, best_n, rng, strength)
            h_pert = project_feasible(h_pert, best_n)
            a5 = [100, 10000, 500000]
            i5 = [1500, 2000, 2500]
            h_t, c_t = run_lbfgs(h_pert, best_n, min(time.time() + per_trial, p5_end), a5, 0.01, i5)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t.copy()

    # Phase 6: Active set + high alpha L-BFGS at best (100s)
    if time.time() < deadline - 130:
        p6_end = min(time.time() + 100, deadline - 130)
        corr = compute_corr(best_h)
        cmax = corr.max()
        for frac in [0.001, 0.003, 0.005]:
            if time.time() > p6_end - 5:
                break
            thresh = cmax - frac * cmax
            active = np.where(corr >= thresh)[0]
            if len(active) > 20:
                order = np.argsort(-corr[active])[:20]
                active = active[order]
            a6 = [200000, 2000000, 20000000]
            i6 = [2000, 3000, 4000]
            h6, c6 = run_lbfgs(best_h, best_n, min(time.time() + 35, p6_end), a6, 0.002, i6)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Hinge bisection on best (60s)
    if time.time() < deadline - 80:
        p7_end = min(time.time() + 60, deadline - 80)
        h7, c7 = run_hinge_bisection(best_h, best_n, p7_end, c_hi=best_c)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()

    # Phase 8: Coordinate descent (60s)
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 60, deadline - 30)
        h8, c8 = coordinate_descent(best_h, best_n, p8_end, n_inner=2)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()

    # Phase 9: Final SLSQP polish (25s)
    if time.time() < deadline - 3:
        p9_end = deadline - 2
        h9, c9 = slqp_polish(best_h, best_n, p9_end, alpha=1000000, maxiter=800)
        if c9 < best_c:
            best_c = c9
            best_h = h9.copy()

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
