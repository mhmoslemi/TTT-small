```python
def compute_delta_a(h, a, n):
    """Change in correlation vector when h[a] goes 0→1."""
    delta = np.zeros(2*n-1)
    # First part: new contribution at offset m for m where 0<=a+m<n
    m_min = 0  # m >= -a (always true since m >= -(n-1) and a >= 0)
    # Actually m ranges from -(n-1) to n-1
    # 0 <= a+m < n means -a <= m < n-a
    lo = max(-(n-1), -a)
    hi = min(n-1, n-1-a)  # m < n-a means m <= n-1-a
    if lo <= hi:
        offsets = np.arange(lo, hi+1)
        indices = a + offsets  # = a+m, ranges from a+lo to a+hi
        delta[offsets + (n-1)] += (1 - h[indices])
    # Second part: removal of contribution at offset m for m where 0<=a-m<n
    # 0 <= a-m < n means a-n < m <= a
    lo2 = max(-(n-1), a-n+1)
    hi2 = min(n-1, a)
    if lo2 <= hi2:
        offsets = np.arange(lo2, hi2+1)
        indices = a - offsets  # = a-m
        delta[offsets + (n-1)] -= h[indices]
    return delta
```
