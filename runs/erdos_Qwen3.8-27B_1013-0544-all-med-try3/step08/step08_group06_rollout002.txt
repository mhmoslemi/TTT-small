    best_h = h.copy()

        if it % 1000 == 0 and time.time() > deadline:
            break

        c_max = c.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()

        eps = 0.01 * c_max + 1e-10
        active = np.where(c >= c_max - eps)[0]

            active = np.array([np.argmax(c)])

                elif t_plus_L < 0 or t_plus_L >= n:
                    contrib -= h[t_minus_L]
                else:
                    contrib += (1.0 - h[t_plus_L]) - h[t_minus_L]
                g[t] += contrib
        g /= len(active)
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.0001 * it)
        
        # Projected step
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:

            h_new2 = project_feasible(h_new2, n)

            if c_new2 < best_c:
                best_c = c_new2
                best_h = h_new2.copy()
                h = h_new2
            else:
                break  # No improvement

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smoothed_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 50:
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scales = [0.03, 0.05, 0.08, 0.12, 0.02]
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 20) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_single(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final intensive polish at best resolution
    if best_h is not None and time.time() < deadline - 5:
        n_fin = len(best_h)
        remaining = deadline - time.time() - 3
        
        if remaining > 30:
            ht = slsqp_optimize(best_h, n_fin, 2000000, maxiter=3000, ftol=1e-16)
            c = compute_c5(ht, n_fin)
            if c < best_c:
                best_c = c
                best_h = ht
        
        if remaining > 20:
            best_h, best_c = subgradient_descent(
                best_h, n_fin, time.time() + min(40, remaining * 0.4),
                n_iter=60000, lr_init=0.001, tol_frac=0.002)
        
        if remaining > 15:
            best_h, best_c = micro_pair_search(best_h, n_fin, time.time() + min(30, remaining * 0.4))
        
        if remaining > 10:
            ht = slsqp_optimize(best_h, n_fin, 5000000, maxiter=4000, ftol=1e-16)
            c = compute_c5(ht, n_fin)
            if c < best_c:
                best_c = c
                best_h = ht
        
        if remaining > 5:
            best_h, best_c = micro_pair_search(best_h, n_fin, time.time() + min(10, remaining * 0.4))

    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_parent, 0.5), n_parent)
        best_c = compute_c5(best_h, n_parent)
        n_best = n_parent

    return best_h, best_c, len(best_h)
