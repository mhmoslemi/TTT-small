import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    n_half = n // 2
    sorted_idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[sorted_idx[:n_half]] = 1.0
    return hb

def compute_F(h, n):
    length = 2 * n - 1
    F = np.zeros((n, length))
    idxs = np.arange(length)
    m_vals = idxs - (n - 1)
    for p in range(n):
        pm = p - m_vals
        mask1 = (pm >= 0) & (pm < n)
        F[p, mask1] += 1.0 - h[pm[mask1]]
        pm2 = p + m_vals
        mask2 = (pm2 >= 0) & (pm2 < n)
        F[p, mask2] -= h[pm2[mask2]]
    return F

def binary_local_search(h_init, n, deadline, max_sweeps=3000, n_restarts=5, seed=42):
    n_half = n // 2
    length = 2 * n - 1
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = to_binary(h_init, n)
    C = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c5 = C.max() * dx
    best_h = h.copy()
    best_c5 = cur_c5
    F = compute_F(h, n)
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            k = rng.randint(2, 8)
            for _ in range(k):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) == 0 or len(zi) == 0:
                    break
                a = int(rng.choice(oi))
                b = int(rng.choice(zi))
                h[a], h[b] = h[b], h[a]
            C = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c5 = C.max() * dx
            F = compute_F(h, n)
        stagnant = 0
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            F_ones = F[ones]
            best_val = cur_c5 + 1.0
            best_i = -1
            best_j = -1
            for ii in range(len(zeros)):
                zi = zeros[ii]
                Gi = C + F[zi]
                vals = (Gi[None, :] - F_ones).max(axis=1)
                bj = int(np.argmin(vals))
                if vals[bj] < best_val:
                    best_val = vals[bj]
                    best_j = bj
                    best_i = zi
                if time.time() > deadline:
                    break
            if best_i < 0:
                break
            j_val = ones[best_j]
            delta = F[best_i] - F[j_val].copy()
            idx1 = best_i - j_val + (n - 1)
            idx2 = j_val - best_i + (n - 1)
            if 0 <= idx1 < length:
                delta[idx1] += 2.0
            if 0 <= idx2 < length:
                if idx2 != idx1:
                    delta[idx2] -= 2.0
            new_C_max = (C + delta).max() * dx
            if new_C_max < cur_c5 - 1e-14:
                h[best_i] = 1.0
                h[j_val] = 0.0
                C = C + delta
                cur_c5 = new_C_max
                F = compute_F(h, n)
                stagnant = 0
            else:
                stagnant += 1
                if stagnant > 50:
                    break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_smooth(h_init, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    length = 2 * n - 1
    idxs = np.arange(length)
    m_vals = idxs - (n - 1)

    def obj_and_grad(x):
        hh = np.clip(x, 0.0, 1.0)
        C = np.correlate(hh, 1 - hh, mode='full')
        g = C * dx
        m = g.max()
        shifted = alpha * (g - m)
        shifted = np.clip(shifted, -60.0, 0.0)
        max_s = shifted.max()
        exps = np.exp(shifted - max_s)
        log_sum = max_s + np.log(exps.sum())
        obj_val = m + log_sum / alpha
        softmax = exps / exps.sum()
        grad = np.zeros(n)
        for i in range(n):
            im = i - m_vals
            mask1 = (im >= 0) & (im < n)
            term1 = np.dot(softmax[mask1], 1.0 - hh[im[mask1]])
            ik = i + m_vals
            mask2 = (ik >= 0) & (ik < n)
            term2 = np.dot(softmax[mask2], hh[ik[mask2]])
            grad[i] = dx * (term1 - term2)
        return obj_val, grad

    def obj(x):
        v, _ = obj_and_grad(x)
        return v

    def jac(x):
        _, j = obj_and_grad(x)
        return j

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_init, n)
