def compute_delta(h, a, b, n):
    """Compute change in correlation when h[a] 0→1, h[b] 1→0"""
    delta = np.zeros(2*n - 1)
    # k ranges from -(n-1) to (n-1), stored at index k + n - 1
    # C[k] = sum_i h[i+k] * (1-h[i]) for valid i, i+k in [0, n-1]
    
    for k in range(-(n-1), n):
        ki = k + n - 1  # index in delta array
        affected = set()
        # i=a
        if 0 <= a < n and 0 <= a+k < n:
            affected.add(a)
        # i=b
        if 0 <= b < n and 0 <= b+k < n:
            affected.add(b)
        # i=a-k (first factor at position a)
        ik = a - k
        if 0 <= ik < n and 0 <= a < n:
            affected.add(ik)
        # i=b-k (first factor at position b)
        ik = b - k
        if 0 <= ik < n and 0 <= b < n:
            affected.add(ik)
        
        for i in affected:
            # new term
            h_new_ik = 1.0 if (i+k == a) else (0.0 if (i+k == b) else h[i+k])
            omh_new_i = 0.0 if i == a else (1.0 if i == b else (1.0 - h[i]))
            new_term = h_new_ik * omh_new_i
            # old term
            old_term = h[i+k] * (1.0 - h[i])
            delta[ki] += new_term - old_term
    
    return delta

def swap_delta(h, a, b, n):
    """Compute C' - C when h[a]:0→1, h[b]:1→0. Returns delta array of length 2n-1."""
    # k ranges from -(n-1) to n-1, stored at index k + (n-1)
    # C[k] = sum_i h[i+k]*(1-h[i])
    
    delta = np.zeros(2*n-1)
    offset = n - 1
    
    # Contribution from i=a (second factor change: (1-h)[a]: 1→0)
    # For valid k where i+k = a+k is in [0,n-1]:
    # delta[k] -= h[a+k] (old) but new value is 0 since (1-h')[a]=0
    # But h'[a+k] might differ if a+k==b
    k_range = np.arange(-(n-1), n)
    
    # Term at i=a: -(h[a+k]) for valid a+k, but adjust if a+k==b
    # a+k valid: -(n-1) ≤ k ≤ n-1-a... wait, a+k in [0,n-1] means k in [-a, n-1-a]
    ka_min = max(-(n-1), -a)
    ka_max = min(n-1, n-1-a)
    if ka_min <= ka_max:
        k_idx = np.arange(ka_min, ka_max+1)
        vals = h[a + k_idx]  # h[a+k] for these k
        # Adjust: if a+k == b (k = b-a), the new value of h'[a+k] = h'[b] = 0
        # But we're computing the CHANGE, which is: new_term - old_term = 0 - h[a+k]*(1-h[a])
        # Since h[a]=0, old_term = h[a+k]*1 = h[a+k], new_term = 0
        # So delta += -h[a+k] (regardless of what h' is)
        delta[k_idx + offset] += -vals
    
    # Term at i=b: +(h'[b+k]) for valid b+k
    kb_min = max(-(n-1), -b)
    kb_max = min(n-1, n-1-b)
    if kb_min <= kb_max:
        k_idx = np.arange(kb_min, kb_max+1)
        vals = h[b + k_idx].copy()  # h[b+k]
        # Adjust: if b+k == a (k = a-b), h'[b+k] = h'[a] = 1
        # if b+k == b (k = 0), h'[b+k] = h'[b] = 0
        mask_a = k_idx == (a - b)
        mask_b = k_idx == 0
        vals[mask_a] = 1.0
        vals[mask_b] = 0.0
        delta[k_idx + offset] += vals
    
    # Term at i=a-k (first factor at position a: h[a] 0→1)
    # i = a-k, valid: 0≤a-k<n, i.e., a-n+1≤k≤a. And i+k=a valid: always.
    # But we need (1-h')[a-k]:
    #   if a-k==a (k=0): (1-h')[a]=0
    #   if a-k==b (k=a-b): (1-h')[b]=1
    #   else: (1-h[a-k])
    kc_min = max(-(n-1), a-n+1)
    kc_max = min(n-1, a)
    if kc_min <= kc_max:
        k_idx = np.arange(kc_min, kc_max+1)
        i_vals = a - k_idx  # these are the i values
        # (1-h')[i]: 0 if i==a, 1 if i==b, else 1-h[i]
        vals = 1.0 - h[i_vals]
        vals[i_vals == a] = 0.0
        vals[i_vals == b] = 1.0
        # new term = h'[a] * (1-h')[i] = 1 * vals = vals
        # old term = h[a] * (1-h)[i] = 0 * (1-h)[i] = 0
        # delta = vals - 0 = vals
        delta[k_idx + offset] += vals
    
    # Term at i=b-k (first factor at position b: h[b] 1→0)
    # i = b-k, valid: 0≤b-k<n, i.e., b-n+1≤k≤b
    kd_min = max(-(n-1), b-n+1)
    kd_max = min(n-1, b)
    if kd_min <= kd_max:
        k_idx = np.arange(kd_min, kd_max+1)
        i_vals = b - k_idx
        # new term = h'[b] * (1-h')[i] = 0
        # old term = h[b] * (1-h)[i] = 1 * (1-h)[i] = 1-h[i]
        #   if i==a: 1-h[a]=1
        #   if i==b: 1-h[b]=0
        # delta = 0 - old_term = -(1-h[i])
        vals = 1.0 - h[i_vals]
        vals[i_vals == a] = 1.0  # 1-h[a] = 1
        vals[i_vals == b] = 0.0  # 1-h[b] = 0
        delta[k_idx + offset] += -vals
    
    return delta

The issue: the set of affected i's is {a, b, a-k, b-k}. If any of these coincide, we should only count once.

For k=0: a-k=a and b-k=b, so the set is {a, b}. We should only compute the delta at i=a and i=b.
