def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # distribute among free variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)

            h = np.clip(h, 0, 1)

            if tot < 1e-16:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_c(x, n):
    return x.sum() - n / 2.0

def _sum_j(x, n):
    return np.ones(n)

def obj_logsumexp(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    if sm > 400:
        return float(m + (sm + np.log(np.sum(np.exp(sh - sm)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def obj_topk(x, n, idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    ca = c[idx]
    m = ca.max()
    sh = alpha * (ca - m)
    sm = sh.max()
    if sm > 400:
        return float(m + (sm + np.log(np.sum(np.exp(sh - sm)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp_full(h, n, alpha, maxiter=5000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_logsumexp, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_topk(h, n, idx, alpha, maxiter=5000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_topk, h, args=(n, idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def get_topk_idx(h, n, K):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return np.argsort(c)[::-1][:min(K, len(c))]

def fine_pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def subgradient_method(h, n, deadline, n_iters=2000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.05
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        max_val = c.max()
        thresh = max_val - 0.002
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            if k >= 0:
                a = h[:n - k]
                b = h[k:n]
                grad[:n - k] += (1.0 - b) * dx
                grad[k:n] += (-a) * dx
            else:
                j = -k
                a = h[j:n]
                b = h[:n - j]
                grad[j:n] += (1.0 - b) * dx
                grad[:n - j] += (-a) * dx
        grad /= len(active)
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr = max(0.005, lr * 0.998)
    return best_h, best_c

def anneal_slsqp(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [200, 800, 3000, 12000, 50000, 200000, 800000, 3000000, 10000000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        iters = 5000 if a < 10000 else (3000 if a < 500000 else 2000)
        ht = slsqp_full(h, n, a, maxiter=iters)
        ct = compute_c5(ht, n)
        if ct < best_c - 1e-16:
            best_c = ct
            h = ht
    return h, best_c

def active_set_optimize(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for K in [5, 10, 20, 40, 80]:
        if time.time() > deadline - 8:
            break
        active = get_topk_idx(h, n, K)
        for a in [5000, 50000, 500000, 5000000]:
            if time.time() > deadline - 5:
                break
            ht = slsqp_topk(h, n, active, a, maxiter=3000)
            ct = compute_c5(ht, n)
            if ct < best_c - 1e-16:
                best_c = ct
                h = ht
                active = get_topk_idx(h, n, K)
    return h, best_c

def try_optimize(h_in, n, deadline):
    h = project_feasible(h_in.copy(), n)
    best_c = compute_c5(h, n)
    
    # Annealed SLSQP
    h, c = anneal_slsqp(h, n, min(deadline, time.time() + 200))
    if c < best_c:
        best_c = c
    
    # Active set
    h, c = active_set_optimize(h, n, min(deadline, time.time() + 150))
    if c < best_c:
        best_c = c
        h = h
    
    # Subgradient
    if time.time() < deadline - 20:
        h_sg, c_sg = subgradient_method(h, n, time.time() + 40, n_iters=1000)
        if c_sg < best_c:
            best_c = c_sg
            h = h_sg
    
    # Fine local search
    remaining = deadline - time.time()
    if remaining > 10:
        if n <= 130:
            h, best_c = fine_pair_search(h, n, deadline - 2,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            h, best_c = fine_pair_search(h, n, deadline - 2,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002],
                                         radius=max(30, n // 8))
    
    # Final ultra-high alpha SLSQP
    if time.time() < deadline - 3:
        ht = slsqp_full(h, n, 20000000, maxiter=10000)
        ct = compute_c5(ht, n)
        if ct < best_c:
            best_c = ct
            h = ht
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
