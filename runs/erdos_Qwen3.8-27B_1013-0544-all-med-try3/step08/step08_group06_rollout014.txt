    log_sum = c_max + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / beta

        # Second sum: x[i-ell] for ell where 0<=i-ell<n
        # ell ranges from max(-(n-1), i-(n-1)) to min(n-1, i)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, beta, penalty=50.0):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = c.max()
    shifted = beta * (c - c_max)
    sm = shifted.max()
    lse = c_max + (sm + np.log(np.sum(np.exp(shifted - sm)))) / beta
    pen = penalty * (x.sum() - n / 2.0) ** 2
    return lse + pen

def grad_lse(x, n, beta, penalty=50.0):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = c.max()
    shifted = beta * (c - c_max)
    sm = shifted.max()
    ew = np.exp(shifted - sm)
    w = ew / ew.sum()
    grad = np.zeros(n)
    for i in range(n):
        s1 = 0.0
        s2 = 0.0
        ell_lo = max(-(n-1), -i)
        ell_hi = min(n-1, n-1-i)
        if ell_lo <= ell_hi:
            p_idx = np.arange(ell_lo + (n-1), ell_hi + (n-1) + 1)
            j_idx = i + np.arange(ell_lo, ell_hi + 1)
            s1 = np.dot(w[p_idx], 1.0 - x[j_idx])
        ell_lo2 = max(-(n-1), i - (n-1))
        ell_hi2 = min(n-1, i)
        if ell_lo2 <= ell_hi2:
            p_idx2 = np.arange(ell_lo2 + (n-1), ell_hi2 + (n-1) + 1)
            j_idx2 = i - np.arange(ell_lo2, ell_hi2 + 1)
            s2 = np.dot(w[p_idx2], x[j_idx2])
        grad[i] = dx * (s1 - s2)
    grad += 2.0 * penalty * (x.sum() - n / 2.0)
    return grad

    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    n_betas = 6
    betas = np.geomspace(beta_start, beta_end, n_betas)
    trial_x = best_h.copy()
    for bi, beta in enumerate(betas):
        if time.time() > deadline - 5:
            break
        time_per_beta = max(2, (deadline - time.time() - 5) / (n_betas - bi))
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(lse_objective, trial_x,
                           jac=lambda x: grad_lse(x, n, beta),
                           args=(n, beta),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'gtol': 1e-10,
                                    'ftol': 1e-14, 'maxfun': 20000,
                                    'iprint': -1,
                                    'maxls': 50})
            trial_x = project_feasible(res.x, n)
        except Exception:
            pass
        c_new = compute_c5(trial_x, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = trial_x.copy()
    return best_h, best_c
