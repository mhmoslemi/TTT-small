def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value_and_weights(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    a_max = shifted.max()
    if a_max > 500:
        e = np.exp(shifted - a_max)
        s = e.sum()
        w = e / s
        val = m + (a_max + np.log(s)) / alpha
    else:
        e = np.exp(shifted)
        s = e.sum()
        w = e / s
        val = m + np.log(s) / alpha
    return val, w, m

def compute_gradient(h, n, w, top_k=50):
    grad = np.zeros(n)
    k_values = np.arange(2 * n - 1) - (n - 1)
    dx = 2.0 / n
    idx = np.argsort(w)[::-1][:top_k]
    for i in idx:
        wi = w[i]
        if wi < 1e-10:
            break
        k = k_values[i]
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo > i_hi:
            continue
        j_lo1 = max(0, i_lo + k)
        j_hi1 = min(n - 1, i_hi + k)
        if j_lo1 <= j_hi1:
            grad[j_lo1:j_hi1 + 1] += wi * dx * (1.0 - h[j_lo1 - k:j_hi1 - k + 1])
        j_lo2 = max(0, i_lo, max(0, -k))
        j_hi2 = min(n - 1, i_hi, n - 1 - k)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= wi * dx * h[j_lo2 + k:j_hi2 + k + 1]
    return grad
