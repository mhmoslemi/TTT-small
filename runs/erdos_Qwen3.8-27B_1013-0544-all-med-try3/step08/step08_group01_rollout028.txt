def normalize_direction(d, h, rho):

   d = np.asarray(d,float).copy()

   # subtract mean to sum zero
   d -= d.mean()
   # clip to box
   low = -h
   high = 1.0 - h
   d = np.clip(d, low, high)
   d = np.clip(d, -rho, rho)

def project_box_sum(x, target=None):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if target is None:
        target = 0.5 * float(n)
    if n == 0:
        return x

    lo = float(np.min(x) - 1.0)
    hi = float(np.max(x))

    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    theta = 0.5 * (lo + hi)
    h = np.clip(x - theta, 0.0, 1.0)

    for _ in range(20):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:

def select_active(cvec, tol_abs=1e-5, max_count=16):
    c = np.asarray(cvec, dtype=float)
    if c.size == 0:
        return np.array([], dtype=int)
    C = float(np.max(c))
    if c.size <= max_count:
        return np.argsort(-c).astype(int)

    thr = C - max(tol_abs, 0.0)
    idx = np.where(c >= thr)[0]

    if idx.size == 0:
        return np.argsort(-c)[:max_count].astype(int)
    if idx.size <= max_count:
        return idx[np.argsort(-c[idx])].astype(int)

    part = np.argpartition(-c[idx], max_count - 1)[:max_count]
    idx = idx[part]
    return idx[np.argsort(-c[idx])].astype(int)


def add_shift_grad(grad, m, h, wm):
    n = h.size
    if wm == 0.0 or n == 0:
        return
    k = int(m) - (n - 1)
    if k >= 0:
        nv = n - k
        if nv <= 0:
            return
        grad[:nv] += wm * (1.0 - h[k:k + nv])
        grad[k:n] += -wm * h[0:nv]
    else:
        kk = -k
        nv = n - kk
        if nv <= 0:
            return
        grad[kk:n] += wm * (1.0 - h[0:nv])
        grad[0:nv] += -wm * h[kk:nv]


def shift_grad(h, m):
    n = len(h)
    grad = np.zeros(n, dtype=float)
    add_shift_grad(grad, int(m), h, 2.0 / float(n))
    return grad


def active_grad(h, active, weights=None):

    if active is None or len(active) == 0:
        return np.zeros(n, dtype=float)
    if weights is None:
        w = np.full(len(active), 1.0 / float(len(active)))
    else:

        s = float(np.sum(w))

            w = np.full(len(active), 1.0 / float(len(active)))
        else:

    dx = 2.0 / float(n)

    for idx, wt in zip(active, w):
        if wt > 1e-16:
            add_shift_grad(grad, int(idx), h, float(wt) * dx)
    return grad


def logsumexp(y):
    y = np.asarray(y, dtype=float)
    if y.size == 0:
        return 0.0
    m = float(np.max(y))
    if not np.isfinite(m):
        return 0.0
    return m + float(np.log(np.sum(np.exp(y - m))))

    if mu != 0.0:
        target = 0.5 * float(x.size)
        d = float(np.sum(x)) - target
        val += mu * d * d
    return float(val)


def smooth_grad(x, active, alpha, mu):
    if active is None or len(active) == 0:
        g = np.zeros_like(x, dtype=float)
    else:
        c = cvec_of(x)[active]
        y = alpha * c
        m = float(np.max(y))
        w = np.exp(y - m)
        s = float(np.sum(w))
        if s <= 0.0:
            w = np.full(len(c), 1.0 / float(len(c)))
        else:
            w /= s
        g = active_grad(x, active, w)

    if mu != 0.0:
        g = g + 2.0 * mu * (float(np.sum(x)) - 0.5 * float(x.size))
    return g


def sum_eq(x):
    return float(np.sum(x) - 0.5 * float(x.size))


def sum_eq_jac(x):
    return np.ones_like(x, dtype=float)


def bounds_for(n):
    return [(0.0, 1.0) for _ in range(n)]


def refine_lbfgs_multi(h, deadline, max_active=16, maxiter=800, tol_abs=1e-5):
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = cmax_of(best_h)
    n = best_h.size
    if n == 0:
        return best_h, best_c

    configs = [
        (200, 1e-6),
        (1000, 1e-5),
        (4000, 1e-4),
        (12000, 1e-3),
        (50000, 1e-2),
        (200000, 0.1),
        (800000, 1.0),
        (2500000, 10.0),
    ]
    bounds = bounds_for(n)

    for alpha, mu in configs:
        if time.time() > deadline - 2.0:
            break
        cvec = cvec_of(best_h)
        active = select_active(cvec, tol_abs=max(tol_abs, best_c * 1e-5), max_count=max_active)
        if len(active) == 0:
            break
        try:
            res = minimize(
                smooth_obj,
                best_h,
                jac=smooth_grad,
                args=(active, float(alpha), float(mu)),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxiter,
                    "maxfun": maxiter * 2,
                    "ftol": 1e-15,
                    "gtol": 1e-12,
                    "maxls": 50,
                },
            )
            cand = project_box_sum(res.x)
        except Exception:
            cand = None

        if cand is not None:
            cc = cmax_of(cand)
            if cc < best_c - 1e-14:
                best_h = cand
                best_c = cc
    return best_h, best_c


def refine_slqp_multi(h, deadline, max_active=10, alphas=(2000.0, 10000.0, 40000.0), maxiter=50):
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = cmax_of(best_h)
    n = best_h.size
    if n == 0 or n > 900:
        return best_h, best_c

    bounds = bounds_for(n)
    constraints = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac}]

    for alpha in alphas:
        if time.time() > deadline - 3.0:
            break
        cvec = cvec_of(best_h)
        active = select_active(cvec, max(1e-5, best_c * 1e-4), max_count=max_active)
        if len(active) == 0:
            break
        try:
            res = minimize(
                smooth_obj,
                best_h,
                jac=smooth_grad,
                args=(active, float(alpha), 0.0),
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-12},
            )
            cand = project_box_sum(res.x)
        except Exception:
            cand = None

        if cand is not None:
            cc = cmax_of(cand)
            if cc < best_c - 1e-14:
                best_h = cand
                best_c = cc
    return best_h, best_c


def lp_solve_direction(gstack, h, rho, use_sum=True):
    n = len(h)
    a = gstack.shape[0]
    if a == 0:
        return None, 0.0

    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    A_ub = np.empty((a, n + 1), dtype=float)
    A_ub[:, :n] = gstack
    A_ub[:, n] = -1.0
    b_ub = np.zeros(a, dtype=float)

    if use_sum:
        A_eq = np.zeros((1, n + 1), dtype=float)
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1, dtype=float)
    else:
        A_eq = None
        b_eq = None

    bounds = []
    for val in h:
        lo = -rho
        if lo < -val:
            lo = -val
        hi = rho
        if hi > (1.0 - val):
            hi = 1.0 - val
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return None, 0.0

    if not res.success or res.x is None:
        return None, 0.0

    return res.x[:-1].copy(), float(res.x[-1])


def normalize_direction(d, h, rho):
    d = np.asarray(d, dtype=float).copy()
    n = len(h)
    d = np.where(np.isfinite(d), d, 0.0)
    d -= float(np.mean(d))

    low_box = -h
    high_box = 1.0 - h
    d = np.clip(d, low_box, high_box)
    d = np.clip(d, -rho, rho)

    for _ in range(6):
        diff = -float(np.sum(d))
        if abs(diff) <= 1e-12:
            break
        lo = np.maximum(-h, -rho) - d
        hi = np.minimum(1.0 - h, rho) - d
        mask = (lo < 0.0) | (hi > 0.0)
        if not np.any(mask):
            break
        d[mask] += diff / float(np.count_nonzero(mask))
        d = np.clip(d, -h, 1.0 - h)
        d = np.clip(d, -rho, rho)

    return d


def lp_minimax_refine(h, deadline, max_active=40, max_iters=150, tol_active=1e-5):
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = cmax_of(best_h)
    n = h.size
    if n == 0:
        return best_h, best_c

    base_rho = max(0.003, min(0.06, 30.0 / float(n)))
    alphas = (1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02)
    total = 0
    fail_streak = 0

    while total < max_iters and time.time() < deadline - 2.0:
        cvec = cvec_of(h)
        C = float(np.max(cvec))
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()

        active = select_active(cvec, max(1e-7, tol_active * best_c), max_count=max_active)
        if len(active) == 0:
            break

        gstack = np.empty((len(active), n), dtype=float)
        for r, m in enumerate(active):
            gstack[r] = shift_grad(h, int(m))

        rho = base_rho * (0.72 ** total)
        if rho < 1e-5:
            break

        improved = False

        for use_sum in (True, False):
            d, tval = lp_solve_direction(gstack, h, rho, use_sum=use_sum)
            if d is None or tval > -1e-10:
                continue
            if not use_sum:
                d = normalize_direction(d, h, rho)
            if float(np.linalg.norm(d)) < 1e-13:
                continue

            for alpha in alphas:
                cand = project_box_sum(h + alpha * d)
                cc = cmax_of(cand)
                if cc < best_c - 1e-14:
                    best_c = cc
                    best_h = cand.copy()
                    h = cand.copy()
                    improved = True
                    break
            if improved:
                fail_streak = 0
                break

        if not improved:
            G = gstack.mean(axis=0)
            d = -G
            d = normalize_direction(d, h, min(rho, 0.01))
            if float(np.linalg.norm(d)) > 1e-13:
                for alpha in (0.1, 0.05, 0.02, 0.01, 0.005):
                    cand = project_box_sum(h + alpha * d)
                    cc = cmax_of(cand)
                    if cc < best_c - 1e-14:
                        best_c = cc
                        best_h = cand.copy()
                        h = cand.copy()
                        improved = True
                        break
            if not improved:
                fail_streak += 1
                if fail_streak >= 5:
                    break

        total += 1

    return best_h, best_c


def tau_ineq_fun(x, active, tau):
    return np.asarray(tau - cvec_of(x)[active], dtype=float)


def tau_ineq_jac(x, active, tau):
    n = len(x)
    active = np.asarray(active, dtype=int)
    J = np.empty((len(active), n), dtype=float)
    for r, m in enumerate(active):
        J[r] = -shift_grad(x, int(m))
    return J


def tau_one_fun(x, active, tau, idx):
    m = int(active[idx])
    return float(tau - cvec_of(x)[m])


def tau_one_jac(x, active, tau, idx):
    m = int(active[idx])
    return -shift_grad(x, m)


def prox_obj(x, anchor, mu):
    d = x - anchor
    return float(mu * np.dot(d, d))


def prox_grad(x, anchor, mu):
    return 2.0 * mu * (x - anchor)


def slqp_tau_once(h0, anchor, active, tau, mu, maxiter):
    h0 = project_box_sum(h0)
    n = len(h0)
    bounds = bounds_for(n)

    cons_group = [
        {"type": "ineq", "fun": tau_ineq_fun, "jac": tau_ineq_jac, "args": (active, float(tau))},
        {"type": "eq", "fun": sum_eq, "jac": sum_eq_jac},
    ]

    try:
        res = minimize(
            prox_obj,
            h0,
            jac=prox_grad,
            args=(anchor, float(mu)),
            method="SLSQP",
            bounds=bounds,
            constraints=cons_group,
            options={"maxiter": maxiter, "ftol": 1e-12},
        )
        if res.x is not None:
            cand = project_box_sum(res.x)
            cv = cvec_of(cand)[active]
            viol = float(np.max(cv - tau)) if len(active) > 0 else 0.0
            if viol <= max(1e-7, 1e-6 * max(1.0, abs(tau))):
                return cand
    except Exception:
        pass

    cons_ind = []
    for idx in range(len(active)):
        cons_ind.append(
            {
                "type": "ineq",
                "fun": tau_one_fun,
                "jac": tau_one_jac,
                "args": (active, float(tau), idx),
            }
        )
    cons_ind.append({"type": "eq", "fun": sum_eq, "jac": sum_eq_jac})

    try:
        res = minimize(
            prox_obj,
            h0,
            jac=prox_grad,
            args=(anchor, float(mu)),
            method="SLSQP",
            bounds=bounds,
            constraints=cons_ind,
            options={"maxiter": maxiter, "ftol": 1e-12},
        )
        if res.x is not None:
            cand = project_box_sum(res.x)
            cv = cvec_of(cand)[active]
            viol = float(np.max(cv - tau)) if len(active) > 0 else 0.0
            if viol <= max(1e-6, 2e-5 * max(1.0, abs(tau))):
                return cand
    except Exception:
        pass

    return None


def slqp_tau_refine(h, deadline, max_active=10, max_solves=24):
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = cmax_of(best_h)
    if len(best_h) == 0 or len(best_h) > 900:
        return best_h, best_c

    anchor = best_h.copy()
    drop = 5e-5
    solves = 0

    while time.time() < deadline - 5.0 and solves < max_solves:
        cvec = cvec_of(best_h)
        C = float(np.max(cvec))
        active = select_active(cvec, max(1e-5, 2e-4 * C), max_count=max_active)
        if len(active) == 0:
            break

        success = False
        for mu in (1e-4, 1e-3):
            for offset in (drop, 0.5 * drop, 0.25 * drop):
                if time.time() > deadline - 5.0 or solves >= max_solves:
                    break
                tau = C - offset
                cand = slqp_tau_once(best_h, anchor, active, tau, mu, 35)
                solves += 1
                if cand is None:
                    continue
                cc = cmax_of(cand)
                if cc < best_c - 1e-14:
                    best_h = cand.copy()
                    best_c = cc
                    anchor = best_h.copy()
                    drop = min(5e-4, max(drop * 1.7, 3e-5))
                    success = True
                    break
            if success:
                break

        if not success:
            drop *= 0.5
            if drop < 2e-7:
                break

    return best_h, best_c


def refine_subgrad(h, deadline, max_iter=400, top_count=16):
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = cmax_of(best_h)
    if len(h) == 0 or len(h) > 900:
        return best_h, best_c

    step = 10.0
    for _ in range(max_iter):
        if time.time() > deadline - 1.0:
            break
        cvec = cvec_of(h)
        C = float(np.max(cvec))
        active = select_active(cvec, max(1e-5, 1e-4 * C), max_count=top_count)
        if len(active) == 0:
            break

        G = active_grad(h, active, None)
        gn = float(np.linalg.norm(G))
        if gn < 1e-11:
            break

        improved = False
        for mult in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125):
            a = step * mult
            cand = project_box_sum(h - a * G)
            cc = cmax_of(cand)
            if cc < best_c - 1e-14:
                best_c = cc
                best_h = cand.copy()
                h = cand.copy()
                step = min(50.0, step * 1.1)
                improved = True
                break

        if not improved:
            step *= 0.5
            if step < 1e-7:
                break

    return best_h, best_c


def refine_pair_descent(h, deadline, max_high=18, max_low=18, max_sweeps=250, max_pairs=800):
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = cmax_of(best_h)
    n = len(h)
    if n == 0 or n > 900:
        return best_h, best_c

    fracs = (1.0, 0.6, 0.35, 0.2, 0.1, 0.05, 0.02)
    sweeps = 0

    while time.time() < deadline - 2.0 and sweeps < max_sweeps:
        cvec = cvec_of(best_h)
        C = float(np.max(cvec))
        active = select_active(cvec, max(1e-5, 1e-4 * C), max_count=12)
        G = active_grad(best_h, active, None)
        order = np.argsort(G)
        low = order[:max_low]
        high = order[-1:-1 - max_high:-1]
        if len(low) == 0 or len(high) == 0:
            break

        improved = False
        tested = 0

        for i in high:
            if improved:
                break
            i = int(i)
            for j in low:
                if tested >= max_pairs:
                    break
                j = int(j)
                if i == j:
                    continue

                tested += 1
                if G[i] <= G[j]:
                    ii, jj = j, i
                else:
                    ii, jj = i, j

                a = best_h[ii]
                b = best_h[jj]
                maxd = min(float(a), float(1.0 - b))
                if maxd <= 1e-12:
                    continue

                for f in fracs:
                    d = maxd * f
                    best_h[ii] += d
                    best_h[jj] -= d
                    cc = cmax_of(best_h)
                    if cc < best_c - 1e-14:
                        best_c = cc
                        improved = True
                        break
                    else:
                        best_h[ii] -= d
                        best_h[jj] += d
                if improved or tested >= max_pairs:
                    break
            if time.time() > deadline - 2.0:
                break

        h = best_h.copy()
        sweeps += 1
        if not improved:
            break

    return best_h, best_c


def resample_h(h, n_new):
    h = np.asarray(h, dtype=float)
    n_old = h.size
    if n_new == n_old:
        return project_box_sum(h)
    if n_old == 0:
        return np.empty(0, dtype=float)
    x_old = np.arange(n_old, dtype=float) * (2.0 / float(n_old))
    x_new = np.arange(int(n_new), dtype=float) * (2.0 / float(n_new))
    y = np.interp(x_new, x_old, h)
    return project_box_sum(y, 0.5 * float(n_new))


def box_smooth(h, radius=1):
    h = np.asarray(h, dtype=float)
    n = h.size
    if radius <= 0 or n <= 1:
        return project_box_sum(h)
    r = int(radius)
    kern = np.ones(2 * r + 1, dtype=float) / float(2 * r + 1)
    y = np.convolve(h, kern, mode="same")
    return project_box_sum(y, 0.5 * float(n))


def default_start(n, seed=42):
    rng = np.random.default_rng(seed)
    n = int(n)
    x = np.arange(n, dtype=float) * (2.0 / float(n))
    h = 0.5 + 0.3 * np.sin(np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
    h += 0.2 * np.sin(2.0 * np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
    return project_box_sum(h, 0.5 * float(n))


def fourier_start(n, seed, parent=None):
    rng = np.random.default_rng(seed)
    n = int(n)
    x = np.arange(n, dtype=float) * (2.0 / float(n))
    h = 0.5 * np.ones(n, dtype=float)
    for r in range(1, 5):
        a = rng.uniform(-0.35, 0.35) / float(r)
        b = rng.uniform(-0.35, 0.35) / float(r)
        h += a * np.cos(np.pi * r * x) + b * np.sin(np.pi * r * x)
    if parent is not None and len(parent) == n:
        h = 0.65 * np.asarray(parent, dtype=float) + 0.35 * h
    return project_box_sum(h, 0.5 * float(n))


def threshold_smooth_start(parent):
    p = np.asarray(parent, dtype=float)
    n = len(p)
    if n == 0:
        return p
    q = float(np.quantile(p, 0.5))
    h = (p > q).astype(float)
    h = box_smooth(h, 1)
    h = project_box_sum(h, 0.5 * float(n))
    h = box_smooth(h, 1)
    return project_box_sum(h, 0.5 * float(n))


def optimize_fixed(h, deadline, mode="full"):
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = cmax_of(best_h)
    n = len(best_h)
    now = time.time()
    if n == 0 or deadline <= now + 4.0:
        return best_h, best_c

    rem = max(4.0, deadline - now)
    max_active = 12 if mode == "quick" else 20
    lb_maxiter = 150 if rem < 120 else 500 if rem < 360 else 1000

    if time.time() < deadline - 15.0:
        d1 = min(deadline, time.time() + max(6.0, 0.35 * rem))
        best_h, best_c = refine_lbfgs_multi(
            best_h, d1, max_active=max_active, maxiter=lb_maxiter
        )

    if mode == "full" and n <= 750 and time.time() < deadline - 25.0:
        d2 = min(deadline, time.time() + max(6.0, 0.18 * rem))
        best_h, best_c = refine_slqp_multi(
            best_h, d2, max_active=min(10, max_active)
        )

    if time.time() < deadline - 8.0:
        max_lp = 64 if n <= 600 else 40 if n <= 900 else 24
        d3 = min(deadline, time.time() + max(6.0, 0.30 * rem))
        max_lp_it = 120 if rem > 220 else 50
        best_h, best_c = lp_minimax_refine(
            best_h, d3, max_active=max_lp, max_iters=max_lp_it
        )

    if mode == "full" and n <= 900 and time.time() < deadline - 10.0:
        d4 = min(deadline, time.time() + max(5.0, 0.10 * rem))
        best_h, best_c = refine_subgrad(best_h, d4, top_count=max_active)

    if mode == "full" and n <= 850 and time.time() < deadline - 8.0:
        d5 = min(deadline, time.time() + max(5.0, 0.15 * rem))
        best_h, best_c = refine_pair_descent(best_h, d5)

    if mode == "full" and n <= 800 and time.time() < deadline - 25.0:
        d6 = min(deadline, time.time() + max(8.0, 0.12 * rem))
        best_h, best_c = slqp_tau_refine(best_h, d6, max_active=8, max_solves=10)

    if time.time() < deadline - 3.0:
        best_h, best_c = lp_minimax_refine(
            best_h,
            deadline - 1.0,
            max_active=min(32, 64 if n <= 600 else 40),
            max_iters=40,
        )

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + budget_s - 6.0

    try:
        base_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        base_h = default_start(500, int(seed))

    if len(base_h) < 8:
        base_h = default_start(max(50, 2 * int(round(len(base_h))) if len(base_h) > 0 else 500), int(seed))

    base_h = project_box_sum(base_h)
    best_h = base_h.copy()
    best_c = cmax_of(best_h)
    best_n = int(best_h.size)

    if budget_s > 800.0:
        d1 = min(t0 + 470.0, deadline - 380.0)
    elif budget_s > 400.0:
        d1 = min(t0 + 220.0, deadline - 170.0)
    else:
        d1 = min(t0 + 90.0, deadline - 30.0)

    if d1 > t0 + 5.0:
        best_h, best_c = optimize_fixed(best_h, d1, mode="full")
        best_n = int(best_h.size)

    if budget_s > 700.0 and time.time() < deadline - 320.0:
        d2 = min(deadline, time.time() + max(120.0, 0.22 * (deadline - time.time())))
        best_h, best_c = optimize_fixed(best_h, d2, mode="full")
        best_n = int(best_h.size)

    if time.time() < deadline - 160.0:
        d3 = min(deadline, time.time() + max(60.0, 0.18 * (deadline - time.time())))
        max_lp = 64 if best_n <= 600 else 40
        best_h, best_c = lp_minimax_refine(best_h, d3, max_active=max_lp, max_iters=220)
        best_n = int(best_h.size)

    if time.time() < deadline - 130.0 and best_n <= 900:
        d4 = min(deadline, time.time() + max(30.0, 0.10 * (deadline - time.time())))
        best_h, best_c = refine_pair_descent(best_h, d4, max_high=20, max_low=20)
        best_n = int(best_h.size)

    resolutions = []
    for nt in (600, 800, 400, 300, 1000, best_n // 2):
        nt = int(nt)
        if nt <= 40 or nt > 1200:
            continue
        if nt % 2 != 0:
            nt += 1
            if nt > 1200:
                continue
        if nt == best_n or nt in resolutions:
            continue
        resolutions.append(nt)

    for nt in resolutions:
        if time.time() > deadline - 90.0:
            break
        h2 = resample_h(best_h, nt)
        c2 = cmax_of(h2)
        if c2 < best_c + 0.02:
            dtr = min(deadline, time.time() + min(130.0, max(30.0, 0.14 * (deadline - time.time()))))
            h2, c2 = optimize_fixed(h2, dtr, mode="quick")
            if c2 < best_c - 1e-14:
                best_h = h2.copy()
                best_c = c2
                best_n = int(h2.size)

    if time.time() < deadline - 110.0:
        try:
            alt1 = fourier_start(best_n, int(seed) + 101, parent=best_h)
            alt2 = threshold_smooth_start(best_h)
            alt3 = fourier_start(best_n, int(seed) + 211, parent=best_h)
        except Exception:
            alt1 = alt2 = alt3 = None

        for st in (alt1, alt2, alt3):
            if st is None or time.time() > deadline - 60.0:
                continue
            dalt = min(deadline, time.time() + min(70.0, max(25.0, 0.09 * (deadline - time.time()))))
            st, cst = optimize_fixed(st, dalt, mode="quick")
            if cst < best_c - 1e-14:
                best_h = st.copy()
                best_c = cst
                best_n = int(st.size)

    if time.time() < deadline - 18.0:
        best_h, best_c = lp_minimax_refine(
            best_h,
            deadline - 2.0,
            max_active=64 if best_n <= 600 else 40,
            max_iters=160,
        )

    best_h = project_box_sum(best_h)
    if len(best_h) == 0:
        best_h = default_start(500, int(seed))

    final_c = cmax_of(best_h)
    return best_h, float(final_c), int(best_h.size)
