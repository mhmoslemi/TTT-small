def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (target / tot)
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def grad_max(h, n, alpha):
    c = compute_c5_all(h, n)
    mx = c.max()
    sh = alpha * (c - mx)
    sh = np.clip(sh, -500.0, 500.0)
    mx_sh = sh.max()
    w = np.exp(sh - mx_sh)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros(len(c))
        idx = np.argmax(c)
        w[idx] = 1.0
    else:
        w /= sw
    dx = 2.0 / n
    grad = np.zeros(n)
    nfull = 2 * n - 1
    for m in range(nfull):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue

                                       [0.003, 0.001, 0.0005, 0.0002, 0.0001],
                                       radius=60, T_init=0.001, T_final=1e-6, rng=rng)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Multi-start perturbation
    if time.time() < deadline - 40:
        n_tries = 3
        for trial in range(n_tries):
            if time.time() > deadline - 30:
                break
            t_trial = min(30, deadline - time.time() - 20)
            if t_trial < 15:
                break
            h_pert = make_perturbation(best_h, best_n, rng, 0.02)
            h_pert, c_pert = lbfgs_optimize(h_pert, best_n, time.time() + min(20, t_trial * 0.6),
                                             alphas=[(1000, 10.0), (10000, 100.0), (100000, 1000.0)],
                                             max_iters=200)
            h_pert, c_pert = pair_search(h_pert, best_n, time.time() + min(10, t_trial * 0.3),
                                          [0.002, 0.001, 0.0005, 0.0002], radius=50)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert

    # Phase 10: Final polishing
    if time.time() < deadline - 10:
        h_f, c_f = slsqp_optimize(best_h, best_n, min(time.time() + 30, deadline - 10),
                                   alpha=2000000, maxiter=2000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    if time.time() < deadline - 3:
        h_fp, c_fp = pair_search(best_h, best_n, deadline - 2,
                                  [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                  radius=50)
        if c_fp < best_c:
            best_c = c_fp
            best_h = h_fp

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
