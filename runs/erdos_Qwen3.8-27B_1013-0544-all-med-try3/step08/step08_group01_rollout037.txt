        t5 = min(time.time() + 120, deadline - 100)
        h_high, c_high = run_lbfgs_phase(h_high, n_high, t5,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
            max_iter=1500, max_k_shifts=6, mu=30.0)
        
        if time.time() < deadline - 50:
            t5b = min(time.time() + 40, deadline - 50)
            h_high, c_high = grad_local_search(h_high, n_high, t5b, top_count=10)
        
        if c_high < best_c:
            best_c = c_high
            best_h = h_high.copy()
    
    # Phase 6: Final polish at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 30:
        t6 = min(time.time() + 25, deadline - 30)
        h, c6 = grad_local_search(best_h, n_best, t6, top_count=8)
        if c6 < best_c:
            best_c = c6
            best_h = h.copy()
    
    # Phase 7: Ultra-fine local search
    if time.time() < deadline - 10:
        t7 = min(time.time() + 10, deadline - 10)
        h, c7 = grad_local_search(best_h, len(best_h), t7, top_count=5)
        if c7 < best_c:
            best_c = c7
            best_h = h.copy()
    
    # Final evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
