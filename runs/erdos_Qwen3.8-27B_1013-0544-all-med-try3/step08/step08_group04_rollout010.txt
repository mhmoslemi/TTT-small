def compute_all_shifts(h, n):

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            frac = diff / tot
            if frac >= 1.0:
                h = np.ones(n)
            else:
                h += frac * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            frac = -diff / tot

    sh = alpha * (c - m)

    h_curr = h.copy()
    
    for it in range(max_iter):

        active_k = int(np.argmax(c_all))
        shift = active_k - (n - 1)

        grad = np.zeros(n)
        for i in range(n):
            h_is = 1.0 - h_curr[i - shift] if 0 <= i - shift < n else 0.0
            h_psh = h_curr[i + shift] if 0 <= i + shift < n else 0.0
            grad[i] = h_is - h_psh

        grad_mean = grad.mean()

        # Binary search for best t
        best_t = 0.0
        best_c_ls = best_c
        for t in [1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 0.1, 0.2, 0.5]:

                best_t = t
                h_curr = h_try_p

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_shifts(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            frac = diff / tot
            if frac >= 1.0:
                h[:] = 1.0
            else:
                h += frac * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            frac = -diff / tot
            if frac >= 1.0:
                h[:] = 0.0
            else:
                h -= frac * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(max(np.sum(np.exp(sh - mx)), 1e-300))) / alpha)
    val = np.sum(np.exp(sh))
    if val < 1e-300:
        return float(m)
    return float(m + np.log(val) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        c_new = compute_c5(res.x, n)
        c_old = compute_c5(h, n)
        if c_new < c_old:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def smooth_gradient(h, n, alpha):
    """Compute gradient of log-sum-exp smooth max using vectorized ops."""
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    ac = alpha * (c - c.max())
    ac = ac - ac.max()
    w = np.exp(ac)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c)); w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        shift = k_idx - (n - 1)
        if shift == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif shift > 0:
            grad[shift:] += wk * (1.0 - h[:-shift])
            grad[:n - shift] -= wk * h[shift:]
        else:
            s = -shift
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]
    
    grad *= dx
    return grad

def subgradient_line_search(h, n, deadline, alpha=10000, max_iter=200):
    """Line search along projected smooth subgradient direction."""
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_gradient(h_curr, n, alpha)
        
        # Project to sum-zero
        grad_proj = grad - grad.mean()
        
        # Line search along -grad_proj
        found = False
        for t in [0.1, 0.05, 0.01, 0.005, 0.001, 0.0005, 0.0001]:
            h_try = h_curr - t * grad_proj
            if h_try.min() < -0.01 or h_try.max() > 1.01:
                continue
            h_try_p = project_feasible(h_try, n)
            c_try = compute_c5(h_try_p, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                h_curr = h_try_p
                found = True
                break
        
        if not found:
            break
    
    return h_curr, best_c

def guided_pair_search(h, n, deadline, deltas, max_passes_per_delta=50):
    """Pair search guided by subgradient of active shifts."""
    best_c = compute_c5(h, n)
    
    for d in deltas:
        passes = 0
        while time.time() < deadline and passes < max_passes_per_delta:
            improved = False
            c_all = compute_all_shifts(h, n)
            # Get top active shifts
            top_k = np.argsort(c_all)[::-1][:5]
            
            # Compute combined subgradient
            combined_grad = np.zeros(n)
            for k_idx in top_k:
                shift = k_idx - (n - 1)
                if shift == 0:
                    combined_grad += (1.0 - 2.0 * h) * c_all[k_idx]
                elif shift > 0:
                    combined_grad[shift:] += c_all[k_idx] * (1.0 - h[:-shift])
                    combined_grad[:n - shift] -= c_all[k_idx] * h[shift:]
                else:
                    s = -shift
                    combined_grad[:n - s] += c_all[k_idx] * (1.0 - h[s:])
                    combined_grad[s:] -= c_all[k_idx] * h[:n - s]
            
            # Determine direction for each index
            inc_set = set()
            dec_set = set()
            for i in range(n):
                if combined_grad[i] < -1e-10:
                    inc_set.add(i)
                elif combined_grad[i] > 1e-10:
                    dec_set.add(i)
            
            if not inc_set or not dec_set:
                # Fall back to all pairs
                inc_list = list(range(n))
                dec_list = list(range(n))
            else:
                inc_list = list(inc_set)
                dec_list = list(dec_set)
            
            for i in inc_list:
                if time.time() > deadline:
                    return h, best_c
                for j in dec_list:
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            
            passes += 1
            if not improved:
                break
    
    return h, best_c

def brute_pair_search(h, n, deadline, deltas, max_passes=30):
    """Unrestricted pair search."""
    best_c = compute_c5(h, n)
    for d in deltas:
        passes = 0
        while time.time() < deadline and passes < max_passes:
            improved = False
            for i in range(n):

                    return h, best_c

                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] -
