    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, shift_s):
    """Gradient of c[shift_s] w.r.t. h, where c = correlate(h, 1-h, 'full')*dx"""
    dx = 2.0 / n
    grad = np.zeros(n)

    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def lse_gradient(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, s in zip(w, active):
        if wi > 1e-12:
            grad += wi * shift_gradient(h, n, s)
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def optimize_lse(h, n, deadline, alpha_schedule, top_frac, mu):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alpha_schedule:
        if time.time() > deadline - 3:
            break
        for _ in range(3):
            if time.time() > deadline - 2:
                break
            c_vec = compute_c5_vec(best_h, n)
            mx = c_vec.max()
            thresh = mx * (1.0 - top_frac)
            active = np.where(c_vec >= thresh)[0]
            if len(active) < 3:
                active = np.argsort(-c_vec)[:5]
            try:
                res = minimize(lse_objective, best_h, jac=lse_gradient,
                             args=(n, active, alpha, mu), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 5000, 'ftol': 1e-20,
                                      'gtol': 1e-14, 'maxfun': 30000, 'maxls': 100})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c
