    """One pass of pair search with step size d. Returns (h, best_c, improved)."""

    c_max = c.max()

    # S[i][k] = dc[k]/dh[i] = (1-h[i+k-n+1]) - h[i-k+n-1] (out of range = 0)

    S = np.zeros((n, 2*n-1))
    for i in range(n):

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def _slsqp_obj(x):
    return x[-1]

def _slsqp_con_k(x, k, n):
    dx = 2.0 / n
    hv = x[:-1]
    return x[-1] - float(np.correlate(hv, 1-hv, mode='full')[k]) * dx

def _slsqp_con_sum(x, n):
    return x[:-1].sum() - n / 2.0

def _slsqp_con_sum_jac(x, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def slsqp_optimize_t(h, n, max_active=40, maxiter=3000, ftol=1e-15):
    dx = 2.0 / n
    c_full = np.correlate(h, 1-h, mode='full') * dx
    c_max = float(c_full.max())
    sorted_idx = np.argsort(c_full)[::-1]
    active_k = sorted_idx[:max_active].tolist()
    
    x0 = np.concatenate([h, [c_max]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    cons = []
    for k in active_k:
        cons.append({'type': 'ineq', 'fun': _slsqp_con_k, 'args': (k, n)})
    cons.append({'type': 'eq', 'fun': _slsqp_con_sum, 'jac': _slsqp_con_sum_jac, 'args': (n,)})
    
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = np.clip(res.x[:-1], 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_max:
            return h_new, c_new
    except Exception:
        pass
    return h, c_max

def compute_sensitivity(h, n):
    one_mh = 1.0 - h
    S = np.zeros((n, 2*n-1))
    for i in range(n):
        k_lo = max(0, n-1-i)
        k_hi = min(2*n-2, 2*n-1-i)
        if k_lo <= k_hi:
            idx = np.arange(i+k_lo-n+1, i+k_hi-n+2)
            S[i, k_lo:k_hi+1] = one_mh[idx]
        k_lo2 = max(0, i)
        k_hi2 = min(2*n-2, i+n-1)
        if k_lo2 <= k_hi2:
            idx2 = np.arange(i-k_lo2+n-1, i-k_hi2+n-1, -1)  # decreasing
            S[i, k_lo2:k_hi2+1] -= h[idx2]
    return S

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        for _pass in range(500):
            if time.time() > deadline:
                return h, best_c
            S = compute_sensitivity(h, n)
            dx = 2.0 / n
            c = np.correlate(h, 1-h, mode='full') * dx
            c_max = float(c.max())
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        delta = d * (S[i] - S[j]) * dx
                        nm = (c + delta).max()
                        if nm < c_max - 1e-14:
                            h[i] += d
                            h[j] -= d
                            c += delta
                            c_max = float(nm)
                            improved = True
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        delta = -d * (S[i] - S[j]) * dx
                        nm = (c + delta).max()
                        if nm < c_max - 1e-14:
                            h[i] -= d
                            h[j] += d
                            c += delta
                            c_max = float(nm)
                            improved = True
            if not improved:
                break
            if c_max < best_c:
                best_c = c_max
    return h, best_c
