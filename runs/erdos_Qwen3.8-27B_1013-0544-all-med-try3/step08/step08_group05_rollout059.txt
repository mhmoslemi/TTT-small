def project_sum(h, target):
    """Project h onto {h: sum(h) = target} ∩ [0,1]^n."""
    h = h.copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # Add/subtract uniformly where possible
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h += diff / s * h
    return np.clip(h, 0, 1)

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-13:
            break

            if s < 1e-15:

            h += diff / s * h
