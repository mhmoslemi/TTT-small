    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    obj = m + (z_max + np.log(s)) / alpha
    weights = exp_z / s
    
    # Gradient via correlate
    t1 = np.correlate(weights, 1.0 - h, mode='full')[:n]
    w_flipped = weights[::-1]
    t2 = np.correlate(w_flipped, h, mode='full')[:n]
    
    grad = dx * (t1 - t2)
    grad += 2.0 * penalty * (h.sum() - n / 2.0)
    
    return obj, grad

def coordinate_descent(h, n, deadline, step_sizes):
    best_c = compute_c5(h, n)
    for d in step_sizes:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    # Try moving d from j to i
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try moving d from i to j
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_coordinate_descent(h, n, deadline, step_sizes, radius=50):
    best_c = compute_c5(h, n)
    for d in step_sizes:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def make_structured_init(n, rng, variant):

        # Alternating blocks
        block_size = n // 10
        for i in range(10):

            end = min((i+1)*block_size, n)
            if i % 2 == 0:
                h[start:end] = 1.0
            else:
                h[start:end] = 0.0
    elif variant == 1:
        # Three blocks: 1, 0, 1
        third = n // 3
        h[:third] = 1.0
        h[third:2*third] = 0.0
        h[2*third:] = 1.0
    elif variant == 2:
        # Sine wave
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.cos(2 * np.pi * x)
    elif variant == 3:

    elif variant == 4:
        # Concentrated in middle
        center = n // 2
        width = n // 4
        h[center-width:center+width] = 1.0
    elif variant == 5:
        # Two pairs of blocks
        quarter = n // 4
        h[:quarter] = 1.0
        h[2*quarter:3*quarter] = 1.0
    elif variant == 6:
        # Cosine with different frequency
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.45 * np.cos(3 * np.pi * x)
    elif variant == 7:
        # Ramp up and down
        h = np.abs(np.linspace(-1, 1, n))
    elif variant == 8:
        # Four alternating blocks
        quarter = n // 4
        h[:quarter] = 1.0
        h[quarter:2*quarter] = 0.0
        h[2*quarter:3*quarter] = 1.0
        h[3*quarter:] = 0.0

    return h

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

                best_c = c_new
                best_h = h_new.copy()
    
    # Phase 4: Perturbation restarts on best
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        for r in range(6):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.06 * r / 5.0
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 20) * 0.4)
            if alloc < 15:
                break
            h_new, c_new = optimize_lbfgsb(hp, n, time.time() + alloc)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    
    # Phase 5: Final intensive polish
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        # High-alpha L-BFGS-B
        alloc_final = min(80, (deadline - time.time()) * 0.3)
        if alloc_final > 10:
            h_new, c_new = optimize_lbfgsb(best_h, n, time.time() + alloc_final, max_alphas=4)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Fine pair search
        if time.time() < deadline - 5:
            steps_fine = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
            rad = min(n, 100) if n > 200 else n
            best_h, best_c = pair_search(best_h, n, deadline - 2, steps_fine, radius=rad)
    
    if best_h is None:
        n = 250
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
