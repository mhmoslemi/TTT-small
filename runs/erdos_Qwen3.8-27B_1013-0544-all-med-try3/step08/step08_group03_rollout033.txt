from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_smooth_max(h, n, k=10, alpha=100.0):
    """Compute top-k smooth max of the correlation values."""
    c = compute_c5_vec(h, n)
    # Get top-k values

    c_top = c[idx]
    m = c_top.max()
    sh = alpha * (c_top - m)
    res = m + np.log(np.sum(np.exp(sh))) / alpha
    return res, idx

            grad[kk2:n] += wm * (1.0 - h[0:nv])

def pair_polish(h, n, deadline, deltas=None, radius=None):
    """Polish by trying small moves on pairs of points."""
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return project_feasible(h, n), compute_c5(h, n)
                lo = 0 if radius is None else max(0, i - radius)
                hi = n if radius is None else min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try moving mass from i to j
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d; h[j] -= d
    return project_feasible(h, n), compute_c5(h, n)

        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def cyclic_coord_descent(h, n, deadline):
    """Cyclic coordinate descent: optimize one point at a time."""
    best_c = compute_c5(h, n)
    n_iter = 0
    while time.time() < deadline:
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return project_feasible(h, n), compute_c5(h, n)
            c_vec = compute_c5_vec(h, n)
            max_k = int(np.argmax(c_vec))
            kk = max_k - (n - 1)
            # Compute subgradient of c_{max_k} w.r.t. h[j]
            sg = 0.0
            if 0 <= j <= n - 1:
                if j + kk >= 0 and j + kk < n:
                    sg += (1.0 - h[j + kk])
                if j - kk >= 0 and j - kk < n:
                    sg += (-h[j - kk])
            if sg > 1e-10:
                # Decrease h[j]
                step = min(0.01, h[j] * 0.1)
                h_new = h.copy()
                h_new[j] -= step
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h = h_new
                    improved = True
            elif sg < -1e-10:
                # Increase h[j]
                step = min(0.01, (1.0 - h[j]) * 0.1)
                h_new = h.copy()
                h_new[j] += step
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h = h_new
                    improved = True
        n_iter += 1
        if not improved and n_iter > 3:
            break
    return project_feasible(h, n), compute_c5(h, n)

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return project_feasible(h, n), compute_c5(h, n)
                lo = 0 if radius is None else max(0, i - radius)
                hi = n if radius is None else min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return project_feasible(h, n), compute_c5(h, n)

def gradient_descent(h, n, deadline, lr_init=0.005, alpha_init=500.0, topk=5):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    for _ in range(20000):
        if time.time() > deadline:
            break
        g = topk_smooth_grad(h, n, topk, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        step = lr
        improved = False
        for _ in range(40):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if improved:
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.5
            alpha *= 1.4
        if lr < 1e-13:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    
    # Phase 1: L-BFGS with top-k smooth max, increasing alpha
    t1 = min(t0 + 280, deadline - 400)
    best_h, best_c = lbfgs_phase(best_h, best_n, t1, k=8,
                                  alpha_seq=[30, 100, 500, 2000, 10000, 50000, 200000])
    best_n = len(best_h)
    
    # Phase 2: Full smooth max L-BFGS (all correlations weighted)
    if time.time() < deadline - 300:
        t2 = min(time.time() + 80, deadline - 300)
        h2, c2 = full_smooth_lbfgs(best_h, best_n, t2,
                                    alpha_seq=[200, 1000, 5000, 20000, 100000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: SLSQP with high-alpha top-k
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 200)
        for a, kk in [(200000, 5), (1000000, 4), (5000000, 3)]:
            if time.time() > t3:
                break
            h3, c3 = slsqp_phase(best_h, best_n, min(time.time() + 35, t3), alpha=a, k=kk)
            if c3 < best_c:
                best_c = c3
                best_h = project_feasible(h3, best_n)
    
    # Phase 4: Try other resolutions
    for n_try in [250, 350, 400, 450, 500]:
        if time.time() > deadline - 180:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc = min(70, deadline - time.time() - 100)
        if alloc < 25:
            break
        
        h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.6,
                                k=6, alpha_seq=[50, 500, 5000, 50000, 200000])
        if time.time() < deadline - 60:
            h_t, c_t = slsqp_phase(h_t, n_try, min(time.time() + 25, deadline - 55),
                                    alpha=500000, k=4)
        if c_t < best_c:
            best_c = c_t
            best_h = project_feasible(h_t, n_try)
            best_n = n_try
    
    # Phase 5: Final L-BFGS on best
    if time.time() < deadline - 100:
        t5 = min(time.time() + 50, deadline - 80)
        best_h, best_c = lbfgs_phase(best_h, best_n, t5, k=4,
                                      alpha_seq=[10000, 100000, 1000000])
    
    # Phase 6: Gradient descent with backtracking
    if time.time() < deadline - 60:
        t6 = min(time.time() + 40, deadline - 45)
        h6, c6 = gradient_descent(best_h, best_n, t6, lr_init=0.003,
                                   alpha_init=1000.0, topk=3)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Cyclic coordinate descent
    if time.time() < deadline - 40:
        t7 = min(time.time() + 30, deadline - 25)
        h7, c7 = cyclic_coord_descent(best_h, best_n, t7)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Pair polishing (coarse to fine)
    if time.time() < deadline - 15:
        t8 = min(time.time() + 50, deadline - 10)
        best_h, best_c = pair_search(best_h, best_n, t8,
                                      deltas=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                      radius=60)
    
    # Phase 9: Very fine pair polishing
    if time.time() < deadline - 3:
        t9 = deadline - 2
        best_h, best_c = pair_search(best_h, best_n, t9,
                                      deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
                                      radius=40)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
