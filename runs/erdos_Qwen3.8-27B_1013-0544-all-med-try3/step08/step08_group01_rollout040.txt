def coordinate_descent(h, best_c, n, deadline):
    dx = 2.0 / n
    h = h.copy()
    current_c = best_c
    
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_c = c_vec.max()
            if max_c < current_c:
                current_c = max_c
                best_c = max_c
            
            # Find active shifts (within 1% of max)
            thresh = max_c * 0.999
            active = np.where(c_vec >= thresh)[0]
            
            # Compute combined gradient for active shifts
            n_act = len(active)
            if n_act == 0:
                break
            
            # For efficiency, compute gradient only for active shifts
            grad = np.zeros(n)
            for mm in active:
                k = mm - (n-1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - h[k:n])
                    grad[k:n] += (-h[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - h[0:nv])
                    grad[0:nv] += (-h[kk:n])
            grad /= n_act * n  # normalize
            
            # Try moving each coordinate in its best direction
            order = np.argsort(-np.abs(grad))
            for idx in order[:200]:  # top 200 coordinates
                if time.time() > deadline:
                    return h, best_c
                i = idx
                # Try +step
                if h[i] + step <= 1.0:
                    h[i] += step
                    c = compute_c5(h, n)
                    if c < current_c - 1e-15:
                        current_c = c
                        improved = True

                        h[i] -= step
                # Try -step
                if h[i] - step >= 0.0:
                    h[i] -= step
                    c = compute_c5(h, n)
                    if c < current_c - 1e-15:
                        current_c = c
                        improved = True
                    else:
                        h[i] += step
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

    dx = 2.0 / n

        mm = k + n - 1
        wk = w[mm] * dx
        if wk < 1e-16:

        nv = n - k
        grad[:nv] += wk * (1.0 - x[k:])
        grad[k:] += wk * (-x[:nv])

        mm = n - 1 - k
        wk = w[mm] * dx

        nv = n - k
        grad[k:] += wk * (1.0 - x[:nv])

    return grad

def lbfgs_phase(h, n, deadline, alphas, mu_scale=0.1, max_iter=2000):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        mu = max(0.1, alpha * mu_scale)
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad_fast,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iter, 'ftol': 1e-16,
                                  'gtol': 1e-12, 'maxfun': max_iter * 5})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fast_coord_descent(h, n, deadline, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    h = h.copy()
    c_vec = compute_c5_vec(h, n)
    current_c = c_vec.max()
    best_c = current_c
    dx = 2.0 / n
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        sweeps = 0
        while improved and time.time() < deadline:
            improved = False
            sweeps += 1
            c_vec = compute_c5_vec(h, n)
            current_c = c_vec.max()
            if current_c < best_c:
                best_c = current_c
            
            # Compute gradient for active (top) shifts
            max_c = current_c
            thresh = max_c - 0.001 * max_c
            active = np.where(c_vec >= thresh)[0]
            n_act = len(active)
            if n_act == 0:
                break
            
            # Compute per-coordinate gradient contribution
            # For coordinate i, the change in C(k) is:
            # delta * dx * [(1-h[i+k]) if 0<=i+k<n - h[i-k] if 0<=i-k<n]
            # averaged over active shifts
            
            # For efficiency, precompute for each active shift the gradient at each i
            # But that's O(n * n_act). Instead, compute on the fly.
            
            order = np.random.permutation(n)  # Random order to avoid bias
            
            for i in order:
                if time.time() > deadline:
                    return best_h if 'best_h' in dir() else h, best_c
                
                # Compute sum of (1-h[i+k]) over active shifts where i+k valid
                # and sum of h[i-k] over active shifts where i-k valid
                g = 0.0
                for mm in active:
                    k = mm - (n - 1)
                    if 0 <= i + k < n:
                        g += (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        g += (-h[i - k])
                g *= dx / n_act
                
                # Try +step
                if h[i] + step <= 1.0 and g < 0:  # moving up decreases C
                    h[i] += step
                    # Update c_vec efficiently
                    # For each active shift mm, update c_vec[mm]
                    for mm in active:
                        k = mm - (n - 1)
                        delta = 0.0
                        if 0 <= i + k < n:
                            delta += step * (1.0 - h[i + k] - step)  # h changed, so 1-h[i+k] changed too if i+k != i
                            # Actually this is getting complicated. Let me just recompute.
                    # Just recompute c_vec
                    c_new = compute_c5(h, n)
                    if c_new < current_c - 1e-15:
                        current_c = c_new
                        if c_new < best_c:
                            best_c = c_new
                        c_vec = compute_c5_vec(h, n)
                        improved = True
                    else:
                        h[i] -= step
                elif h[i] - step >= 0.0 and g > 0:  # moving down decreases C
                    h[i] -= step
                    c_new = compute_c5(h, n)
                    if c_new < current_c - 1e-15:
                        current_c = c_new
                        if c_new < best_c:
                            best_c = c_new
                        c_vec = compute_c5_vec(h, n)
                        improved = True
                    else:
                        h[i] += step
    return h, best_c

Hmm, this coordinate descent is still O(n × n_act) per coordinate due to the inner loop over active shifts. If n_act is large (say 100), that's 100 × 1000 = 10⁵ per coordinate, and 10⁸ per sweep. Too slow.

Let me optimize: instead of looping over active shifts for each coordinate, I can precompute the gradient vector.

For all active shifts, the combined gradient at coordinate i is:
$g[i] = \frac{dx}{n\_act} \sum_{mm \in active} [(1-h[i+k_{mm}]) \cdot \mathbf{1}_{valid} - h[i-k_{mm}] \cdot \mathbf{1}_{valid}]$

This can be computed as:
$g[i] = \frac{dx}{n\_act} [\sum_{mm} (1-h[i+k_{mm}]) \cdot \mathbf{1}_{0 \leq i+k_{mm} < n} - \sum_{mm} h[i-k_{mm}] \cdot \mathbf{1}_{0 \leq i-k_{mm} < n}]$

The first sum is a "masked convolution" of $(1-h)$ with an indicator of active shifts.
The second sum is a "masked convolution" of $h$ with an indicator of active shifts.

If I create a vector $a$ of length $2n-1$ where $a[mm] = 1$ for $mm \in active$ and 0 otherwise (then normalize by $n\_act$), then:

First sum at position i: $\sum_{mm} a[mm] \cdot (1-h[i + (mm - (n-1))])$ where $0 \leq i + mm - (n-1) < n$.

Let $k = mm - (n-1)$, so $mm = k + (n-1)$:
$= \sum_k a[k+(n-1)] \cdot (1-h[i+k]) \cdot \mathbf{1}_{0 \leq i+k < n}$

This is the cross-correlation of $(1-h)$ with $a$ (shifted appropriately).

I can compute this using `np.correlate(1-h, a_shifted, mode='full')` and extract the right part!

Let me define $a$ as a vector of length $n$ (for positive shifts only) and handle negative shifts separately. Actually, it's easier to just use the full vector.

Let $a$ be a vector of length $n$ where $a[k] = 1$ if shift $+k$ is active, else 0.
Let $b$ be a vector of length $n$ where $b[k] = 1$ if shift $-k$ is active, else 0.

Then:
First sum (positive shifts): $\sum_{k=0}^{n-1} a[k] \cdot (1-h[i+k])$ for $i+k < n$.
This is `np.correlate(1-h, a, mode='full')` appropriately indexed.

Actually, let me just use a simpler approach. Since the number of active shifts is usually small (maybe 5-50), I can compute the gradient for each coordinate by summing over active shifts. For n=1000 and 50 active shifts, that's 50 × 1000 = 50,000 operations per gradient vector computation. That's fine!

```python
def compute_active_grad(h, n, active_idx):
    """Compute the combined gradient at all coordinates for the given active shifts."""
    dx = 2.0 / n
    grad = np.zeros(n)
    n_act = len(active_idx)
    for mm in active_idx:
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += (1.0 - h[k:])
            grad[k:] += (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += (1.0 - h[:nv])
            grad[:nv] += (-h[kk:])
    grad *= dx / n_act
    return grad
```

This is O(n × n_act) which for n=1000, n_act=50 is 50,000 operations. Fast!

Then for each coordinate i, I know $g[i]$ (the directional derivative). If $g[i] < 0$, increasing $h[i]$ decreases the (average) active correlation. If $g[i] > 0$, decreasing $h[i]$ helps.

After a move, I need to recompute $C(k)$ for all shifts to check if the max decreased. This is O(n²) per move. If I make n moves per sweep, that's O(n³) per sweep. For n=1000, that's 10⁹ per sweep. Too slow!

Alternative: after a move at coordinate i, I can update $C(k)$ for all k in O(n) time:
$\Delta C(k) = \delta \cdot dx \cdot [(1-h[i+k]) \cdot \mathbf{1}_{valid} - h[i-k] \cdot \mathbf{1}_{valid}]$ (before the move)

Wait, but $h[i+k]$ might change if $i+k = i$ (i.e., $k=0$). And for other shifts, only the term involving $h[i]$ changes.

Actually, $C(k) = \sum_j h[j](1-h[j+k]) \cdot dx$. If I change $h[i]$ to $h[i] + \delta$:
- The term $j = i$ changes: $\delta(1-h[i+k]) \cdot dx$ (if $i+k$ valid and $i+k \neq i$)
- The term $j = i-k$ changes: $-$ (if $i-k$ valid and $i-k \neq i$): $h[i-k] \cdot \delta \cdot dx$

Wait, I need to be more careful. $C(k) = \sum_j h[j](1-h[j+k]) \cdot dx$.

If $h[i] \to h[i] + \delta$:
- Term where $j = i$: $(h[i]+\delta)(1-h[i+k]) - h[i](1-h[i+k]) = \delta(1-h[i+k])$
- Term where $j+k = i$, i.e., $j = i-k$: $h[i-k]((1-h[i]-\delta) - (1-h[i])) = -h[i-k] \cdot \delta$

So: $\Delta C(k) = dx \cdot [\delta(1-h[i+k]) \cdot \mathbf{1}_{0 \leq i+k < n, k \neq 0} - h[i-k] \cdot \delta \cdot \mathbf{1}_{0 \leq i-k < n, k \neq 0}]$

Wait, when $k = 0$: $C(0) = \sum_j h[j](1-h[j]) \cdot dx$. If $h[i] \to h[i]+\delta$:
$\Delta C(0) = (\delta(1-h[i]) - h[i]\delta) \cdot dx = \delta(1-2h[i]) \cdot dx$

For $k \neq 0$:
$\Delta C(k) = dx \cdot \delta \cdot [(1-h[i+k]) \cdot \mathbf{1}_{0 \leq i+k < n} - h[i-k] \cdot \mathbf{1}_{0 \leq i-k < n}]$

Note: when $k \neq 0$, $i+k \neq i$ and $i-k \neq i$, so we don't have the special case.

So I can update all $C(k)$ in O(n) time after a single coordinate change!

This means a full coordinate descent sweep is O(n²) total (n coordinates × O(n) update each). For n=1000, that's 10⁶ per sweep. I can do many sweeps!
