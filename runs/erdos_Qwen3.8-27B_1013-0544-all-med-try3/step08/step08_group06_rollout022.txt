def compute_grad_logsumexp(h, n, alpha):
    """Compute gradient of (1/alpha)*log(sum_k exp(alpha * c_k)) where c_k = correlate(h,1-h,'full')[k]*dx"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    max_c = c.max()
    shifted = alpha * (c - max_c)
    p = np.exp(shifted)
    p /= p.sum()
    
    grad = np.zeros(n)
    # For each lag s = k - (n-1), s ranges from -(n-1) to n-1
    for s in range(-(n-1), n):
        w = p[s + n - 1]
        if w < 1e-15:
            continue
        if s > 0:
            # ∂c_s/∂h[i] = (1-h[i+s]) for i=0..n-1-s, minus h[i-s] for i=s..n-1
            grad[:n-s] += w * (1 - h[s:])  # first part: (1-h[i+s]) for i=0..n-1-s
            grad[s:] -= w * h[:n-s]        # second part: -h[i-s] for i=s..n-1
        elif s < 0:
            s_abs = -s
            # ∂c_{-s_abs}/∂h[i] = (1-h[i-s_abs]) for i=s_abs..n-1, minus h[i+s_abs] for i=0..n-1-s_abs
            grad[s_abs:] += w * (1 - h[:n-s_abs])  # (1-h[i-s_abs]) for i=s_abs..n-1
            grad[:n-s_abs] -= w * h[s_abs:]        # -h[i+s_abs] for i=0..n-1-s_abs
        else:
            # s=0: ∂c_0/∂h[i] = (1-h[i]) - h[i] = 1-2h[i]
            grad += w * (1 - 2 * h)
    
    return grad

def obj_grad(x, n, alpha):
    """Objective and gradient for SLSQP."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    e = np.exp(shifted)
    total = e.sum()
    val = max_c + np.log(total) / alpha
    grad = compute_grad_logsumexp(x, n, alpha)

    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            if tot < 1e-14:

                i = order[ii]
                j = order[-1 - ii]
                if i == j:
                    continue
                d_max = min(1.0 - h[i], h[j])
                if d_max < 1e-12:
                    continue
                for d_frac in [0.1, 0.2, 0.3]:
                    d = d_frac * d_max
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
            if not improved:
                break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def sum_eq_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_eq_jac(x, n):
    return np.ones(n)

def lse_objective(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def lse_gradient(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    e = np.exp(sh)
    p = e / e.sum()
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = p[s + n - 1]
        if w < 1e-17:
            continue
        if s > 0:
            grad[:n - s] += w * (1.0 - x[s:])
            grad[s:] -= w * x[:n - s]
        elif s < 0:
            a = -s
            grad[a:] += w * (1.0 - x[:n - a])
            grad[:n - a] -= w * x[a:]
        else:
            grad += w * (1.0 - 2.0 * x)
    return grad

def slsqp_step(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(lse_objective, h, jac=lse_gradient, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_descent(h, n, deadline, max_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    step = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_c_array(h, n)
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        if s > 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
        elif s < 0:
            a = -s
            grad[a:] += (1.0 - h[:n - a])
            grad[:n - a] -= h[a:]
        else:
            grad = 1.0 - 2.0 * h
        grad -= grad.mean()
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.95
        if step < 1e-8:
            break
    return best_h, best_c

def gradient_pair_search(h, n, deadline, max_rounds=300):
    best_c = compute_c5(h, n)
    h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        g = lse_gradient(h, n, 5000)
        gc = g - g.mean()
        order_neg = np.argsort(gc)[:min(8, n)]
        order_pos = np.argsort(gc)[-min(8, n):][::-1]
        improved = False
        for i in order_neg:
            if gc[i] > -1e-8:
                break
            for j in order_pos:
                if gc[j] < 1e-8:
                    break
                if i == j:
                    continue
                d_max_up = min(1.0 - h[i], h[j])
                d_max_dn = min(h[i], 1.0 - h[j])
                for (a, b, d_max) in [(i, j, d_max_up), (j, i, d_max_dn)]:
                    if d_max < 1e-12:
                        continue
                    for frac in [0.5, 0.25, 0.75, 0.1, 0.9]:
                        d = frac * d_max
                        h[a] += d
                        h[b] -= d
                        c_val = compute_c5(h, n)
                        if c_val < best_c - 1e-14:
                            best_c = c_val
                            improved = True
                        else:
                            h[a] -= d
                            h[b] += d
        for i in order_neg[:3]:
            if gc[i] > -1e-8:
                break
            for j in order_pos[:3]:
                if gc[j] < 1e-8:
                    break
                if i == j:
                    continue
                d_max = min(1.0 - h[i], h[j])
                if d_max < 1e-12:
                    continue
                for frac in [0.05, 0.1]:
                    d = frac * d_max
                    h[i] += d
                    h[j] -= d
                    c_val = compute_c5(h, n)
                    if c_val < best_c - 1e-14:
                        best_c = c_val
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
        if not improved:
            break
    return h, best_c

def full_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(60):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                hi = h[i] + d
                lo = h[i] - d
                for j in range(i + 1, n):
                    # Try h[i]+=d, h[j]-=d
                    if hi <= 1.0 and h[j] - d >= 0.0:
                        h[i], h[j] = hi, h[j] - d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, h[j] + d
                    # Try h[i]-=d, h[j]+=d
                    if lo >= 0.0 and h[j] + d <= 1.0:
                        h[i], h[j] = lo, h[j] + d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = lo + d, h[j] - d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(40):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                hi = h[i] + d
                lo = h[i] - d
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if hi <= 1.0 and h[j] - d >= 0.0:
                        h[i], h[j] = hi, h[j] - d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, h[j] + d
                    if lo >= 0.0 and h[j] + d <= 1.0:
                        h[i], h[j] = lo, h[j] + d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = lo + d, h[j] - d
            if not improved:
                break
    return h, best_c

def optimize_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 6:
            break
        rem = deadline - time.time() - 6
        mi = 5000 if rem > 60 else 2000
        ht = slsqp_step(best_h, n, a, maxiter=mi, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    # Subgradient descent
    if time.time() < deadline - 10:
        st = deadline - 10
        best_h, c = subgradient_descent(best_h, n, st, max_iter=2000)
        if c < best_c:
            best_c = c
    # Gradient-guided pair search
    if time.time() < deadline - 5:
        best_h, c = gradient_pair_search(best_h, n, deadline - 5, max_rounds=300)
        if c < best_c:
            best_c = c
    # Final SLSQP
    if time.time() < deadline - 4:
        ht = slsqp_step(best_h, n, 2000000, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    # Very fine pair search
    if n <= 120 and time.time() < deadline - 2:
        best_h, c = full_pair_search(best_h, n, deadline - 1, [0.002, 0.001, 0.0005])
        if c < best_c:
            best_c = c
    elif time.time() < deadline - 2:
        best_h, c = near_pair_search(best_h, n, deadline - 1, [0.002, 0.001, 0.0005], radius=30)
        if c < best_c:
            best_c = c
    # Final gradient search
    if time.time() < deadline - 1:
        best_h, c = gradient_pair_search(best_h, n, deadline - 0.5, max_rounds=200)
        if c < best_c:
            best_c = c
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
