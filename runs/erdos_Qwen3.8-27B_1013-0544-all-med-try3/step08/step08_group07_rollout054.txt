```python
def simulated_annealing_binary(h, n, deadline, T0=1.0, T1=0.001, steps_per_temp=100):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    g = compute_all_g(h, n)  # all shift values
    max_g = g.max()
    
    best_h = h.copy()
    best_max_g = max_g
    
    T = T0
    n_temps = 100
    for temp_step in range(n_temps):
        for s in range(steps_per_temp):
            if time.time() > deadline:
                return best_h, best_max_g * (2.0/n)
            # Random swap
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            if a == b:
                continue
            # Compute Δg(k) for all k
            # ... O(n) computation
            new_max = max(g + delta_g)
            if new_max < max_g or rng.random() < exp((max_g - new_max) / T):
                # Accept
                h[a], h[b] = 0, 1
                g += delta_g
                max_g = g.max()
                # Update ones and zeros
                ones = ones[ones != a]
                ones = np.append(ones, b)
                zeros = zeros[zeros != b]
                zeros = np.append(zeros, a)
                if max_g < best_max_g:
                    best_max_g = max_g
                    best_h = h.copy()
        T *= (T1/T0)**(1.0/n_temps)
    return best_h, best_max_g * (2.0/n)
```

```python
            s = room.sum()

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)
```

```python
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + np.log(np.exp(sh - sm).sum()) / alpha

def smooth_max_grad(x, n, alpha):
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    return (2.0 / n) * w
```
