import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    # Stable log-sum-exp
    mx = sh.max()
    if mx > 500:
        lse = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        lse = np.log(np.sum(np.exp(sh)))
    val = m + lse / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            if nv > 0:
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def fista_optimize(h, n, deadline, configs=None):
    """FISTA (accelerated projected gradient) with smooth max."""
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    if configs is None:
        configs = [(100, 0.0), (500, 0.0), (2000, 0.0), (5000, 0.0)]
    
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        x = best_h.copy()
        y = x.copy()
        t_k = 1.0
        L = alpha * (2.0 / n) * n + 2.0 * mu  # Lipschitz estimate (crude)
        # Better: L = alpha * dx * n (each c_k involves n terms, derivative bounded by dx)
        # Actually the max gradient norm is bounded by alpha * max(w) * max|∂c/∂h| 
        # ∂c_k/∂h_j is at most 2*dx per element
        # So L ≈ alpha * 2 * dx * (some factor)
        L = alpha * 4.0 / n + 2.0 * mu
        L = max(L, 1.0)
        
        for it in range(5000):
            if time.time() > deadline - 10:
                break
            # Backtracking line search
            step = 1.0 / L
            g = smooth_max_grad(y, n, alpha, mu)
            # Project: x_new = P(x - step * g)
            x_trial = project_feasible(y - step * g, n)
            f_trial = smooth_max_obj(x_trial, n, alpha, mu)
            f_y = smooth_max_obj(y, n, alpha, mu)
            
            # Backtracking
            max_bt = 10
            bt = 0
            while f_trial > f_y - 0.5 * step * np.dot(g, x_trial - y) + 1e-15 * step:
                step *= 0.5
                x_trial = project_feasible(y - step * g, n)
                f_trial = smooth_max_obj(x_trial, n, alpha, mu)
                bt += 1
                if bt > max_bt:
                    break
            if bt > max_bt:
                break
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_trial + ((t_k - 1.0) / t_new) * (x_trial - x)
            x = x_trial
            t_k = t_new
            L = max(L * 1.1, 1.0)  # Slightly increase L
            
            c_trial = compute_c5(x, n)
            if c_trial < best_c:
                best_c = c_trial
                best_h = x.copy()
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):

    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

def fine_local_search(h, n, deadline, radius=30):
    """Very fine local search with pair moves."""
    h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            
            # Compute influence: which indices affect active lags?
            influence = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    influence[0:nv] += 1.0
                    influence[k:n] += 1.0
                else:
                    kk = -k
                    nv = n - kk
                    influence[kk:n] += 1.0
                    influence[0:nv] += 1.0
            
            top_idx = np.argsort(-influence)[:min(radius * 2, n)]
            
            for i_val in top_idx:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Move mass from j to i
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Move mass from i to j
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step

def multi_restart(h_base, n, deadline, n_restarts=5, noise_scale=0.01):
