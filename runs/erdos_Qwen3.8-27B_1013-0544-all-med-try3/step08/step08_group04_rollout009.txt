def project_box_sum(y, target=None):
    y = np.array(y, dtype=float, copy=True)
    n=len(y)
    if n==0: return y
    if target is None: target=0.5*float(n)
    y=np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    y=np.clip(y,0.0,1.0)
    if abs(float(np.sum(y))-target) < 1e-12:
        return y
    lo=float(np.min(y))-1.0
    hi=float(np.max(y))+1.0
    for _ in range(5):
       f_lo=float(np.sum(np.clip(y-lo,0,1)))
       if f_lo < target: lo -= 1.0
       f_hi=float(np.sum(np.clip(y-hi,0,1)))
       if f_hi > target: hi += 1.0
       else: break
    for _ in range(80):
       mid=0.5*(lo+hi)
       val=float(np.sum(np.clip(y-mid,0.0,1.0)))
       if val > target: lo=mid
       else: hi=mid

    x=_sum_adjust(x,target)
    return x

def apply_coord(C,h,idx,delta,n):
    if delta == 0.0: return
    # first-index
    C[idx:idx+n] += delta * (1.0 - h[::-1])

import numpy as np
from scipy.optimize import minimize

def _sum_adjust(x, target):
    x = np.array(x, dtype=float, copy=True)
    for _ in range(60):
        s = float(np.sum(x))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - x
            tot = float(np.sum(room))
            if tot <= 1e-15:
                break
            x += diff * room / tot
        else:
            tot = s
            if tot <= 1e-15:
                break
            x += (diff / tot) * x
    x = np.clip(x, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
            x[idx] = min(1.0, float(x[idx]) + diff)
        else:
            idx = int(np.argmax(x))
            x[idx] = max(0.0, float(x[idx]) + diff)
    return np.clip(x, 0.0, 1.0)

def project_box_sum(y, target=None):
    y = np.array(y, dtype=float, copy=True)
    n = len(y)
    if n == 0:
        return y
    if target is None:
        target = 0.5 * float(n)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    y = np.clip(y, 0.0, 1.0)
    if abs(float(np.sum(y)) - target) < 1e-12:
        return y
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(6):
        flo = float(np.sum(np.clip(y - lo, 0.0, 1.0)))
        fhi = float(np.sum(np.clip(y - hi, 0.0, 1.0)))
        if flo < target:
            lo -= 1.0
        if fhi > target:
            hi += 1.0
        if flo >= target and fhi <= target:
            break
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    x = _sum_adjust(x, target)
    return x

def interpolate_h(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    if n_old == 1:
        return np.full(n_new, float(h_old[0]), dtype=float)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return np.interp(x_new, x_old, h_old)

def compute_C(h, n=None):
    hf = np.asarray(h, dtype=float)
    if n is None:
        n = len(hf)
    return np.correlate(hf, 1.0 - hf, mode='full')

        n = len(hf)

def soft_max_value(C, alpha):
    if alpha <= 0:
        return float(np.max(C))
    m = float(np.max(C))
    z = C - m
    e = np.exp(alpha * z)
    s = float(np.sum(e))
    if s <= 0.0:
        return m
    return m + float(np.log(s) / alpha)

def softmax_weights(C, alpha):
    if alpha <= 0:
        return np.full(len(C), 1.0 / float(len(C)), dtype=float)
    m = float(np.max(C))
    z = C - m
    e = np.exp(alpha * z)
    s = float(np.sum(e))
    if s <= 0.0:
        w = np.zeros(len(C), dtype=float)
        w[int(np.argmax(C))] = 1.0
        return w
    return e / s

def gradient_weights(w, h, n):
    w = np.asarray(w, dtype=float)
    h = np.asarray(h, dtype=float)
    grad = np.zeros(n, dtype=float)
    c = n - 1
    w0 = float(w[c])
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wd = float(w[c + d])
        if wd > 1e-15:
            L = n - d
            grad[d:] += wd * (1.0 - h[:L])
            grad[:L] -= wd * h[d:]
    for L in range(1, n):
        wd = float(w[c - L])
        if wd > 1e-15:
            L2 = n - L
            grad[:L2] += wd * (1.0 - h[L:])
            grad[L:] -= wd * h[:L2]
    return grad

def extreme_min(a, n):
    a = np.asarray(a, dtype=float)
    order = np.argsort(a)
    target = 0.5 * float(n)
    x = np.zeros(n, dtype=float)
    k = int(target)
    if k > 0:
        x[order[:k]] = 1.0
    rem = target - float(k)
    if rem > 1e-12 and k < n:
        x[order[k]] = rem
    return x

def logsum_jac(x, n, alpha):

def slsqp_opt(h_start, n, alpha, maxiter, deadline):

def apply_coord(C, h, idx, delta, n):
    if delta == 0.0:
        return

                        if time.time() >= deadline:
                            break
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break
        if not improved:
            break
    C, h, best = light_resync(C, h, n, best)
    return C, h, best

def radius_scan(C, h, n, best, deadline, radius, deltas):
    C = compute_C(h, n)
    best = float(np.max(C))
    radius = max(2, int(radius))
    max_sweeps = 12
    attempts = 0
    for delta in deltas:
        sweep = 0
        while sweep < max_sweeps and time.time() < deadline - 1.0:
            improved = False
            for i in range(n):
                hi = min(n, i + radius + 1)
                for j in range(i + 1, hi):
                    na = try_pair(C, h, n, i, j, delta, best)
                    attempts += 1
                    if na < best:
                        best = na
                        improved = True
                    nb = try_pair(C, h, n, j, i, delta, best)
                    attempts += 1
                    if nb < best:
                        best = nb
                        improved = True
                    if attempts >= 1000:
                        C, h, best = light_resync(C, h, n, best)
                        attempts = 0
                        if time.time() >= deadline - 1.0:
                            break
                if time.time() >= deadline - 1.0:
                    break
            sweep += 1
            if not improved:
                break
            C, h, best = light_resync(C, h, n, best)
    C, h, best = light_resync(C, h, n, best)
    return C, h, best

def optimize_candidate(h_start, n, deadline, rng=None):
    h = interpolate_h(h_start, int(n))
    h = project_box_sum(h, int(n))
    C = compute_C(h, int(n))
    best = float(np.max(C))
    n = int(n)
    if n <= 600 and time.time() < deadline - 30:
        alphas = [30, 120, 500]
        for a in alphas:
            if time.time() > deadline - 25:
                break
            rem = deadline - time.time()
            if rem < 25:
                break
            maxit = int(max(20, min(100, 30000.0 / float(n))))
            if rem < 60:
                maxit = min(maxit, 40)
            h_new = slsqp_opt(h, n, a, maxit, deadline - 12)
            C2 = compute_C(h_new, n)
            v2 = float(np.max(C2))
            if v2 < best - 1e-12:
                best = v2
                C = C2
                h = h_new
            elif rem <= 20:
                break
    if n <= 1200 and time.time() < deadline - 15:
        h, best = fw_line_search(h, n, best, deadline - 10, rng)
        C, h, best = light_resync(compute_C(h, n), h, n, best)
    if time.time() < deadline - 10:
        C, h, best = directed_pair_search(C, h, n, best, deadline - 6, rng)
    if time.time() < deadline - 4:
        if n <= 300:
            radius = 35
            deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 1e-4]
        elif n <= 600:
            radius = 22
            deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 2e-4, 1e-4]
        else:
            radius = 14
            deltas = [0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]
        C, h, best = radius_scan(C, h, n, best, deadline - 3, radius, deltas)
    C, h, best = light_resync(C, h, n, best)
    return h, best * (2.0 / float(n))

def build_candidate_ns(n0):
    raw = [int(n0)]
    ds = [-10, -5, -2, -1, 1, 2, 5, 10, 15]
    for d0 in ds:
        raw.append(int(n0) + d0)
    fixed = [80, 100, 120, 150, 180, 200, 220, 250, 280, 320, 400, 500, 650]
    raw.extend(fixed)
    for m in [0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.1]:
        raw.append(int(round(float(n0) * m)))
    filtered = []
    for x in raw:
        if x < 20:
            x = 20
        if x > 1200:
            x = 1200
        x = int(x)
        if x not in filtered:
            filtered.append(x)
    rest = []
    for x in filtered:
        if x != int(n0):
            rest.append((abs(x - int(n0)), x))
    rest.sort()
    return [int(n0)] + [z[1] for z in rest[:10]]

def fallback_initial(n, rng=None):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    noise = 0.05 * rng.randn(n) if rng is not None else 0.0
    h = 0.5 + 0.25 * np.sin(2.0 * np.pi * x) + 0.10 * np.sin(4.0 * np.pi * x + 0.3) + noise
    return project_box_sum(h, int(n))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    t_end = t0 + max(25.0, budget - 12.0)
    rng = np.random.RandomState(seed)

    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_init = int(len(init_h))
        init_h = project_box_sum(init_h, n_init)
    except Exception:
        n_init = 250
        init_h = fallback_initial(n_init, rng)

    best_h = init_h.copy()
    best_c = compute_c5(best_h, n_init)

    if budget < 60 or time.time() > t_end - 20:
        best_h = project_box_sum(best_h, int(len(best_h)))
        return best_h, float(compute_c5(best_h, int(len(best_h)))), int(len(best_h))

    cands = build_candidate_ns(n_init)

    if cands:
        alloc1 = min(320.0, max(60.0, 0.35 * (t_end - time.time())))
        dl1 = min(t_end - 60.0, time.time() + alloc1)
        if dl1 > time.time() + 20:
            try:
                h2, c2 = optimize_candidate(best_h, cands[0], dl1, rng)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2
            except Exception:
                pass

    tried = set([cands[0]] if cands else [])
    for ntry in cands[1:5]:
        if time.time() > t_end - 100.0:
            break
        if ntry in tried:
            continue
        tried.add(ntry)
        alloc = min(150.0, max(30.0, 0.20 * (t_end - time.time())))
        if alloc < 35:
            break
        dl = min(t_end - 80.0, time.time() + alloc)
        if dl <= time.time() + 20:
            break
        try:
            h2, c2 = optimize_candidate(best_h, int(ntry), dl, rng)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        except Exception:
            pass

    n_best = int(len(best_h))
    if time.time() < t_end - 80:
        for r in range(2):
            if time.time() > t_end - 70:
                break
            scale = 0.02 + 0.03 * r
            hp = best_h + rng.normal(0.0, scale, n_best)
            hp = project_box_sum(hp, n_best)
            alloc = min(80.0, max(25.0, 0.25 * (t_end - time.time())))
            if alloc < 25:
                break
            dl = min(t_end - 60.0, time.time() + alloc)
            if dl <= time.time() + 15:
                break
            try:
                h2, c2 = optimize_candidate(hp, n_best, dl, rng)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2
            except Exception:
                pass

    if time.time() < t_end - 20:
        n = int(len(best_h))
        C = compute_C(best_h, n)
        uns = float(np.max(C))
        if n <= 500 and time.time() < t_end - 30:
            rem = t_end - time.time()
            maxit = int(min(60, max(20, 20000.0 / float(n))))
            h2 = slsqp_opt(best_h, n, 1000, maxit, time.time() + min(30.0, 0.35 * rem))
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        if n <= 1000 and time.time() < t_end - 12:
            C = compute_C(best_h, n)
            uns = float(np.max(C))
            h2, _ = fw_line_search(h2 if n <= 500 and time.time() < t_end else best_h, n, uns,
                                   time.time() + min(50.0, 0.35 * (t_end - time.time())), rng)
            h2 = project_box_sum(h2, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        if time.time() < t_end - 8:
            C = compute_C(best_h, n)
            uns = float(np.max(C))
            C, h2, uns = directed_pair_search(C, best_h, n, uns,
                                              time.time() + min(30.0, 0.25 * (t_end - time.time())), rng)
            c2 = uns * (2.0 / float(n))
            if c2 < best_c:
                best_h = h2
                best_c = c2
        if time.time() < t_end - 4:
            C = compute_C(best_h, n)
            uns = float(np.max(C))
            if n <= 300:
                radius = 20
                deltas = [0.002, 0.001, 0.0005, 0.0002, 1e-4, 5e-5]
            else:
                radius = 12
                deltas = [0.001, 0.0005, 0.0002, 1e-4, 5e-5]
            C, h2, uns = radius_scan(C, best_h, n, uns, t_end - 3, radius, deltas)
            c2 = uns * (2.0 / float(n))
            if c2 < best_c:
                best_h = h2
                best_c = c2

    n = int(len(best_h))
    best_h = project_box_sum(best_h, n)
    c_final = compute_c5(best_h, n)
    return best_h, float(c_final), n
