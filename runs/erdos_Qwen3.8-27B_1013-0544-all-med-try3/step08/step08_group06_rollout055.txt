                                    vj_val += omh[idx]
                                if j >= lo2 and j < hi2:
                                    idx2 = j - k + (n - 1)
                                    vj_val -= h[idx2]
                                v[j, k] = vj_val
                            improved = True
                            break
                if time.time() > deadline - 1:
                    break
            if not improved:
                break
    return h, best_c

def compute_unit_grad_row(idx, h, n, sign):
    """Compute dc/dh[idx] for a single index (without dx)."""
    omh = 1.0 - h
    row = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        val = 0.0
        m1 = idx + k - (n - 1)
        if 0 <= m1 < n:
            val += omh[m1]
        m2 = idx - k + (n - 1)
        if 0 <= m2 < n:
            val -= h[m2]
        row[k] = val
    return row

def pair_search_simple(h, n, deadline, deltas, max_pairs_per_pass=None):
    """Simpler pair search - recompute c each time."""
    h = h.copy()
    best_c = float(np.max(compute_c5_vec(h, n)))
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(50):
            if time.time() > deadline:
                break
            improved = False
            count = 0
            for i in range(n):
                if time.time() > deadline - 0.5:
                    break
                if max_pairs_per_pass and count > max_pairs_per_pass:
                    break
                for j in range(i + 1, n):
                    count += 1
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi and hi <= 1.0 and 0.0 <= hj and hj <= 1.0:
                        h[i] = hi
                        h[j] = hj
                        c = compute_c5_vec(h, n)
                        cm = float(c.max())
                        if cm < best_c - 1e-14:
                            best_c = cm
                            improved = True
                        else:
                            h[i] = hi - d
                            h[j] = hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 and hi2 <= 1.0 and 0.0 <= hj2 and hj2 <= 1.0:
                        h[i] = hi2
                        h[j] = hj2
                        c = compute_c5_vec(h, n)
                        cm = float(c.max())
                        if cm < best_c - 1e-14:
                            best_c = cm
                            improved = True
                        else:
                            h[i] = hi2 + d
                            h[j] = hj2 - d
                        if time.time() > deadline - 0.5:
                            break
                if time.time() > deadline - 0.5:
                    break
            if not improved:
                break
    return h, best_c

def resample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slsqp_refine(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sh_max = sh.max()
        if sh_max > 500:
            return float(m + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def full_optimize(h_start, n, deadline, rng):
    t_start = time.time()
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: PGD with increasing alpha
    t1 = t_start + min(30, (deadline - t_start) * 0.3)
    h_pgd, c_pgd = pgd_optimize(best_h, n, t1,
                                 alpha_start=50.0, alpha_end=10000.0, n_iters=3000)
    if c_pgd < best_c:
        best_c = c_pgd
        best_h = h_pgd
    
    # Phase 2: PGD with higher alpha
    if time.time() < deadline - 15:
        t2 = time.time() + min(30, (deadline - time.time()) * 0.3)
        h_pgd2, c_pgd2 = pgd_optimize(best_h, n, t2,
                                       alpha_start=500.0, alpha_end=100000.0, n_iters=2000)
        if c_pgd2 < best_c:
            best_c = c_pgd2
            best_h = h_pgd2
    
    # Phase 3: SLSQP refinement
    if time.time() < deadline - 10:
        t3 = min(40, deadline - time.time() - 5)
        h_s = slsqp_refine(best_h, n, alpha=50000, maxiter=3000)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
    
    # Phase 4: Pair search
    if time.time() < deadline - 3:
        t4 = deadline - 1
        if n <= 150:
            best_h, best_c = pair_search_fast(best_h, n, t4,
                                               [0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            best_h, best_c = pair_search_simple(best_h, n, t4,
                                                 [0.01, 0.005, 0.002, 0.001])
    
    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 2:
        h_f = slsqp_refine(best_h, n, alpha=200000, maxiter=2000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200
    
    if h_init is None:
        h_init = np.full(n_init, 0.5)
    
    best_h = None
    best_c = float('inf')
    
    # Primary optimization on initial resolution
    alloc1 = min(300, (deadline - t0) * 0.4)
    h_opt, c_opt = full_optimize(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt
    
    # Try different resolutions
    x_init = np.linspace(0, 2, n_init)
    resolutions = []
    if n_init != 100:
        resolutions.append(100)
    if n_init != 150:
        resolutions.append(150)
    if n_init != 200:
        resolutions.append(200)
    resolutions.append(300)
    resolutions.append(80)
    
    for n_try in resolutions:
        if time.time() > deadline - 40:
            break
        remaining = deadline - time.time()
        alloc = min(80, remaining * 0.15)
        if alloc < 20:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        h_opt2, c_opt2 = full_optimize(h_interp, n_try, time.time() + alloc, rng)
        if c_opt2 < best_c:
            best_c = c_opt2
            best_h = h_opt2
    
    # Random restarts at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 60:
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.03 + 0.15 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 20) * 0.25)
            if alloc < 10:
                break
            h_opt3, c_opt3 = full_optimize(hp, n_best, time.time() + alloc, rng)
            if c_opt3 < best_c:
                best_c = c_opt3
                best_h = h_opt3
    
    # Final polish
    if time.time() < deadline - 5:
        n_fin = len(best_h)
        if n_fin <= 150:
            t_fin = deadline - 1
            best_h, best_c = pair_search_fast(best_h, n_fin, t_fin,
                                               [0.005, 0.002, 0.001, 0.0005, 0.0003])
        else:
            t_fin = deadline - 1
            best_h, best_c = pair_search_simple(best_h, n_fin, t_fin,
                                                 [0.005, 0.002, 0.001, 0.0005])
    
    # Very final SLSQP if time allows
    if time.time() < deadline - 3:
        h_final = slsqp_refine(best_h, len(best_h), alpha=500000, maxiter=3000)
        c_final = compute_c5(h_final, len(best_h))
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    # Ensure we return before deadline
    if time.time() > deadline + 1:
        pass  # We should be fine at this point
    
    return best_h, best_c, len(best_h)
