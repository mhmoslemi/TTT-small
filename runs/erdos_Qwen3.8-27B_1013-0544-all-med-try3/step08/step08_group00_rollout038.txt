    """Minimize max_i (a[i]*x + b[i]) for x in [lo, hi]."""
    # Find the minimum of the upper envelope
    # The minimum is at a vertex of the upper envelope
    # Use a simple approach: evaluate at many points, then refine
    
    # Start with a grid
    best_x = lo
    best_val = np.inf
    # Evaluate at 0, 1, and some interior points
    points = [lo, hi, 0.25, 0.5, 0.75]
    for x in points:
        val = np.max(a * x + b)
        if val < best_val:
            best_val = val
            best_x = x
    # Refine with golden section search (since function is convex)
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a_gs, b_gs = lo, hi
    c = b_gs - gr * (b_gs - a_gs)
    d = a_gs + gr * (b_gs - a_gs)
    fc = np.max(a * c + b)
    fd = np.max(a * d + b)
    for _ in range(50):
        if fc < fd:
            b_gs = d
            d = c
            fd = fc
            c = b_gs - gr * (b_gs - a_gs)
            fc = np.max(a * c + b)
        else:
            a_gs = c
            c = d
            fc = fd
            d = a_gs + gr * (b_gs - a_gs)
            fd = np.max(a * d + b)
    x_opt = (a_gs + b_gs) / 2
    return x_opt

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_a_for_coord(i, h, n, dx):
    a = np.zeros(2 * n - 1)
    n1 = n - 1
    for m in range(2 * n - 1):
        k = m - n1
        am = 0.0
        ik = i - k
        if 0 <= ik <= n1:
            am += (1.0 - h[ik])
        iek = i + k
        if 0 <= iek <= n1:
            am -= h[iek]
        a[m] = am * dx
    return a

def minimize_max_lines(a, b):
    lo = 0.0
    hi = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = float(np.max(a * c + b))
    fd = float(np.max(a * d + b))
    for _ in range(45):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = float(np.max(a * c + b))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = float(np.max(a * d + b))
    return (lo + hi) * 0.5

def coord_descent(h, n, deadline, max_sweeps=5000, random_order=False, seed_offset=0):

    best_c = compute_c5(h, n)

                max_change = max(max_change, abs(delta))

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        k = mm - (n - 1)

            grad[0:nv] += wm * (1.0 - x[k:n])

            if nv > 0:

        return float(x.sum() - n / 2.0)

        return np.ones(n)

def pairwise_cd(h, n, deadline, max_rounds=200):

    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()

            j = np.random.randint(n)
