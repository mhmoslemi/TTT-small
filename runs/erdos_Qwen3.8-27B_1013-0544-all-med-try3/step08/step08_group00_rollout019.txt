def adam_optimize(h, n, deadline, lr=0.001, beta1=0.9, beta2=0.999,
                  alpha=5000, top_m=20, momentum=0.9):
    """Custom Adam optimizer with projected steps."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    eps = 1e-8
    n_active = max(5, top_m)
    
    while time.time() < deadline:
        t += 1
        
        # Compute gradient
        grad = smooth_max_grad(h, n, alpha, top_m=n_active)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Projected step
        h_new = h - step
        h_new = project_feasible(h_new, n)
        
        # Evaluate
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            # Occasionally increase alpha

                alpha = min(alpha * 1.5, 100000)
                n_active = min(n_active + 2, 40)
        else:

            lr *= 0.95
            if lr < 1e-8:
                break
        
        # Check time
        if time.time() > deadline:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _corr_grad_single(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            g[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        g[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            g[0:nv] += (-h[kk:n]) * dx
    return g

def smooth_max_grad_full(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx]
        g = _corr_grad_single(h, n, k)
        grad += wm * g
    return grad

def smooth_max_grad_top(h, n, alpha, top_m):
    c = compute_c5_vec(h, n)
    if top_m >= len(c):
        top_idx = np.arange(len(c))
        c_sub = c
        m_sub = c.max()
    else:
        top_idx = np.argsort(-c)[:top_m]
        c_sub = c[top_idx]
        m_sub = c_sub.max()
    sh = alpha * (c_sub - m_sub)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for j, idx in enumerate(top_idx):
        if w[j] < 1e-14:
            continue
        k = int(idx) - (n - 1)
        g = _corr_grad_single(h, n, k)
        grad += w[j] * g
    return grad

def obj_ls(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_ls(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)

        k = idx - (n - 1)
        g = _corr_grad_single(x, n, k)
        grad += w[idx] * g
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

        if len(active) > 15:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:15]]
        delta, _ = sqp_lp(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-9:
                break
            continue
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_imp = 0
            if rho < 0.08:
                rho = min(rho * 1.4, 0.1)
        else:
            no_imp += 1
            rho *= 0.5
            if rho < 1e-10 or no_imp > 40:
                break
    return best_h, best_c

def upsample_func(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def perturb_func(h, n, scale):
    h_p = h + scale * np.random.randn(n)
    return project_feasible(h_p, n)

def local_search_pairs(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    steps = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for idx in active[:10]:
                grad += np.abs(_corr_grad_single(h, n, int(idx) - (n - 1)))
            grad /= min(len(active), 10)
            top_i = np.argsort(-grad)[:min(radius, n)]
            for i_val in top_i:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    np.random.seed(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS at original resolution (25s)
    t_phase = min(t0 + 25, deadline - 950)
    if t_phase > time.time() + 3:
        h, c = run_lbfgs(h, n, t_phase, [500, 2000], mu=50.0)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Adam at original resolution (25s)
    t_phase = min(t0 + 50, deadline - 900)
    if t_phase > time.time() + 3:
        h, c = run_adam(best_h, n, t_phase, lr_init=0.02, alpha=3000, top_m=15)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: SQP at original resolution (25s)
    t_phase = min(t0 + 75, deadline - 850)
    if t_phase > time.time() + 5:
        h, c = run_sqp(best_h, n, t_phase, max_iter=150, rho_init=0.05)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Upsample to n=1000
    n2 = 1000
    h2 = upsample_func(best_h, n, n2)

    # Phase 4a: L-BFGS at n=1000 (90s)
    t_phase = min(time.time() + 90, deadline - 400)
    if t_phase > time.time() + 5:
        h2, c2 = run_lbfgs(h2, n2, t_phase, [1000, 3000, 8000], mu=200.0)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 4b: Adam at n=1000 (80s)
    t_phase = min(time.time() + 80, deadline - 250)
    if t_phase > time.time() + 5:
        h2, c2 = run_adam(best_h, n2, t_phase, lr_init=0.008, alpha=5000, top_m=20)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 4c: SQP at n=1000 (50s)
    t_phase = min(time.time() + 50, deadline - 150)
    if t_phase > time.time() + 5:
        h2, c2 = run_sqp(best_h, n2, t_phase, max_iter=200, rho_init=0.04)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 4d: L-BFGS sharp at n=1000 (30s)
    t_phase = min(time.time() + 30, deadline - 100)
    if t_phase > time.time() + 5:
        h2, c2 = run_lbfgs(best_h, n2, t_phase, [20000, 50000], mu=2000.0)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 5: Local search at n=1000 (30s)
    t_phase = min(time.time() + 30, deadline - 60)
    if t_phase > time.time() + 3:
        h2, c2 = local_search_pairs(best_h, n2, t_phase, radius=25)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 6: Perturbation restarts (40s)
    n_curr = len(best_h)
    for trial in range(3):
        t_phase = min(time.time() + 15, deadline - 40)
        if t_phase <= time.time() + 3:
            break
        h_p = perturb_func(best_h, n_curr, 0.008)
        h_p, c_p = run_lbfgs(h_p, n_curr, t_phase, [5000, 15000], mu=1000.0)
        if c_p < best_c:
            best_c, best_h = c_p, h_p.copy()

    # Phase 7: Try n=2000 if time permits
    n3 = 2000
    if time.time() < deadline - 120:
        h3 = upsample_func(best_h, len(best_h), n3)
        t_phase = min(time.time() + 60, deadline - 50)
        if t_phase > time.time() + 5:
            h3, c3 = run_lbfgs(h3, n3, t_phase, [3000, 10000], mu=1000.0)
            if c3 < best_c:
                best_c, best_h = c3, h3.copy()

        t_phase = min(time.time() + 40, deadline - 25)
        if t_phase > time.time() + 3:
            h3, c3 = run_adam(best_h, n3, t_phase, lr_init=0.005, alpha=8000, top_m=30)
            if c3 < best_c:
                best_c, best_h = c3, h3.copy()

    # Phase 8: Final SQP polish
    n_f = len(best_h)
    t_phase = deadline - 2
    if t_phase > time.time() + 5:
        h_f, c_f = run_sqp(best_h, n_f, t_phase, max_iter=100, rho_init=0.02)
        if c_f < best_c:
            best_c, best_h = c_f, h_f.copy()

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
