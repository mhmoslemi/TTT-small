```
    target=0.5*n

    diff=target-h.sum()
    if abs(diff)<=1e-14: return np.clip(h,0,1)
```

```
def adjust_sum(h,n):
    h=np.asarray(h, dtype=float)
    target=0.5*float(n)
    diff=target - float(h.sum())
    if abs(diff) <= 1e-14:
        return np.clip(h,0,1)
    if diff >0:
        idx=np.where(h < 1.0)[0]
        if idx.size==0: return np.clip(h,0,1)
        order=idx[np.argsort(-(1.0 - h[idx]))]
    else:
        idx=np.where(h > 0.0)[0]
        if idx.size==0: return np.clip(h,0,1)
        order=idx[np.argsort(-h[idx])]
    for ii in order:
        diff=target - float(h.sum())
        if abs(diff) <= 1e-15: break
        if diff > 0:
            add=diff
            room=1.0 - h[ii]
            if add > room: add=room
            h[ii] += add
        else:
            add=diff
            room=-h[ii]
            if add < room: add=room
            h[ii] += add
    return np.clip(h,0,1)
```

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog


def adjust_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    diff = target - float(h.sum())
    if abs(diff) <= 1e-14:
        return np.clip(h, 0.0, 1.0)

    if diff > 0.0:
        idx = np.where(h < 1.0)[0]
        if idx.size == 0:
            return np.clip(h, 0.0, 1.0)
        order = idx[np.argsort(-(1.0 - h[idx]))]
    else:
        idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            return np.clip(h, 0.0, 1.0)
        order = idx[np.argsort(-h[idx])]

    for ii in order:
        diff = target - float(h.sum())
        if abs(diff) <= 1e-15:
            break
        if diff > 0.0:
            room = 1.0 - h[ii]
            add = diff if diff <= room else room
            h[ii] += add
        else:
            room = -h[ii]
            add = diff if diff >= room else room
            h[ii] += add

    return np.clip(h, 0.0, 1.0)


def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1).copy()
    n = int(n)
    target = 0.5 * float(n)

    if y.size == 0:
        return np.zeros(0, dtype=float)

    y_min = float(np.min(y))
    y_max = float(np.max(y))
    pad = max(2.0, (y_max - y_min) + 2.0)
    lo = y_min - pad
    hi = y_max + pad

    mid = 0.5 * (lo + hi)
    h = np.clip(y + mid, 0.0, 1.0)

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo <= 1e-14:
            break

    h = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    return adjust_sum(h, n)


def compute_c5_vec(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))


def shift_grad_single(h, n, idx):
    """
    Gradient of the exact np.correlate(h, 1-h, 'full') entry idx.
    For idx, let p = (n-1) - idx.  The entry is
      sum_i h_i (1 - h_{i+p}) when p >= 0,
      sum_i h_i (1 - h_{i+|p|}) for the opposite side when p < 0.
    """
    h = np.asarray(h, dtype=float)
    n = int(n)
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    p = (n - 1) - int(idx)

    if p >= 0:
        nv = n - p
        grad[0:nv] += (1.0 - h[p:n])
        grad[p:n] += -h[0:nv]
    else:
        q = -p
        nv = n - q
        grad[q:n] += (1.0 - h[0:nv])
        grad[0:nv] += -h[q:n]

    return grad * dx


def top_k_active(c_vec, k):
    k = int(k)
    if k <= 0:
        return []
    if k > len(c_vec):
        k = len(c_vec)
    return list(np.argsort(-c_vec)[:k])


def safe_alpha(h, d):
    h = np.asarray(h, dtype=float)
    d = np.asarray(d, dtype=float)
    amax = 10.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        amax = min(amax, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        amax = min(amax, float(np.min((-h[neg]) / (-d[neg]))))
    if not np.isfinite(amax) or amax <= 0.0:
        return 0.0
    return max(amax, 0.0)


def line_search_dir(h, d, cur_c, n, deadline):
    h = np.asarray(h, dtype=float)
    d = np.asarray(d, dtype=float)

    if not np.any(np.abs(d) > 1e-12):
        return None, cur_c, 0.0

    amax = safe_alpha(h, d)
    if amax <= 1e-12:
        return None, cur_c, 0.0

    raw_alphas = (min(amax, 4.0), 2.0, 1.0, 0.5, 0.25, 0.125, 0.05, 0.01)
    alphas = []
    for a in raw_alphas:
        if not np.isfinite(a) or a <= 0.0:
            continue
        if a > amax * (1.0 + 1e-12):
            continue
        add = True
        for b in alphas:
            if abs(a - b) <= 1e-14:
                add = False
                break
        if add:
            alphas.append(a)
        if len(alphas) >= 8:
            break

    best_c = cur_c
    best_h = None
    best_a = 1.0

    for a in alphas:
        if time.time() > deadline:
            break
        h_try = project_box_sum(h + a * d, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-13:
            best_c = c_try
            best_h = h_try
            best_a = a

    if best_h is None:
        return None, cur_c, 0.0
    return best_h, best_c, best_a


def lp_unified_dir(h, n, active, rho, c_vec=None):
    if len(active) == 0:
        return None, 0.0

    h = np.asarray(h, dtype=float)
    n = int(n)
    if c_vec is None:
        c_vec = compute_c5_vec(h, n)
    else:
        c_vec = np.asarray(c_vec, dtype=float)

    cmax = float(c_vec.max())
    m = len(active)
    A_ub = np.zeros((m, n + 1), dtype=float)
    b_ub = np.zeros(m, dtype=float)

    for row, idx in enumerate(active):
        idx = int(idx)
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = 1.0
        rhs = cmax - float(c_vec[idx])
        if rhs < 0.0:
            rhs = 0.0
        b_ub[row] = rhs

    lo = np.maximum(-rho, -h)
    hi = np.minimum(rho, 1.0 - h)
    bad = lo > hi
    if np.any(bad):
        lo[bad] = 0.5 * (lo[bad] + hi[bad])
        hi[bad] = lo[bad]

    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[-1] = -1.0

    bounds = list(zip(lo, hi))
    bounds.append((0.0, 1.0))

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1, dtype=float)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
        if res.success:
            d = np.asarray(res.x[0:n], dtype=float)
            if np.max(np.abs(d)) > 1e-12:
                s = float(res.x[n])
                return d, s
    except Exception:
        pass

    return None, 0.0

def active_lp_polish(h, n, deadline, active_count=12, rho_init=0.05,

        active = top_k_active(c_vec, act)

        d, s = lp_unified_dir(cur_h, n, active, rho, c_vec)

            if dm is not None:
                d = dm

        if d is None or not np.any(np.abs(d) > 1e-12):
            fails += 1
            rho = max(0.002, rho * 0.6)
            if fails % 8 == 0:
                act = min(24, act + 4)
            if rho < 1e-4 or fails > max_no_improve:
                break
            continue

        if h_try is not None:
            cur_h = h_try
            cur_c = c_try
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = cur_h.copy()
                fails = 0
                if alpha > 0.95:
                    rho = min(0.12, max(rho * 1.5, 0.02))
                else:
                    rho = min(0.12, max(rho * 1.1, 0.015))
        else:

            rho = max(0.002, rho * 0.7)
            if fails > max_no_improve:

    cur_h = project_box_sum(cur_h, n)
    cur_c = compute_c5(cur_h, n)
    if cur_c < best_c:
        best_h = cur_h.copy()
        best_c = cur_c

    best_h = project_box_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def pair_transfer_search(h, n, deadline, kinc=25, kdec=25):
    h = project_box_sum(h, int(n))
    n = int(n)
    cur_h = h.copy()
    cur_c = compute_c5(cur_h, n)
    best_h = cur_h.copy()
    best_c = cur_c

    steps = (0.08, 0.04, 0.02, 0.01, 0.005)
    no_improve = 0

    while time.time() < deadline - 2.0:
        c_vec = compute_c5_vec(cur_h, n)
        active = top_k_active(c_vec, 35)

        g = np.zeros(n, dtype=float)
        for idx in active:
            g += shift_grad_single(cur_h, n, int(idx))
        g /= max(1, len(active))
        g -= float(g.mean())

        inc = np.argsort(g)[:min(int(kinc), n)]
        dec = np.argsort(-g)[:min(int(kdec), n)]

        changed = False
        for i in inc:
            if changed or time.time() > deadline - 2.0:
                break
            i = int(i)
            for j in dec:
                if changed or time.time() > deadline - 2.0:
                    break
                j = int(j)
                if i == j:
                    continue
                maxeps = min(float(cur_h[j]), float(1.0 - cur_h[i]))
                for eps in steps:
                    if eps > maxeps:
                        continue
                    if time.time() > deadline - 2.0:
                        break
                    cur_h[i] += eps
                    cur_h[j] -= eps
                    c_try = compute_c5(cur_h, n)
                    if c_try < cur_c - 1e-13:
                        cur_c = c_try
                        changed = True
                        break
                    else:
                        cur_h[i] -= eps
                        cur_h[j] += eps
                if changed:
                    break

        if changed:
            cur_h = adjust_sum(cur_h, n)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = cur_h.copy()
            no_improve = 0
            continue

        no_improve += 1
        if no_improve > 3:
            break

    best_h = project_box_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def soft_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)


def soft_max_jac(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    w = np.exp(sh - mx)
    wsum = float(w.sum())
    if wsum <= 0.0:
        return np.zeros(n, dtype=float)
    w = w / wsum

    maxw = float(w.max())
    thresh = max(1e-14, 1e-10 * maxw)
    idxs = np.where(w > thresh)[0]
    if idxs.size > 80:
        order = np.argsort(-w[idxs])
        idxs = idxs[order[:80]]

    w = w[idxs]
    s = float(w.sum())
    if s <= 0.0:
        return np.zeros(n, dtype=float)
    w = w / s

    grad = np.zeros(n, dtype=float)
    for ii, wi in zip(idxs, w):
        if wi > 0.0:
            grad += wi * shift_grad_single(x, n, int(ii))
    return grad


def smooth_pen_obj(x, n, alpha, mu):
    val = soft_max_obj(x, n, alpha)
    s = float(x.sum() - 0.5 * float(n))
    return float(val + 0.5 * mu * s * s)


def smooth_pen_jac(x, n, alpha, mu):
    s = float(x.sum() - 0.5 * float(n))
    return soft_max_jac(x, n, alpha) + mu * s * np.ones(n, dtype=float)


def lbfgs_smooth(h, n, deadline, configs=None):
    h = project_box_sum(h, int(n))
    n = int(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    if configs is None:
        configs = [
            (300.0, 50.0),
            (1500.0, 500.0),
            (5000.0, 2000.0),
        ]

    for alpha, mu in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                smooth_pen_obj,
                h,
                args=(n, alpha, mu),
                jac=smooth_pen_jac,
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 220, "ftol": 1e-18, "gtol": 1e-14,
                         "maxls": 40, "maxfun": 2000}
            )
            if res.x is not None:
                h_try = project_box_sum(res.x, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try.copy()
        except Exception:
            pass

    best_h = project_box_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def sum_con_fun(x, n):
    return float(x.sum() - 0.5 * float(n))


def sum_con_jac(x, n):
    return np.ones(n, dtype=float)


def slqp_soft_max(h, n, deadline, alpha=20000.0, maxiter=80):
    h = project_box_sum(h, int(n))
    n = int(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if time.time() > deadline - 10.0:
        return best_h, best_c

    bounds = [(0.0, 1.0)] * n
    cons = [
        {
            "type": "eq",
            "fun": sum_con_fun,
            "jac": sum_con_jac,
            "args": (n,)
        }
    ]

    try:
        res = minimize(
            soft_max_obj,
            h,
            args=(n, alpha),
            method="SLSQP",
            jac=soft_max_jac,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-17, "disp": 0}
        )
        if res.x is not None:
            h_try = project_box_sum(res.x, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                best_h = h_try.copy()
                best_c = c_try
    except Exception:
        pass

    best_h = project_box_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def make_anti_periodic(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    if n % 2 != 0:
        return project_box_sum(h, n)
    m = n // 2
    a = 0.5 * (h[:m] + (1.0 - h[m:]))
    a = np.clip(a, 0.0, 1.0)
    g = np.empty(n, dtype=float)
    g[:m] = a
    g[m:] = 1.0 - a
    return adjust_sum(g, n)


def make_reflection_symmetric(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    f = np.flip(h.copy())
    g = 0.5 * (h + f)
    return project_box_sum(g, n)


def make_periodic_one(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    if n % 2 != 0:
        return project_box_sum(h, n)
    m = n // 2
    g = np.empty(n, dtype=float)
    g[:m] = h[:m]
    g[m:] = h[:m]
    return adjust_sum(g, n)


def structured_perturb(h, n, strength, rng):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    d = np.zeros(n, dtype=float)
    mode = int(rng.randint(0, 3))

    if mode == 0:
        d += rng.normal(size=n)
    elif mode == 1 and n % 2 == 0:
        m = n // 2
        a = rng.normal(size=m)
        d[:m] = a
        d[m:] = -a
    else:
        f = min(4, max(1, n // 2))
        x = np.linspace(0.0, 2.0, n)
        for mvar in range(1, f + 1):
            d += float(rng.normal()) * np.cos(np.pi * mvar * x)
            d += float(rng.normal()) * np.sin(np.pi * mvar * x)

    d -= float(d.mean())
    amax = float(np.max(np.abs(d)))
    if not np.isfinite(amax) or amax <= 1e-15:
        return project_box_sum(h, n)

    d *= float(strength) / amax
    return project_box_sum(h + d, n)


def downsample_interp(h, n_src, n_dst):
    h = np.asarray(h, dtype=float).copy()
    n_src = int(n_src)
    n_dst = int(n_dst)
    if n_dst == n_src:
        return project_box_sum(h, n_dst)

    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    y = np.interp(x_new, x_old, h)
    return project_box_sum(y, n_dst)


def try_high_resolution(best_h, n, deadline, rng):
    cur = time.time()
    if deadline - cur < 170.0:
        return None, None, None

    n_high = 1000
    if n >= n_high:
        n_high = int(min(2000, n + 200))

    h = downsample_interp(best_h, n, n_high)
    c = compute_c5(h, n_high)

    if not np.isfinite(c) or c > 0.40000:
        h = structured_perturb(h, n_high, 0.10, rng)
        c = compute_c5(h, n_high)

    t1 = min(time.time() + 120.0, deadline - 100.0)
    if time.time() < t1:
        h, c = active_lp_polish(h, n_high, t1,
                                active_count=8, rho_init=0.025,
                                max_no_improve=20)

    if n_high <= 900 and time.time() < deadline - 40.0:
        t2 = min(time.time() + 40.0, deadline - 20.0)
        h, c = pair_transfer_search(h, n_high, t2, kinc=20, kdec=20)

    h = project_box_sum(h, n_high)
    c = compute_c5(h, n_high)
    return h, c, n_high


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(1.0, float(budget_s)) - 8.0
    rng = np.random.RandomState(seed)

    h_init = None
    try:
        h_init = np.asarray(initial_h_values, dtype=float).copy().reshape(-1)
    except Exception:
        h_init = None

    if h_init is None or h_init.size == 0 or not np.all(np.isfinite(h_init)):
        n0 = 800
        m0 = n0 // 2
        a = rng.uniform(0.0, 1.0, size=m0)
        h0 = np.empty(n0, dtype=float)
        h0[:m0] = a
        h0[m0:] = 1.0 - a
        h_init = h0

    n = int(h_init.size)
    best_h = project_box_sum(h_init, n)
    best_c = compute_c5(best_h, n)

    # Cheap symmetric transformations, kept only if they immediately improve.
    variants = []
    if n % 2 == 0:
        variants.append(make_anti_periodic(best_h, n))
        variants.append(make_periodic_one(best_h, n))
    variants.append(make_reflection_symmetric(best_h, n))

    for hh in variants:
        if hh is not None and hh.size == n:
            hh = project_box_sum(hh, n)
            cc = compute_c5(hh, n)
            if np.isfinite(cc) and cc < best_c - 1e-15:
                best_c = cc
                best_h = hh.copy()
                n = int(hh.size)

    # Short quasi-Newton warmup using a stable soft-max model.
    t1 = min(t0 + 80.0, deadline - 500.0)
    if time.time() < t1:
        h_try, c_try = lbfgs_smooth(
            best_h, n, t1,
            configs=[(300.0, 50.0), (1500.0, 500.0), (5000.0, 2000.0)]
        )
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try

    # Main active-set LP line search.
    t2 = min(t0 + 270.0, deadline - 460.0)
    if time.time() < t2:
        h_try, c_try = active_lp_polish(
            best_h, n, t2, active_count=14, rho_init=0.05, max_no_improve=30
        )
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try

    # Exact two-coordinate transfer local search.
    t3 = min(t0 + 360.0, deadline - 400.0)
    if time.time() < t3:
        h_try, c_try = pair_transfer_search(
            best_h, n, t3, kinc=28, kdec=28
        )
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try

    # Another LP pass after the discrete transfers.
    t4 = min(t0 + 450.0, deadline - 340.0)
    if time.time() < t4:
        h_try, c_try = active_lp_polish(
            best_h, n, t4, active_count=12, rho_init=0.03, max_no_improve=25
        )
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try

    n = int(best_h.size)
    pool = []
    scales = (0.30, 0.20, 0.12, 0.07, 0.04, 0.025, 0.012)

    high_need = (n < 900) and (deadline - time.time() > 280.0)
    high_reserve = 230.0 if high_need else 0.0

    attempt = 0
    while time.time() < deadline - high_reserve - 90.0:
        attempt += 1
        scale = scales[attempt % len(scales)]

        if attempt % 4 == 0 and pool:
            base_idx = (attempt // 4) % len(pool)
            base_h, base_n = pool[base_idx]
        else:
            base_h, base_n = best_h, n

        if attempt % 5 == 0 and base_n >= 120:
            n_small = max(70, base_n // 3)
            cand = downsample_interp(base_h, base_n, n_small)
            cand = downsample_interp(cand, n_small, base_n)
            cand = structured_perturb(cand, base_n, 0.18, rng)
        else:
            cand = structured_perturb(base_h, base_n, scale, rng)

        cand_n = int(cand.size)
        cand_c = compute_c5(cand, cand_n)

        if cand_c < best_c + 0.004:
            local_budget = 55.0
        elif cand_c < best_c + 0.015:
            local_budget = 28.0
        else:
            local_budget = 12.0

        local_end = min(time.time() + local_budget,
                        deadline - high_reserve - 60.0)
        if time.time() < local_end:
            cand, cand_c = active_lp_polish(
                cand, cand_n, local_end,
                active_count=10, rho_init=0.04, max_no_improve=15
            )
            cand_n = int(cand.size)
            cand_c = compute_c5(cand, cand_n)

        if cand_c < best_c + 0.003 and cand_n == n and time.time() < deadline - high_reserve - 40.0:
            part_end = min(time.time() + 15.0, deadline - high_reserve - 25.0)
            if time.time() < part_end:
                cand, cand_c = pair_transfer_search(
                    cand, cand_n, part_end, kinc=22, kdec=22
                )
                cand_c = compute_c5(cand, cand_n)

        if cand_c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = float(cand_c)
            n = int(cand.size)
            pool = []
        elif np.isfinite(cand_c) and cand_c < best_c + 0.006 and len(pool) < 10:
            keep = True
            for ph, pn in pool:
                if pn == n and cand_n == n:
                    if float(np.linalg.norm(cand - ph)) < 0.045:
                        keep = False
                        break
            if keep:
                pool.append((cand.copy(), cand_n))

    # Final low-resolution polish before attempting higher resolution.
    t5 = min(time.time() + 35.0, deadline - 40.0)
    if time.time() < t5:
        h_try, c_try = active_lp_polish(
            best_h, n, t5, active_count=10, rho_init=0.015, max_no_improve=20
        )
        if c_try < best_c - 1e-15:
            best_h = h_try
            best_c = c_try
            n = int(best_h.size)

    # Higher-resolution continuation if there is enough time.
    if high_need and time.time() < deadline - 25.0:
        h_high, c_high, n_high = try_high_resolution(best_h, n, deadline - 5.0, rng)
        if h_high is not None and np.isfinite(c_high) and c_high < best_c - 1e-15:
            best_h = h_high
            best_c = float(c_high)
            n = int(n_high)
            pool = []

    # Last polish on the best current resolution.
    if time.time() < deadline - 4.0:
        h_try, c_try = active_lp_polish(
            best_h, n, deadline - 1.0,
            active_count=8, rho_init=0.012, max_no_improve=18
        )
        if c_try < best_c - 1e-15:
            best_h = h_try
            best_c = c_try
            n = int(best_h.size)

    best_h = project_box_sum(best_h, int(n))
    best_c = compute_c5(best_h, n)

    if not np.isfinite(best_c) or not np.all(np.isfinite(best_h)):
        best_h = project_box_sum(h_init, n)
        best_c = compute_c5(best_h, n)

    return best_h, float(best_c), int(n)
```
