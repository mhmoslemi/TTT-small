def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # Softmax approximation of max
    v = shifted.max()
    return float(m + (v + np.log(np.sum(np.exp(shifted - v)))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

    except Exception:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_subgradient(h, n, active_idx):
    """Compute subgradient of average of active overlap lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    count = len(active_idx)
    if count == 0:
        return grad
    for idx in active_idx:
        k = idx - (n - 1)  # lag
        if k == 0:
            grad += (1.0 - 2.0 * h) * dx / count
        elif k > 0:
            grad[:n-k] += (1.0 - h[k:]) * dx / count
            grad[k:] -= h[:n-k] * dx / count
        else:
            k_neg = -k
            grad[k_neg:] += (1.0 - h[:n-k_neg]) * dx / count
            grad[:n-k_neg] -= h[k_neg:] * dx / count
    return grad

def subgradient_descent(h, n, deadline, num_active=5, lr_init=0.1):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stall = 0
    
    while time.time() < deadline:
        c_all = compute_all_overlaps(h, n)
        # Get top num_active lags (with a small tolerance)
        sorted_idx = np.argsort(c_all)[::-1]
        top_val = c_all[sorted_idx[0]]
        active_idx = [i for i in sorted_idx if c_all[i] > top_val - 1e-6][:num_active]
        
        grad = compute_subgradient(h, n, active_idx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        h_new = project_feasible(h - lr * grad / gnorm, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        h = h_new
        # Adapt lr
        if stall > 50:
            lr *= 0.7
            stall = 0
            if lr < 1e-8:
                break
        elif c_new > best_c:
            lr *= 0.95
        else:
            lr = min(lr * 1.02, 0.5)
    
    return best_h, best_c

def block_to_full(h_block, m, n):
    """Upsample m block values to n points."""
    # Each block has n/m points (assume n is divisible by m)
    return np.repeat(h_block, n // m)

def block_c5(h_block, m, n):
    h_full = block_to_full(h_block, m, n)
    return compute_c5(h_full, n)

def optimize_blocks(m, n, deadline, h_init_block):
    """Optimize m block heights to minimize C5."""
    def obj(params):
        h_block = params
        h_full = block_to_full(h_block, m, n)
        return compute_c5(h_full, n)
    
    bounds = [(0.0, 1.0)] * m
    
    def con_fun(params):
        return np.sum(params) - m / 2.0
    
    def con_jac(params):
        return np.ones(m)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    best_params = h_init_block.copy()
    best_c = obj(best_params)
    
    # Try multiple SLSQP runs with different alphas
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > deadline:

            h_full = block_to_full(params, m, n)

            mx = c.max()
            shifted = alpha * (c - mx)
            v = shifted.max()
            return mx + (v + np.log(np.sum(np.exp(shifted - v)))) / alpha
        
        try:
            res = minimize(smooth_obj, best_params, method='SLSQP',
                         bounds=bounds, constraints=cons,

            c_new = obj(res.x)
            if c_new < best_c:
                best_c = c_new

        except Exception:
            pass
    
    # Also try Nelder-Mead on the block parameters
    if time.time() < deadline - 10:
        try:
            res = minimize(obj, best_params, method='Nelder-Mead',
                         bounds=bounds, options={'maxiter': 5000, 'xatol': 1e-10})
            c_new = obj(np.clip(res.x, 0, 1))

                best_c = c_new
                best_params = np.clip(res.x, 0, 1)
        except Exception:
            pass
    
    best_params = project_feasible_block(best_params, m)
    best_c = obj(best_params)
    return block_to_full(best_params, m, n), best_c

def project_feasible_block(h_block, m):
    target = m / 2.0
    h_block = np.clip(h_block.copy(), 0, 1)
    for _ in range(100):
        diff = target - h_block.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h_block
            tot = room.sum()
            if tot < 1e-12: break
            h_block += diff / tot * room
        else:
            tot = h_block.sum()
            if tot < 1e-12: break
            h_block += diff / tot * h_block
    return np.clip(h_block, 0, 1)

Let me also think about what blocks the optimal solution might use. Given the symmetry of the problem (domain [0,2], shifts up to 2), the optimal might have:
- 2-fold symmetry: h(x) = h(2-x)
- Or 2-fold anti-symmetry: h(x) + h(2-x) = 1

If h(x) + h(2-x) = 1, then at lag k, the overlap involves h(x) and h(x+k). If k = 2-2x for some x, then h(x+k) = h(2-x) = 1-h(x), so overlap = h(x)(1-(1-h(x))) = h(x)². This doesn't simplify things.

Let me just go with the optimization and not over-constrain.

One more idea: what if the optimal h has the form:
h(x) = 1/2 + f(x) where f is a "spread out" function?

The constraint ∫h = 1 means ∫f = 0.
The overlap at lag k:
∫(1/2+f(x))(1/2-f(x+k))dx = 1/4 - 1/2∫f(x+k) - 1/2∫f(x) + ∫f(x)f(x+k) 
= 1/4 + 0 + 0 + ∫f(x)f(x+k) (since ∫f=0 and the domain is symmetric)

Wait, that's not quite right because of the boundary effects. But approximately:
O_k ≈ 1/4 + ∫f(x)f(x+k)dx

So we want ∫f(x)f(x+k) to be as negative as possible for all k, or at least as uniform as possible.

If f has a "flat" autocorrelation (like a random function), then ∫f(x)f(x+k) ≈ 0 for k ≠ 0, and ≈ Var(f) for k=0.

But we want it to be negative for all k to minimize the max. This means f should be "anti-autocorrelated" - like a Zadoff-Chu sequence or a Barker sequence in continuous form.

In 1D, a function with anti-autocorrelation would be something like:
f(x) = sign(sin(2πx * frequency)) (a square wave)

For a square wave of frequency 1: f(x) = 1 for x ∈ [0,1/2], -1 for x ∈ [1/2,1], repeated.
Autocorrelation at lag k: for k=0, it's 1. For k=1/2, it's -1. For other k, it varies.

This gives h(x) = 1/2 + A·sign(sin(4πx)) which alternates between 1/2+A and 1/2-A every 1/4 units (with period 1/2).

Actually, let me just compute: h(x) = 0 or 1, alternating in blocks of length 1/4:
h = [1, 0, 1, 0, 1, 0, 1, 0] (8 blocks of 1/4 each)
Sum = 4/8 * 2 = 1 ✓

At lag 0: overlap = sum of h(1-h) = 0 (since h is 0 or 1)
At lag 1/4: h shifts by one block: [0, 1, 0, 1, ...] overlap = sum of h(x)*h(x+1/4) = 0
At lag 1/2: h shifts by two: same as h, overlap = sum h(1-h) = 0
At lag 3/4: shifts by three: anti-pattern, overlap = 0
At lag 1: shifts by four: same as h, overlap = 0
...

So for the alternating binary pattern, C₅ = 0! But wait, this is at the discrete level. The issue is that at non-block-aligned lags, there's partial overlap.

Let me check more carefully with n=200, blocks of 25 samples:
h = [1]*25 + [0]*25 + [1]*25 + [0]*25 + [1]*25 + [0]*25 + [1]*25 + [0]*25

At lag 1 (one sample): the overlap involves h[i] and h[i+1]. Most terms are 0 (since adjacent blocks are different), but at the boundaries (4 boundaries, each contributing 1*0 or 0*1 = 0), actually all boundary terms are 0 too since h goes 1→0 or 0→1.

Wait: at the boundary between block 0 (all 1s) and block 1 (all 0s): h[24]=1, h[25]=0.
Overlap term: h[24]*(1-h[25]) = 1*1 = 1. And h[25]*(1-h[24]) = 0*0 = 0.
So at lag 1, the overlap at the boundary is 1 (one sample contributes 1 to the correlation).
Similarly at each boundary, one sample contributes.
There are 4 boundaries in the "1→0" direction, contributing 4 to the positive-lag correlation.
In the "0→1" direction, h[i]=0, h[i+1]=1: h[i]*(1-h[i+1]) = 0. No contribution.

So at lag 1: correlation = 4, times dx = 4 * 2/200 = 0.04.

At lag 25 (one block): h shifts by one block. The pattern becomes [0,1,0,1,...] relative to [1,0,1,0,...].
Overlap: h[i]*(1-h[i+25]). Since h[i+25] = 1-h[i] (anti-pattern), 1-h[i+25] = h[i].
So overlap = sum h[i]² = sum h[i] = 100. Times dx = 100*2/200 = 1.

That's bad! At lag = block_width, the overlap is 1.

So the alternating pattern is bad at lags that are multiples of the block width. This makes sense: if h alternates with period 1/2, then at lag 1/2, h(x+1/2) = 1-h(x), so (1-h(x+1/2)) = h(x), and overlap = ∫h² = ∫h = 1.

So we can't use a simple periodic pattern. We need something more complex.

What about a "3-level" or "5-level" pattern? Or a pattern with incommensurate periods?

Actually, I think the key is that the optimal solution has a "quasi-random" structure - it looks random but is carefully designed to have low peak autocorrelation.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_subgrad(h, n, active_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    count = len(active_idx)
    if count == 0:
        return grad
    inv_count = 1.0 / count
    for idx in active_idx:
        k = idx - (n - 1)
        if k == 0:
            grad += (1.0 - 2.0 * h) * dx * inv_count
        elif k > 0:
            grad[:n - k] += (1.0 - h[k:]) * dx * inv_count
            grad[k:] -= h[:n - k] * dx * inv_count
        else:
            kp = -k
            grad[kp:] += (1.0 - h[:n - kp]) * dx * inv_count
            grad[:n - kp] -= h[kp:] * dx * inv_count
    return grad

def subgrad_descent(h, n, deadline, num_active=5, lr=0.3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    while time.time() < deadline - 2:
        c_all = compute_all_overlaps(h, n)
        top_val = c_all.max()
        tol = max(1e-9, 0.01 * top_val)
        active_idx = np.where(c_all >= top_val - tol)[0]
        if len(active_idx) > num_active:
            sorted_sub = np.argsort(c_all[active_idx])[::-1][:num_active]
            active_idx = active_idx[sorted_sub]
        grad = compute_subgrad(h, n, list(active_idx))
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = (lr / gnorm) * grad
        h_new = project_feasible(h - step, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 100:
            lr *= 0.6
            stall = 0
            if lr < 0.001:
                break
        elif c_new >= best_c + 0.005:
            lr *= 0.8
    return best_h, best_c

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    v = shifted.max()
    return float(m + (v + np.log(np.sum(np.exp(shifted - v)))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def block_to_full(h_block, m, n):
    reps = n // m
    return np.repeat(h_block, reps)[:n]

def project_block(h_block, m):
    target = m / 2.0
    h_block = np.clip(h_block.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h_block.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h_block
            tot = room.sum()
            if tot < 1e-12:
                break
            h_block += diff / tot * room
        else:
            tot = h_block.sum()
            if tot < 1e-12:
                break
            h_block += diff / tot * h_block
    return np.clip(h_block, 0.0, 1.0)

def optimize_blocks(m, n, deadline, h_init_block=None):
    if h_init_block is None:
        h_init_block = project_block(np.full(m, 0.5), m)
    else:
        h_init_block = project_block(h_init_block.copy(), m)
    
    best_params = h_init_block.copy()
    best_c = compute_c5(block_to_full(best_params, m, n), n)
    
    def block_obj(params):
        return compute_c5(block_to_full(params, m, n), n)
    
    bounds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': lambda p: float(np.sum(p) - m / 2.0),
             'jac': lambda p: np.ones(m)}]
    
    for alpha in [100, 500, 2000, 10000, 50000]:
        if time.time() > deadline - 5:
            break
        def smooth_block_obj(params):
            h_full = block_to_full(params, m, n)
            return smooth_max_obj(h_full, n, alpha)
        try:
            res = minimize(smooth_block_obj, best_params, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-15})
            c_new = block_obj(np.clip(res.x, 0, 1))
            if c_new < best_c:
                best_c = c_new
                best_params = project_block(res.x, m)
        except Exception:
            pass
    
    if time.time() < deadline - 10:
        try:
            res = minimize(block_obj, best_params, method='Nelder-Mead',
                           bounds=bounds, options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-14})
            c_new = block_obj(np.clip(res.x, 0, 1))
            if c_new < best_c:
                best_c = c_new
                best_params = project_block(res.x, m)
        except Exception:
            pass
    
    best_params = project_block(best_params, m)
    best_c = block_obj(best_params)
    return block_to_full(best_params, m, n), best_c, best_params

def targeted_local_search(h, n, deadline, max_passes=30):
    best_c = compute_c5(h, n)
    h = h.copy()
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    radius = max(10, n // 10)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            c_all = compute_all_overlaps(h, n)
            max_c = c_all.max()
            for i in range(n):
                if time.time() > deadline:
                    break
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if h[i] - d >= 0 and h[j] + d <= 1:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if h[j] - d >= 0 and h[i] + d <= 1:
                        h[j] -= d
                        h[i] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
            if not improved:
                break
    return h, best_c
