def binary_greedy_search(h, n, deadline, max_sweeps=10000):
    """Greedy binary local search: try all swaps, take the best improvement each sweep."""
    target = n // 2
    h = h.astype(float).copy()
    # Round to binary
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_swap = None
        best_new_c = best_c
        one_list = list(ones)
        zero_list = list(zeros)
        
        # Try all swaps
        for i in zero_list:
            for j in one_list:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                h_bin[i], h_bin[j] = 0.0, 1.0
                if c < best_new_c - 1e-15:

        if best_swap is None:

        i, j = best_swap
        h_bin[i], h_bin[j] = 1.0, 0.0
        ones.remove(j)
        zeros.remove(i)
        ones.add(i)
        zeros.add(j)
        best_c = best_new_c
    
    return h_bin, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff * room / s
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff * h / s
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsumexp = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsumexp / alpha
    
    def eq_constraint(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_constraint}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
    except Exception:
        h_new = h.copy()
    return h_new
