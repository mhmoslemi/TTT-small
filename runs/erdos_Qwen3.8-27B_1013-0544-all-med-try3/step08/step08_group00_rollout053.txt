import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_gradient(h, n, active_idx):
    """Compute gradient of sum of active C_m values."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def benders_lp(h, n, active_idx, rho):
    """Solve LP: min t s.t. linearized C_m ≤ t, sum(h)=n/2, |delta|≤rho, 0≤h+delta≤1."""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[0:nv] = (1.0 - h[k:n])
            g[k:n] += (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:n] = (1.0 - h[0:nv])
            g[0:nv] += (-h[kk:n])
        g *= (2.0 / n)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

            active = [int(np.argmax(c_vec))]
        if len(active) > 15:

            active = [active[i] for i in top_order[:15]]
        delta, t_opt = benders_lp(h, n, active, rho)

            continue

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    
    def sq_jac_fn(x, n):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def interpolate_h(h, n_old, n_new):

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
