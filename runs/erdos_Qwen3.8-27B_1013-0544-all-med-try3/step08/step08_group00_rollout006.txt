def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_vec(x, n, alpha, mu):
    """Vectorized smooth gradient computation."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    
    # grad[j] = dx * sum_k w[k] * [(1-x[j+k])*I(j+k in range) - x[j-k]*I(j-k in range)]
    # This is a weighted sum of shifted (1-x) and -x
    grad = np.zeros(n)
    
    for idx in range(2*n-1):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:

def binary_block_search(h, n, deadline, max_block=8):
    """Try flipping blocks of consecutive values in a binary-like solution."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    target = n / 2.0

    h_bin = np.where(h > 0.5, 1.0, 0.0)

    n_ones = int(target)
    # Sort h values, keep top n_ones as 1
    indices = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[indices[:n_ones]] = 1.0
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
        h = h_bin.copy()
    
    # Block flip search
    steps = list(range(1, max_block+1))
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for blk_len in steps:
            if time.time() > deadline:
                break
            for start in range(0, n - blk_len + 1):
                if time.time() > deadline:
                    return best_h, best_c
                # Try flipping block
                h_new = h.copy()
                h_new[start:start+blk_len] = 1.0 - h_new[start:start+blk_len]

                    h = h_new
                    improved = True
                    start += blk_len  # skip ahead
        # After one pass, try again with smaller blocks

    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

    n2m1 = 2 * n - 1

            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
