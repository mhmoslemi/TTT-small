```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alpha_list=None, maxiter=None):
    if alpha_list is None:
        alpha_list = [100, 500, 2000, 10000, 50000, 200000]
    if maxiter is None:
        maxiter = 10000
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline - 3:
            break
        mu = alpha * 0.01  # scale mu with alpha
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxiter*10, 'maxls': 50})

            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def binary_swap_search(h, n, deadline, max_rounds=200):

    hb = (h > 0.5).astype(float)
    # Fix to have exactly n/2 ones
    target_ones = n // 2

        idx = zeros[np.argmax(h[zeros])]
        hb[idx] = 1.0

        ones = np.where(hb == 1)[0]
        if len(ones) == 0: break
        idx = ones[np.argmin(h[ones])]
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def lbfgs_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
    
def run_lbfgs(h, n, deadline, alphas=None, iters_per_alpha=5000):
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000, 500000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline - 5:
            break
        mu = a * 0.005
        try:
            res = minimize(lbfgs_smooth_obj, h, jac=lbfgs_smooth_grad,
                          args=(n, a, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': iters_per_alpha, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': iters_per_alpha * 10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
```

Let me also think about the **binary swap search** more carefully.

For the binary case, h[i] ∈ {0,1} and sum(h) = n/2. A swap is: change h[a] from 1 to 0 and h[b] from 0 to 1. The change in c[k] is:
Δc[k] = (change in ∑_i h[i]*(1-h[i+k])) * dx

For k ≥ 0:
c[k] = ∑_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx

If we swap h[a]: 1→0 and h[b]: 0→1:
- Terms involving h[a] in the h[i] position: for k ≥ 0, if a ≤ n-1-k, the term h[a]*(1-h[a+k]) changes from 1*(1-h[a+k]) to 0*(1-h[a+k]) = 0. Loss: (1-h[a+k]).
- Terms involving h[a] in the h[i+k] position: if a-k ≥ 0 and a-k ≤ n-1-k, i.e., k ≤ a ≤ n-1, the term h[a-k]*(1-h[a]) changes from h[a-k]*0 to h[a-k]*1 = h[a-k]. Gain: h[a-k].
- Similarly for h[b].

This gets complicated. Let me just evaluate the correlation after each swap. At n=500, computing the correlation is O(n) = 500 operations (using np.correlate which is O(n²) but with small constant). Actually, np.correlate at n=500 is O(n²) = 250K operations. So each swap evaluation is 250K. If I try 1000 swaps per round, that's 250M per round.

For a more efficient approach: compute the delta for each swap analytically, then only evaluate the full correlation for promising swaps.

For a swap (a: 1→0, b: 0→1), the change in c[k] for each k:

Let me define:
- For k ≥ 0: c[k] = ∑_{i=0}^{n-1-k} h[i]*(1-h[i+k])
- For k < 0: c[k] = ∑_{i=0}^{n-1+k} h[i-k]*(1-h[i]) (using i-k ≥ 0)

After swap: h'[a] = 0, h'[b] = 1, rest same.

Δc[k] for k ≥ 0:
= [contribution from i=a in h[i] term] + [contribution from i+k=a in h[i+k] term] + [contribution from i=b in h[i] term] + [contribution from i+k=b in h[i+k] term]

For h[a]: 1→0
- In h[i] position (i=a): if 0 ≤ a ≤ n-1-k, old = 1*(1-h[a+k]), new = 0. Δ = -(1-h[a+k])
- In h[i+k] position (i+k=a, i=a-k): if 0 ≤ a-k ≤ n-1-k, old = h[a-k]*(1-1) = 0, new = h[a-k]*(1-0) = h[a-k]. Δ = +h[a-k]

For h[b]: 0→1
- In h[i] position (i=b): if 0 ≤ b ≤ n-1-k, old = 0*(1-h[b+k]), new = 1*(1-h[b+k]). Δ = +(1-h[b+k])
- In h[i+k] position (i+k=b, i=b-k): if 0 ≤ b-k ≤ n-1-k, old = h[b-k]*(1-0) = h[b-k], new = h[b-k]*(1-1) = 0. Δ = -h[b-k]

So for k ≥ 0:
Δc[k] = [-(1-h[a+k]) if a ≤ n-1-k] + [h[a-k] if a-k ≥ 0 and a-k ≤ n-1-k]
       + [(1-h[b+k]) if b ≤ n-1-k] + [-h[b-k] if b-k ≥ 0 and b-k ≤ n-1-k] (all * dx)

This is O(1) per k per swap! So to evaluate the effect on all 2n-1 lags, it's O(n) per swap. Much faster than O(n²) for the full correlation.

But I need to check the max of c+Δc, which is O(n) anyway.

So per swap: O(n) to compute Δc for all lags, then O(n) to check max. Total O(n) per swap. At n=500, that's 500 ops per swap. 10000 swaps = 5M ops. Very fast!
