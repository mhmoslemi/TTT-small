def project_feasible(h, n):

            remaining = deadline - time.time() - 15
            n_restarts = max(2, int(remaining / 35))
            for r in range(n_restarts):
                if time.time() > deadline - 18:
                    break
                scale = 0.02 + 0.12 * (r / max(1, n_restarts))
                hp = best_h + rng.normal(0, scale, n_best)
                hp = project_feasible(hp, n_best)
                alloc = min(30, (deadline - time.time() - 18) * 0.4)
                if alloc < 10:
                    break
                h_o4, c_o4 = optimize_full(hp, n_best, time.time() + alloc, rng)
                if c_o4 < best_c:
                    best_c, best_h = c_o4, h_o4

        # === Final polish ===
        if time.time() < deadline - 8:
            n_best = len(best_h)
            # Very high alpha SLSQP
            ht = slsqp_opt(best_h, n_best, 500000, maxiter=3000, ftol=1e-16)
            c = compute_c5(ht, n_best)
            if c < best_c:
                best_c, best_h = c, ht

            # Ultra-fine local search
            if time.time() < deadline - 3:
                best_h, best_c = randomized_local_search(best_h, n_best, deadline - 1,
                                                          rng, base_delta=0.001,
                                                          max_iters=300000)
            best_h = project_feasible(best_h, n_best)
            best_c = compute_c5(best_h, n_best)

    else:
        # No parent - build from scratch
        for n_try in [200, 150]:
            if time.time() > deadline - 30:
                break
            h_start = make_smooth_init(n_try, rng)
            h_opt, c_opt = optimize_full(h_start, n_try,
                                         min(time.time() + 100, deadline - 15), rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
