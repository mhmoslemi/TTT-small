from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(len(x))

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    # Clip to avoid overflow
    sh = np.clip(sh, -500, 500)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def run_slsqp(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def pair_search_optimal(h, n, deadline, deltas):
    """Pair search with multiple step sizes."""
    best_c = compute_c5(h, n)
    improved_any = True
    while improved_any and time.time() < deadline:

        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

                        if c < best_c - 1e-15:

                        h[j] += d
                        c = compute_c5(h, n)

                            improved_any = True

                            h[i] += d

    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        out = project_feasible(res.x, n)
        return out
    except:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        improved = False
        for d in deltas:
            if time.time() > deadline:
                return h, best_c
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
        if not improved:
            break
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    # SLSQP with ramping alpha
    for alpha in [500, 5000, 50000]:
        if time.time() > deadline - 20:
            break
        t_start = time.time()
        maxit = max(500, int((deadline - 20 - time.time()) * 50))
        ht = run_slsqp(h, n, alpha, maxiter=maxit)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            h = ht
    
    # Pair search
    if time.time() < deadline - 10:
        h, best_c = pair_search(h, n, deadline - 5,
                                [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
    
    # Final SLSQP
    if time.time() < deadline - 5:
        ht = run_slsqp(h, n, 200000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    
    # Ultra-fine pair search
    if time.time() < deadline - 2:
        h, best_c = pair_search(h, n, deadline - 1,
                                [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
