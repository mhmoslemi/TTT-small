import time
import math
import numpy as np

def next_power_of_two(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def sanitize_vector(h):
    try:
        h = np.asarray(h, dtype=float).ravel()
    except Exception:
        return None
    if h.size == 0:
        return None
    h = np.where(np.isfinite(h), h, 0.0)
    return h


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if n <= 0:
        return h
    h = np.where(np.isfinite(h), h, 0.0)
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(60):
        s = float(h.sum())
        diff = target - s
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = target - s

    if abs(diff) > 1e-10:
        lo = float(h.min()) - 1.0
        hi = float(h.max()) + 1.0
        mid = 0.5 * (lo + hi)
        for _ in range(70):
            mid = 0.5 * (lo + hi)
            val = float(np.clip(h - mid, 0.0, 1.0).sum())
            if val > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - mid, 0.0, 1.0)
        s = float(h.sum())
        diff = target - s

    if abs(diff) > 1e-13:
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot > 0.0:
                h += (diff / tot) * room
            else:
                h += diff / float(n)
        else:
            tot = float(h.sum())
            if tot > 0.0:
                h += (diff / tot) * h
            else:
                h += diff / float(n)

    h = np.clip(h, 0.0, 1.0)

    # Tiny final correction, useful when the evaluator tests the sum tightly.
    diff = target - float(h.sum())
    if abs(diff) > 0.0:
        if diff > 0.0:
            idx = int(np.argmin(h))
            room = 1.0 - h[idx]
            add = min(diff, room)
            h[idx] += add
            diff -= add
            if diff > 0.0:
                order = np.argsort(h)
                for ii in order:
                    if diff <= 0.0:
                        break
                    room = 1.0 - h[ii]
                    add = min(diff, room)
                    h[ii] += add
                    diff -= add
        else:
            diff = -diff
            order = np.argsort(-h)
            for ii in order:
                if diff <= 0.0:
                    break
                take = min(h[ii], diff)
                h[ii] -= take
                diff -= take

    return np.clip(h, 0.0, 1.0)


def compute_c_values(h, n):
    """Return the 2n-1 correlation counts, without the dx factor, and their max."""
    h = np.asarray(h, dtype=float).ravel()
    if n <= 1:
        val = float(h[0] * (1.0 - h[0])) if n == 1 else 0.0
        c = np.array([val], dtype=float)
        return c, float(c.max())

    n = int(n)
    Nfft = next_power_of_two(2 * n - 1)
    H = np.fft.rfft(h, Nfft)
    ac = np.fft.irfft(H * H.conj(), Nfft)[:n].real

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[-1]

    d = np.arange(n, dtype=int)
    D = pref[n - d] - ac
    F = total - pref[d] - ac

    c = np.empty(2 * n - 1, dtype=float)
    c[0] = D[0]
    c[1:n] = D[1:]
    c[n:2 * n - 1] = F[1:]

    m = float(c.max())
    return c, m


def compute_c5_fast(h, n):
    h = np.asarray(h, dtype=float).ravel()
    m = compute_c_values(h, n)[1]
    return float(m * 2.0 / float(n))


def compute_c5_exact(h, n):
    h = np.asarray(h, dtype=float).ravel()
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))


def smooth_value_from_c(c, alpha):
    m = float(c.max())
    if alpha <= 0.0:
        return m
    sh = alpha * (c - m)
    # sh.max() is 0 in exact arithmetic; guard against roundoff.
    mx = float(sh.max())
    if not np.isfinite(mx):
        return m
    if mx > 500.0:
        s = float(np.sum(np.exp(sh - mx)))
        val = m + (mx + math.log(s + 1e-300)) / alpha
    else:
        s = float(np.sum(np.exp(sh)))
        val = m + math.log(s + 1e-300) / alpha
    return float(val)


def smooth_weights_from_c(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = (c >= m - 1e-12).astype(float)
        s = float(w.sum())
        if s <= 0.0:
            w = np.zeros_like(c)
            w[0] = 1.0
            s = 1.0
        w /= s
    else:
        w = e
        w /= s
    return w, m


def smooth_grad_weights(w, h, n):
    """
    Gradient of sum_i w_i c_i, where c is ordered as
      [D_0, D_1, ..., D_{n-1}, F_1, ..., F_{n-1}]
    with D_d = sum h[i](1-h[i+d]), F_d = sum h[i+d](1-h[i]).
    """
    n = int(n)
    h = np.asarray(h, dtype=float).ravel()
    w = np.asarray(w, dtype=float).ravel()

    if n == 1:
        return np.array([float(w[0]) * (1.0 - 2.0 * h[0])], dtype=float)

    a0 = float(w[0])
    a = w[1:n].copy()
    b = w[n:2 * n - 1].copy()

    s = a + b
    t = 1.0 - h

    K = np.zeros(2 * n - 1, dtype=float)
    center = n - 1
    if n - 1 > 0:
        K[center + 1:2 * n - 1] = s
        K[0:center] = s[::-1]

    Lconv = next_power_of_two(n + (2 * n - 1) - 1)
    T = np.fft.rfft(t, Lconv)
    KF = np.fft.rfft(K, Lconv)
    y = np.fft.irfft(T * KF, Lconv)
    g = y[center:center + n].real

    pa = np.zeros(n, dtype=float)
    pb = np.zeros(n, dtype=float)
    if n - 1 > 0:
        np.cumsum(a, out=pa[1:])
        np.cumsum(b, out=pb[1:])

    idx = np.arange(n, dtype=int)
    left = pa[idx]
    right = pb[n - idx - 1]

    grad = g - left - right
    grad += a0 * (1.0 - 2.0 * h)
    return grad


def penalty_obj(x, n, alpha, mu):
    x = np.asarray(x, dtype=float).ravel()
    c, _ = compute_c_values(x, n)
    val = smooth_value_from_c(c, alpha)
    if mu != 0.0:
        d = float(x.sum()) - 0.5 * float(n)
        val += mu * d * d
    return float(val)


def penalty_grad(x, n, alpha, mu):
    x = np.asarray(x, dtype=float).ravel()
    c, m = compute_c_values(x, n)
    w, _ = smooth_weights_from_c(c, alpha)
    grad = smooth_grad_weights(w, x, n)
    if mu != 0.0:
        grad = grad + 2.0 * mu * (float(x.sum()) - 0.5 * float(n))
    return grad


def interpolate_h_vector(h, n_new):
    h = np.asarray(h, dtype=float).ravel()
    n_old = len(h)
    n_new = int(n_new)
    if n_old == n_new:
        return h.copy()
    if n_old == 1:
        return np.full(n_new, float(h[0]), dtype=float)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)


def make_kronecker(n, alpha, beta):
    n = int(n)
    x = np.arange(n, dtype=float)
    f = alpha * x + beta
    f = f - np.floor(f)
    h = (f < 0.5).astype(float)

    target = n // 2
    cnt = int(h.sum())
    diff = target - cnt
    if diff > 0:
        cand = np.where(h == 0.0)[0]
        if cand.size > 0:
            order = np.argsort(f[cand])
            take = min(diff, cand.size)
            h[order[:take]] = 1.0
    elif diff < 0:
        cand = np.where(h == 1.0)[0]
        if cand.size > 0:
            order = np.argsort(-f[cand])
            take = min(-diff, cand.size)
            h[order[:take]] = 0.0

    return project_feasible(h, n)


def make_threshold_binary(h, n, threshold=0.5):
    h = np.asarray(h, dtype=float).ravel()
    if len(h) != n:
        h = interpolate_h_vector(h, n)
    b = (h > float(threshold)).astype(float)

    target = n // 2
    cnt = int(b.sum())
    diff = target - cnt
    if diff > 0:
        cand = np.where(b == 0.0)[0]
        if cand.size > 0:
            order = np.argsort(-h[cand])
            take = min(diff, cand.size)
            b[order[:take]] = 1.0
    elif diff < 0:
        cand = np.where(b == 1.0)[0]
        if cand.size > 0:
            order = np.argsort(h[cand])
            take = min(-diff, cand.size)
            b[order[:take]] = 0.0

    return project_feasible(b, n)


def generate_parametric_starts(n):
    starts = []
    alphas = [
        (math.sqrt(5.0) - 1.0) / 2.0,
        math.sqrt(2.0) - 1.0,
        1.0 / math.pi,
        (math.sqrt(3.0) - 1.0) / 2.0,
        0.5,
        0.25,
        0.1,
        0.7,
    ]
    betas = [0.0, 1.0 / 7.0, 2.0 / 7.0, 3.0 / 7.0, 4.0 / 7.0, 5.0 / 7.0]
    for a in alphas:
        for b in betas:
            starts.append(make_kronecker(n, a, b))
    return starts


def polish_lbfgs(h, n, deadline, max_calls=6, alphas=None, mus=None, maxiter=None):
    h = project_feasible(np.asarray(h, dtype=float).ravel(), int(n))
    n = int(n)
    best_h = h.copy()
    best_c = compute_c5_exact(best_h, n)

    if max_calls <= 0:
        return best_h, best_c

    if alphas is None:
        alphas = [20.0, 100.0, 500.0, 2000.0] if n <= 1200 else [20.0, 100.0, 500.0]
    if mus is None:
        mus = [1.0, 5.0]
    if maxiter is None:
        if n <= 400:
            maxiter = 800
        elif n <= 800:
            maxiter = 500
        elif n <= 1200:
            maxiter = 250
        else:
            maxiter = 120

    bounds = [(0.0, 1.0)] * n
    calls = 0
    h_cur = best_h.copy()

    for alpha in alphas:
        for mu in mus:
            if calls >= max_calls or time.time() >= deadline - 5.0:
                break
            rem = deadline - time.time()
            if rem < 15.0:
                break
            mi = int(maxiter)
            if n > 1500 and rem < 80.0:
                mi = min(mi, 80)
            if n > 1000 and rem < 45.0:
                mi = min(mi, 100)
            if rem < 20.0:
                mi = min(mi, 50)
            mi = max(10, mi)

            try:
                res = minimize(
                    penalty_obj,
                    h_cur,
                    jac=penalty_grad,
                    args=(n, float(alpha), float(mu)),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={
                        "maxiter": mi,
                        "ftol": 1e-15,
                        "gtol": 1e-11,
                        "maxfun": max(20, 4 * mi),
                        "maxls": 20,
                    },
                )
            except Exception:
                continue

            calls += 1
            if not hasattr(res, "x"):
                continue
            x = np.asarray(res.x, dtype=float).ravel()
            if x.size != n or not np.all(np.isfinite(x)):
                continue
            x = project_feasible(x, n)
            c = compute_c5_exact(x, n)

            if c < best_c:
                best_c = c
                best_h = x.copy()

            if c <= best_c + max(1e-9, 1e-6):
                h_cur = x
            else:
                h_cur = best_h

    if not np.all(np.isfinite(best_h)):
        best_h = project_feasible(h, n)
        best_c = compute_c5_exact(best_h, n)

    return best_h, float(best_c)


def estimate_max_after_move(h, c, watch, i, j, eps):
    """
    Estimated maximum over watch lags after h[i] += eps, h[j] -= eps.
    Exact for each watched lag; a non-watched lag may become active, so
    full evaluation is still used for acceptance.
    """
    n = len(h)
    i = int(i)
    j = int(j)
    e2 = float(eps) * float(eps)
    abs_ij = abs(i - j)
    maxv = -1.0e300

    for idx in watch:
        idx = int(idx)
        if idx == 0:
            delta = 2.0 * eps * (h[j] - h[i]) - 2.0 * e2
            val = c[idx] + delta
        elif idx < n:
            dd = idx
            L = n - dd
            if L <= 0:
                val = c[idx]
            else:
                delta = 0.0
                if i < L:
                    delta += eps * (1.0 - h[i + dd])
                if i >= dd:
                    delta -= eps * h[i - dd]
                if j < L:
                    delta -= eps * (1.0 - h[j + dd])
                if j >= dd:
                    delta += eps * h[j - dd]
                if abs_ij == dd:
                    delta += e2
                val = c[idx] + delta
        else:
            dd = idx - n + 1
            L = n - dd
            if L <= 0:
                val = c[idx]
            else:
                delta = 0.0
                if i >= dd:
                    delta += eps * (1.0 - h[i - dd])
                if i < L:
                    delta -= eps * h[i + dd]
                if j >= dd:
                    delta -= eps * (1.0 - h[j - dd])
                if j < L:
                    delta += eps * h[j - dd]
                if abs_ij == dd:
                    delta += e2
                val = c[idx] + delta

        if val > maxv:
            maxv = val

    return float(maxv)


def screened_local_search(h, n, deadline, watch_size=None, max_full_evals=None,
                          max_stagnation=30, max_rounds=100000):
    h = project_feasible(np.asarray(h, dtype=float).ravel(), int(n))
    n = int(n)
    h_best = h.copy()

    c, m = compute_c_values(h, n)
    if not np.isfinite(m):
        return project_feasible(h, n), compute_c5_exact(h, n)

    best_count = float(m)
    best_obj = float(best_count * 2.0 / n)

    if watch_size is None:
        watch_size = 60 if n <= 1024 else 40
    if max_full_evals is None:
        if n <= 512:
            max_full_evals = 90
        elif n <= 1024:
            max_full_evals = 50
        else:
            max_full_evals = 24

    eps_fracs = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625,
        0.03125, 0.015625, 0.0078125, 0.00390625
    ], dtype=float)

    low_idx = np.empty(0, dtype=int)
    high_idx = np.empty(0, dtype=int)

    stagnation = 0
    rounds = 0

    while time.time() < deadline and rounds < max_rounds:
        rounds += 1

        c, m = compute_c_values(h, n)
        if not np.isfinite(m):
            break
        if m < best_count - 1e-12:
            best_count = float(m)
            h_best = h.copy()
            best_obj = float(best_count * 2.0 / n)

        if c.size == 0:
            break

        watch = np.argsort(-c)[:watch_size]

        candidates = []
        active = watch[:12]
        top_per = 4

        for idx in active:
            idx = int(idx)
            if idx == 0:
                continue

            if idx < n:
                dd = idx
                L = n - dd
                if L <= 0:
                    continue
                contrib = h[:L] * (1.0 - h[dd:])
                if L > top_per:
                    part = np.argpartition(contrib, -top_per)[-top_per:]
                else:
                    part = np.arange(L)
                order = part[np.argsort(-contrib[part])]
                for off in order:
                    off = int(off)
                    i = off + dd
                    j = off
                    if h[j] > 1e-12 and (1.0 - h[i]) > 1e-12:
                        pr = float(contrib[off])
                        if np.isfinite(pr):
                            candidates.append((pr, i, j))
            else:
                dd = idx - n + 1
                L = n - dd
                if L <= 0:
                    continue
                contrib = h[dd:] * (1.0 - h[:L])
                if L > top_per:
                    part = np.argpartition(contrib, -top_per)[-top_per:]
                else:
                    part = np.arange(L)
                order = part[np.argsort(-contrib[part])]
                for off in order:
                    off = int(off)
                    i = off
                    j = off + dd
                    if h[j] > 1e-12 and (1.0 - h[i]) > 1e-12:
                        pr = float(contrib[off])
                        if np.isfinite(pr):
                            candidates.append((pr, i, j))

        # Gradient-like candidates from the current dominant lags.
        w = np.zeros(2 * n - 1, dtype=float)
        cw = c[watch]
        mxw = float(cw.max())
        sh = 30.0 * (cw - mxw)
        mxsh = float(sh.max())
        wt = np.exp(sh - mxsh)
        ssum = float(wt.sum())
        if ssum > 0.0 and np.isfinite(ssum):
            w[watch] = wt / ssum
        else:
            w[watch[0]] = 1.0

        grad = smooth_grad_weights(w, h, n)
        if n > 1 and np.all(np.isfinite(grad)):
            med = float(np.median(grad))
            low_mask = (h < 1.0 - 1e-8) & (grad < med)
            high_mask = (h > 1e-8) & (grad > med)
            low_idx = np.where(low_mask)[0]
            high_idx = np.where(high_mask)[0]

            if low_idx.size > 0 and high_idx.size > 0:
                low = low_idx[np.argsort(grad[low_idx])[:15]]
                high = high_idx[np.argsort(-grad[high_idx])[:15]]
                for i in low:
                    i = int(i)
                    for j in high:
                        j = int(j)
                        if i == j:
                            continue
                        pr = float(-(grad[i] - grad[j]))
                        if pr > 0.0 and np.isfinite(pr):
                            candidates.append((pr, i, j))

        cand_dict = {}
        for pr, i, j in candidates:
            key = i * n + j
            old = cand_dict.get(key)
            if old is None or pr > old[0]:
                cand_dict[key] = (pr, i, j)

        cand_list = list(cand_dict.values())
        cand_list.sort(reverse=True)
        if len(cand_list) > 500:
            cand_list = cand_list[:500]

        promising = []
        screened = 0
        for pr, i, j in cand_list:
            if screened >= 400:
                break
            screened += 1
            i = int(i)
            j = int(j)
            max_eps = min(1.0 - h[i], h[j])
            if max_eps <= 1e-12:
                continue
            for frac in eps_fracs:
                eps = float(frac) * max_eps
                if eps <= 1e-13:
                    continue
                est = estimate_max_after_move(h, c, watch, i, j, eps)
                if est < m - 1e-10:
                    promising.append((est, eps, i, j))
                    break

        if promising:
            promising.sort()

        improved = False
        fulls = 0

        for est, eps, i, j in promising:
            if fulls >= max_full_evals:
                break
            h2 = h.copy()
            h2[i] += eps
            h2[j] -= eps
            # Numerical guard; sum is preserved to roundoff.
            if h2[i] > 1.0:
                h2[i] = 1.0
            if h2[j] < 0.0:
                h2[j] = 0.0
            _, m2 = compute_c_values(h2, n)
            fulls += 1
            if m2 < m - 1e-12:
                h = h2
                improved = True
                break

        if not improved and cand_list:
            fulls = 0
            for pr, i, j in cand_list[:20]:
                if fulls >= max_full_evals:
                    break
                i = int(i)
                j = int(j)
                max_eps = min(1.0 - h[i], h[j])
                if max_eps <= 1e-12:
                    continue
                for eps in (0.010 * max_eps, 0.003 * max_eps):
                    if eps <= 1e-13:
                        break
                    h2 = h.copy()
                    h2[i] += eps
                    h2[j] -= eps
                    if h2[i] > 1.0:
                        h2[i] = 1.0
                    if h2[j] < 0.0:
                        h2[j] = 0.0
                    _, m2 = compute_c_values(h2, n)
                    fulls += 1
                    if m2 < m - 1e-12:
                        h = h2
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stagnation = 0
            continue

        stagnation += 1
        if stagnation > max_stagnation:
            break

        # Small stochastic kicks using only feasible moves; rejected if not improving.
        if low_idx.size > 0 and high_idx.size > 0:
            for _ in range(3):
                ii = int(low_idx[np.random.randint(low_idx.size)])
                jj = int(high_idx[np.random.randint(high_idx.size)])
                max_eps = min(1.0 - h[ii], h[jj])
                eps = 0.002 * max_eps
                if eps <= 1e-13:
                    continue
                h2 = h.copy()
                h2[ii] += eps
                h2[jj] -= eps
                if h2[ii] > 1.0:
                    h2[ii] = 1.0
                if h2[jj] < 0.0:
                    h2[jj] = 0.0
                _, m2 = compute_c_values(h2, n)
                if m2 < m - 1e-12:
                    h = h2
                    improved = True
                    break

        if improved:
            stagnation = 0

    c, m = compute_c_values(h, n)
    if np.isfinite(m) and m < best_count:
        best_count = float(m)
        h_best = h.copy()
        best_obj = float(best_count * 2.0 / n)

    # Return the exact-evaluated best between the final point and best fast point.
    c_try1 = compute_c5_exact(project_feasible(h_best, n), n)
    c_try2 = compute_c5_exact(project_feasible(h, n), n)
    if c_try1 <= c_try2:
        return project_feasible(h_best, n), float(c_try1)
    return project_feasible(h, n), float(c_try2)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(0.0, budget_s) - 8.0
    try:
        np.random.seed(int(seed))
    except Exception:
        np.random.seed(0)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        parent = sanitize_vector(initial_h_values)
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        npar = int(parent.size)
        hpar = project_feasible(parent, npar)
        cpar = compute_c5_exact(hpar, npar)
        best_h = hpar.copy()
        best_c = float(cpar)
        best_n = npar
    else:
        npar = 512
        hpar = make_kronecker(npar, (math.sqrt(5.0) - 1.0) / 2.0, 0.0)
        cpar = compute_c5_exact(hpar, npar)
        best_h = hpar
        best_c = float(cpar)
        best_n = npar

    if best_h is None or not np.all(np.isfinite(best_h)) or not np.isfinite(best_c):
        best_h = project_feasible(np.full(512, 0.5, dtype=float), 512)
        best_c = compute_c5_exact(best_h, 512)
        best_n = 512

    b = max(1.0, budget_s)
    t1 = t0 + 0.10 * b
    t2 = t0 + 0.33 * b
    t3 = t0 + 0.48 * b
    t4 = t0 + 0.70 * b

    # Phase 1: polish the supplied parent at its native resolution.
    if best_n <= 1500 and time.time() < deadline - 30.0:
        limit1 = min(deadline, t1)
        if limit1 > time.time() + 10.0:
            h1, c1 = polish_lbfgs(best_h, best_n, limit1, max_calls=4, maxiter=400)
            if c1 < best_c:
                best_h = h1.copy()
                best_c = float(c1)
                best_n = int(best_n)
            if limit1 > time.time() + 10.0:
                h1, c1 = screened_local_search(
                    best_h, best_n, limit1,
                    max_stagnation=20,
                    watch_size=50 if best_n <= 800 else 40,
                    max_full_evals=60 if best_n <= 800 else 30
                )
                if c1 < best_c:
                    best_h = h1.copy()
                    best_c = float(c1)
                    best_n = int(best_n)

    # Phase 2: refine at 1024 samples.
    n2 = 1024
    if best_n < 1200 and time.time() < deadline - 120.0:
        h2 = project_feasible(interpolate_h_vector(best_h, n2), n2)
        c2 = compute_c5_exact(h2, n2)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = float(c2)
            best_n = n2

        limit2 = min(deadline, t2)
        if limit2 > time.time() + 20.0:
            h2, c2 = polish_lbfgs(h2, n2, limit2, max_calls=5, maxiter=250)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = float(c2)
                best_n = n2

        if time.time() < limit2 - 10.0:
            h2, c2 = screened_local_search(
                h2, n2, min(limit2, deadline - 5.0),
                max_stagnation=25,
                watch_size=60,
                max_full_evals=55
            )
            if c2 < best_c:
                best_h = h2.copy()
                best_c = float(c2)
                best_n = n2

        if time.time() < limit2 - 5.0:
            h2, c2 = polish_lbfgs(
                h2, n2, limit2,
                max_calls=1,
                alphas=[5000.0],
                mus=[5.0],
                maxiter=100
            )
            if c2 < best_c:
                best_h = h2.copy()
                best_c = float(c2)
                best_n = n2

    # Phase 3: low-discrepancy/parametric restarts at moderate resolution.
    if time.time() < deadline - 80.0:
        limit3 = min(deadline, t3)
        n3 = 512
        starts = generate_parametric_starts(n3)
        if best_n > 0:
            h3_seed = project_feasible(interpolate_h_vector(best_h, n3), n3)
            starts.append(make_threshold_binary(h3_seed, n3, 0.5))

        scored = []
        idx = 0
        for s in starts:
            s = project_feasible(s, n3)
            if s.size != n3:
                continue
            try:
                mfast = compute_c_values(s, n3)[1]
                Cf = float(mfast * 2.0 / float(n3))
            except Exception:
                Cf = 1.0
            if not np.isfinite(Cf):
                Cf = 1.0
            scored.append((Cf, idx, s.copy()))
            idx += 1

        if scored:
            scored.sort()
            n_try = min(3, len(scored))
            remaining = max(1.0, limit3 - time.time())
            per = remaining / float(n_try)
            for rank in range(n_try):
                if time.time() >= limit3 - 12.0:
                    break
                Cstart, _, srank = scored[rank]
                if rank > 0 and Cstart > best_c + 0.10:
                    break
                stage_limit = min(limit3 - 5.0, time.time() + max(20.0, per))
                sr, cr = polish_lbfgs(srank, n3, stage_limit, max_calls=2, maxiter=200,
                                      alphas=[50.0, 500.0], mus=[1.0])
                if cr < best_c:
                    best_h = sr.copy()
                    best_c = float(cr)
                    best_n = n3
                if time.time() < stage_limit - 5.0:
                    sr, cr = screened_local_search(
                        sr, n3, stage_limit,
                        max_stagnation=15,
                        watch_size=50,
                        max_full_evals=35
                    )
                    if cr < best_c:
                        best_h = sr.copy()
                        best_c = float(cr)
                        best_n = n3

    # Phase 4: high-resolution refinement from the best known function.
    if time.time() < deadline - 80.0:
        if best_n < 2048 and budget_s > 700.0:
            n4 = 2048
        elif best_n < 1024:
            n4 = 1024
        else:
            n4 = best_n

        if n4 != best_n:
            h4 = project_feasible(interpolate_h_vector(best_h, n4), n4)
            c4 = compute_c5_exact(h4, n4)
            if c4 < best_c:
                best_h = h4.copy()
                best_c = float(c4)
                best_n = n4
        else:
            h4 = best_h.copy()

        limit4 = min(deadline, t4)
        if limit4 > time.time() + 20.0:
            calls4 = 3 if n4 <= 1024 else 2
            hi_iter = 200 if n4 <= 1024 else 120
            h4, c4 = polish_lbfgs(h4, n4, limit4, max_calls=calls4, maxiter=hi_iter)
            if c4 < best_c:
                best_h = h4.copy()
                best_c = float(c4)
                best_n = int(n4)

        if time.time() < limit4 - 10.0:
            h4, c4 = screened_local_search(
                h4, int(n4), min(limit4, deadline - 5.0),
                max_stagnation=20,
                watch_size=40 if int(n4) > 1200 else 60,
                max_full_evals=24 if int(n4) > 1200 else 50
            )
            if c4 < best_c:
                best_h = h4.copy()
                best_c = float(c4)
                best_n = int(n4)

    # Final polishing sequence on the best retained resolution.
    if time.time() < deadline - 30.0:
        hfin, cfin = screened_local_search(
            best_h, int(best_n), deadline - 30.0,
            max_stagnation=25,
            watch_size=60 if int(best_n) <= 1024 else 40,
            max_full_evals=50 if int(best_n) <= 1024 else 24
        )
        if cfin < best_c:
            best_h = hfin.copy()
            best_c = float(cfin)
            best_n = int(best_n)

    if time.time() < deadline - 15.0:
        hfin, cfin = polish_lbfgs(
            best_h, int(best_n), deadline - 5.0,
            max_calls=1,
            alphas=[2000.0],
            mus=[5.0],
            maxiter=80 if int(best_n) > 1024 else 150
        )
        if cfin < best_c:
            best_h = hfin.copy()
            best_c = float(cfin)
            best_n = int(best_n)

    if time.time() < deadline - 5.0:
        hfin, cfin = screened_local_search(
            best_h, int(best_n), deadline - 2.0,
            max_stagnation=12,
            watch_size=60 if int(best_n) <= 1024 else 35,
            max_full_evals=30 if int(best_n) <= 1024 else 16
        )
        if cfin < best_c:
            best_h = hfin.copy()
            best_c = float(cfin)
            best_n = int(best_n)

    # Final exact evaluation and normalization.
    best_n = int(len(best_h))
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5_exact(best_h, best_n)

    if not np.isfinite(best_c):
        best_h = project_feasible(np.full(500, 0.5, dtype=float), 500)
        best_n = 500
        best_c = compute_c5_exact(best_h, best_n)

    return best_h, float(best_c), int(best_n)
