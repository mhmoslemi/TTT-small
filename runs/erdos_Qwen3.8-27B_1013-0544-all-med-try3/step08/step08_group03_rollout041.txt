    """Compute d(c_k)/dh where c_k is the correlation at shift k"""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[:nv] = (1.0 - h[k:])
        if k > 0:
            grad[k:] -= h[:nv]
    else:
        kk = -k
        nv = n - kk
        grad[kk:] = (1.0 - h[:nv])
        if kk > 0:
            grad[:nv] -= h[kk:]
    return grad * dx

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

def smooth_max_jac(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()

    for idx in range(2*n-1):
        if w[idx] < 1e-12:
            continue
        k = idx - (n-1)
        sg = compute_shift_grad(x, n, k)
        grad += w[idx] * sg
    return grad

        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        total_shifts = 2 * n - 1
        if top_k >= total_shifts:
            top_idx = np.arange(total_shifts)
            top_vals = c
        else:
            top_idx = np.argsort(c)[-top_k:]
            top_vals = c[top_idx]
        
        max_val = top_vals.max()
        alpha_w = 200.0
        w = np.exp(alpha_w * (top_vals - max_val))
        wsum = w.sum()
        if wsum < 1e-30:
            break
        w /= wsum
        
        grad = np.zeros(n)
        for idx in range(len(top_idx)):
            if w[idx] < 1e-10:
                continue
            k = top_idx[idx] - (n - 1)
            grad += w[idx] * compute_shift_grad(h, n, k)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        step = lr
        improved = False
        for _ in range(50):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                no_improve = 0
                break
            step *= 0.5
        
        if not improved:
            no_improve += 1
            lr *= 0.6
            top_k = min(int(top_k * 1.5) + 1, total_shifts)
            if no_improve > 150 or lr < 1e-12:
                break
    
    return best_h, best_c

def gradient_guided_pairs(h, n, deadline, d=0.005, top_k=50, n_try=30):
    best_c = compute_c5(h, n)
    for pass_num in range(200):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        total_shifts = 2 * n - 1
        tk = min(top_k, total_shifts)
        top_idx = np.argsort(c)[-tk:]
        top_vals = c[top_idx]
        max_val = top_vals.max()
        w = np.exp(200.0 * (top_vals - max_val))
        wsum = w.sum()
        if wsum < 1e-30:
            break
        w /= wsum
        
        grad = np.zeros(n)
        for idx in range(tk):
            if w[idx] < 1e-10:
                continue
            k = top_idx[idx] - (n - 1)
            grad += w[idx] * compute_shift_grad(h, n, k)
        
        order = np.argsort(grad)
        improved = False
        nt = min(n_try, n)
        for ri in range(nt):
            if time.time() > deadline:
                return h, best_c
            for rj in range(nt):
                i = order[n - 1 - ri]
                j = order[rj]
                if i == j:
                    continue
                if h[i] - d >= -1e-15 and h[j] + d <= 1 + 1e-15:
                    old_i, old_j = h[i], h[j]
                    h[i] = max(0, old_i - d)
                    h[j] = min(1, old_j + d)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] = old_i
                        h[j] = old_j
        if not improved:
            break
    return h, best_c

def local_pair_search(h, n, deadline, deltas=None, radius=15):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] - d >= -1e-15 and h[j] + d <= 1 + 1e-15:
                        old_i, old_j = h[i], h[j]
                        h[i] = max(0, old_i - d)
                        h[j] = min(1, old_j + d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c
                            improved = True
                        else:
                            h[i] = old_i
                            h[j] = old_j
            if not improved:
                break
    return h, best_c

def interpolate_to_resolution(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def optimize_at_resolution(h, n, deadline):
    remaining = deadline - time.time()
    if remaining < 10:
        return h, compute_c5(h, n)
    
    c0 = compute_c5(h, n)
    
    # Phase A: L-BFGS-B
    t_a = time.time() + remaining * 0.35
    h, c = lbfgs_optimize(h, n, t_a, max_iters=150)
    c = compute_c5(h, n)
    
    # Phase B: Subgradient
    t_b = time.time() + (deadline - time.time()) * 0.35
    if time.time() < t_b:
        h, c = subgradient_descent(h, n, t_b, top_k_init=20, lr_init=0.03)
    
    # Phase C: Gradient-guided pairs
    t_c = time.time() + (deadline - time.time()) * 0.2
    if time.time() < t_c and deadline - time.time() > 10:
        h, c = gradient_guided_pairs(h, n, t_c, d=0.005, top_k=50, n_try=25)
    
    # Phase D: Local pair search
    if time.time() < deadline - 3:
        h, c = local_pair_search(h, n, deadline - 2,
                                 deltas=[0.002, 0.001, 0.0005],
                                 radius=min(15, n // 10))
    
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    
    # Get initial solution from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    
    # Phase 1: Quick polish at original resolution
    if time.time() < deadline - 400:
        t1 = min(time.time() + 80, deadline - 400)
        h1, c1 = lbfgs_optimize(best_h, best_n, t1, max_iters=100)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Try multiple higher resolutions
    # Allocate time based on remaining budget
    resolutions = []
    remaining_for_res = deadline - time.time()
    
    # Choose resolutions adaptively
    if best_n <= 300:
        resolutions = [400, 500, 650, 850]
    elif best_n <= 500:
        resolutions = [600, 800, 1000]
    else:
        resolutions = [best_n + 200, best_n + 500]
    
    for n_try in resolutions:
        if time.time() > deadline - 40:
            break
        remaining = deadline - time.time()
        if remaining < 40:
            break
        
        # Interpolate from best solution
        h_try = interpolate_to_resolution(best_h, best_n, n_try)
        c_try = compute_c5(h_try, n_try)
        
        # Also try a perturbed version for diversity
        h_pert = h_try + rng.randn(n_try) * 0.02
        h_pert = project_feasible(h_pert, n_try)
        c_pert = compute_c5(h_pert, n_try)
        
        # Optimize the better starting point
        if c_try <= c_pert:
            h_opt = h_try
        else:
            h_opt = h_pert
        
        # Time allocation
        alloc = min(remaining * 0.3, 250)
        h_try, c_try = optimize_at_resolution(h_opt, n_try, time.time() + alloc)
        
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
            best_n = n_try
    
    # Phase 3: Try a different initialization strategy - "spread" initialization
    if time.time() < deadline - 80:
        n_spread = max(best_n, 500)
        # Initialize with a structured pattern
        x = np.linspace(0, 2, n_spread)
        # Use a "comb" pattern that spreads mass evenly
        h_spread = np.ones(n_spread) * 0.5
        # Add sinusoidal perturbation
        h_spread += 0.15 * np.sin(2 * np.pi * 3 * x / 2.0)
        h_spread += 0.08 * np.sin(2 * np.pi * 7 * x / 2.0)
        h_spread = project_feasible(h_spread, n_spread)
        c_spread = compute_c5(h_spread, n_spread)
        
        remaining = deadline - time.time()
        if remaining > 30:
            alloc = min(remaining * 0.3, 150)
            h_spread, c_spread = optimize_at_resolution(h_spread, n_spread, time.time() + alloc)
        
        if c_spread < best_c:
            best_c = c_spread
            best_h = h_spread
            best_n = n_spread
    
    # Phase 4: Final intensive optimization on best
    if time.time() < deadline - 5:
        remaining = deadline - time.time()
        if remaining > 30:
            best_h, best_c = optimize_at_resolution(best_h, best_n, deadline - 3)
    
    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
