from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using iterative water-filling."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, lam, nu):
    """Smooth max of correlations + augmented Lagrangian for sum constraint."""
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
    obj = m + logsum / alpha + lam * (h.sum() - n/2) + nu * 0.5 * (h.sum() - n/2)**2
    return float(obj)

def smooth_max_grad(h, n, alpha, lam, nu):
    """Gradient of smooth max + augmented Lagrangian."""
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
        w /= w.sum()
    else:
        w = np.exp(shifted)
        w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    # Add constraint gradient
    s = h.sum() - n/2
    grad += (lam + nu * s) * np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max_and_weights(c, alpha, k_top=None):
    """Compute smooth max and weights, optionally restricted to top-k values."""
    if k_top is not None and k_top < c.size:
        idx = np.argpartition(c, -k_top)[-k_top:]
        c_work = c[idx]
        use_subset = True
    else:
        c_work = c
        use_subset = False
    m = c_work.max()
    shifted = alpha * (c_work - m)
    mx = shifted.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
        w_work = np.exp(shifted - mx)
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
        w_work = np.exp(shifted)
    w_work = w_work / w_work.sum()
    smooth_val = m + logsum / alpha
    if use_subset:
        full_w = np.zeros(c.size)
        full_w[idx] = w_work
        return smooth_val, full_w
    return smooth_val, w_work

def _obj_aug_lag(h, n, alpha, lam, nu, k_top):
    c = compute_c5_vec(h, n)
    smooth_val, _ = _smooth_max_and_weights(c, alpha, k_top)
    s = h.sum() - n / 2.0
    return smooth_val + lam * s + 0.5 * nu * s * s

def _grad_aug_lag(h, n, alpha, lam, nu, k_top):
    c = compute_c5_vec(h, n)
    _, w = _smooth_max_and_weights(c, alpha, k_top)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    s = h.sum() - n / 2.0
    grad += (lam + nu * s) * np.ones(n)
    return grad

def _slsqp_obj(h, n, alpha, k_top):
    c = compute_c5_vec(h, n)
    smooth_val, _ = _smooth_max_and_weights(c, alpha, k_top)
    return smooth_val

def _slsqp_eq(h, n):
    return h.sum() - n / 2.0

def _slsqp_eq_jac(h, n):
    return np.ones(n)

def augmented_lag_lbgfs(h, n, deadline, alpha_schedule, k_top=5, max_outer=40):
    """Augmented Lagrangian with L-BFGS-B inner loop."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lam = 0.0
    nu = 1.0
    bounds = [(0.0, 1.0)] * n
    
    for outer in range(max_outer):
        if time.time() > deadline - 3:
            break
        alpha = alpha_schedule[min(outer, len(alpha_schedule) - 1)]
        t_inner = min(deadline - 3, time.time() + 30)
        
        try:
            res = minimize(_obj_aug_lag, h, jac=_grad_aug_lag,
                          args=(n, alpha, lam, nu, k_top), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 2000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
        except Exception:
            break
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        
        # Update Lagrange multiplier
        s = h.sum() - n / 2.0
        lam = lam + nu * s
        if abs(s) > 0.01:
            nu = min(nu * 2.0, 10000.0)
        elif abs(s) < 1e-6:
            nu = max(nu * 0.5, 0.1)
    
    return best_h, best_c

def slsqp_refine(h, n, deadline, alpha, k_top):
    """SLSQP refinement for a fixed alpha."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (n,)}]
    best_c = compute_c5(project_feasible(h, n), n)
    h = project_feasible(h, n)
    t_rem = max(1, deadline - time.time() - 2)
    maxiter = max(50, int(t_rem * 5))
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha, k_top), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def gradient_descent_adapt(h, n, deadline, alpha_init=50.0, lr_init=0.005):
    """Adaptive gradient descent with backtracking line search."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    alpha = alpha_init
    stuck_count = 0
    
    for it in range(50000):
        if time.time() > deadline:
            break
        if (it + 1) % 10 == 0:
            alpha *= 1.3
            if stuck_count > 20:
                lr *= 0.7
                stuck_count = 0
                if lr < 1e-10:
                    break
        
        g = _grad_aug_lag(h, n, alpha, 0.0, 0.0, 3)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_h = h.copy()
                best_c = c_new
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            stuck_count += 1
        else:
            stuck_count = max(0, stuck_count - 1)
    
    return best_h, best_c

def pair_search_gradient(h, n, deadline, step_sizes, radius=None):
    """Pair search guided by gradient differences."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    for ss in step_sizes:
        improved = False
        while time.time() < deadline and improved:
            improved = False
            c_vec = compute_c5_vec(h, n)
            m = c_vec.max()
            w = np.exp(2000 * (c_vec - m))
            w_sum = w.sum()
            if w_sum < 1e-14:
                break
            w = w / w_sum
            dx = 2.0 / n
            g = np.zeros(n)
            for m_idx in range(2 * n - 1):
                if w[m_idx] < 1e-12:
                    continue
                k = m_idx - (n - 1)
                wm = w[m_idx] * dx
                if k >= 0:
                    nv = n - k
                    g[0:nv] += wm * (1.0 - h[k:n])
                    g[k:n] += wm * (-h[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += wm * (1.0 - h[0:nv])
                    g[0:nv] += wm * (-h[kk:n])
            
            # Sort by gradient magnitude to focus on high-impact moves
            order = np.argsort(np.abs(g))[::-1]
            count = 0
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                count += 1
                if count > 80:
                    break
                if radius is not None:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                else:
                    lo, hi = 0, n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    move = -ss * (g[i] - g[j])
                    if abs(move) < 1e-10:
                        continue
                    if h[i] + move >= 0.0 and h[i] + move <= 1.0:
                        if h[j] - move >= 0.0 and h[j] - move <= 1.0:
                            h[i] += move
                            h[j] -= move
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= move
                                h[j] += move
    return h, best_c

def pair_search_simple(h, n, deadline, deltas, radius):
    """Simple pair search with fixed deltas."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = False
        while time.time() < deadline and improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] += (-d) * 0 + d  # undo
                            h[i] += d
                            h[j] -= d
    return h, best_c

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old

    h_new = h_new + (2.0 / n_new - 2.0 / n_old) * 0  # no shift needed
    return project_feasible(h_new, n_new)
