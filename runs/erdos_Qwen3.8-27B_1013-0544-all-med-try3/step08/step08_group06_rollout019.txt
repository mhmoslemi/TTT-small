        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    size = n
    if radius is None:
        radius = size
    for d in deltas:
        for _pass in range(300):
            if time.time() > deadline:
                break
            improved = False
            for i in range(size):
                if time.time() > deadline:
                    return best_h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(size, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        oi, oj = best_h[i], best_h[j]
                        best_h[i] = oi + d
                        best_h[j] = oj - d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = oi
                            best_h[j] = oj
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        oi, oj = best_h[i], best_h[j]
                        best_h[i] = oi - d
                        best_h[j] = oj + d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = oi
                            best_h[j] = oj
            if not improved:
                break
    return best_h, best_c

def subgradient_polish(h, n, deadline, max_steps=2000, lr_init=0.02):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = c.max()
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()
        top_k = max(3, int(8 * np.exp(-step / 300.0)))
        c_sorted_idx = np.argsort(c)[-top_k:]
        w = np.zeros(2 * n - 1)
        for idx in c_sorted_idx:
            w[idx] = c[idx]
        wsum = w.sum()
        if wsum < 1e-20:
            break
        w /= wsum
        grad = np.zeros(n)
        for k_idx in c_sorted_idx:
            k = k_idx - (n - 1)
            wi = w[k_idx]
            if k >= 0:
                if k == 0:
                    grad += wi * (1.0 - 2.0 * h)
                else:
                    grad[:n - k] += wi * (1.0 - h[k:])
                    grad[k:] -= wi * h[:n - k]
            else:
                kp = -k
                if kp == 0:
                    grad += wi * (1.0 - 2.0 * h)
                else:
                    grad[kp:] += wi * (1.0 - h[:-kp])
                    grad[:-kp] -= wi * h[kp:]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h = h - lr * grad / gnorm
        h = project_feasible(h, n)
        if step % 50 == 0:
            lr *= 0.99
            if lr < 1e-7:
                lr = 1e-7
    return best_h, best_c

def optimize_one(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        alloc_check = time.time()
        ht = slsqp_optimize(best_h, n, a, maxiter=2000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    # Subgradient polish
    if time.time() < deadline - 15:
        ht, c = subgradient_polish(best_h, n, time.time() + min(30, (deadline - time.time()) - 15))
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    # SLSQP again with very high alpha
    if time.time() < deadline - 10:
        ht = slsqp_optimize(best_h, n, 10000000, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    # Local search
    if n <= 250:
        if time.time() < deadline - 5:
            best_h, best_c = local_search(best_h, n, deadline - 3,
                                          [0.01, 0.005, 0.002, 0.001])
    else:
        if time.time() < deadline - 5:
            rad = max(30, n // 4)
            best_h, best_c = local_search(best_h, n, deadline - 3,
                                          [0.01, 0.005, 0.002], radius=rad)

    # Final SLSQP
    if time.time() < deadline - 3:
        ht = slsqp_optimize(best_h, n, 10000000, maxiter=2000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    if n <= 250 and time.time() < deadline - 1:
        best_h, best_c = local_search(best_h, n, deadline - 0.5,
                                      [0.002, 0.001, 0.0005])
    elif time.time() < deadline - 1:
        rad = max(20, n // 5)
        best_h, best_c = local_search(best_h, n, deadline - 0.5,
                                      [0.002, 0.001], radius=rad)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    x_init = np.linspace(0, 2, n_init) if h_init is not None else np.linspace(0, 2, 200)

    # Try multiple resolutions
    resolutions = [n_init]
    for nr in [100, 150, 200, 250, 300, 400, 500]:
        if nr != n_init and nr not in resolutions:
            resolutions.append(nr)
    resolutions.sort()

    # Allocate time: give more to resolutions near the parent
    total_alloc = deadline - t0 - 20
    weights = {}
    for nr in resolutions:
        if h_init is not None:
            weights[nr] = max(0.5, 2.0 - abs(nr - n_init) / 200.0)
        else:
            weights[nr] = 1.0

    # Process resolutions
    processed = []
    for nr in resolutions:
        remaining = deadline - time.time() - 20
        if remaining < 30:
            break
        w = weights.get(nr, 1.0)
        total_w = sum(weights.get(r, 1.0) for r in resolutions if r not in processed for _ in [1])
        frac = w / total_w
        alloc = min(remaining * 0.85, total_alloc * frac)
        alloc = max(alloc, 25)
        alloc = min(alloc, remaining - 15)
        if alloc < 20:
            continue

        if h_init is not None:
            x_new = np.linspace(0, 2, nr)
            h_new = np.interp(x_new, x_init, h_init)
        else:
            x_new = np.linspace(0, 2, nr)
            h_new = np.full(nr, 0.5)
        h_new = project_feasible(h_new, nr)

        # Add small random perturbation for diversity
        if len(processed) > 0:
            h_new = h_new + rng.normal(0, 0.01, nr)
            h_new = project_feasible(h_new, nr)

        t_start = time.time()
        h_opt, c_opt = optimize_one(h_new, nr, t_start + alloc, rng)
        processed.append(nr)

        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            x_init = x_new
            n_init = nr
            h_init = best_h.copy()

    # Random restarts from best solution
    if best_h is not None and time.time() < deadline - 30:
        n_best = len(best_h)
        remaining = deadline - time.time() - 15
        n_restarts = max(2, int(remaining / 40))
        n_restarts = min(n_restarts, 5)
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            scale = 0.03 + 0.1 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(35, (deadline - time.time() - 12) * 0.35)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_one(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    # Final polish on best solution
    if best_h is not None and time.time() < deadline - 3:
        n_best = len(best_h)
        ht = slsqp_optimize(best_h, n_best, 50000000, maxiter=2000, ftol=1e-17)
        c = compute_c5(ht, n_best)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht.copy()

        if n_best <= 250 and time.time() < deadline - 1:
            best_h, best_c = local_search(best_h, n_best, deadline - 0.5,
                                          [0.001, 0.0005, 0.0002])
        elif time.time() < deadline - 1:
            rad = max(20, n_best // 5)
            best_h, best_c = local_search(best_h, n_best, deadline - 0.5,
                                          [0.001, 0.0005], radius=rad)

    # Final feasibility check
    if best_h is not None:
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))

    if best_h is None:
        n = 200
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
