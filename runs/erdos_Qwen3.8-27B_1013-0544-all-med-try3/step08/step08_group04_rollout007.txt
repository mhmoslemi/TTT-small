def swap_update(c, h, i, j, d, n):
    """
    Compute new correlation after h[i] += d, h[j] -= d.
    c is the current full correlation (length 2n-1).
    Returns new c (length 2n-1).
    """
    offset = n - 1
    c_new = c.copy()
    
    # For each lag ℓ (0 to 2n-2):
    # Let a = i + ℓ - offset (the g-index for the h-change at i)
    # Let b = j + ℓ - offset (the g-index for the h-change at j)
    
    # ℓ ranges 0 to 2n-2
    # a = i + ℓ - (n-1), valid when 0 <= a < n, i.e., (n-1-i) <= ℓ <= (n-1-i+n-1) = 2(n-1)-i
    # b = j + ℓ - (n-1), valid when 0 <= b < n, i.e., (n-1-j) <= ℓ <= 2(n-1)-j
    
    # For valid a: c_new[ℓ] += d*(1 - 2*h[a])
    # For valid b: c_new[ℓ] += d*(2*h[b] - 1)
    
    a_lo = max(0, n - 1 - i)
    a_hi = min(2*n - 2, 2*(n - 1) - i)
    for ell in range(a_lo, a_hi + 1):
        a_idx = i + ell - offset
        c_new[ell] += d * (1.0 - 2.0 * h[a_idx])
    
    b_lo = max(0, n - 1 - j)
    b_hi = min(2*n - 2, 2*(n - 1) - j)
    for ell in range(b_lo, b_hi + 1):
        b_idx = j + ell - offset
        c_new[ell] += d * (2.0 * h[b_idx] - 1.0)
    
    return c_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def optimize_slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new, n) < compute_c5(h, n):
            return h_new
    except Exception:
        pass
    return h

def optimize_lbfgsb(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    target = n / 2.0
    
    def penalized(x):
        return smooth_obj(x, n, alpha) + 1e5 * (x.sum() - target)**2
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalized, h, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-13, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new, n) < compute_c5(h, n):
            return h_new
    except Exception:
        pass
    return h

def pair_search_fast(h, n, deadline, deltas):
    """Full pair search with early termination."""
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                        else:

                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
