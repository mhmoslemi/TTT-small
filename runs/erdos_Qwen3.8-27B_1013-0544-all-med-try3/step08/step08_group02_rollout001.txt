def vec_grad(h, n, weights):

    # Let shift = k - (n-1) for the actual lag

            kk = -k

            grad[kk:n] += w * (1.0 - h[0:nv])
            grad[0:nv] += w * (-h[kk:n])
    return grad

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            frac = diff / tot
            h += frac * room
            np.clip(h, 0, 1, out=h)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            frac = diff / tot
            h += frac * h
            np.clip(h, 0, 1, out=h)
    return np.clip(h, 0.0, 1.0)

# Objective: smooth max using log-sum-exp
def lse_obj(h, n, alpha, active_mask=None, mu=0.0):
    c = compute_c5_vec(h, n)
    if active_mask is not None:
        c = c[active_mask]
    m = c.max()
    shifted = alpha * (c - m)
    s = shifted.max()
    val = m + (s + np.log(np.sum(np.exp(shifted - s)))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n/2.0)**2
    return float(val)

def lse_grad(h, n, alpha, active_mask=None, mu=0.0):
    c = compute_c5_vec(h, n)
    if active_mask is not None:
        c_full = c
        c = c[active_mask]
    else:
        c_full = c
    m = c.max()
    shifted = alpha * (c - m)
    s = shifted.max()
    w = np.exp(shifted - s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    
    if active_mask is not None:
        for wi, idx in zip(w, active_mask):
            if wi < 1e-14:
                continue
            k = idx - (n-1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    else:
        for idx in range(2*n-1):
            if w[idx] < 1e-14:

            k = idx - (n-1)
            wm = w[idx] * dx
            if k >= 0:
                nv = n - k

                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n/2.0)
    return grad

def coordinate_descent(h, n, deadline, n_passes=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0/n
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):  # stride 2 for speed
                if time.time() > deadline:
                    break
                for j in range(max(0,i-100), min(n, i+101)):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if best_h[j] >= step and best_h[i] + step <= 1.0:
                        best_h[i] += step
                        best_h[j] -= step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= step
                            best_h[j] += step
                    # Try moving mass from i to j
                    if best_h[i] >= step and best_h[j] + step <= 1.0:
                        best_h[i] -= step
                        best_h[j] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += step
                            best_h[j] -= step
    return best_h, best_c
