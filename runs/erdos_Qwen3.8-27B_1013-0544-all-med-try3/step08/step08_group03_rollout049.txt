At k=2 (the full period), c_2 = ∫h(x)(1-h(x+2))dx. But since h is defined on [0,2], what does h(x+2) mean? In the discrete case, the correlation wraps around... actually no, the 'full' mode doesn't wrap. The shift k=2 in discrete corresponds to index 2n which doesn't exist. The valid shifts are from -(n-1) to (n-1) in discrete, or -(2-dx) to (2-dx) in continuous.

So at k=2-dx (shift by almost the full period), we're comparing h(x) with h(x+2-dx), which is just h at a slightly different position. This is similar to k=0.

The most constraining shifts are likely:
- k=0: c_0 = 1 - ∫h²
- k=±small: compares adjacent regions
- k=±large (near ±2): also compares regions that are far apart

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    c_max = c.max()
    sh = alpha * (c - c_max)
    sh_max = sh.max()
    return c_max + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha

def smooth_max_weights(c, alpha):
    c_max = c.max()
    sh = alpha * (c - c_max)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w /= w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_fn(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    val = smooth_max_val(c, alpha)
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def grad_fn(h, n, alpha, mu):
    g = grad_smooth_max(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def obj_slsqp_fn(h, n, alpha):
    c = compute_c5_vec(h, n)
    return float(smooth_max_val(c, alpha))

def constraint_eq(h, n):
    return float(h.sum() - n / 2.0)

def constraint_jac(h, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [100, 1000, 10000, 50000, 200000, 1000000, 5000000]
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    h = project_feasible(h, n)
    mus = [1000.0, 100.0, 10.0, 1.0, 0.1, 0.01]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        for mu in mus:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(obj_fn, h, jac=grad_fn,
                             args=(n, alpha, mu), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 300, 'ftol': 1e-18,
                                      'gtol': 1e-15, 'maxfun': 8000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_jac, 'args': (n,)}]
    try:
        res = minimize(obj_slsqp_fn, h, args=(n, alpha), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-17, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def binary_push(h, n, strength=0.5):
    h_new = h.copy()
    for i in range(n):
        if h_new[i] > 0.5:
            h_new[i] = h_new[i] + strength * (1.0 - h_new[i])
        else:
            h_new[i] = h_new[i] - strength * h_new[i]
    return project_feasible(h_new, n)

def v_binary_push(h, n, v=0.4):
    """Push toward h*(1-h) reduction: nudge toward 0 or 1 based on 0.5 threshold with variance."""
    h_new = h.copy()
    thresh = 0.5
    for i in range(n):
        d = h_new[i] - thresh
        sign = 1.0 if d >= 0 else -1.0
        h_new[i] += v * sign * (0.5 - abs(d))
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n)

def interpolate_to_resolution(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] = min(1.0, h[i] + d)
                        h[j] = max(0.0, h[j] - d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                        else:
                            continue
                    elif h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] = max(0.0, h[i] - d)
                        h[j] = min(1.0, h[j] + d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                        else:
                            continue
    return h, best_c

def block_search(h, n, deadline, deltas, block_sizes, radius=50):
    """Try moving blocks of consecutive points."""
    best_c = compute_c5(h, n)
    for bs in block_sizes:
        for d in deltas:
            any_improved = True
            while any_improved and time.time() < deadline:
                any_improved = False
                for i in range(0, n - bs + 1):
                    if time.time() > deadline:
                        return h, best_c
                    lo = max(0, i - radius)
                    hi = min(n - bs + 1, i + radius + 1)
                    for j in range(lo, hi):
                        if j <= i + bs - 1 and j + bs - 1 >= i:
                            continue
                        if np.all(h[i:i+bs] + d <= 1.0) and np.all(h[j:j+bs] - d >= 0.0):
                            h[i:i+bs] += d
                            h[j:j+bs] -= d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                any_improved = True
                            else:
                                h[i:i+bs] -= d
                                h[j:j+bs] += d
    return h, best_c

def gradient_line_search(h, n, deadline, alpha=50000, max_steps=500):
    """Gradient descent with backtracking line search."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.05
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        improved = False
        for _ in range(50):
            h_try = project_feasible(h - lr * g, n)
            c_try = compute_c5(h_try, n)
            if c_try <= best_c - 1e-15:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            alpha = min(alpha * 1.3, 10000000)
            if lr < 1e-14:
                lr = 0.01
                alpha = 50000
    return best_h, best_c

def anneal_local_search(h, n, deadline, n_iter=2000, temp_init=0.002, temp_min=1e-6):
    """Simulated annealing on pair moves."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    temp = temp_init
    cool_rate = max(temp_min / temp_init, (temp_min / temp_init)) ** (1.0 / max(n_iter, 1))
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        temp *= cool_rate
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        d = np.random.choice([-1, 1]) * np.random.uniform(0.001, 0.02)
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i] = hi
        h[j] = hj
        c_new = compute_c5(h, n)
        delta = c_new - cur_c
        if delta < 0 or np.random.rand() < np.exp(-delta / max(temp, 1e-10)):
            cur_c = c_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] = hi - d
            h[j] = hj + d
    return best_h, best_c
