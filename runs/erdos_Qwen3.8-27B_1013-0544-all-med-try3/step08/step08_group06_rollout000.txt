```python
                    grad_h[lag:] -= p_k * h[:n-lag] * dx

            h_new = project_feasible(h_new, n)

            if val < best_val - 1e-12:
                best_val = val
                best_step = step
        
        if best_step == 1e-5:
            break # no improvement
            
        h = np.clip(h - best_step * grad_h, 0, 1)
        h = project_feasible(h, n)
        best_c = best_val
    return h, best_c
```

```python
def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    # We want to add a scalar to h, or scale the non-bound parts
    # A good way: add diff / count_free to the free parts
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-6) & (h < 1 - 1e-6)
        if not np.any(free):
            # if no free vars, we have to break bounds slightly or scale everything
            if diff > 0:
                room = 1.0 - h
                tot = room.sum()
                if tot > 1e-12: h += (diff / tot) * room
            else:
                tot = h.sum()
                if tot > 1e-12: h += (diff / tot) * h
            break
        h[free] += diff / np.sum(free)
    return np.clip(h, 0.0, 1.0)
```
Wait, if we just add `diff / np.sum(free)` to the free variables, they might cross the boundaries.
The cluster algorithm for projection onto the simplex (with bounds) is $O(n)$ and perfectly robust.
But a simple loop that adds `diff / count` and then re-clips will converge quickly.
Let's write a robust $O(n)$ projection:
To project $v$ onto $\{h : 0 \le h_i \le 1, \sum h_i = n/2\}$:
We want to find $\tau$ such that $\sum \text{clip}(v_i - \tau, 0, 1) = n/2$.
We can find $\tau$ by binary search! Since the function is monotonic in $\tau$, binary search takes $O(n \log \text{precision})$ operations, which is extremely fast in NumPy.
```python
def project_feasible_fast(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi)/2.0, 0.0, 1.0)
```
This is PERFECT. It's exact, $O(n)$ per iteration, 100 iterations is $100 \times O(n)$ which is microseconds for $n=1000$.
No loops in Python, just NumPy vectorized operations. This will make the local search incredibly fast.

```python
def continuous_step(h, n, deadline):

    best_h = h.copy()

    while time.time() < deadline:
        c = np.correlate(h, 1-h, mode='full') * (2.0 / n)
        m = c.max()

        p = np.exp(500 * (c - m))
        p /= np.sum(p)
        
        grad_h = np.zeros(n)
        dx = 2.0 / n
        active_idx = np.where(p > 1e-6)[0]
        for idx in active_idx:
            lag = idx - (n - 1)
            p_k = p[idx]
            if lag == 0:
                grad_h += p_k * (1 - 2 * h) * dx
            else:
                if lag > 0:
                    grad_h[lag:] += p_k * (1 - h[lag:]) * dx
                    grad_h[:n-lag] -= p_k * h[lag:] * dx
                else:
                    lag = -lag
                    grad_h[:n-lag] += p_k * (1 - h[:n-lag]) * dx
                    grad_h[lag:] -= p_k * h[:n-lag] * dx
                    
        grad_h -= np.mean(grad_h)
        
        found_improvement = False
        for _ in range(20):
            h_new = np.clip(h - step * grad_h, 0, 1)
            h_new = project_feasible_fast(h_new, n)
            val = compute_c5(h_new, n)
            if val < best_c - 1e-12:
                best_c = val
                best_h = h_new
                found_improvement = True
                break
            step /= 2.0
            
        if not found_improvement:
            # reset step size and try a different perturbation or break
            step = 1e-2
            # maybe break if we've converged
            # to avoid infinite loops, we can just break or do a random kick
            if np.linalg.norm(grad_h) < 1e-6:
                break
            # do a small random perturbation to escape
            h = best_h + np.random.normal(0, 1e-4, n)
            h = project_feasible_fast(h, n)
        else:
            h = best_h
            step = min(step * 1.5, 1e-1)
            
    return best_h, best_c
```

```python
            h_new = project_feasible_fast(h_new, n)
            val = compute_c5(h_new, n)
            if val < best_c - 1e-12:
                best_c = val
                best_h = h_new
                found_improvement = True
                break
            step /= 2.0
            
        if not found_improvement:
            step = 1e-2
            if np.linalg.norm(grad_h) < 1e-8:
                break
            # Small random kick to escape plateaus
            h = best_h + np.random.normal(0, 1e-5, n)
            h = project_feasible_fast(h, n)
        else:
            h = best_h
            step = min(step * 1.5, 1e-1)
            
    return best_h, best_c

def optimize_h(h_init, n, deadline):
    h = project_feasible_fast(h_init, n)
    x = h[:n-1].copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    alphas = [10, 50, 200, 1000, 5000]
    for a in alphas:
        if time.time() > deadline - 5:
            break
        
        maxiter = max(200, int(200000 / (n / 100.0)))
        maxiter = min(maxiter, 1500)
        
        try:
            res = scipy.optimize.minimize(
                obj_func, x, 
                args=(n, a), 
                method='L-BFGS-B', 
                jac=jac_func,
                bounds=[(0, 1)] * (n - 1),
                options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-4}
            )
            x = res.x
            h = np.zeros(n)
            h[:n-1] = x
            h[n-1] = n / 2.0 - np.sum(x)
            h = project_feasible_fast(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        except Exception:
            pass
            
    if time.time() < deadline - 3:
        best_h, best_c = continuous_step(best_h, n, deadline - 1)
        
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100
        
    best_h = project_feasible_fast(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    current_h = best_h
    current_n = n_init
    
    res_list = []
    if current_n < 200:
        res_list = [200, 400, 800, 1600, 3200]
    else:
        res_list = [400, 800, 1600, 3200]
        
    if time.time() < deadline - 300 and current_n < 200:
        x = np.linspace(0, 2, 200)
        h_200 = np.interp(x, np.linspace(0, 2, current_n), current_h)
        h_200 = project_feasible_fast(h_200, 200)
        h_200, c_200 = optimize_h(h_200, 200, time.time() + 40)
        if c_200 < best_c:
            best_c = c_200
            best_h = h_200
        current_h = h_200
        current_n = 200
        
    remaining_budget = deadline - time.time()
    allocs = {400: 0.20, 800: 0.30, 1600: 0.40, 3200: 0.10}
    
    for n_next in res_list:
        if n_next <= current_n:
            continue
        alloc = remaining_budget * allocs.get(n_next, 0.1)
        if time.time() + alloc > deadline and alloc < 20:
            break
            
        x = np.linspace(0, 2, n_next)
        x_old = np.linspace(0, 2, current_n)
        h_new = np.interp(x, x_old, current_h)
        h_new = project_feasible_fast(h_new, n_next)
        
        t_start = time.time()
        h_opt, c_opt = optimize_h(h_new, n_next, t_start + alloc)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            current_h = h_opt
            current_n = n_next
        elif c_opt < best_c + 0.001:
            current_h = h_opt
            current_n = n_next
                
    if time.time() < deadline - 5:
        best_h, best_c = continuous_step(best_h, len(best_h), deadline - 2)
        
    return best_h, best_c, len(best_h)
```
