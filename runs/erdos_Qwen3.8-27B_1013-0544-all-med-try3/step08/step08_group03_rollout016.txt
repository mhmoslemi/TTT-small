import numpy as np
from scipy.optimize import linprog, minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def build_lp(n, h0, dx):
    """Build the LP for DDA step.
    Variables: h[0..n-1], t
    Constraints: L_k(h) <= t for all k, 0<=h<=1, sum(h) = n/2
    
    L_k(h) = sum_{valid} (h[i+k]*(1-h[i])) linearized around h0
    = sum_{valid} [h[i+k] - h0[i+k]*h[i] - h[i+k]*h0[i] + h0[i+k]*h0[i]]
    
    Returns: c, A_ub, b_ub, A_eq, b_eq, bounds
    for scipy linprog: min c'x s.t. A_ub x <= b_ub, A_eq x = b_eq
    """
    # Variable vector: [h[0], ..., h[n-1], t]
    nv = n + 1
    
    # Objective: minimize t → c = [0,...,0,1]
    c = np.zeros(nv)
    c[n] = 1.0
    
    # For each shift m = 0, ..., 2n-2 (k = m - (n-1)):
    # L_m(h) - t <= 0
    # L_m(h) = sum_j a_m[j]*h[j] + b_m
    # So: sum_j a_m[j]*h[j] - t <= -b_m
    
    A_rows = []
    b_rows = []
    
    for m in range(2*n - 1):
        k = m - (n - 1)
        a = np.zeros(n)
        b_const = 0.0
        
        if k >= 0:
            nv_ = n - k
            for i in range(nv_):
                # term: h[i+k]*(1-h[i])
                # linearized: h[i+k] - h0[i+k]*h[i] - h[i+k]*h0[i] + h0[i+k]*h0[i]
                a[i+k] += (1.0 - h0[i])
                a[i] += -h0[i+k]
                b_const += h0[i+k]*h0[i]
        else:
            kk = -k
            nv_ = n - kk
            for i in range(kk, n):
                # term: h[i]*(1-h[i-kk])
                # linearized: h[i] - h0[i]*h[i-kk] - h[i]*h0[i-kk] + h0[i]*h0[i-kk]
                a[i] += (1.0 - h0[i-kk])
                a[i-kk] += -h0[i]
                b_const += h0[i]*h0[i-kk]
        
        # L_m(h) - t <= -b_m → sum a[j]*h[j] - t <= -b_const
        row = a.copy()
        row[n] = -1.0  # coefficient of t
        A_rows.append(row)
        b_rows.append(-b_const)

    A_eq[0, 0:n] = 1.0

    # Bounds
    lb = [0.0]*n + [-np.inf]
    ub = [1.0]*n + [np.inf]
    
    return c, A_ub, b_ub, A_eq, b_eq, (lb, ub)

from scipy.optimize import linprog, minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_lp_dda(n, h0, dx):
    """Solve the DDA LP step."""
    nv = n + 1
    c = np.zeros(nv)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((2*n-1, nv))
    b_ub = np.zeros(2*n-1)
    
    for m in range(2*n-1):
        k = m - (n-1)
        a = np.zeros(n)
        if k >= 0:
            e = n - k
            a[k:] += (1.0 - h0[:e])
            a[:e] += -h0[k:]
            bconst = float(np.dot(h0[:e], h0[k:]))
        else:
            kk = -k
            e = n - kk
            a[kk:] += (1.0 - h0[:e])
            a[:e] += -h0[kk:]
            bconst = float(np.dot(h0[kk:], h0[:e]))
        A_ub[m, :n] = a * dx
        A_ub[m, n] = -dx
        b_ub[m] = -bconst * dx
    
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0.0, 1.0)]*n + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], float(res.x[n])
    else:
        # Fallback: use bound constraint approach
        # Find feasible point minimizing t with relaxed constraints
        return h0.copy(), compute_c5(h0, n)

def ddA_optimize(h_init, n, deadline, max_iters=50, tol=1e-8):
    """DDA (Direct Descent Algorithm) for the bilinear problem."""
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_lp, t_lp = solve_lp_dda(n, h, dx)
        h_lp = project_feasible(h_lp, n)
        c_lp = compute_c5(h_lp, n)
        
        # Check convergence
        if np.linalg.norm(h_lp - h) < tol:
            break
        
        # Duality gap check: if t_lp >= best_c, we've found the optimum
        if t_lp >= best_c - 1e-10:
            # Convergence: LP relaxation is tight
            break
        
        h = h_lp
        if c_lp < best_c:
            best_c = c_lp
            best_h = h.copy()
    
    return best_h, best_c

= sum_{j=0}^{n-1-k} h0[j+k]*(1-h0[j])  [letting j=i-k]

import numpy as np
from scipy.optimize import linprog, minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_lp_dda(n, h0, dx):
    nv = n + 1
    c = np.zeros(nv)
    c[n] = 1.0
    n_con = 2 * n - 1
    A_ub = np.zeros((n_con, nv))
    b_ub = np.zeros(n_con)
    for m in range(n_con):
        k = m - (n - 1)
        if k >= 0:
            e = n - k
            grad = np.zeros(n)
            grad[:e] += (1.0 - h0[k:])
            grad[k:] += -h0[:e]
            C_k = float(np.dot(h0[:e], 1.0 - h0[k:]))
        else:
            kk = -k
            e = n - kk
            grad = np.zeros(n)
            grad[kk:] += (1.0 - h0[:e])
            grad[:e] += -h0[kk:]
            C_k = float(np.dot(h0[kk:], 1.0 - h0[:e]))
        A_ub[m, :n] = grad * dx
        A_ub[m, n] = -dx
        b_ub[m] = dx * (float(np.dot(grad, h0)) - C_k)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': max(1.0, 30.0)})
    if res.success:
        return res.x[:n], float(res.x[n])
    return None, None

def dda_optimize(h_init, n, deadline, max_iters=80, tol=1e-9):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        res_h, res_t = solve_lp_dda(n, h, dx)
        if res_h is None:
            break
        h_lp = project_feasible(res_h, n)
        c_lp = compute_c5(h_lp, n)
        step = np.linalg.norm(h_lp - h)
        if step < tol:
            break
        if res_t >= best_c - 1e-10:
            h = h_lp
            best_c = c_lp
            best_h = h.copy()
            break
        h = h_lp
        if c_lp < best_c:
            best_c = c_lp
            best_h = h.copy()
    return best_h, best_c

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            e = n - k
            grad[:e] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:e])
        else:
            kk = -k
            e = n - kk
            grad[kk:] += wm * (1.0 - h[:e])
            grad[:e] += wm * (-h[kk:])
    return grad

def _smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _smooth_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_polish(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 10.0), (500, 50.0), (2000, 200.0), (10000, 1000.0), (100000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        t_alloc = deadline - time.time() - 5
        if t_alloc < 3:
            break
        try:
            res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-13,
                                   'maxfun': max_iters * 8, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_polish(h, n, deadline, alpha=200000, maxiter=1000):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    def _cons_fun(x, nn):
        return float(x.sum() - nn / 2.0)
    def _cons_jac(x, nn):
        return np.ones(nn)
    def _obj(x, nn, a):
        c = compute_c5_vec(x, nn)
        m = c.max()
        sh = a * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a)
        return float(m + np.log(np.sum(np.exp(sh))) / a)
    def _grad(x, nn, a):
        return grad_smooth_max(x, nn, a)
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    try:
        t_end = time.time() + min(deadline - time.time() - 3, 80)
        res = minimize(_obj, h, args=(n, alpha), jac=lambda x: _grad(x, n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            n_iter = 0
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
                n_iter += 1
            if not improved:
                break
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def coordinate_descent(h, n, deadline, max_sweeps=20):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_val = h[i]
            # Compute the effect of changing h[i] on each correlation
            # h[i] appears in shifts where it's part of the overlap
            # Try a few candidate values
            candidates = [old_val]
            if old_val > 0.01:
                candidates.append(0.0)
                candidates.append(old_val * 0.5)
                candidates.append(old_val - 0.05)
                candidates.append(old_val - 0.01)
            if old_val < 0.99:
                candidates.append(1.0)
                candidates.append(old_val + 0.05)
                candidates.append(old_val + 0.01)
                candidates.append(old_val * 1.5)
            candidates.append(0.5)
            best_local = old_val
            best_local_c = best_c
            for cv in candidates:
                cv = max(0.0, min(1.0, cv))
                if abs(cv - old_val) < 1e-10:
                    continue
                # Adjust sum to maintain constraint
                adj = (old_val - cv) / (n - 1) if n > 1 else 0
                h[i] = cv
                for j in range(n):
                    if j != i:
                        h[j] = max(0.0, min(1.0, h[j] - adj))
                c = compute_c5(h, n)
                if c < best_local_c:
                    best_local_c = c
                    best_local = cv
                    best_h_copy = h.copy()
            # Restore or apply
            h[i] = old_val
            if best_local != old_val:
                for j in range(n):
                    if j != i:
                        h[j] = best_h_copy[j]
                h[i] = best_local
                if best_local_c < best_c:
                    best_c = best_local_c
                    improved = True
        if not improved:
            break
    return h, best_c
