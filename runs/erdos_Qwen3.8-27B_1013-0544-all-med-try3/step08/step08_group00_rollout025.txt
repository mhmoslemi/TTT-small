def pair_search(h, n, deadline, max_sweeps=50):
    best_c = compute_c5(h, n)
    dx = 2.0/n
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Get subgradient (average of active gradients)
        active = np.where(c_vec >= max_val - 1e-6)[0]
        subgrad = np.zeros(n)
        for idx in active:
            g = shift_grad_single(h, n, idx)
            subgrad += g
        
        # Find top "increase" and "decrease" candidates
        inc_candidates = np.argsort(-subgrad)[:20]
        dec_candidates = np.argsort(subgrad)[:20]
        
        improved = False
        for i in inc_candidates:
            for j in dec_candidates:
                if i == j:
                    continue
                # Compute optimal delta for pair (i, j)
                delta = optimal_pair_delta(h, n, i, j, c_vec, dx)
                if delta is not None and abs(delta) > 1e-10:
                    h_new = h.copy()
                    h_new[i] += delta
                    h_new[j] -= delta
                    if h_new[i] >= -1e-12 and h_new[i] <= 1+1e-12 and h_new[j] >= -1e-12 and h_new[j] <= 1+1e-12:
                        h_new = np.clip(h_new, 0, 1)
                        c_new = compute_c5(h_new, n)
                        if c_new < best_c - 1e-12:
                            best_c = c_new
                            h = h_new
                            improved = True
        
        if not improved:
            break
    
    return h, best_c

def optimal_pair_delta(h, n, i, j, c_vec, dx):
    # For pair (i, j), compute g_k for all k
    # Δf_k = δ * dx * g_k
    # Want to minimize max_k (c_vec[k] + δ*dx*g_k)
    # Subject to: -h[i] ≤ δ ≤ 1-h[j]
    
    g = np.zeros(2*n-1)  # indexed 0 to 2n-2, corresponding to k = -(n-1) to n-1
    
    k_offset = n - 1
    for k in range(-(n-1), n):
        # Contribution from h[i] (changes by +δ)
        coeff_i = 0.0
        # h[i] as first factor: valid if 0 ≤ i ≤ n-1-|k|... 
        # Actually, let me just check the conditions directly
        # f_k = sum_m h[m]*(1-h[m+k]) for valid m, m+k
        # h[i] appears as h[m] when m=i: need 0≤i≤n-1 and 0≤i+k≤n-1
        if 0 <= i+k <= n-1:
            coeff_i += (1 - h[i+k])
        # h[i] appears as h[m+k] when m+k=i, i.e., m=i-k: need 0≤i-k≤n-1
        if 0 <= i-k <= n-1:
            coeff_i += (-h[i-k])
        
        # Contribution from h[j] (changes by -δ)
        coeff_j = 0.0
        if 0 <= j+k <= n-1:
            coeff_j += (1 - h[j+k])
        if 0 <= j-k <= n-1:
            coeff_j += (-h[j-k])
        
        g[k + k_offset] = coeff_i - coeff_j  # total coefficient for δ
    
    # Now find δ that minimizes max_k (c_vec[k] + δ*dx*g[k])
    # This is a 1D LP: minimize t s.t. c_vec[k] + δ*dx*g[k] ≤ t for all k
    # The optimal δ is where two or more constraints are tight
    
    # Since it's 1D, we can check all "crossover" points
    # The function f(δ) = max_k (a_k + b_k*δ) where a_k = c_vec[k], b_k = dx*g[k]
    # The minimum occurs where two lines cross, at the "upper envelope"
    
    b = dx * g
    a = c_vec
    
    # Feasible range for δ
    delta_lo = -h[i]
    delta_hi = 1 - h[j]
    
    if delta_lo >= delta_hi:
        return 0.0
    
    # Find the minimum of the max of linear functions
    # The optimal δ is at a point where the upper envelope has a "kink"
    # i.e., where two lines a_k + b_k*δ intersect
    
    # For efficiency, only consider the active (near-max) constraints
    # Actually for n=500, we have 999 lines, so checking all pairs is ~500K which is fine
    
    # Better: the optimal δ minimizes max(a_k + b_kδ)
    # This is equivalent to: find δ* such that the "ceiling" function is minimized
    # The ceiling changes at intersection points of pairs of lines
    
    # Simple approach: ternary search on [delta_lo, delta_hi]
    # f(δ) is convex (max of linear functions), so ternary search works
    
    def f_of_delta(d):
        return np.max(a + d * b)
    
    # Ternary search
    lo, hi = delta_lo, delta_hi
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f_of_delta(m1) < f_of_delta(m2):
            hi = m2
        else:
            lo = m1
    
    delta_opt = (lo + hi) / 2
    
    # Check if it actually improves
    if f_of_delta(delta_opt) < f_of_delta(0) - 1e-10:

def pair_delta(h, n, i, j, c_vec):
    """Find optimal delta for pair (i, j): h[i]+delta, h[j]-delta"""
    dx = 2.0 / n
    # Compute g_k for all k (2n-1 values, indexed 0 to 2n-2)
    # g_k = coefficient of delta in f_k
    # f_k changes by delta*dx*[(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
    # where out-of-range terms are 0
    
    # Vectorized computation:
    k_range = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For h[i] as first factor: valid when 0 <= i + k <= n-1, i.e., -i <= k <= n-1-i
    mask_i1 = (k_range >= -i) & (k_range <= n - 1 - i)
    # h[i] as second factor: valid when 0 <= i - k <= n-1, i.e., i-(n-1) <= k <= i
    mask_i2 = (k_range >= i - (n - 1)) & (k_range <= i)
    
    # Similarly for j
    mask_j1 = (k_range >= -j) & (k_range <= n - 1 - j)
    mask_j2 = (k_range >= j - (n - 1)) & (k_range <= j)
    
    # Contribution: (1-h[m]) where m = i+k (first factor) or m = i-k (second factor)
    g = np.zeros(2 * n - 1)
    
    # h[i] first factor: (1 - h[i+k])
    if mask_i1.any():
        indices = i + k_range[mask_i1]
        g[mask_i1] += (1.0 - h[indices])
    
    # h[i] second factor: -h[i-k]
    if mask_i2.any():
        indices = i - k_range[mask_i2]
        g[mask_i2] += (-h[indices])
    
    # h[j] first factor: -(1 - h[j+k])
    if mask_j1.any():
        indices = j + k_range[mask_j1]
        g[mask_j1] -= (1.0 - h[indices])
    
    # h[j] second factor: +h[j-k]
    if mask_j2.any():
        indices = j - k_range[mask_j2]
        g[mask_j2] += h[indices]
    
    # Now g[k + (n-1)] is the coefficient of delta in f_k (before multiplying by dx)
    # f_k(delta) = c_vec[k+(n-1)] + delta * dx * g[k+(n-1)]
    
    b = dx * g  # shape (2n-1,)
    a = c_vec  # shape (2n-1,)
    
    # Minimize max_k (a[k] + delta * b[k]) subject to -h[i] <= delta <= 1-h[j]
    dlo = -h[i]
    dhi = 1.0 - h[j]
    
    if dlo >= dhi - 1e-15:
        return 0.0
    
    # The function f(d) = max(a + d*b) is convex in d
    # Find minimum using the fact that it's piecewise linear
    # Optimal d is where the slope crosses 0
    
    # Quick check: if f(0) is already the min at a corner
    f_at_lo = np.max(a + dlo * b)
    f_at_hi = np.max(a + dhi * b)
    f_at_0 = np.max(a)
    
    # Use the subgradient: at optimum, there exist k1, k2 with b[k1] < 0 < b[k2]
    # (or b=0 for some active k) and a[k1]+b[k1]*d* = a[k2]+b[k2]*d*
    
    # Efficient approach: only consider "active" constraints (near the max at d=0)
    # Plus constraints that become active for d != 0
    
    # For precision, use a two-phase approach:
    # Phase 1: find approximate d* using subgradient
    # Phase 2: refine using the exact kink
    
    # Simple approach: ternary search (convex function)
    lo, hi = dlo, dhi
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a + m1 * b) <= np.max(a + m2 * b):
            hi = m2
        else:
            lo = m1
    
    d_opt = (lo + hi) / 2.0
    
    # Check improvement
    f_opt = np.max(a + d_opt * b)
    f_0 = np.max(a)
    
    if f_opt < f_0 - 1e-12:
        return float(d_opt)
    return 0.0

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def obj_lbfgs(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_lbfgs(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 50.0), (2000, 500.0), (5000, 5000.0), (20000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(obj_lbfgs, best_h, jac=grad_lbfgs,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def compute_pair_gradient(h, n, i, j):
    dx = 2.0 / n
    k_range = np.arange(-(n-1), n)
    g = np.zeros(2 * n - 1)
    mask_i1 = (k_range >= -i) & (k_range <= n - 1 - i)
    mask_i2 = (k_range >= i - (n-1)) & (k_range <= i)
    mask_j1 = (k_range >= -j) & (k_range <= n - 1 - j)
    mask_j2 = (k_range >= j - (n-1)) & (k_range <= j)
    if np.any(mask_i1):
        g[mask_i1] += (1.0 - h[i + k_range[mask_i1]])
    if np.any(mask_i2):
        g[mask_i2] += (-h[i - k_range[mask_i2]])
    if np.any(mask_j1):
        g[mask_j1] -= (1.0 - h[j + k_range[mask_j1]])
    if np.any(mask_j2):
        g[mask_j2] += h[j - k_range[mask_j2]]
    return g * dx

def pair_search_single(h, n, i, j, c_vec):
    g = compute_pair_gradient(h, n, i, j)
    a = c_vec
    dlo = -h[i]
    dhi = 1.0 - h[j]
    if dlo >= dhi - 1e-15:
        return 0.0
    lo, hi = dlo, dhi
    for _ in range(55):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a + m1 * g) <= np.max(a + m2 * g):
            hi = m2
        else:
            lo = m1
    d_opt = (lo + hi) / 2.0
    f_opt = float(np.max(a + d_opt * g))
    f_0 = float(np.max(a))
    if f_opt < f_0 - 1e-13:
        return float(d_opt)
    return 0.0

def subgrad_vector(h, n, c_vec, thresh_frac=0.0005):
    max_val = c_vec.max()
    thresh = max_val - max(1e-10, thresh_frac * max_val)
    active = np.where(c_vec >= thresh)[0]
    subgrad = np.zeros(n)
    dx = 2.0 / n
    for idx in active:
        k = int(idx) - (n - 1)
        if k >= 0:
            nv = n - k
            subgrad[0:nv] += (1.0 - h[k:n]) * dx
            subgrad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            subgrad[kk:n] += (1.0 - h[0:nv]) * dx
            subgrad[0:nv] += (-h[kk:n]) * dx
    return subgrad

def pair_exact_optimize(h, n, deadline, top_k=12, max_sweeps=10000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    noop_count = 0
    sweep = 0
    while time.time() < deadline:
        sweep += 1
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        if max_val < best_c - 1e-12:
            best_c = max_val
            best_h = h.copy()
        subgrad = subgrad_vector(h, n, c_vec)
        inc_top = np.argsort(-subgrad)[:top_k]
        dec_top = np.argsort(subgrad)[:top_k]
        improved = False
        for ii in range(len(inc_top)):
            i = int(inc_top[ii])
            for jj in range(len(dec_top)):
                j = int(dec_top[jj])
                if i == j:
                    continue
                d = pair_search_single(h, n, i, j, c_vec)
                if d != 0.0:
                    h[i] += d
                    h[j] -= d
                    h[i] = np.clip(h[i], 0.0, 1.0)
                    h[j] = np.clip(h[j], 0.0, 1.0)
                    improved = True
                    c_vec = compute_c5_vec(h, n)
        if sweep % 20 == 0:
            n_rand = 8
            rng = np.random.RandomState(sweep + 7)
            for _ in range(n_rand):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                d = pair_search_single(h, n, i, j, c_vec)
                if d != 0.0:
                    h[i] += d
                    h[j] -= d
                    h[i] = np.clip(h[i], 0.0, 1.0)
                    h[j] = np.clip(h[j], 0.0, 1.0)
                    improved = True
                    c_vec = compute_c5_vec(h, n)
        if improved:
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            noop_count = 0
        else:
            noop_count += 1
            if noop_count > 80:
                break
            if noop_count > 30:
                top_k = max(8, top_k - 1)
    return best_h, best_c

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slqp_eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def slqp_eq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=500000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_eq_fun, 'jac': slqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-18})

        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def block_swap_optimize(h, n, deadline, block_sizes=(2, 3, 5)):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    noop = 0
    for bs in block_sizes:
        if time.time() > deadline:
            break
        for _ in range(200):
            if time.time() > deadline:
                return best_h, best_c
            c_vec = compute_c5_vec(h, n)
            subgrad = subgrad_vector(h, n, c_vec)
            i = int(np.argmax(subgrad)[:max(1, min(bs, n))])
            j = int(np.argmin(subgrad)[max(0, n - bs):])
            if i + bs > n or j + bs > n:
                i = min(int(np.argmax(subgrad)), n - bs)
                j = min(int(np.argmin(subgrad)), n - bs)
            if abs(i - j) < bs:
                continue
            block_i = h[i:i+bs].copy()
            block_j = h[j:j+bs].copy()
            h[i:i+bs] = block_j
            h[j:j+bs] = block_i
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h.copy()
                noop = 0
            else:
                h[i:i+bs] = block_i
                h[j:j+bs] = block_j
                noop += 1
            if noop > 50:
                break
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_passes=100):
