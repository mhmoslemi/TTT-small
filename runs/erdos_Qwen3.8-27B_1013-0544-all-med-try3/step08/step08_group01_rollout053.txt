                if h[hi] - step >= 0.0 and h[hj] + step <= 1.0:
                    h[hi] -= step
                    h[hj] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_improved = True
                    else:
                        h[hi] += step
                        h[hj] -= step
        if not best_improved:
            step *= 0.5
            if step < 1e-6:
                break
    return h, best_c


def multiresolution(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for n_coarse in [n // 2, n // 4]:
        if n_coarse < 50:
            continue
        if time.time() > deadline - 30:
            break
        x_fine = np.linspace(0, 2, n)
        x_coarse = np.linspace(0, 2, n_coarse)
        h_c = np.interp(x_coarse, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)
        t_sub = min(40, (deadline - time.time()) * 0.3)
        if t_sub > 20:
            h_c, c_c = run_lbfgs(h_c, n_coarse, time.time() + t_sub,
                                  alphas=[50, 200, 1000, 5000], mu=100.0, maxiter=1500)
        x_fine2 = np.linspace(0, 2, n)
        h_f = np.interp(x_fine2, x_coarse, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_parent = 500

    n = 1000
    x_parent = np.linspace(0, 2, n_parent)
    x_new = np.linspace(0, 2, n)
    h = np.interp(x_new, x_parent, h_init)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Low-alpha warm-up
    if time.time() < deadline - 500:
        t1 = min(time.time() + 150, deadline - 500)
        h, c1 = run_lbfgs(h, n, t1, alphas=[30, 100, 300], mu=50.0, maxiter=2000)
        if c1 < best_c:
            best_c = c1
            best_h = h.copy()

    # Phase 2: Medium alpha
    if time.time() < deadline - 400:
        t2 = min(time.time() + 150, deadline - 400)
        h, c2 = run_lbfgs(h, n, t2, alphas=[1000, 3000, 10000], mu=200.0, maxiter=3000)
        if c2 < best_c:
            best_c = c2
            best_h = h.copy()

    # Phase 3: Active-set optimization
    if time.time() < deadline - 280:
        t3 = min(time.time() + 120, deadline - 280)
        h, c3 = run_active(h, n, t3, top_k=15, alphas=[5000, 20000, 100000],
                           mu=500.0, maxiter=5000)
        if c3 < best_c:
            best_c = c3
            best_h = h.copy()

    # Phase 4: Very high alpha full
    if time.time() < deadline - 180:
        t4 = min(time.time() + 80, deadline - 180)
        h, c4 = run_lbfgs(h, n, t4, alphas=[50000, 200000], mu=1000.0, maxiter=2000)
        if c4 < best_c:
            best_c = c4
            best_h = h.copy()

    # Phase 5: Multi-resolution
    if time.time() < deadline - 120:
        t5 = min(time.time() + 60, deadline - 120)
        h, c5 = multiresolution(h, n, t5)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()

    # Phase 6: Active-set again (tighter)
    if time.time() < deadline - 80:
        t6 = min(time.time() + 50, deadline - 80)
        h, c6 = run_active(h, n, t6, top_k=8, alphas=[50000, 200000, 1000000],
                           mu=2000.0, maxiter=5000)
        if c6 < best_c:
            best_c = c6
            best_h = h.copy()

    # Phase 7: Gradient-directed search
    if time.time() < deadline - 40:
        t7 = min(time.time() + 30, deadline - 40)
        h, c7 = gradient_directed_search(h, n, t7, base_step=0.003)
        if c7 < best_c:
            best_c = c7
            best_h = h.copy()

    # Phase 8: Local refinement
    if time.time() < deadline - 10:
        t8 = deadline - 3
        h, c8 = local_refine(h, n, t8)
        if c8 < best_c:
            best_c = c8
            best_h = h.copy()

    # Final
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
