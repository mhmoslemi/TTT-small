def fine_pair_search(h, n, deadline, radius=30, deltas=None):

                    # Try moving mass from j to i

                        c = compute_c5(h, n)

def active_only_obj(x, n, idx, alpha, mu):
    """Compute smoothed max only over active shifts (fast)."""
    dx = 2.0 / n
    vals = np.zeros(len(idx))
    for i, si in enumerate(idx):
        k = int(si) - (n - 1)
        if k >= 0:
            nv = n - k
            vals[i] = np.dot(x[0:nv], 1.0 - x[k:n]) * dx
        else:
            kk = -k
            nv = n - kk
            vals[i] = np.dot(x[kk:n], 1.0 - x[0:nv]) * dx
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_only_grad(x, n, idx, alpha, mu):
    """Gradient of the active-only objective."""
    dx = 2.0 / n
    vals = np.zeros(len(idx))
    for i, si in enumerate(idx):
        k = int(si) - (n - 1)
        if k >= 0:
            nv = n - k
            vals[i] = np.dot(x[0:nv], 1.0 - x[k:n]) * dx
        else:
            kk = -k
            nv = n - kk
            vals[i] = np.dot(x[kk:n], 1.0 - x[0:nv]) * dx
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    
    grad = np.zeros(n)
    for w_i, si in zip(w, idx):
        k = int(si) - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += w_i * dx * (1.0 - x[k:n])
            grad[k:n] += w_i * dx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += w_i * dx * (1.0 - x[0:nv])
            grad[0:nv] += w_i * dx * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 0:
                h = np.clip(h + diff / tot * room, 0, 1)
        else:
            if h.sum() > 0:
                h = np.clip(h + diff / h.sum() * h, 0, 1)

            nv = n - kk
            if nv > 0:
                vals[i] = np.dot(x[kk:n], 1.0 - x[0:nv]) * dx
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    for w_i, si in zip(w, idx):
        k = int(si) - (n - 1)
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += w_i * dx * (1.0 - x[k:n])
                grad[k:n] += w_i * dx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += w_i * dx * (1.0 - x[0:nv])
                grad[0:nv] += w_i * dx * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_reactive_lbfgs(h, n, deadline, top_k=5, alpha=5000, mu=0.01, batch_size=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    no_improve_batches = 0

    while time.time() < deadline - 3:
        c_vec = compute_c5_vec(h, n)
        active = find_active(c_vec, top_k, 0.003)
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        try:
            res = minimize(active_only_obj, h, jac=active_only_grad,
                          args=(n, active, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': batch_size, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': batch_size * 60, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
                h = h_new
                no_improve_batches = 0
            else:
                no_improve_batches += 1
        except Exception:
            no_improve_batches += 1

        if no_improve_batches >= 4:
            break

    return best_h, best_c

def run_reactive_lbfgs_multi(h, n, deadline, top_k=5, alphas=None, mu=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [1000, 10000, 100000, 500000]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        batch = 3000 if n <= 500 else 2000
        h, best_c = run_reactive_lbfgs(h, n, deadline, top_k, alpha, mu, batch_size=batch)

    return best_h, best_c

def run_subgradient(h, n, deadline, top_k=5, init_lr=0.004, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = init_lr
    stale = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break

        c_vec = compute_c5_vec(h, n)
        cur_c = c_vec.max()
        active = find_active(c_vec, top_k, 0.004)

        vals = c_vec[active]
        sh = 8000 * (vals - vals.max())
        w = np.exp(sh)
        w /= w.sum()

        grad = np.zeros(n)
        for w_i, idx_i in zip(w, active):
            grad += w_i * shift_grad_single(h, n, idx_i)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            step = lr * mult
            h_trial = h - step * grad / gnorm
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < cur_c - 1e-16:
                h = h_trial
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = h.copy()
                    stale = 0
                else:
                    stale += 1
                lr = min(lr * 1.05, 0.02)
                improved = True
                break

        if not improved:
            lr *= 0.7
            stale += 1

        if stale > 300:
            break

    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=30, deltas=None):
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.004, 0.002, 0.001, 0.0005, 0.0002]

    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        n_iter = 0
        while improved and n_iter < 50:
            improved = False
            n_iter += 1
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            h = project_feasible(h, n)
    return h, best_c

def gradient_guided(h, n, deadline, base_step=0.003):
    best_c = compute_c5(h, n)

    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        active = find_active(c_vec, 5, 0.004)

        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad_single(h, n, idx)
        grad /= len(active)

        g_sorted = np.argsort(-np.abs(grad))[:30]
        found = False

        for i_idx in range(len(g_sorted)):
            if time.time() > deadline:
                return h, best_c
            i = g_sorted[i_idx]
            for j_idx in range(i_idx + 1, len(g_sorted)):
                j = g_sorted[j_idx]
                gi, gj = grad[i], grad[j]
                if gi > gj:
                    di, dj = -1, 1
                else:
                    di, dj = 1, -1
                for mult in [1.0, 0.5, 0.25]:
                    step = base_step * mult
                    hi_val = h[i] + di * step
                    hj_val = h[j] + dj * step
                    if 0.0 <= hi_val <= 1.0 and 0.0 <= hj_val <= 1.0:
                        h[i] = hi_val
                        h[j] = hj_val
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            found = True
                        else:
                            h[i] -= di * step
                            h[j] -= dj * step

        if not found:
            base_step *= 0.5
            if base_step < 1e-7:
                break

    return h, best_c

def random_restart_refine(h, n, deadline, n_restarts=4, pert_scale=0.008):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    time_per = max(8.0, (deadline - time.time()) / max(1, n_restarts) - 5)
    bounds = [(0.0, 1.0)] * n

    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        h_pert = h + rng.randn(n) * pert_scale
        h_pert = project_feasible(h_pert, n)
        t_sub = min(time.time() + time_per, deadline - 12)

        c_vec = compute_c5_vec(h_pert, n)
        active = find_active(c_vec, 4, 0.005)
        try:
            res = minimize(active_only_obj, h_pert, jac=active_only_grad,
                          args=(n, active, 50000, 0.01), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 1500, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 15000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_orig = 300

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick active-set L-BFGS at parent resolution
    if time.time() < deadline - 850:
        t1 = min(time.time() + 50, deadline - 850)
        best_h, best_c = run_reactive_lbfgs_multi(
            best_h, n_orig, t1, top_k=5, alphas=[1000, 10000], mu=0.01)

    # Phase 2: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    h_1000 = np.interp(x_new, x_old, best_h)
    h_1000 = project_feasible(h_1000, n)
    c_1000 = compute_c5(h_1000, n)

    if time.time() > deadline - 20:
        if best_c <= c_1000:
            return best_h, best_c, n_orig
        return h_1000, c_1000, n

    # Phase 3: Main reactive L-BFGS at n=1000 (low alpha)
    if time.time() < deadline - 550:
        t3 = min(time.time() + 180, deadline - 550)
        h_1000, c_1000 = run_reactive_lbfgs(
            h_1000, n, t3, top_k=5, alpha=2000, mu=0.01, batch_size=3000)

    # Phase 4: Reactive L-BFGS (medium alpha)
    if time.time() < deadline - 350:
        t4 = min(time.time() + 130, deadline - 350)
        h_1000, c_1000 = run_reactive_lbfgs(
            h_1000, n, t4, top_k=4, alpha=20000, mu=0.005, batch_size=3000)

    # Phase 5: Reactive L-BFGS (high alpha)
    if time.time() < deadline - 250:
        t5 = min(time.time() + 90, deadline - 250)
        h_1000, c_1000 = run_reactive_lbfgs(
            h_1000, n, t5, top_k=3, alpha=200000, mu=0.003, batch_size=3000)

    # Phase 6: Projected subgradient
    if time.time() < deadline - 150:
        t6 = min(time.time() + 100, deadline - 150)
        h6, c6 = run_subgradient(h_1000, n, t6, top_k=5, init_lr=0.003)
        if c6 < c_1000:
            c_1000 = c6
            h_1000 = h6

    # Phase 7: Another round of reactive L-BFGS (very high alpha, few shifts)
    if time.time() < deadline - 80:
        t7 = min(time.time() + 60, deadline - 80)
        h_1000, c_1000 = run_reactive_lbfgs(
            h_1000, n, t7, top_k=3, alpha=500000, mu=0.002, batch_size=2000)

    # Phase 8: Fine pair search
    if time.time() < deadline - 40:
        t8 = min(time.time() + 35, deadline - 40)
        h_1000, c_1000 = fine_pair_search(
            h_1000, n, t8, radius=20, deltas=[0.002, 0.001, 0.0005, 0.0002])

    # Phase 9: Gradient-guided
    if time.time() < deadline - 20:
        t9 = min(time.time() + 18, deadline - 20)
        h_1000, c_1000 = gradient_guided(h_1000, n, t9, base_step=0.002)

    # Final: return best between original and n=1000
    h_1000 = project_feasible(h_1000, n)
    c_1000 = compute_c5(h_1000, n)

    if best_c <= c_1000:
        best_h = project_feasible(best_h, n_orig)
        best_c = compute_c5(best_h, n_orig)
        return best_h, best_c, n_orig

    return h_1000, c_1000, n
