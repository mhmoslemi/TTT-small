                cur_top_k -= 1
            else:
                break
    
    return best_h, best_c

def lbfgs_phase(h, n, deadline, alpha, mu, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad_fn,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-15, 'maxfun': 50000,
                               'maxls': 100, 'iprint': -1})
        h_new = project_feasible(np.asarray(res.x), n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def pairwise_local_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            c_max = float(c_vec.max())
            
            order = np.argsort(-c_vec)
            active = order[:20]
            
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad(h, n, int(idx)))
            grad /= float(len(active))
            
            top_pos = np.argsort(-grad)[:min(radius * 2, n)]
            
            for i_val in top_pos:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x):
        return smooth_obj(x, n, alpha, 0.0)
    def eq_con(x):
        return [float(x.sum() - n / 2.0)]
    def eq_jac(x):
        return [np.ones(n)]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}]
    try:
        res = minimize(obj_fn, h, jac=lambda x: smooth_grad_fn(x, n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17, 'iprint': 0})
        h_new = project_feasible(np.asarray(res.x), n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: L-BFGS warmup with moderate smoothing (60s)
    t1 = min(t0 + 60, deadline - 700)
    if time.time() < t1:
        h, c = lbfgs_phase(h, n, t1, alpha=2000, mu=500, maxiter=3000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Active-set optimized gradient (180s)
    t2 = min(time.time() + 180, deadline - 480)
    if time.time() < t2:
        h, c = active_set_optimize(h, n, t2, top_k=5, max_iter=8000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 3: High-alpha L-BFGS (120s)
    t3 = min(time.time() + 120, deadline - 320)
    if time.time() < t3:
        h, c = lbfgs_phase(h, n, t3, alpha=50000, mu=10000, maxiter=3000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 4: Active-set again (150s)
    t4 = min(time.time() + 150, deadline - 200)
    if time.time() < t4:
        h, c = active_set_optimize(h, n, t4, top_k=3, max_iter=8000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 5: Pairwise local search (100s)
    t5 = min(time.time() + 100, deadline - 120)
    if time.time() < t5:
        h, c = pairwise_local_search(h, n, t5, radius=40)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 6: More active-set (80s)
    t6 = min(time.time() + 80, deadline - 60)
    if time.time() < t6:
        h, c = active_set_optimize(h, n, t6, top_k=3, maxiter=6000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 7: SLSQP polish (60s)
    t7 = deadline - 10
    if time.time() < t7:
        h, c = slqp_polish(h, n, t7, alpha=1000000, maxiter=2000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Final: verify and return
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
