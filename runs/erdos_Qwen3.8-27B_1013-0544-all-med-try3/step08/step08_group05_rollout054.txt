def swap_delta(h, i, j, n):
    """Compute change in full correlation when h[i] 0->1, h[j] 1->0
    Assumes h[i]==0, h[j]==1 (the reverse swap for our case)
    
    Wait, in our binary search we swap: position with 1 becomes 0, position with 0 becomes 1.
    So h[i] was 1, becomes 0. h[j] was 0, becomes 1.
    """
    # h[i]: 1 -> 0, h[j]: 0 -> 1
    delta = np.zeros(2*n - 1)
    
    for lag in range(-(n-1), n):
        k = lag + (n-1)
        
        # Contribution from h[i] changing 1->0:
        # Term m=i: was 1*(1-h[i+lag]) = 1-h[i+lag], now 0. Delta: -(1-h[i+lag])
        ip = i + lag

        # Term m=i-lag: was h[i-lag]*(1-1) = 0, now h[i-lag]*(1-0) = h[i-lag]. Delta: +h[i-lag]
        im = i - lag
        if 0 <= im < n:
            delta[k] += h[im]

import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return (m + (sm + np.log(np.exp(sh - sm).sum())) / alpha) * dx

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})

        return h

    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    T_start = 0.2
    T_end = 0.00005
    total_planned = 5000000
    it = 0
    stale = 0
    
    while time.time() < deadline:
        it += 1
        frac = min(it / total_planned, 1.0)
        T = T_start * (T_end / T_start) ** frac
        if T < T_end:
            T = T_end
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        if i == j:
            continue
        
        delta = swap_delta_binary(h, i, j, n)
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        dE = new_c - cur_c
        
        if dE <= 0 or (dE < 0.5 and rng.random() < np.exp(-dE / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = new_c
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
        else:
            stale += 1
        
        if stale > 200000 and it % 500000 == 0:
            # Apply a few random swaps to escape
            for _ in range(10):
                a = zeros[rng.randint(len(zeros))]
                b = ones[rng.randint(len(ones))]
                if a != b:
                    h[a], h[b] = 1.0, 0.0
                    ones.remove(b)
                    ones.append(a)
                    zeros.remove(a)
                    zeros.append(b)
            corr = np.correlate(h, 1 - h, mode='full').copy()
            cur_c = corr.max() * dx
            stale = 0
    
    return best_h, best_c
