```python
def compute_delta(i, j, h, n):
    """Compute change in correlation when swapping h[i] (0->1) and h[j] (1->0)."""
    delta = np.zeros(2*n-1)
    # -h[i+d] for valid i+d
    for d in range(-(n-1), n):
        idx = i + d
        if 0 <= idx < n:
            delta[d + n - 1] -= h[idx]
    # +(1-h[i-d]) for valid i-d
    for d in range(-(n-1), n):
        idx = i - d
        if 0 <= idx < n:
            delta[d + n - 1] += 1 - h[idx]
    # +h[j+d] for valid j+d
    for d in range(-(n-1), n):
        idx = j + d
        if 0 <= idx < n:
            delta[d + n - 1] += h[idx]
    # -(1-h[j-d]) for valid j-d
    for d in range(-(n-1), n):
        idx = j - d
        if 0 <= idx < n:
            delta[d + n - 1] -= 1 - h[idx]
    return delta
```

Change: h'[q-d] (which is h[q-d] if q-d≠p, 1 if q-d=p)

So the total change at lag d (array index k = d+n-1) is:
Δ(d) = [1-h'[p+d]] - [1-h'[q+d]] - [h'[p-d]] + [h'[q-d]]

where [x] means 0 if the index is out of range.

For the "generic" case (no cross terms, i.e., p+d≠q, p-d≠q, q+d≠p, q-d≠p):
Δ(d) = (1-h[p+d]) - (1-h[q+d]) - h[p-d] + h[q-d]

Since h is binary:

- (1-h[q+d]) is 1 if h[q+d]=0, 0 if h[q+d]=1
- h[p-d] is 1 if h[p-d]=1, 0 if h[p-d]=0
- h[q-d] is 1 if h[q-d]=1, 0 if h[q-d]=0

So Δ(d) ∈ {-2, -1, 0, 1, 2} in the generic case.

To minimize Δ(d*) (make it as negative as possible):
- Want (1-h[p+d*]) = 0, i.e., h[p+d*] = 1
- Want (1-h[q+d*]) = 1, i.e., h[q+d*] = 0
- Want h[p-d*] = 1
- Want h[q-d*] = 0

This gives Δ(d*) = 0 - 1 - 1 + 0 = -2.

So the "best" swap for reducing corr at lag d* is one where:
- h[p+d*] = 1 (p is "below" a 1 at distance d*)
- h[q+d*] = 0 (q is "below" a 0 at distance d*)
- h[p-d*] = 1 (p is "above" a 1 at distance d*)
- h[q-d*] = 0 (q is "above" a 0 at distance d*)

This means p is surrounded by 1's at distance ±d*, and q is surrounded by 0's at distance ±d*.

This gives a nice heuristic for finding good swaps!

```python
def compute_c5(h, n):

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n, rng):
    """Convert h to binary with exactly n//2 ones."""
    target = n // 2
    h = np.array(h, dtype=float)
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def swap_delta(i, j, h, n):
    """
    Compute delta in correlation when h[i]: 0->1, h[j]: 1->0.
    Returns array of length 2n-1.
    """
    e = np.arange(-(n-1), n)
    
    # A: (1 - h'[i+e]) for valid i+e
    idx_a = i + e
    mask_a = (idx_a >= 0) & (idx_a < n)
    A = np.zeros(2*n-1)
    A[mask_a] = 1.0 - h[idx_a[mask_a]]
    # Cross term: if i+e == j, h'[i+e] = 0, so A should be 1
    cross_a = (idx_a == j) & mask_a
    A[cross_a] = 1.0
    
    # B: -h'[i-e] for valid i-e
    idx_b = i - e
    mask_b = (idx_b >= 0) & (idx_b < n)
    B = np.zeros(2*n-1)
    B[mask_b] = -h[idx_b[mask_b]]
    # Cross term: if i-e == j, h'[i-e] = 0, so B = 0
    cross_b = (idx_b == j) & mask_b
    B[cross_b] = 0.0
    
    # C: -(1 - h'[j+e]) for valid j+e
    idx_c = j + e
    mask_c = (idx_c >= 0) & (idx_c < n)
    C = np.zeros(2*n-1)
    C[mask_c] = -(1.0 - h[idx_c[mask_c]])
    # Cross term: if j+e == i, h'[j+e] = 1, so C = 0
    cross_c = (idx_c == i) & mask_c
    C[cross_c] = 0.0
    
    # D: h'[j-e] for valid j-e
    idx_d = j - e
    mask_d = (idx_d >= 0) & (idx_d < n)
    D = np.zeros(2*n-1)
    D[mask_d] = h[idx_d[mask_d]]
    # Cross term: if j-e == i, h'[j-e] = 1, so D = 1
    cross_d = (idx_d == i) & mask_d
    D[cross_d] = 1.0
    
    return A + B + C + D
```

```python
def fast_local_search(h, n, deadline, seed=42):
    """Fast local search focusing on critical lags."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = compute_corr(h, n)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    stall = 0
    max_stall = 50
    
    while time.time() < deadline:
        # Find critical lag
        d_star = corr.argmax()  # array index
        d = d_star - (n - 1)  # actual lag
        
        # Find candidate swaps that reduce corr at d_star
        # Best case: h[i+d]=1, h[i-d]=1, h[j+d]=0, h[j-d]=0
        # This gives delta at d_star = -2
        
        candidates = []
        
        # For zeros i: check conditions at position i
        for i in zeros:
            cond_i = 0
            if 0 <= i + d < n and h[i+d] == 1:
                cond_i += 1  # A term = 0 (good, no increase)
            if 0 <= i - d < n and h[i-d] == 1:
                cond_i += 1  # B term = -1 (good, decrease)
            
            if cond_i >= 1:  # At least one condition met
                # Find j that complements
                for j in ones:
                    cond_j = 0
                    if 0 <= j + d < n and h[j+d] == 0:
                        cond_j += 1  # C term = -1 (good)
                    if 0 <= j - d < n and h[j-d] == 0:
                        cond_j += 1  # D term = 0 (good)
                    
                    score = cond_i + cond_j
                    if score >= 2:
                        candidates.append((score, i, j))
        
        # Sort by score and try top candidates
        candidates.sort(key=lambda x: -x[0])
        
        improved = False
        for score, i, j in candidates[:20]:
            delta = swap_delta(i, j, h, n)
            new_corr = corr + delta
            new_max = new_corr.max()
            if new_max < cur_max - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_max = new_max
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                break
        
        if not improved:
            stall += 1
            if stall > max_stall:
                # Perturbation
                if ones and zeros:
                    a = rng.choice(ones)
                    b = rng.choice(zeros)
                    delta = swap_delta(b, a, h, n)
                    h[b], h[a] = 1.0, 0.0
                    corr = corr + delta
                    cur_max = corr.max()
                    ones.remove(a)
                    ones.append(b)
                    zeros.remove(b)
                    zeros.append(a)
                stall = 0
        
        # Also try random swaps occasionally
        if rng.random() < 0.1 and ones and zeros:
            a = rng.choice(ones)
            b = rng.choice(zeros)
            delta = swap_delta(b, a, h, n)
            new_corr = corr + delta
            new_max = new_corr.max()
            if new_max <= cur_max:
                h[b], h[a] = 1.0, 0.0
                corr = new_corr
                cur_max = new_max
                ones.remove(a)
                ones.append(b)
                zeros.remove(b)
                zeros.append(a)
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
    
    return best_h, best_max * dx
```
