    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-13:

            if tot < 1e-15:

            h += (diff / tot) * h

    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 0)
