from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgrad_line_search(h, n, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        
        # Find active set (within tolerance)
        tol = max(1e-8, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        # Compute subgradient (average of gradients of active functions)
        grad = np.zeros(n)
        dx = 2.0 / n
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= max(len(active), 1)
        
        # Line search: try a few step sizes
        best_step_c = c_max
        best_step_h = h
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = eta * factor
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_step_c:
                best_step_c = c_trial
                best_step_h = h_trial
        
        if best_step_c < best_c - 1e-15:
            best_c = best_step_c
            best_h = best_step_h
            h = best_step_h
            eta = min(eta * 1.02, 0.1)
        else:
            h = project_feasible(h - 0.01 * eta * grad, n)
            eta *= 0.95
    
    return best_h, best_c

Actually, let me reconsider the problem once more. The key question is: what's the theoretical minimum of C5?

For a continuous function h: [0,2] → [0,1] with ∫h = 1:
C5 = max_k ∫₀^{2-|k|} h(x)(1-h(x+k)) dx (for k > 0, and similar for k < 0)

Note that ∫₀^{2-k} h(x)(1-h(x+k)) dx + ∫₀^{2-k} h(x+k)(1-h(x)) dx = ∫₀^{2-k} [h(x) + h(x+k) - 2h(x)h(x+k)] dx

The first term: ∫₀^{2-k} h(x) dx + ∫₀^{2-k} h(x+k) dx = ∫₀^{2-k} h(x) dx + ∫_{k}^{2} h(x) dx

The second term: 2∫₀^{2-k} h(x)h(x+k) dx = 2 * autocorrelation at lag k

So: overlap(k) + cross(k) = ∫₀^{2-k} h dx + ∫_k^2 h dx - 2*R(h)(k)

where R(h)(k) = ∫₀^{2-k} h(x)h(x+k) dx

This doesn't immediately help, but it shows the relationship between the overlap and the autocorrelation.

At the optimum, we want all the overlaps to be equal (by the "minimax" principle). So if the optimal value is C*, then for all active k: overlap(k) = C*.

The number of active k's at the optimum constrains the solution. If there are m active constraints, then the solution lies in an m-dimensional "ridge" in the space of feasible functions.

from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
