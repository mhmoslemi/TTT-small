import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(x, n, alpha):
    """Objective and gradient for LSE-smoothed Erdos problem."""
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    k_star = int(np.argmax(g))
    lag = k_star - (n - 1)
    
    # Gradient of max(g)
    grad_m = np.zeros(n)
    jm = max(0, lag)
    jM = min(n, n + lag)
    if jm < jM:
        grad_m[jm:jM] += (1.0 - x[jm - lag:jM - lag])
    jm2 = max(0, -lag)
    jM2 = min(n, n - lag)
    if jm2 < jM2:
        grad_m[jm2:jM2] -= x[jm2 + lag:jM2 + lag]
    grad_m *= dx
    
    # LSE
    sh = alpha * (g - m)
    sm = sh.max()
    ev = np.exp(sh - sm)
    p = ev / ev.sum()
    lse_val = (sm + np.log(ev.sum())) / alpha
    
    obj = m + lse_val
    
    # Gradient of LSE/alpha
    if alpha > 100:
        A = np.zeros(n)
        B = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            lk = k - (n - 1)
            a, b = max(0, lk), min(n, n + lk)
            if a < b:
                A[a:b] += pk * (1.0 - x[a - lk:b - lk])
            a2, b2 = max(0, -lk), min(n, n - lk)
            if a2 < b2:
                B[a2:b2] += pk * x[a2 + lk:b2 + lk]
        grad = dx * (A - B) / alpha
    else:
        # Numerical
        eps = 1e-8
        grad = np.zeros(n)
        for i in range(n):
            xi = x[i]
            x[i] = xi + eps
            g2 = np.correlate(x, 1 - x, mode='full') * dx
            sh2 = alpha * (g2 - g2.max())
            sm2 = sh2.max()
            lse2 = (sm2 + np.log(np.exp(sh2 - sm2).sum())) / alpha
            x[i] = xi
            grad[i] = (lse2 - lse_val) / eps
    
    return obj, grad_m + grad

def slsqp_optimize(h, n, alpha, deadline, maxiter=500):
    """SLSQP optimization with LSE-smoothed objective."""
    def fun(x):
        return obj_grad(x, n, alpha)[0]
    def jac(x):
        return obj_grad(x, n, alpha)[1]
    def eq_fun(x):
        return x.sum() - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
