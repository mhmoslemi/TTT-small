    x_old = np.linspace(0, 2.0, n_old)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def make_binary_from_continuous(h, n):
    h_bin = np.zeros(n)
    n_ones = int(n / 2)
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)

def make_periodic(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(ones_per_period):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: L-BFGS annealing at parent resolution
    p1_end = t0 + 250
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[30, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000],
        mu=0.01, maxiters=[1500, 1500, 2000, 2500, 3000, 3000, 3500, 4000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active-set top-k optimization at parent resolution
    if time.time() < deadline - 300:
        p2_end = min(time.time() + 120, deadline - 300)
        h2, c2 = run_topk_active(best_h, best_n, p2_end, top_k=5,
            alphas=[10000, 50000, 200000, 1000000, 5000000], rounds=20)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Upsample to n=1000
    n_hi = 1000
    if time.time() < deadline - 280:
        h_hi = upsample(best_h, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        p3_end = min(time.time() + 250, deadline - 200)
        h3, c3 = run_lbfgs(h_hi, n_hi, p3_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
            mu=0.005, maxiters=[1000, 1500, 2000, 2500, 3000, 3000, 3500, 3500])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_hi

    # Phase 4: Active-set at best resolution
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 100, deadline - 130)
        h4, c4 = run_topk_active(best_h, best_n, p4_end, top_k=4,
            alphas=[50000, 200000, 1000000, 5000000], rounds=25)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try binary-like start at parent resolution
    if time.time() < deadline - 120:
        h_bin = make_binary_from_continuous(best_h, n0)
        c_bin = compute_c5(h_bin, n0)
        if c_bin < best_c:
            pass
        h5, c5 = run_lbfgs(h_bin, n0, min(time.time() + 60, deadline - 100),
            alphas=[100, 1000, 10000, 100000, 1000000],
            mu=0.01, maxiters=[2000, 2000, 2500, 3000, 3000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n0

    # Phase 6: Upsample to n=2000 if time permits
    if time.time() < deadline - 100:
        n_hi2 = 2000
        h_hi2 = upsample(best_h, n_hi2)
        p6_end = min(time.time() + 80, deadline - 60)
        h6, c6 = run_lbfgs(h_hi2, n_hi2, p6_end,
            alphas=[10, 50, 200, 1000, 5000],
            mu=0.003, maxiters=[500, 800, 1000, 1500, 1500])
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = n_hi2

    # Phase 7: SLSQP polish
    if time.time() < deadline - 40:
        for alpha in [500000, 2000000]:
            if time.time() > deadline - 30:
                break
            h7, c7 = slqp_opt(best_h, best_n, deadline - 25, alpha=alpha, maxiter=1000)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Subgradient refinement
    if time.time() < deadline - 25:
        p8_end = deadline - 10
        h8, c8 = subgrad_step(best_h, best_n, p8_end, n_iter=5000)

        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
