import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n, rng):
    """Convert to binary by keeping top n/2 values as 1."""
    h = np.array(h, dtype=float)
    # Add tiny random perturbation to break ties
    h_tied = h + rng.random(n) * 1e-10
    idx = np.argsort(h_tied)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def make_thue_morse(n, rng):
    """Thue-Morse inspired balanced binary sequence."""
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    # Balance
    ones = np.where(t == 1)[0]
    zeros = np.where(t == 0)[0]
    target = n // 2
    if len(ones) > target:
        excess = len(ones) - target
        perm = rng.permutation(len(ones))
        for idx in perm[:excess]:
            t[ones[idx]] = 0
    elif len(ones) < target:
        need = target - len(ones)
        perm = rng.permutation(len(zeros))
        for idx in perm[:need]:
            t[zeros[idx]] = 1
    return t.astype(float)

def sa_binary(h_init, n, deadline, rng, T_init=0.04, T_min=0.001, max_steps=None):
    """Simulated annealing on binary solutions with single-swap moves."""
    if max_steps is None:
        max_steps = 2000000
    
    h = h_init.copy()
    # Ensure binary
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    current_c = best_c
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    no_improve = 0
    improve_thresh = 50000  # reheat after this many steps without improvement
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        
        progress = step / max_steps
        T = T_init * (T_min / T_init) ** progress
        if T < T_min:
            T = T_min
        
        # Reheat
        if no_improve > improve_thresh:
            T = T_init * 0.5
            no_improve = 0
        
        # Random swap
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        ip = ones[ii]
        jp = zeros[jj]
        
        h_bin[ip] = 0.0
        h_bin[jp] = 1.0
        c_new = compute_c5(h_bin, n)
        
        delta = c_new - current_c
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c = c_new
            ones[ii] = jp
            zeros[jj] = ip
        else:
            h_bin[ip] = 1.0
            h_bin[jp] = 0.0
    
    return best_h, best_c

def sa_binary_multi(h_init, n, deadline, rng, T_init=0.04, T_min=0.001, max_steps=None, multi_prob=0.15):
    """SA with occasional multi-swap moves for better exploration."""
    if max_steps is None:
        max_steps = 2000000
    
    h = h_init.copy()
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    current_c = best_c
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    no_improve = 0
    improve_thresh = 50000
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        
        progress = step / max_steps
        T = T_init * (T_min / T_init) ** progress
        if T < T_min:
            T = T_min
        if no_improve > improve_thresh:
            T = T_init * 0.5
            no_improve = 0
        
        if rng.random() < multi_prob:
            # Multi-swap: swap 2 or 3 pairs
            m = rng.randint(2, 4)  # 2 or 3 pairs
            # Select m random ones and m random zeros
            sel_ones = rng.choice(len(ones), m, replace=False)
            sel_zeros = rng.choice(len(zeros), m, replace=False)
            
            orig_ones = ones[sel_ones].copy()
            orig_zeros = zeros[sel_zeros].copy()
            
            for k in range(m):
                h_bin[orig_ones[k]] = 0.0
                h_bin[orig_zeros[k]] = 1.0
            
            c_new = compute_c5(h_bin, n)
            delta = c_new - current_c
            
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if delta < 0 or rng.random() < np.exp(-delta / T):
                current_c = c_new
                ones[sel_ones] = orig_zeros
                zeros[sel_zeros] = orig_ones
            else:
                for k in range(m):
                    h_bin[orig_ones[k]] = 1.0
                    h_bin[orig_zeros[k]] = 0.0
        else:
            # Single swap
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            ip = ones[ii]
            jp = zeros[jj]
            
            h_bin[ip] = 0.0
            h_bin[jp] = 1.0
            c_new = compute_c5(h_bin, n)
            delta = c_new - current_c
            
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if delta < 0 or rng.random() < np.exp(-delta / T):
                current_c = c_new
                ones[ii] = jp
                zeros[jj] = ip
            else:
                h_bin[ip] = 1.0
                h_bin[jp] = 0.0
    
    return best_h, best_c

def continuous_refine(h_bin, n, deadline, rng):
    """Try to improve a binary solution with small continuous perturbations."""
    h = h_bin.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Try pushing pairs of values: one up, one down (maintaining sum)
    lr = 0.01
    for it in range(20000):
        if time.time() > deadline:
            break
        
        # Pick a 1-position and a 0-position
        ones_mask = h > 0.9
        zeros_mask = h < 0.1
        if ones_mask.sum() == 0 or zeros_mask.sum() == 0:
            break
        
        i = rng.choice(np.where(ones_mask)[0])
        j = rng.choice(np.where(zeros_mask)[0])
        
        h[i] -= lr
        h[j] += lr
        h = np.clip(h, 0, 1)
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            lr = min(lr * 1.1, 0.1)
        else:
            h[i] += lr
            h[j] -= lr
            h = np.clip(h, 0, 1)
            lr *= 0.95
    
    return best_h, best_c

def greedy_improve_binary(h_bin, n, deadline, rng, max_iters=5000):
    """Greedy: try all single swaps, keep the best one each time."""
    h = h_bin.copy()
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        best_delta = 0
        best_i = -1
        best_j = -1
        
        # Try a random subset of swaps (too slow to try all n/2 * n/2)
        n_trials = min(200, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            ip = ones[i]
            jp = zeros[j]
            h_bin[ip] = 0.0
            h_bin[jp] = 1.0
            c = compute_c5(h_bin, n)
            h_bin[ip] = 1.0
            h_bin[jp] = 0.0
            delta = c - best_c
            if delta < best_delta:
                best_delta = delta
                best_i = i
                best_j = j
        
        if best_i < 0:
            break
        
        ip = ones[best_i]
        jp = zeros[best_j]
        h_bin[ip] = 0.0
        h_bin[jp] = 1.0
        best_c += best_delta
        ones[best_i] = jp
        zeros[best_j] = ip
    
    return h_bin, best_c

def continuous_refine(h_in, n, deadline, rng):

    resolutions = [256, 250, 300, 400, 200]
    time_alloc = {256: 3.0, 250: 2.5, 300: 2.0, 400: 1.5, 200: 1.0}
    total_w = sum(time_alloc.values())
    sa_time_total = (dl - t0) * 0.65

    for n in resolutions:
        if time.time() > dl - 80:
            break
        alloc = sa_time_total * time_alloc[n] / total_w
        phase_dl = min(time.time() + alloc, dl - 80)

        inits = []
        if n0 > 0:
            x0 = np.linspace(0, 2, n0)
            xn = np.linspace(0, 2, n)
            h_interp = np.interp(xn, x0, h0)
            inits.append(binarize(h_interp, n, rng))
        inits.append(make_thue_morse(n, rng))
        h_rand = np.zeros(n)
        h_rand[rng.permutation(n)[:n // 2]] = 1.0
        inits.append(h_rand)
        if n <= 300:
            inits.append(greedy_construct(n, seed=seed))

        for init_h in inits:
            if time.time() > phase_dl - 10:
                break
            h_res, c_res = sa_binary(init_h, n, phase_dl, rng,
                                      T_init=0.05, T_min=0.0005,
                                      max_steps=5000000, multi_prob=0.12)
            if c_res < best_c:
                best_c = c_res
                best_h = h_res.copy()
                best_n = n

        if time.time() < dl - 60:
            h_tl, c_tl = targeted_local_search(best_h if best_n == n else inits[0],
                                                n, min(time.time() + 30, dl - 60), rng)
            if c_tl < best_c:
                best_c = c_tl
                best_h = h_tl.copy()
                best_n = n

        if time.time() < dl - 40:
            h_bs, c_bs = block_swap_sa(best_h if best_n == n else inits[0],
                                        n, min(time.time() + 20, dl - 40), rng, max_steps=200000)
            if c_bs < best_c:
                best_c = c_bs
                best_h = h_bs.copy()
                best_n = n

    if time.time() < dl - 50:
        h_res, c_res = sa_binary(best_h, best_n, dl - 35, rng,
                                  T_init=0.02, T_min=0.0002,
                                  max_steps=8000000, multi_prob=0.08)
        if c_res < best_c:
            best_c = c_res
            best_h = h_res.copy()
            best_n = best_n

    if time.time() < dl - 30:
        h_tl, c_tl = targeted_local_search(best_h, best_n, dl - 20, rng, max_iters=30000)
        if c_tl < best_c:
            best_c = c_tl
            best_h = h_tl.copy()
            best_n = best_n

    if time.time() < dl - 10:
        h_cr, c_cr = continuous_refine(best_h, best_n, dl - 5, rng)
        if c_cr < best_c:
            best_c = c_cr
            best_h = h_cr
            best_n = best_n

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
