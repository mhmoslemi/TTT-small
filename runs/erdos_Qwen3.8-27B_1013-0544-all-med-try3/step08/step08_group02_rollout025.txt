Let me optimize: precompute the scores for all a and b at each active shift.

Actually, for a given shift k_val, the score for position i is:
score(i, k_val) = h[i+k_val]*I(valid) + h[i-k_val]*I(valid)

This can be precomputed for all i in O(n) for each k_val. Then finding the best a (min score among ones) and best b (max score among zeros) is O(n).

So per iteration: O(n) for the scores + O(n) for the best candidates + O(n) for computing c_vec. Total O(n) per iteration. For n=500 and 50000 iterations, that's 25M operations. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def linalg_obj(x, n):

    best_c = compute_c5(best_h, n)

    for i, alpha in enumerate(alphas):

        if remaining < 10:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 4))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 6,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass

def make_binary(h, n):
    n_ones = int(round(n / 2.0))
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-12:
        if n_ones < n:
            h_bin[order[n_ones]] += remaining
    return project_feasible(h_bin, n)

def periodic_binary(n, period, ones_per_period, phase=0):
    h = np.zeros(n)
    for p in range(ones_per_period):
        for i in range(n):
            if (i - phase - p) % period == 0:
                h[i] = 1.0
    return project_feasible(h, n)

def slaq_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def sum_fun(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, deadline, alpha=500000, maxiter=2000):

    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    try:

                      bounds=bounds, constraints=cons,

        c_new = compute_c5(h_new, n)

        h5, c5 = binary_local_search(h_bin, best_n, p5_end, max_iters=200000)
        if c5 < best_c:
            best_c, best_h, best_n = c5, h5, best_n

    # Phase 6: Try different resolutions for binary search
    for n_try in [800, 1200, 2000]:
        if time.time() > deadline - 100:
            break
        h_try = upsample(best_h, best_n, n_try)
        h_try_bin = make_binary(h_try, n_try)
        c_try = compute_c5(h_try_bin, n_try)
        p6_end = min(time.time() + 60, deadline - 80)
        h6, c6 = binary_local_search(h_try_bin, n_try, p6_end, max_iters=100000)
        if c6 < best_c:
            best_c, best_h, best_n = c6, h6, n_try

    # Phase 7: SLSQP polish
    if time.time() < deadline - 50:
        for alpha_try in [200000, 1000000]:
            if time.time() > deadline - 40:
                break
            h7, c7 = slsqp_refine(best_h, best_n, deadline - 35, alpha=alpha_try, maxiter=1500)
            if c7 < best_c:
                best_c, best_h = c7, h7

    # Phase 8: Subgradient refinement
    if time.time() < deadline - 20:
        h8, c8 = subgrad_refine(best_h, best_n, deadline - 5, n_iter=15000)
        if c8 < best_c:
            best_c, best_h = c8, h8

    # Final: project and verify
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
