        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, k_idx):

    dx = 2.0 / n
    grad = np.zeros(n)

        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx

        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx

def nest_a_subgradient(h, n, deadline, max_iters=3000):
    """Nesterov-accelerated subgradient method for min max_k C(k)."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    h_prev = h.copy()
    y = h.copy()
    
    for t in range(max_iters):
        if time.time() > deadline:
            break
        t_idx = t + 1
        step = 0.05 / math.sqrt(t_idx + 1)
        
        c = compute_c5_vec(y, n)
        c_max = c.max()
        thr = c_max - max(1e-8, 1e-4 * c_max)
        active = np.where(c >= thr)[0]
        
        if len(active) == 0:
            active = np.array([(n - 1) - 1, (n - 1), (n - 1) + 1])
            active = active[(active >= 0) & (active < 2 * n - 1)]
        
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad_single(y, n, idx)
        grad /= len(active)
        
        h_new = y - step * grad
        h_new = project_feasible(h_new, n)
        
        beta = (1.0 + math.sqrt(1.0 + 4.0 * t_idx * t_idx)) / 2.0

        c_new = compute_c5(h_new, n)

            best_c = c_new
            best_h = h_new.copy()
        
        h_prev = h_curr
        h_curr = h_new

    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    thr = max_val - max(1e-8, 1e-3 * max_val)
    active_shifts = np.where(c_vec >= thr)[0]
    
    grad = np.zeros(n)
    for idx in active_shifts:
        grad += shift_grad_single(h, n, idx)
    grad /= max(1, len(active_shifts))
    
    # Focus on indices with |grad| in top range, bounded by 80 indices
    grad_abs = np.abs(grad)
    sorted_idx = np.argsort(-grad_abs)
    candidate_idx = sorted_idx[:80]
    
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for ii in range(len(candidate_idx)):
                if time.time() > deadline:
                    return best_h if False else h, best_c
                i = candidate_idx[ii]
                for jj in range(ii + 1, len(candidate_idx)):
                    j = candidate_idx[jj]
                    # Try h[i]+d, h[j]-d
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i]-d, h[j]+d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    
    h = project_feasible(h, n)
    return h, best_c

def directed_pair_search(h, n, deadline):
    """Use gradient direction to guide pair moves with line search."""
    best_c = compute_c5(h, n)
    
    for outer in range(50):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thr = max_val - max(1e-8, 2e-3 * max_val)
        active_shifts = np.where(c_vec >= thr)[0]
        
        grad = np.zeros(n)
        for idx in active_shifts:
            grad += shift_grad_single(h, n, idx)
        grad /= max(1, len(active_shifts))
        
        # Find best 2-point move: decrease grad at one, increase at other
        # We want to move in -grad direction: where grad is positive, decrease h; where negative, increase
        pos_idx = np.where(grad > 0.001)[0]
        neg_idx = np.where(grad < -0.001)[0]
        
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            # Try within same-sign
            top_pos = np.argsort(-grad)[:30]
            top_neg = np.argsort(grad)[:30]
            pos_idx = top_pos[grad[top_pos] > 0]
            neg_idx = top_neg[grad[top_neg] < 0]
            if len(pos_idx) == 0 or len(neg_idx) == 0:
                break
        
        # Limit search
        pos_idx = pos_idx[:40]
        neg_idx = neg_idx[:40]
        
        best_improve = 0
        best_move = None
        
        for i in pos_idx:
            if time.time() > deadline:
                break
            for j in neg_idx:
                if i == j:
                    continue
                for step_mult in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
                    t = step_mult
                    if h[i] - t >= 0 and h[j] + t <= 1:
                        h[i] -= t
                        h[j] += t
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_improve = best_improve  # track
                            best_move = (i, j, -t, t)
                        else:
                            h[i] += t
                            h[j] -= t
                            break
                    else:
                        break
        
        if best_move is None:
            break
    
    return h, best_c

def find_top_shifts(h, n, k=5):
    c_vec = compute_c5_vec(h, n)
    idx = np.argsort(-c_vec)[:k]
    return idx.tolist()

def iterative_active_refine(h, n, deadline, n_rounds=30):
    """Iteratively identify top shifts and optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [1000, 5000, 20000, 100000, 200000]
    
    for rnd in range(n_rounds):
        if time.time() > deadline - 10:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Top shifts within 0.05% of max
        thr = max_val - max(1e-9, 5e-5 * max_val)
        active = np.where(c_vec >= thr)[0]
        if len(active) > 8:
            top_active = np.argsort(-c_vec[active])[:8]
            active = active[top_active]
        if len(active) == 0:
            active = find_top_shifts(best_h, n, 3)
        
        alpha = alphas[min(rnd % len(alphas), len(alphas) - 1)]
        mu = 1.0
        
        t_sub = min(time.time() + 60, deadline - 10)
        h_new, c_new = lbfgs_on_active(best_h, n, active, alpha, mu, t_sub)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def multi_start_refine(h, n, deadline, n_starts=5):
    """Try small random perturbations to escape local optima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    remaining = deadline - time.time()
    per_start = remaining / (n_starts + 1)
    
    for s in range(n_starts):
        t_end = time.time() + per_start
        if time.time() > deadline - 20:
            break
        
        # Small random perturbation
        rng = np.random.RandomState(42 + s)
        pert = rng.normal(0, 0.01, n)
        h_pert = best_h + pert
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Quick L-BFGS on the perturbed solution
        active = find_top_shifts(h_pert, n, 5)
        h_opt, c_opt = lbfgs_on_active(h_pert, n, active, 10000, 1.0, t_end)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    n = n_orig
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: Full smooth L-BFGS (quick, to get in the right basin)
    if time.time() < deadline - 500:
        t1_end = min(t0 + 60, deadline - 500)
        h, c = lbfgs_full(h, n, alpha=100, mu=10.0, deadline=t1_end)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Active-set focused L-BFGS with increasing alpha
    if time.time() < deadline - 400:
        t2_end = min(t0 + 200, deadline - 400)
        for alpha in [1000, 5000, 20000, 50000]:
            if time.time() > t2_end - 10:
                break
            active = find_top_shifts(best_h, n, 5)
            h_opt, c_opt = lbfgs_on_active(best_h, n, active, alpha, 1.0, t2_end)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
    
    # Phase 3: Iterative active-set refinement
    if time.time() < deadline - 250:
        t3_end = min(t0 + 400, deadline - 250)
        best_h, best_c = iterative_active_refine(best_h, n, t3_end, n_rounds=20)
    
    # Phase 4: Nesterov subgradient (handles the non-smooth max directly)
    if time.time() < deadline - 150:
        t4_end = min(t0 + 550, deadline - 150)
        h_ns, c_ns = nest_a_subgradient(best_h, n, t4_end, max_iters=2000)
        if c_ns < best_c:
            best_c = c_ns
            best_h = h_ns.copy()
    
    # Phase 5: Multi-start to escape local optima
    if time.time() < deadline - 120:
        t5_end = min(t0 + 700, deadline - 120)
        best_h, best_c = multi_start_refine(best_h, n, t5_end, n_starts=4)
    
    # Phase 6: Another active-set round
    if time.time() < deadline - 80:
        t6_end = min(t0 + 850, deadline - 80)
        best_h, best_c = iterative_active_refine(best_h, n, t6_end, n_rounds=10)
    
    # Phase 7: Directed pair search (gradient-guided)
    if time.time() < deadline - 40:
        t7_end = min(t0 + 950, deadline - 40)
        best_h, best_c = directed_pair_search(best_h, n, t7_end)
    
    # Phase 8: Fine pair local search
    if time.time() < deadline - 10:
        t8_end = deadline - 5
        best_h, best_c = pair_local_search(best_h, n, t8_end)
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    # Verify feasibility
    assert np.all(best_h >= -1e-12), "h has negative values"
    assert np.all(best_h <= 1.0 + 1e-12), "h exceeds 1"
    integral = best_h.sum() * (2.0 / n)
    assert abs(integral - 1.0) < 1e-6, f"Integral {integral} not close to 1"
    
    best_h = np.clip(best_h, 0.0, 1.0)
    # Final exact projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
