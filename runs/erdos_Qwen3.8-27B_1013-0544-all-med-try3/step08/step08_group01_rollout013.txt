    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for outer_iter in range(100):
        if time.time() > deadline - 10:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        # Update weights: put more weight on large c_k
        T_eff = max(0.001, T / (1 + outer_iter * 0.1))
        w = np.exp(c_vec / T_eff)
        w /= w.sum()
        
        # Inner: minimize weighted sum with L-BFGS-B
        # Define objective and gradient
        def obj(x):
            cv = compute_c5_vec(x, n)
            return float(np.dot(w, cv))
        
        def grad(x):
            cv = compute_c5_vec(x, n)
            # Gradient of weighted sum
            dx_ = 2.0 / n
            g = np.zeros(n)
            for kk in range(2*n-1):
                if w[kk] < 1e-12:
                    continue
                k = kk - (n-1)
                wm = w[kk] * dx_
                if k >= 0:
                    nv = n - k
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
                else:
                    kk2 = -k
                    nv = n - kk2
                    g[kk2:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk2:n])
            return g
        
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=[(0,1)]*n,
                          options={'maxiter': max_inner, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except:
            break
    
    return best_h, best_c

    # This is equivalent to cross-correlation
    # Using FFT: cross-correlation can be computed via FFT
    m = len(h) + len(1-h) - 1  # = 2n-1
    # Pad to next power of 2
    from scipy.fft import next_fast_len
    N = next_fast_len(2*m)
    H = np.fft.rfft(h, N)

        c = compute_c5(best_h, n)
        if c < best_c:
            best_c = c
    
    return best_h, best_c

# --- Subgradient with momentum ---

def subgrad_step(h, n, active_shifts):
    dx_ = 2.0 / n
    grad = np.zeros(n)
    for idx in active_shifts:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += (1.0 - h[k:]) * dx_
            grad[k:] += (-h[:nv]) * dx_
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += (1.0 - h[:nv]) * dx_
            grad[:nv] += (-h[kk:]) * dx_
    return grad / len(active_shifts)

def subgradient_momentum(h, n, deadline, eta0=0.005, momentum=0.9, tolerance=0.005):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    no_improve = 0
    
    for it in range(100000):
        if time.time() > deadline - 2:
            break
        
        eta = eta0 / (1.0 + 0.005 * it)
        c_vec = compute_c5_vec(best_h, n)
        m = c_vec.max()
        threshold = m - max(1e-10, tolerance * m)
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) == 0:
            break
        
        grad = subgrad_step(best_h, n, active)
        
        # Nesterov
        y = best_h - momentum * v
        v_new = eta * grad
        y = y - v_new
        v = v_new
        best_h = project_feasible(y, n)
        
        c = compute_c5(best_h, n)
        if c < best_c - 1e-12:
            best_c = c
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break
    
    return best_h, best_c

# --- Pair interchange search ---

def pair_search(h, n, deadline, tol_frac=0.01, max_pairs=50):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, tol_frac * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) == 0:
            break
        
        dx_ = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:]) * dx_
                grad[k:] += (-best_h[:nv]) * dx_
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - best_h[:nv]) * dx_
                grad[:nv] += (-best_h[kk:]) * dx_
        grad /= len(active)
        
        # Rank indices by gradient
        idx_sorted = np.argsort(grad)
        low = idx_sorted[:max_pairs]
        high = idx_sorted[-max_pairs:]
        
        improved = False
        for delta in [0.005, 0.002, 0.001, 0.0005]:
            if time.time() > deadline:
                break
            
            for i in low:
                if time.time() > deadline:
                    return best_h, best_c
                for j in high:
                    if i == j:
                        continue
                    if best_h[i] + delta <= 1.0 + 1e-10 and best_h[j] - delta >= -1e-10:
                        best_h[i] += delta
                        best_h[j] -= delta
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= delta
                            best_h[j] += delta
                            best_h = project_feasible(best_h, n)
                    
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c

# --- SLSQP polish ---

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def sq_constraint(x, n):
    return x.sum() - n / 2.0

def sq_jacobian(x, n):
    return np.ones(n)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_constraint, 'jac': sq_jacobian, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
