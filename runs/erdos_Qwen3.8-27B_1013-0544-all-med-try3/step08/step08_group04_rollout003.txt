                if diff2 > 0.0:
                    add = 1.0 - v
                    idx = int(np.argmax(add))
                    v[idx] += min(diff2, add[idx])
                else:
                    idx = int(np.argmin(v))
                    v[idx] += max(diff2, -v[idx])
                v = np.clip(v, 0.0, 1.0)
    return v


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def compute_c5_arrays(h, omh, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, omh, mode="full")) * dx)


def lag_values(h):
    n = h.size
    dx = 2.0 / float(n)

    ac_full = np.correlate(h, h, mode="full")
    ac = ac_full[n - 1:]
    if ac.size != n:
        tmp = np.zeros(n, dtype=float)
        m = min(n, ac.size)
        tmp[:m] = ac[:m]
        ac = tmp

    cum = np.cumsum(h)
    prefix = cum[n - 1::-1]

    suffix = np.empty(n, dtype=float)
    suffix[0] = cum[-1]
    if n > 1:
        suffix[1:] = cum[-1] - cum[:-1]

    P = (prefix - ac) * dx
    N = (suffix - ac) * dx

    p0 = 0.5 * (float(P[0]) + float(N[0]))
    P[0] = p0
    N[0] = p0
    return P, N


def compute_smooth(h, alpha):
    P, N = lag_values(h)
    n = h.size
    dx = 2.0 / float(n)

    F = np.empty(2 * n - 1, dtype=float)
    F[0] = P[0]
    if n > 1:
        F[1:n] = P[1:n]
        F[n:2 * n - 1] = N[1:n]

    Fmax = float(np.max(F))
    base = F - Fmax
    exps = np.exp(alpha * base)
    s = float(np.sum(exps))

    if s <= 0.0 or not np.isfinite(s):
        weights = np.zeros_like(F)
        weights[int(np.argargmax(F) if False else np.argmax(F))] = 1.0
        f = Fmax
    else:
        weights = exps / s
        f = Fmax + float(np.log(s)) / alpha

    grad = np.zeros(n, dtype=float)

    w0 = float(weights[0]) * dx
    if w0 != 0.0:
        grad += w0 * (1.0 - 2.0 * h)

    if n > 1:
        for t in range(1, n):
            wp = float(weights[t]) * dx
            if wp != 0.0:
                m = n - t
                grad[:m] += wp * (1.0 - h[t:])
                grad[t:] -= wp * h[:m]

            wn = float(weights[n + t - 1]) * dx
            if wn != 0.0:
                m = n - t
                grad[t:] += wn * (1.0 - h[:m])
                grad[:m] -= wn * h[t:]

    return f, grad, Fmax


def smooth_fun(x, n, alpha):
    return compute_smooth(x, alpha)[0]


def smooth_jac(x, n, alpha):
    return compute_smooth(x, alpha)[1]


def sum_eq(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_eq_jac(x, n):
    return np.ones(x.size, dtype=float)


def slsqp_smooth(h, n, alpha, maxiter):
    h = project_feasible(h, n)
    c0 = compute_c5(h, n)

    try:
        bounds = [(0.0, 1.0)] * n
        cons = [
            {
                "type": "eq",
                "fun": sum_eq,
                "jac": sum_eq_jac,
                "args": (n,),
            }
        ]
        res = minimize(
            smooth_fun,
            h,
            args=(n, alpha),
            method="SLSQP",
            jac=smooth_jac,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-12},
        )
        out = np.asarray(res.x, dtype=float)
        out = project_feasible(out, n)
        c = compute_c5(out, n)
        return out, c
    except Exception:
        return h, c0


def projected_gradient(h, n, alpha, iters, deadline):
    h = project_feasible(h, n).copy()
    f, grad, c = compute_smooth(h, alpha)
    best_h = h.copy()
    best_c = c
    step = 0.5
    no_improve = 0

    for it in range(iters):
        if time.time() > deadline:
            break

        d = -grad
        d = d - float(np.mean(d))
        gd = float(np.dot(grad, d))
        if not np.isfinite(gd) or gd > -1e-14:
            break

        f_old = f
        s = step
        accepted = False

        for _ in range(35):
            cand = project_feasible(h + s * d, n)
            f2, g2, c2 = compute_smooth(cand, alpha)
            if np.isfinite(f2) and (f2 <= f_old + 1e-4 * s * gd + 1e-15):
                h = cand
                f = f2
                grad = g2
                c = c2
                step = s
                accepted = True
                if f2 < f_old - 1e-13:
                    no_improve = 0
                else:
                    no_improve += 1
                break
            s *= 0.5
            if s < 1e-14:
                break

        if not accepted:
            break

        if c < best_c:
            best_c = c
            best_h = h.copy()

        if no_improve > 30:
            break

    c_final = compute_c5(h, n)
    if c_final < best_c:
        return h, c_final
    if c_final <= best_c + 1e-10:
        return h, c_final
    return best_h, best_c


def pair_objective_omh(d, h, omh, n, i, j):
    oldi = float(h[i])
    oldj = float(h[j])
    ni = oldi + float(d)
    nj = oldj - float(d)

    if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
        return float("inf")

    h[i] = ni
    h[j] = nj
    omh[i] = 1.0 - ni
    omh[j] = 1.0 - nj

    val = compute_c5_arrays(h, omh, n)

    h[i] = oldi
    h[j] = oldj
    omh[i] = 1.0 - oldi
    omh[j] = 1.0 - oldj
    return val


def pair_grid_search(h, n, deadline, topB=80, passes=8, alpha=50000, grid=25):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)

    for _pass in range(passes):
        if time.time() > deadline:
            break

        _, grad, c_now = compute_smooth(h, alpha, )
        if c_now < best_c:
            best_c = c_now

        if n > 650:
            break

        diff = np.abs(np.subtract.outer(grad, grad))
        np.fill_diagonal(diff, -np.inf)
        flat = diff.reshape(-1)

        k = min(int(topB), flat.size)
        if k <= 0:
            break
        if k >= flat.size:
            idxs = np.arange(flat.size)
        else:
            idxs = np.argpartition(flat, flat.size - k)[flat.size - k:]
        idxs = idxs[np.argsort(flat[idxs])[::-1]]

        improved = False
        accepted = 0

        for idx in idxs:
            if time.time() > deadline:
                break

            i = int(idx // n)
            j = int(idx % n)
            if i == j:
                continue

            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-14:
                continue

            omh = 1.0 - h
            base_i = float(h[i])
            base_j = float(h[j])

            best_local = best_c
            best_d = 0.0
            step = (hi - lo) / float(grid)

            for m in range(grid + 1):
                d = lo + float(m) * step
                if abs(d) < 1e-15:
                    continue

                h[i] = base_i + d
                h[j] = base_j - d
                omh[i] = 1.0 - h[i]
                omh[j] = 1.0 - h[j]

                cv = compute_c5_arrays(h, omh, n)
                if cv < best_local:
                    best_local = cv
                    best_d = d

            h[i] = base_i
            h[j] = base_j
            omh[i] = 1.0 - base_i
            omh[j] = 1.0 - base_j

            if best_local >= best_c - 1e-13:
                continue

            m0 = int(round((best_d - lo) / (hi - lo) * float(grid)))
            width = (hi - lo) / float(grid)
            a = max(lo, best_d - width)
            b = min(hi, best_d + width)
            if b - a < 1e-12:
                a = max(lo, best_d - 1e-4)
                b = min(hi, best_d + 1e-4)
                if b - a < 1e-12:
                    continue

            try:
                res = minimize_scalar(
                    pair_objective_omh,
                    args=(h, omh, n, i, j),
                    bounds=(a, b),
                    options={"xatol": 1e-10, "maxiter": 40},
                )
                rd = float(res.x)
            except Exception:
                rd = float("nan")

            if not np.isfinite(rd):
                h[i] = base_i
                h[j] = base_j
                omh[i] = 1.0 - base_i
                omh[j] = 1.0 - base_j
                continue

            h[i] = base_i + rd
            h[j] = base_j - rd
            omh[i] = 1.0 - h[i]
            omh[j] = 1.0 - h[j]
            rc = compute_c5_arrays(h, omh, n)

            if rc < best_c - 1e-13:
                best_c = rc
                improved = True
                accepted += 1
                if accepted >= 10:
                    break
            else:
                h[i] = base_i
                h[j] = base_j

        if not improved:
            break

    h = project_feasible(h, n)
    c = compute_c5(h, n)
    return h, c


def pair_search_fixed(h, n, deadline, deltas, radius=None):
    h = project_feasible(h, n).copy()
    omh = 1.0 - h
    best_c = compute_c5_arrays(h, omh, n)

    if radius is None:
        if n <= 160:
            radius = max(1, n - 1)
        else:
            radius = max(20, n // 4)

    for d in deltas:
        improved = True
        passes = 0
        while improved and passes < 3 and time.time() < deadline:
            improved = False
            passes += 1
            for i in range(n):
                if time.time() > deadline:
                    break
                upper = min(n, i + radius + 1)
                for j in range(i + 1, upper):
                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        o1 = float(h[i])
                        o2 = float(h[j])
                        h[i] = o1 + d
                        h[j] = o2 - d
                        omh[i] = 1.0 - h[i]
                        omh[j] = 1.0 - h[j]
                        c = compute_c5_arrays(h, omh, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] = o1
                            h[j] = o2
                            omh[i] = 1.0 - o1
                            omh[j] = 1.0 - o2

                    if h[i] - d >= -1e-15 and h[j] + d <= 1.0 + 1e-15:
                        o1 = float(h[i])
                        o2 = float(h[j])
                        h[i] = o1 - d
                        h[j] = o2 + d
                        omh[i] = 1.0 - h[i]
                        omh[j] = 1.0 - h[j]
                        c = compute_c5_arrays(h, omh, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] = o1
                            h[j] = o2
                            omh[i] = 1.0 - o1
                            omh[j] = 1.0 - o2

                if time.time() > deadline:
                    break

    h = project_feasible(h, n)
    c = compute_c5(h, n)
    return h, c


def interpolate_to_n(h_src, n_new):
    n_src = len(h_src)
    if n_src == n_new:
        return h_src.copy()
    x_old = (np.arange(n_src) + 0.5) * (2.0 / float(n_src))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return np.interp(x_new, x_old, h_src)


def smooth_noise(rng, n, scale):
    x = np.arange(n, dtype=float)
    noise = np.zeros(n, dtype=float)
    nmodes = max(2, min(10, n // 20))
    for k in range(1, nmodes + 1):
        amp = float(scale) / float(k + 1)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        noise += amp * np.sin(2.0 * np.pi * float(k) * x / float(n) + phase)
    noise += 0.25 * float(scale) * rng.standard_normal(n)
    return noise


def update_best(h2, c2, best_h, best_c):
    if c2 < best_c - 1e-13:
        return h2.copy(), c2
    return best_h, best_c


def next_h(h2, c2, best_h, best_c, tol=1e-8):
    if c2 <= best_c + tol:
        return h2.copy()
    return best_h


def optimize_candidate(h, n, deadline, full=True):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    slsqp_allowed = n <= 360

    if full and slsqp_allowed and time.time() < deadline - 150:
        h2, c2 = slsqp_smooth(h, n, 2000, 200)
        best_h, best_c = update_best(h2, c2, best_h, best_c)
        h = next_h(h2, c2, best_h, best_c)

    if n <= 700:
        if full:
            settings = [(100, 180), (1000, 140), (10000, 120)]
        else:
            settings = [(200, 80), (5000, 80)]

        for a, it in settings:
            if time.time() > deadline - 80:
                break
            sub_deadline = min(deadline - 40, time.time() + (70 if full else 40))
            h2, c2 = projected_gradient(h, n, a, it, sub_deadline)
            best_h, best_c = update_best(h2, c2, best_h, best_c)
            h = next_h(h2, c2, best_h, best_c)

        if full and time.time() < deadline - 70:
            h2, c2 = projected_gradient(h, n, 40000, 80, min(deadline - 30, time.time() + 45))
            best_h, best_c = update_best(h2, c2, best_h, best_c)
            h = next_h(h2, c2, best_h, best_c)

    if n <= 650 and time.time() < deadline - 60:
        h2, c2 = pair_grid_search(
            h,
            n,
            deadline - 30,
            topB=80 if full else 40,
            passes=8,
            alpha=50000,
        )
        best_h, best_c = update_best(h2, c2, best_h, best_c)
        h = next_h(h2, c2, best_h, best_c)

    if slsqp_allowed and time.time() < deadline - 80:
        h2, c2 = slsqp_smooth(h, n, 50000, 250)
        best_h, best_c = update_best(h2, c2, best_h, best_c)
        h = next_h(h2, c2, best_h, best_c)

    if full and slsqp_allowed and time.time() < deadline - 60:
        h2, c2 = slsqp_smooth(h, n, 200000, 200)
        best_h, best_c = update_best(h2, c2, best_h, best_c)
        h = next_h(h2, c2, best_h, best_c)

    if n <= 650 and time.time() < deadline - 35:
        h2, c2 = pair_grid_search(
            h,
            n,
            deadline - 20,
            topB=60,
            passes=6,
            alpha=100000,
        )
        best_h, best_c = update_best(h2, c2, best_h, best_c)
        h = next_h(h2, c2, best_h, best_c)

    if time.time() < deadline - 15:
        h2, c2 = pair_search_fixed(
            h,
            n,
            deadline - 5,
            [1e-4, 5e-5, 2e-5, 1e-5],
            radius=max(20, n // 4),
        )
        best_h, best_c = update_best(h2, c2, best_h, best_c)
        h = next_h(h2, c2, best_h, best_c)

    if slsqp_allowed and time.time() < deadline - 12:
        h2, c2 = slsqp_smooth(h, n, 200000, 150)
        best_h, best_c = update_best(h2, c2, best_h, best_c)
        h = next_h(h2, c2, best_h, best_c)

    if time.time() < deadline - 5:
        h2, c2 = pair_search_fixed(
            h,
            n,
            deadline - 2,
            [5e-6, 2e-6, 1e-6],
            radius=max(20, n // 5),
        )
        best_h, best_c = update_best(h2, c2, best_h, best_c)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = int(len(h_init))
    except Exception:
        n_init = 250
        h_init = np.full(n_init, 0.5, dtype=float)

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)

    if n_init > 600:
        n_start = 600
        h_start = interpolate_to_n(h_init, n_start)
    else:
        n_start = n_init
        h_start = h_init

    alloc1 = min(420.0, max(180.0, (deadline - t0) * 0.45))
    h1, c1 = optimize_candidate(h_start, n_start, t0 + alloc1, full=True)
    best_h, best_c = update_best(h1, c1, best_h, best_c)

    targets = []
    used = {n_start: True}
    base_n = max(1, len(best_h))
    for factor in (1.08, 1.25, 1.5):
        m = int(round(float(base_n) * factor))
        m = max(120, min(600, m))
        if m not in used:
            targets.append(m)
            used[m] = True

    for idx, m in enumerate(targets):
        if time.time() > deadline - 250:
            break
        source = best_h if idx % 2 == 0 else h_init
        h_try = interpolate_to_n(source, m)
        h_try = project_feasible(h_try + smooth_noise(rng, m, 0.02 + 0.01 * idx), m)

        alloc = min(110.0, max(35.0, (deadline - time.time()) * 0.18))
        if time.time() + alloc > deadline - 80:
            alloc = max(20.0, deadline - time.time() - 80)
        if alloc < 25:
            break

        h2, c2 = optimize_candidate(h_try, m, time.time() + alloc, full=False)
        best_h, best_c = update_best(h2, c2, best_h, best_c)

    n_best = len(best_h)
    if time.time() < deadline - 150:
        for r in range(3):
            if time.time() > deadline - 90:
                break
            scale = 0.05 + 0.04 * r
            if r == 2:
                hp = project_feasible(0.5 + 0.25 * smooth_noise(rng, n_best, 1.0), n_best)
            else:
                hp = project_feasible(best_h + smooth_noise(rng, n_best, scale), n_best)

            alloc = min(75.0, max(25.0, (deadline - time.time() - 60.0) * 0.25))
            if time.time() + alloc > deadline - 60:
                alloc = max(20.0, deadline - time.time() - 60)
            if alloc < 20:
                break

            h2, c2 = optimize_candidate(hp, n_best, time.time() + alloc, full=False)
            best_h, best_c = update_best(h2, c2, best_h, best_c)
            n_best = len(best_h)

    n_best = len(best_h)
    if time.time() < deadline - 30:
        h2, c2 = pair_grid_search(
            best_h,
            n_best,
            deadline - 15,
            topB=120,
            passes=10,
            alpha=100000,
        )
        best_h, best_c = update_best(h2, c2, best_h, best_c)

    if time.time() < deadline - 15:
        h2, c2 = pair_search_fixed(
            best_h,
            len(best_h),
            deadline - 5,
            [5e-6, 2e-6, 1e-6],
            radius=max(20, len(best_h) // 4),
        )
        best_h, best_c = update_best(h2, c2, best_h, best_c)

    if len(best_h) <= 360 and time.time() < deadline - 12:
        h2, c2 = slsqp_smooth(best_h, len(best_h), 200000, 200)
        best_h, best_c = update_best(h2, c2, best_h, best_c)

    if time.time() < deadline - 5:
        h2, c2 = pair_search_fixed(
            best_h,
            len(best_h),
            deadline - 2,
            [2e-6, 1e-6],
            radius=max(20, len(best_h) // 5),
        )
        best_h, best_c = update_best(h2, c2, best_h, best_c)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
