from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_func(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()

        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_func(x, n, alpha, mu):

    m = c.max()

    smx = sh.max()

        if wi < 1e-15:

        k = mm - (n - 1)
        wm = wi * dx_val
        if k >= 0:

            grad[:nv] += wm * (1.0 - x[k:])

- dx = 2.0 / n_points
- sum(h) * dx = 1, i.e., sum(h) = n / 2

When I interpolate, I should use:
x = np.arange(n) * dx = np.arange(n) * 2.0/n

This is equivalent to np.linspace(0, 2, n, endpoint=False).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (target - s) / tot * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (s - target) / tot * h
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_func(x, n, alpha, mu):
    dx_val = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx_val
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wi = w[mm]
        if wi < 1e-15:
            continue
        k = mm - (n - 1)
        wm = wi * dx_val
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, top_idx, alpha, mu):
    dx_val = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx_val
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)

            cur_c = new_c
            if cur_c < best_c - 1e-16:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c

def fine_pair_search(h, n, deadline, max_radius=50):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    
    for delta in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        if time.time() > deadline:
            break
        c_vec = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_idx = int(np.argmax(c_vec))
        k_max = max_idx - (n - 1)
        if k_max >= 0:
            center_lo = max(0, k_max - max_radius)
            center_hi = min(n, n - k_max + max_radius)
        else:
            kk = -k_max
            center_lo = max(0, kk - max_radius)
            center_hi = min(n, n - kk + max_radius)
        
        passes = 0
        while passes < 5:
            if time.time() > deadline:
                break
            improved = False
            passes += 1
            for i in range(center_lo, center_hi):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + delta <= 1.0 and best_h[j] - delta >= 0.0:
                        best_h[i] += delta
                        best_h[j] -= delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= delta
                            best_h[j] += delta
                    if best_h[i] - delta >= 0.0 and best_h[j] + delta <= 1.0:
                        best_h[i] -= delta
                        best_h[j] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += delta
                            best_h[j] -= delta
            if not improved:
                break
    return best_h, best_c

def multi_resolution(h_best, c_best, n_best, deadline):
    """Try different resolutions and pick the best."""
    results = [(h_best.copy(), c_best, n_best)]
    
    for n_try in [750, 1000, 1500]:
        if time.time() > deadline - 20:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_feasible(h_try, n_try)
        c_try = compute_c5(h_try, n_try)
        results.append((h_try, c_try, n_try))
    
    best_r = results[0]
    for r in results[1:]:
        if r[1] < best_r[1]:
            best_r = r
    return best_r[0], best_r[1], best_r[2]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_orig = 400
    
    h = project_feasible(h_init, n_orig)
    best_h = h.copy()
    best_c = compute_c5(h, n_orig)
    
    # Phase 1: Quick optimization at parent resolution (25s)
    if time.time() < deadline - 700:
        t1 = min(time.time() + 25, deadline - 700)
        h, c = lbfgs_opt(h, n_orig, t1,
                         [500, 5000, 50000, 500000], [50, 500, 5000, 50000])
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Multi-resolution selection (10s)
    n_work = 1000
    x_old = np.linspace(0, 2, len(best_h), endpoint=False)
    x_new = np.linspace(0, 2, n_work, endpoint=False)
    h_work = np.interp(x_new, x_old, best_h)
    h_work = project_feasible(h_work, n_work)
    c_work = compute_c5(h_work, n_work)
    if c_work < best_c:
        best_c = c_work
        best_h = h_work.copy()
    else:
        best_h = h_work.copy()
        best_c = max(best_c, c_work)
    
    # Also try n=1500
    if time.time() < deadline - 600:
        n_alt = 1500
        x_alt = np.linspace(0, 2, n_alt, endpoint=False)
        h_alt = np.interp(x_alt, x_old, best_h[:len(x_old)] if len(best_h) == len(x_old) else best_h)
        x_src = np.linspace(0, 2, len(h_alt) if False else n_work, endpoint=False)
        h_alt = np.interp(x_alt, x_src, h_work)
        h_alt = project_feasible(h_alt, n_alt)
        c_alt = compute_c5(h_alt, n_alt)
        if c_alt < c_work - 0.0001:
            n_work = n_alt
            best_h = h_alt.copy()
            best_c = c_alt
    
    # Phase 3: Main L-BFGS at working resolution (280s)
    if time.time() < deadline - 400:
        t3 = min(time.time() + 280, deadline - 400)
        alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
        mus = [10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
        best_h, best_c = lbfgs_opt(best_h, n_work, t3, alphas, mus, maxiter=5000)
    
    # Phase 4: Top-shifts focused (180s)
    if time.time() < deadline - 250:
        t4 = min(time.time() + 180, deadline - 250)
        best_h, best_c = top_shifts_opt(best_h, n_work, t4, top_k=15)
    
    # Phase 5: Simulated annealing (250s)
    if time.time() < deadline - 220:
        t5 = min(time.time() + 250, deadline - 220)
        h_sa, c_sa = simulated_annealing(best_h, n_work, t5)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
    
    # Phase 6: L-BFGS polish after SA (130s)
    if time.time() < deadline - 100:
        t6 = min(time.time() + 130, deadline - 100)
        best_h, best_c = lbfgs_opt(best_h, n_work, t6,
                                   [100000, 500000, 2000000, 10000000],
                                   [5000, 20000, 100000, 500000], maxiter=5000)
    
    # Phase 7: Top-shifts round 2 (80s)
    if time.time() < deadline - 60:
        t7 = min(time.time() + 80, deadline - 60)
        best_h, best_c = top_shifts_opt(best_h, n_work, t7, top_k=5)
    
    # Phase 8: Subgradient descent (40s)
    if time.time() < deadline - 30:
        t8 = min(time.time() + 40, deadline - 30)
        best_h, best_c = subgrad_descent(best_h, n_work, t8, n_steps=20000)
    
    # Phase 9: Fine pair search (40s)
    if time.time() < deadline - 5:
        t9 = min(time.time() + 40, deadline - 5)
        best_h, best_c = fine_pair_search(best_h, n_work, t9, max_radius=30)
    
    # Final
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
