    """Compute new correlation after swapping h[p]:0→1 and h[q]:1→0"""
    # delta[k] for k in 0..2n-2, corresponding to shift d = k-(n-1)
    delta = np.zeros(2*n - 1)
    # p: 0→1 contribution: +(1-h[p+d]) at corr index n-1+d, if 0≤p+d<n
    #                                   -h[p-d] at corr index n-1+d, if 0≤p-d<n
    # q: 1→0 contribution: -(1-h[q+d]) at corr index n-1+d, if 0≤q+d<n
    #                                    +h[q-d] at corr index n-1+d, if 0≤q-d<n
    
    # Vectorized computation:
    # For p: 0→1
    # (1-h[p+d]) for d = -p, ..., n-1-p (where p+d is in [0,n))
    # This goes into corr indices n-1-p, ..., n-1+n-1-p = 2n-2-p
    d_range_p = np.arange(-p, n - p)  # values of d
    idx_p = n - 1 + d_range_p  # corr indices

    pd_vals = p + d_range_p  # = 0, 1, ..., n-1
    delta[idx_p] += (1 - h[pd_vals])

    # p-d in [0,n) → d in (p-n, p] → d = p-n+1, ..., p
    d_range_p2 = np.arange(p - n + 1, p + 1)
    idx_p2 = n - 1 + d_range_p2
    pd_vals2 = p - d_range_p2  # = p-(p-n+1), ..., p-0 = n-1, ..., p
    delta[idx_p2] -= h[pd_vals2]
    
    # For q: 1→0 (negate the above with p→q)
    d_range_q = np.arange(-q, n - q)
    idx_q = n - 1 + d_range_q
    qd_vals = q + d_range_q
    delta[idx_q] -= (1 - h[qd_vals])

    idx_q2 = n - 1 + d_range_q2
    qd_vals2 = q - d_range_q2
    delta[idx_q2] += h[qd_vals2]
    
    return delta
