import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    # Project onto {0 <= h <= 1, sum(h) = n/2}
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_gradient_active(h, n, active_idx):
    """Compute gradient of max over active set (subgradient = mean of active gradients)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    n_active = len(active_idx)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    
    grad /= n_active
    return grad

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def active_max_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(shifted - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def active_max_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    w = np.exp(shifted - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])

        else:
            kk = -k

    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = 2000
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def run_subgradient(h, n, deadline, n_iters=10000):
    """Direct subgradient method with adaptive step size"""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    eta = 0.01
    eta_min = 1e-6
    eta_max = 0.05
    patience = 0
    
    for it in range(n_iters):
        if time.time() > deadline:

        active = np.where(c_vec >= c_max - tol)[0]

def random_eta():
    return 0.005 * (0.5 + np.random.random())

def fine_local_search(h, n, deadline):

    best_h = project_feasible(h.copy(), n)

            improved = False
            # Try moving mass from j to i

                if time.time() > deadline:
                    break
                if best_h[i] + step > 1.0:
                    continue
                for j in range(n):

                        continue
                    # Only try if j is far from i (to find meaningful changes)

                        continue
                    
                    old_i, old_j = best_h[i], best_h[j]
                    best_h[i] = old_i + step
                    best_h[j] = old_j - step
                    
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] = old_i
                        best_h[j] = old_j
                    
                    # Also try reverse
                    if best_h[i] - step >= 0.0 and best_h[j] + step <= 1.0:

                            best_c = c

def upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new points"""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old/2  # cell centers
    # Actually, let's just use standard grid
    x_old = np.arange(n_old) * (2.0/n_old)
    x_new = np.arange(n_new) * (2.0/n_new)
    h_new = np.interp(x_new, x_old, h, period=2.0) if False else np.interp(x_new, x_old, h)
    # Actually np.interp doesn't support period. Let's just do linear interp
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_new)
    return h_new

def downsample(h, n_old, n_new):
    """Downsample h from n_old to n_new points (average pooling)"""
    if n_old < n_new:
        return h.copy()
    ratio = n_old / n_new
    h_new = np.zeros(n_new)
    for i in range(n_new):
        lo = int(i * ratio)
        hi = int((i+1) * ratio)
        if hi <= lo:
            hi = lo + 1
        h_new[i] = h[lo:hi].mean()
    h_new = project_feasible(h_new, n_new)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(shifted)] = 1.0
        w_sum = 1.0
    w /= w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def active_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(shifted - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    w = np.exp(shifted - smx)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(active))
        w[np.argmax(shifted)] = 1.0
        w_sum = 1.0
    w /= w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = 3000
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-24,
                                   'gtol': 1e-18, 'maxfun': maxit * 8,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

        if len(active) > top_k:

        if len(active) == 0:

                              options={'maxiter': maxit, 'ftol': 1e-24,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    return best_h, best_c

def run_subgradient(h, n, deadline, n_iters=50000):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    eta = 0.01
    eta_min = 1e-7
    eta_max = 0.05
    no_improve = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.003 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:]) * dx
                grad[k:] += (-best_h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - best_h[:nv]) * dx
                grad[:nv] += (-best_h[kk:]) * dx
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.02, eta_max)
            no_improve = 0
        else:
            no_improve += 1
            eta = max(eta * 0.97, eta_min)
        if no_improve > 300:
            noise_scale = 0.0005 * (no_improve // 300)
            noise = np.random.randn(n) * noise_scale
            h_try = project_feasible(best_h + noise, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_h = h_try
                best_c = c_try
            eta = 0.01 * (0.5 + np.random.random())
            no_improve = 0
    return best_h, best_c

def coord_descent_active(h, n, deadline, n_iters=5000):

    best_c = compute_c5(best_h, n)
    dx_val = 2.0 / n
    for it in range(n_iters):

            active = np.array([np.argmax(c_vec)])
        # Compute subgradient
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:]) * dx_val
                grad[k:] += (-best_h[:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - best_h[:nv]) * dx_val
                grad[:nv] += (-best_h[kk:]) * dx_val
        grad /= len(active)
        # Find best coordinate to move
        best_step = 0
        best_delta = 0
        best_idx = -1
        # Try moving each coordinate in -gradient direction
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        g_dir = grad / gnorm
        # Try a few step sizes along -g_dir projected onto feasible set
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:

    return h_new

            best_c = c7
            best_h = h7.copy()
    
    # Phase 8: Final subgradient (30s)
    if time.time() < deadline - 10:
        p8_end = deadline - 3
        h8, c8 = run_subgradient(best_h, best_n, p8_end, n_iters=10000)
        if c8 < best_c - 1e-14:
            best_c = c8
            best_h = h8.copy()
    
    # Final projection and verification
    fn = len(best_h)
    best_h = project_feasible(best_h, fn)
    best_c = compute_c5(best_h, fn)
    
    return best_h, best_c, fn
