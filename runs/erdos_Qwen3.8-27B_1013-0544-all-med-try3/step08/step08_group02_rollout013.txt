def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    alphas = [20, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    h_best = h_new.copy()
    c_best = c_new
    for a in alphas:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = min(1500, max(200, int(remaining * 3)))
        h_trial = do_lbfgs(h_best, n_new, a, 0.003, maxit, bounds_full)
        c_trial = compute_c5(h_trial, n_new)
        if c_trial < c_best:
            c_best = c_trial
            h_best = h_trial
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(1000, 0.5)
        n_orig = 1000

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick convergence at original resolution
    p1_end = t0 + 60
    bounds_orig = [(0.0, 1.0)] * n_orig
    for a in [100, 1000, 10000, 100000]:
        if time.time() > p1_end:
            break
        remaining = p1_end - time.time()
        if remaining < 5:
            break
        maxit = min(800, max(100, int(remaining * 4)))
        h1 = do_lbfgs(best_h, best_n, a, 0.01, maxit, bounds_orig)
        c1 = compute_c5(h1, best_n)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upsample to n=1600 and optimize
    n2 = 1600
    bounds2 = [(0.0, 1.0)] * n2
    h2, c2 = try_upsample(best_h, best_n, n2, t0 + 300, bounds2)
    if c2 < best_c:
        best_c = c2
        best_h = h2
        best_n = n2

    # Phase 3: More L-BFGS at n2 with higher alphas
    p3_end = t0 + 500
    for a in [100000, 500000, 2000000, 10000000]:
        if time.time() > p3_end:
            break
        remaining = p3_end - time.time()
        if remaining < 10:
            break
        maxit = min(1200, max(200, int(remaining * 2)))
        h3 = do_lbfgs(best_h, best_n, a, 0.001, maxit, bounds2)
        c3 = compute_c5(h3, best_n)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Alternating active-set optimization
    p4_end = t0 + 700
    h4, c4 = alternating_active(best_h, best_n, p4_end, bounds2)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: SLSQP passes
    p5_end = t0 + 850
    for a in [100000, 500000, 2000000, 10000000]:
        if time.time() > p5_end:
            break
        remaining = p5_end - time.time()
        if remaining < 15:
            break
        maxit = min(1500, max(300, int(remaining * 3)))
        h5 = do_slqp(best_h, best_n, a, maxit, bounds2)
        c5 = compute_c5(h5, best_n)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Gradient step refinement
    p6_end = t0 + 970
    h6, c6 = grad_step_refine(best_h, best_n, p6_end)
    if c6 < best_c:
        best_c = c6
        best_h = h6

    # Phase 7: Local pair search (fine)
    p7_end = t0 + 1060
    h7, c7 = local_pair_search(best_h, best_n, p7_end, radius=20)
    if c7 < best_c:
        best_c = c7
        best_h = h7

    # Phase 8: Final SLSQP
    if time.time() < deadline - 3:
        h8 = do_slqp(best_h, best_n, 5000000, 500, bounds2)
        c8 = compute_c5(h8, best_n)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Very short gradient polish
    if time.time() < deadline - 1:
        h9, c9 = grad_step_refine(best_h, best_n, deadline)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
