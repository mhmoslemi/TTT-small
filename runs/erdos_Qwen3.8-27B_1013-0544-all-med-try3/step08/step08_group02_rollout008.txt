```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, active=None, mu=0.0):
    c = compute_c5_vec(x, n)
    if active is not None:
        c = c[active]
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, active=None, mu=0.0):
    c_full = compute_c5_vec(x, n)
    if active is not None:
        c = c_full[active]
    else:
        c = c_full
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    indices = active if active is not None else np.arange(len(c_full))
    for wi, idx in zip(w, indices):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

        remaining = deadline - time.time()
        if remaining < 10:

            maxit = maxiters[i]
        else:

        try:

                          args=(n, alpha, None, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c
```

```python
import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14 * max(1, n):
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, active, mu):
    c = compute_c5_vec(x, n)
    if active is not None:
        c = c[active]
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + (np.log(np.sum(np.exp(sh)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, active, mu):
    c_full = compute_c5_vec(x, n)
    if active is not None:
        c = c_full[active]
        idxs = active
    else:
        c = c_full
        idxs = np.arange(len(c_full))
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wval, idx in zip(w, idxs):
        if wval < 1e-15:
            continue
        k = int(idx) - (n - 1)
        wm = wval * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    for i, alpha in enumerate(alphas):

            break
        if i < len(maxit_per):

        else:
            maxit = max(500, int(rem * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, None, mu), method='L-BFGS-B',

                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def binary_project(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    if n % 2 == 0:
        n_ones = n // 2
    else:
        n_ones = (n - 1) // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] += remaining
    return project_feasible(h_bin, n)

def pair_search(h, n, deadline, radius=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, max(1, n // 500)):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if abs(best_h[i] - best_h[j]) < 1e-12:
                        continue
                    src = i if best_h[i] > best_h[j] else j
                    dst = j if best_h[i] > best_h[j] else i
                    if best_h[src] - d < 0 or best_h[dst] + d > 1:
                        continue
                    test_h = best_h.copy()
                    test_h[src] -= d
                    test_h[dst] += d
                    c = compute_c5(test_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        best_h = test_h
                        improved = True
                    else:
                        test_h = best_h.copy()

                        test_h[dst] -= d
                        if test_h[src] > 1 or test_h[dst] < 0:
                            continue
                        c = compute_c5(test_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = test_h
                            improved = True
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003

        cm = cv.max()
        tol = max(1e-9, 0.0005 * cm)
        active = np.where(cv >= cm - tol)[0]
        if len(active) == 0:

        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx

                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        h_try = best_h - eta * grad
        h_try = project_feasible(h_try, n)

            best_h = h_try.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
            best_h = project_feasible(best_h, n)
    return best_h, best_c

        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        p4_end = min(time.time() + 80, deadline - 100)
        h4, c4 = phase_lbfgs(h_bin, best_n, p4_end,
            alphas=[100, 1000, 10000, 100000, 500000, 2000000],
            mu=0.01,
            maxit_per=[1000, 1500, 2000, 2000, 2000, 2000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = best_n

    # Phase 5: Perturbation + re-optimize (escape local optima)
    if time.time() < deadline - 100:
        for trial in range(3):
            if time.time() > deadline - 80:
                break
            noise_scale = [0.05, 0.02, 0.01][trial]
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_n)
            c_pert = compute_c5(h_pert, best_n)
            p5_end = min(time.time() + 40, deadline - 60)
            h5, c5 = phase_active_adaptive(h_pert, best_n, p5_end,
                top_k=6, alpha=200000, maxit=800)
            if c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = best_n

    # Phase 6: Subgradient refinement
    if time.time() < deadline - 40:
        p6_end = min(time.time() + 30, deadline - 20)
        h6, c6 = subgrad_refine(best_h, best_n, p6_end, n_iter=5000)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = best_n

    # Phase 7: Pair search (fine local search)
    if time.time() < deadline - 5:
        p7_end = deadline - 2
        h7, c7 = pair_search(best_h, best_n, p7_end, radius=50)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = best_n

    # Final: project and verify
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
```
