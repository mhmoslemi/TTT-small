def fft_correlate(a, b):
    """Compute np.correlate(a, b, 'full') using FFT"""
    n = len(a)
    m = len(b)
    size = n + m - 1
    # Pad to next power of 2
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.fft(a, fft_size)
    fb = np.fft.fft(b, fft_size)
    # Cross-correlation: ifft(fa * conj(fb))
    result = np.fft.ifft(fa * np.conj(fb)).real[:size]
    return result

    deadline = t0 + budget_s - 3

    # Load initial solution
    try:
        h = np.array(initial_h_values, dtype=float).copy()

    except Exception:
        h = np.full(300, 0.5)
        n = 300

    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: Multi-alpha L-BFGS-B warmup
    alphas_warm = [50, 200, 1000, 5000]
    for a in alphas_warm:
        if time.time() > deadline - 500:

        t_alloc = min(40, deadline - time.time() - 400)
        if t_alloc < 15:

        h_new, c_new = lfgs_optimize(h, n, a, time.time() + t_alloc, maxiter=200, mu=1e-3)
        if c_new < best_c:

            best_h = h_new.copy()
        h = h_new

    alphas_main = [10000, 50000, 200000, 500000, 2000000]

        t_alloc = min(80, deadline - time.time() - 300)

            break
        h_new, c_new = lfgs_optimize(h, n, a, time.time() + t_alloc, maxiter=400, mu=1e-5)
        if c_new < best_c:
            best_c = c_new

        h = h_new
    
    # Phase 3: SLSQP for exact constraint handling
    if time.time() < deadline - 250:
        for a in [100000, 1000000, 5000000]:
            if time.time() > deadline - 200:

            h_new, c_new = slsqp_optimize(h, n, a, time.time() + t_alloc, maxiter=800)

            h = h_new

    if time.time() < deadline - 200:

    if time.time() < deadline - 80:
        h400, c400 = try_resolution(best_h, n, 400, deadline - 40, seed=seed + 1)

            best_h = h_final.copy()

    best_h = project_feasible(best_h, len(best_h))
