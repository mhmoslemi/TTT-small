    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    idx = np.argsort(h)[::-1]

def binary_local_search_2opt(h, n, deadline, max_sweeps=500):
    """Local search with both 1-opt (single swap) and 2-opt (double swap) moves."""
    dx = 2.0 / n
    g = compute_all_g(h, n)
    best_c = float(np.max(g) * dx)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        g = compute_all_g(h, n)
        cur_c = float(np.max(g))
        
        # 1-opt: try all single swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_delta = 0
        best_move = None
        
        for j in zeros:
            for i in ones:
                # Swap: h[i] 1->0, h[j] 0->1
                h[i] = 0
                h[j] = 1
                g_new = np.correlate(h, 1-h, 'full')
                new_max = float(np.max(g_new))
                delta = new_max - cur_c
                if delta < best_delta:
                    best_delta = delta
                    best_move = (i, j)
                h[i] = 1
                h[j] = 0
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0
            h[j] = 1
            g = compute_all_g(h, n)
            cur_c = float(np.max(g))
            improved = True
            if cur_c + dx * 0 < best_c:
                best_c = cur_c * dx
                best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c

The key is to be smart about WHICH swaps to evaluate. Here's my strategy:

1. Compute current g = correlate(h, 1-h, 'full')
2. Find the top K lags (say K=10) where g is at maximum
3. For each top lag k, identify the "problem" positions:
   - m where h[m]=1 and 1-h[m+k-(n-1)]=1, i.e., h[m]=1 and h[m+k-(n-1)]=0
   - These are the positions contributing to the high g[k]
4. For each problem position m, consider moving the 1 from m to some other position j where h[j]=0:
   - Prefer j where h[j+k-(n-1)]=0 (won't add to this lag)
   - Prefer j where h[j-lag+(n-1)]=1 for other hot lags (won't create new peaks)
5. Try the most promising few swaps and keep the best

This should be much faster than trying all O(n²) swaps.

def focused_binary_search(h, n, deadline, k_hot=5):

        cur_max = g.max()
        
        # Find hot lags
        top_k = np.argsort(g)[::-1][:k_hot]
        
        # Find all candidate swaps that might help
        candidates = []
        for k in top_k:
            # Problem positions: h[m]=1 and h[m+k-(n-1)]=0 (valid)
            lo = max(0, n-1-k)
            hi = min(n, n-1-k+n)  # = min(n, 2n-1-k) = 2n-1-k if k<n-1
            # m ranges from lo to min(n-1, 2n-2-(k-(n-1))) = min(n-1, n-1) = n-1
            # So m from max(0, n-1-k) to n-1
            for m in range(lo, n):
                p = m + k - (n-1)
                if 0 <= p < n and h[m] == 1 and h[p] == 0:
                    # Moving 1 from m: reduces g[k] by 1
                    # Adding 1 at j: changes g[k] by (1 if h[j+k-(n-1)]=0 else 0)
                    candidates.append((m, k))
        
        # For each problem position m, find best target j
        best_swap = None
        best_new_max = cur_max
        
        # Sample a subset of candidates for efficiency
        if len(candidates) > 200:
            candidates = np.random.choice(len(candidates), 200, replace=False)
            candidates = [candidates[i] for i in range(len(candidates))]
        
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
        
        for m in set(m for m, k in candidates):
            # Try moving 1 from m to various zeros
            # Prefer zeros that are "inert" for the hot lags
            best_j_for_m = None
            for j in zeros:
                if j == m:
                    continue
                # Quick check: will this swap help the main hot lag?
                # Compute delta for the main hot lag k0 = top_k[0]
                k0 = top_k[0]
                p1 = m + k0 - (n-1)
                p2 = j + k0 - (n-1)
                # Old contribution of m to g[k0]: (1 if h[m]=1, h[p1]=0) = 1
                # New contribution of m: 0
                # Old contribution of j: 0 (h[j]=0)
                # New contribution of j: 1 if h[p2]=0
                delta_k0 = -1 + (1 if (0 <= p2 < n and h[p2] == 0) else 0)
                if delta_k0 >= 0:
                    continue  # This swap doesn't help the main lag
                
                # Full evaluation
                h[m] = 0
                h[j] = 1
                g_new = np.correlate(h, 1-h, 'full')
                new_max = g_new.max()
                h[m] = 1
                h[j] = 0
                
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (m, j)
        
        if best_swap is not None:
            m, j = best_swap
            h[m] = 0
            h[j] = 1
            new_max = np.correlate(h, 1-h, 'full').max()
            if new_max * dx < best_c:
                best_c = new_max * dx
                best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 50:
                # Perturbation: random swap
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                h[i] = 0
                h[j] = 1
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                no_improve_count = 0
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)


def to_binary(h, n):
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def slsqp_smoothed(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h


def multi_resolution_search(h_best, c_best, n_best, rng, deadline):
    for nh in [int(n_best * 0.8), int(n_best * 1.2), int(n_best * 1.5)]:
        if time.time() > deadline - 60:
            break
        if nh <= 50 or nh > 1500:
            continue
        if nh % 2 != 0:
            nh += 1
        if nh == n_best:
            continue
        # Interpolate
        xo = np.linspace(0, 2, n_best)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, h_best)
        hh = project_feasible(hh, nh)
        # Binary round
        hb = to_binary(hh, nh)
        # Quick local search (limited time)
        sub_deadline = min(deadline - 30, time.time() + 60)
        hb, cb = greedy_binary_search(hb, nh, sub_deadline, k_hot=6)
        if cb < c_best:
            c_best = cb
            h_best = hb.copy()
            n_best = nh
    return h_best, c_best, n_best
