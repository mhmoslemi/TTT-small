def swap_update(h, n, i, j, c_vals):
    """
    Update c_vals after swapping h[i]=1 to 0 and h[j]=0 to 1.
    i is currently 1, j is currently 0.
    Returns updated c_vals and new h.
    """
    dx = 2.0 / n
    # Don't modify h yet, just compute delta
    for L in range(-(n-1), n):
        if L == 0:
            continue
        delta = 0.0
        k_idx = L + (n - 1)  # index in 'full' output
        # i: 1 -> 0
        # Term where n+L = i, i.e., n = i-L
        n_val = i - L
        if 0 <= n_val < n and 0 <= i < n:
            # Valid pair (n_val, i) at lag L
            a = max(0, -L)
            b = n - 1 - max(0, L)
            if a <= n_val <= b:
                delta -= (1.0 - h[n_val]) * dx
        # Term where n = i
        if 0 <= i < n:
            a = max(0, -L)
            b = n - 1 - max(0, L)
            if a <= i <= b:
                err = i + L
                if 0 <= err < n:
                    delta += h[err] * dx
        # j: 0 -> 1
        # Term where n+L = j, i.e., n = j-L
        n_val = j - L
        if 0 <= n_val < n:
            a = max(0, -L)
            b = n - 1 - max(0, L)
            if a <= n_val <= b:
                delta += (1.0 - h[n_val]) * dx
        # Term where n = j
        if 0 <= j < n:
            a = max(0, -L)
            b = n - 1 - max(0, L)
            if a <= j <= b:
                err = j + L
                if 0 <= err < n:
                    delta -= h[err] * dx
        c_vals[k_idx] += delta
    
    h[i] = 0.0
    h[j] = 1.0
    return h, c_vals
