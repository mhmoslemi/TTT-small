```python
def project_feasible(h, n):

        if abs(diff) < 1e-16:
            break
        abs_diff = abs(diff)
        if abs_diff < 1e-18:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-18:
                break
            h += (diff / max(tot, 1e-30)) * room
        else:
            tot = h.sum()
            if tot < 1e-18:
                break
            h += (diff / max(tot, 1e-30)) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 5e-15:
        # Distribute to non-boundary elements

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
def sca_iteration(h0, n):
    """One SCA iteration: solve LP to get new h."""
    from scipy.optimize import linprog
    
    # Build constraint matrix
    # Variables: [h_0, ..., h_{n-1}, t] (n+1 variables)
    nx = n + 1
    
    # Objective: minimize t
    c_obj = np.zeros(nx)
    c_obj[n] = 1.0  # coefficient of t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag m (0 to 2n-2):
    #   dx * (a_m^T h) - t <= -dx * C_m
    # where a_m is the gradient vector of c_m at h0
    
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, nx))
    b_ub = np.zeros(num_constraints)
    dx = 2.0 / n
    
    for m in range(num_constraints):
        k = m - (n - 1)
        # Compute a_m = gradient of c_k at h0
        # a_m[j] = (1-h0[j+k])*1{j in I_k} - h0[j-k]*1{j-k in I_k}
        am = np.zeros(n)
        if k >= 0:
            nv = n - k
            am[0:nv] = (1.0 - h0[k:n])  # (1-h0[j+k]) for j in [0, nv-1]
            if k > 0:
                am[k:n] -= h0[0:nv]  # -h0[j-k] for j in [k, n-1]
        else:
            kk = -k
            nv = n - kk
            am[kk:n] = (1.0 - h0[0:nv])  # (1-h0[j-kk]) for j in [kk, n-1]
            if kk > 0:
                am[0:nv] -= h0[kk:n]  # -h0[j+kk] for j in [0, nv-1]
        
        # Constant term C_k
        C_k = 0.0
        if k >= 0:
            C_k = np.dot(h0[0:nv], h0[k:n])
        else:
            C_k = np.dot(h0[kk:n], h0[0:nv])
        
        A_ub[m, 0:n] = dx * am
        A_ub[m, n] = -1.0
        b_ub[m] = -dx * C_k
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, nx))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h <= 1, t is free (but we can bound it)
    bounds = [(0.0, 1.0)] * n + [(-1.0, 2.0)]  # t is bounded loosely
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[0:n]
        t_new = res.x[n]
        return h_new, t_new
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

# Helper functions (compute_c5, compute_c5_vec, project_feasible, grad_smooth_max, etc.)

# SCA iteration
def sca_lp_solve(h0, n):
    dx = 2.0/n
    nx = n + 1
    c_obj = np.zeros(nx)
    c_obj[n] = 1.0
    
    num_m = 2*n - 1
    A_ub = np.zeros((num_m, nx))
    b_ub = np.zeros(num_m)
    
    for m in range(num_m):
        k = m - (n-1)
        am = np.zeros(n)
        Pk = 0.0
        if k >= 0:
            nv = n - k
            am[:nv] = 1.0 - h0[k:]
            if k > 0:
                am[k:] -= h0[:nv]
            Pk = np.dot(h0[:nv], h0[k:])
        else:
            kk = -k
            nv = n - kk
            am[kk:] = 1.0 - h0[:nv]
            if kk > 0:
                am[:nv] -= h0[kk:]
            Pk = np.dot(h0[kk:], h0[:nv])
        
        A_ub[m, :n] = dx * am
        A_ub[m, n] = -1.0
        b_ub[m] = dx * Pk
    
    A_eq = np.zeros((1, nx))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0.0, 1.0)]*n + [(-1.0, 2.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], float(res.x[n])
    return h0.copy(), np.inf

def sca_optimize(h, n, deadline, max_iter=200, damping=0.5):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    lam = damping
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_new, t_new = sca_lp_solve(h, n)
        h_damped = (1-lam)*h + lam*h_new
        h_damped = project_feasible(h_damped, n)
        c_new = compute_c5(h_damped, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_damped.copy()
            h = h_damped
            lam = min(1.0, lam * 1.05)  # increase damping slowly
        else:
            h = h_damped  # still update for linearization
            lam = max(0.1, lam * 0.8)  # decrease damping
        
        if it > 10 and abs(c_new - best_c) < 1e-10:
            break
    
    return best_h, best_c
```
