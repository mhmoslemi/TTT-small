    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def multi_phase_opt(h, n, deadline, rng):
    """Multi-phase optimization: L-BFGS-B + SLSQP + threshold + pair search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase A: L-BFGS-B with increasing alpha
    alphas_lbfgs = [500, 2000, 5000, 20000, 50000]
    mus = [1e6, 1e8, 1e10]
    for a in alphas_lbfgs:
        if time.time() > deadline - 15:
            break
        for mu in mus[:2]:
            ht = lbfgs_opt(best_h, n, a, mu, maxiter=2000)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht
            if time.time() > deadline - 15:
                break

    # Phase B: SLSQP with high alpha
    if time.time() < deadline - 10:
        for a in [50000, 200000, 500000]:
            if time.time() > deadline - 10:
                break
            ht = slsqp_opt(best_h, n, a, maxiter=1500)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht

    # Phase C: Threshold-based optimization
    if time.time() < deadline - 10:
        T = best_c - 0.001
        for T_try in [best_c * 0.999, best_c - 0.002, best_c - 0.005]:
            if T_try < 0:
                continue
            if time.time() > deadline - 5:
                break
            ht = threshold_opt(best_h, n, T_try, 1e8, maxiter=2000)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial construction at its native resolution (250s)
    if time.time() < deadline - 300:
        h1 = project_feasible(h_init.copy(), n_init)
        c1 = compute_c5(h1, n_init)
        alloc1 = min(250, deadline - time.time() - 300)
        h1, c1 = multi_phase_opt(h1, n_init, t0 + alloc1, rng)
        # Pair search on initial
        if time.time() < deadline - 280:
            h1, c1 = pair_search_full(h1, n_init, deadline - 280,
                                      [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upsample to n=400 and optimize (200s)
    if time.time() < deadline - 200:
        n2 = 400
        h2 = interpolate_h(h_init, n_init, n2)
        alloc2 = min(180, deadline - time.time() - 200)
        h2, c2 = multi_phase_opt(h2, n2, t0 + alloc2, rng)
        if time.time() < deadline - 180:
            h2, c2 = pair_search_local(h2, n2, deadline - 180,
                                       [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=50)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Upsample to n=500 and optimize (200s)
    if time.time() < deadline - 180:
        n3 = 500
        h3 = interpolate_h(h_init, n_init, n3)
        alloc3 = min(160, deadline - time.time() - 180)
        h3, c3 = multi_phase_opt(h3, n3, t0 + alloc3, rng)
        if time.time() < deadline - 160:
            h3, c3 = pair_search_local(h3, n3, deadline - 160,
                                       [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=50)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Multi-restart with perturbation (150s)
    if best_h is not None and time.time() < deadline - 120:
        n = len(best_h)
        remaining = deadline - time.time() - 100
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 100:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 80) * 0.4)
            if alloc < 15:
                break
            hp, cp = multi_phase_opt(hp, n, time.time() + alloc, rng)
            if cp < best_c - 1e-14:
                best_c = cp
                best_h = hp

    # Phase 5: Intensive local search on best (remaining time)
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        rem = deadline - time.time() - 5
        if n <= 300 and rem > 30:
            best_h, best_c = pair_search_full(best_h, n, deadline - 3,
                                              [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        else:
            best_h, best_c = pair_search_local(best_h, n, deadline - 3,
                                               [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=60)

    # Phase 6: Final SLSQP polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 1000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)

    # Fallback
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
