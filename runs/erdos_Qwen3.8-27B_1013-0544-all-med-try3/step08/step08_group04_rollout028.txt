def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Compute gradient of smooth max of correlate values, scaled by dx."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    c_shift = alpha * c - alpha * c.max()
    exp_c = np.exp(c_shift)
    w = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        # (1-h[m-lag]) for valid m
        ms = max(0, lag)
        me = min(n, n + lag)
        if ms < me:
            grad[ms:me] += wk * (1 - h[ms - lag:me - lag])
        # -h[m+lag] for valid m
        ms2 = max(0, -lag)
        me2 = min(n, n - lag)
        if ms2 < me2:
            grad[ms2:me2] -= wk * h[ms2 + lag:me2 + lag]
    
    grad *= dx
    return grad

def projected_gradient(h, n, alpha, lr, n_iters, deadline, lambda_pen=0.0):

    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    """SLSQP with smooth max and analytic gradient."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        cm = c.max()
        cs = alpha * (c - cm)
        val = (cm + np.log(np.sum(np.exp(cs))) / alpha) * dx
        # Add penalty for sum constraint (SLSQP also has it, but this helps)
        val += 10 * (x.sum() - n/2.0)**2
        return val
    
    def grad(x):
        g = grad_smooth_max(x, n, alpha)
