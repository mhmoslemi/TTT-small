def compute_B(h, j, n):
    B = np.zeros(2*n-1)
    # k >= 0: p = n-1+k, k = 0..n-1
    ks = np.arange(n)  # k values
    ps = n - 1 + ks  # corresponding p values
    # Condition: k <= j <= n-1 → k <= j (and j <= n-1 always)
    mask1 = ks <= j
    B[ps[mask1]] += (1.0 - h[j - ks[mask1]])
    # Condition: j <= n-1-k → k <= n-1-j
    mask2 = ks <= n-1-j
    B[ps[mask2]] += (-h[j + ks[mask2]])
    
    # k < 0: p = n-1-kk, kk = 1..n-1
    kks = np.arange(1, n)  # pp values

    # Condition: 0 <= j <= n-1-pp → pp <= n-1-j
    mask3 = kks <= n-1-j

    # Condition: pp <= j <= n-1 → pp <= j
    mask4 = kks <= j
    B[ps_neg[mask4]] += (-h[j - kks[mask4]])
    
    return B

def solve_1d(A, B):
    n = len(A)
    # Check t=0
    f0 = A.max()
    active0 = np.where(A == f0)[0]
    slope0 = B[active0].max()  # right derivative at 0
    if slope0 >= 0:
        return 0.0
    
    # Check t=1
    vals1 = A + B
    f1 = vals1.max()
    active1 = np.where(vals1 == f1)[0]
    slope1 = B[active1].min()  # left derivative at 1
    if slope1 <= 0:
        return 1.0
    
    # Bisection for the minimum
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = A + B * mid
        fmid = vals.max()
        active = np.where(vals == fmid)[0]
        slope_lo = B[active].min()  # left subgradient
        slope_hi = B[active].max()  # right subgradient
        if slope_lo > 0:
            # f is increasing at mid → minimum is to the left
            hi = mid
        elif slope_hi < 0:
            # f is decreasing at mid → minimum is to the right
            lo = mid
        else:
            # 0 is in the subgradient → we found the minimum
            return mid
    
    return (lo + hi) / 2

    f0 = A.max()
    # Right derivative at 0: max B among active constraints
    tol = 1e-12 * max(1.0, abs(f0))
    active0 = A >= f0 - tol
    if active0.any():
        slope0 = B[active0].max()
        if slope0 >= -1e-14:
            return 0.0
    
    # Check t=1
    vals1 = A + B
    f1 = vals1.max()
    active1 = vals1 >= f1 - tol
    if active1.any():
        slope1 = B[active1].min()
        if slope1 <= 1e-14:
            return 1.0
    
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2
        vals = A + B * mid
        fmid = vals.max()
        act = vals >= fmid - 1e-12 * max(1.0, abs(fmid))
        s_lo = B[act].min() if act.any() else 0
        s_hi = B[act].max() if act.any() else 0
        if s_lo > 1e-14:
            hi = mid
        elif s_hi < -1e-14:
            lo = mid
        else:
            break
    
    return (lo + hi) / 2

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_vec(h, j, n):
    B = np.zeros(2 * n - 1)
    k = np.arange(-(n - 1), n)
    m1_lo = j - (n - 1)
    m1_hi = j
    mask1 = (k >= m1_lo) & (k <= m1_hi)
    if mask1.any():
        B[mask1] += (1.0 - h[j - k[mask1]])
    m2_lo = -j
    m2_hi = n - 1 - j
    mask2 = (k >= m2_lo) & (k <= m2_hi)
    if mask2.any():
        B[mask2] += (-h[j + k[mask2]])
    return B

def solve_1d(A, B):
    f0 = A.max()
    tol0 = 1e-12 * max(1.0, abs(f0))
    act0 = A >= f0 - tol0
    if act0.any():
        if B[act0].max() >= -1e-14:
            return 0.0
    vals1 = A + B
    f1 = vals1.max()
    tol1 = 1e-12 * max(1.0, abs(f1))
    act1 = vals1 >= f1 - tol1
    if act1.any():
        if B[act1].min() <= 1e-14:
            return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = A + B * mid
        fmid = vals.max()
        tol = 1e-12 * max(1.0, abs(fmid))
        act = vals >= fmid - tol
        if not act.any():
            break
        s_lo = B[act].min()
        s_hi = B[act].max()
        if s_lo > 1e-13:
            hi = mid
        elif s_hi < -1e-13:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, max_sweeps, deadline, rng):
    h = project_feasible(h, n).copy()
    c_base = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    best_c = float((c_base * dx).max())
    best_h = h.copy()
    order = np.arange(n)
    stall_count = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        rng.shuffle(order)
        prev_best = best_c
        for j in order:
            B = compute_B_vec(h, j, n)
            A = c_base - B * h[j]
            t_new = solve_1d(A, B)
            delta = t_new - h[j]
            if abs(delta) > 1e-15:
                c_base += B * delta
                h[j] = t_new
        h = project_feasible(h, n)
        c_base = np.correlate(h, 1 - h, mode='full')
        c = float((c_base * dx).max())
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 30:
            break
    return best_h, best_c

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = max(500, int(remaining * 4))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 8,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0, 1)
    h_new = project_feasible(h_new, n_new)
    return h_new

def pair_refine(h, n, deadline, n_trials=50000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        for _ in range(n_trials):
            if time.time() > deadline:
                return best_h, best_c
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            for sign in [1, -1]:
                hi = h[i] + sign * d
                hj = h[j] - sign * d
                if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                    continue
                h[i] = hi
                h[j] = hj
                c = compute_c5(h, n)
                if c < best_c - 1e-16:
                    best_c = c
                    best_h = h.copy()
                else:
                    h[i] = best_h[i]
                    h[j] = best_h[j]
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Coordinate descent at original resolution
    p1_deadline = min(t0 + 200, deadline - 500)
    h1, c1 = coord_descent(best_h, best_n, 5000, p1_deadline, rng)
    if c1 < best_c:
        best_c, best_h, best_n = c1, h1, n0

    # Phase 2: L-BFGS polish at original resolution
    if time.time() < deadline - 400:
        p2_deadline = min(time.time() + 80, deadline - 400)
        h2, c2 = lbfgs_phase(best_h, best_n, p2_deadline,
                             alphas=[500, 5000, 50000, 500000], mu=0.005)
        if c2 < best_c:
            best_c, best_h, best_n = c2, h2, best_n

    # Phase 3: Interpolate to n=1000 and coordinate descent
    n1 = max(1000, 2 * best_n)
    if time.time() < deadline - 350:
        h_fine = interpolate_h(best_h, best_n, n1)
        p3_deadline = min(time.time() + 250, deadline - 250)
        h3, c3 = coord_descent(h_fine, n1, 5000, p3_deadline, rng)
        if c3 < best_c:
            best_c, best_h, best_n = c3, h3, n1

    # Phase 4: L-BFGS polish at n1
    if time.time() < deadline - 250:
        p4_deadline = min(time.time() + 80, deadline - 200)
        h4, c4 = lbfgs_phase(best_h, best_n, p4_deadline,
                             alphas=[1000, 10000, 100000, 500000], mu=0.003)
        if c4 < best_c:
            best_c, best_h, best_n = c4, h4, best_n

    # Phase 5: Interpolate to n=2000 and coordinate descent
    n2 = max(2000, 2 * best_n)
    if time.time() < deadline - 200:
        h_fine2 = interpolate_h(best_h, best_n, n2)
        p5_deadline = min(time.time() + 200, deadline - 100)
        h5, c5 = coord_descent(h_fine2, n2, 3000, p5_deadline, rng)
        if c5 < best_c:
            best_c, best_h, best_n = c5, h5, n2

    # Phase 6: L-BFGS polish at final resolution
    if time.time() < deadline - 100:
        p6_deadline = min(time.time() + 60, deadline - 50)
        h6, c6 = lbfgs_phase(best_h, best_n, p6_deadline,
                             alphas=[5000, 50000, 200000], mu=0.002)
        if c6 < best_c:
            best_c, best_h, best_n = c6, h6, best_n

    # Phase 7: More coordinate descent at final resolution
    if time.time() < deadline - 40:
        p7_deadline = min(time.time() + 30, deadline - 15)
        h7, c7 = coord_descent(best_h, best_n, 2000, p7_deadline, rng)
        if c7 < best_c:
            best_c, best_h, best_n = c7, h7, best_n

    # Phase 8: Pair refinement
    if time.time() < deadline - 10:
        p8_deadline = min(time.time() + 15, deadline - 3)
        h8, c8 = pair_refine(best_h, best_n, p8_deadline, n_trials=30000, rng=rng)
        if c8 < best_c:
            best_c, best_h, best_n = c8, h8, best_n

    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
