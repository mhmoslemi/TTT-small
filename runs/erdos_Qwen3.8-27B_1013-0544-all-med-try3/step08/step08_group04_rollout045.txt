```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    """Gradient of c[k] w.r.t. h where c = correlate(h,1-h)*dx."""
    dx = 2.0 / n
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] -= h[lag:n]
    else:
        L = -lag
        grad[:n-L] += (1.0 - h[L:n])
        grad[L:] -= h[:n-L]
    return grad * dx

def projected_grad_step(h, n, c, top_m, alpha, lr):
    """One step of projected gradient descent."""
    c_full = c  # already computed
    m = min(top_m, len(c_full))
    idx = np.argpartition(c_full, -m)[-m:]
    idx = idx[np.argsort(c_full[idx])[::-1]]
    
    if alpha > 0:
        # Log-sum-exp over top-m
        vals = c_full[idx]
        mx = vals.max()
        shifted = alpha * (vals - mx)
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, idx):
            grad += w * grad_single(h, n, k)
    else:
        # Just the max (use small average of top few for stability)
        m_eff = min(5, m)
        idx_eff = idx[:m_eff]
        grad = np.zeros(n)
        for k in idx_eff:
            grad += grad_single(h, n, k)
        grad /= len(idx_eff)
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, c_full.max()

def prox_grad_descent(h, n, deadline, top_m_schedule, lr_schedule):
    """Main projected gradient loop."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_c(h, n)
    
    for phase in range(len(top_m_schedule)):
        if time.time() > deadline:
            break
        top_m = top_m_schedule[phase]
        lr_start = lr_schedule[phase]
        n_iters = 500  # iterations per phase
        
        for it in range(n_iters):
            if time.time() > deadline:
                break
            # Adaptive lr
            lr = lr_start * (0.995 ** it)
            if lr < 1e-8:
                break
            
            h_new, _ = projected_grad_step(h, n, c, top_m, 0, lr)
            c_new = compute_c(h_new, n)
            c5_new = c_new.max()
            
            if c5_new < best_c - 1e-15:
                best_c = c5_new
                best_h = h_new.copy()
            
            h = h_new
            c = c_new
    
    return best_h, best_c
```
