            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for count in range(min(500, n)):
                if time.time() > deadline:
                    return h, best_c
                i = int(order[count % len(order)])
                if count % 3 == 0:
                    j = int(order[(count + max(1, len(order)//4)) % len(order)])
                elif count % 3 == 1:
                    j = int(order[(count + len(order)//2) % len(order)])
                else:
                    j = int(rng.integers(0, n))
                
                if i == j:
                    continue
                
                sign = 1.0 if rng.random() < 0.5 else -1.0
                t = sign * step_scale
                
                if h[i] + t < -1e-12 or h[i] + t > 1.0 + 1e-12:
                    continue
                if h[j] - t < -1e-12 or h[j] - t > 1.0 + 1e-12:
                    continue
                
                h[i] += t
                h[j] -= t
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    h[i] -= t
                    h[j] += t
                    h = project_feasible(h, n)
    
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    obj_fn_holder = {'n': n, 'alpha': alpha}
    
    def _slqp_obj(x):
        nn = obj_fn_holder['n']
        aa = obj_fn_holder['alpha']
        return lse_obj(x, nn, aa)
    
    def _slqp_jac(x):
        nn = obj_fn_holder['n']
        aa = obj_fn_holder['alpha']
        return lse_grad(x, nn, aa)
    
    def _cons_fun(x):
        return [x.sum() - n / 2.0]
    
    def _cons_jac(x):
        return [np.ones(n)]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_slqp_obj, h, jac=_slqp_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def multi_resolution_optimize(h_base, n_base, deadline):
    best_h = h_base.copy()
    best_c = compute_c5(h_base, n_base)
    best_n = n_base
    
    resolutions = [400, 600, 800]
    time_per_res = (deadline - time.time()) / (len(resolutions) + 1)
    
    for n_try in resolutions:
        if time.time() > deadline - 30:
            break
        h_try = interpolate_h(h_base, n_base, n_try)
        c_try = compute_c5(h_try, n_try)
        
        t_phase1 = min(time.time() + time_per_res * 0.3, deadline - 60)
        h_try, c_try = run_lbfgs(h_try, n_try, t_phase1,
                                  alphas=[500, 3000, 10000], maxiter=4000)
        
        t_phase2 = min(time.time() + time_per_res * 0.4, deadline - 30)
        h_try, c_try = newton_active_optimize(h_try, n_try, t_phase2, max_iter=80)
        
        t_phase3 = min(time.time() + time_per_res * 0.3, deadline - 10)
        h_try, c_try = fine_coordinate_descent(h_try, n_try, t_phase3)
        
        c_try = compute_c5(h_try, n_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
            best_n = n_try
    
    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n
    
    # Phase 1: L-BFGS smooth optimization (40s)
    t1 = min(t0 + 40, deadline - 800)
    if t1 > time.time() + 3:
        h, best_c = run_lbfgs(h, n, t1,
                              alphas=[200, 1000, 5000, 20000, 80000],
                              maxiter=10000)
        best_h = h.copy()
    
    # Phase 2: Newton active-set optimization (180s)
    t2 = min(time.time() + 180, deadline - 500)
    if t2 > time.time() + 5:
        h2, c2 = newton_active_optimize(h, n, t2, max_iter=400)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            h = h2
    
    # Phase 3: Fine coordinate descent (250s)
    t3 = min(time.time() + 250, deadline - 300)
    if t3 > time.time() + 5:
        h3, c3 = fine_coordinate_descent(h, n, t3)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
            h = h3
    
    # Phase 4: Another Newton round after coordinate descent (60s)
    t4 = min(time.time() + 60, deadline - 200)
    if t4 > time.time() + 5:
        h4, c4 = newton_active_optimize(h, n, t4, max_iter=150)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
            h = h4
    
    # Phase 5: Very fine local search (100s)
    t5 = min(time.time() + 100, deadline - 150)
    if t5 > time.time() + 5:
        h5, c5 = fine_coordinate_descent(h, n, t5)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
            h = h5
    
    # Phase 6: SLSQP polish (60s)
    t6 = min(time.time() + 60, deadline - 60)
    if t6 > time.time() + 5:
        h6, c6 = slqp_polish(h, n, t6, alpha=1000000, maxiter=3000)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
            h = h6
    
    # Phase 7: Multi-resolution check
    if time.time() < deadline - 120:
        h7, c7, n7 = multi_resolution_optimize(best_h, best_n, deadline - 60)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
            best_n = n7
            h = h7
            n = n7
    
    # Phase 8: Final very fine polish on best solution
    if time.time() < deadline - 10:
        t8 = deadline - 5
        h8, c8 = fine_coordinate_descent(best_h, len(best_h), t8)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final SLSQP if time permits
    if time.time() < deadline - 15:
        h9, c9 = slqp_polish(best_h, len(best_h), deadline - 3, alpha=2000000, maxiter=2000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final feasibility and validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    # Verify constraints
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(best_h.sum() * 2.0 / n_final - 1.0) < 1e-8
    
    return best_h, best_c, n_final
