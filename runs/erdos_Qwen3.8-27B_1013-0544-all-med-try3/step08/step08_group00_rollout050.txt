import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:k+nv]) * dx
            grad[k:k+nv] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:kk+nv] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:kk+nv]) * dx
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * alpha * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:k+nv])
                grad[k:k+nv] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:kk+nv] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:kk+nv])
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    grads = []
    c_vals = []
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        grads.append(g)
        c_vals.append(c_vec[idx])
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        A_ub[row, 0:n] = grads[j]
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 5.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.05, max_active=25):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    consec_no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > max_active:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:max_active]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            consec_no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.4, 0.12)
        else:
            consec_no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if consec_no_improve > 30:
                break
    return best_h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 100.0), (2000, 500.0), (10000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': 100000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def projected_subgradient(h, n, deadline, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_size = 0.005
    momentum = np.zeros(n)
    mom_coeff = 0.3
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        cv = c_vec
        mv = cv.max()
        sh = 5000.0 * (cv - mv)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for mm in range(len(w)):
            if w[mm] < 1e-13:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += wm * (1.0 - h[k:k+nv])
                    grad[k:k+nv] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:kk+nv] += wm * (1.0 - h[0:nv])
                    grad[0:nv] += wm * (-h[kk:kk+nv])
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        momentum = mom_coeff * momentum + grad
        h_trial = h - step_size * momentum
        h_new = project_feasible(h_trial, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if it % 100 == 0 and step_size < 0.02:
                step_size *= 1.1
        else:
            h = h_new
            no_improve += 1
            step_size *= 0.9
            if no_improve > 50:
                step_size = max(0.001, step_size * 0.5)
                no_improve = 0
    return best_h, best_c

def targeted_local_search(h, n, deadline, center_radius=30):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad_single(h, n, idx))
            if len(active) > 0:
                grad /= len(active)
            top_pos = np.argsort(-grad)[:min(center_radius * 3, n)]
            tried = 0
            for i_val in top_pos:
                if tried > 50 and improved:
                    break
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    tried += 1
                    if i == j:
                        continue
                    if tried > 200:
                        break
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def coordinate_local_search(h, n, deadline, max_sweeps=50):
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweep_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            orig = h[i]
            for trial in [0.0, 1.0, orig + 0.05, orig - 0.05]:
                t = max(0.0, min(1.0, trial))
                if abs(t - orig) < 1e-10:
                    continue
                delta = t - orig
                h[i] = t
                h_proj = project_feasible(h, n)
                c = compute_c5(h_proj, n)
                if c < best_c - 1e-14:
                    best_c = c
                    h = h_proj.copy()
                    sweep_improved = True
                    break
                else:
                    h[i] = orig
        if not sweep_improved:
            break
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_orig = 400

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick L-BFGS at original resolution (40s)
    t1 = min(t0 + 40, deadline - 900)
    if t1 > time.time() + 5:
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            configs=[(500, 100.0), (2000, 500.0), (8000, 3000.0)])

    # Phase 2: SQP at original resolution (50s)
    t2 = min(t0 + 100, deadline - 800)
    if t2 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=300, rho_init=0.05)

    # Phase 3: Upscale to n=2000
    n_high = 2000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n_high)
    h_high = np.interp(x_new, x_old, best_h)
    h_high = project_feasible(h_high, n_high)
    c_high = compute_c5(h_high, n_high)
    if c_high < best_c:
        best_c = c_high
        best_h = h_high.copy()

    # Phase 4: L-BFGS at n=2000 (150s)
    t4 = min(time.time() + 150, deadline - 450)
    if t4 > time.time() + 10:
        h_high, c_high = run_lbfgs(h_high, n_high, t4,
            configs=[(1000, 200.0), (5000, 1000.0), (20000, 5000.0)])
        if c_high < best_c:
            best_c = c_high
            best_h = h_high

    # Phase 5: SQP at n=2000 (200s)
    t5 = min(time.time() + 200, deadline - 250)
    if t5 > time.time() + 10:
        h_high, c_high = sqp_optimize(h_high, n_high, t5, max_iter=400, rho_init=0.04, max_active=30)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high

    # Phase 6: Projected subgradient at n=2000 (150s)
    t6 = min(time.time() + 150, deadline - 150)
    if t6 > time.time() + 10:
        h_high, c_high = projected_subgradient(h_high, n_high, t6, max_iter=80000)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high

    # Phase 7: Targeted local search at n=2000 (100s)
    t7 = min(time.time() + 100, deadline - 60)
    if t7 > time.time() + 5:
        h_high, c_high = targeted_local_search(h_high, n_high, t7, center_radius=40)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high

    # Phase 8: Another SQP round (60s)
    if time.time() < deadline - 30:
        h_high, c_high = sqp_optimize(h_high, n_high, deadline - 10,
                                      max_iter=200, rho_init=0.02, max_active=20)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high

    # Phase 9: Final SLSQP polish
    if time.time() < deadline - 10:
        h_final, c_final = slqp_polish(best_h, len(best_h), deadline - 2,
                                       alpha=1000000, maxiter=2000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final: tiny local search on best solution
    n_final = len(best_h)
    if time.time() < deadline:
        best_h, best_c = targeted_local_search(best_h, n_final, deadline, center_radius=20)

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    # Verify constraints
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
