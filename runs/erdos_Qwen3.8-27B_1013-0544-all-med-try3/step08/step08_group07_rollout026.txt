import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def greedy_targeted(h_bin, n, deadline, max_rounds=20000):
    dx = 2.0 / n
    rng = np.random.RandomState(123)
    best_c = float(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')) * dx)
    cur_c = best_c
    ones_set = set(np.where(h_bin == 1)[0].tolist())
    zeros_set = set(np.where(h_bin == 0)[0].tolist())
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        cv = corr * dx
        m = float(cv.max())
        
        if m <= best_c:
            break
        
        # Find top-3 bottleneck shifts
        top_idx = np.argsort(-cv)[:3]
        improved_this_round = False
        
        for tidx in top_idx:
            d = int(tidx) - (n - 1)
            if d >= 0:
                rng_i = range(0, n - d)
            else:
                rng_i = range(-d, n)
            
            bad_i = []
            for i in rng_i:
                if h_bin[i] == 1 and h_bin[i + d] == 0:
                    bad_i.append(i)
            
            if not bad_i:
                continue
            
            # Try to fix by moving a 1 away from bad position or filling the gap
            tried = 0
            for i in bad_i:
                j = i + d
                # Swap h[i] (1) with a random zero
                zlist = [x for x in zeros_set if x != i]
                if not zlist:
                    continue
                ntry = min(15, len(zlist))
                idxs = rng.choice(len(zlist), size=min(ntry, len(zlist)), replace=False)
                for ii in idxs:
                    z = zlist[ii]
                    h_bin[i] = 0.0
                    h_bin[z] = 1.0
                    c_new = float(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')) * dx)
                    if c_new < m - 1e-14:
                        ones_set.discard(i)
                        ones_set.add(z)
                        zeros_set.add(i)
                        zeros_set.discard(z)
                        cur_c = c_new
                        if c_new < best_c:
                            best_c = c_new
                        improved_this_round = True
                        tried += 1
                        break
                    else:
                        h_bin[i] = 1.0
                        h_bin[z] = 0.0
                        tried += 1
                if improved_this_round and tried > 3:
                    break
            
            if improved_this_round:
                break
        
        if not improved_this_round:
            # Escape: random perturbation (swap 2-3 pairs)
            esc_found = False
            n_esc = 3
            for _ in range(200):
                if not ones_set or not zeros_set:
                    break
                # Collect pairs to swap
                olist = list(ones_set)
                zlist = list(zeros_set)
                swaps = []
                for _ in range(min(n_esc, len(olist), len(zlist))):
                    o = olist[rng.randint(len(olist))]
                    z = zlist[rng.randint(len(zlist))]
                    swaps.append((o, z))
                    olist.remove(o)
                    zlist.remove(z)
                
                for o, z in swaps:
                    h_bin[o], h_bin[z] = 0.0, 1.0
                    ones_set.discard(o)
                    ones_set.add(z)
                    zeros_set.add(o)
                    zeros_set.discard(z)
                
                c_new = float(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')) * dx)
                if c_new <= best_c + 0.002:
                    cur_c = c_new
                    esc_found = True
                    break
                else:
                    for o, z in swaps:
                        h_bin[o], h_bin[z] = 1.0, 0.0
                        ones_set.add(o)
                        ones_set.discard(z)
                        zeros_set.discard(o)
                        zeros_set.add(z)
            
            if not esc_found:
                break
    
    return h_bin, best_c

def sim_anneal_binary(h_bin, n, deadline, t_start=0.5, t_end=0.0001):
    dx = 2.0 / n
    rng = np.random.RandomState(999)
    best_h = h_bin.copy()
    corr = np.correlate(h_bin, 1 - h_bin, mode='full')
    best_c = float(np.max(corr) * dx)
    cur_c = best_c
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    
    # Estimate iters
    t0 = time.time()
    # Time one correlation
    _ = np.correlate(h_bin, 1 - h_bin, mode='full')
    t_corr = time.time() - t0
    
    iters_per_sec = 0.7 / max(t_corr, 1e-6)
    remaining = max(deadline - time.time(), 0)
    max_iters = int(remaining * min(iters_per_sec, 20000))
    max_iters = max(max_iters, 1000)
    
    t = t_start
    if max_iters > 1:
        cool = (t_end / t_start) ** (1.0 / max_iters)
    else:
        cool = t_end / t_start
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_bin[i], h_bin[j] = 0.0, 1.0
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        new_c = float(np.max(corr) * dx)
        
        delta = new_c - cur_c
        if delta <= 0 or rng.rand() < np.exp(-delta / max(t, 1e-10)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h_bin[i], h_bin[j] = 1.0, 0.0
        
        t *= cool
    
    return best_h, best_c

def two_swap_local(h_bin, n, deadline, max_iters=50000):
    dx = 2.0 / n
    rng = np.random.RandomState(555)
    best_c = float(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')) * dx)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Pick 2 ones and 2 zeros to swap
        o1 = ones[rng.randint(len(ones))]
        o2 = ones[rng.randint(len(ones) - 1)]
        if o1 == o2:
            continue
        z1 = zeros[rng.randint(len(zeros))]
        z2 = zeros[rng.randint(len(zeros) - 1)]
        if z1 == z2:
            continue
        
        h_bin[o1], h_bin[o2] = 0.0, 0.0
        h_bin[z1], h_bin[z2] = 1.0, 1.0
        c_new = float(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')) * dx)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            ones.remove(o1)
            ones.remove(o2)
            ones.extend([z1, z2])
            zeros.remove(z1)
            zeros.remove(z2)
            zeros.extend([o1, o2])
        else:
            h_bin[o1], h_bin[o2] = 1.0, 1.0
            h_bin[z1], h_bin[z2] = 0.0, 0.0
    
    return h_bin, best_c

def continuous_optimize(h, n, deadline, alpha_list=(500, 2000, 10000, 50000, 200000)):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        
        def objective(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            m = c.max()
            shifted = alpha * (c - m)
            sm = shifted.max()
            logsum = sm + np.log(np.exp(shifted - sm).sum())
            return m + logsum / alpha
        
        def constraint_fun(x):
            return x.sum() - n / 2.0
        
        res = minimize(objective, best_h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': constraint_fun}],
                      options={'maxiter': 500, 'ftol': 1e-14})
        
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    return best_h, best_c

def subgrad_polish(h, n, deadline, max_iter=3000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        k = int(np.argmax(c))
        
        # Compute gradient of c_k w.r.t. h
        # c_k = sum_i h[i]*(1-h[i+k]) for valid pairs
        grad = np.zeros(n)
        # Forward differences for speed
        eps = 1e-8
        for i in range(0, n, max(1, n // 50)):
            old = h[i]
            h[i] = old + eps
            c_new = np.correlate(h, 1 - h, mode='full')[k] * dx
            grad[i] = (c_new - c[k]) / eps
            h[i] = old
        
        # Scale gradient to cover all positions
        grad_full = np.zeros(n)
        for i in range(n):
            old = h[i]
            h[i] = old + eps
            c_new = np.correlate(h, 1 - h, mode='full')[k] * dx
            grad_full[i] = (c_new - c[k]) / eps
            h[i] = old
        
        ng = np.linalg.norm(grad_full)
        if ng < 1e-12:
            break
        
        accepted = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad_full, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                h = ht
                accepted = True
                if ct < best_c:
                    best_c = ct
                    best_h = h.copy()
                break
        
        if not accepted:
            lr *= 0.9
            if lr < 1e-12:
                break
    
    return best_h, best_c

                best_c = cc
                best_h = hc.copy()
    
    # Final verification and projection
    best_h = project_feasible(best_h, n)
    final_c = compute_c5(best_h, n)
    
    return best_h, final_c, n
