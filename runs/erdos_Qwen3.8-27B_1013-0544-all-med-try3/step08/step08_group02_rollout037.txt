                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    return best_h, best_c

def active_set_phase(h, n, end_time, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000]

    for rnd in range(30):
        remaining = end_time - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        active = get_active_set(c_vec, top_k, tol_frac=0.0005)
        if len(active) < 2:
            active = get_active_set(c_vec, top_k, tol_frac=0.01)
        if len(active) < 2:
            break

        for alpha in alphas:
            remaining = end_time - time.time()
            if remaining < 10:
                break
            maxit = max(300, int(remaining * 2))
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                              args=(n, alpha, active, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    return best_h, best_c

def local_pair_search(h, n, end_time, radius=25, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]

    for d in deltas:
        improved = True
        while improved and time.time() < end_time:
            improved = False
            for i in range(n):
                if time.time() > end_time:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] >= d and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def subgradient_refine(h, n, end_time, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003

    for it in range(n_iter):
        if time.time() > end_time:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.0005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)

        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97

        h = h_new

    return best_h, best_c

def upsample(h_from, n_from, n_to):
    x_from = np.linspace(0, 2, n_from)
    x_to = np.linspace(0, 2, n_to)
    h_to = np.interp(x_to, x_from, h_from)
    return project_feasible(h_to, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    c0 = compute_c5(h0, n_orig)

    # Choose primary resolution
    n_primary = 1000
    h_primary = upsample(h0, n_orig, n_primary)
    best_h = h_primary.copy()
    best_c = min(c0, compute_c5(h_primary, n_primary))
    best_n = n_primary

    # Phase 1: Coarse L-BFGS at n_primary
    p1_end = t0 + 250
    h1, c1 = lbfgs_phase(best_h, best_n, p1_end,
                         alphas=[30, 100, 500, 2000, 10000, 50000, 200000],
                         mu=0.005,
                         max_iters=[2000, 2000, 3000, 3000, 3000, 4000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set L-BFGS
    p2_end = min(time.time() + 200, deadline - 300)
    h2, c2 = active_set_phase(best_h, best_n, p2_end, top_k=8,
                              alphas=[50000, 200000, 1000000, 5000000])
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: LP-based linearized min-max
    c_vec = compute_c5_vec(best_h, best_n)
    active = get_active_set(c_vec, 10, tol_frac=0.001)
    p3_end = min(time.time() + 200, deadline - 150)
    h3, c3 = lp_minmax(best_h, best_n, active, max_iter=300)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Update active set and do another LP round
    c_vec = compute_c5_vec(best_h, best_n)
    active = get_active_set(c_vec, 12, tol_frac=0.002)
    if time.time() < deadline - 100:
        p3b_end = min(time.time() + 150, deadline - 80)
        h3b, c3b = lp_minmax(best_h, best_n, active, max_iter=200)
        if c3b < best_c:
            best_c = c3b
            best_h = h3b

    # Phase 4: Try higher resolution n=2000
    n_high = 2000
    if time.time() < deadline - 200:
        h_high = upsample(best_h, len(best_h), n_high)
        c_high_init = compute_c5(h_high, n_high)

        p4_end = min(time.time() + 200, deadline - 150)
        h4, c4 = lbfgs_phase(h_high, n_high, p4_end,
                             alphas=[20, 100, 500, 2000, 10000, 50000],
                             mu=0.003,
                             max_iters=[1500, 2000, 2000, 2500, 3000, 3000])

        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_high

        if time.time() < deadline - 80:
            p4b_end = min(time.time() + 100, deadline - 60)
            h4b, c4b = active_set_phase(best_h, best_n, p4b_end, top_k=8,
                                        alphas=[100000, 500000, 2000000])
            if c4b < best_c:
                best_c = c4b
                best_h = h4b

        # LP refinement at high resolution
        if time.time() < deadline - 50:
            c_vec_h = compute_c5_vec(best_h, best_n)
            active_h = get_active_set(c_vec_h, 8, tol_frac=0.001)
            p4c_end = min(time.time() + 60, deadline - 30)
            h4c, c4c = lp_minmax(best_h, best_n, active_h, max_iter=100)
            if c4c < best_c:
                best_c = c4c
                best_h = h4c

    # Phase 5: Subgradient refinement
    if time.time() < deadline - 30:
        p5_end = min(time.time() + 40, deadline - 20)
        h5, c5 = subgradient_refine(best_h, best_n, p5_end, n_iter=3000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Final active set + LP
    if time.time() < deadline - 20:
        c_vec_f = compute_c5_vec(best_h, best_n)
        active_f = get_active_set(c_vec_f, 6, tol_frac=0.0005)
        p6_end = deadline - 5
        h6, c6 = active_set_phase(best_h, best_n, p6_end, top_k=6,
                                  alphas=[500000, 2000000, 10000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final LP
    if time.time() < deadline - 5:
        c_vec_f2 = compute_c5_vec(best_h, best_n)
        active_f2 = get_active_set(c_vec_f2, 8, tol_frac=0.001)
        p7_end = deadline - 1
        h7, c7 = lp_minmax(best_h, best_n, active_f2, max_iter=100)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
