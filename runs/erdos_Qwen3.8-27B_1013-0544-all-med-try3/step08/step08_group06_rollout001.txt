import time
import numpy as np
from scipy.optimize import minimize

_GLOBAL_STOP_TIME = 0.0


def stop_callback(*args):
    return time.time() > _GLOBAL_STOP_TIME


def project_feasible(x, n=None):
    if n is None:
        x = np.asarray(x, dtype=float).ravel()
        n = len(x)
    else:
        x = np.asarray(x, dtype=float).ravel()
        if len(x) != n:
            if len(x) == 0:
                x = np.full(n, 0.5)
            else:
                x = np.interp(np.linspace(0.0, 1.0, n),
                              np.linspace(0.0, 1.0, len(x)), x)
        n = int(n)

    if n == 0:
        return x

    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.5)

    H = 0.5 * n

    lo = -float(np.max(x)) - 1.0
    hi = 1.0 - float(np.min(x)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x + mid, 0.0, 1.0).sum())
        if s < H:
            lo = mid
        else:
            hi = mid

    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = H - float(y.sum())

    if diff > 0.0:
        idxs = np.argsort(1.0 - y)[::-1]
        for idx in idxs:
            if diff <= 1e-15:
                break
            room = 1.0 - y[idx]
            if room > 0.0:
                add = min(diff, room)
                y[idx] += add
                diff -= add
    elif diff < 0.0:
        idxs = np.argsort(y)
        for idx in idxs:
            if diff >= -1e-15:
                break
            avail = y[idx]
            if avail > 0.0:
                sub = min(-diff, avail)
                y[idx] -= sub
                diff += sub

    np.clip(y, 0.0, 1.0, out=y)
    return y


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return float("inf")
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return m * dx


def interpolate_h(h, n):
    h = np.asarray(h, dtype=float).ravel()
    if len(h) == 0:
        return np.full(n, 0.5)
    if len(h) == n:
        return h.copy()
    old = np.linspace(0.0, 2.0, len(h))
    new = np.linspace(0.0, 2.0, n)
    return np.interp(new, old, h)


def symmetrize_h(h):
    h = np.asarray(h, dtype=float).ravel()
    if len(h) == 0:
        return h
    return project_feasible(0.5 * (h + h[::-1]))


def smooth_objective_gradient(h, alpha):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 0.0, np.zeros(0)

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    v = c * dx

    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.0)

    m = float(np.max(v))
    if not np.isfinite(m):
        m = 0.0

    if alpha <= 0.0:
        return m, np.zeros_like(h)

    z = alpha * (v - m)
    ez = np.exp(z)
    s = float(np.sum(ez))

    if not np.isfinite(s) or s <= 0.0:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        obj = m
    else:
        w = ez / s
        obj = m + np.log(s) / alpha

    mid = n - 1
    wr = np.zeros(n)
    wl = np.zeros(n)
    wr[0] = w[mid]
    if n > 1:
        wr[1:] = w[mid - 1::-1]
        wl[1:] = w[mid + 1:]

    q = 1.0 - h

    A = np.correlate(q, wr, mode="full")[mid:mid + n]
    B = np.convolve(wr, h, mode="full")[:n]

    C = np.convolve(wl, q, mode="full")[:n]
    D = np.correlate(h, wl, mode="full")[mid:mid + n]

    grad = dx * ((A - B) + (C - D))

    if not np.all(np.isfinite(grad)):
        grad = np.where(np.isfinite(grad), grad, 0.0)

    return float(obj), grad


def fun_alpha(x, alpha):
    ob, _ = smooth_objective_gradient(x, alpha)
    return ob


def jac_alpha(x, alpha):
    _, gr = smooth_objective_gradient(x, alpha)
    return gr


def penalty_fun(x, alpha, lam):
    ob, _ = smooth_objective_gradient(x, alpha)
    d = float(np.sum(x) - 0.5 * len(x))
    return ob + 0.5 * lam * d * d


def penalty_jac(x, alpha, lam):
    _, gr = smooth_objective_gradient(x, alpha)
    d = float(np.sum(x) - 0.5 * len(x))
    if d == 0.0:
        return gr
    return gr + lam * d * np.ones_like(x)


def sum_con_fun(x):
    return float(np.sum(x) - 0.5 * len(x))


def sum_con_jac(x):
    return np.ones_like(x)


def make_bounds(n):
    return [(0.0, 1.0) for _ in range(n)]


def slsqp_run(h, n, alpha, maxiter, stop_after):
    global _GLOBAL_STOP_TIME
    h = np.asarray(h, dtype=float).copy()
    if n == 0:
        return h

    _GLOBAL_STOP_TIME = time.time() + max(1.0, float(stop_after))
    bounds = make_bounds(n)
    cons = [{"type": "eq", "fun": sum_con_fun, "jac": sum_con_jac}]

    res = None
    try:
        res = minimize(
            fun_alpha,
            h,
            args=(alpha,),
            jac=jac_alpha,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-14, "iprint": 0},
            callback=stop_callback,
        )
    except Exception:
        res = None
    finally:
        _GLOBAL_STOP_TIME = 0.0

    if res is not None and hasattr(res, "x") and res.x is not None:
        return project_feasible(res.x, n)
    return h


def lbfgb_run(h, n, alpha, lam, maxiter, stop_after):
    global _GLOBAL_STOP_TIME
    h = np.asarray(h, dtype=float).copy()
    if n == 0:
        return h

    _GLOBAL_STOP_TIME = time.time() + max(1.0, float(stop_after))
    bounds = make_bounds(n)

    res = None
    try:
        res = minimize(
            penalty_fun,
            h,
            args=(alpha, lam),
            jac=penalty_jac,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": int(maxiter), "ftol": 1e-14, "gtol": 1e-8, "disp": 0},
            callback=stop_callback,
        )
    except Exception:
        res = None
    finally:
        _GLOBAL_STOP_TIME = 0.0

    if res is not None and hasattr(res, "x") and res.x is not None:
        return project_feasible(res.x, n)
    return h


def optimize_resolution(h_start, n, end_time):

    best = h.copy()
    best_c = compute_c5(best)

    n = len(best)
    if n == 0:
        return best, float("inf")

    if n <= 140:
        alphas = [100.0, 400.0, 1500.0, 6000.0, 25000.0, 100000.0, 350000.0]
        maxiter = 450
    elif n <= 260:
        alphas = [50.0, 200.0, 800.0, 3000.0, 12000.0, 50000.0, 200000.0, 700000.0]
        maxiter = 320
    elif n <= 400:
        alphas = [100.0, 500.0, 2500.0, 10000.0, 50000.0, 200000.0]
        maxiter = 220
    else:
        alphas = [500.0, 3000.0, 20000.0, 100000.0]
        maxiter = 120

    for alpha in alphas:
        if time.time() > end_time - 5.0:
            break
        rem = end_time - time.time()
        stop = min(70.0, max(10.0, 0.25 * rem))
        h2 = slsqp_run(best, n, alpha, maxiter, stop)
        c2 = compute_c5(h2)
        if np.isfinite(c2) and c2 < best_c - 1e-15:
            best_c = c2
            best = h2

    if time.time() < end_time - 8.0:
        for alpha, lam in [(200000.0, 1e4), (500000.0, 1e5)]:
            if time.time() > end_time - 6.0:
                break
            rem = end_time - time.time()
            stop = min(40.0, max(8.0, 0.35 * rem))
            h2 = lbfgb_run(best, n, alpha, lam, 220, stop)
            c2 = compute_c5(h2)
            if np.isfinite(c2) and c2 < best_c - 1e-15:
                best_c = c2
                best = h2

    best = project_feasible(best, n)
    best_c = compute_c5(best)
    return best, best_c


def get_active_lags(c_arr, dx, n, tol=0.002):
    if len(c_arr) == 0:
        return [0]
    v = c_arr * dx
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.0)
    m = float(np.max(v))
    if not np.isfinite(m) or m <= 0.0:
        return [n - 1]

    idxs = np.where(v >= m - tol)[0]
    if len(idxs) == 0:
        idxs = np.array([n - 1])

    if len(idxs) > 64:
        order = np.argsort(v[idxs])[::-1][:64]
        idxs = idxs[order]

    return [int(i) for i in idxs]


def local_search(h_in, n, end_time, seed=12345):
    h = project_feasible(h_in, n)
    n = len(h)
    if n == 0:
        return h, float("inf")

    dx = 2.0 / n
    mid = n - 1

    t_cal = time.time()
    c_arr = np.correlate(h, 1.0 - h, mode="full")
    _ = np.correlate(h, 1.0 - h, mode="full")
    per_eval = time.time() - t_cal
    if per_eval < 1e-6:
        per_eval = 1e-5

    best_c = float(c_arr.max() * dx)
    rng = np.random.RandomState(seed)

    remaining = max(0.0, end_time - time.time())
    max_evals = int(min(20000, max(500, 0.5 * remaining / max(per_eval, 1e-9))))
    if max_evals < 300:
        return project_feasible(h, n), best_c

    active = get_active_lags(c_arr, dx, n)
    nonself_active = [lag for lag in active if lag != mid]

    evals = 0
    total = 0
    stall = 0

    while time.time() < end_time - 0.5 and evals < max_evals:
        if total % 80 == 0 or stall > 80 or len(active) == 0:
            c_arr = np.correlate(h, 1.0 - h, mode="full")
            best_c = float(c_arr.max() * dx)
            active = get_active_lags(c_arr, dx, n)
            nonself_active = [lag for lag in active if lag != mid]

        base_d = 0.002 * (0.5 ** min(6, stall // 100))
        if base_d < 1e-6:
            base_d = 1e-6

        improved_this_cycle = False

        for _batch in range(3):
            if time.time() > end_time - 0.5 or evals >= max_evals:
                break

            if len(active) == 0:
                typ = 0
            else:
                r = float(rng.rand())
                if r < 0.25:
                    typ = 0
                elif (mid in active) and r < 0.48:
                    typ = 1
                else:
                    typ = 2

            candidates = []

            if typ == 0:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                tries = 0
                while i == j and tries < 12:
                    j = int(rng.randint(n))
                    tries += 1
                if i != j:
                    sgn = 1.0 if rng.rand() < 0.5 else -1.0
                    candidates.append((i, j, sgn * base_d, -sgn * base_d))
                    candidates.append((i, j, -sgn * base_d, sgn * base_d))

            elif typ == 1:
                cnt = min(24, n)
                idxs = rng.choice(n, cnt, replace=False)
                vals = h[idxs]
                if len(vals) > 0:
                    i = int(idxs[int(np.argmax(vals))])
                    j = int(idxs[int(np.argmin(vals))])
                    if i != j and h[i] > 0.5 + 1e-12 and h[j] < 0.5 - 1e-12:
                        candidates.append((i, j, base_d, -base_d))
                        candidates.append((i, j, -base_d, base_d))

            else:
                if len(nonself_active) > 0:
                    lag = int(nonself_active[int(rng.randint(len(nonself_active)))])
                    t = abs(lag - mid)
                    max_i = n - t
                    if max_i > 0:
                        sample = min(32, max_i)
                        if sample == max_i:
                            idxs = np.arange(max_i)
                        else:
                            idxs = rng.randint(0, max_i, size=sample)
                        if lag < mid:
                            scores = h[idxs] * (1.0 - h[idxs + t])
                        else:
                            scores = h[idxs + t] * (1.0 - h[idxs])
                        pos = int(np.argmax(scores))
                        a = int(idxs[pos])
                        b = a + t
                        if lag < mid:
                            candidates.append((a, b, -base_d, base_d))
                            candidates.append((a, b, base_d, -base_d))
                        else:
                            candidates.append((b, a, -base_d, base_d))
                            candidates.append((b, a, base_d, -base_d))
                else:
                    i = int(rng.randint(n))
                    j = int(rng.randint(n))
                    tries = 0
                    while i == j and tries < 12:
                        j = int(rng.randint(n))
                        tries += 1
                    if i != j:
                        sgn = 1.0 if rng.rand() < 0.5 else -1.0
                        candidates.append((i, j, sgn * base_d, -sgn * base_d))
                        candidates.append((i, j, -sgn * base_d, sgn * base_d))

            if len(candidates) == 0:
                stall += 1
                continue

            for (i, j, di, dj) in candidates:
                if evals >= max_evals or time.time() > end_time - 0.5:
                    break
                if i == j:
                    continue
                accepted = False
                for mult in (1.0, 0.35):
                    di2 = di * mult
                    dj2 = dj * mult
                    hi = h[i] + di2
                    hj = h[j] + dj2
                    if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                        continue

                    vi = h[i]
                    vj = h[j]
                    h[i] = hi
                    h[j] = hj
                    evals += 1

                    c_new = np.correlate(h, 1.0 - h, mode="full")
                    c5_new = float(c_new.max() * dx)

                    if c5_new < best_c - 1e-16:
                        best_c = c5_new
                        c_arr = c_new
                        improved_this_cycle = True
                        accepted = True
                        break
                    else:
                        h[i] = vi
                        h[j] = vj

                if time.time() > end_time - 0.5:
                    break
                if accepted:
                    break

            if improved_this_cycle:
                stall = 0
            else:
                stall += 1

        total += 1

    h = project_feasible(h, n)
    c_final = compute_c5(h)
    if np.isfinite(c_final) and c_final < best_c:
        best_c = c_final
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    end = t0 + max(10.0, float(budget_s)) - 3.0
    rng = np.random.RandomState(seed)

    h_parent = None
    n_parent = 200
    try:
        if len(initial_h_values) > 0:
            h_parent = np.array(initial_h_values, dtype=float).copy()
            n_parent = len(h_parent)
    except Exception:
        h_parent = None

    if h_parent is None:
        h_parent = project_feasible(np.full(200, 0.5), 200)

    best_h = h_parent.copy()
    if len(best_h) > 600:
        best_h = interpolate_h(best_h, 256)
    elif len(best_h) < 80 and len(best_h) > 0:
        best_h = interpolate_h(best_h, 128)

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)

    # Primary optimization on the parent resolution.
    if time.time() < end - 30.0:
        alloc = min(430.0, max(45.0, (end - time.time()) * 0.40))
        h2, c2 = optimize_resolution(best_h, len(best_h), time.time() + alloc)
        if np.isfinite(c2) and c2 < best_c - 1e-15:
            best_c = c2
            best_h = h2

    # Symmetrized variant.
    if time.time() < end - 45.0:
        alloc = min(95.0, max(20.0, (end - time.time()) * 0.10))
        h_sym = symmetrize_h(best_h)
        h2, c2 = optimize_resolution(h_sym, len(best_h), time.time() + alloc)
        if np.isfinite(c2) and c2 < best_c - 1e-15:
            best_c = c2
            best_h = h2

    # Nearby and coarser/finer resolutions.
    base_n = len(best_h)
    seen = {base_n}
    cand_list = [
        base_n + 16,
        256,
        base_n + 32,
        288,
        base_n + 64,
        320,
        max(64, base_n - 16),
        max(64, base_n - 32),
    ]
    candidates = []
    for nn in cand_list:
        if nn < 64 or nn > 360:
            continue
        if nn in seen:
            continue
        seen.add(nn)
        candidates.append(int(nn))

    for nn in candidates:
        if time.time() > end - 75.0:
            break
        rem = end - time.time()
        alloc = min(130.0, max(20.0, 0.18 * rem))
        if alloc < 18.0:
            break
        h_try = project_feasible(interpolate_h(best_h, nn), nn)
        h2, c2 = optimize_resolution(h_try, nn, time.time() + alloc)
        if np.isfinite(c2) and c2 < best_c - 1e-15:
            best_c = c2
            best_h = h2

    # Randomized perturbations of the current best.
    scales = [0.015, 0.035, 0.07, 0.11]
    for scale in scales:
        if time.time() > end - 50.0:
            break
        rem = end - time.time()
        alloc = min(65.0, max(15.0, 0.12 * rem))
        if alloc < 12.0:
            break
        n_now = len(best_h)
        pert = best_h + rng.normal(0.0, scale, n_now)
        pert = project_feasible(pert, n_now)
        h2, c2 = optimize_resolution(pert, n_now, time.time() + alloc)
        if np.isfinite(c2) and c2 < best_c - 1e-15:
            best_c = c2
            best_h = h2

    # Final exact local refinement.
    if time.time() < end - 5.0:
        best_h, best_c = local_search(best_h, len(best_h), end - 0.2, seed=987654 + seed)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)

    return best_h, float(best_c), int(len(best_h))
