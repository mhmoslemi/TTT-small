            active = np.array([np.argmax(c_vec)])

        c_active = c_vec[active]
        sh = soft_beta * (c_active - c_active.max())
        w = np.exp(sh)
        w /= w.sum()

        grad = np.zeros(n)
        for wi, ai in zip(w, active):
            if wi < 1e-15:
                continue
            grad += wi * grad_shift(h, n, ai)

        v = mom * v - lr * grad
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            stagnation = 0
        else:
            h = h_new if c_new < max_val + 0.01 * max_val else h
            stagnation += 1
            if stagnation > 30:
                lr *= 0.7
                v *= 0.5
                if lr < 1e-12:
                    break
            elif stagnation > 10:
                lr *= 0.9
                v *= 0.7

    return h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 5.0), (1000, 50.0), (5000, 500.0), (20000, 5000.0), (100000, 50000.0)]

    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            def obj(x):
                return smooth_obj(x, n, alpha, None) + mu * (x.sum() - n / 2.0) ** 2

            def gobj(x):
                g = smooth_grad(x, n, alpha, None)
                g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
                return g

            res = minimize(obj, h, jac=gobj, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-16,
                                    'gtol': 1e-13, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fine_pair_search(h, n, deadline, radius=50):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            active = np.where(c_vec >= c_vec.max() - 0.01 * c_vec.max())[0]

            grad = np.zeros(n)
            for ai in active[:15]:
                grad += grad_shift(h, n, ai)

            hot = np.argsort(-np.abs(grad))[:100]

            for i in hot:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def binary_rounding(h, n):
    """Try rounding to binary and see if it improves."""
    c_cont = compute_c5(h, n)
    target = int(n // 2)
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    c_bin = compute_c5(h_bin, n)
    if c_bin < c_cont:
        return h_bin, c_bin
    # Try intermediate thresholds
    for frac in [0.4, 0.45, 0.55, 0.6]:
        thresh = np.quantile(h, 1 - frac)
        h_try = (h > thresh).astype(float)
        s = int(h_try.sum())
        if s > target:
            ones_idx = np.where(h_try == 1)[0]
            order = np.argsort(h[ones_idx])
            h_try[ones_idx[order[:s - target]]] = 0
        elif s < target:
            zeros_idx = np.where(h_try == 0)[0]
            order = np.argsort(-h[zeros_idx])
            h_try[zeros_idx[order[:target - s]]] = 1
        c_try = compute_c5(h_try, n)
        if c_try < c_cont:
            return h_try, c_try
    return h, c_cont

def random_restart_opt(h, n, deadline, n_restarts=3, perturb_scale=0.02):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    time_per = (deadline - time.time()) / max(1, n_restarts) - 5

    for r in range(n_restarts):
        if time.time() > deadline:
            break
        h_pert = h + np.random.randn(n) * perturb_scale
        h_pert = project_feasible(h_pert, n)
        t_sub = time.time() + max(10, time_per)
        h_r, c_r = momentum_active_opt(h_pert, n, t_sub,
                                        mom=0.9, lr_init=0.003, eps_frac=0.005,
                                        max_iter=5000)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS warmup at original resolution
    t1 = min(t0 + 80, deadline - 500)
    h1, c1 = run_lbfgs(best_h, n, t1,
        configs=[(200, 5.0), (1000, 50.0), (5000, 500.0), (50000, 5000.0)])
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Momentum active set optimization (aggressive)
    if time.time() < deadline - 350:
        t2 = min(time.time() + 180, deadline - 350)
        h2, c2 = momentum_active_opt(best_h, n, t2,
                                      mom=0.9, lr_init=0.005, eps_frac=0.002,
                                      soft_beta=800.0, max_iter=20000)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: Binary rounding heuristic
    h3, c3 = binary_rounding(best_h, n)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Try n=1000
    n2 = 1000
    if time.time() < deadline - 300:
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h1000 = np.interp(x_new, x_old, best_h)
        h1000 = project_feasible(h1000, n2)
        c1000 = compute_c5(h1000, n2)

        if time.time() < deadline - 250:
            t4a = min(time.time() + 80, deadline - 250)
            h1000, c1000 = run_lbfgs(h1000, n2, t4a,
                configs=[(500, 10.0), (3000, 100.0), (10000, 1000.0)])

        if time.time() < deadline - 150:
            t4b = min(time.time() + 120, deadline - 150)
            h1000, c1000 = momentum_active_opt(h1000, n2, t4b,
                                                mom=0.9, lr_init=0.003, eps_frac=0.002,
                                                soft_beta=800.0, max_iter=15000)

        # Binary rounding at n=1000
        h1000, c1000 = binary_rounding(h1000, n2)

        if c1000 < best_c:
            best_c = c1000
            best_h = h1000.copy()
            n = n2

    # Phase 5: Momentum at best resolution (gentle)
    if time.time() < deadline - 150:
        t5 = min(time.time() + 100, deadline - 150)
        h5, c5 = momentum_active_opt(best_h, n, t5,
                                      mom=0.85, lr_init=0.002, eps_frac=0.001,
                                      soft_beta=1000.0, max_iter=15000)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    # Phase 6: Random restarts
    if time.time() < deadline - 80:
        t6 = min(time.time() + 50, deadline - 80)
        h6, c6 = random_restart_opt(best_h, n, t6, n_restarts=2, perturb_scale=0.01)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()

    # Phase 7: Fine pair search
    if time.time() < deadline - 30:
        t7 = min(time.time() + 40, deadline - 30)
        h7, c7 = fine_pair_search(best_h, n, t7, radius=40)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()

    # Phase 8: Final momentum polish
    if time.time() < deadline - 5:
        t8 = deadline - 3
        h8, c8 = momentum_active_opt(best_h, n, t8,
                                      mom=0.8, lr_init=0.001, eps_frac=0.001,
                                      soft_beta=2000.0, max_iter=5000)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()

    # Final verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
