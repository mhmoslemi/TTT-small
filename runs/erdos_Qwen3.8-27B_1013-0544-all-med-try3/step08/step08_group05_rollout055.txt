        # Gradient of log-sum-exp w.r.t. x
        # d(log_sum)/dx[p] = sum_d p[d] * dg_d/dx[p]
        # dg_d/dx[p] = (1-x[p+d])*I(valid) - x[p-d]*I(valid)
        grad = np.zeros(n)
        for d_idx in range(n_shifts):
            d = m_range[d_idx]
            if p[d_idx] < 1e-10:
                continue
            # (1-x[p+d]) for valid p+d
            ip = slice(max(0, -d), min(n, n-d))  # p where p+d is valid: 0<=p+d<n => -d<=p<n-d
            lo1 = max(0, -d)
            hi1 = min(n, n - d)
            grad[lo1:hi1] += p[d_idx] * (1 - x[lo1+d:hi1+d])
            # -x[p-d] for valid p-d
            lo2 = max(0, d)
            hi2 = min(n, n + d)
            grad[lo2:hi2] -= p[d_idx] * x[lo2-d:hi2-d]
        
        return log_sum, grad * 1.0  # grad is already for the smooth function
        
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_grad(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_grad}]
    
    try:
        res = minimize(obj_and_grad, h, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
