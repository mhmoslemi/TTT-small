import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def all_cors(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smemax(vals, alpha):
    m = np.max(vals)
    sh = alpha * (vals - m)
    mx = np.max(sh)
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def obj_lbfgsb(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    val = smemax(c, alpha)
    val += mu * (h.sum() - n / 2.0) ** 2
    return val

def obj_slsqp(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    return smemax(c, alpha)

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(n)

def lbfgsb_run(h, n, alpha, mu, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_lbfgsb, h, method='L-BFGS-B', bounds=bounds,
                   args=(n, alpha, mu),
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                            'maxfun': maxiter * 3})
    return project_feasible(res.x, n)

def slsqp_run(h, n, alpha, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (n,)}]
    res = minimize(obj_slsqp, h, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(n, alpha),
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, n)
