import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_lag_values(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 400:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

    h = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_val, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            h = project_feasible(res.x, n)
    except Exception:
        pass
    if time.time() > deadline:
        pass
    return h

def gradient_descent(h, n, alpha, deadline, step_scale=0.01):
    """Simple projected gradient descent on the smooth max."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    step = step_scale
    for it in range(50000):
        if time.time() > deadline:
            break
        # Compute gradient of smooth max (numerical)
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 400:
            w = np.exp(shifted - mx)
            w /= w.sum()
        else:
            w = np.exp(shifted)
            w /= w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for p in range(2 * n - 1):
            wp = w[p]
            if wp < 1e-10:
                continue
            k = p - (n - 1)
            j_lo = max(0, -k)
            j_hi = min(n, n - k)
            if j_lo < j_hi:
                idx = np.arange(j_lo, j_hi)
                grad[idx] += wp * dx * (1 - h[idx + k])
            j_lo2 = max(0, k)
            j_hi2 = min(n, n + k)
            if j_lo2 < j_hi2:
                idx2 = np.arange(j_lo2, j_hi2)
                grad[idx2] -= wp * dx * h[idx2 - k]
        
        # Projected gradient step
        lr = step / (np.linalg.norm(grad) + 1e-10)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step *= 1.01
        else:
            step *= 0.7
            if step < 1e-8:
                break
    
    return best_h, best_c

        j_lo = max(0, -k)

    """Take one gradient step on smooth max with projection."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-12:
            continue
        k = p - (n - 1)
        # First factor contribution: x[j] in h[i]*(1-h[i+k]), i=j
        j_lo = max(0, -k)
        j_hi = min(n, n - k)
        if j_lo < j_hi:
            idx = np.arange(j_lo, j_hi)
            grad[idx] += wp * dx * (1.0 - h[idx + k])
        # Second factor contribution: x[j] in h[i-k]*(1-h[j]), i=j+k
        j_lo2 = max(0, k)
        j_hi2 = min(n, n + k)
        if j_lo2 < j_hi2:

            grad[idx2] -= wp * dx * h[idx2 - k]

    return h_new
