    # For shifts s from -(n-1) to (n-1):
    # delta(s) = h[b+s] - h[a+s] - h[a-s] + h[b-s]

    valid1 = (idx1 >= 0) & (idx1 < n)
    delta[valid1] += h[idx1[valid1]]

    valid2 = (idx2 >= 0) & (idx2 < n)

    delta[valid3] -= h[idx3[valid3]]
    
    # +h[b-s]
    idx4 = b - s
    valid4 = (idx4 >= 0) & (idx4 < n)
    delta[valid4] += h[idx4[valid4]]

        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def make_swap_delta(h, a, b, n, s_arr):
    L = 2*n - 1
    delta = np.zeros(L)
    
    # +I(a+s valid) : 1 where 0 <= a+s < n
    i_a_plus = a + s_arr
    v_ap = (i_a_plus >= 0) & (i_a_plus < n)
    delta[v_ap] += 1.0
    delta[v_ap] -= h[i_a_plus[v_ap]]  # -h[a+s]*I(valid)
    
    # -h[a-s]*I(a-s valid)
    i_a_minus = a - s_arr
    v_am = (i_a_minus >= 0) & (i_a_minus < n)
    delta[v_am] -= h[i_a_minus[v_am]]
    
    # -I(b+s valid)
    i_b_plus = b + s_arr
    v_bp = (i_b_plus >= 0) & (i_b_plus < n)
    delta[v_bp] -= 1.0
    delta[v_bp] += h[i_b_plus[v_bp]]  # +h[b+s]*I(valid)
    
    # +h[b-s]*I(b-s valid)
    i_b_minus = b - s_arr
    v_bm = (i_b_minus >= 0) & (i_b_minus < n)
    delta[v_bm] += h[i_b_minus[v_bm]]
    
    return delta
