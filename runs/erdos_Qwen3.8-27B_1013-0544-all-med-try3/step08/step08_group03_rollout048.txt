Let me code this up. The gradient of sum h[i]*h[i+k] with respect to h[j] is:
- For j in [0, n-k-1]: h[j+k] (from term h[j]*h[j+k])
- For j in [k, n-1]: h[j-k] (from term h[j-k]*h[j])
- Combined: h[j+k] (if j+k < n) + h[j-k] (if j-k >= 0)

So grad[j] = h[j+k]*(j+k<n) + h[j-k]*(j-k>=0)

And we want to ascend this while keeping sum(h) = n/2 and 0 <= h <= 1.

The projected gradient step would be:
h_new = project(h + lr * (grad - mean(grad)))

where project handles the [0,1] bounds and the sum constraint.

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def grad_smooth_max(h, n, alpha):

    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n

            grad[k:n] += wm * (-h[0:nv])

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])

            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 1.0), (500, 5.0), (2000, 10.0),
        (10000, 50.0), (50000, 200.0), (200000, 500.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 5,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def targeted_correlation_descent(h, n, deadline, top_m=3):
    """Iteratively reduce top correlations by ascending on h[i]*h[i+k]."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    lrs = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for lr in lrs:
        stalled = 0
        while time.time() < deadline and stalled < 3:
            c = compute_c5_vec(h, n)
            idx = np.argsort(-c)[:top_m]
            improved = False
            for mi in idx:
                k = int(mi) - (n - 1)
                grad = np.zeros(n)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] = h[k:n].copy()
                    grad[k:n] += h[0:nv]
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] = h[0:nv].copy()
                    grad[0:nv] += h[kk:n]
                grad -= grad.mean()
                h_new = np.clip(h + lr * grad, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
                    stalled = 0
                    break
                else:
                    # Try smaller step
                    h_new2 = np.clip(h + lr * 0.3 * grad, 0.0, 1.0)
                    h_new2 = project_feasible(h_new2, n)
                    c_new2 = compute_c5(h_new2, n)
                    if c_new2 < best_c - 1e-14:
                        best_c = c_new2
                        best_h = h_new2.copy()
                        h = h_new2
                        improved = True
                        stalled = 0
                        break
            if not improved:
                stalled += 1
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def multi_move_search(h, n, deadline, deltas, radius=40):
    """Try 2-simultaneous moves: shift from i to j and from p to q."""
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(0, n, 3):
                if time.time() > deadline:
                    return h, best_c
                lo1 = max(0, i - radius)
                hi1 = min(n, i + radius + 1)
                lo2 = max(0, i + 1 - radius)
                hi2 = min(n, i + 1 + radius + 1)
                for j in range(lo1, hi1):
                    if j == i:
                        continue
                    for p in range(lo2, hi2):
                        if p == i or p == j:
                            continue
                        # Move d from j to i, and d from p to... wait that changes sum
                        # Actually: h[i]+=d, h[j]-=d, h[p]+=d, h[q]-=d keeps sum
                        # Let me just do: h[i]+=d, h[j]-=d (single move)
                        pass
            if not improved:
                break
    return h, best_c

def gradient_descent_refine(h, n, deadline, lr_init=0.02, alpha_init=500.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    stall = 0
    for _ in range(50000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step_lr = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step_lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                stall = 0
                break
            step_lr *= 0.5
        if not improved:
            stall += 1
            lr *= 0.7
            alpha *= 1.3
            if stall > 20:
                break
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def random_perturb(h, n, rng, magnitude=0.05):
    h_p = h.copy()
    mask = rng.random(n) < 0.3
    h_p[mask] += rng.uniform(-magnitude, magnitude, mask.sum())
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B at original resolution (0-80s)
    t1 = min(t0 + 80, deadline - 600)
    if t1 > t0 + 10:
        best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_iters=150)

    # Phase 2: SLSQP at original resolution (80-150s)
    if time.time() < deadline - 500:
        t2 = min(time.time() + 60, deadline - 500)
        for a in [100000, 500000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_optimize(best_h, best_n, min(time.time() + 35, t2), alpha=a, maxiter=1500)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Targeted correlation descent (150-230s)
    if time.time() < deadline - 400:
        t3 = min(time.time() + 70, deadline - 400)
        h3, c3 = targeted_correlation_descent(best_h, best_n, t3, top_m=3)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Gradient descent refinement (230-310s)
    if time.time() < deadline - 350:
        t4 = min(time.time() + 60, deadline - 350)
        h4, c4 = gradient_descent_refine(best_h, best_n, t4, lr_init=0.01, alpha_init=500.0)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Pair search at original resolution (310-400s)
    if time.time() < deadline - 280:
        t5 = min(time.time() + 70, deadline - 280)
        best_h, best_c = pair_search(best_h, best_n, t5,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005],
                                     radius=50)

    # Phase 6: Try higher resolutions
    for n_try in [400, 500, 600]:
        if time.time() > deadline - 180:
            break
        h_up = upsample(best_h, best_n, n_try)
        c_up = compute_c5(h_up, n_try)
        # L-BFGS at new resolution
        t_up = min(time.time() + 50, deadline - 130)
        if t_up > time.time() + 15:
            h_up, c_up = lbfgs_optimize(h_up, n_try, t_up, max_iters=120)
        # SLSQP
        if time.time() < deadline - 100:
            h_up, c_up = slsqp_optimize(h_up, n_try, min(time.time() + 25, deadline - 80),
                                        alpha=200000, maxiter=800)
        # Targeted descent
        if time.time() < deadline - 70:
            h_up, c_up = targeted_correlation_descent(h_up, n_try,
                                                      min(time.time() + 15, deadline - 55), top_m=3)
        # Quick pair search
        if time.time() < deadline - 45:
            h_up, c_up = pair_search(h_up, n_try, min(time.time() + 10, deadline - 35),
                                     [0.01, 0.005, 0.002],
                                     radius=30)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            best_n = n_try

    # Phase 7: Try nearby resolutions of best_n
    if time.time() < deadline - 100:
        for dn in [-2, -1, 1, 2, 3]:
            if time.time() > deadline - 60:
                break
            n_near = best_n + dn
            if n_near < 100 or n_near > 2000:
                continue
            h_near = upsample(best_h, best_n, n_near)
            c_near = compute_c5(h_near, n_near)
            t_near = min(time.time() + 20, deadline - 40)
            if t_near > time.time() + 5:
                h_near, c_near = lbfgs_optimize(h_near, n_near, t_near, max_iters=80)
            if c_near < best_c:
                best_c = c_near
                best_h = h_near
                best_n = n_near

    # Phase 8: Intensive local search at best resolution
    if time.time() < deadline - 25:
        t8 = min(time.time() + 40, deadline - 15)
        best_h, best_c = pair_search(best_h, best_n, t8,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=60)

    # Phase 9: Final targeted descent + gradient refine
    if time.time() < deadline - 10:
        t9 = min(time.time() + 20, deadline - 5)
        best_h, best_c = targeted_correlation_descent(best_h, best_n, t9, top_m=2)

    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, best_n, deadline - 1,
                                     [0.001, 0.0005, 0.0002, 0.0001],
                                     radius=40)

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
