import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def compute_delta_for_swap(i, j, h, n, s_all):
    """Compute the change in correlation when h[i] 0->1 and h[j] 1->0.
    s_all is a precomputed array of all shifts.
    Returns delta array of length 2n-1."""
    valid_i_plus = (i + s_all >= 0) & (i + s_all < n)
    valid_i_minus = (i - s_all >= 0) & (i - s_all < n)
    valid_j_plus = (j + s_all >= 0) & (j + s_all < n)
    valid_j_minus = (j - s_all >= 0) & (j - s_all < n)
    
    h_ip = np.zeros(2*n-1)
    h_im = np.zeros(2*n-1)
    h_jp = np.zeros(2*n-1)
    h_jm = np.zeros(2*n-1)
    
    h_ip[valid_i_plus] = h[i + s_all[valid_i_plus]]
    h_im[valid_i_minus] = h[i - s_all[valid_i_minus]]
    h_jp[valid_j_plus] = h[j + s_all[valid_j_plus]]
    h_jm[valid_j_minus] = h[j - s_all[valid_j_minus]]
    
    return -h_ip - h_im + h_jp + h_jm

def binary_local_search(h_init, n, deadline, max_sweeps=5000, n_random_trials=30, 
                        annealing=False, seed=42):

    dx = 2.0 / n
    target = n // 2
    rng = np.random.RandomState(seed)
    
    h = to_binary(h_init, n)
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    s_all = np.arange(-(n-1), n)
    
    ones = h == 1
    zeros = ~ones
    n_ones = target
    n_zeros = n - target
    
    # Simulated annealing parameters
    if annealing:
        T = 0.01  # initial temperature
        T_min = 1e-5
        cooling = 0.999
    
    no_improve = 0
    max_no_improve = 200
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        improved_this_sweep = False
        
        for trial in range(n_random_trials):
            if time.time() > deadline:
                break
            
            # Pick random zero to flip to one, and random one to flip to zero
            zi = np.where(zeros)[0]
            oi = np.where(ones)[0]
            if len(zi) == 0 or len(oi) == 0:
                break
            
            i = zi[rng.randint(len(zi))]
            j = oi[rng.randint(len(oi))]
            if i == j:
                continue
            
            delta = compute_delta_for_swap(i, j, h, n, s_all)
            new_corr = corr + delta
            new_c = new_corr.max() * dx

            if new_c < cur_c - 1e-12:
                accept = True
            elif annealing and new_c > cur_c:

                if new_c - cur_c < T * 5 and rng.random() < np.exp(-(new_c - cur_c) / T):
                    accept = True
            
            if accept:
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = new_c
                ones[i] = False
                ones[j] = True
                zeros[i] = True
                zeros[j] = False
                improved_this_sweep = True
                no_improve = 0
                
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        
        if not improved_this_sweep:
            no_improve += 1

                        break
                else:

                        ones[j] = True
                        zeros[i] = True
                        zeros[j] = False

                    cur_c = corr.max() * dx
                    no_improve = 0
            
            if not annealing and no_improve > 500:
                break
    
    return best_h, best_c

def slsqp_refine(h_init, n, deadline, alphas=[1000, 5000, 50000, 500000]):
    """SLSQP refinement with log-sum-exp smoothing."""
    dx = 2.0 / n
    target = n / 2.0
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            shifted = (g - m) * alpha
            return m + np.log(np.exp(shifted).sum()) / alpha
        
        def cons(x):
            return x.sum() - target
        
        bounds = [(0.0, 1.0)] * n
        cons_dict = [{'type': 'eq', 'fun': cons}]
        
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons_dict,
                          options={'maxiter': 500, 'ftol': 1e-12})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin
