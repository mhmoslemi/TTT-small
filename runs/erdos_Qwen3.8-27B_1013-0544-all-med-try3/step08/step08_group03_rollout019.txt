def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d

                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def binary_push(h, n, deadline):
    """Push values toward 0 or 1 while maintaining sum constraint."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for sweep in range(200):
        if time.time() > deadline:
            break
        improved = False
        # Try to push each value toward 0 or 1
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            # Try pushing toward 1 (if h[i] > 0.5)
            # Or toward 0 (if h[i] < 0.5)
            # Need to maintain sum = n/2, so pair with another position
            # Simple approach: find a partner
            target = 1.0 if h[i] > 0.5 else 0.0
            needed = target - h[i]
            if abs(needed) < 1e-6:
                continue
            # Find partner j that can absorb -needed
            # Try the position most opposite
            if needed > 0:
                # Need to decrease some h[j]
                candidates = np.where(h < 0.99)[0]
            else:
                # Need to increase some h[j]
                candidates = np.where(h > 0.01)[0]
            if len(candidates) == 0:
                continue
            # Just try the one with most room
            if needed > 0:
                j = candidates[np.argmin(h[candidates])]
            else:
                j = candidates[np.argmax(h[candidates])]
            if j == i:
                continue
            step = min(abs(needed), abs(-needed))  # limited by partner
            if needed > 0:
                step = min(needed, h[j])
            else:
                step = min(-needed, 1.0 - h[j])
            if step < 1e-6:
                continue
            h[i] += np.sign(needed) * step
            h[j] -= np.sign(needed) * step
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                improved = True
            else:
                h[i] -= np.sign(needed) * step
                h[j] += np.sign(needed) * step
        if not improved:
            break
    return best_h, best_c

    idx = np.argpartition(c, -p)[-p:]
    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)


def lag_grad_single(h, n, m):
    """Gradient of the m-th correlate entry w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = m - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += dx * (1.0 - h[k:n])
        grad[k:n] += dx * (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += dx * (1.0 - h[0:nv])
        grad[0:nv] += dx * (-h[kk:n])
    return grad


def smooth_max_grad(h, n, alpha):
    c = compute_c5_all(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-12:
            continue
        lm = w[m]
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += lm * (1.0 - h[k:n])
            grad[k:n] += lm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += lm * (1.0 - h[0:nv])
            grad[0:nv] += lm * (-h[kk:n])
    return grad


def top_p_grad(h, n, p):
    c = compute_c5_all(h, n)
    idx = np.argpartition(c, -p)[-p:]
    grad = np.zeros(n)
    for m in idx:
        grad += lag_grad_single(h, n, m)
    return grad


def pen_obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g


def slq_obj(x, n, alpha):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha


def slq_grad(x, n, alpha):
    return smooth_max_grad(x, n, alpha)


def slq_con(x, n):
    return float(x.sum() - n / 2.0)


def slq_jac(x, n):
    return np.ones(n)


def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (50, 1.0), (200, 5.0), (1000, 10.0),
        (5000, 50.0), (20000, 200.0), (100000, 1000.0),
        (500000, 5000.0), (2000000, 20000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c


def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slq_con, 'jac': slq_jac, 'args': (n,)}]
    try:
        t_end = time.time() + (deadline - time.time())
        res = minimize(slq_obj, h, jac=slq_grad, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)


def projected_subgrad(h, n, deadline, p=5, lr_init=0.005, max_iter=5000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = top_p_grad(h, n, p)
        g += 0.1 * (h.sum() - n / 2.0) * np.ones(n) / n
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        improved = False
        step = lr
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
        if it % 200 == 0:
            p = max(1, p // 2)
    return best_h, best_c


def pair_search_all(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    improved_total = True
    for d in deltas:
        while improved_total and time.time() < deadline:
            improved_total = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_total = True

                            h[i] -= d

def swap_search(h, n, deadline):
    best_c = compute_c5(h, n)
    improved_total = True
    while improved_total and time.time() < deadline:
        improved_total = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-10:
                    continue
                h[i], h[j] = h[j], h[i]
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved_total = True
                else:
                    h[i], h[j] = h[j], h[i]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c


def make_stripy(n, n_strips):
    h = np.zeros(n)
    per = n // n_strips
    for s in range(n_strips):
        lo = s * per
        hi = min(lo + per // 2, n)
        h[lo:hi] = 0.9
        lo2 = min(lo + per // 2, n)
        hi2 = min(lo + per, n)
        h[lo2:hi2] = 0.1
    return project_feasible(h, n)


def make_sinusoidal(n, freq, amp):
    i = np.arange(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * i / n)
    return project_feasible(h, n)
