import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    # Water-filling projection
    target = n / 2.0
    h = h.copy()
    # Bisection on lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid  # need to increase lambda (decrease x)
        else:
            hi = mid  # need to decrease lambda (increase x)
    x = np.clip(h - mid, 0.0, 1.0)
    # Tiny adjustment for floating point
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        room = (1.0 - x) if diff > 0 else x
        tot = room.sum()
        if tot > 1e-14:
            x += (diff / tot) * room
    return np.clip(x, 0.0, 1.0)

def c5_grad_at_shift(h, n, shift_idx):
    """Gradient of c_shift with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return smooth_grad_with_weights(x, n, w, alpha, mu)

def smooth_grad_with_weights(x, n, w, alpha, mu):
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def adam_optimize(h, n, deadline, lr=0.005, max_iter=20000, alpha=10000, mu=1.0):
    """Adam optimization with log-sum-exp smoothing and projection."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    for i in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        c = compute_c5_vec(h, n)
        m_c = c.max()
        sh = alpha * (c - m_c)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        
        # Compute gradient
        grad = smooth_grad_with_weights(h, n, w, alpha, mu)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_next = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h_next = project_feasible(h_next, n)
        
        c_new = compute_c5(h_next, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_next.copy()
        h = h_next
        
        # Periodic re-computation
        if i % 5000 == 0 and i > 0:
            lr *= 0.5
    
    return best_h, best_c

        threshold = max_val - top_frac * max_val
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-c_vec)[:10]
        
        for mu in [0.01, 0.1, 1.0]:
            if time.time() > deadline - 2:

                    else:
                        val = mv + np.log(np.sum(np.exp(sh))) / alpha
