import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, constraint_weight):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += constraint_weight * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, constraint_weight):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * constraint_weight * (x.sum() - n / 2.0)
    return grad

def do_lbfgs(h, n, deadline, alpha, mu, maxiter=5000):

    try:

                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,

                               'gtol': 1e-14, 'maxfun': 20000, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

    """Coordinate descent: for each coordinate, find best value in [0,1]."""

        old_val = h[i]

        candidates.add(0.5)
        # Points near old_val
        for delta in [1e-3, 5e-3, 1e-2, 5e-2, -1e-3, -5e-3, -1e-2, -5e-2]:

            if 0 <= t <= 1:

        # Quantized values

                best_t = t
        
        h[i] = best_t
    
    # Project to feasible set
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def do_lbfgs(h, n, deadline, alpha, mu, maxiter=3000, maxfun=10000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxfun, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def coord_descent_1d(h, n, i, c5_vec, num_grid=200):
    """Solve exact 1D subproblem for coordinate i given current correlation."""
    dx = 2.0 / n
    old = h[i]
    center = n - 1
    s = np.zeros(2 * n - 1)
    b = np.zeros(2 * n - 1)
    
    for mm in range(2 * n - 1):
        k = mm - center
        if k == 0:
            # Quadratic: g_0(t) = t - t^2 + b_0
            b[mm] = c5_vec[mm] - old * (1.0 - old)
            s[mm] = 0.0  # marker
        else:
            # Linear: g_k(t) = t * s_k + b_k
            slope = 0.0
            ik = i - k
            if 0 <= ik < n:
                slope += (1.0 - h[ik])
            ik2 = i + k
            if 0 <= ik2 < n:
                slope -= h[ik2]
            s[mm] = slope
            b[mm] = c5_vec[mm] - old * slope
    
    # Evaluate f(t) = max(t - t^2 + b[center], max_{mm!=center}(t*s[mm]+b[mm]))
    best_t = old
    best_f = float('inf')
    ts = np.linspace(0.0, 1.0, num_grid)
    for t in ts:
        val_0 = t - t * t + b[center]
        vals = t * s + b
        vals[center] = val_0
        f_val = np.max(vals)
        if f_val < best_f:
            best_f = f_val
            best_t = t
    
    # Refine around best_t
    if 0 < best_t < 1:
        lo = max(0, best_t - 2.0 / num_grid)
        hi = min(1, best_t + 2.0 / num_grid)
        ts_fine = np.linspace(lo, hi, 50)
        for t in ts_fine:
            val_0 = t - t * t + b[center]
            vals = t * s + b
            vals[center] = val_0
            f_val = np.max(vals)
            if f_val < best_f:
                best_f = f_val
                best_t = t
    
    return best_t, best_f

def coord_descent_pass(h, n, deadline, num_grid=200, random_order=False):
    """Full coordinate descent pass."""
    target = n / 2.0
    c5_vec = compute_c5_vec(h, n)
    best_c = c5_vec.max()
    
    if random_order:
        order = np.random.permutation(n)
    else:
        order = np.arange(n)
    
    for i in order:
        if time.time() > deadline:
            break
        best_t, _ = coord_descent_1d(h, n, i, c5_vec, num_grid)
        if abs(best_t - h[i]) > 1e-10:
            h[i] = best_t
            c5_vec = compute_c5_vec(h, n)
            new_c = c5_vec.max()
            if new_c < best_c:
                best_c = new_c
    
    # Project to maintain sum constraint
    h = project_feasible(h, n)
    c5_vec = compute_c5_vec(h, n)
    best_c = c5_vec.max()
    return h, best_c, c5_vec

def active_shift_lbfgs(h, n, deadline, alpha, mu, top_k=5, maxiter=3000):
    """L-BFGS focusing on top-k active shifts."""
    c5_vec = compute_c5_vec(h, n)
    max_val = c5_vec.max()
    threshold = max_val - max(1e-8, 0.005 * max_val)
    active = np.where(c5_vec >= threshold)[0]
    if len(active) > top_k:
        active = active[np.argsort(-c5_vec[active])[:top_k]]
    
    def top_obj(x, n, act):
        c = compute_c5_vec(x, n)
        vals = c[act]
        mx = vals.max()
        sh = alpha * (vals - mx)
        smx = sh.max()
        if smx > 500:
            val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
        else:
            val = mx + np.log(np.sum(np.exp(sh))) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    
    def top_grad(x, n, act):
        c = compute_c5_vec(x, n)
        vals = c[act]
        mx = vals.max()
        sh = alpha * (vals - mx)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for wi, idx in zip(w, act):
            k = idx - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])

                kk = -k
                nv = n - kk

                grad[0:nv] += wm * (-x[kk:n])
        grad += 2.0 * mu * (x.sum() - n / 2.0)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(top_obj, h, jac=top_grad,
                      args=(n, active), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': 8000, 'maxls': 40})

        return h_new, c_new

        return h.copy(), compute_c5(h, n)

            best_c = c_up
            best_h = h_up
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    return best_h, best_c, n_orig
