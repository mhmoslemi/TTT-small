    """Minimize overlap at shift k_idx, with penalty for other active shifts."""
    c = compute_c5_vec(x, n)
    main_val = c[k_idx]

    # Soft penalty for constraint violations
    penalty = 0.0
    for cv in constr_vals:

            penalty += alpha_constraint * (cv - main_val) ** 2
    return float(main_val + penalty)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def shift_obj_pen(x, n, target_idx, penalty_indices, pen_alpha, mu):
    c = compute_c5_vec(x, n)
    main_val = c[target_idx]
    pen = 0.0
    for pi in penalty_indices:
        if c[pi] > main_val:
            pen += pen_alpha * (c[pi] - main_val) ** 2
    val = main_val + pen
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def shift_grad_pen(x, n, target_idx, penalty_indices, pen_alpha, mu):
    c = compute_c5_vec(x, n)
    main_val = c[target_idx]
    dx = 2.0 / n
    k = target_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += dx * (1.0 - x[k:n])
        grad[k:n] += dx * (-x[0:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += dx * (1.0 - x[0:nv])
        grad[0:nv] += dx * (-x[kk:n])
    for pi in penalty_indices:
        if c[pi] > main_val:
            diff = c[pi] - main_val
            kc = pi - (n - 1)
            if kc >= 0:
                nvc = n - kc
                grad[0:nvc] += 2.0 * pen_alpha * diff * dx * (1.0 - x[kc:n])
                grad[kc:n] += 2.0 * pen_alpha * diff * dx * (-x[0:nvc])
            else:
                kkc = -kc
                nvc = n - kkc
                grad[kkc:n] += 2.0 * pen_alpha * diff * dx * (1.0 - x[0:nvc])
                grad[0:nvc] += 2.0 * pen_alpha * diff * dx * (-x[kkc:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu, max_iters):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = min(max_iters[i] if i < len(max_iters) else 2000, max(500, int(remaining * 3)))
        try:

                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': maxit, 'ftol': 1e-20,

                                  'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def pair_search(h, n, deadline, radius=60, deltas=None, n_passes=3):

                for i in range(n):

                        break
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                    for j in range(lo, hi):
                        if i == j:
                            continue

                            h[i] += d
                            h[j] -= d
                            h = np.clip(h, 0.0, 1.0)
                            c = compute_c5(h, n)
                            if c < best_c - 1e-16:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i] -= d
                                h[j] += d
                                h = np.clip(h, 0.0, 1.0)
                        if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                            h[i] -= d
                            h[j] += d
                            h = np.clip(h, 0.0, 1.0)

                                improved = True

def make_binary_from_continuous(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)

def golden_ratio_start(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        if (i * phi) % 1.0 < 0.5:
            h[i] = 1.0
    return project_feasible(h, n)

def thue_morse_start(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        if cnt % 2 == 0:
            h[i] = 1.0
    return project_feasible(h, n)

def periodic_start(n, period, ones_count):
    h = np.zeros(n)
    ones_in_period = min(ones_count, period)
    for i in range(n):
        if (i % period) < ones_in_period:
            h[i] = 1.0
    return project_feasible(h, n)
