import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_shift_matrix(h, n):
    tw = 2 * n - 1
    rows = np.arange(n)[:, None]
    cols = np.arange(tw)[None, :]
    diff = cols - rows
    valid = (diff >= 0) & (diff < n)
    M = np.where(valid, h[diff], 0.0)
    return M

def lbbgsm(h_init, n, alpha, maxiter, deadline, penalty=1e4):
    dx = 2.0 / n
    tw = 2 * n - 1

    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        max_s = shifted.max()
        exp_s = np.exp(shifted - max_s)
        lse = m + (max_s + np.log(exp_s.sum())) / alpha
        weights = exp_s / exp_s.sum()
        conv = np.correlate(weights, x, mode='full')
        grad_corr = dx * (1.0 - 2.0 * conv[n-1:2*n-1])
        pen = penalty * (x.sum() - n / 2.0)
        grad = grad_corr + 2.0 * penalty * (x.sum() - n / 2.0)
        return lse + penalty * (x.sum() - n / 2.0)**2, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h_init, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def vectorized_binary_hill_climb(h, n, deadline, max_iters=100000):
    dx = 2.0 / n
    tw = 2 * n - 1
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_cost = corr.max() * dx
    best_cost = cur_cost
    best_h = h.copy()
    consecutive_no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        M = build_shift_matrix(h, n)
        d = 2.0 * (M[ones][:, None, :] - M[zeros][None, :, :])
        new_costs = (d + corr[None, None, :]).max(axis=2) * dx
        flat_idx = new_costs.argmin()
        bi, bj = np.unravel_index(flat_idx, new_costs.shape)
        min_cost = new_costs[bi, bj]

        if min_cost >= cur_cost - 1e-13:
            consecutive_no_improve += 1
            if consecutive_no_improve > 2:
                break
            # Try a random perturbation
            ri = zeros[np.random.randint(len(zeros))]
            rj = ones[np.random.randint(len(ones))]
            rd = 2.0 * (M[rj] - M[ri])
            rcost = (corr + rd).max() * dx
            if rcost < cur_cost - 1e-13:
                h[ri], h[rj] = 1.0, 0.0
                corr += rd
                cur_cost = rcost
                consecutive_no_improve = 0
            else:
                break
        else:
            consecutive_no_improve = 0
            j = ones[bi]
            i = zeros[bj]
            h[i], h[j] = 1.0, 0.0
            corr = corr + d[bi, bj, :]
            cur_cost = min_cost
            if cur_cost < best_cost - 1e-13:
                best_cost = cur_cost
                best_h = h.copy()

    return best_h, best_cost

def binary_from_continuous(hc, n):
    target = n // 2
    idx = np.argsort(hc)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h
