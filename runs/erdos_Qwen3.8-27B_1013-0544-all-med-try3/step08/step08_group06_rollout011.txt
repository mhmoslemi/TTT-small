import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh = np.clip(sh, -400, 400)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=4000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def corr_gradient(h, n, k_idx):
    dx = 2.0 / n
    offset = k_idx - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        j1 = i - offset
        if 0 <= j1 < n:
            val += (1.0 - h[j1])
        j2 = i + offset
        if 0 <= j2 < n:
            val -= h[j2]
        grad[i] = val
    return grad * dx

    best_c = compute_c5(h, n)
    for lr in lr_scales:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        stall = 0

            if time.time() > deadline:
                break
            c = compute_c5_vec(h_cur, n)
            c_max = c.max()
            if c_max >= best_c - 1e-15:
                h_cur = best_h.copy()
                stall += 1
            else:
                stall = 0
            if stall > 30:
                break
            top_indices = np.argsort(c)[-n_top:]
            grad = np.zeros(n)
            for k_idx in top_indices:
                grad += corr_gradient(h_cur, n, int(k_idx))
            grad /= n_top
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
            h_cur = h_new
    return best_h, best_c

                amp = 0.04 + 0.04 * rng.rand()
                x = np.linspace(0, 1, n)
                hp = best_h + amp * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
            else:
                freq1 = rng.randint(1, 4)
                freq2 = rng.randint(4, 12)
                amp = 0.03 + 0.05 * rng.rand()
                x = np.linspace(0, 1, n)
                hp = best_h + amp * (np.sin(2 * np.pi * freq1 * x) + 0.5 * np.cos(2 * np.pi * freq2 * x))
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 18) * 0.3)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        if n <= 200:
            remaining_time = deadline - time.time() - 2
            best_h, best_c = pair_search(best_h, n, time.time() + remaining_time,
                                         [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        else:
            remaining_time = deadline - time.time() - 2
            best_h, best_c = targeted_pair_search(best_h, n, time.time() + remaining_time,
                                                  [0.002, 0.001, 0.0005, 0.0002], n_top=20)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        best_h, best_c = subgradient_refine(best_h, n, deadline - 1, n_top=8,
                                            lr_scales=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
