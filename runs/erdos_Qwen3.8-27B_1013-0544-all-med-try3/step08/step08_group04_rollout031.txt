            if tot < 1e-15:

    return float(m + log_sum / alpha)

    if time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(2, int(remaining / 50))
        rng = np.random.RandomState(seed + 7777)
        for r in range(n_restarts):
            if time.time() > deadline - 22:
                break
            scale = 0.01 + 0.06 * (r / max(1, n_restarts - 1))
            # Mix of Gaussian and structured perturbation
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 18) * 0.35)
            if alloc < 18:
                break
            h_opt, c_opt = optimize_resolution(hp, n, time.time() + alloc, seed + 5000 + r * 13)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Intensive final local search
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        radius = min(70, n // 3)
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        h_pol, c_pol = local_search(best_h.copy(), n, deadline - 5, steps, radius, seed=seed + 999)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol
    
    # Phase 5: Ultra-final SLSQP + micro local search
    if best_h is not None and time.time() < deadline - 4:
        n = len(best_h)
        ht = slsqp_pass(best_h, n, 10000000, maxiter=10000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = project_feasible(ht, n)
        if time.time() < deadline - 1:
            h_fin, c_fin = local_search(best_h, n, deadline - 0.5,
                                         [0.00005, 0.00002, 0.00001], 30, seed=seed + 1)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin
    
    # Ensure feasibility and return
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
