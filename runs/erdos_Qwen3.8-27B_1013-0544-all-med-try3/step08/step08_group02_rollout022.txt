For coordinate descent: fix all variables except h[j], and find the h[j] that minimizes C₅ = max_k c_k. Since each c_k is linear in h[j] (for fixed other variables), C₅ as a function of h[j] is the max of linear functions, which is convex. The minimum is where the top two or more linear functions cross.

This is actually a nice property! For each coordinate j:
- c_k(j) = a_k * h[j] + b_k (linear in h[j])
- C₅(j) = max_k c_k(j) is a piecewise linear convex function of h[j]
- The minimum is at a "kink" point where two or more c_k values are equal and maximal
- Or at the boundary h[j] = 0 or h[j] = 1

So for each coordinate, I can:
1. Compute all c_k as functions of h[j] (they're linear: c_k = α_k * h_j + β_k)
2. Find the point in [0,1] that minimizes max_k(α_k * h_j + β_k)
3. This is a 1D linear programming problem (or just find the min of the upper envelope)

The min of max of linear functions over [0,1] occurs at:
- A point where two lines cross and are both ≥ all others
- Or at 0 or 1

Finding this exactly: The upper envelope of M lines over [0,1] changes at O(M) points. The minimum of the upper envelope is at one of these points or the endpoints.

For n=500, there are 999 shift values. For each coordinate, I need to find the min of the max of 999 linear functions over [0,1]. This can be done in O(n log n) by finding the upper envelope of the lines.

Actually, a simpler approach: binary search on the target t. For a given t, check if there exists h_j ∈ [0,1] such that α_k * h_j + β_k ≤ t for all k. This is:
- For α_k > 0: h_j ≤ (t - β_k) / α_k
- For α_k < 0: h_j ≥ (t - β_k) / α_k (note: dividing by negative flips inequality)
- For α_k = 0: need β_k ≤ t

So: max_{k: α_k<0} (t-β_k)/α_k ≤ h_j ≤ min_{k: α_k>0} (t-β_k)/α_k, and h_j ∈ [0,1], and all β_k ≤ t for α_k=0.

Binary search on t to find the minimum feasible t. This is O(n log n) per coordinate.

Doing this for all n=500 coordinates in one full pass: O(n² log n) ≈ 500 * 1000 * 10 = 5 × 10⁶ operations. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    lo = float(h.min()) - 1.0
    hi = 1.0 - float(h.min())
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_proj.sum()
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h_proj
            tot = room.sum()
            if tot < 1e-15:
                break
            h_proj += (diff / tot) * room
        else:
            tot = h_proj.sum()
            if tot < 1e-15:
                break
            h_proj += (diff / tot) * h_proj
        diff = target - h_proj.sum()
    return np.clip(h_proj, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:

    return grad

    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:

                          args=(n, alpha, mu), method='L-BFGS-B',

                                   'gtol': 1e-18, 'maxfun': max(maxit * 10, 10000),
                                   'maxls': 60, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active(h, n, deadline, top_k, alphas):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for round_idx in range(30):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(-c_vec[active])[:top_k]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(400, int((deadline - time.time()) * 3))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': max(maxit * 10, 8000),
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new <= best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def coord_descent(h, n, deadline, max_sweeps=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    num_shifts = 2 * n - 1
    sweep = 0
    while time.time() < deadline and sweep < max_sweeps:
        sweep += 1
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old_hj = best_h[j]
            current_c = compute_c5_vec(best_h, n)
            alpha_arr = np.zeros(num_shifts)
            beta_arr = np.zeros(num_shifts)
            s_arr = np.arange(num_shifts) - (n - 1)
            for idx in range(num_shifts):
                s = s_arr[idx]
                if s == 0:
                    beta_arr[idx] = current_c[idx]
                    continue
                a = 0.0
                if 0 <= j + s and j + s < n:
                    a += (1.0 - best_h[j + s])
                if 0 <= j - s and j - s < n:
                    a -= best_h[j - s]
                alpha_arr[idx] = dx * a
                beta_arr[idx] = current_c[idx] - alpha_arr[idx] * best_h[j]
            # Binary search on target t for the linear constraints (s != 0)
            # and handle s=0 separately by checking candidate h values
            zero_mask = np.abs(alpha_arr) < 1e-16
            pos_mask = alpha_arr > 1e-16
            neg_mask = alpha_arr < -1e-16
            t_lo = 0.0
            if np.any(zero_mask):
                t_lo = max(t_lo, float(np.max(beta_arr[zero_mask])))
            t_hi = current_c.max() + 1e-12
            for _ in range(50):
                t_mid = (t_lo + t_hi) / 2.0
                lo_b = 0.0
                hi_b = 1.0
                if np.any(pos_mask):
                    hi_b = min(hi_b, float(np.min((t_mid - beta_arr[pos_mask]) / alpha_arr[pos_mask])))
                if np.any(neg_mask):
                    lo_b = max(lo_b, float(np.max((t_mid - beta_arr[neg_mask]) / alpha_arr[neg_mask])))
                if hi_b < lo_b - 1e-13:
                    t_lo = t_mid
                else:
                    t_hi = t_mid
            t_star = t_hi
            # Get h[j] from feasibility at t_star
            lo_b = 0.0
            hi_b = 1.0
            if np.any(pos_mask):
                hi_b = min(hi_b, float(np.min((t_star - beta_arr[pos_mask]) / alpha_arr[pos_mask])))
            if np.any(neg_mask):
                lo_b = max(lo_b, float(np.max((t_star - beta_arr[neg_mask]) / alpha_arr[neg_mask])))
            candidates = [0.0, 1.0, np.clip((lo_b + hi_b) / 2.0, 0.0, 1.0)]
            candidates.append(np.clip(lo_b, 0.0, 1.0))
            candidates.append(np.clip(hi_b, 0.0, 1.0))
            best_candidate = old_hj
            best_candidate_c = best_c
            for cv in candidates:
                best_h[j] = cv
                # Check s=0 constraint: c_0 = dx*(h[j]-h[j]^2) + beta_arr[n-1]
                c0_new = dx * (cv - cv * cv) + beta_arr[n - 1]
                actual_max = max(t_star, c0_new)
                # Verify by computing actual c
                c_check = compute_c5(best_h, n)
                if c_check < best_candidate_c - 1e-15:
                    best_candidate_c = c_check
                    best_candidate = cv
            best_h[j] = best_candidate
            if best_candidate != old_hj:
                improved = True
            if best_candidate_c < best_c:
                best_c = best_candidate_c
        if not improved and sweep > 20:
            break
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=25):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h[i] -= d
                            best_h[j] += d
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return project_feasible(best_h, n), best_c

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.004
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.0008 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        h_new = project_feasible(best_h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.97
        best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: L-BFGS at original resolution
    p1_end = t0 + 200
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.008,
        maxiter_list=[2000, 2000, 3000, 3000, 3000, 4000, 4000, 5000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    best_n = len(best_h)
    
    # Phase 2: Active-set focused
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 120, deadline - 400)
        h2, c2 = run_active(best_h, best_n, p2_end, top_k=5,
            alphas=[50000, 200000, 1000000, 5000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
        best_n = len(best_h)
    
    # Phase 3: Coordinate descent (key innovation)
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 200, deadline - 350)
        h3, c3 = coord_descent(best_h, best_n, p3_end, max_sweeps=300)
        if c3 < best_c:
            best_c = c3
            best_h = h3
        best_n = len(best_h)
    
    # Phase 4: Resolution probe - try different n values
    if time.time() < deadline - 250:
        probe_ns = [best_n - 10, best_n - 5, best_n + 5, best_n + 10,
                    best_n + 20, best_n + 40, best_n + 60]
        for pn in probe_ns:
            if pn < 50 or pn > 2000 or pn % 2 != 0:
                continue
            if time.time() > deadline - 250:
                break
            h_probe = resample_h(best_h, pn)
            c_probe = compute_c5(h_probe, pn)
            if c_probe < best_c - 1e-6:
                # Quick optimize at this resolution
                p_probe_end = min(time.time() + 30, deadline - 220)
                h_pr, c_pr = run_lbfgs(h_probe, pn, p_probe_end,
                    alphas=[500, 5000, 50000, 500000],
                    mu=0.005, maxiter_list=[1000, 1500, 2000, 2000])
                if c_pr < best_c:
                    best_c = c_pr
                    best_h = h_pr
                    best_n = pn
        # Also try a lower resolution
        low_n = best_n // 2
        if low_n >= 100 and low_n % 2 == 0:
            if time.time() < deadline - 250:
                h_low = resample_h(best_h, low_n)
                c_low = compute_c5(h_low, low_n)
                if c_low < best_c:
                    p_low_end = min(time.time() + 40, deadline - 220)
                    h_lr, c_lr = run_lbfgs(h_low, low_n, p_low_end,
                        alphas=[200, 2000, 20000, 200000],
                        mu=0.005,
                        maxiter_list=[1500, 2000, 2500, 2500])
                    if c_lr < best_c:
                        best_c = c_lr
                        best_h = h_lr
                        best_n = low_n
    best_n = len(best_h)
    
    # Phase 5: More L-BFGS at best resolution
    if time.time() < deadline - 200:
        p5_end = min(time.time() + 100, deadline - 200)
        h5, c5 = run_lbfgs(best_h, best_n, p5_end,
            alphas=[100000, 500000, 2000000, 10000000],
            mu=0.003,
            maxiter_list=[2000, 2500, 3000, 3000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
        best_n = len(best_h)
    
    # Phase 6: Coordinate descent again
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 100, deadline - 150)
        h6, c6 = coord_descent(best_h, best_n, p6_end, max_sweeps=200)
        if c6 < best_c:
            best_c = c6
            best_h = h6
        best_n = len(best_h)
    
    # Phase 7: Active-set at high alpha
    if time.time() < deadline - 100:
        p7_end = min(time.time() + 50, deadline - 100)
        h7, c7 = run_active(best_h, best_n, p7_end, top_k=3,
            alphas=[500000, 2000000, 10000000])
        if c7 < best_c:
            best_c = c7
            best_h = h7
        best_n = len(best_h)
    
    # Phase 8: Subgradient refinement
    if time.time() < deadline - 50:
        p8_end = min(time.time() + 30, deadline - 50)
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=3000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        best_n = len(best_h)
    
    # Phase 9: Fine pair search
    if time.time() < deadline - 10:
        p9_end = deadline - 3
        h9, c9 = fine_pair_search(best_h, best_n, p9_end, radius=20)
        if c9 < best_c:
            best_c = c9
            best_h = h9
        best_n = len(best_h)
    
    # Phase 10: Final coordinate micro-optimization
    if time.time() < deadline - 2:
        p10_end = deadline - 1
        h10, c10 = coord_descent(best_h, best_n, p10_end, max_sweeps=50)
        if c10 < best_c:
            best_c = c10
            best_h = h10
        best_n = len(best_h)
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
