def fast_swap_search(h, n, deadline, max_sweeps=1000, seed=42):
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    c = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = c.max()
    dx = 2.0/n
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zero_list = sorted(zeros)
        # Shuffle for diversification
        rng.shuffle(zero_list)
        
        for i in zero_list:
            if time.time() > deadline:
                break
            if i not in zeros:
                continue
            one_list = sorted(ones)
            # Precompute i-part of delta
            # A1[k] = 1-h[i+k-n+1] for valid k
            # A3[k] = h[i-k+n-1] for valid k
            a1 = np.zeros(2*n-1)
            a3 = np.zeros(2*n-1)
            # k range for a1: 0<=i+k-n+1<n => n-1-i <= k <= n-1+i
            k_lo = max(0, n-1-i)
            k_hi = min(2*n-2, n-1+i)
            idx = np.arange(k_lo, k_hi+1)
            a1[idx] = 1 - h[i + idx - (n-1)]
            
            # k range for a3: 0<=i-k+n-1<n => i-n+1 <= k <= i+n-1
            k_lo3 = max(0, i-n+1)
            k_hi3 = min(2*n-2, i+n-1)
            idx3 = np.arange(k_lo3, k_hi3+1)
            a3[idx3] = h[i - idx3 + (n-1)]
            
            i_part = a1 - a3  # part that depends only on i
            
            # For each j in ones, compute j-part
            # B1[k] = 1-h[j+k-n+1] for valid k
            # B3[k] = h[j-k+n-1] for valid k
            # j_part = -B1 + B3
            # Total delta = i_part + j_part = a1 - a3 - B1 + B3
            
            best_j = -1
            best_new_max = cur_max + 1  # worse than current
            
            for j in one_list:
                if time.time() > deadline:
                    break
                if j not in ones:
                    continue
                b1 = np.zeros(2*n-1)
                b3 = np.zeros(2*n-1)
                k_lo_j = max(0, n-1-j)
                k_hi_j = min(2*n-2, n-1+j)
                idxj = np.arange(k_lo_j, k_hi_j+1)
                b1[idxj] = 1 - h[j + idxj - (n-1)]
                
                k_lo3j = max(0, j-n+1)
                k_hi3j = min(2*n-2, j+n-1)
                idx3j = np.arange(k_lo3j, k_hi3j+1)
                b3[idx3j] = h[j - idx3j + (n-1)]
                
                j_part = -b1 + b3
                new_c = c + i_part + j_part
                nm = new_c.max()
                if nm < best_new_max:
                    best_new_max = nm
                    best_j = j
                    # Early termination: if we found a move that reduces max
                    if nm < cur_max - 1e-15:
                        break
            
            if best_j >= 0 and best_new_max < cur_max - 1e-15:
                h[i] = 1.0
                h[best_j] = 0.0
                c += i_part + (-np.zeros(2*n-1))  # recompute
                # Actually recompute c for correctness
                c = np.correlate(h, 1-h, mode='full').astype(float)
                cur_max = c.max()
                zeros.discard(i)
                zeros.add(best_j)
                ones.discard(best_j)
                ones.add(i)
                improved = True
        
        if not improved:
            break
    
    return h, cur_max * dx

                idx_minus = j_range - k + (n - 1)  # i - k + (n-1) = j_range - (k - n + 1)
                # dc[k]/dx[j] contribution: w[k] * dx * [(1-x[idx_plus]) - x[idx_minus]]
                grad[j_range] += w[k] * dx * ((1 - x[idx_plus]) - x[idx_minus])
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        try:
            res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            return project_feasible(res.x, n)
        except Exception:
            return h_init
