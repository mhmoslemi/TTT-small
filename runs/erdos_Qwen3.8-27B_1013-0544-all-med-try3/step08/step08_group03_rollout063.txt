                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return best_h, best_c

def gradient_pair_search(h, n, deadline, step_size=0.003, radius=60):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in [100, 500, 2000, 10000]:
        while time.time() < deadline - 5:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline - 5:
                    return best_h, best_c
                for j in range(max(0, i-radius), min(n, i+radius)):
                    if i == j:
                        continue
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    if 0.0 <= h[i] + move <= 1.0 and 0.0 <= h[j] - move <= 1.0:
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B at original resolution (aggressive)
    t1 = min(t0 + 60, deadline - 600)
    best_h, best_c = lbfgs_multi_alpha(best_h, best_n, t1,
        alpha_list=[200, 1000, 5000, 20000, 100000, 500000, 2000000])
    best_n = best_h.shape[0]

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 500:
        for a in [200000, 1000000, 5000000]:
            if time.time() > deadline - 450:
                break
            h2, c2 = slsqp_phase(best_h, best_n, min(time.time() + 30, deadline - 450),
                                alpha=a, maxiter=800)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Gradient focused descent
    if time.time() < deadline - 400:
        t3 = min(time.time() + 40, deadline - 400)
        h3, c3 = grad_focused_descent(best_h, best_n, t3)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Try n=600
    if time.time() < deadline - 250:
        n_try = 600
        h600 = upsample(best_h, best_n, n_try)
        t4 = min(time.time() + 50, deadline - 200)
        h600, c600 = lbfgs_multi_alpha(h600, n_try, t4,
            alpha_list=[500, 5000, 50000, 200000, 1000000])
        if c600 < best_c:
            best_c = c600
            best_h = h600
            best_n = n_try

    # Phase 5: Try n=900
    if time.time() < deadline - 180:
        n_try2 = 900
        h900 = upsample(best_h, best_n, n_try2)
        t5 = min(time.time() + 40, deadline - 140)
        h900, c900 = lbfgs_multi_alpha(h900, n_try2, t5,
            alpha_list=[1000, 10000, 100000, 500000])
        if c900 < best_c:
            best_c = c900
            best_h = h900
            best_n = n_try2

    # Phase 6: Try n=450 (between 300 and 600)
    if time.time() < deadline - 130:
        n_try3 = 450
        h450 = upsample(best_h, best_n, n_try3)
        t6 = min(time.time() + 30, deadline - 100)
        h450, c450 = lbfgs_multi_alpha(h450, n_try3, t6,
            alpha_list=[500, 5000, 50000, 200000])
        if c450 < best_c:
            best_c = c450
            best_h = h450
            best_n = n_try3

    # Phase 7: Final L-BFGS + SLSQP on best
    if time.time() < deadline - 150:
        t7 = min(time.time() + 40, deadline - 120)
        best_h, best_c = lbfgs_multi_alpha(best_h, best_n, t7,
            alpha_list=[100000, 500000, 2000000, 10000000])
        if time.time() < deadline - 100:
            h7, c7 = slsqp_phase(best_h, best_n, min(time.time() + 30, deadline - 90),
                                alpha=2000000, maxiter=1000)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Gradient pair search
    if time.time() < deadline - 80:
        t8 = min(time.time() + 100, deadline - 60)
        best_h, best_c = gradient_pair_search(best_h, best_n, t8, step_size=0.002, radius=50)

    # Phase 9: Fine pair search
    if time.time() < deadline - 30:
        t9 = deadline - 5
        best_h, best_c = pair_search_local(best_h, best_n, t9, radius=35)

    # Phase 10: Random perturbations (if time remains)
    if time.time() < deadline - 15:
        for _ in range(3):
            if time.time() > deadline - 10:
                break
            h_pert = best_h + rng.randn(best_n) * 0.01
            h_pert = project_feasible(h_pert, best_n)
            t_p = min(time.time() + 8, deadline - 5)
            h_p, c_p = lbfgs_multi_alpha(h_pert, best_n, t_p,
                alpha_list=[50000, 500000, 2000000])
            if c_p < best_c:
                best_c = c_p
                best_h = h_p

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
