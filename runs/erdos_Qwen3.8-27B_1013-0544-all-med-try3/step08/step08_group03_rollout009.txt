import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))


def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def grad_smooth_max(h, n, alpha):
    c = compute_c5_all(h, n)
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        wm = w[m]
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (-h[k:n])
            grad[k:n] += wm * (1.0 - h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[0:nv] += wm * (1.0 - h[kk:n])
            grad[kk:n] += wm * (-h[0:nv])
    grad *= dx
    return grad


def smooth_max_obj(x, n, alpha):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)


def penalty_obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g


def slsqp_equal(x, n):
    return float(x.sum() - n / 2.0)


def slsqp_equal_jac(x, n):
    return np.ones(n)


def lbfgs_phase(h, n, t_end, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

        (200, 10.0), (1000, 50.0), (10000, 100.0),

def slsqp_phase(h, n, t_end, alpha=500000, maxiter=2000):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

        made_progress = True
        while made_progress and time.time() < deadline:
            made_progress = False

                    return h, best_c
                if radius is None:
                    lo, hi = 0, n

                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):

                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            made_progress = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d

def gradient_pair_search(h, n, deadline, step_size=0.003):
    h = h.copy()

    alphas = [200, 1000, 5000]
    for alpha in alphas:

                break
    return h, best_c
