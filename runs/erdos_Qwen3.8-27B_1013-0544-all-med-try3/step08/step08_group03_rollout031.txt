    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

        s = h.sum()

            break

            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room

                break
            h += (diff / s) * h

    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        return c.max() + (mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha

    mx = c_s.max()
