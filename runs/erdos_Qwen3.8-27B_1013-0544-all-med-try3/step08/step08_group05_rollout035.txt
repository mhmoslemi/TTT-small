    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    c = fft_cross_correlate(h, 1 - h)
    cur_energy = np.max(c)
    best_energy = cur_energy
    best_h = h.copy()

    T0 = 0.1

    cooling = (T1 / T0) ** (1.0 / n_steps)

        oj = ones[j]

import time

def fft_cross_correlate(a, b):
    n = len(a)
    size = 2 * n - 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b[n::-1], size)
    return np.fft.irfft(A * B, size)

def compute_c5(h, n):
    dx = 2.0 / n
    c = fft_cross_correlate(h, 1.0 - h)
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = fft_cross_correlate(x, 1.0 - x) * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

def fft_cross_correlate(a, b):
    n = len(a)
    size = 2 * n - 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b[n::-1], size)
    return np.fft.irfft(A * B, size)

def compute_c5(h, n):
    dx = 2.0 / n
    c = fft_cross_correlate(h, 1.0 - h)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

def fft_cross_correlate(a, b):
    n = len(a)
    size = 2 * n - 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b[n-1::-1], size)
    return np.fft.irfft(A * B, size)

def compute_c5(h, n):
    dx = 2.0 / n
    c = fft_cross_correlate(h, 1.0 - h)
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = fft_cross_correlate(x, 1.0 - x) * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

    c = fft_cross_correlate(h, omh)
    cur_energy = c.max()
    best_energy = cur_energy
    best_h = h.copy()
    
    if T0 is None:
        T0 = max(cur_energy * 0.05, 0.005)
    T1 = T0 * 1e-3
    ratio = (T1 / T0) ** (1.0 / n_steps)
    
    for step in range(n_steps):
        if time.time() > t_end:
            break
        T = T0 * (ratio ** step)
        
        i = rng.randint(len(zeros))
        j = rng.randint(len(ones))
        zi = zeros[i]
        oj = ones[j]
        
        h[zi] = 1.0
        h[oj] = 0.0
        omh[zi] = 0.0
        omh[oj] = 1.0
        zeros[i] = oj
        ones[j] = zi
        
        c = fft_cross_correlate(h, omh)
        new_energy = c.max()
        delta = new_energy - cur_energy
        
        if delta <= 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            cur_energy = new_energy
            if new_energy < best_energy:
                best_energy = new_energy
                best_h = h.copy()
        else:
            h[zi] = 0.0
            h[oj] = 1.0
            omh[zi] = 1.0
            omh[oj] = 0.0
            zeros[i] = zi
            ones[j] = oj
    
    return best_h, best_energy * dx

            break
        improved = False

            if time.time() > t_end:
                break
            for j in o_list:
                if time.time() > t_end:
                    break
                h[i] = 1.0
                h[j] = 0.0
                omh[i] = 0.0
                omh[j] = 1.0
                c = fft_cross_correlate(h, omh)
                nc = c.max() * dx
                if nc < cur_c - 1e-15:
                    cur_c = nc
                    c = c.copy()
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    break
                else:
                    h[i] = 0.0
                    h[j] = 1.0
                    omh[i] = 1.0
                    omh[j] = 0.0
        
        if not improved:
            break
    
    return best_h, best_c

    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:

    elif len(ones) < target:
        deficit = target - len(ones)
        to_one = rng.choice(zeros, deficit, replace=False)
        h[to_one] = 1.0
    return h

        T0 = max(cur_energy * 0.04, 0.003)
    T1 = T0 * 5e-4
    ratio = (T1 / T0) ** (1.0 / n_steps)

    for step in range(n_steps):
        if time.time() > t_end:
            break
        T = T0 * (ratio ** step)

        if beam == 1:
            i = rng.randint(len(zeros))
            j = rng.randint(len(ones))
            zi = zeros[i]
            oj = ones[j]

            h[zi] = 1.0
            h[oj] = 0.0
            omh[zi] = 0.0
            omh[oj] = 1.0
            zeros[i] = oj
            ones[j] = zi

            c = fft_cross_correlate(h, omh)
            new_energy = c.max()
            delta = new_energy - cur_energy

            if delta <= 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
                cur_energy = new_energy
                if new_energy < best_energy:
                    best_energy = new_energy
                    best_h = h.copy()
            else:
                h[zi] = 0.0
                h[oj] = 1.0
                omh[zi] = 1.0
                omh[oj] = 0.0
                zeros[i] = zi
                ones[j] = oj
        else:
            best_delta = np.inf
            best_zi = -1
            best_oj = -1
            best_i_idx = -1
            best_j_idx = -1
            best_c = None
            for _ in range(beam):
                ii = rng.randint(len(zeros))
                jj = rng.randint(len(ones))
                zi = zeros[ii]
                oj = ones[jj]
                h[zi] = 1.0
                h[oj] = 0.0
                omh[zi] = 0.0
                omh[oj] = 1.0
                c = fft_cross_correlate(h, omh)
                ne = c.max()
                d = ne - cur_energy
                if d < best_delta:
                    best_delta = d
                    best_zi = zi
                    best_oj = oj
                    best_i_idx = ii
                    best_j_idx = jj
                    best_c = c
                if best_i_idx < 0 or zi != best_zi:
                    h[zi] = 0.0
                    h[oj] = 1.0
                    omh[zi] = 1.0
                    omh[oj] = 0.0
            if best_zi >= 0:
                h[best_zi] = 1.0
                h[best_oj] = 0.0
                omh[best_zi] = 0.0
                omh[best_oj] = 1.0
                zeros[best_i_idx] = best_oj
                ones[best_j_idx] = best_zi
                accept = best_delta <= 0 or (T > 1e-15 and rng.random() < np.exp(-best_delta / T))
                if accept:
                    cur_energy = best_delta + cur_energy
                    if cur_energy < best_energy:
                        best_energy = cur_energy
                        best_h = h.copy()
                else:
                    h[best_zi] = 0.0
                    h[best_oj] = 1.0
                    omh[best_zi] = 1.0
                    omh[best_oj] = 0.0
                    zeros[best_i_idx] = best_zi
                    ones[best_j_idx] = best_oj

    return best_h, best_energy * dx


def greedy_binary(h0, n, dx, t_end, max_sweeps=2000):
    target = n // 2
    idx = np.argsort(-h0)
    h = np.zeros(n, dtype=np.float64)
    h[idx[:target]] = 1.0
    omh = 1.0 - h

    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())

    c = fft_cross_correlate(h, omh).copy()
    cur_c = c.max() * dx
    best_c = cur_c
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        z_list = sorted(zeros)
        o_list = sorted(ones)

        for i in z_list:
            if time.time() > t_end:
                break
            for j in o_list:
                if time.time() > t_end:
                    break
                h[i] = 1.0
                h[j] = 0.0
                omh[i] = 0.0
                omh[j] = 1.0
                c = fft_cross_correlate(h, omh)
                nc = c.max() * dx
                if nc < cur_c - 1e-14:
                    cur_c = nc
                    c = c.copy()
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    break
                else:
                    h[i] = 0.0
                    h[j] = 1.0
                    omh[i] = 1.0
                    omh[j] = 0.0

        if not improved:
            break

    return best_h, best_c

    m_range = np.arange(-(n - 1), n)

def make_periodic_init(n, period, duty, rng):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        excess = len(ones) - target
        to_zero = rng.choice(ones, excess, replace=False)
        h[to_zero] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_one = rng.choice(zeros, deficit, replace=False)
        h[to_one] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    dx = 2.0 / n
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SA from initial solution (primary search)
    p1_end = t0 + 0.20 * budget_s
    hb, cb = binary_sa(best_h, n, dx, p1_end, rng, n_steps=8000000, beam=1)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Greedy binary polish
    p2_end = t0 + 0.40 * budget_s
    hb, cb = greedy_binary(best_h, n, dx, p2_end, max_sweeps=3000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: EA-style SA with beam (wider search)
    p3_end = t0 + 0.55 * budget_s
    hb, cb = binary_sa(best_h, n, dx, p3_end, rng, n_steps=4000000, T0=0.02, beam=3)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    if time.time() < p3_end:
        hb, cb = greedy_binary(best_h, n, dx, p3_end, max_sweeps=2000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 4: SLSQP continuous polish
    p4_end = t0 + 0.62 * budget_s
    if time.time() < p4_end:
        for alpha in [500, 5000, 50000]:
            if time.time() > p4_end:
                break
            hp = slsqp_smooth(best_h.copy(), n, alpha, 1500, p4_end)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 5: Alternative starting structures
    p5_end = t0 + 0.78 * budget_s
    if time.time() < p5_end - 30:
        inits = []
        inits.append(make_periodic_init(n, 4, 2, rng))
        inits.append(make_periodic_init(n, 6, 3, rng))
        inits.append(make_periodic_init(n, 10, 5, rng))
        inits.append(make_periodic_init(n, 12, 6, rng))
        inits.append(np.full(n, 0.5) + 0.5 * np.sin(np.linspace(0, 2 * np.pi * 3, n)))

        per_init_time = min(50, max(20, (p5_end - time.time()) / len(inits)))
        for h_init in inits:
            if time.time() > p5_end - 10:
                break
            h_bin_init = make_periodic_init(n, 4, 2, rng)
            h_bin_init = np.where(h_init > 0.5, 1.0, 0.0)
            h_bin_init = project_feasible(h_bin_init, n)
            sub_end = min(p5_end - 5, time.time() + per_init_time)
            hb, cb = binary_sa(h_bin_init, n, dx, sub_end, rng, n_steps=2000000)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                if time.time() < p5_end - 5:
                    hb2, cb2 = greedy_binary(hb, n, dx, min(p5_end - 3, time.time() + 30), max_sweeps=500)
                    if cb2 < best_c:
                        best_c = cb2
                        best_h = hb2.copy()

    # Phase 6: Higher resolution exploration
    p6_end = deadline - 100
    if time.time() < p6_end - 80:
        for nh in [400, 500]:
            if time.time() > p6_end - 20:
                break
            if nh == n:
                continue
            xo = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            xn = np.linspace(0, 2, nh, endpoint=False) + 1.0 / nh
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            sub_end = min(p6_end - 10, time.time() + 120)
            hb2, cb2 = binary_sa(hh, nh, 2.0 / nh, sub_end, rng, n_steps=2000000)
            if cb2 < best_c:
                if time.time() < p6_end - 10:
                    hb2, cb2 = greedy_binary(hb2, nh, 2.0 / nh, min(p6_end - 5, time.time() + 40), max_sweeps=300)
                if cb2 < best_c:
                    best_c = cb2
                    best_h = hb2.copy()
                    n = nh
                    dx = 2.0 / n
            else:
                if time.time() < p6_end - 10:
                    hh2 = slsqp_smooth(hh, nh, 10000, 500, min(p6_end - 5, time.time() + 40))
                    ch = compute_c5(hh2, nh)
                    if ch < best_c:
                        best_c = ch
                        best_h = hh2.copy()
                        n = nh
                        dx = 2.0 / n

    # Phase 7: Final SA + Greedy + SLSQP polish
    if time.time() < deadline - 40:
        hb, cb = binary_sa(best_h, n, dx, min(deadline - 20, time.time() + 60), rng, n_steps=3000000, T0=0.005)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    if time.time() < deadline - 15:
        hb, cb = greedy_binary(best_h, n, dx, min(deadline - 8, time.time() + 30), max_sweeps=500)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    if time.time() < deadline - 8:
        hp = slsqp_smooth(best_h.copy() + 0.01 * rng.randn(n) * 0.05, n, 200000, 2000, deadline - 5)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()

    # Final: ensure feasibility and pick best of continuous vs binary
    best_h = project_feasible(best_h, n)
    c_cont = compute_c5(best_h, n)

    target = n // 2
    idx = np.argsort(-best_h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    c_bin = compute_c5(h_bin, n)

    if c_bin < c_cont:
        best_h = h_bin
        best_c = c_bin
    else:
        best_c = c_cont

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
