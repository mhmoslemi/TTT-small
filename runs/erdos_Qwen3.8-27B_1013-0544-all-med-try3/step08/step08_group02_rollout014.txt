                                       'gtol': 1e-15, 'maxfun': maxit * 8,
                                       'maxls': 60, 'maxcor': 60})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run_slqp_phase(h_init, n, deadline, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_smooth, h_init, jac=grad_smooth,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16,
                               'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h_init.copy(), n), compute_c5(h_init, n)

def run_penalty_bisect(h_init, n, deadline, c_start, c_end, n_bisect=5):
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for _ in range(n_bisect):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        c_target = best_c - max(0.0005, 0.002 * (best_c - 0.3))
        for gamma in [50.0, 200.0, 1000.0]:
            if time.time() > deadline - 15:
                break
            maxit = max(500, int((deadline - time.time()) * 3))
            try:
                res = minimize(obj_penalty, best_h, jac=grad_penalty,
                              args=(n, c_target, gamma), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit * 8,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgrad_refine(h_init, n, deadline, n_iter=8000):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for p in active:
            k = p - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:]) * dx
                grad[k:] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[kk:]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.99
        h = h_new
    return best_h, best_c

def fine_local_search(h_init, n, deadline):
    h = h_init.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for delta in deltas:
        improved = True
        passes = 0
        while improved and passes < 5 and time.time() < deadline - 5:
            improved = False
            passes += 1
            grad_info = None
            for i in range(n):
                if time.time() > deadline - 3:
                    break
                for j in range((i + 1) % n, n):
                    if time.time() > deadline - 3:
                        break
                    if i == j:
                        continue
                    # Try moving mass from i to j
                    if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                        h[i] -= delta
                        h[j] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    # Try moving mass from j to i
                    if h[j] - delta >= 0.0 and h[i] + delta <= 1.0:
                        h[j] -= delta
                        h[i] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
            h = best_h.copy()
    return best_h, best_c

def make_binary_start(n, pattern='block'):
    h = np.zeros(n)
    half = n // 2
    if pattern == 'block':
        h[:half] = 1.0
    elif pattern == 'two_block':
        q = n // 4
        h[:q] = 1.0
        h[2 * q:3 * q] = 1.0
    elif pattern == 'random':
        idx = np.random.choice(n, half, replace=False)
        h[idx] = 1.0
    return project_feasible(h, n)

def upsample(h_from, n_from, n_to):
    x_fine = np.linspace(0, 2, n_to, endpoint=False)
    x_coarse = np.linspace(0, 2, n_from, endpoint=False)
    h_fine = np.interp(x_fine, x_coarse, h_from, period=2)
    return project_feasible(h_fine, n_to)

def downsample(h_from, n_from, n_to):
    x_coarse = np.linspace(0, 2, n_to, endpoint=False)
    x_fine = np.linspace(0, 2, n_from, endpoint=False)
    h_coarse = np.interp(x_coarse, x_fine, h_from, period=2)
    return project_feasible(h_coarse, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    rng = np.random.RandomState(seed)

    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    bounds_500 = [(0.0, 1.0)] * n0

    # Phase 1: L-BFGS from parent at n0 with careful alpha schedule
    p1_end = t0 + 250
    h1, c1 = run_lbfgs(best_h, n0, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.01,
        maxiters=[1500, 1500, 2000, 2500, 2500, 3000, 3000, 3500],
        bounds=bounds_500)
    if c1 < best_c:
        best_c, best_h, best_n = c1, h1, n0

    # Phase 2: Active set refinement
    p2_end = min(time.time() + 150, deadline - 400)
    h2, c2 = run_active_refine(best_h, n0, p2_end, top_k=6, mu=0.005)
    if c2 < best_c:
        best_c, best_h, best_n = c2, h2, n0

    # Phase 3: Penalty bisection
    p3_end = min(time.time() + 150, deadline - 300)
    h3, c3 = run_penalty_bisect(best_h, n0, p3_end, best_c, 0.3, n_bisect=4)
    if c3 < best_c:
        best_c, best_h, best_n = c3, h3, n0

    # Phase 4: Try n=1000 (upsample + optimize)
    n_hi = 1000
    bounds_1000 = [(0.0, 1.0)] * n_hi
    if time.time() < deadline - 300:
        h_up = upsample(best_h, n0, n_hi)
        c_up = compute_c5(h_up, n_hi)
        p4_end = min(time.time() + 250, deadline - 200)
        h4, c4 = run_lbfgs(h_up, n_hi, p4_end,
            alphas=[30, 100, 500, 2000, 10000, 50000, 200000],
            mu=0.005,
            maxiters=[1500, 2000, 2000, 2500, 2500, 2500, 2500],
            bounds=bounds_1000)
        if c4 < best_c:
            best_c, best_h, best_n = c4, h4, n_hi

    # Phase 5: Try binary/random starts
    if time.time() < deadline - 200:
        for pat in ['block', 'two_block', 'random']:
            if time.time() > deadline - 80:
                break
            h_bin = make_binary_start(n0, pat)
            c_bin = compute_c5(h_bin, n0)
            p5_end = min(time.time() + 80, deadline - 70)
            h5, c5 = run_lbfgs(h_bin, n0, p5_end,
                alphas=[100, 1000, 5000, 50000, 500000],
                mu=0.01,
                maxiters=[2000, 2000, 2500, 2500, 3000],
                bounds=bounds_500)
            if c5 < best_c:
                best_c, best_h, best_n = c5, h5, n0
            if c_bin < best_c:
                best_c, best_h = c_bin, h_bin

    # Phase 6: Perturbation + re-optimize
    if time.time() < deadline - 150:
        for noise_scale in [0.02, 0.01, 0.005]:
            if time.time() > deadline - 50:
                break
            h_pert = best_h + noise_scale * rng.randn(best_n)
            h_pert = project_feasible(h_pert, best_n)
            bt = [(0.0, 1.0)] * best_n
            p6_end = min(time.time() + 50, deadline - 45)
            h6, c6 = run_lbfgs(h_pert, best_n, p6_end,
                alphas=[5000, 50000, 500000],
                mu=0.01,
                maxiters=[2000, 2500, 2500],
                bounds=bt)
            if c6 < best_c:
                best_c, best_h, best_n = c6, h6, best_n

    # Phase 7: SLSQP polish at current best
    if time.time() < deadline - 60:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 40:
                break
            p7_end = min(time.time() + 30, deadline - 35)
            h7, c7 = run_slqp_phase(best_h, best_n, p7_end, alpha, maxiter=1000)
            if c7 < best_c:
                best_c, best_h, best_n = c7, h7, best_n

    # Phase 8: Subgradient refinement
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 35, deadline - 20)
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=10000)
        if c8 < best_c:
            best_c, best_h, best_n = c8, h8, best_n

    # Phase 9: Fine local search (limited)
    if time.time() < deadline - 15:
        p9_end = deadline - 3
        h9, c9 = fine_local_search(best_h, best_n, p9_end)
        if c9 < best_c:
            best_c, best_h, best_n = c9, h9, best_n

    # Phase 10: Final SLSQP
    if time.time() < deadline - 2:
        h10, c10 = run_slqp_phase(best_h, best_n, deadline - 1, 500000, maxiter=500)
        if c10 < best_c:
            best_c, best_h, best_n = c10, h10, best_n

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
