Actually, for the pair transfer, I can be smarter. Instead of trying many δ values, I can compute the derivative of c_max with respect to δ and use that to find the optimal δ analytically or with fewer evaluations.

For a transfer of δ from i to j:
h[i] → h[i] - δ
h[j] → h[j] + δ

The change in c_k:
c_k(δ) = c_k(0) + δ * (∂c_k/∂h[j] - ∂c_k/∂h[i])
= c_k(0) + δ * g_k

where g_k = ∂c_k/∂h[j] - ∂c_k/∂h[i]

And c_
eqk(0) = c_k(0) is known.

We want to find δ ∈ [0, δ_max] that minimizes max_k (c_k(0) + δ * g_k).

This is a 1D problem: minimize f(δ) = max_k (a_k + b_k * δ) where a_k = c_k(0) and b_k = g_k.

The function f(δ) is the upper envelope of lines, which is convex in δ. So we can find the minimum by:
1. Finding all breakpoints of the upper envelope
2. Evaluating f at each breakpoint and at the boundaries
3. Taking the minimum

The breakpoints occur where two lines cross: a_i + b_i * δ = a_j + b_j * δ → δ = (a_j - a_i) / (b_i - b_j)

For 2n-1 lines, there are O(n²) potential breakpoints, but the upper envelope has at most O(n) breakpoints (in the worst case, it's a convex piecewise-linear function with at most 2n-1 pieces).

To find the minimum of the upper envelope:
- The minimum occurs at a breakpoint (where two or more lines achieve the max simultaneously) or at a boundary δ = 0 or δ = δ_max.
- We can find it by: start at δ = 0, identify which line is on top, then find where the next line crosses it, etc.

Actually, a simpler approach: since f(δ) is convex, we can use ternary search or binary search on the derivative. The derivative of f(δ) at any point is the slope of the line that's on top at that point. We want to find where the derivative changes from negative to positive.

Binary search: evaluate f at midpoint, compute the subgradient (which is the slope of the active line). If the active line has negative slope, move right. If positive, move left. If zero, we're at the minimum.

This takes O(log(1/ε)) iterations, each requiring O(n) work to evaluate f and find the active line. Total: O(n log(1/ε)) per pair.

For n=500, that's about 500 * 30 = 15000 ops per pair. With 400 pairs (20×20), that's 6M ops per sweep. Very fast!

    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxiter_per=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 4))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 8,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_phase(h, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000]
    for round_idx in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': maxit * 8,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def steepen(h, n, beta):
    h_new = 0.5 + np.tanh(beta * (h - 0.5))
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n)

def pair_refine(h, n, deadline, max_sweeps=15, radius=50):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-6, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        
        pos_idx = np.where(grad > 1e-10)[0]
        neg_idx = np.where(grad < -1e-10)[0]
        
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            break
        
        pos_idx = pos_idx[np.argsort(-grad[pos_idx])][:15]
        neg_idx = neg_idx[np.argsort(grad[neg_idx])][:15]
        
        n_deltas = 12
        for i in pos_idx:
            if time.time() > deadline:
                break
            lo_i = max(0, i - radius)
            hi_i = min(n, i + radius + 1)
            for j in neg_idx:
                if i == j or j < lo_i or j >= hi_i:
                    continue
                delta_max = min(best_h[i], 1.0 - best_h[j])
                if delta_max < 1e-12:
                    continue
                deltas = np.linspace(0, delta_max, n_deltas)
                best_delta = 0.0
                best_c_trial = best_c
                for d in deltas[1:]:
                    best_h[i] -= d
                    best_h[j] += d
                    c_trial = compute_c5(best_h, n)
                    if c_trial < best_c_trial - 1e-16:
                        best_c_trial = c_trial
                        best_delta = d
                    best_h[i] += d
                    best_h[j] -= d
                if best_delta > 0:
                    best_h[i] -= best_delta
                    best_h[j] += best_delta
                    best_c = best_c_trial
                    improved = True
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
    return best_h, best_c

def slqp_phase(h, n, deadline, alpha=500000, maxiter=1000):
    def sq_fun(x):
        return x.sum() - n / 2.0
    def sq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac}]
    try:
        res = minimize(sm_obj, h, args=(n, alpha, 0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def fine_pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h[i] -= d
                            best_h[j] += d
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] += remaining
    return project_feasible(h_bin, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n = 500
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    best_n = n
    
    # Phase 1: L-BFGS with soft-max annealing at native resolution
    p1_end = min(t0 + 250, deadline - 400)
    h1, c1 = lbfgs_phase(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.01, maxiter_per=[2000, 2000, 2500, 2500, 3000, 3000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Steepen + re-optimize (push toward binary)
    if time.time() < deadline - 300:
        for beta in [10, 30, 80]:
            if time.time() > deadline - 250:
                break
            h_steep = steepen(best_h, best_n, beta)
            c_steep = compute_c5(h_steep, best_n)
            p_end = min(time.time() + 80, deadline - 200)
            h_s, c_s = lbfgs_phase(h_steep, best_n, p_end,
                alphas=[500, 5000, 50000, 500000],
                mu=0.005, maxiter_per=[1500, 2000, 2500, 2500])
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
    
    # Phase 3: Active-set focused optimization
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 120, deadline - 200)
        h3, c3 = active_phase(best_h, best_n, p3_end, top_k=6,
            alphas=[20000, 100000, 500000, 2000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Try upsampling to higher resolution
    n_hi = 1000
    if time.time() < deadline - 250:
        x_fine = np.linspace(0, 2, n_hi + 1)
        x_coarse = np.linspace(0, 2, best_n + 1)
        h_fine = np.interp(x_fine[:-1], x_coarse[:-1], best_h)
        h_fine = project_feasible(h_fine, n_hi)
        c_fine = compute_c5(h_fine, n_hi)
        
        p4_end = min(time.time() + 250, deadline - 200)
        h4, c4 = lbfgs_phase(h_fine, n_hi, p4_end,
            alphas=[20, 100, 1000, 5000, 20000, 100000],
            mu=0.005, maxiter_per=[1000, 1500, 2000, 2000, 2500, 2500])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_hi
    
    # Phase 5: Binary start exploration
    if time.time() < deadline - 200:
        h_bin = make_binary_start(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        p5_end = min(time.time() + 100, deadline - 150)
        h5, c5 = lbfgs_phase(h_bin, best_n, p5_end,
            alphas=[100, 1000, 10000, 100000, 500000],
            mu=0.01, maxiter_per=[1500, 2000, 2000, 2500, 2500])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Gradient-guided pair refinement
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 100, deadline - 100)
        h6, c6 = pair_refine(best_h, best_n, p6_end, max_sweeps=10, radius=40)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Active-set with very high alpha
    if time.time() < deadline - 100:
        p7_end = min(time.time() + 60, deadline - 80)
        h7, c7 = active_phase(best_h, best_n, p7_end, top_k=4,
            alphas=[100000, 500000, 2000000, 5000000])
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Subgradient refinement
    if time.time() < deadline - 60:
        p8_end = min(time.time() + 40, deadline - 40)
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=3000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: SLSQP polish
    if time.time() < deadline - 40:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 30:
                break
            p9_end = min(time.time() + 20, deadline - 25)
            h9, c9 = slqp_phase(best_h, best_n, p9_end, alpha=alpha, maxiter=800)
            if c9 < best_c:
                best_c = c9
                best_h = h9
    
    # Phase 10: Fine pair search (last resort)
    if time.time() < deadline - 10:
        p10_end = deadline - 2
        h10, c10 = fine_pair_search(best_h, best_n, p10_end, radius=20)
        if c10 < best_c:
            best_c = c10
            best_h = h10
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
