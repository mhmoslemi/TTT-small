def compute_subgradient(h, n, active_set):
    """Compute subgradient of max_k c(k) at h, averaged over active_set."""
    # c(k) = dx * sum_i h[i] * (1 - h[i+k]) for valid indices
    # In 'full' mode, the result has length 2n-1
    # c_full[k] = sum_{i} h[i] * (1-h)[i + (k - (n-1))]
    #           = sum_{i} h[i] * (1 - h[i + k - n + 1])
    
    # Let shift s = k - (n-1), so c at shift s = sum_i h[i] * (1-h[i+s])
    # where i+s must be in [0, n-1] and i in [0, n-1]
    
    # dc/dh[j] for shift s:
    # = dx * sum_{i: i+s in [0,n-1], i in [0,n-1]} [delta(j,i)*(1-h[i+s]) + delta(j,i+s)*(-h[i])]
    # = dx * (1-h[j+s]) + dx * (-h[j-s])  [if indices valid]
    
    # More carefully:
    # c(s) = sum_{i=max(0,-s)}^{min(n-1, n-1-s)} h[i] * (1-h[i+s])
    # dc/dh[j] = dx * [(1 if j in valid range as first factor) * (1-h[j+s]) 
    #                 + (1 if j in valid range as second factor) * (-h[j-s])]
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for s in active_set:
        # s ranges from -(n-1) to (n-1) in the correlate formula
        # In np.correlate(a, v, 'full'), the k-th element is sum_i a[i+k] * v[i]
        # So c_full[k] = sum_i h[i+k] * (1-h)[i] = sum_i h[i+k] * (1-h[i])
        # Let's use the convention: shift = k - (n-1)
        
        shift = s - (n - 1)
        
        # c(shift) = sum_i h[i] * (1 - h[i + shift]) where both indices are valid
        # First term: sum over i where i valid and i+shift valid: h[i] * (1-h[i+shift])
        # dc/dh[j] from first term: (1-h[j+shift]) if j valid and j+shift valid
        # dc/dh[j] from second term: -h[j-shift] if j-shift valid and j valid
        
        # First contribution: for each j, if 0<=j<n and 0<=j+shift<n, add (1-h[j+shift])
        # Second contribution: for each j, if 0<=j<n and 0<=j-shift<n, add -h[j-shift]
        
        # First: j in [0,n-1] and j+shift in [0,n-1] => j in [max(0,-shift), min(n-1, n-1-shift)]
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)  # exclusive
        if lo1 < hi1:
            grad[lo1:hi1] += (1 - h[lo1+shift:hi1+shift])
        
        # Second: j in [0,n-1] and j-shift in [0,n-1] => j in [max(0,shift), min(n-1, n-1+shift)]
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)  # exclusive
        if lo2 < hi2:
            grad[lo2:hi2] += -h[lo2-shift:hi2-shift]
    
    grad *= dx / len(active_set)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def obj_lse(x, n, alpha):
    """Log-sum-exp smooth approximation of max c(k)."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    # Use stable log-sum-exp
    sh_max = sh.max()
    if sh_max > 500:
        lse = sh_max + np.log(np.sum(np.exp(sh - sh_max)))
    else:
        lse = np.log(np.sum(np.exp(sh)))
    return float(m + lse / alpha)

def obj_active(x, n, active_idx, alpha):
    """Focus on active set: smooth max over active correlations only."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_act = c[active_idx]
    m = c_act.max()
    sh = alpha * (c_act - m)
    lse = np.log(np.sum(np.exp(sh)))
    return float(m + lse / alpha)

def find_active_set(c_all, m, eps):
    """Find indices where c is within eps of max."""
    return np.where(c_all >= m - eps)[0]

def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h.copy(), args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, active_idx, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_active, h.copy(), args=(n, active_idx, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=None):
    """Local search: try moving mass from one position to another."""
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_range = list(range(i+1, n))
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                    j_range = [j for j in range(lo, hi) if j != i]
                
                for j in j_range:
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

def subgradient_step(h, n, step_size, tol=0.01):
    """One projected subgradient step."""
    dx = 2.0 / n
    c_all = np.correlate(h, 1 - h, mode='full') * dx
    m = c_all.max()
    # Active set: indices within tol of max
    active = np.where(c_all >= m - tol)[0]
    
    # Compute subgradient (average gradient over active set)
    grad = np.zeros(n)
    for idx in active:
        shift = idx - (n - 1)
        # c(shift) = sum_i h[i] * (1 - h[i+shift]) with valid indices
        # dc/dh[j] = (1-h[j+shift]) - h[j-shift] (with boundary handling)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            grad[lo1:hi1] += (1 - h[lo1+shift:hi1+shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            grad[lo2:hi2] += -h[lo2-shift:hi2-shift]
    
    grad *= dx / len(active)
    
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def subgradient_method(h, n, deadline, n_iter_max=5000):

def coordinate_descent(h, n, deadline, n_sweeps=10):
    """Coordinate descent: optimize one coordinate at a time."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:

            # Try to find optimal h[i] given all others fixed
            # Constraint: sum(h) = n/2, so if we change h[i] to v,
            # we need to adjust the sum. Simplest: adjust h[i] and distribute
            # the difference equally among all other coordinates.
            # But this might violate [0,1] bounds.
            # Alternative: just try a few values.
            current = h[i]
            # Try values: current ± small amounts
            for delta in [0.01, 0.005, 0.001]:
                for sign in [1, -1]:
                    v = current + sign * delta
                    if v < 0 or v > 1:
                        continue
                    diff = v - current
                    # Distribute -diff among other coordinates proportionally
                    others = np.delete(h, i)
                    # We need others_sum + v = n/2
                    # others_sum_new = n/2 - v = others_sum - diff
                    # Scale others: others_new = others * (others_sum - diff) / others_sum
                    # But this might violate bounds
                    # Simpler: just add -diff/n to all others
                    h_trial = h.copy()
                    h_trial[i] = v
                    h_trial[np.arange(n) != i] += -diff / (n - 1)
                    h_trial = project_feasible(h_trial, n)
                    c = compute_c5(h_trial, n)
                    if c < best_c - 1e-12:
                        best_c = c
                        h = h_trial
                        improved = True
                        break
            # Also try pushing to 0 or 1
            for v in [0.0, 1.0]:
                if abs(v - h[i]) < 1e-6:
                    continue
                diff = v - h[i]
                h_trial = h.copy()
                h_trial[i] = v
                h_trial[np.arange(n) != i] += -diff / (n - 1)
                h_trial = project_feasible(h_trial, n)

                if c < best_c - 1e-12:

                    h = h_trial
                    improved = True
        if not improved:

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 250
        h_init = np.full(n_init, 0.5)

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize at native resolution (most important)
    t1 = time.time()
    alloc1 = min(350, (deadline - t1) * 0.35)
    h_opt, c_opt = optimize_full(h_init, n_init, t1 + alloc1)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try alternate resolutions
    x_init = np.linspace(0, 2, n_init)
    n_tries = []
    for nt in [125, 200, 300, 400]:
        if nt != n_init:
            n_tries.append(nt)
    for n_try in n_tries:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        remaining = deadline - time.time() - 60
        alloc = min(130, remaining * 0.25)
        if alloc < 50:
            break
        h_opt, c_opt = optimize_full(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Subgradient exploration
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        alloc = min(60, (deadline - time.time()) * 0.2)
        if alloc > 25:
            h_sg, c_sg = subgradient_method(best_h, n, time.time() + alloc, max_iter=3000)
            if c_sg < best_c:
                best_c = c_sg
                best_h = h_sg
            # Also try from perturbed version
            if time.time() < deadline - 40:
                hp = best_h + rng.normal(0, 0.03, n)
                hp = project_feasible(hp, n)
                h_sg2, c_sg2 = subgradient_method(hp, n, time.time() + min(30, deadline - time.time() - 30), max_iter=2000)
                if c_sg2 < best_c:
                    best_c = c_sg2
                    best_h = h_sg2

    # Phase 4: Multi-restart with perturbation
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.015 + 0.05 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 20) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_full(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Intensive final polish on best solution
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)

        # High-alpha SLSQP
        ht = slsqp_optimize(best_h, n, 200000, maxiter=3000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

        # Active set SLSQP (very tight)
        c_all = compute_all_c(best_h, n)
        m = c_all.max()
        active = find_active_set(c_all, m, 0.003)
        ht = slsqp_active(best_h, n, active, 100000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

        # Boundary CD
        if time.time() < deadline - 8:
            best_h, best_c = boundary_cd(best_h, n, deadline - 5, n_sweeps=5)

        # Ultra-fine pair search
        if time.time() < deadline - 5:
            if n <= 200:
                best_h, best_c = pair_search(best_h, n, deadline - 2,
                                            [0.00005, 0.00002, 0.00001, 0.000005])
            elif n <= 350:
                best_h, best_c = pair_search(best_h, n, deadline - 2,
                                            [0.00005, 0.00002, 0.00001],
                                            radius=50)
            else:
                best_h, best_c = pair_search(best_h, n, deadline - 2,
                                            [0.00005, 0.00002],
                                            radius=40)

        # Final SLSQP
        if time.time() < deadline - 3:
            ht = slsqp_optimize(best_h, n, 500000, maxiter=2000, ftol=1e-15)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht

        # One more boundary CD pass
        if time.time() < deadline - 2:
            best_h, best_c = boundary_cd(best_h, n, deadline - 1, n_sweeps=3)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
