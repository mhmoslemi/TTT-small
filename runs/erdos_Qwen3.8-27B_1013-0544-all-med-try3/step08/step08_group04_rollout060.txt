from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target / tot)
    return np.clip(h, 0.0, 1.0)

def smoothed_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smoothed_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-12:
            continue
        shift = k - n + 1
        
        # First term: (1-h[j-shift]) for j where j-shift is valid i
        # j-shift in [max(0,-shift), min(n-1,n-1-shift)]
        # j in [max(shift,0), min(n-1+shift, n-1)]
        j_lo1 = max(shift, 0)
        j_hi1 = min(n - 1 + shift, n - 1)
        if j_lo1 <= j_hi1:
            grad[j_lo1:j_hi1+1] += wk * (1.0 - h[j_lo1-shift:j_hi1+1-shift])
        
        # Second term: -h[j+shift] for j where j is valid i
        # j in [max(0,-shift), min(n-1,n-1-shift)]
        j_lo2 = max(0, -shift)
        j_hi2 = min(n - 1, n - 1 - shift)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] += wk * (-h[j_lo2+shift:j_hi2+1+shift])
    
    return dx * grad

def proj_gradient_method(h, n, alpha_seq, deadline, momentum_beta=0.9):
    """Projected gradient method with Nesterov momentum and backtracking"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)  # momentum velocity
    
    for alpha in alpha_seq:
        eta = 0.5 / max(np.max(np.abs(smoothed_grad(h, n, alpha))), 1e-10)
        eta = min(eta, 5.0)
        
        for it in range(500):
            if time.time() > deadline:
                return best_h, best_c
            
            g = smoothed_grad(h, n, alpha)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-12:
                break
            
            # Nesterov: evaluate gradient at y = h + beta * v
            y = h + momentum_beta * v
            g_y = smoothed_grad(y, n, alpha)
            v = momentum_beta * v - eta * g_y
            h_new = h + v
            h_new = project_feasible(h_new, n)
            
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            
            # Check if we made progress on the smoothed objective
            sv_new = smoothed_val(h_new, n, alpha)
            sv_old = smoothed_val(y, n, alpha)
            if sv_new > sv_old + 1e-8:
                # Backtrack
                v *= 0.5
                h_new = project_feasible(h + 0.5 * v, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            
            h = h_new
            
            # Adapt learning rate
            if it % 100 == 99:

    return best_h, best_c

    def obj(x):
        return smoothed_val(x, n, alpha)
    
    def jac(x):
        return smoothed_grad(x, n, alpha)
    
    def sum_con(x, n):

        return np.ones(n)

    try:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target / tot)
    return np.clip(h, 0.0, 1.0)

def smoothed_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smoothed_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        shift = k - n + 1
        i_lo = max(0, -shift)
        i_hi = min(n - 1, n - 1 - shift)
        # First term: j in [i_lo, i_hi] contributes (1-h[j+shift])
        # to grad[j]
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wk * (1.0 - h[idx + shift])
        # Second term: j = i+shift, i in [i_lo, i_hi], so j in [i_lo+shift, i_hi+shift]
        # contributes -h[i] = -h[j-shift] to grad[j]
        j_lo = i_lo + shift
        j_hi = i_hi + shift
        j_lo = max(j_lo, 0)
        j_hi = min(j_hi, n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += wk * (-h[idx - shift])
    return dx * grad

    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.9  # momentum
    
    # Alpha schedule: start low, increase
    alpha_seq = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    
    iters_per_alpha = 200
    total_budget = 0
    
    for alpha in alpha_seq:
        if time.time() > deadline:
            break
        eta = 1.0 / (alpha * 2.0 / n + 1e-10)  # initial step size
        eta = min(eta, 2.0)
        
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                return best_h, best_c
            
            # Nesterov step
            y = h + beta * v
            g = smoothed_grad(np.clip(y, 0, 1), n, alpha)
            v = beta * v - eta * g
            h_new = h - eta * g  # Actually for Nesterov: h_new = h + v_new
            # Let me use the standard Nesterov scheme:
            # y = h + beta * v
            # v_new = (1-beta) * (h - y_prev) - eta * grad(y)  -- no this is wrong
            
            # Standard Nesterov:
            # y_{k+1} = x_k + (beta_{k+1} - 1)/(1+beta_{k+1}) * (x_k - x_{k-1})
            # x_{k+1} = P(y_{k+1} - eta * grad(y_{k+1}))
            # Where beta_k = (1+sqrt(1+4k²))/(2k+2) approximately 1 - 1/(k+1)
            
            # Simpler: just use heavy ball momentum
            v = 0.9 * v - eta * g
            h_new = h + v
            h_new = project_feasible(h_new, n)
            
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            
            h = h_new
            eta *= 0.997  # slow decay

    best_h = h.copy()
    
    for it in range(n_iters):

            break
        
        # Anneal alpha
        progress = it / n_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        g = smoothed_grad(h, n, alpha)

        if g_norm < 1e-14:
            break

        # Backtracking line search
        eta = min(2.0, 1.0 / g_norm)
