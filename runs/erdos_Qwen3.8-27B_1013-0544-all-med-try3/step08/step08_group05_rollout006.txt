def simulated_annealing_binary(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = np.correlate(h.astype(float), 1-h, mode='full').astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    best_corr = corr.copy()
    
    T = cur_max * 0.01  # initial temperature
    T_min = cur_max * 1e-8
    cooling = 0.9995
    
    while time.time() < deadline:
        # Pick random zero and one to swap
        i = rng.choice(zeros)
        j = rng.choice(ones)
        
        # Compute new correlation
        # ... O(n) update
        new_corr = compute_new_corr(corr, h, i, j, n)
        new_max = new_corr.max()
        
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            # Accept
            h[i] = 1; h[j] = 0
            corr = new_corr
            cur_max = new_max
            # Update ones, zeros sets
            # ...
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                best_corr = corr.copy()
        
        T = max(T * cooling, T_min)
    
    return best_h, best_max * dx
