    # softmax weights
    ac = alpha * c
    ac_shifted = ac - ac.max()
    w = np.exp(ac_shifted)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj_and_grad(h, n, alpha):
    """Objective and gradient of smooth max using exp weights."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    ac = alpha * c
    ac -= ac.max()
    e = np.exp(ac)
    w = e / e.sum()
    obj = float((c.max() + np.log(e.sum()) / alpha) * dx)
    # Gradient
    # c[i] = sum_j h[j] * (1-h[j+n-1-i])
    # dc_i/dh[p] = (1-h[p+n-1-i]) if valid - h[p-n+1+i] if valid
    grad = np.zeros(n)
    for p in range(n):
        s1 = 0.0
        s2 = 0.0
        # First term: sum over valid i where p+n-1-i in [0,n-1]
        # i ranges from n+1-p to 2n-1-p, but also 0 <= i < 2n-1
        i_lo = max(0, n + 1 - p)
        i_hi = min(2 * n - 1, 2 * n - 1 - p)  # exclusive
        # p+n-1-i in [0,n-1] means 0 <= p+n-1-i < n, i.e., n+1-p <= i <= 2n-1-p
        # Also need 0 <= i < 2n-1
        if i_hi > i_lo:
            idx = p + n - 1 - np.arange(i_lo, i_hi)
            s1 = np.dot(w[i_lo:i_hi], 1.0 - h[idx])
        # Second term: sum over valid i where p-n+1+i in [0,n-1]
        # p-n+1+i in [0,n-1] means n-1-p <= i <= 2n-2-p
        # Also 0 <= i < 2n-1

        if j_hi > j_lo:

            s2 = np.dot(w[j_lo:j_hi], h[idx2])
        grad[p] = (s1 - s2) * dx
    return obj, grad

    """Penalize top-k correlation values."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_sorted = np.sort(c)[::-1]
    top = c_sorted[:k]

    except Exception:
        return h.copy()

def pair_search_full(h, n, deadline, deltas, max_passes=20):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40, max_passes=25):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    if not improved and time.time() > deadline:
                        return h, best_c
            if not improved:
                break
    return h, best_c

def optimize_block(h_start, n, t_limit):
    h = project_feasible(h_start.copy(), n)
    c = compute_c5(h, n)
    alphas = [500, 2000, 10000, 50000]
    for a in alphas:
        if time.time() > t_limit - 12:
            break
        h_new = slsqp_opt(h, n, a, maxiter=1200, ftol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < c - 1e-15:
            h, c = h_new, c_new
            h = project_feasible(h, n)
    remaining = t_limit - time.time()
    if remaining > 8:
        if n <= 200:
            h, c = pair_search_full(h, n, t_limit - 3,
                                    [0.01, 0.005, 0.002, 0.001], max_passes=15)
        else:
            h, c = near_pair_search(h, n, t_limit - 3,
                                    [0.01, 0.005, 0.002], radius=40, max_passes=15)
    if time.time() < t_limit - 10:
        h_new = slsqp_opt(h, n, 200000, maxiter=800, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < c:
            h, c = h_new, c_new
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Optimize on initial resolution
    alloc1 = min(170, (deadline - t0) * 0.18)
    h1, c1 = optimize_block(h_init, n_init, t0 + alloc1)
    if c1 < best_c:
        best_h, best_c, best_n = h1, c1, n_init

    # Phase 2: Try alternative resolutions
    x_ref = np.linspace(0, 2, best_n)
    n_alternatives = [100, 150, 250, 300, 400]
    for n_try in n_alternatives:
        if n_try == best_n:
            continue
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_ref, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(55, (deadline - time.time() - 50) * 0.22)
        if alloc < 18:
            break
        h_t, c_t = optimize_block(h_interp, n_try, time.time() + alloc)
        if c_t < best_c - 1e-15:
            best_c, best_h, best_n = c_t, h_t, n_try

    # Phase 3: Random restarts on best resolution
    remaining = deadline - time.time()
    n_restarts = max(3, int(remaining / 55))
    for r in range(n_restarts):
        if time.time() > deadline - 50:
            break
        scale = 0.02 + 0.10 * rng.rand()
        hp = best_h + rng.normal(0, scale, best_n)
        hp = project_feasible(hp, best_n)
        alloc = min(48, (deadline - time.time() - 40) * 0.35)
        if alloc < 12:
            break
        h_r, c_r = optimize_block(hp, best_n, time.time() + alloc)
        if c_r < best_c - 1e-15:
            best_c, best_h = c_r, h_r

    # Phase 4: Structured restart (smooth bump around current best)
    if time.time() < deadline - 60:
        x_best = np.linspace(0, 2, best_n)
        for bump_pos in [0.3, 0.7, 1.0, 1.5]:
            if time.time() > deadline - 45:
                break
            bump = np.zeros(best_n)
            bump += 0.15 * np.exp(-0.5 * ((x_best - bump_pos) / 0.25) ** 2)
            hp = best_h + (rng.rand() > 0.5) * bump - (rng.rand() > 0.5) * bump
            hp = project_feasible(hp, best_n)
            alloc = min(35, (deadline - time.time() - 30) * 0.3)
            if alloc < 10:
                break
            h_r, c_r = optimize_block(hp, best_n, time.time() + alloc)
            if c_r < best_c - 1e-15:
                best_c, best_h = c_r, h_r

    # Phase 5: Intensive local polish
    if time.time() < deadline - 20:
        n = best_n
        alloc_polish = min(80, deadline - time.time() - 12)
        if n <= 200:
            best_h, best_c = pair_search_full(best_h, n, time.time() + alloc_polish,
                                              [0.005, 0.002, 0.001, 0.0005], max_passes=12)
        else:
            best_h, best_c = near_pair_search(best_h, n, time.time() + alloc_polish,
                                              [0.005, 0.002, 0.001], radius=35, max_passes=12)

    # Phase 6: Final SLSQP with very high alpha
    if time.time() < deadline - 10:
        h_new = slsqp_opt(best_h, best_n, 500000, maxiter=1500, ftol=1e-16)
        c_new = compute_c5(h_new, best_n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 7: Ultra-fine pair search
    if time.time() < deadline - 5:
        n = best_n
        if n <= 200:
            best_h, best_c = pair_search_full(best_h, n, deadline - 2,
                                              [0.002, 0.001, 0.0005, 0.0002, 0.0001], max_passes=8)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 2,
                                              [0.002, 0.001, 0.0005], radius=25, max_passes=8)

    # Phase 8: Final SLSQP + tiny pair nudge
    if time.time() < deadline - 3:
        h_new = slsqp_opt(best_h, best_n, 1000000, maxiter=800, ftol=1e-17)
        c_new = compute_c5(h_new, best_n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    if time.time() < deadline - 1.5:
        n = best_n
        deltas_final = [0.001, 0.0005, 0.0002, 0.0001]
        if n <= 200:
            best_h, best_c = pair_search_full(best_h, n, deadline - 0.5,
                                              deltas_final, max_passes=4)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 0.5,
                                              deltas_final[:2], radius=20, max_passes=4)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
