def g_gradient(h, n, k):
    """Gradient of g[k] (the k-th element of full correlation) with respect to h."""
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # S = {0, ..., n-1-lag}
        # j in S: 0 <= j <= n-1-lag
        # j-lag in S: j >= lag, j <= n-1, j-lag <= n-1-lag => j <= n-1
        for j in range(n - lag):
            grad[j] += (1.0 - h[j + lag])  # j in S term
        for j in range(lag, n):
            grad[j] -= h[j - lag]  # j-lag in S term
    else:
        # lag' = -lag > 0, S = {lag', ..., n-1}
        lagp = -lag
        for j in range(lagp, n):
            grad[j] += (1.0 - h[j + lag])  # note: j + lag = j - lagp, must be >= 0
            # j in S: lagp <= j <= n-1, and j+lag = j-lagp must be in [0, n-1]
            # j-lagp >= 0 => j >= lagp (already satisfied)
            # j-lagp <= n-1 => j <= n-1+lagp (always true since j <= n-1 and lagp >= 0)
            # But we also need j+lag < n: j-lagp < n => true
            # AND j+lag >= 0: j-lagp >= 0 => j >= lagp (already in range)
        for j in range(0, n - lagp):
            grad[j] -= h[j - lag]  # j-lag = j+lagp in S: lagp <= j+lagp <= n-1
            # j+lagp >= lagp (always true for j>=0)
            # j+lagp <= n-1 => j <= n-1-lagp
            # And j in range: j < n-lagp
    return grad

    g = np.correlate(h, 1 - h, 'full')
    m = g.max()
    shifted = g - m

            ln = n - lp

def coordinate_descent(h0, n, deadline, max_sweeps=200):

    rng = np.random.RandomState(123)

        if time.time() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for s in range(n - 1):
            if time.time() > deadline:
                break
            i, j = int(idx[s]), int(idx[(s + 1) % n])
            g_cur = compute_all_g(h, n)
            ci = h[i]
            cj = h[j]
            dmin = max(-ci, -(1.0 - cj))
            dmax = min(1.0 - ci, cj)
            if dmax - dmin < 1e-12:
                continue
            ntry = 12
            for t in range(ntry):
                d = dmin + (dmax - dmin) * t / (ntry - 1)
                if abs(d) < 1e-12:

                h[i] = ci + d
                h[j] = cj - d

                    best_h = h.copy()
                    improved = True
                    break
                h[i] = ci
                h[j] = cj
        if not improved:
            if sweep > 20:
                break
    return best_h, best_c

        i = rng.choice(zeros)
        j = rng.choice(ones)
        h[i], h[j] = 1.0, 0.0
        c_new = compute_c5(h, n)
        delta = c_new - best_c
        if delta < 0 or rng.random() < np.exp(-max(delta * dx * 50, 0) / max(T, 1e-10)):
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                T = min(T * 1.05, 0.05)
        else:
            h[i], h[j] = 0.0, 1.0
    return best_h, best_c


def interpolate(h, n_old, n_new):
    xo = np.linspace(0, 2, n_old)
    xn = np.linspace(0, 2, n_new)
    hh = np.interp(xn, xo, h)
    return project_feasible(hh, n_new)


def multi_alpha_smooth(h0, n, deadline, alphas, iters_each=3000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h, c = smooth_gradient_descent(best_h.copy(), n, deadline, n_iters=iters_each)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    def elapsed():
        return time.time() - t0

    # Phase 1: Smooth gradient descent at native resolution (25% budget)
    p1_end = t0 + 0.25 * (budget_s - 8)
    h1, c1 = multi_alpha_smooth(best_h.copy(), n, p1_end,
                                [200, 500, 1000, 2000, 5000, 10000], iters_each=4000)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Exact subgradient polish (10% budget)
    p2_end = t0 + 0.35 * (budget_s - 8)
    h2, c2 = subgradient_exact(best_h.copy(), n, p2_end, n_iters=8000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Higher resolution (25% budget)
    p3_end = t0 + 0.60 * (budget_s - 8)
    for nh in [300, 400, 500]:
        if time.time() > p3_end - 40:
            break
        hh = interpolate(best_h, n, nh)
        hh, ch = multi_alpha_smooth(hh, nh, p3_end - 20,
                                    [500, 2000, 8000], iters_each=2000)
        hh, ch = subgradient_exact(hh, nh, p3_end - 10, n_iters=3000)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh

    # Phase 4: Binary SA diversification (15% budget)
    p4_end = t0 + 0.75 * (budget_s - 8)
    if time.time() < p4_end - 30:
        hb, cb = binary_sa(best_h.copy(), n, p4_end, rng, n_iters=40000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        # Try SA from perturbed continuous solution
        hp = project_feasible(best_h + 0.15 * rng.randn(n), n)
        hb2, cb2 = binary_sa(hp, n, p4_end, rng, n_iters=20000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 5: Coordinate descent polish (15% budget)
    p5_end = t0 + 0.90 * (budget_s - 8)
    if time.time() < p5_end - 10:
        h5, c5v = coordinate_descent(best_h.copy(), n, p5_end, max_sweeps=150)
        if c5v < best_c:
            best_c = c5v
            best_h = h5.copy()

    # Phase 6: Final polishing with smooth GD at high alpha (7% budget)
    if time.time() < dl - 15:
        h6, c6 = smooth_gradient_descent(best_h.copy(), n, dl - 10, n_iters=5000)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        h6b, c6b = subgradient_exact(best_h.copy(), n, dl - 5, n_iters=3000)
        if c6b < best_c:
            best_c = c6b
            best_h = h6b.copy()

    # Final coordinate descent if time remains
    if time.time() < dl - 5:
        hf, cf = coordinate_descent(best_h.copy(), n, dl, max_sweeps=50)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
