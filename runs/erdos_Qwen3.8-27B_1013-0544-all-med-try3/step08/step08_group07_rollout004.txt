```python
import cvxpy as cp

def lp_minmax(h, n, K, c_vals, grads):
    """Solve LP: min t s.t. c_k + grad_k^T (h-h0) <= t"""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    for k in K:
        constraints.append(c_vals[k] + grads[k] @ (h_var - h) <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLOP)  # or ECOS, SCS
    
    if h_var.value is not None:
        return h_var.value, t_var.value
    return h, t_var.value
```

def binary_local_search(h, n, deadline, max_sweeps=500):
    """Binary 1-swap local search"""
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0])
    zeros = set(np.where(h_bin == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try all 1-swaps
        for i in list(zeros):
            if time.time() > deadline:
                break
            for j in list(ones):

                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                else:

        if not improved:
            break
    
    return h_bin, best_c
