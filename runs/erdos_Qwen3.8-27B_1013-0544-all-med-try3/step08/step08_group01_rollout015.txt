                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def restart_and_optimize(h, n, deadline, n_restarts=3, noise_scale=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        rng = np.random.RandomState(1000 + r)
        h_pert = h + rng.uniform(-noise_scale, noise_scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        t_sub = min(time.time() + 40, deadline - 20)
        try:
            res = minimize(smooth_obj, h_pert, jac=smooth_grad,
                          args=(n, 100000, 100.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': 20000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick L-BFGS at original resolution
    if time.time() < deadline - 600:
        t1 = min(time.time() + 60, deadline - 600)
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            configs=[(500, 50.0), (2000, 500.0), (5000, 5000.0)])

    # Phase 2: Upsample to n=1000 for higher resolution
    n2 = 1000
    h2 = upsample(best_h, n_orig, n2)
    c2 = compute_c5(h2, n2)
    if c2 > best_c:
        # Upsampling can slightly increase; still proceed
        pass

    # Phase 3: Main L-BFGS at n=1000
    if time.time() < deadline - 400:
        t3 = min(time.time() + 180, deadline - 400)
        h2, c2 = run_lbfgs(h2, n2, t3,
            configs=[(500, 50.0), (2000, 500.0), (5000, 5000.0),
                     (10000, 20000.0), (20000, 100000.0), (50000, 500000.0)])
    if c2 < best_c:
        best_c = c2

    # Phase 4: Active-set focused optimization at n=1000
    if time.time() < deadline - 280:
        t4 = min(time.time() + 100, deadline - 280)
        h4, c4 = run_active_lbfgs(h2, n2, t4, top_k=12,
                                   alphas=[20000, 100000, 500000, 2000000],
                                   mu=0.01)
        if c4 < best_c:
            best_c = c4

    # Phase 5: Restart at n=1000
    if time.time() < deadline - 220:
        t5 = min(time.time() + 80, deadline - 220)
        h5, c5 = restart_and_optimize(h2, n2, t5, n_restarts=3, noise_scale=0.008)
        if c5 < best_c:
            best_c = c5
        else:
            pass

    # Re-sync best_h with n2
    best_h = project_feasible(h2, n2)
    best_c = compute_c5(best_h, n2)

    # Phase 6: Try n=2000
    n3 = 2000
    if time.time() < deadline - 200:
        h3 = upsample(best_h, n2, n3)
        c3 = compute_c5(h3, n3)
        t6 = min(time.time() + 140, deadline - 200)
        h3, c3 = run_lbfgs(h3, n3, t6,
            configs=[(1000, 100.0), (5000, 1000.0), (10000, 5000.0),
                     (20000, 50000.0)])
        if c3 < best_c:
            best_c = c3
            best_h = h3
        else:
            best_h = project_feasible(h2, n2)
            best_c = compute_c5(best_h, n2)

    # Phase 7: Active-set at final resolution
    n_final = len(best_h)
    if time.time() < deadline - 100:
        t7 = min(time.time() + 60, deadline - 100)
        h7, c7 = run_active_lbfgs(best_h, n_final, t7, top_k=8,
                                   alphas=[50000, 200000, 1000000],
                                   mu=0.01)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Another L-BFGS with very high alpha (focus on single worst shift)
    if time.time() < deadline - 50:
        t8 = min(time.time() + 40, deadline - 50)
        h8, c8 = run_lbfgs(best_h, n_final, t8,
            configs=[(100000, 100000.0), (500000, 500000.0)])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Fine pair search at final resolution (small radius for higher n)
    if time.time() < deadline - 20:
        radius = 8 if n_final >= 1000 else 15
        t9 = min(time.time() + 25, deadline - 20)
        h9, c9 = fine_pair_search(best_h, n_final, t9, radius=radius, max_passes=2)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final: ensure feasibility and compute exact value
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
