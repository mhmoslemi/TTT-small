```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_constraint(x, n):
    return float(x.sum() - n / 2.0)

def sum_jacobian(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or res.fun < obj_smooth(h, n, alpha):
            return project_feasible(res.x, n)
        return project_feasible(h, n)
    except Exception:
        return project_feasible(h, n)

    g = np.zeros((n, 2*n-1))
    offset = n - 1
    for j in range(n):
        for lag in range(-(n-1), n):
            idx = lag + offset
            val = 0.0
```

  sum(h') = n/2
  0 ≤ h' ≤ 1

This is a linear program! And we can solve it with scipy or cvxpy.

This is actually a very good approach because:
1. It's exact for linear objectives
2. The trust region keeps us close to the current point (ensuring the linearization is good)
3. It naturally finds the best direction to move
4. We can use the solution as the new point and repeat

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_constraint(x, n):
    return float(x.sum() - n / 2.0)

def sum_jacobian(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new, n) < compute_c5(h, n):
                return h_new
        return project_feasible(h, n)
    except Exception:
        return project_feasible(h, n)

def compute_gradients(h, n):
    g = np.zeros((n, 2*n-1))
    offset = n - 1
    for j in range(n):
        for lag in range(-(n-1), n):
            idx = lag + offset
            val = 0.0
            jp = j + lag
            jm = j - lag
            if 0 <= jp < n:
                val += (1.0 - h[jp])
            if 0 <= jm < n:
                val -= h[jm]
            g[j, idx] = val
    return g

def lp_local_search(h, n, deadline, max_iter=200, eps_start=0.05):
    best_c = compute_c5(h, n)
    eps = eps_start
    n_lags = 2 * n - 1
    
    for iteration in range(max_iter):
        if time.time() > deadline:
            break
        
        c = compute_c_all(h, n)
        g = compute_gradients(h, n)
        
        # Build LP
        # Variables: [t, h'_0, ..., h'_{n-1}]
        c_obj = np.zeros(1 + n)
        c_obj[0] = 1.0
        
        # Inequality constraints: -t + sum_j g[j,ℓ] h'[j] <= -c[ℓ] + sum_j g[j,ℓ] h[j]
        A_ub_rows = []
        b_ub_vals = []
        
        for ell in range(n_lags):
            row = np.zeros(1 + n)
            row[0] = -1.0
            row[1:] = g[:, ell]
            A_ub_rows.append(row)
            b_ub_vals.append(-c[ell] + g[:, ell] @ h)
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        # Equality: sum(h') = n/2
        A_eq = np.zeros((1, 1 + n))
        A_eq[0, 1:] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds with trust region
        bounds = [(0, None)]  # t >= 0
        for j in range(n):
            lo = max(0.0, h[j] - eps)
            hi = min(1.0, h[j] + eps)
            bounds.append((lo, hi))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                         method='highs')
            if res.status == 0:
                h_new = res.x[1:]
                h_new = np.clip(h_new, 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    h = h_new
                    # Keep eps same or slightly increase
                    eps = min(eps * 1.1, 0.1)
                else:
                    eps *= 0.5
                    if eps < 0.005:
                        break
            else:
                break
        except Exception:
            break
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_constraint(x, n):
    return float(x.sum() - n / 2.0)

def sum_jacobian(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new, n) < compute_c5(h, n) + 1e-15:
                return h_new
        return project_feasible(h, n)
    except Exception:
        return project_feasible(h, n)

def compute_gradients(h, n):
    g = np.zeros((n, 2 * n - 1))
    offset = n - 1
    for j in range(n):
        for lag in range(-(n - 1), n):
            idx = lag + offset
            val = 0.0
            jp = j + lag
            jm = j - lag
            if 0 <= jp < n:
                val += (1.0 - h[jp])
            if 0 <= jm < n:
                val -= h[jm]
            g[j, idx] = val
    return g

def lp_local_search(h, n, deadline, max_iter=300, eps_start=0.04):
    best_c = compute_c5(h, n)
    eps = eps_start
    n_lags = 2 * n - 1
    no_improve_count = 0
    
    for iteration in range(max_iter):
        if time.time() > deadline:
            break
        
        c = compute_c_all(h, n)
        g = compute_gradients(h, n)
        
        c_obj = np.zeros(1 + n)
        c_obj[0] = 1.0
        
        A_ub_rows = []
        b_ub_vals = []
        
        for ell in range(n_lags):
            row = np.zeros(1 + n)
            row[0] = -1.0
            row[1:] = g[:, ell]
            A_ub_rows.append(row)
            b_ub_vals.append(-c[ell] + g[:, ell] @ h)
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        A_eq = np.zeros((1, 1 + n))
        A_eq[0, 1:] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, None)]
        for j in range(n):
            lo = max(0.0, h[j] - eps)
            hi = min(1.0, h[j] + eps)
            bounds.append((lo, hi))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                          method='highs')
            if res.status == 0:
                h_new = np.clip(res.x[1:], 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-13:
                    improvement = best_c - c_new
                    best_c = c_new
                    h = h_new
                    no_improve_count = 0
                    if improvement > 1e-5:
                        eps = min(eps * 1.2, 0.08)
                else:
                    no_improve_count += 1
                    eps *= 0.6
                    if eps < 0.003 or no_improve_count > 50:
                        break
            else:
                break
        except Exception:
            break
    
    return h, best_c

def gradient_pair_search(h, n, deadline, max_iter=200):
    best_c = compute_c5(h, n)
    n_lags = 2 * n - 1
    
    for iteration in range(max_iter):
        if time.time() > deadline:
            break
        
        c = compute_c_all(h, n)
        g = compute_gradients(h, n)
        M = c.max()
        
        threshold = M - 0.002
        critical = np.where(c >= threshold)[0]
        if len(critical) == 0:
            critical = np.where(c >= M - 0.01)[0]
        
        score_inc = -g[:, critical].mean(axis=1)
        score_dec = g[:, critical].mean(axis=1)
        
        top_k = min(40, n)
        top_inc = np.argsort(-score_inc)[:top_k]
        top_dec = np.argsort(-score_dec)[:top_k]
        
        best_improvement = 0
        best_move = None
        
        for ii in range(len(top_inc)):
            i = top_inc[ii]
            for jj in range(len(top_dec)):
                j = top_dec[jj]
                if i == j:
                    continue
                d = g[i] - g[j]
                delta_max = min(1.0 - h[i], h[j])
                if delta_max <= 1e-10:
                    continue
                d_crit = d[critical]
                if d_crit.mean() >= 0:
                    continue
                for frac in [0.4, 0.6, 0.8]:
                    delta = delta_max * frac
                    new_c = c + delta * d
                    new_M = new_c.max()
                    if new_M < best_c - 1e-13:
                        if h[i] + delta <= 1.0 + 1e-10 and h[j] - delta >= -1e-10:
                            improvement = best_c - new_M
                            if improvement > best_improvement:
                                best_improvement = improvement
                                best_move = (i, j, delta)
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] += delta
            h[j] -= delta
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
            else:
                h[i] -= delta
                h[j] += delta
                h = project_feasible(h, n)
                break
        else:
            break
    
    return h, best_c
```
