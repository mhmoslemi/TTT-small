import numpy as np
from scipy.optimize import minimize
import time

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_2opt(h, n, deadline, max_sweeps=500000):
    best_c = compute_c5(h, n)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for zi in range(len(zeros)):
            if time.time() > deadline:
                break
            z = zeros[zi]
            for oi in range(len(ones)):
                o = ones[oi]
                h[z] = 1.0
                h[o] = 0.0
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones[oi] = z
                    zeros[zi] = o
                    improved = True
                else:
                    h[z] = 0.0
                    h[o] = 1.0
        if not improved:
            break
    return h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        a = alpha * (g - m)
        am = a.max()
        return m + (am + np.log(np.exp(a - am).sum())) / alpha
    
    def con_fun(x):
        return x.sum() - target
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 1000})
        return project_feasible(res.x, n)
    except Exception:
        return h

def multi_alpha_slsqp(h, n, deadline, alphas, maxiter_each):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        h_new = slsqp_smooth(best_h.copy(), n, alpha, maxiter_each, deadline)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
    return best_h, best_c
