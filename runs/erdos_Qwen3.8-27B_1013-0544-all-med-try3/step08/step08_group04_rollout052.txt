def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

            if tot < 1e-15:

def smooth_max_obj(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh_shift = sh - sh.max()
    logsum = sh.max() + np.log(np.sum(np.exp(sh_shift)))

        h_parent = np.full(250, 0.5)
        n_parent = 250
    
    best_h = h_parent.copy()
    best_c = compute_c5(best_h, n_parent)
    
    # Phase 1: Optimize parent resolution (250s)
    t1 = t0 + min(220, (deadline - t0) * 0.22)
    best_h, best_c = multi_alpha_optimize(best_h, n_parent, t1)
    
    # Phase 2: Try higher resolution n=350
    if time.time() < deadline - 150:
        n_hi = 350
        x_old = np.linspace(0, 2, n_parent)
        x_new = np.linspace(0, 2, n_hi)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_feasible(h_hi, n_hi)
        t2 = time.time() + min(130, (deadline - time.time()) * 0.4)
        h_opt, c_opt = multi_alpha_optimize(h_hi, n_hi, t2)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Try n=400 if time permits
    if time.time() < deadline - 100:
        n_hi2 = 400
        n_cur = len(best_h)
        x_old = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_hi2)
        h_hi2 = np.interp(x_new, x_old, best_h)
        h_hi2 = project_feasible(h_hi2, n_hi2)
        t3 = time.time() + min(80, (deadline - time.time()) * 0.3)
        h_opt, c_opt = multi_alpha_optimize(h_hi2, n_hi2, t3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 4: Multi-restart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 60:
        n_restarts = 4
        perturb_scales = [0.04, 0.08, 0.12, 0.02]
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = perturb_scales[r]
            hp = best_h + rng.normal(0, scale, n_best)
            # For r=2, also try a block perturbation
            if r == 2:
                block = max(1, n_best // 8)
                for b in range(0, n_best, block):
                    e = min(n_best, b + block)
                    hp[b:e] += rng.uniform(-0.15, 0.15)
            hp = project_feasible(hp, n_best)
            alloc = min(55, (deadline - time.time() - 15) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = multi_alpha_optimize(hp, n_best, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Fine-tuning with pair search
    n = len(best_h)
    if time.time() < deadline - 8:
        if n <= 200:
            best_h, best_c = paired_coord_search(best_h, n, deadline - 3, max_passes=2)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 3, radius=min(30, n//5), max_passes=2)
    
    # Phase 6: Peak-directed fine tuning
    if time.time() < deadline - 5:
        best_h, best_c = peak_directed_shave(best_h, n, deadline - 2, rng, n_trials=2000)
    
    # Phase 7: Final L-BFGS-B polish with very high alpha
    if time.time() < deadline - 4:
        ht = lfbg_run(best_h, n, 100000, 300.0, maxiter=300)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Phase 8: Ultra-fine pair search
    if time.time() < deadline - 1:
        n = len(best_h)
        if n <= 200:
            best_h, best_c = paired_coord_search(best_h, n, deadline, max_passes=1)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline, radius=min(20, n//5), max_passes=1)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
