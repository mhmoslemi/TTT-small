def top_shifts_lse(x, n, top_idx, alpha, mu):

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def top_shifts_lse_grad(x, n, top_idx, alpha, mu):

    vals = c[top_idx]
    m = vals.max()

        w[idx] = w_local[i]
    g = corr_grad_weighted(x, n, w)

    return g

def pair_search(h, n, deadline, max_radius=50):
    """Fine pair search with decreasing step sizes."""
    best_c = compute_c5(h, n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    
    for step in step_sizes:
        if time.time() > deadline - 1:
            break
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            # Sample random pairs for speed
            n_trials = 20000

                    return h, best_c

                    h[j] -= step

def slqp_optim(h, n, deadline, alpha=100000, maxiter=5000):

        return lse_obj(x, n, alpha, 0.0)

        return x.sum() - n / 2.0

        return np.ones(n)

                      bounds=bounds, constraints=cons,

def gradient_descent_step(h, n, lr, active_shifts):
    """One step of projected gradient descent on active shifts."""
    c_vec = compute_c5_vec(h, n)
    # Average gradient over active shifts
    grad = np.zeros(n)
    for idx in active_shifts:
        grad += corr_grad_single(h, n, idx)
    grad /= len(active_shifts)
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new

def subgradient_method(h, n, deadline, lr_init=0.01, lr_decay=0.95):
    """Subgradient method with decreasing step sizes."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    
    while time.time() < deadline - 2:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) == 0:
            active = np.argsort(-c_vec)[:5]
        
        # Gradient step
        grad = np.zeros(n)
        for idx in active:
            grad += corr_grad_single(h, n, idx)
        grad /= len(active)
        
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Backtrack
            lr *= lr_decay
        
        if lr < 1e-8:
            break
    
    return best_h, best_c

def perturbation_restart(h, n, deadline, n_restarts=5):

                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def run_slqp(h, n, deadline, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_eq_fun, 'jac': slqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(lse_full_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def coordinate_descent(h, n, deadline, max_passes=200):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n

    for p in range(max_passes):
        if time.time() > deadline - 1:
            break
        for j in range(n):
            if time.time() > deadline - 1:
                return best_h, best_c
            hj = h[j]
            # Optimal h[j] minimizes max over affected shifts
            # Try a range of values around hj
            lo = max(0.0, hj - 0.05)
            hi = min(1.0, hj + 0.05)
            # Coarse search
            candidates = np.linspace(lo, hi, 11)
            best_val = best_c
            best_candidate = hj
            for cv in candidates:
                if abs(cv - hj) < 1e-15:
                    continue
                h[j] = cv
                # Adjust another coordinate to maintain sum
                # Actually, let's just evaluate and track
                c = compute_c5(h, n)
                if c < best_val:
                    best_val = c
                    best_candidate = cv
            h[j] = best_candidate

            if best_val < best_c - 1e-15:
                best_c = best_val
                best_h = h.copy()

        # Re-project to maintain constraint
        h = project_feasible(h, n)
        c_check = compute_c5(h, n)
        if c_check < best_c:
            best_c = c_check
            best_h = h.copy()

    return best_h, best_c

def pair_search_grad_guided(h, n, deadline, base_step=0.01):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = base_step

    while time.time() < deadline - 2 and step > 1e-7:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-c_vec)[:5]

        grad = corr_grad_top(h, n, active)
        grad /= len(active)

        # Find best 2-point move
        n_top = min(80, n)
        g_sorted = np.argsort(-np.abs(grad))[:n_top]
        improved = False

        for _ in range(500):
            if time.time() > deadline - 2:
                break
            i = g_sorted[np.random.randint(n_top)]
            j = g_sorted[np.random.randint(n_top)]
            if i == j:
                continue
            direction = -1 if grad[i] > grad[j] else 1
            for mult in [1.0, 0.5, 0.2, 0.1]:
                s = step * mult
                if direction > 0:
                    if h[i] - s >= 0 and h[j] + s <= 1:
                        h[i] -= s
                        h[j] += s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += s
                            h[j] -= s
                else:
                    if h[i] + s <= 1 and h[j] - s >= 0:
                        h[i] += s
                        h[j] -= s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= s
                            h[j] += s
        if not improved:
            step *= 0.5

    return best_h, best_c

def fine_pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]

    for d in deltas:
        if time.time() > deadline - 1:
            break
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            n_trials = 5000
            for _ in range(n_trials):
                if time.time() > deadline - 1:
                    break
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
                if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    h[i] -= d
                    h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
    return best_h, best_c

def subgradient_polish(h, n, deadline, lr_init=0.002):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init

    while time.time() < deadline - 2 and lr > 1e-9:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-c_vec)[:5]

        grad = corr_grad_top(h, n, active)
        grad /= len(active)

        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.85

    return best_h, best_c

def multi_start_perturb(h, n, deadline, n_starts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t_per = max(10.0, (deadline - time.time()) / max(1, n_starts))

    for _ in range(n_starts):
        if time.time() > deadline - 5:
            break
        h_p = h.copy()
        mask = np.random.random(n) < 0.15
        h_p[mask] += np.random.randn(int(mask.sum())) * 0.03
        h_p = project_feasible(h_p, n)

        t_end = min(time.time() + t_per, deadline - 5)
        h_p, c_p = run_top_shifts(h_p, n, t_end, top_k=5,
                                   alphas=[5000, 50000, 500000], mu=0.01)

        if c_p < best_c:
            best_c = c_p
            best_h = h_p

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_orig = 200

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick L-BFGS at original resolution
    t1 = min(t0 + 40, deadline - 900)
    if t1 > t0 + 5:
        best_h, best_c = run_lbfgs_full(best_h, n_orig, t1,
            configs=[(500, 100.0), (5000, 1000.0), (20000, 10000.0),
                     (50000, 100000.0)])

    # Phase 2: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h_new = np.interp(x_new, x_old, best_h)
    h_new = project_feasible(h_new, n)
    best_h = h_new
    best_c = compute_c5(h_new, n)

    # Phase 3: L-BFGS full at n=1000 (short, low alpha)
    t3 = min(time.time() + 120, deadline - 700)
    if time.time() < t3:
        best_h, best_c = run_lbfgs_full(best_h, n, t3,
            configs=[(100, 20.0), (500, 100.0), (2000, 1000.0)])

    # Phase 4: Top-shifts focused (main optimization phase)
    t4 = min(time.time() + 350, deadline - 350)
    if time.time() < t4:
        best_h, best_c = run_top_shifts(best_h, n, t4, top_k=5,
                                         alphas=[5000, 20000, 100000, 500000, 2000000],
                                         mu=0.01)

    # Phase 5: SLSQP
    if time.time() < deadline - 250:
        t5 = min(time.time() + 80, deadline - 250)
        h5, c5 = run_slqp(best_h, n, t5, alpha=100000, maxiter=3000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Top-shifts with smaller top_k
    t6 = min(time.time() + 150, deadline - 150)
    if time.time() < t6:
        best_h, best_c = run_top_shifts(best_h, n, t6, top_k=3,
                                         alphas=[50000, 200000, 1000000],
                                         mu=0.005)

    # Phase 7: Subgradient polish
    t7 = min(time.time() + 80, deadline - 100)
    if time.time() < t7:
        h7, c7 = subgradient_polish(best_h, n, t7, lr_init=0.003)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Gradient-guided pair search
    t8 = min(time.time() + 100, deadline - 60)
    if time.time() < t8:
        best_h, best_c = pair_search_grad_guided(best_h, n, t8, base_step=0.005)

    # Phase 9: Fine pair search
    t9 = min(time.time() + 80, deadline - 30)
    if time.time() < t9:
        best_h, best_c = fine_pair_search(best_h, n, t9)

    # Phase 10: Final L-BFGS polish
    if time.time() < deadline - 30:
        t10 = min(time.time() + 30, deadline - 10)
        best_h, best_c = run_top_shifts(best_h, n, t10, top_k=5,
                                         alphas=[500000, 2000000, 5000000],
                                         mu=0.01)

    # Final: check at original resolution too
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    if n_orig != n_final:
        x_fin = np.linspace(0, 2, n_final)
        x_og = np.linspace(0, 2, n_orig)
        h_og = np.interp(x_og, x_fin, best_h)
        h_og = project_feasible(h_og, n_orig)
        c_og = compute_c5(h_og, n_orig)
        if c_og < best_c:
            best_h = h_og
            best_c = c_og
            n_final = n_orig

    # One more tiny polish at the chosen resolution
    if time.time() < deadline - 5:
        h_fp, c_fp = fine_pair_search(best_h, n_final, deadline - 3)
        if c_fp < best_c:
            best_c = c_fp
            best_h = h_fp

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
