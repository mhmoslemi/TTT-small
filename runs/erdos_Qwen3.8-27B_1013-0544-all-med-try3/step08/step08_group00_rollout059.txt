def project_feasible(h, n):
    """Project h onto [0,1]^n ∩ {sum = n/2} using bisection."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-15:
        return h.copy()
    # Bisection for water-filling
    # We want sum(clip(h + t, 0, 1)) = target
    # f(t) = sum(clip(h + t, 0, 1)) is monotone in t
    lo, hi = -1.0, 1.0
    if s > target:
        # Need to decrease: find t < 0
        lo, hi = -1.0, 0.0
    else:
        # Need to increase: find t > 0
        lo, hi = 0.0, 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        h_new = np.clip(h + mid, 0.0, 1.0)
        s_new = h_new.sum()
        if s_new > target:
            hi = mid
        else:
            lo = mid
    h_new = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Final correction
    diff = target - h_new.sum()
    if abs(diff) > 1e-15:
        # Distribute remaining diff
        if diff > 0:
            mask = h_new < 1.0
            if mask.sum() > 0:
                h_new[mask] += diff / mask.sum()
        else:
            mask = h_new > 0.0
            if mask.sum() > 0:
                h_new[mask] += diff / mask.sum()
    return np.clip(h_new, 0.0, 1.0)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hc = np.clip(h - mid, 0.0, 1.0)
        sc = hc.sum()
        if sc > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def compute_smooth_gradient(h, n, alpha, top_k=None):
    c = compute_c5_vec(h, n)
    if top_k is not None and top_k < len(c):
        idx_all = np.argsort(-c)
        idx = idx_all[:top_k]
    else:
        idx = np.arange(len(c))
    w = np.exp(alpha * (c[idx] - c[idx].max()))
    s = w.sum()
    if s < 1e-300:
        return np.zeros(n)
    w = w / s
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, mm in zip(w, idx):
        if wi < 1e-14:
            continue
        k = int(mm) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                if k < n:
                    grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                if kk < n:
                    grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_jac(x, n, alpha, mu):
    g = compute_smooth_gradient(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, alphas=None, mu=0.5, top_k=200):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000]
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_jac,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 80000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c
