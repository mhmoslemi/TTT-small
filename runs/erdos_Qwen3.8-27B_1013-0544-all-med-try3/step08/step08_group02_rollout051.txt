import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        logsum = np.log(np.sum(np.exp(sh)))
    val = m + logsum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)

    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slqp_minmax(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slqp_func(x, n):
    return x.sum() - n / 2.0

def slqp_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, max_iters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if max_iters is not None and i < len(max_iters):
            maxit = max_iters[i]
        else:
            maxit = max(500, int(remaining * 4))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-14:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_lbfgs(h, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000, 5000000]
    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 8:
                break
            maxit = max(400, int((deadline - time.time()) * 2.5))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run_slqp_phase(h, n, deadline, top_k=10, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_func, 'jac': slqp_jac, 'args': (n,)}]
    if alphas is None:
        alphas = [50000, 200000, 1000000]
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        maxit = max(500, int(remaining * 1.5))
        try:
            res = minimize(slqp_minmax, best_h, args=(n, active, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxit, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def coord_descent(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = best_h[i]
            # Try a few perturbations
            for delta in [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005, 0.0002, -0.0002]:
                nv = old + delta
                if nv < 0 or nv > 1:
                    continue
                best_h[i] = nv
                # Fix sum constraint by adjusting another coordinate
                diff = (old - nv)
                if abs(diff) > 1e-15:
                    # Find best partner to adjust
                    j = (i + 1) % n
                    for _j in range(n):
                        j = (i + _j + 1) % n
                        if j == i:
                            continue
                        correction = diff / (n - 1) if n > 1 else 0
                        jn = best_h[j] + diff * (1.0 / (n - 1))
                        if 0 <= jn <= 1:
                            best_h[j] = jn
                            break
                best_h = project_feasible(best_h, n)
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                    break
                best_h[i] = old
        if not improved:
            break
    return best_h, best_c

def pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_expand = 0.002
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def upsample(h_coarse, n_coarse, n_fine):
    x_c = np.linspace(0, 2, n_coarse, endpoint=False)
    x_f = np.linspace(0, 2, n_fine, endpoint=False)
    h_f = np.interp(x_f, x_c, h_coarse)
    return project_feasible(h_f, n_fine)

def downsample(h_fine, n_fine, n_coarse):
    x_f = np.linspace(0, 2, n_fine, endpoint=False)
    x_c = np.linspace(0, 2, n_coarse, endpoint=False)
    h_c = np.interp(x_c, x_f, h_fine)
    return project_feasible(h_c, n_coarse)

def perturbed_start(h, n, sigma, rng):
    h_p = h + rng.normal(0, sigma, size=n)
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)

    # Primary resolution: n=1000 for finer control
    n = 1000
    h = upsample(h0, n0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    best_n = n

    # Phase 1: L-BFGS smoothing at n=1000 (200s)
    t1_end = t0 + 200
    h1, c1 = run_lbfgs(h, n, t1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.01, max_iters=[2000, 2000, 3000, 3000, 3000, 4000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active-set L-BFGS (150s)
    if time.time() < deadline - 300:
        t2_end = min(time.time() + 150, deadline - 300)
        h2, c2 = run_active_lbfgs(best_h, n, t2_end, top_k=6,
            alphas=[10000, 100000, 1000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: SLSQP with active constraints (150s)
    if time.time() < deadline - 200:
        t3_end = min(time.time() + 150, deadline - 200)
        h3, c3 = run_slqp_phase(best_h, n, t3_end, top_k=8,
            alphas=[50000, 200000, 1000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: More L-BFGS with very high alpha (150s)
    if time.time() < deadline - 150:
        t4_end = min(time.time() + 150, deadline - 150)
        h4, c4 = run_lbfgs(best_h, n, t4_end,
            alphas=[100000, 500000, 2000000, 5000000],
            mu=0.001, max_iters=[3000, 4000, 4000, 4000])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try n=1500 if time permits (150s)
    if time.time() < deadline - 120:
        n15 = 1500
        h15 = upsample(best_h, n, n15)
        c15 = compute_c5(h15, n15)
        t5_end = min(time.time() + 150, deadline - 100)
        h15o, c15o = run_lbfgs(h15, n15, t5_end,
            alphas=[50, 500, 5000, 50000, 500000],
            mu=0.005, max_iters=[1500, 2000, 2500, 3000, 3000])
        if c15o < best_c:
            best_c = c15o
            best_h = h15o
            best_n = n15

    # Phase 6: Active-set refinement (100s)
    if time.time() < deadline - 60:
        t6_end = min(time.time() + 100, deadline - 50)
        h6, c6 = run_active_lbfgs(best_h, best_n, t6_end, top_k=4,
            alphas=[100000, 1000000, 5000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = len(best_h)

    # Phase 7: Subgradient refinement (60s)
    if time.time() < deadline - 30:
        t7_end = min(time.time() + 60, deadline - 20)
        h7, c7 = subgrad_refine(best_h, best_n, t7_end, n_iter=5000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = len(best_h)

    # Phase 8: Pair search (remaining time minus 5s)
    if time.time() < deadline - 3:
        h8, c8 = pair_search(best_h, best_n, deadline - 2, radius=40)
        if c8 < best_c:
            best_c = c8
            best_h = h8
            best_n = len(best_h)

    # Final: also try at original resolution n0 in case it's better
    if time.time() < deadline - 10 and best_n != n0:
        h_back = downsample(best_h, best_n, n0)
        c_back = compute_c5(h_back, n0)
        # Quick optimization at n0
        h_back2, c_back2 = run_lbfgs(h_back, n0, min(time.time() + 8, deadline - 3),
            alphas=[10000, 100000], mu=0.01, max_iters=[1000, 1500])
        if c_back2 < best_c:
            best_c = c_back2
            best_h = h_back2
            best_n = n0

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
