        for t in range(5000):
            if time.time() > deadline - 3:
                break
            if k_top < 2 * n - 1 and alpha > 5000:
                g = get_grad_topk(h_cur, n, min(k_top, 2 * n - 1), alpha)
            else:
                g = get_grad(h_cur, n, alpha)
            lr = lr_scale / np.sqrt(t + 1)
            h_new = project_feasible(h_cur - lr * g, n)
            h_cur = h_new
            if t % 300 == 299:
                c_cur = compute_c5(h_cur, n)
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h_cur.copy()
    return best_h, best_c

def gradient_pair_search(h, n, deadline, alpha=50000, max_rounds=500):
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        g = get_grad(h, n, alpha)
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(n):
                if i == j:
                    continue
                for sign in [1.0, -1.0]:
                    dir_i = sign * (-g[i])
                    dir_j = sign * g[j]
                    if abs(dir_i) < 1e-12 and abs(dir_j) < 1e-12:
                        continue
                    max_step = 0.03
                    if dir_i > 0:
                        max_step = min(max_step, (1.0 - h[i]) / dir_i)
                    elif dir_i < 0:
                        max_step = min(max_step, h[i] / (-dir_i))
                    if dir_j > 0:
                        max_step = min(max_step, (1.0 - h[j]) / dir_j)
                    elif dir_j < 0:
                        max_step = min(max_step, h[j] / (-dir_j))
                    if max_step < 1e-10:
                        continue
                    for frac in [0.5, 0.25, 0.1, 0.05]:
                        step = max_step * frac
                        h[i] += dir_i * step
                        h[j] += dir_j * step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            break
                        h[i] -= dir_i * step
                        h[j] -= dir_j * step
                    if improved:
                        break
            if improved:
                break
        if not improved:
            alpha = min(alpha * 1.5, 500000)
    return h, best_c

def fixed_delta_pair(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True; improved_any = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True; improved_any = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
        if not improved_any and d < 0.01:
            break
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturb_h(h, n, rng, sigma=0.03):
    h_p = h.copy()
    mask = (h_p > 0.02) & (h_p < 0.98)
    if mask.sum() > 0:
        h_p[mask] += rng.normal(0, sigma, mask.sum())
    return project_feasible(h_p, n)

def slsqp_polish(h, n, deadline, alpha=500000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: float(x.sum() - n / 2.0),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_pen, h, args=(n, alpha, 0.0), method='SLSQP',
                      jac=obj_pen_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 800, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B warmup + cooldown at n_init
    t1 = min(t0 + 250, deadline - 500)
    alphas_warm = [30, 100, 400, 1500]
    best_h, best_c = lbfgs_phase(best_h, n_init, min(t0 + 120, t1), alphas_warm, max_iters=100)
    alphas_cool = [5000, 20000, 100000, 500000, 2000000, 10000000]
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, alphas_cool, max_iters=120)
    best_n = n_init

    # Phase 2: Top-k L-BFGS refinement at n_init
    if time.time() < deadline - 400:
        t2 = min(time.time() + 120, deadline - 400)
        h2, c2 = lbfgs_phase(best_h, n_init, t2, [50000, 200000, 1000000],
                           max_iters=100, topk_mode=True, k_top=6)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    if time.time() < deadline - 300:
        t2b = min(time.time() + 80, deadline - 300)
        h2b, c2b = subgradient_phase(best_h, n_init, t2b, [5000, 50000, 500000], k_top=8)
        if c2b < best_c:
            best_c = c2b
            best_h = h2b

    # Phase 3: Gradient-guided pair search at n_init
    if time.time() < deadline - 250:
        t3 = min(time.time() + 120, deadline - 250)
        best_h, best_c = gradient_pair_search(best_h, n_init, t3, alpha=50000, max_rounds=300)

    # Phase 4: Interpolate to higher resolution and optimize
    if time.time() < deadline - 200:
        for n_try in [450, 600]:
            if time.time() > deadline - 100:
                break
            h_interp = interpolate_h(best_h, best_n, n_try)
            alloc = min(120, deadline - time.time() - 60)
            if alloc < 40:
                break
            t4 = time.time() + alloc * 0.4
            h_t, c_t = lbfgs_phase(h_interp, n_try, t4, [200, 2000, 20000, 200000],
                                  max_iters=80)
            if time.time() < deadline - 60:
                t4b = min(time.time() + 50, deadline - 60)
                h_t, c_t = lbfgs_phase(h_t, n_try, t4b, [1000000, 5000000],
                                      max_iters=60, topk_mode=True, k_top=6)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try

    # Phase 5: Fixed-delta pair search on best solution
    if time.time() < deadline - 100:
        t5 = min(time.time() + 120, deadline - 100)
        best_h, best_c = fixed_delta_pair(best_h, best_n, t5,
                                          [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                          radius=60)

    # Phase 6: Perturb + restart optimization
    if time.time() < deadline - 80:
        for perturb_sigma in [0.02, 0.04, 0.01]:
            if time.time() > deadline - 40:
                break
            h_pert = perturb_h(best_h, best_n, rng, sigma=perturb_sigma)
            alloc = min(50, deadline - time.time() - 30)
            if alloc < 20:
                break
            h_r, c_r = lbfgs_phase(h_pert, best_n, time.time() + alloc,
                                  [10000, 100000, 1000000], max_iters=80)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 15:
        h_s, c_s = slsqp_polish(best_h, best_n, deadline - 10, alpha=1000000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 8: Final pair search
    if time.time() < deadline - 5:
        best_h, best_c = fixed_delta_pair(best_h, best_n, deadline - 3,
                                          [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                          radius=40)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
