```python
            continue

        if k >= 0:

                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])

            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    grad *= dx
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
```

```python
def compute_active_gradient(h, n, active_indices):
    """Compute sum of gradients of c_k for k in active_indices."""
    dx = 2.0 / n
    grad = np.zeros(n)
    n_sub = np.zeros(n)  # count of active terms for off-diagonal
    for idx in active_indices:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += (1.0 - h[k:]) 
                grad[k:] += (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += (1.0 - h[:nv])
                grad[:nv] += (-h[kk:])
    grad *= dx
    return grad
```
