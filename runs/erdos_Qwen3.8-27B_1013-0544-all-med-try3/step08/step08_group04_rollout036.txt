def fft_correlate(a, v):
    n = len(a)
    m = len(v)
    N = n + m - 1
    # Next power of 2 for FFT
    N2 = 1 << (N - 1).bit_length()
    fa = np.fft.rfft(a, N2)
    fv = np.fft.rfft(v, N2)
    # Cross-correlation via FFT
    result = np.fft.irfft(fa * np.conj(fv), N2)[:N]
    return result

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_local_search(h, n, deadline, n_iter=5000):
    """Projected subgradient method for the max objective."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.02
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_all_c(h, n)
        m = c.max()
        if m < best_c - 1e-14:
            best_c = m
            best_h = h.copy()
        # Compute subgradient: use softmax weighting over near-max c[k]
        weights = np.exp((c - m) * 50.0)
        weights /= weights.sum()
        # Gradient of c[k] w.r.t. h[m]:
        # c[k] = sum_i h[i]*(1-h[i+k-(n-1)])*dx
        # d c[k] / d h[p] = dx * [(1-h[p+k-(n-1)]) * I(0<=p+k-(n-1)<=n-1) - h[p-k+(n-1)] * I(0<=p-k+(n-1)<=n-1)]
        grad = np.zeros(n)
        dx = 2.0 / n
        for k in range(2 * n - 1):
            if weights[k] < 1e-10:
                continue
            w = weights[k] * dx
            offset = k - (n - 1)
            # First term: (1 - h[p + offset]) for valid p + offset
            lo1 = max(0, -offset)
            hi1 = min(n, n - offset)
            if lo1 < hi1:
                grad[lo1:hi1] += w * (1.0 - h[lo1 + offset:hi1 + offset])
            # Second term: -h[p - offset] for valid p - offset
            lo2 = max(0, offset)
            hi2 = min(n, n + offset)
            if lo2 < hi2:
                grad[lo2:hi2] -= w * h[lo2 - offset:hi2 - offset]
        # Projected gradient step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            step = min(step * 1.1, 0.05)
        else:
            step *= 0.8
            if step < 1e-6:
                step = 0.01
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def random_pair_search(h, n, deadline, n_steps=500000, delta_range=(0.001, 0.05)):
    """Random pair moves with adaptive step size."""
    rng = np.random.RandomState(12345)
    best_c = compute_c5(h, n)
    delta = 0.01
    for step_i in range(n_steps):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if rng.rand() < 0.5:
            di, dj = delta, -delta
        else:
            di, dj = -delta, delta
        if h[i] + di < -1e-12 or h[i] + di > 1.0 + 1e-12:
            continue
        if h[j] + dj < -1e-12 or h[j] + dj > 1.0 + 1e-12:
            continue
        h[i] += di; h[j] += dj
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
            delta = min(delta * 1.05, 0.05)
        else:
            h[i] -= di; h[j] -= dj
            h = project_feasible(h, n)
            delta = max(delta * 0.95, 0.0005)
    return h, best_c

def triple_search(h, n, deadline, delta=0.005):
    """Try 3-point moves: add d to h[i], subtract d/2 from h[j] and h[k]."""
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(999)
    n_triple = 100000
    for _ in range(n_triple):
        if time.time() > deadline:
            break
        idx = rng.choice(n, size=3, replace=False)
        i, j, k = idx
        di = delta
        dj = -delta / 2.0
        dk = -delta / 2.0
        if h[i] + di > 1.0 or h[j] + dj < 0.0 or h[k] + dk < 0.0:
            continue
        h[i] += di; h[j] += dj; h[k] += dk
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
        else:
            h[i] -= di; h[j] -= dj; h[k] -= dk
            h = project_feasible(h, n)
    return h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    # SLSQP with increasing alpha
    alphas = [2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_opt(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    # Gradient-based local search
    if time.time() < deadline - 30:
        h, best_c = gradient_local_search(h, n, time.time() + min(80, deadline - time.time() - 20))
        h = project_feasible(h, n)
        best_c = compute_c5(h, n)

    # Pair search with decreasing steps
    if time.time() < deadline - 10:
        if n <= 300:
            h, best_c = pair_search(h, n, deadline - 5,
                                    [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            # For larger n, use random pair search
            h, best_c = random_pair_search(h, n, deadline - 5, n_steps=200000)

    # Final SLSQP polish
    if time.time() < deadline - 3:
        ht = slsqp_opt(h, n, 5000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    # Final pair search
    if time.time() < deadline - 1 and n <= 300:
        h, best_c = pair_search(h, n, deadline - 0.5,
                                [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
