The most effective single change: find (i, j) with j = i+k such that h[j] is large and h[i] is small. Transfer from j to i: h[j] -= d, h[i] += d. This reduces c[k] by approximately 2d*(h[j]+d)*(1-h[i]) - ... it's complex.

Actually, the simplest directed search:
- For each worst shift k, for each valid pair (i, i+k):
  - Try h[i] += d, h[i+k] -= d (if feasible)
  - Try h[i] -= d, h[i+k] += d (if feasible)
- This is O(n) per shift, and we only do it for the worst few shifts.

This is much faster than the O(n²) pair search in the previous code!

            # Find worst shifts

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_shift = shifted.max()
    exps = np.exp(shifted - max_shift)
    logsum = max_shift + np.log(exps.sum())
    val = m + logsum / alpha
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    sig_idx = np.where(weights > 1e-10)[0]
    
    for k_idx in sig_idx:
        w = weights[k_idx]
        s = k_idx - (n - 1)
        if s >= 0:
            grad[s:] += w * (1 - x[:n-s])  # (1-x[j-s]) for j in [s, n-1] → x[j-s] for j-s in [0, n-1-s]
            grad[:n-s] -= w * x[s:]  # -x[j+s] for j in [0, n-1-s] → x[j+s] for j+s in [s, n-1]
        else:
            s2 = -s
            grad[:n-s2] += w * (1 - x[s2:])  # (1-x[j+s2]) for j in [0, n-1-s2] → x[j+s2] for j+s2 in [s2, n-1]
            grad[s2:] -= w * x[:n-s2]  # -x[j-s2] for j in [s2, n-1] → x[j-s2] for j-s2 in [0, n-1-s2]
    
    return val, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    ms = shifted.max()
    return float(m + (ms + np.log(np.sum(np.exp(shifted - ms)))) / alpha)

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    ms = shifted.max()
    exps = np.exp(shifted - ms)
    logsum = ms + np.log(exps.sum())
    val = m + logsum / alpha
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    sig_idx = np.where(weights > 1e-12)[0]
    for k_idx in sig_idx:
        w = weights[k_idx]
        s = int(k_idx - (n - 1))
        if s >= 0:
            grad[s:] += w * (1 - x[:n-s])
            grad[:n-s] -= w * x[s:]
        else:
            s2 = -s
            grad[:n-s2] += w * (1 - x[s2:])
            grad[s2:] -= w * x[:n-s2]
    
    return val, grad

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), jac=lambda x, n2, a2: smooth_obj_grad(x, n2, a2)[1],
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            c_all = compute_all_overlaps(h, n)
            worst_idx = np.argsort(c_all)[-num_worst:]
            for widx in worst_idx:
                if time.time() > deadline:
                    return h, best_c
                s = int(widx - (n - 1))
                if s >= 0:
                    for i in range(n - s):
                        j = i + s
                        if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                            h[i] += d; h[j] -= d
                            c = compute_c5(h, n)
                            if c
