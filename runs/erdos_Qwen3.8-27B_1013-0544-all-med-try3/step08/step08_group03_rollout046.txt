def lbfgs_optimize(h, n, deadline, alpha_schedule=None, max_iter_per_alpha=150):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alpha_schedule is None:
        alpha_schedule = [100, 500, 2000, 20000, 200000, 1000000, 5000000]
    for alpha in alpha_schedule:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time() - 3
        if remaining < 15:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, 0.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter_per_alpha, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iter_per_alpha * 30,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
            else:
                h = project_feasible(h, n)
        except Exception:
            h = project_feasible(h, n)
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=1500):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def direct_max_refine(h, n, deadline, n_iters=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        max_val = c.max()
        if max_val <= best_c:
            best_c = max_val
        # Find top-3 peaks for more robust subgradient
        top3 = np.argsort(c)[-3:]
        grad = np.zeros(n)
        for lag in top3:
            k = lag - (n - 1)
            if k >= 0:
                nv = n - k
                grad[k:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[k:n]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[0:nv] += (1.0 - h[kk:n]) * dx
                grad[kk:n] += (-h[0:nv]) * dx
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-16:
            break
        grad /= g_norm
        lr = 0.005
        improved = False
        for _ in range(25):
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            # Try smaller steps
            for trial_lr in [0.001, 0.0005, 0.0001]:
                h_new = project_feasible(h - trial_lr * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    h = h_new
                    break
    return h, best_c

def pair_refine(h, n, deadline, deltas=None, radius=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    if radius is None:
        radius = max(30, n // 8)
    for d in deltas:
        improved_any = False
        passes = 0
        while time.time() < deadline and improved_any and passes < 10:
            improved_any = False
            passes += 1
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
        if not improved_any:
            break
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def multi_alpha_lbfgs(h, n, deadline):
    """Aggressive multi-alpha L-BFGS with progress tracking."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    # Fine-grained alpha schedule
    alphas = [50, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000, 3000000]
    prev_best = best_c
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            maxit = min(200, max(50, int((deadline - time.time() - 10) / max(1, len(alphas) - 1) * 2)))
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, 0.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-13, 'maxfun': maxit * 30, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
                prev_best = c_new
            else:
                h = project_feasible(h, n)
        except Exception:
            h = project_feasible(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Optimize at original resolution (quick)
    t1 = min(t0 + 100, deadline - 600)
    h1, c1 = lbfgs_optimize(best_h, n_init, t1,
                             alpha_schedule=[200, 1000, 5000, 50000, 500000],
                             max_iter_per_alpha=100)
    if c1 < best_c:
        best_c = c1
        best_h = h1
        best_n = n_init

    # Phase 2: Move to n=600 and optimize heavily
    n600 = 600
    h600 = interpolate_h(best_h, best_n, n600)
    h600 = project_feasible(h600, n600)
    t2 = min(t0 + 350, deadline - 350)
    h600, c600 = multi_alpha_lbfgs(h600, n600, t2)
    if c600 < best_c:
        best_c = c600
        best_h = h600
        best_n = n600

    # Phase 3: SLSQP at n=600
    if time.time() < deadline - 250:
        t3 = min(time.time() + 120, deadline - 250)
        for a in [200000, 1000000]:
            if time.time() > t3:
                break
            hs, cs = slsqp_optimize(best_h, best_n, min(time.time() + 60, t3), alpha=a, maxiter=1000)
            if cs < best_c:
                best_c = cs
                best_h = hs

    # Phase 4: Try n=900
    n900 = 900
    if time.time() < deadline - 200:
        h900 = interpolate_h(best_h, best_n, n900)
        h900 = project_feasible(h900, n900)
        t4 = min(time.time() + 180, deadline - 150)
        h900, c900 = multi_alpha_lbfgs(h900, n900, t4)
        if c900 < best_c:
            best_c = c900
            best_h = h900
            best_n = n900

    # Phase 5: Second SLSQP pass at best resolution
    if time.time() < deadline - 100:
        nf = len(best_h)
        t5 = min(time.time() + 60, deadline - 80)
        hs, cs = slsqp_optimize(best_h, nf, t5, alpha=500000, maxiter=1500)
        if cs < best_c:
            best_c = cs
            best_h = hs

    # Phase 6: Direct max gradient refinement
    if time.time() < deadline - 50:
        nf = len(best_h)
        h6, c6 = direct_max_refine(best_h, nf, deadline - 20, n_iters=1500)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = nf

    # Phase 7: Pair refinement for final polish
    if time.time() < deadline - 5:
        nf = len(best_h)
        best_h, best_c = pair_refine(best_h, nf, deadline - 2,
                                      deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                      radius=max(20, nf // 10))

    # Final check
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
