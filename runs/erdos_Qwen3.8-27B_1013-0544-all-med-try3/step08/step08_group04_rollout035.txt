import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):

    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

This is a cutting plane / Kelley cut approach. Each iteration adds a linear cut for the max. It should converge to a local minimum.

import cvxpy as cp

def lp_refine(h, n, max_iter=100, tol=1e-6):
    """Sequential linearization / cutting plane method."""
    dx = 2.0 / n
    h_curr = h.copy()
    
    for iteration in range(max_iter):
        c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        M = float(c.max())
        active = np.where(c > M - 1e-4)[0]
        
        # Compute gradients for active lags
        grads = []
        for m in active:
            k = int(m) - (n - 1)
            grad = np.zeros(n)
            if k >= 0:
                for p in range(n):
                    if p < n - k:  # valid as first factor
                        grad[p] += (1 - h_curr[p + k])
                    if p >= k and p < n:  # valid as second factor (p = i+k, i = p-k)
                        grad[p] -= h_curr[p - k]
            else:
                kp = -k  # positive
                for p in range(n):
                    if p >= kp:  # valid as first factor
                        grad[p] += (1 - h_curr[p - kp])
                    if p < n - kp:  # valid as second factor
                        grad[p] -= h_curr[p + kp]
            grad *= dx
            grads.append(grad)
        
        # Set up LP
        h_var = cp.Variable(n, bounds=[0, 1])
        t_var = cp.Variable()
        
        constraints = [h_var.sum() == n / 2.0]
        for idx, m in enumerate(active):
            k = int(m) - (n - 1)
            c_val = c[m]
            g = grads[idx]
            # Linearization: c[k] ≈ c_val + g · (h_new - h_curr)
            # Constraint: t >= c_val + g · (h_new - h_curr)
            constraints.append(t_var >= c_val + g @ (h_var - h_curr))
        
        problem = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            problem.solve(solver=cp.SCS, verbose=False)
            if h_var.value is None:
                break
            h_new = np.clip(h_var.value, 0, 1)
            h_new = project_feasible(h_new, n)
        except Exception:
            break

def compute_grad_c(h, n, lag):
    """Compute ∂c[lag]/∂h for a specific lag."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        for p in range(n - lag):  # p is first factor
            grad[p] += (1.0 - h[p + lag])
        for p in range(lag, n):  # p is second factor (i = p - lag)
            grad[p] -= h[p - lag]
    else:
        m = -lag
        for p in range(m, n):  # p is first factor (i = p, need 0 <= p-m < n)
            # Actually, for lag < 0: i ranges from -lag to n-1
            # h[i] * (1 - h[i+lag]) = h[i] * (1 - h[i-m])
            # Valid: 0 <= i < n and 0 <= i-m < n → m <= i < n and i-m < n → i < n+m
            # Combined: m <= i < n (since m >= 0 and n <= n+m)
            grad[p] += (1.0 - h[p - m])

            # And 0 <= p+m < n (i valid) → 0 <= p < n-m

    return grad * dx

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)

            if tot < 1e-15:
