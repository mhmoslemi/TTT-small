                    found2 = False
                    for li, i in enumerate(I):
                        if best_h[i] - d < 0.0:
                            continue
                        valid_j = best_h <= (1.0 - d)
                        valid_j[i] = False
                        if not valid_j.any():
                            continue
                        worst_valid = worst[li].copy()
                        worst_valid[~valid_j] = np.inf
                        j_best = np.argmin(worst_valid)
                        new_max = worst[li, j_best]
                        if new_max < best_c - 1e-14:
                            best_h[i] -= d
                            best_h[j_best] += d
                            best_h = project_feasible(best_h, n)
                            best_c = compute_c5(best_h, n)
                            improved = True
                            found2 = True
                            break
                    if found2:
                        break
    return best_h, best_c

def simple_pair_search(h, n, deadline, radius=50):
    """Simpler pair search with local radius."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0004, 0.00015, 0.00006, 0.00002, 0.000008]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i]+=d, h[j]-=d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = project_feasible(best_h, n)
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i]-=d, h[j]+=d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = project_feasible(best_h, n)
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def nesterov_subgrad(h, n, deadline, n_iter=10000):
    """Nesterov-accelerated subgradient method."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    momentum = np.zeros(n)
    eta = 0.003
    gamma = 0.9  # momentum decay
    tol_expand = 0.002
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_curr, n)
        c_max = c_vec.max()
        tol = max(1e-9, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        # Compute average subgradient over active set
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h_curr[k:n]) * dx
                grad[k:n] += (-h_curr[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h_curr[0:nv]) * dx
                grad[0:nv] += (-h_curr[kk:n]) * dx
        grad /= len(active)
        
        # Nesterov momentum
        momentum = gamma * momentum - eta * grad
        h_new = h_curr + momentum
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
        
        h_curr = h_new
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=1000):
    c = compute_c5_vec(h, n)
    
    def obj(x):
        cc = compute_c5_vec(x, n)
        m = cc.max()
        sh = alpha * (cc - m)
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha
    
    def jac(x):
        cc = compute_c5_vec(x, n)
        m = cc.max()
        sh = alpha * (cc - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-14:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                g[0:nv] += wm * (1.0 - x[k:n])
                g[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wm * (1.0 - x[0:nv])
                g[0:nv] += wm * (-x[kk:n])
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n = 500
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    best_n = n
    
    # Phase 1: Full softmax L-BFGS with increasing alpha
    p1_end = min(t0 + 200, deadline - 700)
    alphas_p1 = [50, 200, 1000, 5000, 20000, 100000]
    maxiters_p1 = [2000, 2000, 3000, 3000, 3000, 4000]
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas_p1):
        if time.time() > p1_end - 10:
            break
        maxit = maxiters_p1[i] if i < len(maxiters_p1) else 2000
        try:
            res = minimize(full_softmax_obj, best_h, jac=full_softmax_grad,
                          args=(n, alpha, 0.01), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 8,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    # Phase 2: Active-set focused L-BFGS (top 3-6 shifts)
    p2_end = min(t0 + 450, deadline - 450)
    for top_k in [5, 4, 3]:
        if time.time() > p2_end - 30:
            break
        active, _ = get_active_set(best_h, n, top_k, tol_frac=0.0005)
        for alpha in [50000, 200000, 1000000, 5000000]:
            if time.time() > p2_end - 15:
                break
            maxit = max(1000, int((p2_end - time.time()) * 2))
            try:
                res = minimize(active_softmax_obj, best_h, jac=active_softmax_grad,
                              args=(n, active, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 3: Nesterov subgradient
    p3_end = min(t0 + 600, deadline - 250)
    h3, c3 = nesterov_subgrad(best_h, n, p3_end, n_iter=8000)
    if c3 < best_c:
        best_c = c3
        best_h = h3
    
    # Phase 4: Vectorized pair search
    p4_end = min(t0 + 850, deadline - 150)
    active, _ = get_active_set(best_h, n, 5, tol_frac=0.001)
    h4, c4 = vectorized_pair_search(best_h, n, p4_end, active, max_rounds=30000)
    if c4 < best_c:
        best_c = c4
        best_h = h4
    
    # Phase 5: Simple pair search with smaller radius for final polish
    p5_end = min(t0 + 950, deadline - 80)
    h5, c5 = simple_pair_search(best_h, n, p5_end, radius=30)
    if c5 < best_c:
        best_c = c5
        best_h = h5
    
    # Phase 6: SLSQP polish
    p6_end = deadline - 20
    for alpha in [500000, 2000000]:
        if time.time() > p6_end - 10:
            break
        p6_sub = min(time.time() + 25, p6_end - 5)
        h6, c6 = slqp_polish(best_h, n, p6_sub, alpha=alpha, maxiter=800)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Final: tiny pair search
    if time.time() < deadline - 2:
        h7, c7 = simple_pair_search(best_h, n, deadline - 1, radius=15)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
