        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def topk_subgradient_descent(h, n, deadline, top_k=5, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t_param = 0
    alpha = 0.1
    patience = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        if max_val < best_c - 1e-15:
            best_c = max_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 500:
            # Restart with perturbation
            h = project_feasible(h + np.random.randn(n) * 0.005, n)
            patience = 0
            alpha = max(alpha * 0.5, 0.01)
            continue
        
        # Get top-k shifts
        idx = np.argsort(-c_vec)[:top_k]
        
        # Compute average gradient of top-k
        grad = np.zeros(n)
        for i in idx:
            grad += shift_grad_single(h, n, int(i))
        grad /= top_k
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size
        step = alpha / (1 + t_param * 0.01)**0.75
        t_param += 1
        
        # Update
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new <= max_val + 1e-10:
            h = h_new
        else:
            # Halve step
            h_new = project_feasible(h - (step/2) * grad, n)
            if compute_c5(h_new, n) <= max_val + 1e-10:
                h = h_new
            else:
                alpha *= 0.8
    return best_h, best_c

For the active set (shifts at the maximum), the tangent linearization gives:
c_j(h + d) ≈ c_j(h) + ∇c_j(h)^T d + (1/2) d^T ∇²c_j d

The Hessian of c_j with respect to h is:
∇²c_j = -dx * (S_j + S_j^T)
where S_j is the shift matrix (S_j)[i, i+j] = 1.

So the quadratic term is -(dx/2) * d^T (S_j + S_j^T) d = -dx * ∑_i d[i]*d[i+j].

For the LP subproblem with second-order correction:
minimize t
subject to: c_j(h) + ∇c_j^T d - dx * ∑_i d[i]*d[i+j] ≤ t for j in active
            0 ≤ h[i] + d[i] ≤ 1
            ∑(h[i] + d[i]) = n/2
            ||d||_∞ ≤ ρ

The quadratic constraint makes this a QCQP, not an LP. But if ρ is small, the quadratic term is O(ρ²) while the linear term is O(ρ), so the linear term dominates. The LP is a good approximation for small ρ.

Alternatively, I can use the Hessian to compute a Newton-like step:
Solve: min (1/2) d^T H d + g^T d
subject to constraints
where H is the Hessian of the smooth approximation and g is the gradient.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def smooth_obj_constrained(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_constrained(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_subgradient(h, n, deadline, top_k=5, max_iter=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 0.08
    t_param = 0
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        curr_max = c_vec.max()
        
        if curr_max < best_c - 1e-14:
            best_c = curr_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            h = project_feasible(h + np.random.randn(n) * 0.003, n)

            alpha = 0.05
            continue
        
        idx = np.argsort(-c_vec)[:top_k]
        grad = np.zeros(n)

        grad /= float(top_k)

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    grads = []
    c_vals = []
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        grads.append(g)
        c_vals.append(c_vec[idx])
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        A_ub[row, 0:n] = grads[j]
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i] - rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i] + rho
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.04):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 15:
            top_order = np.argsort(-c_vec[active])
            active = [int(active[i]) for i in top_order[:15]]
        
        delta, _ = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            if rho < 0.06:
                rho = min(rho * 1.2, 0.08)
        else:
            stall += 1
            rho *= 0.6
            if rho < 1e-10:
                break
            if stall > 30:
                break
    return best_h, best_c

def pairwise_local_search(h, n, deadline, n_passes=5):

    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

            break
        improved_any = True
        while improved_any and time.time() < deadline:

            max_val = c_vec.max()
            k_top = int(np.argmax(c_vec))
            k_shift = k_top - (n - 1)
            
            # Find worst pairs for this shift
            if k_shift >= 0:
                pairs = [(i, i + k_shift) for i in range(n - k_shift)]
            else:
                pairs = [(i, i - k_shift) for i in range(-k_shift, n)]
            
            # Sort pairs by contribution
            if k_shift >= 0:
                contribs = [(i, i + k_shift, h[i] * (1 - h[i + k_shift])) for i in range(n - k_shift)]
            else:
                contribs = [(i, i - k_shift, h[i] * (1 - h[i - k_shift])) for i in range(-k_shift, n)]
            contribs.sort(key=lambda x: -x[2])
            
            checked = 0
            for i, j, _ in contribs[:200]:
                if checked >= 5000:
                    break
                checked += 1
                if h[i] >= step and h[j] <= 1.0 - step:

                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:

def slqp_polish(h, n, deadline, alpha=3000000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    def obj_fn(x):
        return smooth_obj(x, n, alpha)
    def grad_fn(x):
        return smooth_grad(x, n, alpha)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

            h_h, c_h = sqp_optimize(h_high, n_high, t_sq, max_iter=150, rho_init=0.03)
            if c_h < c_high:
                c_high = c_h
                h_high = h_h
        
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_orig = n_high
    
    # Final check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
