def project_feasible(y, n):
    target = n / 2.0
    # Find lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0, 1)

def smooth_max_obj_and_grad(h, n, alpha, mu=0.0):
    """Compute smooth max objective and its gradient."""
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    # For numerical stability
    smx = sh.max()
    w = np.exp(sh - smx)
    w_sum = w.sum()
    w /= w_sum
    val = m + (smx + np.log(w_sum)) / alpha
    if mu > 0:
        val += mu * (h.sum() - n/2.0)**2
    
    # Gradient
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only iterate over shifts with significant weight
    threshold = 1e-10 * w.max()

        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:

            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])

        grad += 2.0 * mu * (h.sum() - n/2.0)

def project_feasible(y, n):
    """Project y onto {x : 0 <= x <= 1, sum(x) = n/2}."""
    target = n / 2.0
    if y.sum() == target and y.min() >= 0 and y.max() <= 1:
        return y.copy()

The objective is max_k O(k). For a fixed active set, we're minimizing a function of h. We could:
1. Fix all but one coordinate h[i]
2. Find the optimal h[i] given the others (this is a 1D optimization)
3. Cycle through all coordinates

For a fixed i, the objective as a function of h[i] is:
f(t) = max_k O(k|_{h[i]=t})

where O(k|_{h[i]=t}) = O(k) + (t - h[i]) * ∂O(k)/∂h[i] (first order approximation)

Or more precisely, for each k:
O(k) depends on h[i] through:
- The term h[i](1-h[i+k]) (coefficient of h[i] is 1-h[i+k])
- The term h[i-k](1-h[i]) (coefficient of h[i] is -h[i-k])

So O(k) = A_k + h[i] * B_k where:
B_k = dx * [sum over j where i participates: (1-h[j+k]) - h[j-k]]

And A_k is O(k) with h[i] removed.

So f(t) = max_k (A_k + t * B_k), which is the max of linear functions of t, hence convex in t. We can minimize it by finding where the subgradient contains 0, or by simple 1D search on [0, 1].

This is actually a great approach! It's exact (no smooth approximation needed), and each 1D subproblem is easy to solve.
