    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_improve = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            new_t, new_c = coord_opt_val(h, n, i)
            if abs(new_t - h[i]) > 1e-10:
                h[i] = new_t
            if new_c < best_c - 1e-12:
                best_c = new_c
                any_improve = True
        # Verify and update
        actual_c = compute_c5(h, n)
        if actual_c < best_c:
            best_c = actual_c
        if not any_improve:
            break
    
    return h, best_c

def pair_search_fast(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(50):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(30):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        ht = project_feasible(res.x, n)
        c = compute_c5(ht, n)
        return ht, c
    except Exception:
        return h.copy(), compute_c5(h, n)

def hill_climb(h, n, deadline, rng, n_trials=2000):
    """Try random 3-5 coordinate perturbations."""
    best_c = compute_c5(h, n)
    h = h.copy()
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        n_perturb = rng.randint(2, 6)
        idxs = rng.choice(n, size=n_perturb, replace=False)
        old_vals = h[idxs].copy()
        h[idxs] += rng.normal(0, 0.02, n_perturb)
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
        else:
            h[idxs] = old_vals
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Coordinate descent on initial resolution
    alloc1 = min(250, (deadline - t0) * 0.30)
    best_h, best_c = coordinate_descent(best_h, n_init, t0 + alloc1, max_sweeps=150)
    
    # Phase 2: SLSQP refinement
    if time.time() < deadline - 100:
        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - 80:
                break
            ht, ct = slsqp_step(best_h, n_init, alpha, maxiter=1500)
            if ct < best_c - 1e-14:
                best_c = ct
                best_h = ht
    
    # Phase 3: Pair search on initial resolution
    if n_init <= 200 and time.time() < deadline - 80:
        best_h, best_c = pair_search_fast(best_h, n_init, time.time() + min(80, (deadline - time.time() - 20) * 0.5),
                                          [0.01, 0.005, 0.002, 0.001, 0.0005])
    elif time.time() < deadline - 80:
        best_h, best_c = near_pair_search(best_h, n_init, time.time() + min(80, (deadline - time.time() - 20) * 0.5),
                                          [0.01, 0.005, 0.002], radius=35)
    
    # Phase 4: Try different resolutions
    x_init = np.linspace(0, 2, n_init)
    resolutions = [80, 100, 120, 150, 180, 200, 250, 300]
    resolutions = [r for r in resolutions if r != n_init]
    
    for n_try in resolutions:
        if time.time() > deadline - 50:
            break
        remaining = deadline - time.time() - 30
        alloc = min(remaining * 0.22, 100)
        if alloc < 25:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        h_opt, c_opt = coordinate_descent(h_interp, n_try, time.time() + alloc * 0.6, max_sweeps=80)
        
        if time.time() < deadline - 25 and n_try <= 200:
            h_opt, c_opt = pair_search_fast(h_opt, n_try, time.time() + min(30, (deadline - time.time() - 15) * 0.3),
                                             [0.01, 0.005, 0.002, 0.001])
        elif time.time() < deadline - 25:
            h_opt, c_opt = near_pair_search(h_opt, n_try, time.time() + min(30, (deadline - time.time() - 15) * 0.3),
                                             [0.01, 0.005, 0.002], radius=30)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            x_init = x_new
    
    # Phase 5: Hill climbing for escape from local optima
    if time.time() < deadline - 30:
        n = len(best_h)
        best_h, best_c = hill_climb(best_h, n, time.time() + min(30, (deadline - time.time() - 15) * 0.4), rng, n_trials=1500)
        
        # Coordinate descent after hill climb
        if time.time() < deadline - 15:
            best_h, best_c = coordinate_descent(best_h, n, time.time() + min(40, (deadline - time.time() - 10) * 0.5), max_sweeps=50)
    
    # Phase 6: Final polishing
    if time.time() < deadline - 5:
        n = len(best_h)
        if n <= 150:
            best_h, best_c = pair_search_fast(best_h, n, deadline - 2, [0.002, 0.001, 0.0005, 0.0002])
        elif n <= 250:
            best_h, best_c = pair_search_fast(best_h, n, deadline - 2, [0.002, 0.001, 0.0005])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 2, [0.002, 0.001], radius=25)
        
        # Final coordinate descent
        if time.time() < deadline - 2:
            best_h, best_c = coordinate_descent(best_h, n, deadline - 1, max_sweeps=30)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
