import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            frac = diff / tot
            if frac < 1.0:
                h += frac * room
            else:
                h = np.minimum(h + diff / n, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (-diff) / tot * h
    return np.clip(h, 0.0, 1.0)

def make_legendre_start(n, p_seed=251):
    h = np.zeros(n)
    p = 2 * p_seed + 1
    while p <= n or p % 2 == 0:
        p += 2
    residues = set()
    for k in range(1, p):
        residues.add((k * k) % p)
    residues.discard(0)
    idx = 0
    for i in range(n):
        if i % p in residues:
            h[idx] = 1.0
            idx += 1
    target = n / 2.0
    h = np.zeros(n)
    count = 0
    for i in range(n):
        if count < len(residues) and (i % p) in residues:
            h[i] = 1.0
            count += 1
        if count >= len(residues):
            break
    while h.sum() < target:
        for i in range(n - 1, -1, -1):
            if h[i] < 1.0:
                h[i] = min(1.0, h[i] + 1.0)
                break
    while h.sum() > target:
        for i in range(n):
            if h[i] > 0.0:
                h[i] = max(0.0, h[i] - 1.0)
                break
    return project_feasible(h, n)

def make_greedy_start(n, rng):
    h = np.zeros(n)
    target = n / 2
    ones = set()
    candidates = list(range(n))
    rng.shuffle(candidates)
    for idx in candidates:
        if len(ones) >= target:
            break
        h[idx] = 1.0
        ones.add(idx)
    return project_feasible(h, n)

def make_tm_start(n):
    h = np.zeros(n)
    half = n // 2
    seq = [0]
    for _ in range(n - 1):
        L = len(seq)
        for v in seq:
            seq.append(1 - v)
        if len(seq) >= n:
            break
    h[:n] = seq[:n]
    s = h.sum()
    target = n / 2.0
    if s < target:
        idx = 0
        while h.sum() < target:
            if h[idx] < 1.0:
                h[idx] = 1.0
            idx += 1
            if idx >= n:
                idx = 0
    elif s > target:
        idx = n - 1
        while h.sum() > target:
            if h[idx] > 0.0:
                h[idx] = 0.0
            idx -= 1
            if idx < 0:
                idx = n - 1
    return project_feasible(h, n)

def make_golden_start(n):
    h = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2
    target = int(n / 2)
    positions = set()
    for k in range(target):
        pos = int((k + 1) * phi) % n
        if pos in positions:
            pos = (pos + 1) % n
            while pos in positions:
                pos = (pos + 1) % n
        positions.add(pos)
        h[pos] = 1.0
    return project_feasible(h, n)

def comb_local_search(h, n, deadline, max_sweeps=100):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones_list = np.where(best_h > 0.5)[0].tolist()
    zeros_list = np.where(best_h < 0.5)[0].tolist()
    
    sweep = 0
    while time.time() < deadline and sweep < max_sweeps:
        improved = False
        for _oi in range(len(ones_list)):
            if time.time() > deadline:
                return best_h, best_c
            i = ones_list[_oi]
            for _zi in range(len(zeros_list)):
                j = zeros_list[_zi]
                if abs(i - j) < 2:
                    continue
                best_h[i] = 0.0
                best_h[j] = 1.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
        sweep += 1
        if not improved:
            if sweep > 5:
                break
        if improved:
            sweep = 0
    return best_h, best_c

def soft_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_soft_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_soft_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n) / n

def run_gradient(h, n, deadline, alphas=None, mu=0.005):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = max(500, int(remaining * 8 / max(1, 7 - i)))
        try:
            res = minimize(soft_max_obj, best_h, jac=soft_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 60})
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-14:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active(h, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 50000, 500000, 5000000]
    
    for round_idx in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])[:top_k]
            active = active[order]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(300, int((deadline - time.time()) * 4))
            try:
                res = minimize(active_soft_obj, best_h, jac=active_soft_grad,
                              args=(n, active, alpha, 0.002), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(np.clip(res.x, 0, 1), n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000):
    t0 = time.time()
    maxit = max(200, int((deadline - t0) * 3))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(soft_max_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        h_new = project_feasible(np.clip(res.x, 0, 1), n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def pair_refine(h, n, deadline, radius=60, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] > d and h[j] < 1.0 - d:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def binary_to_continuous(h, n, frac=0.15):
    hc = h.copy()
    one_mask = hc > 0.5
    zero_mask = hc < 0.5
    hc[one_mask] = 1.0 - frac * rng_uniform(one_mask.sum(), seed=42)
    hc[zero_mask] = frac * rng_uniform(zero_mask.sum(), seed=43)
    return project_feasible(hc, n)

def rng_uniform(size, seed=0):
    r = np.random.RandomState(seed + size)
    return r.uniform(0.4, 0.9, size)

def upsample(h, n_from, n_to):
    x_from = np.linspace(0, 2, n_from)
    x_to = np.linspace(0, 2, n_to)
    h_to = np.interp(x_to, x_from, h)
    return project_feasible(h_to, n_to)
