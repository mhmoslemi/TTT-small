from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(h, n, alpha, lam):
    """Objective (smooth max C5 + penalty) and gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    log_sum = max_sh + np.log(exp_sh.sum())
    val = m + log_sum / alpha
    
    # Penalty term
    s_diff = h.sum() - n / 2.0
    val += lam * s_diff * s_diff
    
    # Gradient of smooth max
    p = exp_sh / exp_sh.sum()  # softmax weights
    
    grad = np.zeros(n)
    for k in range(len(c)):
        if p[k] < 1e-12:
            continue
        tau = k - (n - 1)
        i_lo = max(0, -tau)
        i_hi = min(n - 1, n - 1 - tau)
        if i_lo > i_hi:
            continue
        # Term 1: (1-h[j+tau]) for j in [i_lo, i_hi]
        j_lo1 = i_lo
        j_hi1 = i_hi
        js_lo = j_lo1 + tau
        js_hi = j_hi1 + tau
        if js_lo >= 0 and js_hi < n and js_lo <= js_hi:
            grad[j_lo1:j_hi1+1] += p[k] * dx * (1.0 - h[js_lo:js_hi+1])
        # Term 2: -h[j-tau] for j-tau in [i_lo, i_hi], i.e., j in [i_lo+tau, i_hi+tau]
        j_lo2 = i_lo + tau
        j_hi2 = i_hi + tau
        if j_lo2 <= j_hi2:
            j_lo2_c = max(0, j_lo2)
            j_hi2_c = min(n-1, j_hi2)
            if j_lo2_c <= j_hi2_c:
                js_lo2 = j_lo2_c - tau
                js_hi2 = j_hi2_c - tau
                if js_lo2 >= 0 and js_hi2 < n and js_lo2 <= js_hi2:
                    grad[j_lo2_c:j_hi2_c+1] -= p[k] * dx * h[js_lo2:js_hi2+1]
    
    # Penalty gradient
    grad += 2 * lam * s_diff * np.ones(n)
    
    return val, grad

    try:
        res = minimize(obj_and_grad, h, args=(n, alpha, lam),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-14})

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    es = exp_sh.sum()
    val = m + (max_sh + np.log(es)) / alpha
    
    s_diff = x.sum() - n / 2.0
    val += lam * s_diff * s_diff
    
    p = exp_sh / es
    
    grad = np.zeros(n)
    for k in range(len(c)):
        pk = p[k]
        if pk < 1e-12:
            continue
        tau = k - (n - 1)
        i_lo = max(0, -tau)
        i_hi = min(n - 1, n - 1 - tau)
        if i_lo > i_hi:
            continue
        # (1-h[j+tau]) for j in [i_lo, i_hi]
        js_lo = i_lo + tau
        js_hi = i_hi + tau
        if 0 <= js_lo and js_hi < n:
            grad[i_lo:i_hi+1] += pk * dx * (1.0 - x[js_lo:js_hi+1])
        # -h[j-tau] for j in [i_lo+tau, i_hi+tau] clamped to [0,n-1]
        jlo2 = max(0, i_lo + tau)
        jhi2 = min(n - 1, i_hi + tau)
        if jlo2 <= jhi2:
            js_lo2 = jlo2 - tau
            js_hi2 = jhi2 - tau
            if 0 <= js_lo2 and js_hi2 < n:
                grad[jlo2:jhi2+1] -= pk * dx * x[js_lo2:js_hi2+1]
