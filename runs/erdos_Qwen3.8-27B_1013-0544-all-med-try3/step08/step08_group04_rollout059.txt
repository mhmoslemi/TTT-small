import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_values(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_values(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_values(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, n, alpha):
    c = compute_c5_values(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    obj = float(m + (mx + np.log(weights.sum())) / alpha)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jp = j + shifts
        jm = j - shifts
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        t1 = 0.0
        t2 = 0.0
        if vp.any():
            t1 = np.dot(weights[vp], (1.0 - x[jp[vp]]))
        if vm.any():
            t2 = np.dot(weights[vm], x[jm[vm]])
        grad[j] = dx * (t1 - t2)
    return obj, grad

def slsqp_run(h, n, alpha, maxiter=3000, deadline=None):
    h = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj(x, n, alpha)

    def obj_grad(x):
        return smooth_obj_grad(x, n, alpha)

    def sum_con(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj_grad, h, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        if deadline is not None and time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, alpha, deadline):
    h = h.copy()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - (n - 1)
    c_vals = compute_c5_values(h, n)
    best_c = compute_c5(h, n)

    for sweep in range(200):
        if time.time() > deadline:
            break
        for j in range(n):
            jp = j + shifts
            jm = j - shifts
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)

            a = np.zeros(n_corr)
            if vp.any():
                a[vp] = (1.0 - h[jp[vp]])
            if vm.any():
                a[vm] -= h[jm[vm]]
            a *= dx
            b = c_vals - a * h[j]

            def f1d(t):
                exponents = alpha * (a * t + b)
                exponents -= exponents.max()
                return float(np.log(np.sum(np.exp(exponents))) / alpha)

            t_cur = h[j]
            f_cur = f1d(t_cur)
            candidates = [0.0, 1.0, t_cur]
            for frac in [0.25, 0.5, 0.75]:
                candidates.append(t_cur + (1.0 - t_cur) * (frac - 0.5) * 0.3)
                candidates.append(t_cur + (0.0 - t_cur) * (0.5 - frac) * 0.3)
            candidates = [max(0.0, min(1.0, c)) for c in candidates]

            best_t = t_cur
            best_f = f_cur
            for t_cand in candidates:
                if abs(t_cand - t_cur) < 1e-12:
                    continue
                f_cand = f1d(t_cand)
                if f_cand < best_f - 1e-14:
                    best_f = f_cand
                    best_t = t_cand

            if best_t != t_cur:
                delta = best_t - t_cur
                h[j] = best_t
                c_vals += a * delta

        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            c_vals = compute_c5_values(h, n)
        if time.time() > deadline:
            break

    return h, best_c
