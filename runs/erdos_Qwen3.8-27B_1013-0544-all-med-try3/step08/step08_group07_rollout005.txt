    dx = 2.0 / n

            c0 = np.correlate(h, 1 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1 - h, mode='full')
            h[i] = old_val
            
            C = c0 * dx
            D = (c1 - c0) * dx
            
            x_opt = ternary_max(C, D, 0.0, 1.0, iters=35)
            h[i] = x_opt
        
        if (sweep + 1) % 10 == 0:
            h = project_sum(h, n)
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 100:
                break
    
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return val
    
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        val = m + (sm + np.log(w.sum())) / alpha - m  # just m + logsumexp/alpha
        # gradient of logsumexp(alpha*(g-m)): w * alpha * dg/dx
        # dg/dx is the gradient of the correlation
        # For simplicity, use numerical
        return val
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_sum(res.x, n)
    except Exception:
        return h

def subgrad_polish(h, n, deadline, max_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.05
    n_full = 2 * n - 1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        ks = int(np.argmax(g))
        
        # Analytical gradient of g[ks] w.r.t. h
        k = ks - n + 1  # shift in units of dx
        grad = np.zeros(n)
        if k >= 0:
            # g[ks] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx
            # d/dh[i]: (1-h[i+k])*dx for i in [0, n-1-k]
            #          -h[i-k]*dx for i in [k, n-1]
            for i in range(n - k):
                grad[i] += (1 - h[i + k]) * dx
            for i in range(k, n):
                grad[i] -= h[i - k] * dx
        else:
            kk = -k
            # g[ks] = sum_{i=kk}^{n-1} h[i]*(1-h[i-kk]) * dx
            #       = sum_{i=kk}^{n-1} h[i]*(1-h[i-kk]) * dx
            for i in range(kk, n):
                grad[i] += (1 - h[i - kk]) * dx
            for i in range(0, n - kk):
                grad[i] -= h[i + kk] * dx
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f / ng
            ht = project_sum(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.92
            if lr < 1e-12:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

def binary_swap_search(h, n, deadline, max_sweeps=500):
    target = int(n / 2)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_copy = zeros[:]
        for i in z_copy:
            if time.time() > deadline:
                return h_bin, best_c
            if i not in zeros or i in ones:
                continue
            o_copy = ones[:]
            for j in o_copy:
                if j not in ones or j in zeros:
                    continue
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = project_sum(h0.copy(), n0)
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: Single coordinate descent (fast, 30% budget)
    p1 = t0 + 0.30 * (budget_s - 5)
    h, c = single_coord_descent(best_h.copy(), n, p1, max_sweeps=20000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 2: SLSQP with smooth max (15% budget)
    p2 = t0 + 0.45 * (budget_s - 5)
    for alpha in [10000, 50000, 200000]:
        if time.time() > p2:
            break
        h = slsqp_step(best_h.copy(), n, alpha, 1000, p2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 3: Subgradient polish (10% budget)
    p3 = t0 + 0.55 * (budget_s - 5)
    h, c = subgrad_polish(best_h.copy(), n, p3, max_iter=5000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 4: More coordinate descent (20% budget)
    p4 = t0 + 0.75 * (budget_s - 5)
    h, c = single_coord_descent(best_h.copy(), n, p4, max_sweeps=20000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 5: Pair descent on best solution (10% budget)
    p5 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p5:
        h, c = pair_descent(best_h.copy(), n, p5, max_sweeps=200)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 6: Try higher resolution (5% budget)
    p6 = t0 + 0.90 * (budget_s - 5)
    if time.time() < p6 - 20 and n < 1000:
        nh = min(n * 2, 1000)
        if nh % 2 == 1:
            nh = nh + 1
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_sum(hh, nh)
        hh, c = single_coord_descent(hh, nh, p6, max_sweeps=3000)
        if c < best_c:
            best_c = c
            best_h = hh.copy()
            n = nh
    
    # Phase 7: Binary local search (5% budget)
    p7 = t0 + 0.95 * (budget_s - 5)
    if time.time() < p7:
        hb, cb = binary_swap_search(best_h.copy(), n, p7, max_sweeps=300)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Final: SLSQP + coordinate descent polish
    if time.time() < dl - 3:
        h = slsqp_step(best_h.copy(), n, 1000000, 500, dl - 3)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        if time.time() < dl - 1:
            h, c = single_coord_descent(best_h.copy(), n, dl, max_sweeps=5000)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Final projection and evaluation
    best_h = project_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    assert 0 <= best_c <= 1.0
    assert abs(best_h.sum() - n / 2.0) < 1e-8
    
    return best_h, best_c, n
