```python
def project_feasible(h, n):

    target = n / 2.0

        else:

            if not mask.any():

            h[mask] += diff / mask.sum()
```

```python
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:

            if cnt == 0:

            h[mask] += diff / cnt
        else:

            if cnt == 0:

            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def compute_grad(h, n, alpha):
    """Gradient of soft-max of overlaps"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    
    a = alpha * (c - c.max())
    p = np.exp(a)
    s = p.sum()
    if s < 1e-300:
        p = np.full(len(c), 1.0/len(c))
    else:
        p /= s
    
    grad = np.zeros(n)
    
    for i in range(2 * n - 1):
        if p[i] < 1e-18:
            continue
        k = i - (n - 1)
        w = p[i] * dx
        if k >= 0:
            # First term: j in [k, n-1], contribution (1-h[j-k])
            if k < n:
                grad[k:] += w * (1.0 - h[:n-k])
            # Second term: j in [0, n-1-k], contribution -h[j+k]
            if k < n:
                grad[:n-k] -= w * h[k:]
        else:
            t = -k
            # First term: j in [0, n-1-t], contribution (1-h[j+t])
            if t < n:
                grad[:n-t] += w * (1.0 - h[t:])
            # Second term: j in [t, n-1], contribution -h[j-t]
            if t < n:
                grad[t:] -= w * h[:n-t]
    
    return grad

def soft_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return m + log_sum / alpha

def projected_gradient(h, n, deadline, verbose=False):
    """Projected gradient descent with adaptive step size and alpha annealing"""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    t_start = time.time()
    total_budget = deadline - t_start
    
    # Alpha schedule: start low, increase over time
    alphas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000, 200000, 500000]
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        
        # Number of iterations for this alpha
        # More iterations for larger alpha (closer to the true max)
        n_iter = 3000
        lr = 0.5  # initial learning rate
        ftol = 1e-15
        
        for it in range(n_iter):
            if time.time() > deadline - 5:
                break
            
            c_val = soft_max_obj(h, n, alpha)
            grad = compute_grad(h, n, alpha)
            
            # Gradient step
            step = lr * grad
            h_new = h - step
            h_new = project_feasible(h_new, n)
            
            c_new = soft_max_obj(h_new, n, alpha)
            
            # Armijo condition
            if c_new <= c_val - 1e-12 * np.linalg.norm(step):
                h = h_new
                # Optionally increase lr
                if it % 100 == 0:
                    lr = min(lr * 1.1, 2.0)
            else:
                # Reduce step size
                lr *= 0.5
                if lr < 1e-8:
                    break
            
            # Check actual C5
            actual_c = compute_c5(h, n)
            if actual_c < best_c - 1e-14:
                best_c = actual_c
                best_h = h.copy()
        
        # After each alpha, check and update best
        actual_c = compute_c5(h, n)
        if actual_c < best_c:
            best_c = actual_c
            best_h = h.copy()
    
    return best_h, best_c

def pair_refine(h, n, deadline, deltas=None):
```

If we change h[i] → h[i]+d and h[j] → h[j]-d, the change in c_k is:
Δc_k = dx * [contribution of h[i] change + contribution of h[j] change]

For h[i] change (+d):

For h[j] change (-d):
- j appears as n+k: n = j-k, contribution: -d*(1-h[j-k])

```python
def compute_all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def overlap_delta(h, n, i, j, d):
    """Compute change in all overlaps when h[i] += d, h[j] -= d"""
    dx = 2.0 / n
    delta = np.zeros(2 * n - 1)
    
    for shift_idx in range(2 * n - 1):
        k = shift_idx - (n - 1)
        if k >= 0:
            # c_k = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])*dx
            # Change from h[i]: 
            #   if i = n+k (n=i-k): +d*(1-h[i-k]) if 0<=i-k<=n-1-k
            #   if i = n: -d*h[i+k] if 0<=i<=n-1-k
            # Change from h[j]:
            #   if j = n+k (n=j-k): -d*(1-h[j-k]) if 0<=j-k<=n-1-k
            #   if j = n: +d*h[j+k] if 0<=j<=n-1-k
            dc = 0.0
            if 0 <= i - k <= n - 1 - k:
                dc += d * (1.0 - h[i-k])
            if 0 <= i <= n - 1 - k:
                dc -= d * h[i+k]
            if 0 <= j - k <= n - 1 - k:
                dc -= d * (1.0 - h[j-k])
            if 0 <= j <= n - 1 - k:
                dc += d * h[j+k]
            delta[shift_idx] = dx * dc
        else:
            t = -k
            # c_{-t} = sum_{n=0}^{n-1-t} h[n]*(1-h[n+t])*dx
            dc = 0.0
            if 0 <= i <= n - 1 - t:
                dc += d * (1.0 - h[i+t])
            if 0 <= i - t <= n - 1 - t:  # i-t >= 0 and i-t <= n-1-t, i.e., t <= i <= n-1
                dc -= d * h[i-t]
            if 0 <= j <= n - 1 - t:
                dc -= d * (1.0 - h[j+t])
            if 0 <= j - t <= n - 1 - t:
                dc += d * h[j-t]
            delta[shift_idx] = dx * dc
    
    return delta
```
