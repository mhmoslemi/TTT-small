from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            vals = h.copy()
            tot = vals.sum()
            if tot < 1e-15:
                break
            h -= (abs(diff) / tot) * vals
    return np.clip(h, 0.0, 1.0)

def subgradient(h, n, s):
    """Compute subgradient of f_s at h. s is the shift index (0 to 2n-2 for correlate output)."""
    dx = 2.0 / n
    # shift in index space: actual shift = s - (n-1)
    shift = s - (n - 1)
    g = np.zeros(n)
    # ∂f/∂h[j] = dx * [(1-h[j-shift]) if 0<=j-shift<n - h[j+shift] if 0<=j+shift<n]
    for j in range(n):
        val = 0.0
        js = j - shift
        if 0 <= js < n:
            val += (1.0 - h[js])
        jp = j + shift
        if 0 <= jp < n:
            val -= h[jp]
        g[j] = dx * val
    return g

def subgradient_vec(h, n, s):
    """Vectorized subgradient computation."""
    dx = 2.0 / n
    shift = s - (n - 1)
    g = np.zeros(n)
    # (1-h[j-shift]) for valid j-shift
    if shift >= 0:
        # j-shift valid: j >= shift, j < n
        g[shift:] += 1.0 - h[:-shift]
    else:
        # j-shift valid: j < n+shift, j >= 0
        # shift is negative, so j < n + shift means j < n - |shift|
        g[:n + shift] += 1.0 - h[-shift:]
    
    # -h[j+shift] for valid j+shift
    if shift >= 0:
        # j+shift valid: j + shift < n, so j < n - shift
        g[:n - shift] -= h[shift:]
    else:
        # j+shift valid: j + shift >= 0, so j >= -shift
        g[-shift:] -= h[:-shift]
    
    g *= dx
    return g

def projected_subgradient(h, n, deadline, maxit=50000):
    """Projected subgradient method with adaptive step size."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    alpha = 0.1  # initial step size
    patience = 0
    max_patience = 500
    
    for it in range(maxit):
        if time.time() > deadline:
            break
        
        c = compute_correlation(h, n)
        cur_c = c.max()
        
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()
            patience = 0
            alpha = min(alpha * 1.05, 0.5)  # slight increase
        else:
            patience += 1
            if patience % 10 == 0:
                alpha *= 0.95
        
        if patience > max_patience:
            break
        
        # Find active set
        tol = 1e-10
        active = np.where(c >= cur_c - tol)[0]
        
        # Compute averaged subgradient
        g_avg = np.zeros(n)
        for s in active:
            g_avg += subgradient_vec(h, n, s)
        g_avg /= len(active)
        
        # Projected gradient step
        h_new = h - alpha * g_avg
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < cur_c + 1e-12:  # accept if not significantly worse
            h = h_new
        else:
            # Try smaller step
            h_new = h - alpha * 0.5 * g_avg
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < cur_c + 1e-12:
                h = h_new
                alpha *= 0.5
            else:
                alpha *= 0.5
    
    return best_h, best_c

    dx = 2.0 / n
    shift = s - (n - 1)
    g = np.zeros(n)
    if shift > 0:

        g[:n - shift] -= h[shift:]
    elif shift < 0:

        g[:n - sh] += 1.0 - h[sh:]
        g[sh:] -= h[:n - sh]
    else:
        g[:] = 1.0 - 2.0 * h

    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 0.5
    st = 0.0
    patience = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_full_c(h, n)
        cur = float(c.max())
        if cur < best_c - 1e-14:
            best_c = cur
            best_h = h.copy()
            patience = 0
            alpha = min(alpha * 1.02, 2.0)
        else:
            patience += 1
            if patience > 300:
                break
            if patience % 30 == 0:
                alpha *= 0.9
        tol = max(1e-9, 0.005 * cur)
        active = np.where(c >= cur - tol)[0]
        g = np.zeros(n)
        for s in active:
            g += subgrad_single(h, n, s)
        g /= len(active)
        st = 0.9 * st - alpha * g
        h_new = h + st
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < cur + 1e-12:
            h = h_new
        else:
            alpha *= 0.7
            st *= 0.5
    return best_h, best_c

def optimize_one(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    c = compute_full_c(best_h, n)
    cur_max = c.max()
    tol = max(1e-6, 0.02 * cur_max)
    active = np.where(c >= cur_max - tol)[0]
    if len(active) > 15:
        top_idx = np.argsort(c)[-15:]
        active = top_idx

    for _outer in range(30):
        if time.time() > deadline - 10:
            break
        h = slsqp_active(best_h, n, active, maxiter=1500, tol=1e-18)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
        c = compute_full_c(best_h, n)
        cur_max = c.max()
        tol2 = max(1e-7, 0.01 * cur_max)
        active = np.where(c >= cur_max - tol2)[0]
        if len(active) > 15:
            active = np.argsort(c)[-15:]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

    alphas = [5000, 50000, 500000, 2000000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_smooth(best_h, n, a, maxiter=2000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    best_h, best_c = subgrad_step(best_h, n, max_iter=3000, deadline=max(time.time(), deadline - 30))

    if n <= 120:
        if time.time() < deadline - 10:
            best_h, best_c = pair_search(best_h, n, deadline - 5,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005])
    else:
        if time.time() < deadline - 10:
            best_h, best_c = near_pair_search(best_h, n, deadline - 5,
                                              [0.01, 0.005, 0.002, 0.001], radius=40)

    if time.time() < deadline - 5:
        ht = slsqp_active(best_h, n, active, maxiter=3000, tol=1e-19)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()

    if n <= 120:
        if time.time() < deadline - 2:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.002, 0.001, 0.0005, 0.0002])
    else:
        if time.time() < deadline - 2:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.002, 0.001, 0.0005], radius=30)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc1 = min(350, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_one(h_init, n_init, t0 + alloc1)
        best_h, best_c = h_opt, c_opt

        x_init = np.linspace(0, 2, n_init)
        for n_try in [150, 250, 100]:
            if n_try == n_init:
                continue
            if time.time() > deadline - 60:
                break
            remaining = deadline - time.time()
            alloc = min(120, remaining * 0.2)
            if alloc < 30:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_one(h_interp, n_try, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.01 + 0.08 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 20) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_one(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if n <= 120:
            best_h, best_c = pair_search(best_h, n, deadline - 3,
                                         [0.001, 0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 3,
                                              [0.001, 0.0005, 0.0002, 0.0001], radius=30)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        c = compute_full_c(best_h, n)
        cur_max = c.max()
        tol = max(1e-8, 0.005 * cur_max)
        active = np.where(c >= cur_max - tol)[0]
        if len(active) > 10:
            active = np.argsort(c)[-10:]
        ht = slsqp_active(best_h, n, active, maxiter=3000, tol=1e-20)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        if n <= 120:
            best_h, best_c = pair_search(best_h, n, deadline - 1, [0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.0005, 0.0002, 0.0001], radius=25)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
