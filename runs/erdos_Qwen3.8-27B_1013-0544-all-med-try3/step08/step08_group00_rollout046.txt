def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:

            tot = room.sum()

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fine_local_search(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    thresh = max_val - max(1e-8, 0.003 * max_val)
    hot = np.where(c_vec >= thresh)[0]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for idx in hot:
                if time.time() > deadline:
                    return h, best_c
                k = int(idx) - (n - 1)
                if k >= 0:
                    pairs = [(i, i + k) for i in range(n - k)]
                else:
                    pairs = [(i, i + k) for i in range(-k, n) if i + k >= 0]
                for (i, j) in pairs:
                    if i < 0 or i >= n or j < 0 or j >= n:
                        continue
                    for sign in [1, -1]:
                        hi_new = h[i] + sign * step
                        hj_new = h[j] - sign * step
                        if hi_new >= -1e-12 and hi_new <= 1.0 + 1e-12 and hj_new >= -1e-12 and hj_new <= 1.0 + 1e-12:
                            old_i, old_j = h[i], h[j]
                            h[i] = max(0.0, min(1.0, hi_new))
                            h[j] = max(0.0, min(1.0, hj_new))
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] = old_i
                                h[j] = old_j
    return h, best_c

def coordinate_descent(h, n, deadline, n_sweeps=50):
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            best_val = h[i]
            best_c5 = best_c
            for trial in [h[i] + 0.01, h[i] - 0.01, h[i] + 0.005, h[i] - 0.005,
                         h[i] + 0.001, h[i] - 0.001, 0.0, 1.0]:
                if trial < -1e-9 or trial > 1.0 + 1e-9:
                    continue
                trial = max(0.0, min(1.0, trial))
                old = h[i]
                h[i] = trial
                adj = (h[i] - old)
                if abs(adj) > 1e-15:
                    for j in range(n):
                        if j != i:
                            h[j] -= adj * h[j] / max(h.sum() - h[i], 1e-15)
                    h = np.clip(h, 0.0, 1.0)
                c = compute_c5(h, n)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    best_val = trial
                h[i] = old
            if best_val != h[i]:
                h[i] = best_val
                h = project_feasible(h, n)
                best_c = compute_c5(h, n)
    return h, best_c

def bang_bang_project(h, n, strength):
    h2 = h.copy()
    mask_high = h2 > 0.5
    mask_low = ~mask_high
    h2[mask_high] = 1.0
    h2[mask_low] = 0.0
    h_blend = (1.0 - strength) * h + strength * h2
    return project_feasible(h_blend, n)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x):
        return np.ones(len(x))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def multi_start_optimize(h, n, deadline, n_starts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    np_rng = np.random.RandomState(42)
    for s in range(n_starts):
        if time.time() > deadline - 60:
            break
        h_trial = h.copy()
        scale = [0.01, 0.005, 0.002][min(s, 2)]
        h_trial += np_rng.randn(n) * scale
        h_trial = project_feasible(h_trial, n)
        t_sub = min(time.time() + 80, deadline - 30)
        h_trial, c_trial = run_lbfgs(h_trial, n, t_sub,
            configs=[(500, 100.0, 0.0), (2000, 1000.0, 0.5), (5000, 5000.0, 1.0)])
        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick L-BFGS at original resolution (20s)
    t1 = min(t0 + 20, deadline - 900)
    if t1 > time.time() + 3:
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            configs=[(500, 100.0, 0.0), (2000, 1000.0, 0.1), (5000, 5000.0, 0.5)])

    # Phase 2: SQP at original resolution (30s)
    t2 = min(t0 + 50, deadline - 800)
    if t2 > time.time() + 5:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=200, rho_init=0.05)

    # Phase 3: Upsample to n=1000
    n_fine = 1000
    if len(best_h) != n_fine:
        best_h = upsample(best_h, len(best_h), n_fine)
        best_c = compute_c5(best_h, n_fine)
    n_cur = n_fine

    # Phase 4: Extended L-BFGS at n=1000 (250s)
    t4 = min(t0 + 300, deadline - 450)
    if t4 > time.time() + 10:
        best_h, best_c = run_lbfgs(best_h, n_cur, t4,
            configs=[(200, 50.0, 0.0), (1000, 500.0, 0.1), (3000, 2000.0, 0.5),
                     (8000, 10000.0, 1.0), (15000, 20000.0, 2.0), (30000, 50000.0, 3.0)])

    # Phase 5: SQP at n=1000 (150s)
    t5 = min(t0 + 450, deadline - 300)
    if t5 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_cur, t5, max_iter=500, rho_init=0.06, max_active=35)

    # Phase 6: Fine local search at n=1000 (120s)
    t6 = min(t0 + 570, deadline - 220)
    if t6 > time.time() + 5:
        best_h, best_c = fine_local_search(best_h, n_cur, t6, radius=25)

    # Phase 7: Another SQP round (80s)
    t7 = min(t0 + 650, deadline - 150)
    if t7 > time.time() + 5:
        best_h, best_c = sqp_optimize(best_h, n_cur, t7, max_iter=400, rho_init=0.03, max_active=40)

    # Phase 8: Bang-bang projection + re-optimize (60s)
    t8 = min(t0 + 710, deadline - 100)
    if t8 > time.time() + 10:
        for strength in [0.3, 0.5, 0.7]:
            if time.time() > t8 - 5:
                break
            h_bb = bang_bang_project(best_h, n_cur, strength)
            c_bb = compute_c5(h_bb, n_cur)
            if c_bb < best_c:
                best_h, best_c = h_bb, c_bb
        t8b = min(time.time() + 40, t8)
        if t8b > time.time() + 10:
            best_h, best_c = run_lbfgs(best_h, n_cur, t8b,
                configs=[(5000, 5000.0, 0.5), (10000, 10000.0, 1.0)])

    # Phase 9: Upsample to n=2000 and optimize (150s if time)
    n_super = 2000
    t9 = time.time()
    if t9 < deadline - 160:
        h_super = upsample(best_h, n_cur, n_super)
        c_super = compute_c5(h_super, n_super)
        t9_end = min(time.time() + 130, deadline - 30)
        if t9_end > t9 + 10:
            h_super, c_super = run_lbfgs(h_super, n_super, t9_end,
                configs=[(1000, 500.0, 0.0), (3000, 2000.0, 0.5), (8000, 10000.0, 1.0)])
        if c_super < best_c:
            best_c = c_super
            best_h = h_super
            n_cur = n_super

    # Phase 10: Final SLSQP polish (remaining time)
    t10_end = deadline - 3
    if time.time() < t10_end - 10:
        h_final, c_final = slqp_polish(best_h, len(best_h), t10_end, alpha=2000000, maxiter=1500)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
