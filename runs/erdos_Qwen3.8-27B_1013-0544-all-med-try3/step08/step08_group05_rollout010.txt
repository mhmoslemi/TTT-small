    # For each j and each valid s:
    # grad[j] += dx * w[n-1+s] * [(1-h[j+s]) - h[j-s]] (with validity)

    grad = np.zeros(n)
    for s in range(-(n-1), n):
        k = n-1+s
        if w[k] < 1e-15:
            continue
        # For shift s: contributes w[k]*((1-h[j+s]) - h[j-s]) to grad[j]
        # j+s valid: 0 <= j+s < n => max(0,-s) <= j < n-s (if s>=0) or 0 <= j < n+s (if s<0)
        # j-s valid: 0 <= j-s < n => s <= j < n+s (if s>=0) or 0 <= j < n-s (if s<0)
        
        # Forward term: (1-h[j+s]) for valid j+s
        if s >= 0:
            j_min_f = 0
            j_max_f = n-1-s  # j+s <= n-1
        else:
            j_min_f = -s
            j_max_f = n-1
        
        # Backward term: h[j-s] for valid j-s
        if s >= 0:
            j_min_b = s
            j_max_b = n-1
        else:
            j_min_b = 0
            j_max_b = n-1+s  # j-s <= n-1 => j <= n-1+s
        
        # Forward contribution
        if j_min_f <= j_max_f:
            idx_f = np.arange(j_min_f, j_max_f+1)
            grad[idx_f] += dx * w[k] * (1.0 - h[idx_f + s])
        
        # Backward contribution
        if j_min_b <= j_max_b:
            idx_b = np.arange(j_min_b, j_max_b+1)
            grad[idx_b] -= dx * w[k] * h[idx_b - s]
    
    return grad

        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_smoothed(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    
    grad = np.zeros(n)
    # Loop over shifts
    for s in range(-(n-1), n):

        wk = w[k]

            grad[jf] += wk * (1.0 - h[jf + s])

            jb = np.arange(s, n)
        else:
            jb = np.arange(0, n + s)

            grad[jb] -= wk * h[jb - s]
    
    grad *= dx
    return grad, w, m

    target = n / 2.0
    
    def obj_grad(x):
        c = np.correlate(x, 1-x, mode='full')
        dx = 2.0 / n
        cc = c * dx
        m = cc.max()
        sh = alpha * (cc - m)
        sm = sh.max()
        ew = np.exp(sh - sm)
        sw = ew.sum()
        obj = m + np.log(sw) / alpha
        # Constraint penalty
        cv = x.sum() - target
        obj += lambda_pen * cv * cv
        
        grad = np.zeros(n)
        w = ew / sw
        for s in range(-(n-1), n):
            k = n - 1 + s
            wk = w[k]
            if wk < 1e-15:
                continue
            if s >= 0:
                jf = np.arange(0, n - s)
                jb = np.arange(s, n)
            else:

                jb = np.arange(0, n + s)
            if len(jf) > 0:
                grad[jf] += wk * (1.0 - x[jf + s])
            if len(jb) > 0:
                grad[jb] -= wk * x[jb - s]
        grad *= (2.0 / n)
        grad += 2.0 * lambda_pen * cv * np.ones(n)
        
        return obj, grad

def gradient_descent(h, n, alpha_start, alpha_end, n_steps, deadline):
    """Custom projected gradient descent with momentum"""
    h = h.copy()
    target = n / 2.0
    dx = 2.0 / n

    best_h = h.copy()
    momentum = np.zeros(n)
    beta1 = 0.9
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        progress = step / max(n_steps - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = 0.01 * (1 - 0.5 * progress)  # decreasing learning rate
        
        c = np.correlate(h, 1-h, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for s in range(-(n-1), n):
            k = n - 1 + s
            wk = w[k]
            if wk < 1e-15:
                continue
            if s >= 0:
                jf = np.arange(0, n - s)
                jb = np.arange(s, n)
            else:
                jf = np.arange(-s, n)
                jb = np.arange(0, n + s)
            if len(jf) > 0:
                grad[jf] += wk * (1.0 - h[jf + s])
            if len(jb) > 0:
                grad[jb] -= wk * h[jb - s]
        grad *= dx
        
        # Add constraint gradient
        cv = h.sum() - target
        grad += 2.0 * 100.0 * cv * np.ones(n)  # constraint penalty
        
        # Momentum update
        momentum = beta1 * momentum + (1 - beta1) * grad
        h = h - lr * momentum
        h = project_feasible(h, n)
        
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
    
    return best_h, best_c

    momentum = np.zeros(n)
    beta = 0.85

    for step in range(n_steps):
        if time.time() > deadline:
            break
        t = step / max(n_steps - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr0 * (1.0 - 0.7 * t)

        c_raw = np.correlate(h, 1 - h, mode='full')
        cc = c_raw * dx
        m = cc.max()
        sh = alpha * (cc - m)
        sm = sh.max()
        ew = np.exp(sh - sm)
        sw = ew.sum()
        w = ew / sw

        grad = _grad_of_smoothed_max(h, n, alpha, w)
        cv = h.sum() - target
        grad += 100.0 * cv * np.ones(n)

        momentum = beta * momentum + (1 - beta) * grad
        h = h - lr * momentum
        h = project_feasible(h, n)

        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()

    return best_h, best_c

def subgrad_focus(h, n, n_steps, deadline):
    h = h.copy()
    target = n / 2.0
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()

    for step in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        active = np.where(c > m - 0.02)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        lr = 0.01 / (1.0 + step * 0.002)
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                jf = np.arange(0, n - s)
                jb = np.arange(s, n)
            else:
                jf = np.arange(-s, n)
                jb = np.arange(0, n + s)
            if len(jf) > 0:
                grad[jf] += (1.0 - h[jf + s])
            if len(jb) > 0:
                grad[jb] -= h[jb - s]
        grad *= dx / len(active)

        cv = h.sum() - target
        grad += 80.0 * cv * np.ones(n)

        h = h - lr * grad
        h = project_feasible(h, n)
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()

    return best_h, best_c

def smart_local_search(h, n, deadline, n_restarts=3, seed=42):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    target = n / 2.0

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        curr = best_h.copy()
        curr_c = best_c
        stuck = 0
        for sweep in range(3000):
            if time.time() > deadline:
                break
            corr = np.correlate(curr, 1 - curr, mode='full')
            max_k = np.argmax(corr)
            s = max_k - (n - 1)
            # Find pairs that would reduce this specific correlation
            if s >= 0:
                valid_f = np.arange(0, n - s)
                valid_b = np.arange(s, n)
            else:
                valid_f = np.arange(-s, n)
                valid_b = np.arange(0, n + s)

            # Try to reduce overlap: move probability from forward to backward
            # or vice versa
            ntried = 0
            improved = False
            for _ in range(400):
                ntried += 1
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                delta = curr[i] - curr[j]
                if abs(delta) < 1e-10:
                    continue
                # Swap or partial swap
                amt = min(abs(delta), 0.1 + 0.4 * rng.random())
                if delta > 0:
                    new_i = curr[i] - amt
                    new_j = curr[j] + amt
                else:
                    new_i = curr[i] + amt
                    new_j = curr[j] - amt
                if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
                    continue
                h_new = curr.copy()
                h_new[i] = new_i
                h_new[j] = new_j
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < curr_c - 1e-12:
                    curr = h_new
                    curr_c = c_new
                    improved = True
                    if curr_c < best_c:
                        best_c = curr_c
                        best_h = curr.copy()
                    break
            if not improved:
                stuck += 1
                if stuck > 20:
                    # Perturbation
                    curr = project_feasible(curr + 0.03 * rng.randn(n), n)
                    curr_c = compute_c5(curr, n)
                    if curr_c < best_c:
                        best_c = curr_c
                        best_h = curr.copy()
                    stuck = 0
                elif stuck > 60:
                    break
            else:
                stuck = 0

    return best_h, best_c

def make_structured_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    # Sinusoidal-ish
    h = 0.5 + 0.3 * np.sin(np.pi * x + rng.uniform(0, 2 * np.pi))
    h = project_feasible(np.clip(h, 0, 1), n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: L-BFGS-B with increasing alpha (fast continuous optimization)
    p1 = t0 + 0.12 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > p1 - 5:
            break
        h = lbfgsb_optimize(h, n, alpha, 50000, 300, p1 - 5)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Gradient with momentum
    p2 = t0 + 0.25 * (budget_s - 5)
    h2, c2 = gradient_momentum(best_h, n, 5000, 200000, 8000, p2)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Subgradient focus
    p3 = t0 + 0.35 * (budget_s - 5)
    h3, c3 = subgrad_focus(best_h, n, 5000, p3)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: L-BFGS-B polish with higher alpha
    p4 = t0 + 0.45 * (budget_s - 5)
    for alpha in [100000, 500000, 2000000]:
        if time.time() > p4 - 5:
            break
        h = best_h.copy() + 0.005 * rng.randn(n)
        h = project_feasible(h, n)
        h = lbfgsb_optimize(h, n, alpha, 200000, 400, p4 - 5)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Smart local search
    p5 = t0 + 0.60 * (budget_s - 5)
    h5, c5 = smart_local_search(best_h, n, p5, n_restarts=3, seed=seed)
    if c5 < best_c:
        best_c = c5
        best_h = h5.copy()

    # Phase 6: Multi-resolution exploration
    p6 = t0 + 0.78 * (budget_s - 5)
    for nh in [300, 400, 500, 600]:
        if time.time() > p6 - 20:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        # Optimize at new resolution
        hh = lbfgsb_optimize(hh, nh, 50000, 50000, 200, p6 - 10)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh
        hh = lbfgsb_optimize(hh, nh, 200000, 100000, 200, p6 - 5)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh
        # Quick momentum
        hh2, ch2 = gradient_momentum(hh, nh, 10000, 500000, 3000, p6 - 3)
        if ch2 < best_c:
            best_c = ch2
            best_h = hh2.copy()
            n = nh

    # Phase 7: Structured alternatives (different starting points)
    p7 = t0 + 0.85 * (budget_s - 5)
    for alt_seed in [seed + 10, seed + 20]:
        if time.time() > p7 - 10:
            break
        ha = make_structured_init(n, alt_seed)
        ha = lbfgsb_optimize(ha, n, 50000, 50000, 200, p7 - 10)
        ca = compute_c5(ha, n)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()

    # Phase 8: Final polish
    if time.time() < dl - 10:
        h = best_h.copy() + 0.003 * rng.randn(n)
        h = project_feasible(h, n)
        h = lbfgsb_optimize(h, n, 1000000, 500000, 500, dl - 8)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

        h8, c8 = smart_local_search(best_h, n, dl, n_restarts=2, seed=seed + 50)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()

    # Final: ensure feasibility and compute exact objective
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
