def add_single_grad(g, h, n, lag, scale=1.0):
    r = abs(lag)
    if scale == 0.0:
        return
    if r == 0:
        g += scale * (1.0 - 2.0 * h)
        return
    nv = n - r
    if lag > 0:
        g[r:n] += scale * (1.0 - h[:nv])
        g[:nv] -= scale * h[r:n]
    else:
        g[:nv] += scale * (1.0 - h[r:n])
        g[nv? r:n] -= scale * h[:nv]

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    n = int(n)
    T = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)

        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < T:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)

        diff = T - float(h.sum())
        if abs(diff) <= 1e-13: break
        if diff > 0:

            tot = float(room.sum())
            if tot <= 1e-15: break
            h += (diff / tot) * room

    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def eval_c(h):
    return float(np.max(eval_vec(h)))


def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    n = int(n)
    target = 0.5 * float(n)

    h = np.clip(h, 0.0, 1.0)

    lo = -1.0
    hi = 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(60):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)

    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        for _ in range(12):
            if diff > 0.0:
                idx = int(np.argmax(1.0 - h))
                add = min(diff, 1.0 - h[idx])
            else:
                idx = int(np.argmax(h))
                add = max(diff, -h[idx])
            h[idx] += add
            diff = target - float(h.sum())
            if abs(diff) <= 1e-13:
                break

    return h


def add_single_grad(g, h, n, lag, scale=1.0):
    """Add scale * gradient of the np.correlate(h, 1-h, full) term for one lag."""
    if scale == 0.0:
        return
    r = abs(int(lag))
    if r == 0:
        g += scale * (1.0 - 2.0 * h)
        return
    nv = n - r
    if lag > 0:
        g[r:n] += scale * (1.0 - h[0:nv])
        g[0:nv] -= scale * h[r:n]
    else:
        g[0:nv] += scale * (1.0 - h[r:n])
        g[r:n] -= scale * h[0:nv]


def active_gradient_uniform(h, n, idx):
    g = np.zeros(n, dtype=float)
    cnt = float(len(idx))
    if cnt <= 0.0:
        return g
    scale = 1.0 / cnt
    off = n - 1
    for t in idx:
        lag = int(t) - off
        add_single_grad(g, h, n, lag, scale)
    return g


def select_active_indices(c_vec, max_k=80, margin_frac=0.002, min_margin=1e-6):
    max_k = max(1, int(max_k))
    m = float(np.max(c_vec))
    margin = max(min_margin, margin_frac * max(1e-12, m))
    idx = np.flatnonzero(c_vec >= m - margin)
    if idx.size == 0:
        idx = np.array([int(np.argmax(c_vec))], dtype=int)
    else:
        idx = idx.astype(int, copy=False)
        if idx.size > max_k:
            vals = c_vec[idx]
            order = np.argsort(-vals)
            idx = idx[order[:max_k]]
    return idx


def make_t_grid(tlo, thi, grid_count=20):
    if thi <= tlo:
        return np.empty(0, dtype=float)
    grid_count = max(4, int(grid_count))
    base = np.linspace(tlo, thi, grid_count)

    fixed = np.array(
        [
            1e-6, 3e-6, 1e-5, 3e-5, 1e-4, 3e-4,
            1e-3, 3e-3, 1e-2, 3e-2, 1e-1, 3e-1,
        ],
        dtype=float,
    )
    relax = 0.05 * (thi - tlo)
    fixed = np.concatenate((fixed, relax * np.array([1e-3, 3e-3, 1e-2, 3e-2, 1e-1], dtype=float)))

    near = np.array([0.0], dtype=float)
    if tlo <= 0.0 <= thi:
        near = np.concatenate((near, fixed, -fixed))

    ts = np.concatenate((base, near))
    ts = ts[(ts >= tlo - 1e-12) & (ts <= thi + 1e-12)]
    if ts.size == 0:
        return np.empty(0, dtype=float)
    ts = np.clip(ts, tlo, thi)
    ts = np.unique(np.round(ts, 13))

    if ts.size > 52:
        keep = np.unique(np.linspace(0, ts.size - 1, 48).astype(int))
        ts = ts[keep]
    return ts


def line_search_pair(h, n, i, j, c_current, max_evals=36):
    a = float(h[i])
    b = float(h[j])
    tlo = max(-a, b - 1.0)
    thi = min(1.0 - a, b)
    if not (thi > tlo + 1e-11):
        return None

    ts = make_t_grid(tlo, thi, 18)
    best = c_current
    best_t = 0.0
    evals = 0

    for t in ts:
        if abs(t) <= 1e-14:
            continue
        if evals >= max_evals:
            break
        h[i] = a + t
        h[j] = b - t
        if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
            h[i] = a
            h[j] = b
            continue
        c = eval_c(h)
        h[i] = a
        h[j] = b
        evals += 1
        if c < best - 1e-15:
            best = c
            best_t = float(t)

    if abs(best_t) <= 1e-14 or best >= c_current - 1e-14:
        return None

    width = 0.25 * abs(best_t) + 1e-5
    rlo = max(tlo, best_t - width)
    rhi = min(thi, best_t + width)
    for t in np.linspace(rlo, rhi, 9):
        if abs(t) <= 1e-14 or abs(t - best_t) <= 1e-14:
            continue
        if evals >= max_evals + 12:
            break
        h[i] = a + t
        h[j] = b - t
        if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
            h[i] = a
            h[j] = b
            continue
        c = eval_c(h)
        h[i] = a
        h[j] = b
        evals += 1
        if c < best - 1e-15:
            best = c
            best_t = float(t)

    if best < c_current - 1e-14:
        return (float(best_t), float(best))
    return None


def pair_local_search(
    h,
    n,
    deadline,
    top_pairs=24,
    max_k_active=120,
    random_pairs=20,
    grid_count=20,
    max_evals=36,
    rng=None,
):
    h = project_feasible(h, n)
    c_vec = eval_vec(h)
    c = float(np.max(c_vec))
    no_improve = 0

    if rng is None:
        rng = np.random.default_rng(0)

    max_pairs_per_round = max(120, min(520, int(260000 / max(1, n))))

    while time.time() < deadline - 2.0:
        rem = deadline - time.time()
        if rem <= 3.0:
            break

        idx = select_active_indices(c_vec, max_k=max_k_active, margin_frac=0.003, min_margin=1e-6)
        g = active_gradient_uniform(h, n, idx)

        order = np.argsort(g)
        lows = order[:top_pairs]
        highs = order[-top_pairs:]

        pairs = []
        seen = set()
        for hi in highs:
            i = int(hi)
            for lo in lows:
                j = int(lo)
                if i == j:
                    continue
                code = i * n + j
                if code in seen:
                    continue
                seen.add(code)
                pairs.append((i, j))

        if random_pairs > 0:
            for _ in range(random_pairs):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    pairs.append((i, j))

        if len(pairs) > max_pairs_per_round:
            if len(pairs) > 0:
                sel = rng.choice(len(pairs), size=max_pairs_per_round, replace=False)
                pairs = [pairs[int(k)] for k in sel]

        improved = False
        accepts = 0

        for i, j in pairs:
            if time.time() > deadline - 2.0:
                break
            res = line_search_pair(h, n, i, j, c, max_evals=max_evals)
            if res is not None:
                t, bc = res
                h[i] = float(h[i] + t)
                h[j] = float(h[j] - t)
                if h[i] < 0.0:
                    h[i] = 0.0
                elif h[i] > 1.0:
                    h[i] = 1.0
                if h[j] < 0.0:
                    h[j] = 0.0
                elif h[j] > 1.0:
                    h[j] = 1.0
                c = float(bc)
                improved = True
                accepts += 1
                if accepts % 15 == 0:
                    c_vec = eval_vec(h)

        if improved:
            no_improve = 0
            c_vec = eval_vec(h)
        else:
            no_improve += 1
            if no_improve >= 3:
                break

    h = project_feasible(h, n)
    c = eval_c(h)
    return h, c


def active_smooth_obj(x, n, idx, alpha, pen):
    c = eval_vec(x)
    vals = c[idx]
    m = float(np.max(vals))
    if vals.size == 1:
        val = m
    else:
        sh = alpha * (vals - m)
        mx = float(np.max(sh))
        if mx > 500.0:
            val = m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha
        else:
            val = m + float(np.log(np.sum(np.exp(sh)))) / alpha
    val += pen * (float(x.sum()) - 0.5 * float(n)) ** 2
    return float(val)


def active_smooth_grad(x, n, idx, alpha, pen):
    c = eval_vec(x)
    vals = c[idx]
    m = float(np.max(vals))
    if vals.size == 1:
        w = np.array([1.0], dtype=float)
    else:
        sh = alpha * (vals - m)
        mx = float(np.max(sh))
        if mx > 500.0:
            w = np.exp(sh - mx)
        else:
            w = np.exp(sh)
        w = w / float(np.sum(w))

    dx = 2.0 / float(n)
    off = n - 1
    g = np.zeros(n, dtype=float)
    for r, idx_val in enumerate(idx):
        wi = float(w[r])
        if wi <= 1e-15:
            continue
        lag = int(idx_val) - off
        add_single_grad(g, x, n, lag, scale=dx * wi)

    if pen != 0.0:
        g += (2.0 * pen) * (float(x.sum()) - 0.5 * float(n))
    return g


def run_lbgfs_active(
    h,
    n,
    end_time,
    stages=((40, (200.0, 1000.0)), (25, (3000.0, 10000.0)), (12, (50000.0, 200000.0))),
    pen=None,
):
    h = project_feasible(h, n)
    c = eval_c(h)

    if pen is None:
        pen = max(1e-4, 0.01 * (500.0 / max(1, n)))

    bounds = [(0.0, 1.0)] * n

    for topk, alphas in stages:
        for alpha in alphas:
            if time.time() > end_time - 35.0:
                break
            c_vec = eval_vec(h)
            idx = select_active_indices(c_vec, max_k=topk, margin_frac=0.002, min_margin=1e-7)
            if idx.size == 0:
                continue

            rem = max(10.0, end_time - time.time())
            if n <= 700:
                maxiter = min(450, max(60, int(rem * 18)))
            elif n <= 1200:
                maxiter = min(320, max(45, int(rem * 12)))
            else:
                maxiter = min(220, max(35, int(rem * 8)))

            try:
                res = minimize(
                    active_smooth_obj,
                    h,
                    jac=active_smooth_grad,
                    args=(n, idx, float(alpha), pen),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={
                        "maxiter": maxiter,
                        "ftol": 1e-15,
                        "gtol": 1e-11,
                        "maxfun": max(maxiter + 10, maxiter * 2),
                        "maxls": 25,
                    },
                )
                h2 = project_feasible(res.x, n)
                c2 = eval_c(h2)
                if np.isfinite(c2) and c2 < c - 1e-14:
                    h = h2
                    c = c2
            except Exception:
                pass

    h = project_feasible(h, n)
    c = eval_c(h)
    return h, c


def subgrad_search(h, n, end_time, init_step=0.05):
    h = project_feasible(h, n)
    c = eval_c(h)
    step = max(1e-6, float(init_step))
    no_improve = 0

    for _ in range(6000):
        if time.time() > end_time - 2.0:
            break
        c_vec = eval_vec(h)
        idx = select_active_indices(c_vec, max_k=80, margin_frac=0.002, min_margin=1e-6)
        g = active_gradient_uniform(h, n, idx)
        g -= float(np.mean(g))
        norm = float(np.linalg.norm(g))
        if norm <= 1e-13:
            break
        d = -g / norm

        improved = False
        for mult in (1.0, 0.5, 0.25, 0.125, 0.0625):
            h2 = project_feasible(h + step * mult * d, n)
            c2 = eval_c(h2)
            if np.isfinite(c2) and c2 < c - 1e-15:
                h = h2
                c = c2
                step = min(step * 1.3, 0.25)
                no_improve = 0
                improved = True
                break

        if not improved:
            step *= 0.65
            no_improve += 1
            if no_improve >= 10 or step < 1e-8:
                break

    h = project_feasible(h, n)
    c = eval_c(h)
    return h, c


def slqp_obj(z, n):
    return float(z[-1])


def slqp_jac(z, n):
    g = np.zeros(n + 1, dtype=float)
    g[-1] = 1.0
    return g


def slqp_eq_fun(z, n):
    return float(z[:n].sum() - 0.5 * float(n))


def slqp_eq_jac(z, n):
    g = np.zeros(n + 1, dtype=float)
    g[:n] = 1.0
    return g


def slqp_con_fun(z, n, idx):
    h = z[:n]
    t = z[-1]
    return t - eval_vec(h)[idx]


def slqp_con_jac(z, n, idx):
    h = z[:n]
    dx = 2.0 / float(n)
    off = n - 1
    m = int(len(idx))
    J = np.zeros((m, n + 1), dtype=float)

    for a, idx_val in enumerate(idx):
        lag = int(idx_val) - off
        if lag == 0:
            J[a, 0:n] -= dx * (1.0 - 2.0 * h)
        elif lag > 0:
            r = lag
            nv = n - r
            J[a, r:n] -= dx * (1.0 - h[0:nv])
            J[a, 0:nv] += dx * h[r:n]
        else:
            r = -lag
            nv = n - r
            J[a, 0:nv] -= dx * (1.0 - h[r:n])
            J[a, r:n] += dx * h[0:nv]
        J[a, -1] = 1.0

    return J


def run_slqp_active(h, n, end_time, top_k=10):
    h = project_feasible(h, n)
    c = eval_c(h)
    if time.time() > end_time - 15.0:
        return h, c

    c_vec = eval_vec(h)
    idx = select_active_indices(c_vec, max_k=min(int(top_k), 14), margin_frac=0.002, min_margin=1e-7)
    z0 = np.concatenate((h, np.array([c + 1e-5, dtype=float])))

    bounds = [(0.0, 1.0)] * n
    t_upper = max(1.5, c + 1.0)
    bounds.append((0.0, t_upper))

    cons = [
        {"type": "eq", "fun": slqp_eq_fun, "jac": slqp_eq_jac, "args": (n,)},
        {"type": "ineq", "fun": slqp_con_fun, "jac": slqp_con_jac, "args": (n, idx)},
    ]

    maxiter = 70 if n > 1200 else 100
    try:
        res = minimize(
            slqp_obj,
            z0,
            jac=slqp_jac,
            args=(n,),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-12},
        )
        h2 = project_feasible(res.x[:n], n)
        c2 = eval_c(h2)
        if np.isfinite(c2) and c2 < c - 1e-14:
            return h2, c2
    except Exception:
        pass

    return h, c


def binarize(h, n):
    n = int(n)
    target = 0.5 * float(n)
    h2 = np.zeros(n, dtype=float)
    order = np.argsort(-h)
    k = int(float(target))
    if k > 0:
        h2[order[:k]] = 1.0
    rem = target - float(k)
    if rem > 1e-14 and k < n:
        h2[order[k]] = float(rem)
    return project_feasible(h2, n)


def interpolate_h(h, n_new):
    n_old = int(h.size)
    n_new = int(n_new)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    return np.interp(x_new, x_old, h)


def choose_up_resolution(n):
    n = int(n)
    if n <= 500:
        cand = 1000
    elif n <= 1000:
        cand = 1200
    elif n <= 1500:
        cand = 1200
    elif n <= 2500:
        cand = 1600
    else:
        cand = 1000
    if cand % 2 == 1:
        cand += 1
    return max(100, cand)


def refine_candidate(
    h,
    n,
    end_time,
    rng,
    do_lbgfs=True,
    do_slqp=True,
    do_subgrad=True,
    pair_top=24,
):
    if time.time() >= end_time - 5.0:
        h = project_feasible(h, n)
        return h, eval_c(h)

    h = project_feasible(h, n)
    c = eval_c(h)

    max_evals = 44 if n <= 900 else 30

    rem = max(0.0, end_time - time.time())

    # Initial pair search.
    p1 = time.time() + 0.32 * rem
    if p1 > end_time - 15.0:
        p1 = end_time - 15.0
    if time.time() < p1:
        h, c = pair_local_search(
            h,
            n,
            p1,
            top_pairs=pair_top,
            max_k_active=160 if n <= 900 else 110,
            random_pairs=22,
            grid_count=18,
            max_evals=max_evals,
            rng=rng,
        )

    # Gradient-based continuous refinement.
    if do_lbgfs and time.time() < end_time - 60.0:
        rem2 = max(0.0, end_time - time.time())
        p2 = time.time() + 0.26 * rem2
        if p2 > end_time - 25.0:
            p2 = end_time - 25.0
        if time.time() < p2:
            h, c = run_lbgfs_active(h, n, p2)

    # Projected subgradient collective moves.
    if do_subgrad and time.time() < end_time - 45.0:
        rem3 = max(0.0, end_time - time.time())
        p3 = time.time() + 0.12 * rem3
        if p3 > endtime := end_time - 20.0:
            pass
        if p3 > end_time - 20.0:
            p3 = end_time - 20.0
        if time.time() < p3:
            h, c = subgrad_search(h, n, p3)

    # Second pair search.
    slqp_reserve = 55.0 if do_slqp else 10.0
    p4_end = end_time - slqp_reserve - 5.0
    if time.time() < p4_end:
        rem4 = max(0.0, p4_end - time.time())
        p4 = time.time() + 0.22 * rem4
        if p4 > p4_end:
            p4 = p4_end
        if time.time() < p4:
            h, c = pair_local_search(
                h,
                n,
                p4,
                top_pairs=max(12, pair_top - 6),
                max_k_active=110 if n <= 900 else 80,
                random_pairs=14,
                grid_count=16,
                max_evals=max_evals,
                rng=rng,
            )

    # SLSQP polish on active constraints.
    if do_slqp and time.time() < end_time - 20.0:
        p5 = min(end_time - 5.0, time.time() + 32.0)
        top_k = 8 if n > 1200 else 10
        h, c = run_slqp_active(h, n, p5, top_k=top_k)
        if time.time() < end_time - 5.0:
            h, c = pair_local_search(
                h,
                n,
                end_time - 3.0,
                top_pairs=12,
                max_k_active=70,
                random_pairs=10,
                grid_count=12,
                max_evals=26,
                rng=rng,
            )

    h = project_feasible(h, n)
    c = eval_c(h)
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    b = float(budget_s)

    # Keep a small reserve so we always return a feasible point.
    reserve = min(25.0, 0.10 * max(1.0, b))
    deadline = t0 + max(1.0, b - reserve)

    rng = np.random.default_rng(seed)

    # Get the parent construction if present.
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.size < 50 or not np.all(np.isfinite(parent)):
            parent = np.full(250, 0.5, dtype=float)
    except Exception:
        parent = np.full(250, 0.5, dtype=float)

    n = int(parent.size)

    # Keep the working resolution in a regime we can really optimize within budget.
    if n > 2500:
        n_small = 1200
        parent = project_feasible(interpolate_h(parent, n_small), n_small)
        n = n_small
    if n < 100:
        parent = np.full(250, 0.5, dtype=float)
        n = 250

    # Build a few cheap initial candidates and choose the best starting point.
    cands = []

    base = project_feasible(parent, n)
    cands.append(base)
    cands.append(project_feasible(1.0 - base, n))

    for sh in (max(1, n // 8), max(1, n // 4), max(1, n // 2)):
        cands.append(project_feasible(np.roll(base, sh), n))

    cands.append(binarize(base, n))
    cands.append(project_feasible(base + rng.normal(0.0, 0.10, size=n), n))

    best_h = None
    best_c = 1e100
    for a in cands:
        cc = eval_c(a)
        if np.isfinite(cc) and cc < best_c:
            best_c = cc
            best_h = a

    best_n = n

    total = max(0.0, deadline - t0)
    if total < 45.0:
        best_h = project_feasible(best_h, best_n)
        best_c = eval_c(best_h)
        return best_h, best_c, best_n

    # Time slices.
    end_parent = t0 + 0.50 * total
    end_variant = t0 + 0.62 * total
    end_up = t0 + 0.85 * total

    # Main optimization at the parent's resolution.
    if time.time() < end_parent - 20.0:
        h2, c2 = refine_candidate(
            best_h,
            best_n,
            end_parent,
            rng,
            do_lbgfs=True,
            do_slqp=True,
            do_subgrad=True,
            pair_top=24 if best_n <= 1000 else 18,
        )
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # If still not at target, try a couple of transformed basins at the same resolution.
    if best_c > 0.38080 and time.time() < end_variant - 35.0:
        variants = []
        variants.append(project_feasible(1.0 - best_h, best_n))
        variants.append(binarize(best_h, best_n))
        for sh in (max(1, best_n // 6), max(1, best_n // 3), max(1, best_n // 2)):
            variants.append(project_feasible(np.roll(best_h, sh), best_n))
        variants.append(project_feasible(best_h + rng.normal(0.0, 0.05, size=best_n), best_n))

        scored = []
        for v in variants:
            cv = eval_c(v)
            if np.isfinite(cv):
                scored.append((float(cv), v))

        available = list(range(len(scored)))
        chosen = 0
        while available and chosen < 2 and time.time() < end_variant - 35.0:
            min_pos = 0
            min_val = scored[available[0]][0]
            for apos in range(1, len(available)):
                val = scored[available[apos]][0]
                if val < min_val:
                    min_val = val
                    min_pos = apos
            v = scored[available.pop(min_pos)][1]

            t_v = min(time.time() + 38.0, end_variant - 35.0)
            hv, cv = refine_candidate(
                v,
                best_n,
                t_v,
                rng,
                do_lbgfs=True,
                do_slqp=False,
                do_subgrad=False,
                pair_top=18,
            )
            if cv < best_c:
                best_h = hv
                best_c = cv
            chosen += 1

    # Try a different resolution. Finer grids can reduce discretization artifacts;
    # coarser grids can sometimes be easier to optimize in a fixed budget.
    if time.time() < end_up - 80.0:
        n_try = choose_up_resolution(best_n)
        if n_try != best_n:
            h_up = project_feasible(interpolate_h(best_h, n_try), n_try)
            c_up = eval_c(h_up)
            if np.isfinite(c_up) and c_up < best_c + 0.025:
                h_up, c_up = refine_candidate(
                    h_up,
                    n_try,
                    end_up,
                    rng,
                    do_lbgfs=True,
                    do_slqp=True,
                    do_subgrad=True,
                    pair_top=16 if n_try > 900 else 22,
                )
                if c_up < best_c:
                    best_h = h_up
                    best_c = c_up
                    best_n = n_try
        elif time.time() < end_up - 60.0:
            h2, c2 = refine_candidate(
                best_h,
                best_n,
                end_up,
                rng,
                do_lbgfs=True,
                do_slqp=True,
                do_subgrad=True,
                pair_top=22 if best_n <= 1000 else 16,
            )
            if c2 < best_c:
                best_h = h2
                best_c = c2

    # Final polish on the best resolution.
    if time.time() < deadline - 35.0:
        rem = deadline - time.time()
        if rem > 150.0:
            h2, c2 = refine_candidate(
                best_h,
                best_n,
                deadline - 20.0,
                rng,
                do_lbgfs=True,
                do_slqp=True,
                do_subgrad=True,
                pair_top=20 if best_n <= 1000 else 14,
            )
        else:
            h2, c2 = refine_candidate(
                best_h,
                best_n,
                deadline - 12.0,
                rng,
                do_lbgfs=rem > 70.0,
                do_slqp=True,
                do_subgrad=False,
                pair_top=16,
            )
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Last-second conservative pair moves.
    if time.time() < deadline - 8.0:
        h2, c2 = pair_local_search(
            best_h,
            best_n,
            deadline - 4.0,
            top_pairs=12,
            max_k_active=60,
            random_pairs=8,
            grid_count=12,
            max_evals=24,
            rng=rng,
        )
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = eval_c(best_h)

    if not np.isfinite(best_c):
        # Extremely defensive fallback, should not occur.
        fallback = project_feasible(np.full(max(100, best_n), 0.5, dtype=float), best_n)
        best_h = fallback
        best_c = eval_c(best_h)

    return best_h, float(best_c), int(best_n)
