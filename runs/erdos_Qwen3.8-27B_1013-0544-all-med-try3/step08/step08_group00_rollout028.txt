```
def softmax_weights(c,alpha):
    m=np.max(c)
    sh=alpha*(c-m)
    mx=sh.max()
    e=np.exp(sh-mx)
    s=e.sum()
    if s <=0: return np.full_like(c,1/len(c))
    return e/s
```

```
def deriv_lag_dx_shift(h,n,shift):
    dx=2.0/float(n)
    g=np.zeros(n,dtype=float)
    if shift >=0:
        kv=int(shift)
        if kv < n:

           g[0:nv] = (1.0 - h[kv:n]) * dx
```

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(120):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / float(np.sum(mask)))
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / float(np.sum(mask)))

    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-13:
        if diff > 0.0:
            order = np.argsort(1.0 - h)[::-1]
            rem = abs(diff)
            for idx in order:
                idx = int(idx)
                add = min(rem, max(0.0, 1.0 - float(h[idx])))
                if add <= 1e-16:
                    continue
                h[idx] += add
                rem -= add
                if rem <= 1e-14:
                    break
        else:
            order = np.argsort(h)[::-1]
            rem = abs(diff)
            for idx in order:
                idx = int(idx)
                sub = min(rem, max(0.0, float(h[idx])))
                if sub <= 1e-16:
                    continue
                h[idx] -= sub
                rem -= sub
                if rem <= 1e-14:
                    break

    return np.clip(h, 0.0, 1.0)


def c5_vector(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5_value(h, n):
    return float(np.max(c5_vector(h, n)))


def softmax_weights(c, alpha):
    cmax = float(np.max(c))
    sh = alpha * (c - cmax)
    mx = float(np.max(sh))
    e = np.exp(sh - mx)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(c.shape, 1.0 / float(c.size), dtype=float)
    return e / s


def deriv_lag_dx_shift(h, n, shift):
    dx = 2.0 / float(n)
    g = np.zeros(int(n), dtype=float)
    kv = int(shift)
    if kv >= 0:
        if kv < n:
            nv = n - kv
            if nv > 0:
                g[0:nv] += (1.0 - h[kv:n]) * dx
                g[kv:n] += (-h[0:nv]) * dx
    else:
        kv = -kv
        if kv < n:
            nv = n - kv
            if nv > 0:
                g[kv:n] += (1.0 - h[0:nv]) * dx
                g[0:nv] += (-h[kv:n]) * dx
    return g


def grad_row_dx(h, n, i):
    dx = 2.0 / float(n)
    i = int(i)
    g = np.zeros(2 * n - 1, dtype=float)
    g[i:i + n] += (1.0 - h[::-1])
    g[n - 1 - i:n + i] -= h
    return dx * g


def smooth_objective(x, n, alpha, mu):
    c = c5_vector(x, n)
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    val = m + float(mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu != 0.0:
        diff = float(np.sum(x)) - 0.5 * float(n)
        val += mu * diff * diff
    return float(val)


def smooth_grad(x, n, alpha, mu):
    c = c5_vector(x, n)
    w = softmax_weights(c, alpha)
    active = np.where(w > 1e-11)[0]
    if active.size > 120:
        active = active[np.argsort(-w[active])[:120]]
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=int)

    grad = np.zeros(int(n), dtype=float)
    for mv in active:
        m = int(mv)
        weight = float(w[m])
        if weight <= 0.0:
            continue
        grad += weight * deriv_lag_dx_shift(x, n, n - 1 - m)

    if mu != 0.0:
        grad += 2.0 * mu * (float(np.sum(x)) - 0.5 * float(n))
    return grad


def run_lbfgs(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_value(h, n)
    bounds = [(0.0, 1.0)] * int(n)

    for item in configs:
        if time.time() > deadline - 4.0:
            break
        if len(item) == 2:
            alpha, mu, maxiter = float(item[0]), float(item[1]), 150
        else:
            alpha, mu, maxiter = float(item[0]), float(item[1]), int(item[2])

        try:
            res = minimize(
                smooth_objective,
                best_h,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                jac=smooth_grad,
                options={
                    "maxiter": maxiter,
                    "maxfun": int(max(200, 4 * maxiter)),
                    "ftol": 1e-14,
                    "gtol": 1e-10,
                    "maxls": 20,
                },
            )
            h2 = project_feasible(res.x, n)
            c2 = c5_value(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    return best_h, best_c


def sum_eq_fun(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_eq_jac(x, n):
    return np.ones_like(np.asarray(x, dtype=float), dtype=float)


def slsqp_objective(x, n, alpha):
    return smooth_objective(x, n, alpha, 0.0)


def slsqp_grad(x, n, alpha):
    return smooth_grad(x, n, alpha, 0.0)


def slsqp_polish(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_value(h, n)
    bounds = [(0.0, 1.0)] * int(n)
    cons = [
        {
            "type": "eq",
            "fun": sum_eq_fun,
            "jac": sum_eq_jac,
            "args": (n,),
        }
    ]

    for alpha, maxiter in configs:
        if time.time() > deadline - 4.0:
            break
        try:
            res = minimize(
                slsqp_objective,
                best_h,
                args=(n, alpha),
                method="SLSQP",
                bounds=bounds,
                jac=slsqp_grad,
                constraints=cons,
                options={"maxiter": int(maxiter), "ftol": 1e-14},
            )
            h2 = project_feasible(res.x, n)
            c2 = c5_value(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    return best_h, best_c


def sqp_direction_lp(h, n, c_vec, active):
    active = np.asarray(active, dtype=int)
    if active.size == 0:
        return None
    nvar = int(n) + 1
    obj = np.zeros(nvar, dtype=float)
    obj[-1] = 1.0

    aub = np.zeros((int(active.size), nvar), dtype=float)
    bub = np.zeros(int(active.size), dtype=float)

    for r, mv in enumerate(active):
        m = int(mv)
        shift = n - 1 - m
        g = deriv_lag_dx_shift(h, n, shift)
        aub[r, 0:n] = g
        aub[r, n] = -1.0
        bub[r] = -float(c_vec[m])

    aeq = np.zeros((1, nvar), dtype=float)
    aeq[0, 0:n] = 1.0
    beq = np.zeros(1, dtype=float)
    bounds = [(-1.0, 1.0)] * int(n) + [(-2.0, 2.0)]

    try:
        res = linprog(
            obj,
            A_ub=aub,
            b_ub=bub,
            A_eq=aeq,
            b_eq=beq,
            bounds=bounds,
            method="highs",
        )
        if res.success:
            d = np.asarray(res.x[:n], dtype=float)
            if np.all(np.isfinite(d)):
                return d
    except Exception:
        pass

    return None


def sqp_line_search(h, d, n, C, step_scale=0.05):
    d = np.asarray(d, dtype=float)
    max_abs = float(np.max(np.abs(d)))
    if max_abs < 1e-5:
        return None, None, None

    d = d * (float(step_scale) / max_abs)

    pos = d > 1e-14
    neg = d < -1e-14
    alpha_max = 1.0

    if np.any(pos):
        room = (1.0 - h[pos]) / d[pos]
        alpha_max = min(alpha_max, float(np.min(room)))
    if np.any(neg):
        room = h[neg] / (-d[neg])
        alpha_max = min(alpha_max, float(np.min(room)))

    if alpha_max <= 1e-12:
        return None, None, None

    alpha = min(1.0, max(1e-6, alpha_max))

    for _ in range(35):
        h_new = h + alpha * d
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_vec_new = c5_vector(h_new, n)
        C_new = float(np.max(c_vec_new))
        if C_new < C - 1e-14:
            return h_new, C_new, c_vec_new
        alpha *= 0.5
        if alpha <= 1e-12:
            break

    return None, None, None


def sqp_optimize_lp(h, n, deadline, max_iter=3000, active_frac=0.004, max_active=100, step_scale=0.05):
    h = project_feasible(h, n)
    c_vec = c5_vector(h, n)
    C = float(np.max(c_vec))
    best_h = h.copy()
    best_c = C

    no_improve = 0
    cur_frac = float(active_frac)
    cur_active_max = int(max_active)

    for _ in range(int(max_iter)):
        if time.time() >= deadline:
            break

        thresh = C - max(1e-9, C * cur_frac)
        active = np.where(c_vec >= thresh)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c_vec))], dtype=int)
        if active.size > cur_active_max:
            active = active[np.argsort(-c_vec[active])[:cur_active_max]]

        d = sqp_direction_lp(h, n, c_vec, active)
        if d is None or float(np.max(np.abs(d))) < 1e-5:
            no_improve += 1
            if no_improve % 5 == 0:
                cur_active_max = max(8, cur_active_max // 2)
                cur_frac *= 0.6
            if no_improve > 80:
                break
            continue

        h_new, C_new, c_vec_new = sqp_line_search(h, d, n, C, step_scale=step_scale)
        if h_new is not None:
            h = h_new
            c_vec = c_vec_new
            C = C_new
            best_h = h.copy()
            best_c = C
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                cur_active_max = max(8, cur_active_max // 2)
                cur_frac *= 0.6
            if no_improve > 80:
                break

    return best_h, best_c


def local_search_pairs(h, n, deadline, steps=None, max_candidates=24, alpha=2000.0):
    h = project_feasible(h, n)
    c_vec = c5_vector(h, n)
    C = float(np.max(c_vec))
    best_h = h.copy()
    best_c = C

    if steps is None:
        steps = [
            0.02, 0.01, 0.005, 0.002, 0.001,
            0.0005, 0.0002, 0.0001, 5e-05, 2e-05
        ]

    for base_step in steps:
        if time.time() >= deadline:
            break

        improved = True
        while improved and time.time() < deadline - 2.0:
            improved = False

            w = softmax_weights(c_vec, alpha)
            maxw = float(np.max(w))
            active = np.where(w > max(1e-12, maxw * 1e-5))[0]
            if active.size > 30:
                active = active[np.argsort(-w[active])[:30]]
            if active.size == 0:
                active = np.array([int(np.argmax(c_vec))], dtype=int)

            grad = np.zeros(int(n), dtype=float)
            for mv in active:
                m = int(mv)
                weight = float(w[m])
                if weight <= 0.0:
                    continue
                grad += weight * deriv_lag_dx_shift(h, n, n - 1 - m)

            inc = np.argsort(grad)[:max_candidates]
            dec = np.argsort(-grad)[:max_candidates]

            cand_list = []
            for idx in list(inc) + list(dec):
                idx = int(idx)
                if idx not in cand_list:
                    cand_list.append(idx)

            rows = []
            for idx in cand_list:
                rows.append(grad_row_dx(h, n, idx))

            row_pos = {}
            for pp, idx in enumerate(cand_list):
                row_pos[idx] = pp

            found = False
            for ii in inc:
                if found:
                    break
                ii = int(ii)
                if h[ii] >= 1.0 - 1e-10:
                    continue
                ri = rows[row_pos[ii]]

                for jj in dec:
                    if found:
                        break
                    jj = int(jj)
                    if ii == jj:
                        continue
                    if h[jj] <= 1e-10:
                        continue

                    rj = rows[row_pos[jj]]
                    delta = ri - rj

                    pos_idx = np.where(delta > 1e-12)[0]
                    if pos_idx.size > 0:
                        room = C - c_vec[pos_idx]
                        ok = room > 1e-14
                        if not np.any(ok):
                            continue
                        lin_step = float(np.min(room[ok] / delta[pos_idx[ok]]))
                        trial = min(float(base_step), 0.95 * lin_step)
                    else:
                        trial = float(base_step)

                    bound = min(1.0 - float(h[ii]), float(h[jj]))
                    if bound <= 1e-10:
                        continue
                    trial = min(trial, bound * 0.999999)
                    if trial <= 1e-9:
                        continue

                    newC_approx = float(np.max(c_vec + trial * delta))
                    if newC_approx < C - 1e-13:
                        h[ii] += trial
                        h[jj] -= trial
                        c_new = c5_vector(h, n)
                        C_new = float(np.max(c_new))
                        if C_new < C - 1e-15:
                            C = C_new
                            c_vec = c_new
                            best_h = h.copy()
                            best_c = C
                            improved = True
                            found = True
                        else:
                            h[ii] -= trial
                            h[jj] += trial

    return best_h, best_c


def make_default_h(n):
    x = np.linspace(0.0, 2.0, int(n))
    h = np.full(int(n), 0.5, dtype=float)
    h += 0.35 * np.cos(np.pi * x)
    h += 0.20 * np.sin(2.0 * np.pi * x)
    h += 0.08 * np.sin(4.0 * np.pi * x + 0.7)
    return np.clip(h, 0.0, 1.0)


def upsample_candidate(h, n, best_c, deadline):
    n_hi = 1000
    x_old = np.linspace(0.0, 2.0, int(n))
    x_new = np.linspace(0.0, 2.0, n_hi)
    h_hi = np.interp(x_new, x_old, h)
    h_hi = project_feasible(h_hi, n_hi)
    c_hi = c5_value(h_hi, n_hi)

    if time.time() < deadline - 40.0:
        t_lp = min(time.time() + 90.0, deadline - 35.0)
        if time.time() < t_lp - 4.0:
            h_hi, c_hi = sqp_optimize_lp(
                h_hi, n_hi, t_lp,
                max_iter=2500,
                active_frac=0.003,
                max_active=110,
                step_scale=0.03,
            )

    if time.time() < deadline - 12.0:
        t_ls = min(time.time() + 35.0, deadline - 8.0)
        if time.time() < t_ls - 2.0:
            h_hi, c_hi = local_search_pairs(
                h_hi, n_hi, t_ls,
                steps=[0.005, 0.002, 0.001, 0.0005, 0.0002],
                max_candidates=20,
            )

    h_hi = project_feasible(h_hi, n_hi)
    c_hi = c5_value(h_hi, n_hi)
    return h_hi, c_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.default_rng(seed)

    try:
        h_raw = np.asarray(initial_h_values, dtype=float).ravel()
        if h_raw.size < 10:
            raise ValueError
    except Exception:
        h_raw = None

    if h_raw is None:
        h_start = make_default_h(500)
    else:
        n_raw = int(h_raw.size)
        if n_raw == 500:
            h_start = h_raw
        else:
            x_old = np.linspace(0.0, 2.0, n_raw)
            h_start = np.interp(np.linspace(0.0, 2.0, 500), x_old, h_raw)

    n = int(len(h_start))
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = c5_value(h, n)

    t_quick = min(time.time() + 20.0, deadline - 650.0)
    if time.time() < t_quick - 2.0:
        hh, cc = local_search_pairs(
            best_h, n, t_quick,
            steps=[0.005, 0.002, 0.001, 0.0005],
            max_candidates=18,
        )
        if cc < best_c:
            best_h, best_c = hh, cc

    t_lb = min(time.time() + 55.0, deadline - 520.0)
    if time.time() < t_lb - 3.0:
        best_h, best_c = run_lbfgs(
            best_h, n, t_lb,
            configs=[
                (100.0, 5000.0, 110),
                (500.0, 20000.0, 110),
                (2000.0, 60000.0, 70),
            ],
        )

    t_lp1 = min(time.time() + 360.0, deadline - 300.0)
    if time.time() < t_lp1 - 5.0:
        best_h, best_c = sqp_optimize_lp(
            best_h, n, t_lp1,
            max_iter=8000,
            active_frac=0.004,
            max_active=110,
            step_scale=0.05,
        )

    t_ls1 = min(time.time() + 120.0, deadline - 210.0)
    if time.time() < t_ls1 - 4.0:
        best_h, best_c = local_search_pairs(
            best_h, n, t_ls1,
            steps=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05],
            max_candidates=24,
        )

    t_lp2 = min(time.time() + 150.0, deadline - 140.0)
    if time.time() < t_lp2 - 4.0:
        best_h, best_c = sqp_optimize_lp(
            best_h, n, t_lp2,
            max_iter=5000,
            active_frac=0.002,
            max_active=80,
            step_scale=0.03,
        )

    t_ls2 = min(time.time() + 85.0, deadline - 90.0)
    if time.time() < t_ls2 - 3.0:
        best_h, best_c = local_search_pairs(
            best_h, n, t_ls2,
            steps=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05],
            max_candidates=24,
        )

    if n <= 600 and time.time() < deadline - 45.0:
        t_sl = min(time.time() + 45.0, deadline - 12.0)
        if time.time() < t_sl - 3.0:
            best_h, best_c = slsqp_polish(
                best_h, n, t_sl,
                configs=[(500.0, 70), (1200.0, 55)],
            )

    if len(best_h) == 500 and time.time() < deadline - 135.0:
        t_hi = min(time.time() + 120.0, deadline - 45.0)
        if time.time() < t_hi - 5.0:
            h_hi, c_hi = upsample_candidate(best_h, len(best_h), best_c, t_hi)
            if c_hi < best_c:
                best_c = c_hi
                best_h = h_hi

    if len(best_h) == 500 and time.time() < deadline - 75.0:
        t_lo = min(time.time() + 65.0, deadline - 20.0)
        if time.time() < t_lo - 5.0:
            h_lo, c_lo = downsample_candidate(best_h, len(best_h), t_lo)
            if c_lo < best_c:
                best_c = c_lo
                best_h = h_lo

    n_final = int(len(best_h))
    if time.time() < deadline - 6.0:
        t_fin = max(time.time() + 1.0, min(time.time() + 25.0, deadline - 2.0))
        if time.time() < t_fin - 2.0:
            best_h, best_c = local_search_pairs(
                best_h, n_final, t_fin,
                steps=[0.001, 0.0005, 0.0002, 0.0001, 5e-05],
                max_candidates=18,
            )

    best_h = project_feasible(best_h, n_final)
    best_c = c5_value(best_h, n_final)

    return best_h, float(best_c), int(n_final)
```
