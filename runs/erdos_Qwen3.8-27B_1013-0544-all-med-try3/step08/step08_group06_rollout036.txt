def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_linf(x, n, alpha, beta):
    """Smooth max with sum penalty"""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_shift = shifted.max()
    if max_shift > 500:
        logsumexp = max_shift + np.log(np.sum(np.exp(shifted - max_shift)))
    else:
        logsumexp = np.log(np.sum(np.exp(shifted)))
    smooth_max = m + logsumexp / alpha
    sum_diff = np.sum(x) - n / 2.0
    return smooth_max + beta * sum_diff * sum_diff

    h = project_feasible(h.copy(), n)

    best_c = compute_c5(best_h, n)
    
    bounds = [(0.0, 1.0)] * n
    iters_used = 0
    
    for a, b in zip(alphas, betas):
        if time.time() > deadline or iters_used >= max_iter_total:
            break
        mi = max(20, (max_iter_total - iters_used) // len(alphas))
        try:
            res = minimize(obj_linf, best_h, args=(n, a, b), method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': mi, 'ftol': 1e-16, 'maxfun': mi * 10})
            iters_used += mi
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        # Add/subtract from the element with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if abs(diff) > 1e-14:
                room = 1.0 - h
                tot = room.sum()
                if tot > 1e-14:
                    h += (diff / tot) * room
        else:
            idx = np.argmin(h)
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if abs(diff) > 1e-14:
                tot = h.sum()
                if tot > 1e-14:
                    h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    if max_sh > 500:
        val = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
    else:
        val = np.log(np.sum(np.exp(shifted)))
    return float(m + val / alpha)

def obj_penalized(x, n, alpha, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    if max_sh > 500:
        val = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
    else:
        val = np.log(np.sum(np.exp(shifted)))
    smooth_max = m + val / alpha
    sum_diff = float(np.sum(x) - n / 2.0)
    return float(smooth_max + beta * sum_diff * sum_diff)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = np.argmin(h)
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) > 1e-13:
            if diff > 0:
                room = 1.0 - h
                tot = room.sum()
                if tot > 1e-13:
                    h += (diff / tot) * room
            else:
                tot = h.sum()
                if tot > 1e-13:
                    h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    if max_sh > 500:
        val = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
    else:
        val = np.log(np.sum(np.exp(shifted)))
    return float(m + val / alpha)


def obj_penalized(x, n, alpha, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    if max_sh > 500:
        val = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
    else:
        val = np.log(np.sum(np.exp(shifted)))
    smooth_max = m + val / alpha
    sd = float(np.sum(x) - n / 2.0)
    return float(smooth_max + beta * sd * sd)


def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def _sum_con_jac(x, n):
    return np.ones(n)


def optimize_lbfgs(h, n, deadline, alpha_schedule=None, beta_schedule=None):
    if alpha_schedule is None:
        alpha_schedule = [300, 1000, 4000, 15000, 50000, 150000, 500000, 1500000, 5000000]
    if beta_schedule is None:
        beta_schedule = [5, 20, 100, 500, 2000, 10000, 50000, 100000, 200000]
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    n_stages = len(alpha_schedule)
    for i_stage in range(n_stages):
        if time.time() > deadline:
            break
        a = alpha_schedule[i_stage]
        b = beta_schedule[i_stage]
        try:
            opts = {'maxiter': 300, 'ftol': 1e-18, 'maxfun': 3000}
            res = minimize(obj_penalized, best_h, args=(n, a, b), method='L-BFGS-B',
                           bounds=bounds, options=opts)
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c


def optimize_slsqp(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [5000, 25000, 100000, 400000, 1500000, 5000000]
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n, a), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-17, 'iprint': -1})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c


def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(1000):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_lo = i + 1
                    j_hi = n
                else:
                    j_lo = max(i + 1, i - radius + 1)
                    j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    hi_v = h[i] + d
                    hj_v = h[j] - d
                    if 0.0 <= hi_v <= 1.0 and 0.0 <= hj_v <= 1.0:
                        oi, oj = h[i], h[j]
                        h[i], h[j] = hi_v, hj_v
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = oi, oj
                    hi_v = h[i] - d
                    hj_v = h[j] + d
                    if 0.0 <= hi_v <= 1.0 and 0.0 <= hj_v <= 1.0:
                        oi, oj = h[i], h[j]
                        h[i], h[j] = hi_v, hj_v
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = oi, oj
            if not improved:
                break
    return h, best_c


def gradient_flow(h, n, deadline, n_steps=50000, lr_init=0.001, alpha=50000):
    """Projected gradient flow on the smooth max surrogate."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if step % 1000 == 0:
            c_now = compute_c5(h, n)
            if c_now < best_c - 1e-15:
                best_c = c_now
                best_h = h.copy()
            lr *= 0.95
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        max_sh = shifted.max()
        if max_sh > 500:
            exp_shifted = np.exp(shifted - max_sh)
            weights = exp_shifted / exp_shifted.sum()
        else:
            exp_shifted = np.exp(shifted)
            weights = exp_shifted / exp_shifted.sum()
        grad = np.zeros(n)
        for k in range(len(weights)):
            w = weights[k]
            if w < 1e-12:
                continue
            shift = k - (n - 1)
            if shift >= 0:
                for i in range(n - shift):
                    grad[i] += w * (1.0 - h[i + shift]) * dx
                    grad[i + shift] -= w * h[i] * dx
            else:
                s = -shift
                for i in range(n - s):
                    grad[i] -= w * h[i + s] * dx
                    grad[i + s] += w * (1.0 - h[i]) * dx
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c, best_h = c_final, h
    return best_h, best_c


def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    t_remaining = deadline - time.time()
    if t_remaining < 5:
        return best_h, best_c

    lbfgs_deadline = time.time() + t_remaining * 0.35
    h1, c1 = optimize_lbfgs(best_h, n, lbfgs_deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1

    if n <= 300 and time.time() < deadline - 20:
        slsqp_deadline = time.time() + (deadline - 20 - time.time()) * 0.35
        h2, c2 = optimize_slsqp(best_h, n, slsqp_deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if time.time() < deadline - 2:
        if n <= 300:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.02, 0.01, 0.005, 0.002, 0.001], radius=50)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def fine_refine(h, n, deadline, radius=None):
    if n <= 300:
        return pair_search(h, n, deadline, [0.001, 0.0005, 0.0002, 0.0001, 5e-05])
    else:
        return pair_search(h, n, deadline, [0.001, 0.0005, 0.0002, 0.0001], radius=radius or 40)
