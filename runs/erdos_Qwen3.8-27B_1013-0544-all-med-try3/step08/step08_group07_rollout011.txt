    dx = 2.0 / n
    g = np.correlate(h, 1-h, mode='full').astype(float)
    best_h = h.copy()
    best_c = g.max() * dx
    h = h.copy()
    
    # Precompute for fast delta computation
    # For shift s (in -n+1..n-1), the delta at index i is:
    # (1-h[i+s]) - h[i-s] (with 0 for out-of-bounds)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Choose delta
        max_d = min(h[i], 1-h[j])  # can decrease h[i] by this
        min_d = min(1-h[i], h[j])  # can increase h[i] by this
        delta = rng.uniform(-min_d, max_d) * 0.1  # small step
        if abs(delta) < 1e-10:
            continue
            
        # Compute delta_g for all shifts
        # s_k = k - (n-1) for k in 0..2n-2
        s_vals = np.arange(-(n-1), n)  # n*2-1 values
        # For index i with shift s: (1-h[i+s]) if i+s in [0,n-1], else 0
        # minus h[i-s] if i-s in [0,n-1], else 0
        
        delta_g = np.zeros(2*n-1)
        
        # h[i] changes by +delta
        # As "l" in g: contribution delta*(1-h[i+s]) for s where i+s in [0,n-1]
        # As "l+s" in g: contribution -delta*h[i-s] for s where i-s in [0,n-1]

        s_end_i = n - 1 - i
        if s_end_i >= s_start_i:
            delta_g[s_start_i + (n-1):s_end_i + (n-1) + 1] += delta * (1 - h[i + s_vals[s_start_i:s_end_i+1]])
        
        # For h[i] as second factor (l+s=i, so l=i-s): valid s range: i-(n-1) to i
        s_start_i2 = i - (n-1)
        s_end_i2 = i
        if s_end_i2 >= s_start_i2:
            delta_g[s_start_i2 + (n-1):s_end_i2 + (n-1) + 1] -= delta * h[i - s_vals[s_start_i2:s_end_i2+1]]

        s_end_j = n - 1 - j
        if s_end_j >= s_start_j:

    return best_h, best_c

def greedy_shift_reduction(h, n, deadline, rng, max_iter=2000000):

def slsqp_smooth(h, n, alpha, deadline, maxiter=500):
    """SLSQP on smooth max approximation."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + np.log(np.exp(shifted - sm).sum()) / alpha

    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w = w / w.sum()
        # grad of g[k] wrt x[i]
        grad = np.zeros(n)
        for i in range(n):
            for k in range(2 * n - 1):
                s = k - (n - 1)
                val = 0.0
                if 0 <= i + s < n:
                    val += 1.0 - x[i + s]
                if 0 <= i - s < n:
                    val -= x[i - s]
                grad[i] += w[k] * alpha * dx * val
        return grad

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

            # while keeping sum(h) = target
            # g[k_star] contains: h[i]*(1-h[i+s_star]) + h[i+s_star]*(1-h[i]) + ...
            # Let a = h[i+s_star] (if valid), b = h[i-s_star] (if valid)
            # Terms: h[i]*(1-a) + b*(1-h[i]) = h[i] - a*h[i] + b - b*h[i] = b + h[i]*(1-a-b)
            # To minimize: if 1-a-b > 0, set h[i]=0; if 1-a-b < 0, set h[i]=1
            # But we need sum(h)=target, so we adjust another coordinate.
            
            # Simpler: try small adjustments
            old_hi = h[i]
            if old_hi <= 0.001 or old_hi >= 0.999:
                continue
            # Try moving to 0 or 1 (binary push)
            for target_val in [0.0, 1.0]:
                diff = target_val - old_hi
                if abs(diff) < 1e-6:
                    continue
                # Need to compensate on another coordinate
                comp_val = old_hi - diff  # other coordinate changes by -diff... wait
                # h[i] changes by (target_val - old_hi), some h[j] changes by -(target_val - old_hi)
                comp_change = -(target_val - old_hi)
                
                # Find j to compensate: prefer one where h[j]+comp_change stays in [0,1]
                # and that reduces g[k_star]
                best_j = -1
                best_new_max = g_max
                # Try a few candidates
                candidates = []
                if comp_change > 0:
                    # Need to increase some h[j], prefer where h[j] is low
                    candidates = np.argsort(h)[:min(20, n//5)]
                else:
                    candidates = np.argsort(h)[::-1][:min(20, n//5)]
                
                for j in candidates:
                    if j == i:
                        continue
                    hj_new = h[j] + comp_change
                    if hj_new < -1e-8 or hj_new > 1 + 1e-8:
                        continue
                    dg = compute_delta_g(h, n, i, j, target_val - old_hi)
                    new_g = g + dg
                    nm = new_g.max()
                    if nm < best_new_max:
                        best_new_max = nm
                        best_j = j
                
                if best_j >= 0 and best_new_max < g_max - 1e-13:
                    h[i] = target_val
                    h[best_j] = np.clip(h[best_j] + comp_change, 0, 1)
                    g = g + compute_delta_g(h - np.where(np.arange(n)==i, target_val-old_hi, 0) - np.where(np.arange(n)==best_j, comp_change, 0), n, i, best_j, 0)
                    # Recompute to be safe
                    g = np.correlate(h, 1.0 - h, mode='full').astype(float)
                    g_max = g.max()
                    improved = True
                    c = g_max * dx
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                    break
            if improved:
                # Rebalance
                diff_sum = target - h.sum()
                if abs(diff_sum) > 1e-10:
                    h = project_feasible(h, n)
                    g = np.correlate(h, 1.0 - h, mode='full').astype(float)
                    g_max = g.max()

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Fast perturb-project (40% of budget)
    p1 = t0 + 0.40 * (budget_s - 5)
    h_curr = best_h.copy()
    h_curr, c = fast_perturb_project(h_curr, n, p1, rng, base_delta=0.06)
    if c < best_c:
        best_c = c
        best_h = h_curr.copy()

    # Phase 2: Greedy shift reduction (25% of budget)
    p2 = t0 + 0.65 * (budget_s - 5)
    h_curr = best_h.copy()
    h_curr, c = greedy_shift_reduction(h_curr, n, p2, rng)
    if c < best_c:
        best_c = c
        best_h = h_curr.copy()

    # Phase 3: Fine perturb-project (20% of budget)
    p3 = t0 + 0.85 * (budget_s - 5)
    h_curr = best_h.copy()
    h_curr, c = fast_perturb_project(h_curr, n, p3, rng, base_delta=0.02, anneal=True)
    if c < best_c:
        best_c = c
        best_h = h_curr.copy()

    # Phase 4: Multi-resolution attempt
    if time.time() < dl - 100:
        h_new, c_new, n_new = multi_resolution_optimize(best_h, best_c, n, dl - 80, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            n = n_new

    # Phase 5: Final fine polish
    if time.time() < dl - 20:
        h_curr = best_h.copy()
        h_curr, c = fast_perturb_project(h_curr, n, dl - 10, rng, base_delta=0.005)
        if c < best_c:
            best_c = c
            best_h = h_curr.copy()

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
