from scipy.optimize import minimize
import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

            if tot < 1e-14:

    m = c.max()

    max_s = shifted.max()
    w = np.exp(shifted - max_s)

def _obj_lsq(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    return float(m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha)

def _obj_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    val = m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_pen(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _con_eq(x, n):
    return float(x.sum() - n / 2.0)

    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=200):

            h_new = project_feasible(res.x, n)

def pair_search(h, n, deadline, deltas=None, radius=None, algo='forward'):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i]+=d, h[j]-=d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

                        c = compute_c5(h, n)

                        else:

                        c = compute_c5(h, n)

                        else:

        k = k_star - (n - 1)
        dx = 2.0 / n
        g = np.zeros(n)
        if k >= 0:

            g[0:nv] = (1.0 - h[k:n]) * dx
            g[k:n] += -h[0:nv] * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:n] = (1.0 - h[0:nv]) * dx
            g[0:nv] += -h[kk:n] * dx
        # Projected subgradient step
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = project_feasible(h - lr * 0.5 * g, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                lr *= 0.9
                if lr < 1e-8:
                    break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj_lsq(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    return float(m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha)

def _obj_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    val = m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_pen(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _con_eq(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas_mus = [(20, 5.0), (100, 10.0), (500, 50.0), (2000, 200.0),
                  (10000, 1000.0), (50000, 5000.0), (200000, 10000.0),
                  (1000000, 50000.0), (5000000, 200000.0)]
    for alpha, mu in alphas_mus:
        if time.time() > deadline - 3:
            break
        t_rem = deadline - time.time() - 3
        if t_rem < 5:
            break
        try:
            res = minimize(_obj_pen, h.copy(), jac=_grad_pen,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 8,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac, 'args': (n,)}]
    t_rem = min(deadline - time.time() - 3, 150)
    if t_rem < 10:
        return h, best_c
    try:
        res = minimize(_obj_lsq, h.copy(), args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
        if time.time() > deadline:
            break
    return h, best_c

def grad_guided_pair(h, n, deadline, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000]
    for alpha in alphas:
        for ss in step_sizes:
            while time.time() < deadline:
                improved = False
                g = grad_smooth_max(h, n, alpha)
                for i in range(n):
                    if time.time() > deadline:
                        return h, best_c
                    for j in range(n):
                        if i == j:
                            continue
                        move = -ss * (g[i] - g[j])
                        if abs(move) < 1e-12:
                            continue
                        if 0 <= h[i] + move <= 1 and 0 <= h[j] - move <= 1:
                            h[i] += move
                            h[j] -= move
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= move
                                h[j] += move
                if not improved:
                    break
    return h, best_c

def subgradient_method(h, n, deadline, n_iters=10000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.05
    beta = 0.9
    v = np.zeros(n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        k_star = int(np.argmax(c))
        k = k_star - (n - 1)
        dx = 2.0 / n
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[0:nv] = (1.0 - h[k:n]) * dx
            g[k:n] += -h[0:nv] * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:n] = (1.0 - h[0:nv]) * dx
            g[0:nv] += -h[kk:n] * dx
        v = beta * v + (1.0 - beta) * g
        h_new = project_feasible(h - lr * v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            h_try = project_feasible(h - 0.5 * lr * v, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
            else:
                lr *= 0.95
                if lr < 1e-10:
                    break
    return best_h, best_c

def simulated_annealing_pairs(h, n, deadline, rng, n_steps_max=500000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    T_init = 0.01
    T_end = 1e-6
    n_steps = 0
    for step in range(n_steps_max):
        if time.time() > deadline:
            break
        T = T_init * (T_end / T_init) ** (step / max(n_steps_max, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.02, 0.02)
        if abs(d) < 1e-8:
            continue
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d
            h[j] -= d
            c = compute_c5(h, n)
            delta = c - best_c
            if delta <= 0:
                best_c = c
                best_h = h.copy()
            elif rng.rand() < np.exp(-delta / T):
                pass
            else:
                h[i] -= d
                h[j] += d
        n_steps += 1
    return best_h, best_c

def random_perturb_optimize(h, n, deadline, rng, n_tries=4, noise_std=0.04):

            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            t_sub = min(60, deadline - time.time() - 70)
            if t_sub > 25:
                h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + t_sub * 0.6, max_iters=150)
                if time.time() < deadline - 50:
                    h_t, c_t = slsqp_optimize(h_t, n_try,
                         min(time.time() + 30, deadline - 40), alpha=500000, maxiter=800)
                if c_t < best_c:
                    best_c = c_t
                    best_h = h_t
                    n = n_try

    # Phase 11: Final SLSQP + pair search on best
    if time.time() < deadline - 15:
        nf = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, nf, min(time.time() + 30, deadline - 10),
                                   alpha=5000000, maxiter=1500)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = pair_search(best_h, nf, deadline - 2,
                                     deltas=[0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5],
                                     radius=60)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
