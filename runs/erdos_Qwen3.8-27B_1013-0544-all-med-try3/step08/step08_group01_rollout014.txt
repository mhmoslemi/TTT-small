def project_feasible(h, n):

    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)

        diff = h.sum() - target
        if abs(diff) < 1e-14:

        # Adjust: add -diff/n to all, then clip, iterate
        h -= diff / n
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
