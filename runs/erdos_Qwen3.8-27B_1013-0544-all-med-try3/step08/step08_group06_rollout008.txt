```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            if not mask.any():
                break
            h[mask] += (diff / room[mask].sum()) * room[mask]
        else:
            mask = h > 1e-10
            if not mask.any():
                break
            h[mask] += (diff / h[mask].sum()) * h[mask]
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    out = np.exp(sh)
    sm = out.sum()
    return float(m + np.log(sm) / alpha)

        offset = k - (n - 1)
        # c[k] = sum over valid i: h[i+offset] * (1-h[i])
```

Actually, SLSQP with 200 variables will be slow with numerical gradients. Let me provide the analytical gradient.

The softmax objective:
f(h) = m + (1/alpha) * log(sum_k exp(alpha * (c_k - m)))

where c_k = dx * sum_i h[(i+k-(n-1))] * (1-h[i]) for valid i.

Let me denote offset_k = k - (n-1) for k = 0, ..., 2n-2.

c_k = dx * S_k where S_k = sum_{i in valid_k} h[i+offset_k] * (1-h[i])

∂c_k/∂h[j] = dx * [(1-h[j']) * 1_{j appears as j' in h} + (-h[j'']) * 1_{j appears in (1-h)}]

More precisely:
- If j = i + offset_k for some valid i (i.e., j - offset_k >= 0 and j - offset_k <= n-1 and j <= n-1 and j >= 0):
  Contribution from h[j] in the first factor: dx * (1 - h[j - offset_k])
- If j = i for some valid i (i.e., j - offset_k >= 0 and j - offset_k <= n-1):
  Contribution from (1-h[j]) in the second factor: -dx * h[j + offset_k]

So:
∂c_k/∂h[j] = dx * [(1-h[j-offset_k]) * I[0 <= j-offset_k < n, 0 <= j < n] - h[j+offset_k] * I[0 <= j < n, 0 <= j+offset_k < n]]

Where I[condition] is 1 if the condition holds, 0 otherwise.

For the softmax:
Let p_k = exp(alpha*(c_k - m)) / sum_l exp(alpha*(c_l - m))

∂f/∂h[j] = sum_k p_k * ∂c_k/∂h[j]

This is the same as the expected subgradient where we weight by p_k.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            if not mask.any():
                break
            h[mask] += (diff / room[mask].sum()) * room[mask]
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] += (diff / h[mask].sum()) * h[mask]
    return np.clip(h, 0.0, 1.0)

def _sum_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(len(x))

def slsqp_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    c_vals = c * dx
    m = c_vals.max()
    sh = alpha * (c_vals - m)
    exp_sh = np.exp(sh)
    ss = exp_sh.sum()
    val = m + np.log(ss) / alpha
    if ss < 1e-300:
        return float(val), np.zeros(n)
    p = exp_sh / ss
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        valid_i = np.arange(i_lo, i_hi)
        h_idx = valid_i + offset
        dh_idx = valid_i
        grad[h_idx] += (1.0 - x[dh_idx]) * dx * p[k]
        grad[dh_idx] -= x[h_idx] * dx * p[k]
    return float(val), grad

def slsqp_round(h, n, alpha, maxiter=4000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    def obj(x):
        return slsqp_obj_and_grad(x, n, alpha)[0]
    def jac(x):
        return slsqp_obj_and_grad(x, n, alpha)[1]
    def obj_jac(x):
        v, g = slsqp_obj_and_grad(x, n, alpha)
        return v, g
    try:
        res = minimize(obj_jac, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=jac,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgrad_polish(h, n, steps=500, lr0=0.005, lr_end=0.0001):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    for s in range(steps):
        lr = lr0 * (lr_end / lr0) ** (s / max(1, steps - 1))
        c = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        m = c.max()
        binding = c > (m - 1e-12)
        n_bind = max(1, binding.sum())
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if not binding[k]:
                continue
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n, n - offset)
            valid_i = np.arange(i_lo, i_hi)
            h_idx = valid_i + offset
            dh_idx = valid_i
            grad[h_idx] += (1.0 - best_h[dh_idx]) * dx
            grad[dh_idx] -= best_h[h_idx] * dx
        grad /= n_bind
        h_new = best_h - lr * grad
        diff = n / 2.0 - h_new.sum()
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def targeted_local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    c_corr = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0 / n)
    binding_lags = np.where(c_corr > (c_corr.max() - 1e-10))[0]
    affected = set()
    for k in binding_lags:
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        for i in range(i_lo, i_hi + 1):
            affected.add(i)
            affected.add(i + offset)
    affected = list(affected)
    if len(affected) > 60:
        affected = list(range(n))
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(80):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            for idx_a in range(len(affected)):
                if time.time() > deadline:
                    break
                i = affected[idx_a]
                for idx_b in range(idx_a + 1, len(affected)):
                    j = affected[idx_b]
                    if best_h[i] + d <= 1.0 + 1e-15 and best_h[j] - d >= -1e-15:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= -1e-15 and best_h[j] + d <= 1.0 + 1e-15:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
            if not improved:
                break
    return best_h, best_c

def full_pair_search(h, n, deadline, deltas, radius=None):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(100):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                if radius is None:
                    j_range = range(i + 1, n)
                else:
                    j_lo = max(0, i - radius)
                    j_hi = min(n, i + radius + 1)
                    j_range = range(j_lo, j_hi)
                for j in j_range:
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 + 1e-15 and best_h[j] - d >= -1e-15:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
            if not improved:
                break
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    remaining = deadline - time.time()
    alphas = [500, 2000, 8000, 30000, 100000, 300000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_round(best_h, n, a, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    if time.time() < deadline - 10:
        best_h, best_c = subgrad_polish(best_h, n, steps=300, lr0=0.003, lr_end=0.00005)
    if n <= 200 and time.time() < deadline - 5:
        best_h, best_c = targeted_local_search(best_h, n, deadline - 3)
    elif time.time() < deadline - 5:
        best_h, best_c = full_pair_search(best_h, n, deadline - 3, [0.01, 0.005, 0.002], radius=25)
    if time.time() < deadline - 3:
        ht = slsqp_round(best_h, n, 500000, maxiter=4000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    if time.time() < deadline - 2:
        best_h, best_c = subgrad_polish(best_h, n, steps=200, lr0=0.001, lr_end=0.00002)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, n_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    cur_h = h.copy()
    cur_c = best_c
    t0 = 3.0
    tmin = 0.001
    total = n_iter
    for s in range(total):
        if time.time() > deadline:
            break
        T = t0 * (tmin / t0) ** (s / total)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = (rng.rand() - 0.5) * 0.15
        hi = cur_h[i] + d
        hj = cur_h[j] - d
        if hi < -0.01 or hi > 1.01 or hj < -0.01 or hj > 1.01:
            continue
        cur_h[i] = min(1.0, max(0.0, hi))
        cur_h[j] = min(1.0, max(0.0, hj))
        cur_h = project_feasible(cur_h, n)
        c_new = compute_c5(cur_h, n)
        delta = c_new - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = cur_h.copy()
        else:
            cur_h = best_h.copy()
            cur_c = best_c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        alloc1 = (deadline - t0) * 0.35
        h_opt, c_opt = full_optimize(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

        # Simulated annealing from the optimized solution
        if time.time() < deadline - 60:
            sa_deadline = min(deadline - 50, time.time() + 120)
            h_sa, c_sa = simulated_annealing(best_h, n_init, sa_deadline, rng, n_iter=30000)
            if c_sa < best_c:

                best_h = h_sa.copy()

            # Re-optimize the SA result
            if c_sa < best_c + 0.001 and time.time() < deadline - 40:

                if c_re < best_c:
                    best_c = c_re
                    best_h = h_re

        # Try different resolutions
        x_init = np.linspace(0, 2, n_init)
        for n_try in [100, 150, 300, 400]:
            if n_try == n_init:
                continue
            if time.time() > deadline - 50:
                break
            remaining = deadline - time.time() - 10
            alloc = min(100, remaining * 0.2)
            if alloc < 20:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            # Also try from current best
            h_interp2 = np.interp(x_new, np.linspace(0, 2, n_init), best_h)

            if c_opt1 < best_c:

                best_h = h_opt1
            if time.time() < deadline - 25:
                h_opt2, c_opt2 = full_optimize(h_interp2, n_try, time.time() + min(50, deadline - time.time() - 20), rng)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2

        # Random restarts from best
        n_best = len(best_h)
        if time.time() < deadline - 40:
            n_restarts = max(2, int((deadline - time.time() - 30) / 50))
            for r in range(n_restarts):
                if time.time() > deadline - 20:
                    break
                scale = 0.03 + 0.15 * (r / max(1, n_restarts))
                hp = best_h + rng.normal(0, scale, n_best)
                hp = project_feasible(hp, n_best)
                alloc = min(40, (deadline - time.time() - 15) * 0.25)

                h_opt, c_opt = full_optimize(hp, n_best, time.time() + alloc, rng)

                    best_h = h_opt

        best_h = project_feasible(np.full(n_init, 0.5), n_init)

    # Final fine polishing

                                              [0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = full_pair_search(best_h, n_final, deadline - 1,
                                              [0.005, 0.002, 0.001], radius=20)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, len(best_h)
```
