        grad_norm_sq = np.dot(grad, grad) * dx * dx

        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible_sum(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

                                     [0.01, 0.005, 0.002])

    if time.time() < t_end - 5:
        ht = slsqp_optimize(best_h, n, 500000, maxiter=2000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    if time.time() < t_end - 3:
        best_h, best_c = subgradient_optimize(best_h, n, min(time.time() + 10, t_end - 5))

    if n <= 250 and time.time() < t_end - 2:
        best_h, best_c = pair_search(best_h, n, t_end - 1,
                                     [0.005, 0.002, 0.001, 0.0005])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc1 = min(200, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_resolution(h_init, n_init, alloc1, rng)
        best_h, best_c = h_opt, c_opt

        # Try other resolutions
        x_init = np.linspace(0, 2, n_init)
        for n_try in [250, 150, 300]:
            if n_try == n_init or time.time() > deadline - 60:
                continue
            remaining = deadline - time.time()
            alloc = min(100, remaining * 0.2)
            if alloc < 20:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_resolution(h_interp, n_try, alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Random restarts from best solution
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        remaining = deadline - time.time() - 10
        n_restarts = max(2, int(remaining / 50))
        scales = [0.01, 0.03, 0.05, 0.08, 0.12, 0.15, 0.2, 0.25]
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 10) * 0.25)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_resolution(hp, n, alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Structured restart: try low-frequency perturbations
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        for freq in [1, 2, 3, 4, 5]:
            if time.time() > deadline - 15:
                break
            pert = np.sin(np.pi * freq * np.arange(n) / (n - 1))
            for amp in [0.05, 0.1, 0.15]:
                if time.time() > deadline - 12:
                    break
                hp = best_h + amp * pert
                hp = project_feasible(hp, n)
                alloc = min(30, (deadline - time.time() - 10) * 0.3)
                if alloc < 10:
                    break
                h_opt, c_opt = optimize_resolution(hp, n, alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Final polish on best solution
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Subgradient polish
        if time.time() < deadline - 8:
            best_h, best_c = subgradient_optimize(best_h, n, deadline - 8)
        # Final SLSQP
        if time.time() < deadline - 10:
            ht = slsqp_optimize(best_h, n, 1000000, maxiter=3000, ftol=1e-17)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
        # Fine pair search
        if n <= 250 and time.time() < deadline - 3:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.002, 0.001, 0.0005, 0.0002, 0.0001])
        elif time.time() < deadline - 3:
            # Restricted pair search for larger n
            best_h, best_c = pair_search_best_h_near(best_h, n, deadline - 1,
                                                     [0.001, 0.0005, 0.0002], radius=30)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)

def pair_search_best_h_near(h, n, t_end, deltas, radius=30):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for d in deltas:
        while time.time() < t_end:
            improved = False
            for i in range(n):
                if time.time() > t_end:
                    return best_h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = best_h[i] + d
                    hj = best_h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        best_h[i] = hi
                        best_h[j] = hj
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = hi - d
                            best_h[j] = hj + d
            if not improved:
                break
    return best_h, best_c
