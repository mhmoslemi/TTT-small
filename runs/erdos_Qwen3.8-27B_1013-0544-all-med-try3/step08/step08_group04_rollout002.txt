def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s <= 1e-15: break
            h += (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    # final exact correction
    diff = target - h.sum()
    if diff != 0:
       if diff>0:
          idxs = np.where(h < 1.0 - 1e-15)[0]
       else: idxs = np.where(h > 1e-15)[0]
       if len(idxs):
          # distribute evenly to avoid hitting bounds
          add = diff / len(idxs)
          h[idxs] += add
          h = np.clip(h,0,1)
          # one more adjust
          diff2 = target - h.sum()
          if diff2>0: idx=h<1; else idx=h>0
          if any: h[np.where(idx)[0][0]] += diff2
    return h

def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.ndim != 1 or h.shape[0] != n:
        h = np.resize(h, n)
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5, dtype=float)

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    for _ in range(120):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = float(room.sum())
            if s <= 1e-15:
                break
            h += (diff / s) * room
        else:
            s = float(h.sum())
            if s <= 1e-15:
                break
            h += (diff / s) * h
    h = np.clip(h, 0.0, 1.0)

    for _ in range(30):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:

        if idx.size == 0:
            break
        h[idx] += diff / float(idx.size)
        h = np.clip(h, 0.0, 1.0)

    return h


def enforce_sum_exact(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.ndim != 1 or h.shape[0] != n:
        h = np.resize(h, n)
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5, dtype=float)

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h[h < 1e-14] = 0.0
    h[(1.0 - h) < 1e-14] = 1.0

    total = float(h.sum())
    for _ in range(4):
        done = False
        for j in range(n):
            s = total - h[j]
            val = target - s
            if -1e-12 <= val <= 1.0 + 1e-12:
                h[j] = float(np.clip(val, 0.0, 1.0))
                done = True
                break
        if done:
            break
        h = project_feasible(h, n)
        total = float(h.sum())

    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        h = project_feasible(h, n)
    return h


def corr_grad(h, n, scale=1.0):
    h = np.asarray(h, dtype=float)
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=float)
    m0 = n - 1
    for m in range(L):
        k = m - m0
        row = G[m]
        row.fill(0.0)
        if k >= 0:
            row[k:] = 1.0 - h[:n - k]
            row[:n - k] -= h[k:]
        else:
            d = -k
            row[:d] = 1.0 - h[d:]
            row[d:] -= h[:d]
    if scale != 1.0:
        G *= scale
    return G


def smooth_fjac(h, n, alpha):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(c.max())

    if alpha <= 1e-9:
        idx = int(np.argmax(c))
        g = np.zeros_like(h)
        g[idx] = 1.0
        return m, g

    z = alpha * (c - m)
    zm = float(z.max())
    ez = np.exp(z - zm)
    s = float(ez.sum())
    if s <= 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = ez / s

    f = m + (zm + np.log(s)) / alpha
    G = corr_grad(h, n, dx)
    g = p @ G
    return float(f), g


def _slsqp_sum_fun(x, n):
    return float(x.sum()) - n / 2.0


def _slsqp_sum_jac(x, n):
    return np.ones(len(x), dtype=float)


def slsqp_one(h, n, alpha, maxiter=80):
    h = project_feasible(h, n)
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
    cons = [{"type": "eq", "fun": _slsqp_sum_fun, "jac": _slsqp_sum_jac, "args": (n,)}]

    try:
        res = minimize(
            smooth_fjac,
            h,
            args=(n, alpha),
            method="SLSQP",
            jac=True,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12},
        )
        hx = np.asarray(res.x, dtype=float)
        return project_feasible(hx, n)
    except Exception:
        return h


def soft_weights(c, alpha):
    m = float(c.max())
    if alpha <= 1e-9:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        return p
    z = alpha * (c - m)
    zm = float(z.max())
    ez = np.exp(z - zm)
    s = float(ez.sum())
    if s <= 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = ez / s
    return p


def first_order_direction(h, n, tau):
    h = np.asarray(h, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)
    if tau <= 1e-12:
        return np.zeros(n, dtype=float)

    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    G = corr_grad(h, n, dx)
    L = len(c)

    A_ub = np.empty((L, n + 1), dtype=float)
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c

    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[n] = 1.0

    A_eq = np.ones((1, n + 1), dtype=float)
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1, dtype=float)

    bounds = []
    eps = 1e-9
    for i in range(n):
        lo = max(-h[i] - eps, -tau + eps)
        hi = min(1.0 - h[i] + eps, tau - eps)
        if hi < lo:
            mid = 0.5 * (ho if False else (hi + lo))
            lo = hi = mid
        bounds.append((float(lo), float(hi)))
    bounds.append((-1.0, 2.0))

    try:
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
        if res.status not in (0, 2) or res.x is None:
            return np.zeros(n, dtype=float)
        return np.asarray(res.x[:n], dtype=float)
    except Exception:
        return np.zeros(n, dtype=float)


def line_objective(alpha, h, n, delta):
    x = h + alpha * delta
    x = np.clip(x, 0.0, 1.0)
    return compute_c5(x, n)


def line_search(h, n, delta, c_base):
    if delta is None or delta.size == 0:
        return h.copy(), c_base
    dmax = float(np.max(np.abs(delta)))
    if dmax < 1e-12:
        return h.copy(), c_base

    best_c = c_base
    best_a = 0.0

    for a in (1.0, 0.8, 0.6, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01):
        c = line_objective(a, h, n, delta)
        if c < best_c:
            best_c = c
            best_a = a

    if best_a > 0.0:
        lo = max(0.0, 0.35 * best_a)
        hi = min(1.0, 2.5 * best_a)
        if hi - lo < 1e-4:
            lo = max(0.0, best_a - 0.02)
            hi = min(1.0, best_a + 0.02)
        try:
            r = minimize_scalar(
                line_objective,
                args=(h, n, delta),
                bounds=(lo, hi),
                method="bounded",
                options={"xatol": 1e-5},
            )
            if r.fun < best_c:
                best_c = float(r.fun)
                best_a = float(r.x)
        except Exception:
            pass

    x = h + best_a * delta
    x = np.clip(x, 0.0, 1.0)
    best_c = compute_c5(x, n)
    return x, best_c


def slp_optimize(h, n, deadline, tau_start=0.12):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    tau = float(tau_start)
    stall = 0

    while time.time() < deadline - 4.0 and tau > 1e-4:
        improved = 0
        stage_max = 18 if n > 160 else 35

        for _ in range(stage_max):
            if time.time() > deadline - 5.0:
                break
            delta = first_order_direction(h, n, tau)
            dm = float(np.max(np.abs(delta))) if delta.size else 0.0
            if dm < 1e-8:
                break
            h2, c2 = line_search(h, n, delta, best_c)
            h2 = project_feasible(h2, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c - 1e-13:
                h = h2
                best_c = c2
                improved += 1
            else:
                break

        if improved == 0:
            tau *= 0.5
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
            tau *= 0.75

    return h, best_c


def gradient_pair_polish(h, n, deadline, max_passes=4, pcount=12):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / float(n)

    for _ in range(max_passes):
        if time.time() > deadline - 1.0:
            break
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        p = soft_weights(c, 20000.0)
        G = corr_grad(h, n, dx)
        grad = p @ G

        pos = np.where(grad > 1e-9)[0]
        neg = np.where(grad < -1e-9)[0]
        if pos.size == 0 or neg.size == 0:
            break

        pos = pos[np.argsort(-grad[pos])][:pcount]
        neg = neg[np.argsort(grad[neg])][:pcount]

        improved = False
        for i in pos:
            if h[i] <= 1e-12:
                continue
            for j in neg:
                if i == j or (1.0 - h[j]) <= 1e-12:
                    continue
                upper = min(h[i], 1.0 - h[j])
                if upper <= 1e-12:
                    continue
                for frac in (0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005):
                    d = frac * upper
                    h[i] -= d
                    h[j] += d
                    c2 = compute_c5(h, n)
                    if c2 < best_c - 1e-13:
                        best_c = c2
                        improved = True
                        break
                    else:
                        h[i] += d
                        h[j] -= d
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c


def make_even(n):
    try:
        m = int(round(float(n)))
    except Exception:
        m = 100
    if m < 8:
        m = 8
    if m > 360:
        m = 360
    if m % 2 != 0:
        m += 1
    return m


def make_n_list(parent_n):
    p = int(parent_n)
    if p < 8:
        p = 8
    if p > 360:
        p = 360

    raw = [
        p,
        p * 1.2,
        p * 0.8,
        p // 2,
        p // 3,
        p // 4,
        24,
        30,
        40,
        60,
        80,
        100,
        120,
        150,
        200,
        250,
        300,
        320,
    ]

    out = []
    out.append(max(8, min(360, p)))
    for x in raw:
        n = make_even(x)
        if n not in out:
            out.append(n)

    filtered = []
    for n in out:
        if 8 <= n <= 360:
            filtered.append(n)
    if len(filtered) == 0:
        filtered.append(p if p >= 8 else 100)
    return filtered


def random_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    if n % 2 == 0:
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    else:
        m = n // 2
        idx = rng.choice(n, size=m + 1, replace=False)
        h[idx[:m]] = 1.0
        h[idx[m]] = 0.5
    return h


def thue_morse_binary(n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        x = i
        parity = 0
        while x:
            parity ^= 1
            x &= (x - 1)
        h[i] = float(parity)
    return project_feasible(h, n)


def interpolate_source(src, src_n, n):
    src = np.asarray(src, dtype=float).reshape(-1)
    if src.size == n:
        return project_feasible(src.copy(), n)
    x1 = np.linspace(0.0, 2.0, len(src))
    x2 = np.linspace(0.0, 2.0, n)
    y = np.interp(x2, x1, src)
    return project_feasible(y, n)


def default_start(n, rng):
    x = np.linspace(0.0, 2.0, n)

    h1 = project_feasible(1.0 - np.abs(x - 1.0), n)
    c1 = compute_c5(h1, n)
    best_h = h1
    best_c = c1

    h2 = project_feasible(0.5 + 0.5 * np.sin(2.0 * np.pi * x), n)
    c2 = compute_c5(h2, n)
    if c2 < best_c:
        best_h = h2
        best_c = c2

    h3 = project_feasible(0.5 + 0.5 * np.sin(np.pi * x), n)
    c3 = compute_c5(h3, n)
    if c3 < best_c:
        best_h = h3
        best_c = c3

    if n % 2 == 0:
        ht = thue_morse_binary(n)
        ct = compute_c5(ht, n)
        if ct < best_c:
            best_h = ht
            best_c = ct

        hr = random_balanced(n, rng)
        cr = compute_c5(hr, n)
        if cr < best_c:
            best_h = hr
            best_c = cr

    return best_h


def choose_start(n, source_h, source_n, rng):
    if source_h is None or len(source_h) == 0:
        return default_start(n, rng)

    h = interpolate_source(source_h, source_n, n)
    best_h = h
    best_c = compute_c5(best_h, n)

    rv = project_feasible(best_h[::-1], n)
    crv = compute_c5(rv, n)
    if crv < best_c:
        best_h = rv
        best_c = crv

    if n % 2 == 0 and n <= 220:
        ht = thue_morse_binary(n)
        ct = compute_c5(ht, n)
        if ct < best_c:
            best_h = ht
            best_c = ct

        hr = random_balanced(n, rng)
        cr = compute_c5(hr, n)
        if cr < best_c:
            best_h = hr
            best_c = ct if False else cr

    return best_h


def optimize_resolution(h0, n, local_deadline, global_deadline):
    h = project_feasible(h0, n)
    best_c = compute_c5(h, n)
    stop = min(float(local_deadline), float(global_deadline) - 12.0)
    now = time.time()
    if stop <= now + 2.0:
        h = enforce_sum_exact(h, n)
        return h, compute_c5(h, n)

    rem = stop - now
    d1 = min(stop, now + max(10.0, 0.35 * rem))
    h2, c2 = slp_optimize(h, n, d1, tau_start=0.12)
    if c2 < best_c:
        h = h2
        best_c = c2

    rem = stop - time.time()
    alphas = (800.0, 3000.0, 12000.0, 60000.0)
    for a in alphas:
        if rem < 12.0:
            break
        h2 = slsqp_one(h, n, a, maxiter=80)
        c2 = compute_c5(h2, n)
        if c2 < best_c - 1e-14:
            h = h2
            best_c = c2
        rem = stop - time.time()

    rem = stop - time.time()
    if rem > 15.0:
        d2 = min(stop, time.time() + max(5.0, 0.25 * rem))
        h2, c2 = slp_optimize(h, n, d2, tau_start=0.02)
        if c2 < best_c:
            h = h2
            best_c = c2

    rem = stop - time.time()
    if rem > 6.0:
        h2, c2 = gradient_pair_polish(h, n, stop, max_passes=4, pcount=12)
        if c2 < best_c:
            h = h2
            best_c = c2

    if stop - time.time() > 12.0:
        h2 = slsqp_one(h, n, 200000.0, maxiter=50)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            h = h2
            best_c = c2

    h = project_feasible(h, n)
    h = enforce_sum_exact(h, n)
    best_c = compute_c5(h, n)
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(int(seed))

    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
        parent_n = int(len(parent))
    except Exception:
        parent = None
        parent_n = 0

    if parent is None or parent_n < 8 or not np.all(np.isfinite(parent)):
        parent_n = 100
        parent = default_start(parent_n, rng)

    parent = project_feasible(parent, parent_n)
    parent = enforce_sum_exact(parent, parent_n)

    best_h = parent.copy()
    best_n = parent_n
    best_c = compute_c5(best_h, best_n)
    source_h = parent.copy()
    source_n = parent_n

    ns = make_n_list(parent_n)
    first_n = ns[0]

    alloc = min(260.0, max(120.0, 0.25 * float(budget_s)))
    h2, c2 = optimize_resolution(source_h, first_n, t0 + alloc, deadline)
    if c2 < best_c:
        best_h = h2
        best_n = first_n
        best_c = c2
        source_h = h2.copy()
        source_n = first_n

    for n_try in ns:
        if n_try == first_n:
            continue
        if time.time() > deadline - 120.0:
            break

        remaining = deadline - time.time()
        if n_try > 300:
            alloc = min(70.0, max(20.0, 0.18 * remaining))
        else:
            alloc = min(90.0, max(25.0, 0.22 * remaining))
        if alloc < 18.0:
            break

        starts = [choose_start(n_try, source_h, source_n, rng)]
        if n_try <= 90 and n_try % 2 == 0:
            tb = thue_morse_binary(n_try)
            starts.append(tb)
            rb = random_balanced(n_try, rng)
            starts.append(rb)

        per_alloc = alloc / float(len(starts))
        local_best_h = None
        local_best_c = float("inf")
        for st in starts:
            if time.time() > deadline - 90.0:
                break
            ha, ca = optimize_resolution(st, n_try, time.time() + per_alloc, deadline)
            if ca < local_best_c:
                local_best_c = ca
                local_best_h = ha

        if local_best_h is not None and local_best_c < best_c - 1e-15:
            best_h = local_best_h
            best_n = n_try
            best_c = local_best_c
            source_h = local_best_h.copy()
            source_n = n_try

    attempt = 0
    while time.time() < deadline - 70.0:
        attempt += 1
        remaining = deadline - time.time()
        if remaining < 60.0:
            break

        scale = 0.02 + 0.06 * float(attempt % 5) / 4.0
        if attempt % 2 == 0:
            pert = best_h + rng.normal(0.0, scale, best_n)
        else:
            rev = interpolate_source(best_h[::-1], best_n, best_n)
            pert = rev + rng.normal(0.0, 0.5 * scale, best_n)

        pert = project_feasible(pert, best_n)
        alloc = min(70.0, max(25.0, 0.30 * remaining))
        if alloc < 20.0:
            break

        h2, c2 = optimize_resolution(pert, best_n, time.time() + alloc, deadline)
        if c2 < best_c - 1e-15:
            best_h = h2
            best_n = best_n
            best_c = c2
            source_h = h2.copy()
            source_n = best_n

    if time.time() < deadline - 12.0:
        h2, c2 = optimize_resolution(best_h, best_n, deadline - 1.0, deadline)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_h = enforce_sum_exact(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, float(best_c), int(best_n)
