    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def greedy_binary_local(h, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    oh = 1.0 - h
    c = float(np.max(np.correlate(h, oh, 'full')) * dx)
    ones = [i for i in range(n) if h[i] == 1.0]
    zeros = [i for i in range(n) if h[i] == 0.0]
    best_c = c
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_delta = 1e-15
        best_i = -1
        best_j = -1

        for i in zeros:
            if time.time() > deadline:
                break
            h[i] = 1.0
            oh[i] = 0.0
            for j in ones:
                h[j] = 0.0
                oh[j] = 1.0
                nc = float(np.max(np.correlate(h, oh, 'full')) * dx)
                h[j] = 1.0
                oh[j] = 0.0
                d = c - nc
                if d > best_delta:
                    best_delta = d
                    best_i = i
                    best_j = j
            h[i] = 0.0
            oh[i] = 1.0

        if best_i >= 0:
            h[best_i] = 1.0
            oh[best_i] = 0.0
            h[best_j] = 0.0
            oh[best_j] = 1.0
            c = c - best_delta
            best_c = c
            best_h = h.copy()
            zeros.remove(best_i)
            ones.append(best_i)
            ones.remove(best_j)
            zeros.append(best_j)
        else:
            break

    return best_h, best_c

def sa_binary(h, n, deadline, t_init=0.05, t_ratio=0.002, n_temps=60, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    oh = 1.0 - h
    c = float(np.max(np.correlate(h, oh, 'full')) * dx)
    best_h = h.copy()
    best_c = c
    ones = [i for i in range(n) if h[i] == 1.0]
    zeros = [i for i in range(n) if h[i] == 0.0]

    t_final = t_init * t_ratio
    sweeps_per_temp = 150

    for ti in range(n_temps):
        if time.time() > deadline:
            break
        frac = (ti + 1) / n_temps
        T = t_init * (t_final / t_init) ** frac

        for s in range(sweeps_per_temp):
            if time.time() > deadline:
                break
            ii = rng.randint(len(zeros))
            jj = rng.randint(len(ones))
            iz = zeros[ii]
            jo = ones[jj]

            h[iz] = 1.0
            oh[iz] = 0.0
            h[jo] = 0.0
            oh[jo] = 1.0
            new_c = float(np.max(np.correlate(h, oh, 'full')) * dx)

            if new_c < c or (T > 1e-15 and rng.random() < np.exp((c - new_c) / max(T, 1e-15))):
                c = new_c
                zeros[ii] = jo
                ones[jj] = iz
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
            else:
                h[iz] = 0.0
                oh[iz] = 1.0
                h[jo] = 1.0
                oh[jo] = 0.0

        if ti % 12 == 11:
            h, c = greedy_binary_local(h, n, min(deadline, time.time() + 3), max_sweeps=30)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
            ones = [i for i in range(n) if h[i] == 1.0]
            zeros = [i for i in range(n) if h[i] == 0.0]

    return best_h, best_c

def perturbation_restart(h, n, deadline, n_perturb=5, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for p in range(n_perturb):
        if time.time() > deadline - 20:
            break
        hp = h.copy()
        n_flip = rng.randint(3, 10)
        ones = np.where(hp == 1)[0]
        zeros = np.where(hp == 0)[0]
        for _ in range(n_flip):
            i = rng.randint(len(zeros))
            j = rng.randint(len(ones))
            hp[zeros[i]] = 1.0
            hp[ones[j]] = 0.0
            zeros[i] = ones[j]
            ones[j] = zeros[i]

        hp, c = greedy_binary_local(hp, n, min(deadline - 10, time.time() + 15), max_sweeps=100)
        if c < best_c:
            best_c = c
            best_h = hp.copy()

    return best_h, best_c

def slsqp_continuous(h, n, deadline, alpha=50000, maxiter=1500):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return float(m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha)

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def two_swap_search(h, n, deadline, max_iters=2000):
    """Try 2-for-2 swaps for more flexibility."""
    dx = 2.0 / n
    oh = 1.0 - h
    c = float(np.max(np.correlate(h, oh, 'full')) * dx)
    best_c = c
    best_h = h.copy()
    ones = sorted([i for i in range(n) if h[i] == 1.0])
    zeros = sorted([i for i in range(n) if h[i] == 0.0])
    rng = np.random.RandomState(77)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        i1 = rng.randint(len(zeros))
        i2 = (i1 + rng.randint(1, len(zeros))) % len(zeros)
        j1 = rng.randint(len(ones))
        j2 = (j1 + rng.randint(1, len(ones))) % len(ones)

        z1, z2 = zeros[i1], zeros[i2]
        o1, o2 = ones[j1], ones[j2]

        if z1 == z2 or o1 == o2:
            continue

        h[z1] = 1.0; oh[z1] = 0.0
        h[z2] = 1.0; oh[z2] = 0.0
        h[o1] = 0.0; oh[o1] = 1.0
        h[o2] = 0.0; oh[o2] = 1.0
        nc = float(np.max(np.correlate(h, oh, 'full')) * dx)

        if nc < c - 1e-14:
            c = nc
            if c < best_c:
                best_c = c
                best_h = h.copy()
            zeros.remove(z1); zeros.insert(0, o1)
            zeros.remove(z2); zeros.insert(0, o2)
            ones.remove(o1); ones.append(z1)
            ones.remove(o2); ones.append(z2)
        else:
            h[z1] = 0.0; oh[z1] = 1.0
            h[z2] = 0.0; oh[z2] = 1.0
            h[o1] = 1.0; oh[o1] = 0.0
            h[o2] = 1.0; oh[o2] = 0.0

    return best_h, best_c
