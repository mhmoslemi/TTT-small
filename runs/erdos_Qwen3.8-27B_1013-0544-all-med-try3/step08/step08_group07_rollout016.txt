    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_gk(h, n, k_idx):
    """Analytical gradient of g_{k_idx} w.r.t. h."""
    m = abs(k_idx - (n - 1))
    if m == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if k_idx > n - 1:
        # positive lag = m
        grad[m:] = 1.0 - h[:n - m]
        grad[:n - m] += -h[m:]
    else:
        # negative lag = -m
        grad[:n - m] = 1.0 - h[m:]
        grad[m:] += -h[:n - m]
    return grad

def subgrad_optimize(h, n, deadline, max_iter=10000):

    W = compute_all_g(h, n)  # history for Polyak

    # Polyak step: alpha_t = (f(h_t) - f*) / ||g_t||^2
    # We don't know f*, so use optimistic estimate
    f_star_est = best_c * 0.99  # optimistic
    t = 0
    lr = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g_all = compute_all_g(h, n)
        m = g_all.max()
        k_star = int(g_all.argmax())
        
        # Subgradient: use the active constraint's gradient
        grad = grad_gk(h, n, k_star) * dx
        
        # Also consider near-active constraints
        # If multiple k's are close to max, use their average gradient
        threshold = m - 1e-4
        active_ks = np.where(g_all >= threshold)[0]
        if len(active_ks) > 1:
            grad = np.mean([grad_gk(h, n, k) for k in active_ks], axis=0) * dx
        
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        
        # Adaptive step size
        step = lr / ng
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < m - 1e-14:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.9
            if lr < 1e-12:

def binary_local_search(h, n, deadline, max_sweeps=300):
    """Local search on binary solutions via bit swaps."""
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    
    ones = sorted(np.where(h_bin == 1)[0].tolist())
    zeros = sorted(np.where(h_bin == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try moving a 1 from position j to position i
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                if j == i:
                    continue
                # Quick check: only try nearby or random pairs
                h_bin[j] = 0.0
                h_bin[i] = 1.0
                c = compute_c5(h_bin, n)
                if c < best_c:
                    best_c = c
                    improved = True
                    ones.remove(j)
                    ones.append(i)
                    ones.sort()
                    zeros.remove(i)
                    zeros.append(j)
                    zeros.sort()
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_gk(h, n, k_idx):
    m = abs(k_idx - (n - 1))
    if m == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if k_idx > n - 1:
        grad[m:] = 1.0 - h[:n - m]
        grad[:n - m] += -h[m:]
    else:
        grad[:n - m] = 1.0 - h[m:]
        grad[m:] += -h[:n - m]
    return grad

def subgrad_optimize(h, n, deadline, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.5
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g_all = compute_all_g(h, n)
        m = g_all.max()
        k_star = int(g_all.argmax())
        grad = grad_gk(h, n, k_star) * dx
        thresh = m - 0.002
        active_ks = np.where(g_all >= thresh)[0]
        if len(active_ks) > 1:
            gsum = np.zeros(n)
            for k in active_ks:
                gsum += grad_gk(h, n, int(k))
            grad = gsum / len(active_ks) * dx
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        step = lr / ng
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < m - 1e-14:
            h = h_new
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            lr = min(lr * 1.02, 2.0)
        else:
            lr *= 0.85
            stall += 1
            if stall > 500:
                break
            if lr < 1e-14:
                break
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, deadline, max_sweeps=500):
