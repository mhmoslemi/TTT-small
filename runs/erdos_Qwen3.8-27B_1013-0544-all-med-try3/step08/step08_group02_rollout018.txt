def fine_pair_search(h, n, deadline, radius=30):

        while improved and time.time() < deadline:

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

def fourier_optimize(h, n, deadline, K=30):
    """Optimize in a truncated Fourier basis for dimensionality reduction."""

    c_best = compute_c5(h_best, n)

    H = fft(h_best)
    # Keep DC fixed, optimize first K modes
    # Parameters: K complex numbers = 2K real params
    a = np.zeros(2 * K)
    for k in range(1, K + 1):
        a[2*k-2] = H[k].real / n
        a[2*k-1] = H[k].imag / n
    
    def fourier_h(params):
        spectrum = np.zeros(n, dtype=complex)
        spectrum[0] = n / 2.0  # fixed DC for sum = n/2
        for k in range(1, K + 1):
            spectrum[k] = params[2*k-2] + 1j * params[2*k-1]
            spectrum[n-k] = np.conj(spectrum[k])
        hh = np.real(ifft(spectrum))
        return project_feasible(hh, n)
    
    def fourier_obj(params):
        hh = fourier_h(params)
        return compute_c5(hh, n)
    
    def fourier_opt(params):
        hh = fourier_h(params)
        # Use L-BFGS with numerical gradient in low-dim space
        return compute_c5(hh, n)
    
    # Optimize with L-BFGS using numerical gradient (low dim, OK)
    x0 = a.copy()
    maxit = min(300, int((deadline - time.time()) / 2))

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:

        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick L-BFGS at original resolution (60s)
    p1_end = t0 + 60
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[200, 2000, 20000, 200000],
        mu=0.01, maxit_list=[500, 800, 1000, 1200])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active-set focused (100s)
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 100, deadline - 400)
        h2, c2 = run_active(best_h, best_n, p2_end, top_k=6,
            alphas=[20000, 100000, 500000, 2000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: SLSQP with explicit constraint (80s)
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 80, deadline - 350)
        h3, c3 = run_slqp(best_h, best_n, p3_end, alpha=500000, maxiter=800)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Sigmoid reinitialization + reoptimize (80s)
    if time.time() < deadline - 300:
        for beta in [3, 6, 10]:
            if time.time() > deadline - 300:
                break
            h_sig = sigmoid_reinit(best_h, best_n, beta)
            c_sig = compute_c5(h_sig, best_n)
            p4_end = min(time.time() + 25, deadline - 290)
            h4, c4 = run_active(h_sig, best_n, p4_end, top_k=5,
                alphas=[10000, 100000, 1000000])
            if c4 < best_c:
                best_c = c4
                best_h = h4
    
    # Phase 5: Binary-reinit (50s)
    if time.time() < deadline - 250:
        h_bin = make_binary_start(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        p5_end = min(time.time() + 40, deadline - 240)
        h5, c5 = run_active(h_bin, best_n, p5_end, top_k=5,
            alphas=[5000, 50000, 500000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
    
    # Phase 6: Fourier basis optimization (100s)
    if time.time() < deadline - 200:
        p6_end = min(time.time() + 100, deadline - 200)
        h6, c6 = fourier_optimize(best_h, best_n, p6_end, K=25)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Upsample to n=1000 and refine (250s)
    n_hi = 1000
    if time.time() < deadline - 180:
        x_fine = np.linspace(0, 2, n_hi)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_hi)
        c_fine = compute_c5(h_fine, n_hi)
        
        p7a_end = min(time.time() + 120, deadline - 120)
        h7, c7 = run_lbfgs(h_fine, n_hi, p7a_end,
            alphas=[50, 500, 5000, 50000, 200000],
            mu=0.005, maxit_list=[800, 1000, 1200, 1500, 800])
        
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = n_hi
        
        p7b_end = min(time.time() + 80, deadline - 80)
        h7b, c7b = run_active(best_h, best_n, p7b_end, top_k=5,
            alphas=[50000, 200000, 1000000])
        if c7b < best_c:
            best_c = c7b
            best_h = h7b
            best_n = n_hi
    
    # Phase 8: Subgradient refinement (60s)
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 50, deadline - 30)
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=8000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Fine pair search (50s)
    if time.time() < deadline - 10:
        p9_end = deadline - 3
        h9, c9 = fine_pair_search(best_h, best_n, p9_end, radius=25)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: Final SLSQP polish (5s)
    if time.time() < deadline - 2:
        h10, c10 = run_slqp(best_h, best_n, deadline - 1, alpha=1000000, maxiter=300)
        if c10 < best_c:
            best_c = c10
            best_h = h10
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
