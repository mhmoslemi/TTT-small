import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_active(x, n, top_idx, alpha, mu):
    """Softplus objective for top-k active shifts."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_active(x, n, top_idx, alpha, mu):
    """Gradient of softplus for top-k active shifts."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def find_active_shifts(c_vec, n, top_k=5, rel_tol=0.002):
    """Find the top-k shifts that are close to the maximum."""
    max_val = c_vec.max()
    threshold = max_val - max(1e-8, rel_tol * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = active[np.argsort(-c_vec[active])][:top_k]
    return active.tolist(), c_vec[active]

def lbfgs_active(h, n, deadline, alphas, mu=0.01, top_k=5):
    """Iterative L-BFGS with active-set updates."""
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        for _ in range(5):  # Multiple rounds with updated active set
            if time.time() > deadline - 3:
                break
            c_vec = compute_c5_vec(h, n)
            active_idx, _ = find_active_shifts(c_vec, n, top_k=top_k)
            if len(active_idx) == 0:
                break
            try:
                res = minimize(obj_active, h, jac=grad_active,
                              args=(n, active_idx, alpha, mu),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 30000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
            except Exception:
                pass
    return best_h, best_c

def perturbation_restart(h, n, rng, deadline, alphas, mu, top_k):
    """Perturb h slightly and re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for _ in range(3):
        if time.time() > deadline - 10:
            break
        # Small random perturbation
        eps = 0.01
        h_pert = h + eps * (rng.random(n) - 0.5)
        h_pert = project_feasible(h_pert, n)
        
        h_new, c_new = lbfgs_active(h_pert, n, deadline, alphas, mu, top_k)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def pair_search(h, n, deadline, radius=30):
    """Local pair swap search."""
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, 3):  # Stride to save time
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 2):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def obj_full_softplus(x, n, alpha, mu):
    """Full softplus over all shifts."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_full_softplus(x, n, alpha, mu, top_k=20):
    """Gradient using only top-k active shifts for efficiency."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    # Only use top-k weights
    top = np.argsort(-w)[:top_k]
    w_top = w[top]
    w_top /= w_top.sum()  # Renormalize
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx, wi in zip(top, w_top):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def find_active(c_vec, top_k=5, rel_tol=0.001):
    mx = c_vec.max()
    threshold = mx - max(1e-10, rel_tol * mx)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = active[np.argsort(-c_vec[active])][:top_k]
    return active

def obj_top(x, n, idx, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    vals = c[idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_top(x, n, idx, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    vals = c[idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    for ii in range(len(idx)):
        wi = w[ii]
        if wi < 1e-14:
            continue
        k = idx[ii] - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_full(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_full_topk(x, n, alpha, mu, top_k=15):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w_all = np.exp(sh - mx)
    w_all /= w_all.sum()
    top = np.argsort(-w_all)[:top_k]
    wt = w_all[top]
    wt /= wt.sum()
    grad = np.zeros(n)
    for ii in range(len(top)):
        wi = wt[ii]
        if wi < 1e-14:
            continue
        k = top[ii] - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])

            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def optimize_active(h, n, deadline, alphas, mu, top_k, max_rounds=4):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        for rnd in range(max_rounds):
            if time.time() > deadline - 3:
                break

            if len(active) == 0:
                break

                              args=(n, active, alpha, mu),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 50000, 'maxls': 60})
                h_new = project_feasible(res.x, n)

                    best_h = h_new
                    h = h_new
            except Exception:
                pass

def pair_search(h, n, deadline, radius=25, stride_main=2, stride_sub=2):
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, stride_main):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, stride_sub):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def gradient_descent_subgrad(h, n, deadline, lr0=0.01, lr_decay=0.95, max_iter=20000):
    """Projected subgradient method on the max of active shifts."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr0
    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        c_vec = compute_c5_vec(h, n)
        mx = c_vec.max()
        active = find_active(c_vec, top_k=5, rel_tol=0.0005)
        if len(active) == 0:
            break
        # Average gradient over active shifts
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += (1.0 - h[k:n]) * dx
                    grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += (1.0 - h[0:nv]) * dx
                    grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        # Projected gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new
        else:
            lr *= lr_decay
        if lr < 1e-10:
            break
    return best_h, best_c
