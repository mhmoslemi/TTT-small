def compute_gradient(h, n, alpha):
    """Compute gradient of smoothed max correlation"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()  # softmax weights
    
    # gradient of c_k w.r.t. h[j] = (1-h[j+k_offset]) - h[j-k_offset]
    # For the full correlation of length 2n-1, shift k ranges from -(n-1) to n-1
    # c[k] for k=0,...,2n-2 corresponds to shift k-(n-1)
    
    # grad[j] = sum_k w[k] * (1-h[j+k-(n-1)]) - h[j-(k-(n-1))]) * dx
    #         = sum_k w[k] * (1-h[j+k-(n-1)]) - sum_k w[k] * h[j-(k-(n-1))])
    
    # First term: sum_k w[k] * (1-h[j+k-(n-1)])
    # = N - sum_k w[k] * h[j+k-(n-1)]  (with bounds)
    # This is like a weighted correlation of w with h
    
    # Let shift = k - (n-1), ranges from -(n-1) to n-1
    # First term: N - sum_shift w[shift+(n-1)] * h[j+shift] (bounds checked)
    # Second term: sum_shift w[shift+(n-1)] * h[j-shift] (bounds checked)
    
    grad = np.zeros(n)
    for j in range(n):
        for k in range(2*n-1):
            s = k - (n + 1)  # shift
            # term 1: h[j] * (1-h[j+s]) -> grad contribution (1-h[j+s])
            # term 2: h[j-s] * (1-h[j]) -> grad contribution -h[j-s]
            if 0 <= j + s < n:
                grad[j] += w[k] * (1 - h[j+s])
            if 0 <= j - s < n:
                grad[j] -= w[k] * h[j-s]
        grad[j] *= dx
    
    return grad

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h_trial, n):
    """Project onto {0≤h≤1, sum(h)=n/2} via binary search on offset."""
    target = n / 2.0
    tau_lo = h_trial.min() - 1.0
    tau_hi = h_trial.max() + 1.0
    for _ in range(80):
        tau = (tau_lo + tau_hi) / 2.0
        s = np.clip(h_trial - tau, 0.0, 1.0).sum()
        if s > target:
            tau_lo = tau
        else:
            tau_hi = tau
    tau = (tau_lo + tau_hi) / 2.0
    h = np.clip(h_trial - tau, 0.0, 1.0)

        if free.sum() > 0:

            h = np.clip(h, 0.0, 1.0)

        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    n_full = len(c)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    exp_s = np.exp(sh - mx)

def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-14):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        return softmax_obj(x, n, alpha)
    
    def constraint_fun(x):
        return x.sum() - target
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < softmax_obj(h, n, alpha):

    except Exception:
        pass

def projected_gradient(h, n, alpha, deadline, lr_init=0.005, max_iters=500000):
    """Projected gradient descent with backtracking line search."""
    h = project_box_sum(h.copy(), n)
    best_c = compute_c5(h, n)
    lr = lr_init
    iterations = 0
    
    while time.time() < deadline and iterations < max_iters:
        iterations += 1
        obj_val, grad = softmax_obj(h, n, alpha), softmax_grad(h, n, alpha)
        c = compute_c5(h, n)
        
        if c < best_c - 1e-14:
            best_c = c
        
        # Determine step direction (project gradient onto sum=0 subspace)
        grad_centered = grad - grad.mean()
        
        # Backtracking line search
        accepted = False
        for backtrack in range(20):
            h_trial = h - lr * grad_centered
            h_new = project_box_sum(h_trial, n)
            c_new = compute_c5(h_new, n)
            if c_new < c - 1e-15 * max(1, abs(c)):
                h = h_new
                if c_new < best_c:

                accepted = True
                break

            if lr < 1e-10:

def gradient_pair_search(h, n, deadline, max_sweeps=10000):
    """Use gradient to identify best pair moves."""
    h = project_box_sum(h.copy(), n)
    best_c = compute_c5(h, n)
    alpha = 100000  # high for sharp max
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
        
        grad = softmax_grad(h, n, alpha)
        grad_centered = grad - grad.mean()
        
        # Find best pair: maximize -grad_centered[i] * (1 if we increase i)
        # We want to increase h[i] (negative gradient) and decrease h[j] (positive gradient)
        # Score for moving from j to i: grad_centered[j] - grad_centered[i]
        # We want this to be positive (gradient at j is high, at i is low)
        
        # Find the pair with most negative (grad[i] - grad[j]) where grad[i] < grad[j]
        # i.e., maximize grad[j] - grad[i]
        
        # For efficiency, find the most negative and most positive entries
        # But we also need to respect bounds: h[i]+d ≤ 1 and h[j]-d ≥ 0
        
        # Simple approach: find argmin and argmax of grad_centered
        i_best = np.argmin(grad_centered)  # most negative → increase h[i]
        j_best = np.argmax(grad_centered)  # most positive → decrease h[j]
        
        if i_best == j_best:
            break
        
        # Try a few promising pairs
        n_candidates = 5
        idx_sorted = np.argsort(grad_centered)
        candidates = []
        for _i in range(n_candidates):
            i_c = idx_sorted[_i]  # low gradient → increase
            for _j in range(n - n_candidates, n):
                j_c = idx_sorted[_j]  # high gradient → decrease
                if i_c != j_c:
                    candidates.append((i_c, j_c))

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h_trial, n):
    target = n / 2.0
    tau_lo = h_trial.min() - 1.0
    tau_hi = h_trial.max() + 1.0
    for _ in range(80):
        tau = (tau_lo + tau_hi) / 2.0
        s = np.clip(h_trial - tau, 0.0, 1.0).sum()
        if s > target:
            tau_lo = tau
        else:
            tau_hi = tau
    tau = (tau_lo + tau_hi) / 2.0
    h = np.clip(h_trial - tau, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    n_full = len(c)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    exp_s = np.exp(sh - mx)
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    shifts = np.arange(n_full) - (n - 1)
    for j in range(n):
        idx_a = j + shifts
        va = (idx_a >= 0) & (idx_a < n)
        if va.any():
            grad[j] += w[va] @ (1.0 - h[idx_a[va]])
        idx_b = j - shifts
        vb = (idx_b >= 0) & (idx_b < n)
        if vb.any():
            grad[j] -= w[vb] @ h[idx_b[vb]]
    grad *= dx
    return grad
