def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_onto_simplex_box(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for lambda
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Fix small numerical errors
    diff = target - projected.sum()
    if abs(diff) > 1e-14:
        # Distribute among non-bound entries
        mask = (projected > 1e-10) & (projected < 1 - 1e-10)
        if mask.any():
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0.0, 1.0)
    return projected

    grad /= len(active_ks)
    return grad

def coordinate_descent(h, n, deadline, max_sweeps=200, step_sizes=None):

    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h, n)

        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Randomize order
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            for s in step_sizes:
                for sign in (1, -1):
                    if h[j] + sign * s > 1 or h[j] + sign * s < 0:
                        continue
                    # Transfer: increase h[j], decrease another h to maintain sum
                    # Find which other index to decrease
                    # Simple: decrease the index with highest h value (excluding j)
                    h[j] += sign * s
                    # Need to adjust sum
                    diff = -sign * s
                    # Distribute to other elements proportionally to their "room"
                    if diff > 0:
                        # Need to increase others
                        room = 1.0 - h
                        room[j] = 0
                        if room.sum() > 0:
                            h += diff * room / room.sum()
                        else:
                            h[j] -= sign * s
                            continue
                    else:
                        # Need to decrease others
                        h_others = h.copy()
                        h_others[j] = 0
                        if h_others.sum() > 0:
                            h += diff * h_others / h_others.sum()
                        else:
                            h[j] -= sign * s
                            continue
                    h = np.clip(h, 0, 1)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-12:
                        best_c = c
                        improved = True
                    else:
                        h[j] -= sign * s
                        # Re-project
                        h = project_onto_simplex_box(h, n)
        if not improved:
            break
    
    return h, best_c

        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_onto_simplex_box(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if projected.sum() > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - projected.sum()
    if abs(diff) > 1e-12:
        mask = (projected > 1e-12) & (projected < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            projected[mask] += diff / cnt
    return np.clip(projected, 0.0, 1.0)

def grad_for_lag(h, n, k, dx):
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n - k]) * dx
            if k > 0:
                g[:n - k] += -h[k:] * dx
            else:
                g[:] += -h[:] * dx
    else:
        kp = -k
        if kp < n:
            g[:n - kp] += (1.0 - h[kp:]) * dx
            g[kp:] += -h[:n - kp] * dx
    return g

def subgrad_optimize(h, n, deadline, max_iter=50000, top_k=10):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.03
    v = np.zeros(n)
    beta = 0.3
    no_improve = 0
    patience = 300
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if (it + 1) % 50 == 0 and time.time() > deadline:
            break
        
        g = compute_all_g(h, n)
        m = g.max()
        
        idx_sorted = np.argsort(g)[::-1]
        active = []
        threshold = m - 0.1 * (m - g[idx_sorted[min(top_k, len(g) - 1)]])
        for idx in idx_sorted[:top_k]:
            if g[idx] < threshold:
                break
            k = int(idx) - (n - 1)
            active.append(k)
        if not active:
            active = [int(idx_sorted[0]) - (n - 1)]
        
        grad = np.zeros(n)
        for k in active:
            grad += grad_for_lag(h, n, k, dx)
        grad /= len(active)
        
        direction = -grad
        v = beta * v + direction
        beta = min(beta * 1.005, 0.85)
        
        step = lr
        improved = False
        for _ in range(25):
            h_new = project_onto_simplex_box(h + step * v, n)
            c_new = compute_c5(h_new, n)
            if c_new < m - 1e-13:
                h = h_new
                improved = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                no_improve = 0
                break
            step *= 0.5
        if not improved:
            lr *= 0.85
            v *= 0.3
            no_improve += 1
            if no_improve > patience or lr < 1e-10:
                break
    
    return best_h, best_c

def slsqp_polish(h, n, deadline, alpha=50000, maxiter=150):
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_eq(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_onto_simplex_box(res.x, n)
    except Exception:
        return h

def fine_pair_search(h, n, deadline, max_iters=30000):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        d = deltas[rng.randint(len(deltas))]
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        s = 1.0 if rng.rand() < 0.5 else -1.0
        hi, hj = h[i] + s * d, h[j] - s * d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
        else:
            h[i], h[j] = hi - s * d, hj + s * d
    
    return h, best_c
