        if alloc < 5:
            break
        ht = slsqp(best_h, n, a, maxiter=4000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    if time.time() < deadline - 10:
        best_h, best_c2 = subgrad_descent(best_h, n, n_iter=10000,
                                          deadline=min(deadline - 5, time.time() + 80))
        if best_c2 < best_c:
            best_c = best_c2
            best_h = best_h

    if time.time() < deadline - 10:
        if n <= 120:
            best_h, best_c2 = pair_search_optimal(best_h, n, deadline - 5, max_passes=20)
        else:
            best_h, best_c2 = pair_search_optimal(best_h, n, deadline - 5, max_passes=15, radius=40)
        if best_c2 < best_c:
            best_c = best_c2
            best_h = best_h

    if time.time() < deadline - 5:
        if n <= 120:
            best_h, best_c2 = fixed_delta_pair_search(best_h, n, deadline - 2,
                                                       [0.001, 0.0005, 0.0002], max_passes=30)
        else:
            best_h, best_c2 = fixed_delta_pair_search(best_h, n, deadline - 2,
                                                       [0.001, 0.0005], max_passes=20, radius=30)
        if best_c2 < best_c:
            best_c = best_c2
            best_h = best_h

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        # Phase 1: Optimize at native resolution
        alloc1 = min(400, (deadline - t0) * 0.4)
        h_opt, c_opt = optimize_one(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

        # Phase 2: Try other resolutions
        x_init = np.linspace(0, 2, n_init)
        resolutions = []
        for n_try in [100, 120, 150, 200, 250, 300]:
            if n_try != n_init:
                resolutions.append(n_try)

        for n_try in resolutions:
            if time.time() > deadline - 40:
                break
            remaining = deadline - time.time()
            alloc = min(80, remaining * 0.15)
            if alloc < 20:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_one(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is None:
        n_init = 200
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
        alloc = min(500, deadline - t0 - 30)
        best_h, best_c = optimize_one(best_h, n_init, t0 + alloc, rng)

    # Phase 3: Multi-restart with perturbations
    n = len(best_h)
    remaining = deadline - time.time()
    if remaining > 30:
        n_restarts = max(2, int((remaining - 20) / 50))
        n_restarts = min(n_restarts, 5)
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            if r == 0:
                scale = 0.01
            elif r == 1:
                scale = 0.03
            else:
                scale = 0.05 + 0.05 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 12) * 0.35)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_one(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final fine-tuning
    if time.time() < deadline - 3:
        n = len(best_h)
        if n <= 120:
            best_h, best_c = pair_search_optimal(best_h, n, deadline - 1, max_passes=15)
        else:
            best_h, best_c = pair_search_optimal(best_h, n, deadline - 1, max_passes=10, radius=30)
        # Even finer fixed delta
        if time.time() < deadline - 1:
            if n <= 120:
                best_h, best_c = fixed_delta_pair_search(best_h, n, deadline - 0.5,
                                                          [0.0005, 0.0002, 0.0001], max_passes=20)
            else:
                best_h, best_c = fixed_delta_pair_search(best_h, n, deadline - 0.5,
                                                          [0.0005, 0.0002], max_passes=15, radius=25)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
