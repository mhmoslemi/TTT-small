import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            sr = room.sum()
            if sr < 1e-14:
                break
            h = h + diff / sr * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_continuous(h0, n, deadline, lr_init=0.1, top_k=5):
    """FISTA-like accelerated projected subgradient method."""
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    base_lr = lr_init / n  # scale by n for stability
    
    while time.time() < deadline:
        corr = np.correlate(y, 1 - y, mode='full') * dx
        cmax = corr.max()
        
        # Top-k active lags
        top_idx = np.argsort(corr)[-top_k:]
        
        # Subgradient
        grad = np.zeros(n)
        for k_idx in top_idx:
            k = k_idx - (n - 1)
            if k >= 0:
                if n - k > 0:
                    grad[:n-k] += (1 - y[k:]) * dx
                if k > 0:
                    grad[k:] -= y[:n-k] * dx
            else:
                kk = -k
                if n - kk > 0:
                    grad[kk:] += (1 - y[:n-kk]) * dx
                    grad[:n-kk] -= y[kk:] * dx
        grad /= len(top_idx)
        
        lr = base_lr / (t + 1)
        h_new = project_sum(y - lr * grad, target)
        
        if cmax < best_c:
            best_c = cmax
            best_h = h.copy()
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = h_new + ((t - 1) / t_new) * (h_new - h)
        y = project_sum(y, target)
        h = h_new
        t = t_new
    
    return best_h, best_c

def binary_local_search_fast(h, n, deadline, max_sweeps=2000):
    """Binary local search with early termination."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    m_range = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        # Try each zero position
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones:
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2*n-1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n - 1 + (j - i)

                    delta[k1] += 1.0
