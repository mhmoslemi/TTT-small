def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_fn(x, n, alpha):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        logsum = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        logsum = np.log(np.sum(np.exp(sh)))

def smooth_max_grad(x, n, alpha):
