    return best_h, best_c

def run_active_opt(h, n, deadline, start_topk=20, end_topk=3,
                   alphas=(5000, 20000, 100000, 500000, 2000000, 10000000),
                   mu=0.005, rounds_per_alpha=3, margin_frac=0.002):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    total_alphas = len(alphas)
    round_num = 0
    for ai, alpha in enumerate(alphas):
        topk = max(end_topk, start_topk - int((start_topk - end_topk) * ai / max(1, total_alphas - 1)))
        for _round in range(rounds_per_alpha):
            round_num += 1
            if time.time() > deadline - 10:
                return best_h, best_c
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            margin = max(1e-8, margin_frac * max_val)
            active = np.where(c_vec >= max_val - margin)[0]
            if len(active) > topk:
                active = np.argsort(-c_vec[active])[:topk]
            active = active[np.argsort(active)]
            if len(active) < 2:
                active = np.argsort(-c_vec)[:max(3, topk // 2)]
            maxit = max(500, int((deadline - time.time()) * 2.5))
            try:
                res = minimize(_active_obj, best_h, jac=_active_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                elif c_new <= best_c + 1e-14:
                    s = h_new.sum()
                    if abs(s - n/2.0) < 1e-10:
                        best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run_slqp_phase(h, n, deadline, alphas=(100000, 500000, 2000000)):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    def sq_f(x, nn):
        return x.sum() - nn / 2.0
    def sq_j(x, nn):
        return np.ones(nn)
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        maxit = max(500, int((deadline - time.time()) * 3))
        cons = [{'type': 'eq', 'fun': sq_f, 'jac': sq_j, 'args': (n,)}]
        try:
            res = minimize(_full_obj, best_h, args=(n, alpha, 0.0), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxit, 'ftol': 1e-18, 'iprint': -1})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_subgrad(h, n, deadline, n_iter=5000, eta0=0.003):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    eta = eta0
    tol_expand = 0.002
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_curr, n)
        c_max = c_vec.max()
        tol = max(1e-10, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        if len(active) > 30:
            active = np.argsort(-c_vec[active])[:30]
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h_curr[k:n]) * dx
                grad[k:n] += (-h_curr[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h_curr[0:nv]) * dx
                grad[0:nv] += (-h_curr[kk:n]) * dx
        grad /= len(active)
        h_new = h_curr - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        elif c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        else:
            eta *= 0.97
        h_curr = h_new
    return best_h, best_c

def run_pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 5:
            improved = False
            for i in range(n):
                if time.time() > deadline - 5:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def run_ineq_active(h, n, deadline, topk=5):
    """Optimize only the top-k lags with very high alpha, many restarts."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in [50000, 200000, 1000000, 5000000, 20000000]:
        if time.time() > deadline - 10:
            break
        for restart in range(3):
            if time.time() > deadline - 10:
                break
            c_vec = compute_c5_vec(best_h, n)
            active = np.argsort(-c_vec)[:topk]
            active = active[np.argsort(active)]
            maxit = max(800, int((deadline - time.time()) * 2))
            try:
                res = minimize(_active_obj, best_h, jac=_active_grad,
                              args=(n, active, alpha, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': maxit * 12,
                                       'maxls': 60, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                elif c_new <= best_c + 1e-14:
                    if abs(h_new.sum() - n/2.0) < 1e-10:
                        best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def perturb_h(h, n, rng, scale=0.01):
    h_p = h.copy()
    h_p += rng.normal(0, scale, size=n)
    return project_feasible(h_p, n)

def make_periodic_start(n, period, pattern):
    h = np.zeros(n)
    for i in range(n):
        h[i] = pattern[i % period]
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick L-BFGS from initial (100s)
    p1_end = min(t0 + 100, deadline - 800)
    h1, c1 = run_quick_lbfgs(best_h, best_n, p1_end,
        alphas=[100, 500, 2000, 10000, 50000, 200000, 1000000],
        mu=0.01, maxit_list=[1000, 1000, 1500, 2000, 2000, 2000, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
        best_n = best_n

    # Phase 2: Active-set optimization, progressive narrowing (300s)
    p2_end = min(t0 + 400, deadline - 500)
    h2, c2 = run_active_opt(best_h, best_n, p2_end,
        start_topk=15, end_topk=3,
        alphas=(10000, 50000, 200000, 1000000, 5000000, 20000000),
        mu=0.005, rounds_per_alpha=2, margin_frac=0.001)
    if c2 < best_c:
        best_c = c2
        best_h = h2
        best_n = best_n

    # Phase 3: Inequality-focused active optimization (150s)
    p3_end = min(t0 + 550, deadline - 350)
    h3, c3 = run_ineq_active(best_h, best_n, p3_end, topk=4)
    if c3 < best_c:
        best_c = c3
        best_h = h3
        best_n = best_n

    # Phase 4: Random perturbation + quick re-optimization (100s)
    p4_end = min(t0 + 650, deadline - 300)
    for scale in [0.02, 0.005]:
        if time.time() > p4_end - 30:
            break
        h_pert = perturb_h(best_h, best_n, rng, scale)
        c_pert = compute_c5(h_pert, best_n)
        h4, c4 = run_quick_lbfgs(h_pert, best_n, min(time.time() + 40, p4_end - 10),
            alphas=[10000, 100000, 1000000], mu=0.01, maxit_list=[1000, 1500, 2000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = best_n

    # Phase 5: SLSQP with explicit constraint (100s)
    p5_end = min(t0 + 750, deadline - 200)
    h5, c5 = run_slqp_phase(best_h, best_n, p5_end,
        alphas=(100000, 500000, 2000000, 10000000))
    if c5 < best_c:
        best_c = c5
        best_h = h5
        best_n = best_n

    # Phase 6: Subgradient refinement (80s)
    p6_end = min(t0 + 830, deadline - 150)
    h6, c6 = run_subgrad(best_h, best_n, p6_end, n_iter=8000, eta0=0.002)
    if c6 < best_c:
        best_c = c6
        best_h = h6
        best_n = best_n

    # Phase 7: Another round of active optimization (100s)
    p7_end = min(t0 + 930, deadline - 100)
    h7, c7 = run_active_opt(best_h, best_n, p7_end,
        start_topk=10, end_topk=2,
        alphas=(50000, 500000, 5000000, 50000000),
        mu=0.002, rounds_per_alpha=2, margin_frac=0.0005)
    if c7 < best_c:
        best_c = c7
        best_h = h7
        best_n = best_n

    # Phase 8: Pair search (120s)
    p8_end = min(t0 + 1050, deadline - 50)
    h8, c8 = run_pair_search(best_h, best_n, p8_end, radius=40)
    if c8 < best_c:
        best_c = c8
        best_h = h8
        best_n = best_n

    # Phase 9: Final SLSQP (40s)
    p9_end = deadline - 2
    h9, c9 = run_slqp_phase(best_h, best_n, p9_end,
        alphas=(500000, 5000000))
    if c9 < best_c:
        best_c = c9
        best_h = h9
        best_n = best_n

    # Final: tiny pair search
    if time.time() < deadline - 2:
        h_f, c_f = run_pair_search(best_h, best_n, deadline - 1, radius=20)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
            best_n = best_n

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
