```python
import time
import numpy as np
from scipy.optimize import minimize


def project_feasible(x, n):
    x = np.asarray(x, dtype=float).copy()
    if x.size != n:
        x = np.resize(x, n)
    target = 0.5 * n
    x = np.clip(x, 0.0, 1.0)

    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) <= 1e-12:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            if diff > 0.0:
                idx = np.argsort(1.0 - h)[:max(1, n // 10)]
            else:
                idx = np.argsort(h)[:max(1, n // 10)]
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return h


def corr_scaled(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(np.max(corr_scaled(h, n)))


def softmax_weights(c, alpha):
    m = float(c.max())
    z = alpha * (c - m)
    z = np.minimum(z, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, m
    w = ez / s
    return w, m


def grad_from_weights(w, h, n):
    L = 2 * n - 1
    W = w[::-1]
    cs = np.empty(2 * n, dtype=float)
    cs[0] = 0.0
    np.cumsum(W, out=cs[1:])

    t = np.arange(n, dtype=int)
    A = cs[2 * n - 1 - t] - cs[n - 1 - t]

    F = np.correlate(W, h, mode="valid")[::-1]
    E = np.correlate(W[::-1], h, mode="valid")[::-1]

    return A - F - E


def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(c.max())
    z = alpha * (c - m)
    z = np.minimum(z, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())

    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        val = m
    else:
        w = ez / s
        val = m + float(np.log(s)) / alpha

    g = grad_from_weights(w, h, n) * dx
    return float(val), g


def _pen_obj(x, n, alpha, lam):
    val, _ = smooth_obj_grad(x, n, alpha)
    s = x.sum() - 0.5 * n
    return float(val + lam * s * s)


def _pen_jac(x, n, alpha, lam):
    _, g = smooth_obj_grad(x, n, alpha)
    s = x.sum() - 0.5 * n
    return g + (2.0 * lam * s)


def make_bounds(n):
    return [(0.0, 1.0) for _ in range(n)]


def lbfgs_optimize(x, n, alpha, lam, maxiter):
    x = project_feasible(x, n)
    maxiter = int(max(0, maxiter))
    if maxiter <= 0:
        return x
    try:
        res = minimize(
            _pen_obj,
            x,
            jac=_pen_jac,
            args=(n, alpha, lam),
            method="L-BFGS-B",
            bounds=make_bounds(n),
            options={
                "maxiter": maxiter,
                "ftol": 1e-12,
                "gtol": 1e-8,
                "maxls": 20,
                "disp": False,
            },
        )
        y = np.asarray(res.x, dtype=float)
    except Exception:
        y = x
    return project_feasible(y, n)


def recompute_state(h, n, grad_alpha):
    c = corr_scaled(h, n)
    omh = 1.0 - h
    omh_rev = omh[::-1]
    best = float(c.max())
    w, _ = softmax_weights(c, float(grad_alpha))
    g = grad_from_weights(w, h, n) * (2.0 / n)
    return c, omh_rev, best, g


def pair_move_max(i, j, d, h, c, dx, n, omh_rev):
    L = 2 * n - 1
    delta = np.zeros(L, dtype=float)
    lin = d * dx
    d2 = d * d * dx

    delta[i:n + i] += lin * omh_rev
    delta[n - 1 - i:2 * n - 1 - i] -= lin * h

    delta[j:n + j] -= lin * omh_rev
    delta[n - 1 - j:2 * n - 1 - j] += lin * h

    delta[n - 1] -= 2.0 * d2
    diff = abs(i - j)
    if diff > 0:
        delta[n - 1 - diff] += d2
        delta[n - 1 + diff] += d2

    delta += c
    return float(delta.max())


def local_pair_search(h, n, deadline, deltas, grad_alpha=5000.0,
                      improve_tol=1e-12, filter_tol=1e-7):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n

    for d in deltas:
        if time.time() > deadline:
            break

        use_filter = d < 0.01
        pass_count = 0

        while time.time() < deadline - 0.5:
            c, omh_rev, best, g = recompute_state(h, n, grad_alpha)
            improved = False
            pair_check = 0

            for i in range(n - 1):
                if time.time() > deadline - 0.5:
                    break
                for j in range(i + 1, n):
                    pair_check += 1
                    if (pair_check & 1023) == 0 and time.time() > deadline - 0.5:
                        break

                    if use_filter:
                        score = g[i] - g[j]
                        if abs(score) <= filter_tol:
                            do_pos = True
                            do_neg = True
                        elif score < 0.0:
                            do_pos = True
                            do_neg = False
                        else:
                            do_pos = False
                            do_neg = True
                    else:
                        do_pos = True
                        do_neg = True

                    best_pair = best
                    move_i = -1
                    move_j = -1
                    move_d = 0.0

                    if do_pos:
                        ni = h[i] + d
                        nj = h[j] - d
                        if (ni >= -1e-12 and ni <= 1.0 + 1e-12 and
                                nj >= -1e-12 and nj <= 1.0 + 1e-12):
                            mv = pair_move_max(i, j, d, h, c, dx, n, omh_rev)
                            if mv < best_pair - improve_tol:
                                best_pair = mv
                                move_i = i
                                move_j = j
                                move_d = d

                    if do_neg:
                        ni = h[i] - d
                        nj = h[j] + d
                        if (ni >= -1e-12 and ni <= 1.0 + 1e-12 and
                                nj >= -1e-12 and nj <= 1.0 + 1e-12):
                            mv = pair_move_max(i, j, -d, h, c, dx, n, omh_rev)
                            if mv < best_pair - improve_tol:
                                best_pair = mv
                                move_i = i
                                move_j = j
                                move_d = -d

                    if move_i >= 0:
                        h[move_i] += move_d
                        h[move_j] -= move_d
                        h = project_feasible(h, n)
                        c, omh_rev, best, g = recompute_state(h, n, grad_alpha)
                        improved = True

                if time.time() > deadline - 0.5:
                    break

            pass_count += 1
            if not improved or pass_count > 20:
                break

    h = project_feasible(h, n)
    best = compute_c5(h, n)
    return h, best


def resample_step(h_old, n_old, n_new):
    if n_new == len(h_old):
        return np.asarray(h_old, dtype=float).copy()
    dx_old = 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    idx = np.floor(x_new / dx_old).astype(int)
    idx = np.clip(idx, 0, n_old - 1)
    return np.asarray(h_old, dtype=float)[idx].copy()


def default_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + 0.75 * np.sin(np.pi * x)
    return project_feasible(h, n)


def block_perturb(h, n, rng, scale, nblocks=None):
    h = np.asarray(h, dtype=float).copy()
    if nblocks is None:
        nblocks = max(1, n // 20)
    y = h.copy()
    for _ in range(nblocks):
        start = int(rng.randint(0, n))
        length = int(rng.randint(1, max(2, n // 20) + 1))
        end = min(n, start + length)
        if end > start:
            y[start:end] += rng.normal(0.0, scale, end - start)
    return project_feasible(y, n)


def template_start(n, idx, rng):
    if n < 2:
        return default_h(n)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    kind = idx % 5
    amp = 0.3 + 0.08 * ((idx // 5) % 7)

    if kind == 0:
        k = 1 + (idx // 5) % 6
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h = 0.5 + amp * np.sin(np.pi * k * x + phase)
    elif kind == 1:
        k = 1 + (idx % 10)
        h = 0.5 + 0.75 * np.cos(np.pi * k * x)
    elif kind == 2:
        nb = 2 + (idx % 8)
        cuts = np.sort(rng.randint(1, n, nb - 1))
        vals = rng.uniform(0.0, 1.0, nb)
        blocks = np.searchsorted(cuts, np.arange(n), side="left")
        h = vals[blocks]
    elif kind == 3:
        m = max(2, n // 8)
        nodes = np.linspace(0.0, 2.0, m)
        y = rng.uniform(0.0, 1.0, m)
        y = y - y.mean()
        h = 0.5 + 0.5 * np.interp(x, nodes, y)
    else:
        f = np.fft.rfft(rng.normal(size=n))
        keep = max(2, n // 15)
        if len(f) > keep:
            f[keep:] = 0.0
        if len(f) > 1:
            f[1] = 0.0
        sig = np.fft.irfft(f, n)
        sig = sig - sig.mean()
        mx = float(np.max(np.abs(sig)))
        if mx < 1e-12:
            h = np.full(n, 0.5)
        else:
            h = 0.5 + 0.85 * (sig / mx)
    return project_feasible(h, n)


def polish_candidate(h, n, deadline, local_s=30.0):
    h = project_feasible(np.asarray(h, dtype=float).copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    hcur = best_h.copy()
    ccur = best_c

    schedules = [
        (200.0, 50),
        (1000.0, 60),
        (5000.0, 50),
        (20000.0, 35),
    ]

    for alpha, base_iter in schedules:
        rem = deadline - local_s - 2.0 - time.time()
        if rem < 15.0:
            break
        est_cost = 0.04 + 2.0e-5 * n
        iters = max(20, min(base_iter, int(rem / max(1e-6, est_cost))))
        h2 = lbfgs_optimize(hcur, n, alpha, 1e4, iters)
        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < best_c - 1e-12:
            best_h = h2
            best_c = c2
        if c2 < ccur + 0.005:
            hcur = h2
            ccur = c2

    if local_s > 0.0 and time.time() < deadline - 2.0:
        local_deadline = min(deadline - 1.0, time.time() + local_s)
        dlist = [0.02, 0.005, 0.001, 0.0003]
        h2, c2 = local_pair_search(best_h, n, local_deadline, dlist)
        if c2 < best_c - 1e-13:
            best_h = h2
            best_c = c2

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 6.0
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.asarray(initial_h_values, dtype=float).copy()
        n_parent = int(len(h_parent))
    except Exception:
        n_parent = 125
        h_parent = default_h(n_parent)

    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(best_h, n_parent)

    try:
        h_sym = 0.5 * (h_parent + h_parent[::-1])
        h_sym = project_feasible(h_sym, n_parent)
        c_sym = compute_c5(h_sym, n_parent)
        if c_sym < best_c - 1e-12:
            best_h = h_sym
            best_c = c_sym

        total_budget = max(60.0, float(budget_s))
        final_reserve = min(300.0, max(60.0, 0.22 * total_budget))
        candidate_deadline = deadline - final_reserve
        if candidate_deadline < t0 + 45.0:
            candidate_deadline = deadline - 12.0

        stage1_deadline = min(candidate_deadline - 4.0, t0 + 210.0)
        if time.time() < stage1_deadline - 25.0:
            h1, c1 = polish_candidate(best_h, len(best_h), stage1_deadline, local_s=75.0)
            if c1 < best_c - 1e-13:
                best_h = h1
                best_c = c1

        n_curr = len(best_h)
        base_n = n_parent
        alt_ns = []
        for nn in [base_n // 2, base_n // 4, 100, 125, 150]:
            if 20 <= nn < base_n and nn not in alt_ns:
                alt_ns.append(nn)
        for nn in alt_ns[:3]:
            if time.time() >= candidate_deadline - 25.0:
                break
            h_try = resample_step(best_h, len(best_h), nn)
            alloc = min(70.0, max(25.0, (candidate_deadline - time.time()) * 0.20))
            h2, c2 = polish_candidate(h_try, nn, time.time() + alloc, local_s=22.0)
            if c2 < best_c - 1e-13:
                best_h = h2
                best_c = c2

        attempt = 0
        while time.time() < candidate_deadline - 20.0:
            n_use = len(best_h)
            if n_use < 20:
                break

            kind = attempt % 4
            if kind == 0:
                scale = 0.02 + 0.035 * (attempt % 6)
                h_try = best_h + rng.normal(0.0, scale, n_use)
            elif kind == 1:
                h_try = block_perturb(best_h, n_use, rng, 0.08 + 0.045 * (attempt % 4))
            elif kind == 2:
                h_try = template_start(n_use, attempt + rng.randint(0, 8), rng)
            else:
                h_try = (best_h + rng.normal(0.0, 0.05, n_use) > 0.5).astype(float)

            h_try = project_feasible(h_try, n_use)
            c_init = compute_c5(h_try, n_use)

            if c_init < best_c + 0.05:
                rem = candidate_deadline - time.time()
                alloc = min(45.0, max(14.0, rem * 0.14))
                local_s = 10.0 if attempt < 10 else 4.0
                h2, c2 = polish_candidate(h_try, n_use, time.time() + alloc, local_s=local_s)
                if c2 < best_c - 1e-13:
                    best_h = h2
                    best_c = c2

            attempt += 1

        n_best = len(best_h)
        if time.time() < deadline - 80.0:
            rem = deadline - time.time()
            iters = max(25, min(90, int(rem / 0.5)))
            h2 = lbfgs_optimize(best_h, n_best, 20000.0, 1e4, iters)
            h2 = project_feasible(h2, n_best)
            c2 = compute_c5(h2, n_best)
            if c2 < best_c - 1e-13:
                best_h = h2
                best_c = c2

        if time.time() < deadline - 6.0:
            dlist = [0.01, 0.004, 0.0015, 0.0006, 0.0002, 0.0001]
            best_h, best_c = local_pair_search(best_h, n_best, deadline - 3.0, dlist)

    except Exception:
        pass

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    if not np.isfinite(best_c):
        best_h = project_feasible(np.full(n_final, 0.5), n_final)
        best_c = compute_c5(best_h, n_final)

    return best_h, float(best_c), int(n_final)
```
