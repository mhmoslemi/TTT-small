import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # cmax = m + max of (c-m)
    # smooth approx: m + (1/alpha) * log(sum(exp(alpha*(c-m))))
    max_sh = shifted.max()
    if max_sh > 20:
        val = (max_sh + np.log(np.sum(np.exp(shifted - max_sh)))) / alpha
    else:
        val = np.log(np.sum(np.exp(shifted))) / alpha
    return float(m + val)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def coord_descent(h, n, deadline, n_passes=50):
    dx = 2.0 / n
    shifts = np.arange(1 - n, n)  # length 2n-1
    best_c = compute_c5(h, n)
    
    for _pass in range(n_passes):
        if time.time() > deadline: break
        improved = False
        for i in range(n):
            if time.time() > deadline: break
            old = h[i]
            c_all = np.correlate(h, 1-h, 'full') * dx
            
            # Compute slope for each lag
            slope = np.zeros(2*n-1)
            # h[i] as h[j] (j=i): contributes -h[i+s]*dx if 0<=i+s<n
            idx_a = i + shifts
            mask_a = (idx_a >= 0) & (idx_a < n)
            slope[mask_a] -= h[idx_a[mask_a]] * dx
            # h[i] as h[j+s] (j+s=i, j=i-s): contributes (1-h[i-s])*dx if 0<=i-s<n
            idx_b = i - shifts
            mask_b = (idx_b >= 0) & (idx_b < n)
            slope[mask_b] += (1 - h[idx_b[mask_b]]) * dx
            
            # Ternary search for v in [0,1] minimizing max(c_all + (v-old)*slope)
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi-lo)*0.333
                m2 = hi - (hi-lo)*0.333
                f1 = np.max(c_all + (m1-old)*slope)
                f2 = np.max(c_all + (m2-old)*slope)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            v_opt = (lo+hi)*0.5
            
            if abs(v_opt - old) > 1e-10:
                h[i] = v_opt
                improved = True
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
        if not improved:
            break
    return h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline: break
        for _pass in range(200):
            if time.time() > deadline: break
            improved = False
            for i in range(n):
                if time.time() > deadline: return h, best_c
                for j in range(i+1, n):
                    hi, hj = h[i]+d, h[j]-d
                    if 0<=hi<=1 and 0<=hj<=1:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

                            improved = True
                        else:
                            h[i], h[j] = hi-d, hj+d
                    hi, hj = h[i]-d, h[j]+d
                    if 0<=hi<=1 and 0<=hj<=1:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved: break
    return h, best_c

def simulated_annealing(h, n, deadline, t_start=0.01, t_end=0.0001):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    dx = 2.0/n
    t0 = time.time()
    total_time = max(deadline - t0, 1)
    
    while time.time() < deadline:
        progress = (time.time() - t0) / total_time
        T = t_start * (t_end/t_start) ** progress
        if T < t_end: T = t_end
        
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        d = np.random.uniform(-0.05, 0.05)
        
        hi, hj = h[i]+d, h[j]-d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1: continue
        
        h[i], h[j] = hi, hj
        new_c = compute_c5(h, n)
        
        delta = new_c - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = hi-d, hj+d
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 30:
        val = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = np.log(np.sum(np.exp(shifted))) / alpha
    return float(m + val)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_slopes(h, n):
    dx = 2.0 / n
    shifts = np.arange(1 - n, n)
    slopes = np.zeros((n, 2 * n - 1))
    for i in range(n):
        idx1 = i - shifts
        mask1 = (idx1 >= 0) & (idx1 < n)
        slopes[i, mask1] += (1.0 - h[idx1[mask1]]) * dx
        idx2 = i + shifts
        mask2 = (idx2 >= 0) & (idx2 < n)
        slopes[i, mask2] -= h[idx2[mask2]] * dx
    return slopes

def coord_descent(h, n, deadline, n_passes=200):
    dx = 2.0 / n
    shifts = np.arange(1 - n, n)
    best_c = compute_c5(h, n)

    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = h[i]
            c_all = np.correlate(h, 1 - h, 'full') * dx
            slope = np.zeros(2 * n - 1)
            idx1 = i - shifts
            mask1 = (idx1 >= 0) & (idx1 < n)
            slope[mask1] += (1.0 - h[idx1[mask1]]) * dx
            idx2 = i + shifts
            mask2 = (idx2 >= 0) & (idx2 < n)
            slope[mask2] -= h[idx2[mask2]] * dx

            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) * 0.3333
                m2 = hi - (hi - lo) * 0.3333
                f1 = np.max(c_all + (m1 - old) * slope)
                f2 = np.max(c_all + (m2 - old) * slope)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            v_opt = (lo + hi) * 0.5
            v_opt = max(0.0, min(1.0, v_opt))

            if abs(v_opt - old) > 1e-12:
                h[i] = v_opt
                improved = True

        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
        if not improved:
            break
    return h, best_c

def stochastic_search(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    t_start = 0.02
    t_end = 0.00005
    t0 = time.time()
    total = max(deadline - t0, 1.0)

    while time.time() < deadline:
        progress = (time.time() - t0) / total
        T = t_start * (t_end / t_start) ** min(progress, 1.0)
        if T < t_end:
            T = t_end
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.03, 0.03)
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        h[i], h[j] = hi, hj
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c - 1e-13:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = hi - d, hj + d
    return best_h, best_c

    except Exception:
        n_init = 200
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    n_best = n_init

    # Phase 1: SLSQP + coord descent on initial resolution
    if time.time() < deadline - 250:
        alloc1 = (deadline - t0) * 0.25
        best_h, best_c = optimize_resolution(h_init, n_init, t0 + alloc1, rng)
        n_best = n_init

    # Phase 2: Try other resolutions
    x_init = np.linspace(0, 2, n_best)
    trial_ns = [150, 250, 300]
    if n_init != 150 and n_init != 250 and n_init != 300:
        pass
    for n_try in trial_ns:
        if time.time() > deadline - 150:
            break
        if n_try == n_best:
            continue
        remaining = deadline - time.time() - 100
        if remaining < 30:
            break
        alloc = min(100, remaining * 0.4)
        if alloc < 25:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_best = n_try
            x_init = x_new

    # Phase 3: Stochastic search (simulated annealing style)
    if time.time() < deadline - 120:
        alloc_sa = min(200, (deadline - time.time()) * 0.35)
        best_h, best_c = stochastic_search(best_h, n_best, time.time() + alloc_sa, rng)
        n_best = len(best_h)

    # Phase 4: Post-SA optimization
    if time.time() < deadline - 60:
        for a in [50000, 200000]:
            if time.time() > deadline - 60:
                break
            ht = slsqp_optimize(best_h, n_best, a, maxiter=2000, ftol=1e-16)
            c = compute_c5(ht, n_best)
            if c < best_c:
                best_c = c
                best_h = ht
        if time.time() < deadline - 40:
            best_h, best_c = coord_descent(best_h, n_best, deadline - 30, n_passes=100)

    # Phase 5: Fine pair search
    if time.time() < deadline - 20:
        best_h, best_c = fast_pair_search(best_h, n_best, deadline - 10,
                                          [0.005, 0.002, 0.001, 0.0005, 0.0002])

    # Phase 6: Final coordinate descent polish
    if time.time() < deadline - 5:
        best_h, best_c = coord_descent(best_h, n_best, deadline - 2, n_passes=80)

    # Phase 7: One more very fine pair search
    if time.time() < deadline - 2:
        best_h, best_c = fast_pair_search(best_h, n_best, deadline - 1,
                                          [0.001, 0.0005, 0.0002, 0.0001])

    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)

    return best_h, best_c, len(best_h)
