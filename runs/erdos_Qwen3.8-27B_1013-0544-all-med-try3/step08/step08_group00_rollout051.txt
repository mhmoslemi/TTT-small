import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    n_sh = 2 * n - 1
    for mm in range(n_sh):
        wi = w[mm]
        if wi < 1e-18:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    return grad

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[:nv] = (1.0 - h[k:]) * dx
            grad[k:] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:] = (1.0 - h[:nv]) * dx
            grad[:nv] += (-h[kk:]) * dx
    return grad

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.03, top_k=25):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Active set: use a wider tolerance and more constraints
        thresh = max_val - 0.004 * max_val
        active_mask = c_vec >= thresh
        active = np.where(active_mask)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c_vec))])
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        
        n_act = len(active)
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0
        total_ineq = n_act + 4 * n
        A_ub = np.zeros((total_ineq, n_var))
        b_ub = np.zeros(total_ineq)
        row = 0
        for idx in active:
            g = shift_grad_single(h, n, idx)
            A_ub[row, :n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = -c_vec[idx]
            row += 1
        # Bounds: h[i] >= 0 and h[i] <= 1, and |delta| <= rho
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = rho
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = rho
            row += 1
        A_eq = np.ones((1, n_var))
        A_eq[0, n] = 0.0
        b_eq = np.zeros(1)
        bounds = [(None, None)] * n + [(0.0, 2.0)]
        delta = None
        try:

            best_h = h_new.copy()
            h = h_new

            no_improve += 1
            rho *= 0.5

    return best_h, best_c

            break
        improved = True
        while improved and time.time() < deadline - 2:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - 0.002 * max_val
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(c_vec))])
            active = active[:min(10, len(active))]
            
            # Compute average gradient over active set
            grad = np.zeros(n)
            for idx in active:
                grad += shift_grad_single(h, n, idx)
            grad /= len(active)
            
            # Find best pairs to swap
            # Positive gradient → decrease h there
            # Negative gradient → increase h there
            pos_idx = np.argsort(-grad)  # most positive first
            neg_idx = np.argsort(grad)   # most negative first
            
            n_pos = min(60, n)
            n_neg = min(60, n)
            
            for pi in pos_idx[:n_pos]:
                if time.time() > deadline - 2:
                    return best_h, best_c
                i = int(pi)
                if h[i] < 1e-6:
                    continue

                        continue
                    if h[j] > 1 - 1e-6:
                        continue

                    h[j] += step

                        best_h = h.copy()

                    else:
                        h[i] += step
                        h[j] -= step
