            h = np.clip(h, 0.0, 1.0)

- minimize t subject to c_k ≤ t for all k, 0 ≤ h ≤ 1, sum(h) = n/2
- This is a QP (since c_k is quadratic in h)
- Can be solved with cvxpy if the quadratic form is PSD... but it's not in general

The quadratic form for c_k = sum_i h[i] - sum_i h[i]h[i+k]. The term -sum_i h[i]h[i+k] has a symmetric matrix that's a banded matrix with -1 on the k-th diagonal. This matrix is indefinite in general, so it's a non-convex QP.

However, I could use a bisection on t:
- For a given t, check feasibility: does there exist h with c_k ≤ t for all k?
- This is still a non-convex feasibility problem.

Alternatively, I could use a "barrier" approach: minimize sum_k log(1/c_k - 1/t) for the active constraints... but this is getting complicated.

One more idea: **coordinate descent on the correlation values**. Instead of optimizing h directly, I could:
1. Pick the worst correlation k*
2. Gradient descent on h to reduce c_{k*} specifically (while keeping others from increasing too much)
3. Iterate

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
            h = np.clip(h, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-12:
            continue
        k = k_idx - (n - 1)
        wm = w[k_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_lbfgs(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_lbfgs(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_slqp(x, n, alpha):
    c = compute_c5_vec(x, n)
    return float(smooth_max_val(c, alpha))

def cons_fun(x, n):
    return x.sum() - n / 2.0

def con_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    mus = [10.0, 50.0, 100.0, 500.0, 1000.0, 5000.0, 10000.0, 20000.0]
    n_alphas = min(len(alphas), max(2, int((deadline - time.time()) / 20)))
    for i in range(n_alphas):
        if time.time() > deadline - 2:
            break
        alpha = alphas[i]
        mu = mus[i]
        remaining = deadline - time.time() - 2
        iters = min(max_iters, max(20, int(remaining * 3)))
        try:
            res = minimize(obj_lbfgs, best_h, jac=grad_lbfgs,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': iters * 12,
                                   'maxls': 35, 'm': 25})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def slqp_optimize(h, n, deadline, alpha=500000, maxiter=1500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': con_jac, 'args': (n,)}]
    remaining = deadline - time.time() - 2
    iters = min(maxiter, max(50, int(remaining * 2)))
    try:
        res = minimize(obj_slqp, best_h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': iters, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    except Exception:
        pass
    return best_h, best_c

def nesterov_optimize(h, n, deadline, lr_init=0.003, alpha=5000.0, momentum=0.85):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    stuck = 0
    for it in range(80000):
        if time.time() > deadline:
            break
        x = h + v * momentum
        x = np.clip(x, 0.0, 1.0)
        g = smooth_max_grad(x, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        v = momentum * v - lr * g / (g_norm + 1e-10)
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stuck = 0
        else:
            h = h_new
            stuck += 1
            if stuck > 80:
                lr *= 0.85
                v *= 0.5
                stuck = 0
                if lr < 1e-12:
                    break
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-14 and h[j] - d >= -1e-14:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-14 and h[j] + d <= 1.0 + 1e-14:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                        if time.time() > deadline:
                            return h, best_c
    return h, best_c

def gradient_pair_search(h, n, deadline, step_size=0.003, alphas=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            g = smooth_max_grad(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    if (h[i] + move >= -1e-14 and h[i] + move <= 1.0 + 1e-14 and
                        h[j] - move >= -1e-14 and h[j] - move <= 1.0 + 1e-14):
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
    return h, best_c

def interpolate_to(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def random_perturb(h, n, rng, scale=0.01):
    h_p = h.copy()
    h_p += rng.normal(0, scale, size=n)
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

        h_init = np.full(300, 0.5)

        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 400:
        t2_end = min(time.time() + 70, deadline - 400)
        h2, c2 = slqp_optimize(best_h, best_n, t2_end, alpha=500000, maxiter=1500)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Try higher resolutions
    resolutions_to_try = []
    double_n = n_init * 2
    triple_n = n_init * 3
    if double_n != n_init and double_n <= 2000:
        resolutions_to_try.append(double_n)
    if triple_n != n_init and triple_n <= 2000 and triple_n != double_n:
        resolutions_to_try.append(triple_n)
    # Also try intermediate

    if mid_n != n_init and mid_n != double_n and mid_n != triple_n and mid_n <= 2000:
