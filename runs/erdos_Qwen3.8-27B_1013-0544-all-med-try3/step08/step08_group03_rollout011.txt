def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    target = 0.5 * float(n)
    y = np.clip(y, 0.0, 1.0)
    s = float(y.sum())
    if abs(s - target) < 1e-13:
        return y

    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(6):
        diff = target - float(y.sum())
        if abs(diff) < 1e-14:
            break
        idx = np.where((y > 1e-12) & (y < 1.0 - 1e-12))[0]
        if len(idx) > 0:
            amt = diff / float(len(idx))
            y[idx] += amt
            y = np.clip(y, 0.0, 1.0)
        else:
            if diff > 0:
                ii = int(np.argmax(1.0 - y))
                y[ii] = min(1.0, y[ii] + diff)
            else:
                ii = int(np.argmin(y))
                y[ii] = max(0.0, y[ii] + diff)
    return y


def c_max_sum(h):
    n = len(h)
    if n == 0:
        return 0.0
    if n == 1:
        return float(h[0] * (1.0 - h[0]))

    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(h, N)
    ac = np.fft.irfft(fa * np.conjugate(fa), N)[:n]

    P = np.empty(n + 1, dtype=float)
    P[0] = 0.0
    P[1:] = np.cumsum(h)

    idx = np.arange(n)
    D = P[n] - P[idx] - ac
    E = P[n - idx] - ac

    dm = float(D.max())
    em = float(E.max())
    return max(dm, em)


def fast_c5(h):
    n = len(h)
    if n == 0:
        return float("inf")
    return float(c_max_sum(h) * (2.0 / n))


def harness_c5(h):
    n = len(h)
    if n == 0:
        return float("inf")
    v = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(v) * (2.0 / n))


def c_weights(h, alpha):
    n = len(h)
    if n == 1:
        c0 = float(h[0] * (1.0 - h[0]))
        return c0, 1.0, np.zeros(0, dtype=float), np.zeros(0, dtype=float)

    alpha = float(alpha)
    if alpha <= 0.0 or not np.isfinite(alpha):
        alpha = 1.0

    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(h, N)
    ac = np.fft.irfft(fa * np.conjugate(fa), N)[:n]

    P = np.empty(n + 1, dtype=float)
    P[0] = 0.0
    P[1:] = np.cumsum(h)

    idx = np.arange(n)
    D = P[n] - P[idx] - ac
    E = P[n - idx] - ac

    maxc = max(float(D.max()), float(E.max()))
    shift = maxc

    dd = np.exp(alpha * (D[1:] - shift))
    ee = np.exp(alpha * (E[1:] - shift))
    d0 = np.exp(alpha * (D[0] - shift))

    total = float(dd.sum() + ee.sum() + d0)
    if total <= 0.0 or not np.isfinite(total):
        total = 1.0
        d0 = 1.0
        dd = np.zeros(n - 1, dtype=float)
        ee = np.zeros(n - 1, dtype=float)

    F = shift + math.log(total) / alpha
    w0 = float(d0) / total
    wD = dd / total
    wE = ee / total
    return F, w0, wD, wE


def grad_from_weights(h, w0, wD, wE):
    n = len(h)
    if n == 1:
        return np.array([float(w0) * (1.0 - 2.0 * h[0])], dtype=float)

    A = np.zeros(n, dtype=float)
    A[1:] = wD + wE

    K = np.zeros(2 * n - 1, dtype=float)
    K[:n - 1] = A[1:][::-1]
    K[n:] = A[1:]

    if n < 700:
        conv = np.convolve(h, K, mode="full")[n - 1:n - 1 + n]
    else:
        need = 3 * n - 1
        N = 1
        while N < need:
            N <<= 1
        conv = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(K, N), N)[n - 1:n - 1 + n]

    prefD = np.cumsum(wD)
    prefE = np.cumsum(wE)

    const = np.zeros(n, dtype=float)
    const[1:] = prefD
    const[:n - 1] = prefE[::-1]

    return const - conv + float(w0) * (1.0 - 2.0 * h)


def c_weights_gradient(h, alpha):
    F, w0, wD, wE = c_weights(h, alpha)
    g = grad_from_weights(h, w0, wD, wE)
    return F, g


def _al_value(x, n, alpha, lam, rho):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    F, _, _, _ = c_weights(x, alpha)
    s = float(x.sum() - 0.5 * n)
    return float(F + lam * s + 0.5 * rho * s * s)


def _al_grad(x, n, alpha, lam, rho):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    _, g = c_weights_gradient(x, alpha)
    s = float(x.sum() - 0.5 * n)
    return g + (lam + rho * s)

def _expit(z):
    z = np.asarray(z, dtype=float)
    z = np.clip(z, -60.0, 60.0)

        if float(h.sum()) < target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

    return float(F)


def _sig_grad(u, n, alpha):

    beta = solve_beta(u, n)

    ss = float(s.sum())
    if ss < 1e-14:
        return np.zeros_like(u)
    m = float(np.dot(s, g)) / ss
    return s * (g - m)

def pair_gradient_search(h, n, deadline, alphas=None, top=8, base_steps=None):
    if time.time() >= deadline:
        h0 = project_feasible(h, n)
        return h0, fast_c5(h0)

    if alphas is None:
        alphas = [100.0, 400.0, 1500.0]
    if base_steps is None:
        base_steps = [0.05, 0.02, 0.01, 0.005]

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = fast_c5(h)

    for alpha in alphas:
        for base in base_steps:
            count = 0
            while time.time() < deadline and count < 120:
                try:
                    _, g = c_weights_gradient(h, float(alpha))
                except Exception:
                    break

                if not np.all(np.isfinite(g)) or np.max(np.abs(g)) < 1e-14:
                    break

                order = np.argsort(g)
                lows = list(order[:top])
                highs = list(order[-top:][::-1])

                improved = False
                for i in lows:
                    ii = int(i)
                    for j in highs:
                        jj = int(j)
                        if ii == jj:
                            continue
                        if g[ii] >= g[jj] - 1e-15:
                            continue
                        tmax = min(float(base), 1.0 - h[ii], h[jj])
                        t = tmax
                        accepted = False
                        for _ in range(5):
                            if t < 1e-10:
                                break
                            oi, oj = h[ii], h[jj]
                            h[ii] = oi + t
                            h[jj] = oj - t
                            cnew = fast_c5(h)
                            if cnew < best_c - 1e-14:
                                best_c = cnew
                                best_h = h.copy()
                                improved = True
                                accepted = True
                                break
                            h[ii] = oi
                            h[jj] = oj
                            t *= 0.5
                        if accepted:
                            break
                    if improved:
                        break

                if not improved:
                    break
                count += 1

    best_h = project_feasible(best_h, n)
    best_c = fast_c5(best_h)
    return best_h, best_c


def shift_search(h, n, deadline):

        return h, fast_c5(h)

    h = project_feasible(h, n)

        cr = fast_c5(hr)

def interpolate_to(h, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h), n_new)


def random_fourier_h(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    amps = [1.0, 0.6, 0.6, 0.35, 0.35, 0.2, 0.2]
    h = np.zeros(n, dtype=float)
    h += rng.normal(0.0, 0.15)
    for m in range(1, 4):
        h += rng.normal(0.0, amps[m * 2 - 1]) * np.sin(np.pi * m * x + rng.uniform(-math.pi, math.pi))
        h += rng.normal(0.0, amps[m * 2]) * np.cos(np.pi * m * x + rng.uniform(-math.pi, math.pi))
    h = 0.5 + 0.5 * np.tanh(h)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def binary_c5_mask(mask, n, full=None):
    if full is None:
        full = (1 << n) - 1
    mask &= full
    mask0 = full ^ mask
    best = 0
    for k in range(1, n):
        a = (mask0 & (mask >> k)).bit_count()
        if a > best:
            best = a
        b = (mask & (mask0 >> k)).bit_count()
        if b > best:
            best = b
    return float(best * (2.0 / n))


def h_to_mask(h, n):
    m = 0
    for i in range(n):
        if h[i] > 0.5:
            m |= (1 << i)
    return m


def mask_to_h(mask, n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        if (mask >> i) & 1:
            h[i] = 1.0
    return h


def make_binary_from_h(h, n):
    order = np.argsort(h)
    b = np.zeros(n, dtype=float)
    if n % 2 == 0:
        b[order[n // 2:]] = 1.0
    else:
        m = n // 2
        b[order[m:n - m]] = 0.5
        b[order[n - m:]] = 1.0
    return b


def binary_polish_mask(mask, n, deadline, rng, rounds_per_phase=240, max_restarts=3):
    if time.time() >= deadline or n % 2 != 0:
        h = project_feasible(mask if isinstance(mask, np.ndarray) else mask_to_h(mask, n), n)
        return h, fast_c5(h)

    full = (1 << n) - 1
    mask &= full
    current_mask = mask
    current_c = binary_c5_mask(current_mask, n, full)
    best_mask = current_mask
    best_c = current_c

    alphas = [80.0, 350.0, 1200.0]
    tops = [8, 14, 24]
    bits = [1 << i for i in range(n)]

    for restart in range(max_restarts + 1):
        if time.time() >= deadline:
            break
        no_improve = 0
        alpha = alphas[min(restart, len(alphas) - 1)]
        top = tops[min(restart, len(tops) - 1)]

        for _ in range(rounds_per_phase):
            if time.time() >= deadline:
                break

            hcur = mask_to_h(current_mask, n)
            try:
                _, g = c_weights_gradient(hcur, float(alpha))
            except Exception:
                break

            z = np.where(hcur < 0.5)[0]
            o = np.where(hcur > 0.5)[0]
            if len(z) == 0 or len(o) == 0:
                break

            zc = z[np.argsort(g[z])[:top]]
            oc = o[np.argsort(-g[o])[:top]]

            extra = min(3, min(len(z), len(o)))
            if extra > 0:
                zr = rng.choice(z, extra, replace=False)
                ov = rng.choice(o, extra, replace=False)
                zc = np.concatenate((zc, zr))
                oc = np.concatenate((oc, ov))

            best_new = float("inf")
            bi = -1
            bj = -1

            for i in zc:
                ii = int(i)
                if current_mask & bits[ii]:
                    continue
                biti = bits[ii]
                for j in oc:
                    jj = int(j)
                    if ii == jj or not (current_mask & bits[jj]):
                        continue
                    m2 = (current_mask & ~bits[jj]) | biti
                    c2 = binary_c5_mask(m2, n, full)
                    if c2 < best_new:
                        best_new = c2
                        bi = ii
                        bj = jj

            applied = False
            if bi != -1 and best_new < current_c - 1e-15:
                current_mask = (current_mask & ~bits[bj]) | bits[bi]
                current_c = best_new
                if current_c < best_c - 1e-15:
                    best_c = current_c
                    best_mask = current_mask
                no_improve = 0
                applied = True

            if not applied:
                no_improve += 1
                for _r in range(80):
                    if time.time() >= deadline:
                        break
                    ii = int(rng.choice(z))
                    jj = int(rng.choice(o))
                    if ii == jj or (current_mask & bits[ii]) or not (current_mask & bits[jj]):
                        continue
                    m2 = (current_mask & ~bits[jj]) | bits[ii]
                    c2 = binary_c5_mask(m2, n, full)
                    if c2 < current_c - 1e-15:
                        current_mask = m2
                        current_c = c2
                        if current_c < best_c - 1e-15:
                            best_c = current_c
                            best_mask = current_mask
                        no_improve = 0
                        applied = True
                        break

                if not applied and no_improve > 6:
                    break

        if time.time() >= deadline:
            break

        if restart < max_restarts:
            current_mask = best_mask
            current_c = best_c
            hcur = mask_to_h(current_mask, n)
            z = np.where(hcur < 0.5)[0]
            o = np.where(hcur > 0.5)[0]
            pert = max(5, min(40, n // 8))
            for _ in range(pert):
                if time.time() >= deadline:
                    break
                if len(z) == 0 or len(o) == 0:
                    break
                ii = int(rng.choice(z))
                jj = int(rng.choice(o))
                if ii == jj:
                    continue
                current_mask = (current_mask & ~bits[jj]) | bits[ii]
                hcur = mask_to_h(current_mask, n)
                z = np.where(hcur < 0.5)[0]
                o = np.where(hcur > 0.5)[0]
            current_c = binary_c5_mask(current_mask, n, full)

    best_h = mask_to_h(best_mask, n)
    return best_h, binary_c5_mask(best_mask, n, full)


def random_balanced_mask(n, rng):
    if n % 2 != 0:
        return None
    idx = rng.choice(n, n // 2, replace=False)
    m = 0
    for i in idx:
        m |= (1 << int(i))
    return m


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

        h5, c5a = shift_search(best_h, len(best_h), end5)
        best_h, best_c = update_best(best_h, best_c, h5, len(h5))

    n_base = len(best_h)
    cand_ns = []
    for nn in [n_base, n_base + 50, n_base + 100, 512, 640, 720]:
        if nn >= 160 and nn <= 1100 and nn % 2 == 0 and nn not in cand_ns:

    cand_ns = [nn for nn in cand_ns if nn != n_base][:3]

    if len(cand_ns) > 0 and time.time() < deadline - 220:
        total_budget = min(220.0, max(60.0, deadline - time.time() - 220.0))
        per = total_budget / float(len(cand_ns))
        for nn in cand_ns:
            if time.time() >= deadline - 80:
                break
            per_deadline = min(deadline - 80, time.time() + max(30.0, per * 0.95))
            if per_deadline < time.time() + 30:

            hinit = interpolate_to(best_h, len(best_h), nn)
            h_alt = random_fourier_h(nn, rng)
            h_alt = interpolate_to(best_h, len(best_h), nn)

            hn, cn = al_polish(hinit, nn, per_deadline, alphas=[20.0, 100.0, 400.0, 1500.0], max_inner=16)
            best_h, best_c = update_best(best_h, best_c, hn, nn)

            if time.time() < per_deadline - 10:
                hn2, cn2 = pair_gradient_search(hn, nn, per_deadline, alphas=[300.0, 1000.0], top=6, base_steps=[0.03, 0.01, 0.004])
                best_h, best_c = update_best(best_h, best_c, hn2, nn)

    if time.time() < deadline - 150:

    best_h = project_feasible(best_h, len(best_h))

    try:
        final_c = float(harness_c5(best_h))

            pc = float("inf")
        if pc < final_c - 1e-12:
            best_h, final_c = hp.copy(), pc

    return best_h, float(final_c), int(len(best_h))
