import numpy as np
from scipy.optimize import minimize
import time

_TOP_N = 0
_TOP_X = None
_TOP_R = 0
_TOP_ALPHA = 20.0

_GR_N = 0
_GR_ALPHA = 20.0


def _midgrid(n):
    return (np.arange(n) + 0.5) * (2.0 / float(n))


def compute_f_from_a(a, n, center=False):
    a = np.asarray(a, dtype=float).reshape(-1)
    if a.size < n:
        a = np.pad(a, (0, n - a.size), mode='constant')
    elif a.size > n:
        a = a[:n]
    a = np.nan_to_num(a, nan=0.0, posinf=1.0, neginf=-1.0)
    if center:
        a = a - float(a.mean())

    S = np.empty(n + 1, dtype=float)
    S[0] = 0.0
    S[1:] = np.cumsum(a)

    ac = np.correlate(a, a, mode="full")
    R = ac[n - 1:n - 1 + n].astype(float)

    k = np.arange(n, dtype=float)
    L = float(n) - k
    idx = np.arange(n, dtype=int)
    Ssum = S[L.astype(np.int64)] + S[idx]

    f = (L + np.abs(Ssum) - R) / (2.0 * float(n))
    f = np.nan_to_num(f, nan=0.5, posinf=1.0, neginf=0.0)
    return f, float(np.max(f))


def compute_c5(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size < n:
        h = np.pad(h, (0, n - h.size), mode='constant')
    elif h.size > n:
        h = h[:n]
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def project_a(v):
    v = np.asarray(v, dtype=float).reshape(-1)
    n = v.size
    if n == 0:
        return v
    v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=-1.0)

    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        x = np.clip(v - tau, -1.0, 1.0)
        s = float(x.sum())
        if s > 0.0:
            lo = tau
        else:
            hi = tau

    x = np.clip(v - 0.5 * (lo + hi), -1.0, 1.0)
    diff = -float(x.sum())
    for _ in range(20):
        if abs(diff) <= 1e-15:
            break
        mask = (x > -1.0 + 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        x[mask] += diff / float(cnt)
        x = np.clip(x, -1.0, 1.0)
        diff = -float(x.sum())

    if abs(diff) > 1e-12:
        x = x + diff / float(n)
        x = np.clip(x, -1.0, 1.0)
    return x


def enforce_sum_h(h, n):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    if h.size < n:
        h = np.pad(h, (0, n - h.size), mode='constant')
    elif h.size > n:
        h = h[:n]
    h = np.clip(np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0), 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(1000):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = float(room.sum())
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = float(h.sum())
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    h = np.clip(h, 0.0, 1.0)

    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.count_nonzero(mask) > 0:
            h[mask] += diff / float(np.count_nonzero(mask))
            h = np.clip(h, 0.0, 1.0)
    return h


def a_to_h_feasible(a, n):
    a = project_a(a)
    h = 0.5 + 0.5 * a
    return enforce_sum_h(h, n)


def sum_zero_fun(x):
    return float(np.sum(x))


def sum_zero_jac(x):
    return np.ones_like(x, dtype=float)


def smooth_obj_a(x):
    n = _GR_N
    alpha = _GR_ALPHA
    f, _ = compute_f_from_a(x, n, center=True)
    m = float(np.max(f))
    z = np.exp(alpha * (f - m))
    return float(m + (float(np.log(float(z.sum()))) / alpha))


def smooth_grad_a(x):
    n = _GR_N
    alpha = _GR_ALPHA
    f, _ = compute_f_from_a(x, n, center=True)
    m = float(np.max(f))
    z = np.exp(alpha * (f - m))

    if n <= 300 or alpha <= 30.0:
        M = n
    else:
        M = int(max(12, min(n, 5000.0 / alpha)))

    zmax = float(np.max(z))
    thresh = max(1e-16, zmax * 1e-8)
    act = np.where(z >= thresh)[0]
    if act.size == 0:
        act = np.array([int(np.argmax(f))], dtype=int)
    if act.size > M:
        order = np.argsort(f[act])
        act = act[order[len(order) - M:]]

    w = z[act]
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        act = np.arange(n)
        w = np.ones(n)
        s = float(n)
    w = w / s

    a = x - float(x.mean())
    S = np.empty(n + 1, dtype=float)
    S[0] = 0.0
    S[1:] = np.cumsum(a)

    grad = np.zeros(n, dtype=float)
    inv_n = 1.0 / float(n)

    for idx, wk in zip(act, w):
        k = int(idx)
        L = n - k
        sst = S[L] + S[k]
        if sst > 1e-14:
            sig = 1.0
        elif sst < -1e-14:
            sig = -1.0
        else:
            sig = 0.0

        c_sig = wk * sig * (0.5 * inv_n)
        c_neg = -wk * inv_n

        if L > 0:
            grad[:L] += c_sig
            if c_neg != 0.0:
                grad[:L] += c_neg * a[k:k + L]
        if k > 0:
            grad[k:] += c_sig
            if c_neg != 0.0:
                grad[k:] += c_neg * a[:n - k]

    grad -= float(grad.mean())
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return grad


def optimize_a_slsqp(a, n, alphas, deadline, maxiters=150, regress_limit=0.04):
    global _GR_N, _GR_ALPHA

    a = project_a(a)
    best_h = a_to_h_feasible(a, n)
    best_c = compute_c5(best_h, n)
    best_a = project_a(2.0 * best_h - 1.0)

    work = project_a(a)
    work_h = a_to_h_feasible(work, n)
    work_c = compute_c5(work_h, n)

    bounds = [(-1.0, 1.0) for _ in range(n)]
    constraints = [{'type': 'eq', 'fun': sum_zero_fun, 'jac': sum_zero_jac}]

    for alpha in alphas:
        if time.time() > deadline:
            break
        _GR_N = int(n)
        _GR_ALPHA = float(alpha)

        if n <= 300:
            mi = max(20, int(maxiters))
        elif n <= 500:
            mi = max(20, int(0.6 * maxiters))
        else:
            mi = max(20, int(0.4 * maxiters))

        try:
            res = minimize(
                smooth_obj_a,
                work,
                jac=smooth_grad_a,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': mi, 'ftol': 1e-14, 'eps': 1e-8}
            )
            cand = project_a(np.asarray(res.x, dtype=float))
        except Exception:
            continue

        h_cand = a_to_h_feasible(cand, n)
        c_cand = compute_c5(h_cand, n)

        if c_cand < best_c - 1e-14:
            best_c = c_cand
            best_h = h_cand
            best_a = project_a(2.0 * best_h - 1.0)

        if c_cand < max(work_c + regress_limit, best_c + regress_limit):
            work = project_a(cand)
            work_c = c_cand

    return best_h, best_c, best_a


def interp_h_to_n(h_old, n_old, n_new):
    if n_old == n_new:
        return enforce_sum_h(h_old, n_new)
    x_old = _midgrid(n_old)
    x_new = _midgrid(n_new)
    h = np.interp(x_new, x_old, h_old)
    return enforce_sum_h(h, n_new)


def random_smooth_a(n, rng, R=5, sat=1.15):
    x = _midgrid(n)
    a = np.zeros(n, dtype=float)
    for m in range(1, R + 1):
        ca = rng.uniform(-1.0, 1.0)
        sa = rng.uniform(-1.0, 1.0)
        if ca != 0.0:
            a += ca * np.cos(np.pi * float(m) * x)
        if sa != 0.0:
            a += sa * np.sin(np.pi * float(m) * x)

    denom = float(np.max(np.abs(a)))
    if denom > 1e-12:
        a = np.tanh(a / denom)
        scale = rng.uniform(0.85, 1.15) * sat
        a = scale * 1.25 * a
        a = np.clip(a, -1.0, 1.0)

    a = np.clip(a, -1.0, 1.0)
    return project_a(a)


def random_binary_a(n, rng):
    if n % 2 != 0:
        return random_smooth_a(n, rng)
    a = np.empty(n, dtype=float)
    a[:] = -1.0
    on = n // 2
    idx = rng.choice(n, size=on, replace=False)
    a[idx] = 1.0
    return a


def coeffs_to_a(co, n, x, R):
    a = np.zeros(n, dtype=float)
    co = np.asarray(co, dtype=float).ravel()
    if co.size < 2 * R:
        co = np.pad(co, (0, 2 * R - co.size), mode='constant')
    for m in range(1, R + 1):
        c1 = float(co[2 * m - 2])
        c2 = float(co[2 * m - 1])
        if c1 != 0.0:
            a += c1 * np.cos(np.pi * float(m) * x)
        if c2 != 0.0:
            a += c2 * np.sin(np.pi * float(m) * x)
    denom = float(np.max(np.abs(a)))
    if denom > 1e-12:
        a = 1.35 * np.tanh(a / denom)
    a = np.clip(a, -1.0, 1.0)
    return project_a(a)


def param_objective(co):
    co = np.asarray(co, dtype=float).ravel()
    if not np.all(np.isfinite(co)):
        return 1e9
    n = _TOP_N
    x = _TOP_X
    R = _TOP_R
    a = coeffs_to_a(co, n, x, R)
    f, c = compute_f_from_a(a, n)
    alpha = _TOP_ALPHA
    if alpha <= 0.0:
        return float(c)
    m = float(np.max(f))
    z = np.exp(alpha * (f - m))
    return float(m + float(np.log(float(z.sum()))) / alpha)


def optimize_parametric(n, R, deadline, maxiter=200, starts=2, rng=None):
    global _TOP_N, _TOP_X, _TOP_R, _TOP_ALPHA
    if rng is None:
        rng = np.random.RandomState(0)

    if n % 2 != 0:
        n -= 1
    if n < 50:
        return None, 1e9

    x = _midgrid(n)
    _TOP_N = int(n)
    _TOP_X = x
    _TOP_R = int(R)
    _TOP_ALPHA = 25.0

    best_a = None
    best_c = 1e9

    for s in range(starts):
        if time.time() > deadline:
            break
        co0 = rng.normal(0.0, 0.7, size=2 * R)
        try:
            res = minimize(
                param_objective,
                co0,
                method='Nelder-Mead',
                options={'maxiter': maxiter, 'xatol': 1e-7, 'fatol': 1e-10}
            )
            co = res.x
        except Exception:
            co = co0
        a = coeffs_to_a(co, n, x, R)
        h = a_to_h_feasible(a, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_a = a.copy()

    if best_a is None:
        best_a = random_smooth_a(n, rng)
        best_c = compute_c5(a_to_h_feasible(best_a, n), n)
    return best_a, best_c


def cand_key(item):
    return float(item[0])


def add_candidate(cands, a, n):
    if a is None or n <= 0:
        return
    a = project_a(a)
    if a.size != n:
        return
    _, c = compute_f_from_a(a, n)
    cands.append((float(c), int(n), a))


def select_starts(cands, max_total=4, max_per_n=2):
    cands = sorted(cands, key=cand_key)
    selected = []
    count = {}
    for c, n, a in cands:
        if a is None:
            continue
        cnt = count.get(n, 0)
        if cnt >= max_per_n:
            continue
        selected.append((int(n), a))
        count[n] = cnt + 1
        if len(selected) >= max_total:
            break
    return selected


def binary_refine_from_a(a, n, deadline):
    if n % 2 != 0 or n < 50:
        return None, 1e9

    a = np.asarray(a, dtype=float).reshape(-1)
    on = n // 2
    order = np.argsort(a)[::-1]
    h = np.zeros(n, dtype=float)
    h[order[:on]] = 1.0

    best_h = h.copy()
    best_c = compute_c5(h, n)
    W = 22 if n <= 320 else 12

    for _ in range(30):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        zd = zeros[a[zeros] > 0.0]
        if zd.size < W:
            zd = zeros
        zord = np.argsort(a[zd])
        z_cand = zd[zord[len(zord) - W:]]

        od = ones[a[ones] < 0.0]
        if od.size < W:
            od = ones
        oord = np.argsort(a[od])
        o_cand = od[oord[:W]]

        improved = False
        best_move = None
        best_local = best_c

        for i in z_cand:
            for j in o_cand:
                if i == j:
                    continue
                h[i], h[j] = 1.0, 0.0
                c = compute_c5(h, n)
                if c < best_local - 1e-14:
                    best_local = c
                    best_move = (int(i), int(j))
                else:
                    h[i], h[j] = 0.0, 1.0

        if best_move is not None:
            i, j = best_move
            h[i], h[j] = 1.0, 0.0
            best_c = best_local
            best_h = h.copy()
            improved = True

        if not improved:
            break

    return best_h, best_c


def subgrad_polish_a(a, n, deadline, alpha=3000, max_iter=2500):
    global _GR_N, _GR_ALPHA
    a = project_a(a)
    h = a_to_h_feasible(a, n)
    c = compute_c5(h, n)
    best_h = h.copy()
    best_c = c

    _GR_N = int(n)
    _GR_ALPHA = float(alpha)

    eta = 5.0
    min_eta = 1e-3
    max_eta = 10.0 * float(n)

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        g = smooth_grad_a(a)
        gn = float(np.linalg.norm(g))
        if gn < 1e-13:
            break
        eta = float(np.clip(eta, min_eta, max_eta))
        y = a - eta * g
        a_new = project_a(y)
        h_quick = 0.5 + 0.5 * a_new
        c_new = compute_c5(h_quick, n)
        if c_new < c - 1e-14:
            a = a_new
            c = c_new
            eta = min(max_eta, eta * 1.08)
            if c < best_c - 1e-14:
                best_c = c
                best_h = a_to_h_feasible(a, n)
                best_c = compute_c5(best_h, n)
                c = best_c
        else:
            eta *= 0.5

    return best_h, best_c

    # 6) Final high-alpha smoothing and a last cheap subgradient pass.
    if time.time() < hard_deadline - 80:
        final_a = project_a(2.0 * best_h - 1.0)
        if best_n <= 360:
            fin_alphas = (700.0, 2500.0)
            fin_mi = 70
        else:
            fin_alphas = (1000.0,)
            fin_mi = 40
        h3, c3, a3 = optimize_a_slsqp(final_a, best_n, fin_alphas, hard_deadline - 40, fin_mi)
        if c3 < best_c - 1e-14:
            best_c = c3
            best_h = h3
            best_n = int(best_n)
            best_a = a3

    if time.time() < hard_deadline - 35:
        h4, c4 = subgrad_polish_a(project_a(2.0 * best_h - 1.0), best_n,
                                  hard_deadline - 12, alpha=3500, max_iter=1200)
        if c4 < best_c - 1e-14:
            best_c = c4
            best_h = h4
            best_n = int(best_n)
            best_a = project_a(2.0 * best_h - 1.0)

    # Final feasibility enforcement and exact reported value.
    best_h = enforce_sum_h(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, float(best_c), int(best_n)
