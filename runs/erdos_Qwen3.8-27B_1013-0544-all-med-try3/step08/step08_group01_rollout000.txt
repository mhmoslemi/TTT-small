            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def perturb_restart(h, n, seed, scale=0.02):
    rng = np.random.RandomState(seed)
    h_new = h.copy() + rng.randn(n) * scale
    h_new = project_feasible(h_new, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Optimize at original resolution (quick)
    if time.time() < deadline - 500:
        t1 = min(time.time() + 80, deadline - 500)
        best_h, best_c = run_lse_block(best_h, n_orig, t1,
            alphas=[2000, 20000, 200000, 2000000],
            rel_thresh=0.005, max_count=15,
            maxiter=2000, maxfun=10000)
    
    # Phase 2: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h_fine = np.interp(x_new, x_old, best_h)
    h_fine = project_feasible(h_fine, n)
    c_fine = compute_c5(h_fine, n)
    
    # Quick optimization at fine resolution
    if time.time() < deadline - 400:
        t2 = min(time.time() + 100, deadline - 400)
        h_fine, c_fine = run_lse_block(h_fine, n, t2,
            alphas=[5000, 50000, 500000],
            rel_thresh=0.005, max_count=15,
            maxiter=2000, maxfun=10000)
    
    if c_fine < best_c:
        best_c = c_fine
        best_h = h_fine
    else:
        best_h = h_fine
        best_c = c_fine  # work at fine res regardless
    
    # Phase 3: Main benders optimization at n=1000
    if time.time() < deadline - 200:
        t3 = min(time.time() + 350, deadline - 200)
        best_h, best_c = benders_optimize(best_h, n, t3)
    
    # Phase 4: Try perturbation restart if stuck
    if time.time() < deadline - 100:
        improved = False
        for rs in range(5):
            if time.time() > deadline - 50:
                break
            h_pert = perturb_restart(best_h, n, seed + rs * 7, scale=0.01 * (rs + 1))
            c_pert = compute_c5(h_pert, n)
            if c_pert < best_c:
                h_pert, c_pert = benders_optimize(h_pert, n, min(time.time() + 60, deadline - 30))
                if c_pert < best_c:
                    best_c = c_pert
                    best_h = h_pert
                    improved = True
    
    # Phase 5: Final intensive benders
    if time.time() < deadline - 60:
        t5 = min(time.time() + 60, deadline - 60)
        best_h, best_c = benders_optimize(best_h, n, t5)
    
    # Phase 6: Pair refinement
    if time.time() < deadline - 15:
        t6 = min(time.time() + 25, deadline - 15)
        best_h, best_c = pair_refine(best_h, n, t6, radius=30)
    
    # Also check at original resolution
    n_check = n_orig
    x_fin = np.linspace(0, 2, n)
    x_orig = np.linspace(0, 2, n_orig)
    h_orig_res = np.interp(x_orig, x_fin, best_h)
    h_orig_res = project_feasible(h_orig_res, n_orig)
    c_orig_res = compute_c5(h_orig_res, n_orig)
    
    # Choose best
    if c_orig_res < best_c:
        best_h = h_orig_res
        best_c = c_orig_res
    # Otherwise keep fine resolution
    
    best_h = project_feasible(best_h, len(best_h))
    final_n = len(best_h)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
