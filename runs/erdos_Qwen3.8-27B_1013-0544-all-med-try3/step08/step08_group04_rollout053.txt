            h[j] -= t_opt
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                improved_any = True
                c0 = np.correlate(h, 1 - h, mode='full')
            else:
                h[i] = h_old_i
                h[j] = h_old_j
        
        sweep_count += 1
        if not improved_any:
            if current_radius > min_radius:
                current_radius = max(min_radius, current_radius // 2)
            else:
                break
    
    return h, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh - mx))) / alpha)
    
    def constr_fun(x):
        return x.sum() - n / 2.0
    
    def constr_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def fixed_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - 40)
                hi = min(n, i + 41)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(250, 0.5)
        n = 250
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: SLSQP quick optimization (60s)
    p1_deadline = t0 + 60
    for a in [500, 2000, 10000]:
        if time.time() > p1_deadline - 5:
            break
        ht = slsqp_optimize(h, n, a, maxiter=1000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
            h = ht
    
    # Phase 2: Smooth gradient descent (120s)
    p2_deadline = t0 + 180
    if time.time() < p2_deadline:
        ht, c = smooth_grad_descent(h, n, p2_deadline, alpha=3000, lr_init=0.003, lr_decay=0.9995)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
            h = ht
    
    # Phase 3: Exact optimal pair search - main workhorse (400s)
    p3_deadline = t0 + 580
    if time.time() < p3_deadline:
        h, best_c = pair_search_optimal(h, n, p3_deadline, radius=30, global_every=8, min_radius=3)
        best_h = h.copy()
    
    # Phase 4: SLSQP polish + more pair search (200s)
    p4_deadline = t0 + 780
    if time.time() < p4_deadline - 20:
        ht = slsqp_optimize(best_h, n, 50000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
            h = ht
        if time.time() < p4_deadline:
            h, best_c = pair_search_optimal(h, n, p4_deadline, radius=20, global_every=5, min_radius=2)
            best_h = h.copy()
    
    # Phase 5: Fixed-step pair search fine tuning (150s)
    p5_deadline = t0 + 930
    if time.time() < p5_deadline:
        h, best_c = fixed_pair_search(h, n, p5_deadline,
                                       [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        best_h = h.copy()
    
    # Phase 6: Try nearby n values (100s)
    p6_deadline = t0 + 1030
    if best_h is not None and time.time() < p6_deadline:
        x_init = np.linspace(0, 2, len(best_h))
        for n_try in [n - 1, n + 1, n - 2, n + 2]:
            if time.time() > p6_deadline - 15:
                break
            if n_try < 50 or n_try > 500:
                continue
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)
            ht, c = pair_search_optimal(h_interp, n_try, p6_deadline - 15,
                                         radius=20, global_every=5, min_radius=3)
            if c < best_c:
                best_c = c
                best_h = ht.copy()
    
    # Phase 7: Final polish (remaining time)
    if time.time() < deadline - 2 and best_h is not None:
        n_final = len(best_h)
        ht = slsqp_optimize(best_h, n_final, 100000, maxiter=1500)
        c = compute_c5(ht, n_final)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 1:
            best_h, best_c = fixed_pair_search(best_h, n_final, deadline - 1,
                                                [0.0001, 0.00005, 0.00002, 0.00001])
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
