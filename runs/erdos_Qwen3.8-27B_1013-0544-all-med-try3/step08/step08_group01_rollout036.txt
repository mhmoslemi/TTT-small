import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smoothmax_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def grad_smoothmax_full(h, n, alpha):
    c = compute_c5_vec(h, n)
    w = smoothmax_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:

    vals = np.array([c[i] for i in top_idx])

            grad[0:nv] += wm * (-h[kk:n])

def adam_optimize(h, n, deadline, alpha_start=50, alpha_end=200000,

def block_coordinate_descent(h, n, deadline, block_size=8):
    """Optimize one block of variables at a time."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(200):
        if time.time() > deadline:
            break
        for start in range(0, n, block_size):
            if time.time() > deadline:
                break
            end = min(start + block_size, n)
            bsize = end - start
            
            # Try small perturbations of this block
            base = best_h.copy()
            block = best_h[start:end].copy()
            
            # Try each coordinate in the block
            for bi in range(bsize):
                for delta in [0.001, 0.0005, 0.0002, 0.0001, -0.001, -0.0005, -0.0002, -0.0001]:
                    new_val = block[bi] + delta
                    if new_val < 0 or new_val > 1:
                        continue
                    block[bi] = new_val
                    # Renormalize the whole vector
                    test_h = best_h.copy()
                    test_h[start:end] = block
                    test_h = project_feasible(test_h, n)
                    c_test = compute_c5(test_h, n)
                    if c_test < best_c - 1e-15:
                        best_c = c_test
                        best_h = test_h
                    else:
                        block[bi] -= delta
                        break
                block[bi] = best_h[start + bi]
    
    return best_h, best_c
