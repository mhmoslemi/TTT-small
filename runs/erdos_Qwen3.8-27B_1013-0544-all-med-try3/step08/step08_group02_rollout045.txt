                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 80, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_lbfgs_full(h, n, deadline, alphas, mu, maxiters):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 10:
            break
        maxit = maxiters[i] if i < len(maxiters) else 2000
        try:
            res = minimize(full_sm_obj, best_h, jac=full_sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 80, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def random_perturb(h, n, sigma, seed):
    rng = np.random.RandomState(seed)
    h_p = h + sigma * rng.randn(n)
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    # Get starting point
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Top-k L-BFGS at original resolution (focus on worst 5 shifts)
    p1_end = t0 + 200
    alphas1 = [200, 1000, 5000, 20000, 100000, 500000]
    maxit1 = [1500, 2000, 2500, 3000, 3500, 4000]
    h1, c1 = run_lbfgs_topk(best_h, best_n, p1_end, k=5,
                             alphas=alphas1, mu=0.01, maxiters=maxit1)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Full soft-max L-BFGS
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 150, deadline - 400)
        alphas2 = [500, 5000, 50000, 500000, 2000000]
        maxit2 = [2000, 2500, 3000, 3000, 3000]
        h2, c2 = run_lbfgs_full(best_h, best_n, p2_end,
                                 alphas=alphas2, mu=0.005, maxiters=maxit2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Projected subgradient (many iterations, fine)
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 120, deadline - 350)
        h3, c3 = projected_subgrad(best_h, best_n, p3_end, iters=15000, tol_expand=0.003)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to n=1000 and optimize
    n_hi = 1000
    if time.time() < deadline - 300:
        h_hi = try_resolution(best_h, best_n, n_hi, deadline)
        c_hi = compute_c5(h_hi, n_hi)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
            best_n = n_hi

        p4_end = min(time.time() + 250, deadline - 250)
        alphas4 = [100, 500, 2000, 10000, 50000, 200000]
        maxit4 = [1000, 1500, 2000, 2500, 2500, 3000]
        h4, c4 = run_lbfgs_topk(best_h, best_n, p4_end, k=5,
                                 alphas=alphas4, mu=0.005, maxiters=maxit4)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Random restarts in neighborhood
    if time.time() < deadline - 200:
        for rseed in [42, 123, 777, 2024, 5555]:
            if time.time() > deadline - 190:
                break
            h_r = random_perturb(best_h, best_n, 0.005, rseed)
            c_r = compute_c5(h_r, best_n)
            p5_end = min(time.time() + 40, deadline - 185)
            alphas_r = [10000, 100000, 1000000]
            h_r2, c_r2 = run_lbfgs_topk(h_r, best_n, p5_end, k=4,
                                         alphas=alphas_r, mu=0.01, maxiters=[800, 1000, 1000])
            if c_r2 < best_c:
                best_c = c_r2
                best_h = h_r2

    # Phase 6: Top-k L-BFGS again with very high alpha (sharp max)
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 80, deadline - 140)
        alphas6 = [500000, 2000000, 5000000]
        h6, c6 = run_lbfgs_topk(best_h, best_n, p6_end, k=3,
                                 alphas=alphas6, mu=0.001, maxiters=[1000, 1500, 1500])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: More subgradient refinement
    if time.time() < deadline - 80:
        p7_end = min(time.time() + 50, deadline - 70)
        h7, c7 = projected_subgrad(best_h, best_n, p7_end, iters=10000, tol_expand=0.001)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Coordinate descent
    if time.time() < deadline - 60:
        p8_end = min(time.time() + 40, deadline - 50)
        h8, c8 = coord_descent(best_h, best_n, p8_end, max_rounds=30)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Fine pair swaps
    if time.time() < deadline - 20:
        p9_end = deadline - 3
        h9, c9 = fine_pair_swap(best_h, best_n, p9_end, radius=40)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final: one more subgradient pass
    if time.time() < deadline - 2:
        h10, c10 = projected_subgrad(best_h, best_n, deadline - 1, iters=5000, tol_expand=0.001)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
