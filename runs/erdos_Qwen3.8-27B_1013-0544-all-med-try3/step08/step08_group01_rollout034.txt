def project_feasible(h,n):
    target=0.5*n
    h=np.asarray(h,dtype=float).copy()
    h=np.clip(h,0.0,1.0)
    # Solve sum via loop
    for _ in range(200):
        s=h.sum()
        diff=target-s
        if abs(diff) <= 1e-14: break
        if diff > 0:
            room=1.0-h
            tot=room.sum()
            if tot <= 1e-15: break
            h += (diff/tot)*room
        else:
            tot=s
            if tot <= 1e-15: break
            h += (diff/tot)*h
    h=np.clip(h,0.0,1.0)
    # final exact distribute
    diff=target-h.sum()
    if abs(diff) > 0:
        free = np.where((h > 1e-12) & (h < 1.0-1e-12))[0]
        if free.size:
            add=diff/free.size
            h[free] += add
            h=np.clip(h,0.0,1.0)
            diff=target-h.sum()
        else:
            # all at bounds; adjust one within possible by choosing first with enough
            if diff > 0:
                idx=np.argsort(1.0-h)[::-1]
            else:
                idx=np.argsort(h)
            for k in idx:
                room = (1.0-h[k]) if diff>0 else h[k]
                take=min(abs(diff), room)
                h[k] += np.sign(diff)*take
                diff -= np.sign(diff)*take
                if abs(diff) <= 1e-15: break
    # one more tiny correction at an index
    diff=target-h.sum()
    if abs(diff) > 1e-15:
        # choose index with room
        if diff>0:
           idx = int(np.argmax(1.0-h))
        else: idx=int(np.argmin(h))
        h[idx] += diff
        h=np.clip(h,0.0,1.0)
    return h

def compute_c_values(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(np.max(compute_c_values(h, n)))


def is_feasible(h, n, tol=1e-7):
    if int(getattr(h, "size", -1)) != int(n):
        return False
    if not bool(np.all(np.asarray(h, dtype=float) >= -tol)):
        return False
    if not bool(np.all(np.asarray(h, dtype=float) <= 1.0 + tol)):
        return False
    return bool(abs(float(np.sum(h)) - 0.5 * float(n)) <= tol)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(300):
        s = float(h.sum())
        diff = target - s
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot <= 1e-15:
                break
            h += (diff / tot) * h

    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-15:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += diff / float(free.size)
            h = np.clip(h, 0.0, 1.0)
            diff = target - float(h.sum())

    if abs(diff) > 1e-15:

        else:
            order = np.argsort(h)
        for k in order:
            kk = int(k)
            if diff > 0.0:
                take = min(diff, 1.0 - h[kk])

                diff -= take

        else:

    return h


def conv_slice(w, x, n):
    w = np.asarray(w, dtype=float)
    x = np.asarray(x, dtype=float)
    start = int(n) - 1
    if _HAVE_FFT and len(w) * len(x) > 700000:
        try:
            c = fftconvolve(w, x, mode="full")
        except Exception:
            c = np.convolve(w, x, mode="full")
    else:
        c = np.convolve(w, x, mode="full")
    return c[start:start + int(n)]


def top_weights(c, k):
    c = np.asarray(c, dtype=float)
    m = int(len(c))
    k = int(k)
    if k <= 0:
        k = 1
    if k >= m:
        return np.full(m, 1.0 / float(m), dtype=float)
    mx = float(c.max())
    act = np.where(c >= mx - 1e-9)[0]
    if act.size > 0 and act.size <= k:
        w = np.zeros(m, dtype=float)
        w[act] = 1.0 / float(act.size)
        return w
    idx = np.argsort(c)[-k:]
    w = np.zeros(m, dtype=float)
    w[idx] = 1.0 / float(k)
    return w


def gradient_counts(w, n, x):
    n = int(n)
    Nc = 2 * n - 1
    w = np.asarray(w, dtype=float)
    x = np.asarray(x, dtype=float)

    pref = np.empty(Nc + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(w)
    A = pref[n:Nc + 1] - pref[0:n]

    p = conv_slice(w, x, n)
    q = conv_slice(w[::-1], x, n)
    return A - p - q


def smooth_fun(x, n, alpha, mu):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = float(c.max())
    s = alpha * (c - m)
    z = float(s.max())
    e = np.exp(s - z)
    se = float(e.sum())
    val = m + (z + np.log(se)) / float(alpha)
    target = 0.5 * float(n)
    diff = float(np.sum(x)) - target
    val += mu * diff * diff
    return float(val)


def smooth_grad(x, n, alpha, mu):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = float(c.max())
    s = alpha * (c - m)
    z = float(s.max())
    e = np.exp(s - z)
    w = e / float(e.sum())

    g = gradient_counts(w, n, x) * dx
    target = 0.5 * float(n)
    diff = float(np.sum(x)) - target
    g += 2.0 * mu * diff
    return g


def lbfgs_opt(h, n, deadline, configs=None, maxiter=200):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * int(n)

    if configs is None:
        configs = [
            (30.0, 1e-3),
            (150.0, 1e-2),
            (600.0, 0.1),
            (2500.0, 1.0),
            (10000.0, 10.0),
        ]

    for alpha, mu in configs:
        if time.time() > float(deadline) - 4.0:
            break
        try:
            res = minimize(
                smooth_fun,
                np.asarray(h, dtype=float),
                jac=smooth_grad,
                args=(int(n), float(alpha), float(mu)),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": int(maxiter),
                         "maxfun": int(max(maxiter * 4, 300)),
                         "ftol": 1e-18,
                         "gtol": 1e-12,
                         "maxls": 20},
            )
            h2 = project_feasible(np.asarray(res.x, dtype=float), n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                h = h2.copy()
        except Exception:
            pass

    return best_h, best_c


def eval_pair_t_into(tmp, c, Lc, dx, diag, m1, m2, t):
    if t == 0.0:
        return float(np.max(c))

    scale = float(t) * float(dx)
    np.multiply(Lc, scale, out=tmp)
    tmp += c

    old = float(tmp[diag])
    tmp[diag] = -np.inf
    base = float(tmp.max())
    tmp[diag] = old

    f = base
    val = old - 2.0 * dx * float(t) * float(t)
    if val > f:
        f = val

    if m1 >= 0:
        val = float(tmp[m1]) + dx * float(t) * float(t)
        if val > f:
            f = val

    if m2 >= 0:
        val = float(tmp[m2]) + dx * float(t) * float(t)
        if val > f:
            f = val

    return float(f)


def pair_line_search(h, c, n, i, j, lo, hi, F, g, w, dx, refine=2):
    n = int(n)
    Nc = 2 * n - 1
    diag = n - 1
    m1 = diag + int(i) - int(j)
    m2 = diag + int(j) - int(i)

    if not (0 <= m1 < Nc) or m1 == diag:
        m1 = -1
    if not (0 <= m2 < Nc) or m2 == diag:
        m2 = -1

    Lc = np.zeros(Nc, dtype=float)
    hrev = h[::-1]

    Lc[int(i):int(i) + n] += 1.0 - hrev
    Lc[n - 1 - int(i):2 * n - 1 - int(i)] += -h
    Lc[int(j):int(j) + n] += -1.0 + hrev
    Lc[n - 1 - int(j):2 * n - 1 - int(j)] += h

    span = float(hi) - float(lo)
    if span <= 1e-13:
        return 0.0, float(F), Lc, diag, m1, m2

    cand = [0.0, lo, hi]

    for tv in np.linspace(lo, hi, 9):
        cand.append(float(tv))

    pos_scale = max(0.0, float(hi))
    neg_scale = max(0.0, -float(lo))
    for frac in (0.02, 0.05, 0.12, 0.30):
        if pos_scale * frac > 1e-14:
            cand.append(float(pos_scale * frac))
        if neg_scale * frac > 1e-14:
            cand.append(float(-neg_scale * frac))

    G = float(g[int(i)] - g[int(j)])
    H = -2.0 * float(w[diag])
    if m1 >= 0:
        H += float(w[m1])
    if m2 >= 0:
        H += float(w[m2])

    if abs(H) > 1e-15 and H > 0.0:
        t0 = -G / (2.0 * H)
        for mult in (0.25, 0.5, 0.75, 1.0, 1.5, 2.0):
            cand.append(float(t0 * mult))

    uniq = []
    for tv in cand:
        tv = float(tv)
        if tv < lo:
            tv = float(lo)
        if tv > hi:
            tv = float(hi)
        ok = True
        for u in uniq:
            if abs(tv - u) < 1e-12:
                ok = False
                break
        if ok:
            uniq.append(tv)

    if not uniq:
        uniq = [0.0]

    tmp = np.empty_like(c)
    best_t = 0.0
    best_f = float(F)

    for tv in uniq:
        fv = eval_pair_t_into(tmp, c, Lc, dx, diag, m1, m2, tv)
        if fv < best_f:
            best_f = fv
            best_t = tv

    if best_f < F - 1e-15:
        radius = span / 16.0
        for _ in range(int(refine)):
            if radius < 1e-13:
                break
            lo2 = max(lo, best_t - radius)
            hi2 = min(hi, best_t + radius)
            if hi2 - lo2 > 1e-13:
                for tv in np.linspace(lo2, hi2, 11):
                    fv = eval_pair_t_into(tmp, c, Lc, dx, diag, m1, m2, float(tv))
                    if fv < best_f:
                        best_f = fv
                        best_t = float(tv)
            radius *= 0.35

    if best_f < F:
        return float(best_t), float(best_f), Lc, diag, m1, m2

    return 0.0, float(F), Lc, diag, m1, m2

def smooth_1d(h, sigma):
    h = np.asarray(h, dtype=float)
    sig = float(sigma)
    if sig <= 0.0:
        return h.copy()
    half = int(max(1, min(len(h) // 2, 3 * sig + 1)))
    x = np.arange(-half, half + 1, dtype=float)
    k = np.exp(-0.5 * (x / max(sig, 1e-6)) ** 2)
    k /= float(k.sum())
    hp = np.pad(h, half, mode="edge")
    return np.convolve(hp, k, mode="valid")[:len(h)]


def binary_run_start(n, seed, p, smooth_sigma=0.0):
    n = int(n)
    rng = np.random.default_rng(int(seed))
    bits = np.empty(n, dtype=float)
    b = 1.0 if rng.random() < 0.5 else 0.0
    bits[0] = b

    for i in range(1, n):
        if rng.random() < float(p):
            bits[i] = b
        else:
            b = 1.0 - b
            bits[i] = b

    target = 0.5 * float(n)
    s = float(bits.sum())
    if s < target:
        idx = np.where(bits == 0.0)[0]
        if idx.size > 0:
            rng.shuffle(idx)
            need = int(round(target - s))
            if need > 0:
                bits[idx[:max(0, min(need, idx.size))]] = 1.0
    elif s > target:
        idx = np.where(bits == 1.0)[0]
        if idx.size > 0:
            rng.shuffle(idx)
            need = int(round(s - target))
            if need > 0:
                bits[idx[:max(0, min(need, idx.size))]] = 0.0

    if float(smooth_sigma) > 1e-8:
        h = smooth_1d(bits, float(smooth_sigma))
    else:
        h = bits.copy()

    return project_feasible(h, n)


def lowpass_logistic_start(n, seed, sigma_frac=0.08, strength=2.5):
    n = int(n)
    rng = np.random.default_rng(int(seed))
    sig = max(2.0, float(sigma_frac) * float(n))
    half = int(max(2, min(n // 2, 3 * sig + 1)))

    x = np.arange(-half, half + 1, dtype=float)
    k = np.exp(-0.5 * (x / max(sig, 1e-6)) ** 2)
    k /= float(k.sum())

    z = rng.normal(size=n)
    zp = np.pad(z, half, mode="edge")
    z = np.convolve(zp, k, mode="valid")[:n]

    zm = float(z.mean())
    sd = float(z.std())
    if sd < 1e-12:
        z = np.arange(n, dtype=float)
    else:
        z = (z - zm) / sd
    z = np.clip(z * 1.5, -20.0, 20.0)
    h = 1.0 / (1.0 + np.exp(-float(strength) * z))
    return project_feasible(h, n)


def resample_h(h, n_new):
    n_old = int(len(h))
    n_new = int(n_new)
    if n_old == n_new:
        return np.asarray(h, dtype=float).copy()

    dxo = 2.0 / float(n_old)
    dxn = 2.0 / float(n_new)
    xo = (np.arange(n_old, dtype=float) + 0.5) * dxo
    xn = (np.arange(n_new, dtype=float) + 0.5) * dxn
    out = np.interp(xn, xo, np.asarray(h, dtype=float))
    return project_feasible(out, n_new)

                                 k_cycle=(1, 8, 20, 1, 50),
                                 no_imp_limit=10, base_refine=2,
                                 seed=int(seed) + 404 + si)
            if cr < best_c - 1e-15:
                best_c = float(cr)
                best_h = np.asarray(hr, dtype=float).copy()
                best_n = int(hr.size)
        si += 1

    # Refine on a higher resolution grid if time remains.
    if time.time() < deadline - 180.0:
        n_up = 800
        if best_n >= n_up:
            n_up = int(best_n)

        h_up = resample_h(best_h, n_up)
        c_up = compute_c5(h_up, n_up)

        # Upsampled pair descent.
        t_u1 = min(time.time() + 100.0, deadline - 140.0)
        if t_u1 > time.time() + 20.0:
            h_up, c_up = pair_cd_opt(h_up, n_up, t_u1,
                                     top_p=9, max_pairs=36,
                                     k_cycle=(1, 8, 18, 3, 45),
                                     no_imp_limit=12, base_refine=2,
                                     seed=int(seed) + 505)

        # Upsampled L-BFGSB polish only if competitive.
        if c_up < best_c + 0.004 and time.time() < deadline - 90.0:
            t_u2 = min(time.time() + 80.0, deadline - 70.0)
            if t_u2 > time.time() + 15.0:
                h_u2, c_u2 = lbfgs_opt(h_up, n_up, t_u2,
                                       configs=[(50.0, 1e-3), (400.0, 0.05), (2000.0, 0.5)],
                                       maxiter=120)
                if c_u2 < best_c + 0.004:
                    h_up, c_up = np.asarray(h_u2, dtype=float), float(c_u2)

        # Final upsampled pair polish.
        t_u3 = min(time.time() + 30.0, deadline - 15.0)
        if t_u3 > time.time() + 8.0:
            h_up, c_up = pair_cd_opt(h_up, n_up, t_u3,
                                     top_p=8, max_pairs=30,
                                     k_cycle=(1, 7, 20),
                                     no_imp_limit=10, base_refine=2,
                                     seed=int(seed) + 606)

        if c_up < best_c - 1e-15:
            best_c = float(c_up)
            best_h = np.asarray(h_up, dtype=float).copy()
            best_n = int(h_up.size)

    # Last short polish at the best resolution.
    if time.time() < deadline - 4.0:
        t_f = deadline - 2.0
        if t_f > time.time() + 6.0:
            hf, cf = pair_cd_opt(best_h, best_n, t_f,
                                 top_p=8, max_pairs=28,
                                 k_cycle=(1, 7, 18),
                                 no_imp_limit=8, base_refine=2,
                                 seed=int(seed) + 707)
            if cf < best_c - 1e-15:
                best_c = float(cf)
                best_h = np.asarray(hf, dtype=float).copy()
                best_n = int(hf.size)

    # Ensure the returned vector is feasible and the reported value is exact.
    if not is_feasible(best_h, best_n, tol=1e-7):
        best_h = project_feasible(best_h, best_n)
        best_c = compute_c5(best_h, best_n)
    else:
        # A very light cleanup is usually harmless; only keep it if it improves
        # or does not materially worsen the certified objective.
        hc = project_feasible(best_h, best_n)
        cc = compute_c5(hc, best_n)
        if cc <= best_c + 1e-10:
            best_h = hc
            best_c = cc

    best_h = np.asarray(best_h, dtype=float)
    best_c = float(compute_c5(best_h, int(best_n)))
    return best_h, best_c, int(best_n)
