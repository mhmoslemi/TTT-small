import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute among non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jacobian(x, n):
    return np.ones(n)

    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_D(h, n, s):
    """Compute D(i) = (1-h[i+s])*valid(i+s) - h[i-s]*valid(i-s) for shift s"""
    Di = np.zeros(n)
    if s >= 0:
        # 1-h[i+s] valid for i in [0, n-1-s]
        if n - 1 - s >= 0:
            Di[0:n-s] += (1.0 - h[s:n])
        # h[i-s] valid for i in [s, n-1]
        if s < n:
            Di[s:n] -= h[0:n-s]
    else:
        ss = -s
        # 1-h[i-s] ... wait, s is negative here
        # 1-h[i+s] where s < 0: i+s >= 0 => i >= -s = ss
        # i+s < n => i < n - s = n + ss
        # So i in [ss, n-1] (since n+ss > n for ss > 0, cap at n-1)
        Di[ss:n] += (1.0 - h[0:n-ss])
        # h[i-s] where s < 0: i-s = i+ss >= 0 always (for i>=0)
        # i+ss < n => i < n-ss
        # So i in [0, n-ss-1]
        Di[0:n-ss] -= h[ss:n]
    return Di

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_fun(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_D(h, n, s):
    Di = np.zeros(n)
    if s >= 0:
        if n - 1 - s >= 0:
            Di[0:n-s] += (1.0 - h[s:n])
        if s < n:
            Di[s:n] -= h[0:n-s]
    else:
        ss = -s
        Di[ss:n] += (1.0 - h[0:n-ss])
        Di[0:n-ss] -= h[ss:n]
    return Di
