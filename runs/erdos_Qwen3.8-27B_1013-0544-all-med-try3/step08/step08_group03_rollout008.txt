    t5 = min(t0 + 800, deadline - 40)
    if time.time() < t5:
        h_curr, c5 = coord_descent(best_h, n_curr, t5, grid_size=300, passes=25)
        if c5 < best_c:
            best_c = c5
            best_h = h_curr.copy()
    
    # Phase 6: Try different n values
    if time.time() < deadline - 60:
        for n_try in [350, 250]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, n_curr)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            t_alloc = min(50, deadline - time.time() - 10)
            if t_alloc < 20:
                break
            h_t, c_t = coord_descent(h_interp, n_try, time.time() + t_alloc * 0.7, grid_size=100, passes=8)
            if time.time() < deadline - 10:
                h_t, c_t = fine_pair_search(h_t, n_try, deadline - 5, base_step=0.002)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n_curr = n_try
    
    # Phase 7: Final coordinate descent
    if time.time() < deadline - 3:
        best_h, best_c = coord_descent(best_h, len(best_h), deadline - 1, grid_size=200, passes=8)
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
