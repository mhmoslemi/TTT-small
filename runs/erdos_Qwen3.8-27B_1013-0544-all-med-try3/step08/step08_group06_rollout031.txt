            if tot < 1e-14:

    try:
        res = minimize(
            smooth_max_obj, h, args=(n, alpha, penalty),
            jac=smooth_max_grad, method='SLSQP',
            bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18}
        )
        h_opt = project_feasible(res.x, n)
        return h_opt
    except Exception:
        return h.copy()

def coordinate_descent(h, n, deadline, steps=None):
    if steps is None:
        steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    h = h.copy()
    for step in steps:
        for _pass in range(20):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for delta in [step, -step]:
                    hi = h[i] + delta
                    if 0.0 <= hi <= 1.0:
                        # Adjust sum: redistribute to other coordinates
                        adj = -delta
                        non_i = np.arange(n) != i
                        room = np.zeros(n)
                        if adj > 0:
                            room[non_i] = 1.0 - h[non_i]
                            tot = room.sum()
                        else:
                            room[non_i] = h[non_i]
                            tot = h[non_i].sum()
                        if tot < 1e-15:
                            continue
                        trial = h.copy()
                        trial[i] = hi
                        trial += (adj / tot) * room
                        trial = np.clip(trial, 0.0, 1.0)
                        # Re-project for exact sum
                        trial = project_feasible(trial, n)
                        c = compute_c5(trial, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            h = trial
                            improved = True
                            delta = -delta  # try opposite next
            if not improved:
                break
    return h, best_c

def pair_sweep(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(30):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n, 2):
                    # Try h[i]+=d, h[j]-=d
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    # Try h[i]-=d, h[j]+=d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def near_pair_sweep(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(30):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def optimize_at_resolution(h_start, n, deadline, use_slsqp=False):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: LBFGS with increasing alpha and penalty
    alphas = [200, 500, 2000, 8000, 30000, 100000, 300000]
    penalties = [100, 500, 1000, 5000, 10000]
    
    for a in alphas:
        if time.time() > deadline - 10:
            break
        for pen in penalties:
            if time.time() > deadline - 10:
                break
            h_try = lbfgs_optimize(best_h, n, a, pen, maxiter=800, deadline=deadline)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_h = h_try
    
    # Phase 2: SLSQP for finer optimization (small n only)
    if n <= 250 and time.time() < deadline - 20:
        for a in [50000, 150000, 500000]:
            if time.time() > deadline - 15:
                break
            h_try = slsqp_run(best_h, n, a, 5000, maxiter=1500)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_h = h_try
    
    # Phase 3: Coordinate descent
    if time.time() < deadline - 10:
        best_h, best_c = coordinate_descent(best_h, n, deadline - 10)
    
    # Phase 4: Pair sweep
    if time.time() < deadline - 5:
        if n <= 150:
            best_h, best_c = pair_sweep(best_h, n, deadline - 3, [0.01, 0.005, 0.002, 0.001])
        else:
            best_h, best_c = near_pair_sweep(best_h, n, deadline - 3, [0.01, 0.005, 0.002], radius=30)
    
    # Phase 5: Final LBFGS polish
    if time.time() < deadline - 3:
        h_try = lbfgs_optimize(best_h, n, 500000, 10000, maxiter=500)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_h = h_try
            best_c = c_try
    
    # Phase 6: Final coordinate nudge
    if time.time() < deadline - 1:
        best_h, best_c = coordinate_descent(best_h, n, deadline - 0.5,
                                            steps=[0.005, 0.002, 0.001, 0.0005])
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src)
    x_tgt = np.linspace(0, 2, n_tgt)
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return project_feasible(h_tgt, n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = t0 + budget_s - 3
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 200
        h_init = np.full(n_init, 0.5)
    
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    
    # Strategy: try multiple resolutions with aggressive optimization
    # Allocate time based on resolution complexity
    elapsed = time.time() - t0
    remaining = budget_s - 3 - elapsed
    
    if remaining < 10:
        return best_h, best_c, best_n
    
    # Resolution plan: [n_init, n_init+100, n_init+200, n_init*2]
    # Filter to reasonable range [60, 500]
    resolutions = []
    for nr in [n_init, n_init + 50, n_init + 100, n_init + 150, min(n_init * 2, 500)]:
        nr = max(60, min(500, nr))
        if nr not in resolutions:
            resolutions.append(nr)
    
    # Time allocation: more time for higher resolutions (they're more interesting)
    # But SLSQP only for n<=250
    weights = [1.0 + 0.3 * (r - resolutions[0]) / max(1, resolutions[-1] - resolutions[0]) for r in resolutions]
    total_w = sum(weights)
    
    for idx, nr in enumerate(resolutions):
        alloc_time = remaining * weights[idx] / total_w
        alloc_time = max(15, min(alloc_time, remaining * 0.5))
        dl = time.time() + alloc_time
        
        if time.time() > hard_deadline - 20:
            break
        
        # Interpolate to this resolution
        if nr == n_init:
            h_start = h_init.copy()
        else:
            h_start = interpolate_h(h_init, n_init, nr)
        
        # Also try a different starting point for diversity
        if idx > 0 and rng.random() < 0.3:
            # Add small structured perturbation
            h_pert = h_start + 0.05 * rng.randn(nr)
            h_pert = project_feasible(h_pert, nr)
            h_start = h_pert
        
        h_opt, c_opt = optimize_at_resolution(h_start, nr, min(dl, hard_deadline - 5))
        
        if c_opt < best_c - 1e-10:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = nr
            # Update initial for subsequent interpolations
            h_init = best_h
            n_init = best_n
        
        remaining = hard_deadline - time.time()
        if remaining < 30:
            break
    
    # Random restarts at best resolution
    if time.time() < hard_deadline - 30:
        n_best = best_n
        n_restarts = max(2, int((hard_deadline - time.time() - 20) / 40))
        for r in range(n_restarts):
            if time.time() > hard_deadline - 15:
                break
            scale = 0.03 + 0.08 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (hard_deadline - time.time() - 15) * 0.4)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc)
            if c_opt < best_c - 1e-10:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n_best
    
    # Final refinement on the best solution
    if time.time() < hard_deadline - 5:
        n_fin = best_n
        # Very fine coordinate descent
        best_h, best_c = coordinate_descent(best_h, n_fin, hard_deadline - 2,
                                            steps=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        
        # Final LBFGS polish
        if time.time() < hard_deadline - 2:
            h_try = lbfgs_optimize(best_h, n_fin, 1000000, 20000, maxiter=300)
            c_try = compute_c5(h_try, n_fin)
            if c_try < best_c:
                best_h = h_try
                best_c = c_try
        
        # Ultra-fine pair search
        if n_fin <= 200 and time.time() < hard_deadline - 1:
            best_h, best_c = pair_sweep(best_h, n_fin, hard_deadline - 0.5,
                                        [0.002, 0.001, 0.0005, 0.0002])
        elif time.time() < hard_deadline - 1:
            best_h, best_c = near_pair_sweep(best_h, n_fin, hard_deadline - 0.5,
                                              [0.002, 0.001, 0.0005], radius=20)
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
