import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def corr_grad_single(h, k, n):
    """Gradient of c_k = sum h[n+k]*(1-h[n]) w.r.t. h."""
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        a = -k
        g[:n-a] += (1.0 - h[a:])
        g[a:] -= h[:n-a]
    return g

def compute_interpolate(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def sqp_optimize(h, n, deadline, max_lp_iters=60, active_tol_frac=0.02):
    """SQP/active-set method: minimize max_k c_k(h) via sequential LPs."""
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    c_cur = compute_c5(h, n)
    
    for iteration in range(max_lp_iters):
        if time.time() > deadline - 2:
            break
        
        # Compute all correlation values
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_c = corr.max()
        
        # Active set: lags within tolerance of max
        tol = active_tol_frac * max_c + 1e-8
        active = np.where(corr >= max_c - tol)[0]
        
        # Limit active set size for LP efficiency
        if len(active) > 80:
            # Keep only the top 80
            active_diffs = corr[active]
            order = np.argsort(-active_diffs)[:80]
            active = active[order]
        
        # Build LP: minimize t
        # s.t. g_k^T h_new - t <= g_k^T h_cur - c_k(h_cur) for k in active
        #      0 <= h_new <= 1, sum(h_new) = n/2
        n_act = len(active)
        A_ub = np.zeros((n_act, n + 1))
        b_ub = np.zeros(n_act)
        
        for idx, p in enumerate(active):
            k_lag = int(p) - (n - 1)
            g = corr_grad_single(h, k_lag, n)
            A_ub[idx, :n] = g
            A_ub[idx, n] = -1.0
            b_ub[idx] = g @ h - corr[idx]
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'maxiter': 5000})
        except Exception:
            break
        
        if not res.success:
            # Try with relaxed active set
            continue
        
        h_lp = res.x[:n]
        t_lp = res.x[n]
        
        # Line search: ensure actual objective decreases
        best_alpha = 0.0
        best_c = c_cur
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]:
            h_try = h + alpha * (h_lp - h)
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_alpha = alpha
                h = h_try.copy()
        
        # Check convergence

            break
        if max_c - c_cur < 1e-12:
            break
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

    for d in deltas:

            break
        n_pass = 0

            n_pass += 1

                j_lo = max(0, i - radius)

                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    # Try h[i]-d, h[j]+d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi2 + d, hj2 - d
            if not improved:
                break
    return h, best_c

def full_optimize(h_start, n, deadline, rng):
    """Full optimization pipeline for a given resolution."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: SQP
    t_phase = min(deadline * 0.3, 80)
    h1, c1 = sqp_optimize(best_h, n, time.time() + t_phase)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Pair search
    if time.time() < deadline - 10:
        d_phase = min(deadline - time.time() - 10, 60)
        best_h, best_c = pair_search_targeted(best_h, n, time.time() + d_phase,
                                               [0.02, 0.01, 0.005, 0.002])
    
    # Phase 3: SQP again
    if time.time() < deadline - 8:
        h3, c3 = sqp_optimize(best_h, n, time.time() + min(deadline - time.time() - 8, 40),
                              max_lp_iters=30, active_tol_frac=0.05)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Fine pair search
    if time.time() < deadline - 2:
        best_h, best_c = pair_search_targeted(best_h, n, time.time() + min(deadline - time.time() - 2, 30),
                                               [0.01, 0.005, 0.002, 0.001])
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 200
        h_init = np.full(n_init, 0.5)
    
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    
    # Phase A: Optimize at initial resolution
    remaining = deadline - time.time()
    alloc_a = min(remaining * 0.35, 200)
    h_a, c_a = full_optimize(h_init, n_init, time.time() + alloc_a, rng)
    if c_a < best_c:
        best_c = c_a
        best_h = h_a
    
    # Phase B: Try finer resolution
    if time.time() < deadline - 60:
        n_fine = min(n_init + 100, 350)
        if n_fine != n_init:
            h_interp = compute_interpolate(best_h, len(best_h), n_fine)
            alloc_b = min(deadline - time.time() - 60, 120)
            h_b, c_b = full_optimize(h_interp, n_fine, time.time() + alloc_b, rng)
            if c_b < best_c:
                best_c = c_b
                best_h = h_b
    
    # Phase C: Try coarser resolution (can find different basins)
    if time.time() < deadline - 50:
        n_coarse = max(80, n_init // 2)
        h_coarse = compute_interpolate(best_h, len(best_h), n_coarse)
        alloc_c = min(deadline - time.time() - 50, 60)
        h_c, c_c = full_optimize(h_coarse, n_coarse, time.time() + alloc_c, rng)
        if c_c < best_c:
            best_c = c_c
            best_h = h_c
    
    # Phase D: Random restarts
    if time.time() < deadline - 40:
        n_best = len(best_h)
        n_restarts = max(2, int((deadline - time.time() - 40) / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.03 + 0.08 * (r / max(1, n_restarts - 1))
            hp = best_h.copy()
            hp += rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc_r = min(50, deadline - time.time() - 35)
            if alloc_r < 15:
                break
            h_r, c_r = full_optimize(hp, n_best, time.time() + alloc_r, rng)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    
    # Phase E: Targeted SQP with tighter active set
    if time.time() < deadline - 20:
        alloc_e = min(deadline - time.time() - 15, 60)
        h_e, c_e = sqp_optimize(best_h, len(best_h), time.time() + alloc_e,
                                max_lp_iters=80, active_tol_frac=0.005)
        if c_e < best_c:
            best_c = c_e
            best_h = h_e
    
    # Phase F: Final fine pair search
    if time.time() < deadline - 3:
        n_best = len(best_h)
        best_h, best_c = pair_search_targeted(best_h, n_best, deadline - 1,
                                               [0.005, 0.002, 0.001, 0.0005, 0.0002],
                                               radius=40)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
