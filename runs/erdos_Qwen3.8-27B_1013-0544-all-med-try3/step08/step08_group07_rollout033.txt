def compute_correlations(h, n):
    """Compute all correlation values (without dx factor)."""
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h onto the feasible set {0≤h≤1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

You know what, let me just compute the gradient directly in O(n²) time. For n=250, that's 62500 operations per gradient, which is fast (~10μs in numpy).

def compute_gradient(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w = w / w.sum()
    
    # c has length 2n-1, indexed by m=0,...,2n-2
    # c[m] corresponds to effective lag d = m - (n-1)
    # ∂c[m]/∂x[i] = dx * [(1-x[i+d]) - x[i-d]] where d = m-(n-1)
    
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for m in range(2*n-1):
            d = m - (n-1)
            term = 0.0
            if 0 <= i + d < n:
                term += (1 - x[i + d])
            if 0 <= i - d < n:
                term -= x[i - d]
            s += w[m] * term
        grad[i] = dx * s
    return grad
