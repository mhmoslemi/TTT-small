For the binary case with n=250, I can precompute the effect matrix (n × 2n-1) and then for each zero i, compute all deltas to all ones j in one vectorized operation. The cross terms can be added as a small correction.

Let me write the final program now, keeping it clean and correct:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def greedy_binary(h, n):
    """Sort and take top n/2 as 1s."""
    target = n // 2
    h_b = np.zeros(n)
    h_b[np.argsort(h)[::-1][:target]] = 1.0
    return h_b

def vectorized_swap_search(h, n, deadline, max_sweeps=500, verbosity=0):
    """Fast binary swap search with precomputed effect matrix."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    nm1 = n - 1
    
    # Precompute index matrices (recomputed after each swap)
    def compute_effect(h):
        m_idx = np.arange(n_corr)
        fwd = np.arange(n)[:, None] + m_idx[None, :] - nm1
        bwd = np.arange(n)[:, None] - m_idx[None, :] + nm1
        vf = (fwd >= 0) & (fwd < n)
        vb = (bwd >= 0) & (bwd < n)
        sf = np.clip(fwd, 0, n-1)
        sb = np.clip(bwd, 0, n-1)
        return (1.0 - h[sf]) * vf - h[sb] * vb
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    effect = compute_effect(h)
    
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h <= 0.5)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        best_improve = 1e-15
        best_i = -1
        best_j = -1
        best_new_corr = None
        
        for i in zeros_idx:
            # Main delta: effect[i] - effect[j] for all j in ones
            # Shape: (len(ones), n_corr)
            main_delta = effect[i][None, :] - effect[ones_idx]
            
            # Add cross terms:
            # For each j_idx, j = ones_idx[j_idx]:
            #   delta[nm1] -= 2
            #   delta[nm1 + j - i] += 1 (if valid)
            #   delta[nm1 + i - j] += 1 (if valid)
            main_delta[:, nm1] -= 2.0
            m1 = nm1 + (ones_idx - i)  # (len(ones),)
            m2 = nm1 + (i - ones_idx)  # (len(ones),)
            # Add +1 at (j_idx, m1[j_idx]) for valid m1
            vm1 = (m1 >= 0) & (m1 < n_corr)
            for jj in np.where(vm1)[0]:
                main_delta[jj, m1[jj]] += 1.0
            vm2 = (m2 >= 0) & (m2 < n_corr)
            for jj in np.where(vm2)[0]:
                main_delta[jj, m2[jj]] += 1.0
            
            # Compute new max for each j
            new_vals = corr[None, :] + main_delta  # (len(ones), n_corr)
            new_maxes = new_vals.max(axis=1) * dx
            
            # Find best j
            j_best_local = new_maxes.argmin()
            if new_maxes[j_best_local] < cur_c - best_improve:
                best_improve = cur_c - new_maxes[j_best_local]
                best_i = i
                best_j = ones_idx[j_best_local]
                best_new_corr = (corr + main_delta[j_best_local]).copy()
        
        if best_i >= 0:
            # Perform the swap
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = best_new_corr
            cur_c = corr.max() * dx
            effect = compute_effect(h)
            ones_idx = np.where(h > 0.5)[0]
            zeros_idx = np.where(h <= 0.5)[0]
        else:
            break
    
    return h, cur_c
```
