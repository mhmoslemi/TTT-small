        w = 1.0 / n_active
        shift = k - n + 1
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += w * dx * (1.0 - h[idx - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= w * dx * h[idx + shift]
    
    return grad, len(active_indices)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: float(x.sum() - n/2.0),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_run(h, n, deadline, phases):
    """Multi-phase subgradient with Nesterov acceleration."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    y = h.copy()
    
    for phase, (eps, lr, decay, n_iter) in enumerate(phases):
        if time.time() > deadline:
            break
        for it in range(n_iter):
            if time.time() > deadline:
                return best_h, best_c
            grad, _ = subgradient_active(y, n, eps)
            x_new = y - lr * grad
            x_new = project_feasible(x_new, n)
            c_new = compute_c5(x_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = x_new.copy()
            y = 0.5 * (x_new + x) + 0.5 * x_new
            y = project_feasible(y, n)
            x = x_new
            lr *= decay
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas, max_passes_per_delta=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes_per_delta):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(20):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def try_resolution(h_src, n_src, n_target, deadline):
    """Interpolate to new resolution and optimize."""
    x_src = np.linspace(0, 2, n_src)
    x_tgt = np.linspace(0, 2, n_target)
    h_interp = np.interp(x_tgt, x_src, h_src)
    h_interp = project_feasible(h_interp, n_target)
    
    # SLSQP on new resolution
    for alpha in [10000, 100000, 1000000]:
        if time.time() > deadline:
            break
        alloc = min(80, deadline - time.time() - 10)
        if alloc < 15:
            break
        h_opt = slsqp_optimize(h_interp, n_target, alpha, maxiter=2000, ftol=1e-15)
        h_interp = h_opt
    
    # Local search
    if time.time() < deadline - 5:
        if n_target <= 250:
            h_interp, _ = pair_search(h_interp, n_target, deadline - 2, [0.005, 0.002, 0.001, 0.0005])
        else:
            h_interp, _ = near_pair_search(h_interp, n_target, deadline - 2, [0.005, 0.002, 0.001], radius=30)
    
    h_interp = project_feasible(h_interp, n_target)
    return h_interp, compute_c5(h_interp, n_target)

def run(seed=42, budget_s=1100, **kwargs):
