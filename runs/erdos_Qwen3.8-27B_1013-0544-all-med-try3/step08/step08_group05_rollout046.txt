import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def project_binary(h, n):
    target = n // 2
    idx = np.argpartition(-h, target)[:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)


def binary_local_search(h, n, deadline, max_sweeps=5000, seed=42):
    dx = 2.0 / n
    h = project_binary(h, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = float(corr.max())
    cur_c = cur_max * dx
    ncorr = 2 * n - 1
    k_vals = np.arange(-(n - 1), n, dtype=np.int64)

    best_c = cur_c
    best_h = h.copy()

    ones_pos = np.where(h == 1)[0].copy()
    zeros_pos = np.where(h == 0)[0].copy()

    rng = np.random.RandomState(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        improved = False
        # Try each zero position (shuffled), for each find best one to swap with
        z_order = rng.permutation(len(zeros_pos))

        for zi in z_order:
            i = zeros_pos[zi]
            if time.time() > deadline:
                break

            # Compute A_i = (1-h[i-k])_valid - h[i+k]_valid
            ik = i - k_vals
            i_k = i + k_vals
            A_i = np.zeros(ncorr)
            m1 = (ik >= 0) & (ik < n)
            A_i[m1] += 1.0 - h[ik[m1]]
            m2 = (i_k >= 0) & (i_k < n)
            A_i[m2] -= h[i_k[m2]]

            # For each one position j, compute best new_max
            best_j_idx = -1
            best_new_max = cur_max
            oj_order = rng.permutation(len(ones_pos))

            for oj in oj_order:
                j = ones_pos[oj]
                jk = j - k_vals
                j_k = j + k_vals
                B_j = np.zeros(ncorr)
                m3 = (jk >= 0) & (jk < n)
                B_j[m3] -= 1.0 - h[jk[m3]]
                m4 = (j_k >= 0) & (j_k < n)
                B_j[m4] += h[j_k[m4]]

                new_max = float(np.max(corr + A_i + B_j))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_j_idx = oj

            if best_j_idx >= 0 and best_new_max < cur_max - 1e-12:
                j = ones_pos[best_j_idx]
                # Recompute delta for the chosen pair
                jk = j - k_vals
                j_k = j + k_vals
                delta = A_i.copy()
                m3 = (jk >= 0) & (jk < n)
                delta[m3] -= 1.0 - h[jk[m3]]
                m4 = (j_k >= 0) & (j_k < n)
                delta[m4] += h[j_k[m4]]

                h[i] = 1.0
                h[j] = 0.0
                corr += delta
                cur_max = best_new_max
                new_c = cur_max * dx
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()

                # Update index arrays
                ones_pos = np.delete(ones_pos, best_j_idx)
                ones_pos = np.append(ones_pos, i)
                zeros_pos = np.delete(zeros_pos, zi)
                zeros_pos = np.append(zeros_pos, j)
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                break

    return best_h, best_c


def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    p1 = t0 + 0.40 * (budget_s - 5)

        best_h = hb.copy()

    # Phase 2: Try other resolutions (30% time)
    p2 = t0 + 0.70 * (budget_s - 5)
    candidate_ns = [248, 252, 260, 300]
    for nh in candidate_ns:
        if time.time() > p2 - 30:
            break
        if nh == best_n:
            continue
        # Interpolate to new resolution
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hh_bin = project_binary(hh, nh)
        hb2, cb2 = binary_local_search(hh_bin, nh, min(p2 - 20, time.time() + 80),

            best_h = hb2.copy()
            best_n = nh

    p3 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p3:
        h_cont = best_h.copy()
        h_cont = project_feasible(h_cont + 0.03 * rng.randn(best_n), best_n)
        for alpha in [20000, 100000, 500000, 2000000]:
            if time.time() > p3 - 5:

            h_cont = slsqp_smooth(h_cont, best_n, alpha, 800, p3 - 3)
            c = compute_c5(h_cont, best_n)
            if c < best_c:
                best_c = c
                best_h = h_cont.copy()
        # Also try from binary projection of the polished solution
        h_bin_polish = project_binary(h_cont, best_n)
        hb3, cb3 = binary_local_search(h_bin_polish, best_n, p3, max_sweeps=2000, seed=seed + 99)

            best_c = cb3
            best_h = hb3.copy()

                                       max_sweeps=5000, seed=seed + 200)

            best_c = cb4
            best_h = hb4.copy()

    best_h = project_feasible(best_h, best_n)
    # Ensure exact constraint
    if best_n % 2 == 0 and np.allclose(best_h, np.round(best_h)):
        best_h = project_binary(best_h, best_n)
    else:
        best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
