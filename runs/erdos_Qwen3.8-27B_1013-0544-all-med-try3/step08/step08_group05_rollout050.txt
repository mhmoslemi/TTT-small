def smooth_obj_and_grad(x, n, alpha, lam=0.0):
    """
    Smoothed objective: (1/alpha)*log(sum exp(alpha * c_k)) + lam*(sum(x)-n/2)^2
    where c_k are the correlation values * dx.
    Returns (value, gradient).
    """

    m = c.max()

    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    S = exp_shifted.sum()

    # Gradient of the log-sum-exp part
    # d/dx[k] of (1/alpha)*log(S) = (1/(alpha*S)) * sum_m alpha*exp(alpha*c_m) * dc_m/dx[k]
    # = (1/S) * sum_m exp(alpha*c_m) * dc_m/dx[k]
    # dc_m/dx[k] = dx * (1 - 2*x[m-k]) if 0 <= m-k <= n-1, else 0
    # So sum_m exp(alpha*c_m) * dc_m/dx[k] = dx * sum_j exp(alpha*c_{j+k}) * (1-2*x[j])
    # Let w = exp(alpha*(c-m)) and v = 1-2*x
    # Then: sum_j w[j+k] * v[j] for j=0..n-1
    w = exp_shifted / S  # normalized weights, length 2n-1
    v = 1.0 - 2.0 * x  # length n
    grad = np.zeros(n)
    for k in range(n):
        # sum_{j=0}^{n-1} w[j+k] * v[j]
        # w has length 2n-1, indices 0..2n-2
        # j+k ranges from k to k+n-1, need k+n-1 <= 2n-2, i.e., k <= n-1. OK.
        grad[k] = dx * np.dot(w[k:k+n], v)
    
    # Add constraint penalty gradient
    if lam > 0:
        val += lam * (x.sum() - n/2.0)**2
        grad += 2.0 * lam * (x.sum() - n/2.0)
    
    return val, grad

import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha, lam=0.0):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -700, 700)
    exp_shifted = np.exp(shifted)
    S = exp_shifted.sum()
    val = m + np.log(S) / alpha
    w = exp_shifted / S
    v = 1.0 - 2.0 * x
    grad = np.zeros(n)
    for k in range(n):
        grad[k] = dx * np.dot(w[k:k+n], v)
    if lam > 0:
        s_diff = x.sum() - n / 2.0
        val += lam * s_diff * s_diff
        grad += 2.0 * lam * s_diff
    return val, grad

def lbfgs_optimize(h, n, alpha, lam, maxiter, deadline):
    dx = 2.0 / n
    def wrapped(x):
        x = project_feasible(x, n)
        val, g = smooth_obj_grad(x, n, alpha, lam)
        return val, g
    try:
        res = fmin_l_bfgs_b(wrapped, h, pgtol=1e-14, maxiter=maxiter,
                            factr=1e14, iprint=0)
        return project_feasible(res[0], n), res[0]
    except Exception:
        return h, h

def slsqp_run(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        val, _ = smooth_obj_grad(x, n, alpha, 0.0)
        return val
    def obj_grad(x):
        _, g = smooth_obj_grad(x, n, alpha, 0.0)
        return g
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

    if rng is None:
        rng = np.random.RandomState(0)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy().astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = None
        best_val = cur_c
        best_pair = None
        # Evaluate all single swaps: move 1 from j to i
        for i in zeros:
            im = i + m_range
            imn = i - m_range
            for j in ones:
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)

                msk = (jmn >= 0) & (jmn < n)

                    best_pair = (i, j)
        if best_pair is None:

        ones = np.where(h == 1)[0]

        if len(ones) < 2 or len(zeros) < 2:
            break
        # Pick two random ones and two random zeros
        j1, j2 = rng.choice(ones, size=2, replace=False)
        i1, i2 = rng.choice(zeros, size=2, replace=False)
        
        # Compute delta for swapping: h[i1]+=1, h[i2]+=1, h[j1]-=1, h[j2]-=1
        delta = np.zeros(2 * n - 1)
        for i in [i1, i2]:
            im = i + m_range
            imn = i - m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
        for j in [j1, j2]:
            jm = j + m_range
            jmn = j - m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
        # Diagonal corrections
        delta[n-1] -= 4.0
        for a in [i1, i2]:
            for b in [j1, j2]:
                kb = n - 1 + (b - a)
                if 0 <= kb < 2*n-1:
                    delta[kb] += 1.0
                ka = n - 1 + (a - b)
                if 0 <= ka < 2*n-1:
                    delta[ka] += 1.0
        # Cross terms between the two new 1s and two new 0s
        new_max = (corr + delta).max() * dx
        if new_max < cur_c - 1e-14:

            h[j1], h[j2] = 0.0, 0.0

            cur_c = new_max
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
