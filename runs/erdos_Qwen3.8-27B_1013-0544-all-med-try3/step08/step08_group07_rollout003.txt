    """Steepest descent: try all swaps, accept best improvement each step."""
    best_c = compute_c5(h, n)
    h = h.copy()
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])

    for sweep in range(max_sweeps):
        if time.time() > deadline:

        val = m + (sm + np.log(exps.sum())) / alpha
        # Gradient of g[k] w.r.t. h[i]
        # g[k] = sum_x h[x](1-h[x+k]) * dx
        # dg[i] = dx * sum_k weights[k] * (dG_k/dh[i])
        # dG_k/dh[i] = (1-h[i+k_mod]) - h[i-k_mod] (with appropriate indexing)
        # This is complex; use numerical gradient
        grad = np.zeros(n)
        eps = 1e-7
        for i in range(n):
            x[i] += eps
            g2 = np.correlate(x, 1 - x, mode='full') * dx
            m2 = g2.max()
            sh2 = alpha * (g2 - m2)
            sm2 = sh2.max()
            val2 = m2 + (sm2 + np.log(np.exp(sh2 - sm2).sum())) / alpha
            x[i] -= eps
            grad[i] = (val2 - val) / eps
        return grad
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h

def binary_steepest(h, n, deadline, max_sweeps=500):
    """Best-improvement binary local search."""
    best_c = compute_c5(h, n)
    h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        n_z = len(zeros)
        n_o = len(ones)
        if n_z == 0 or n_o == 0:
            break
        
        best_dc = 0.0
        best_i = -1
        best_j = -1
        
        # Try all swaps (or a random subset for very large n)
        if n_z * n_o > 50000:
            # Random subset
            rng = np.random.RandomState(sweep + 1)
            pairs = [(rng.randint(n_z), rng.randint(n_o)) for _ in range(50000)]
        else:
            pairs = [(ii, jj) for ii in range(n_z) for jj in range(n_o)]
        
        for (ii, jj) in pairs:
            i = zeros[ii]
            j = ones[jj]
            h[i], h[j] = 1.0, 0.0
            c = compute_c5(h, n)
            h[i], h[j] = 0.0, 1.0
            dc = c - best_c
            if dc < best_dc:
                best_dc = dc
                best_i = i
                best_j = j
        
        if best_i >= 0:
            h[best_i], h[best_j] = 1.0, 0.0
            best_c += best_dc
            ones.remove(best_j)
            ones.append(best_i)
            zeros.remove(best_i)
            zeros.append(best_j)
        else:
            break
    
    return h, best_c

def first_improvement_binary(h, n, deadline, max_sweeps=500, seed=42):
    """First-improvement binary local search with randomization."""
    best_c = compute_c5(h, n)
    h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()

            break
        
        n_z = len(zeros)
        n_o = len(ones)
        if n_z == 0 or n_o == 0:
            break
        
        # Randomize order
        zero_order = list(range(n_z))
        one_order = list(range(n_o))
        rng.shuffle(zero_order)
        rng.shuffle(one_order)
        
        improved = False
        for ii in zero_order:
            if time.time() > deadline:
                break
            i = zeros[ii]
            for jj in one_order:
                j = ones[jj]
                h[i], h[j] = 1.0, 0.0
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
                else:
                    h[i], h[j] = 0.0, 1.0
            if improved:
                break
        
        if not improved:
            break
    
    return h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)
