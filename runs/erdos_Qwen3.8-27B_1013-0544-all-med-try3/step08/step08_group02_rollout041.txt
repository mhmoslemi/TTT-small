def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-16:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:

    return grad

    else:

def active_set_grad(x, n, active, alpha, mu):

    mx = vals.max()

    smx = sh.max()

    efficacy = build_efficacy(best_h, n)
    idx = np.arange(n)
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d > 1.0 or best_h[j] - d < 0.0:
                        continue
                    delta_c = d * (efficacy[i] - efficacy[j])
                    new_max = (c + delta_c).max()
                    if new_max < best_c - 1e-14:
                        best_h[i] += d
                        best_h[j] -= d
                        c += delta_c
                        best_c = new_max
                        update_efficacy(efficacy, i, d)
                        update_efficacy(efficacy, j, -d)
                        improved = True
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d < 0.0 or best_h[j] + d > 1.0:
                        continue
                    delta_c = -d * (efficacy[i] - efficacy[j])
                    new_max = (c + delta_c).max()
                    if new_max < best_c - 1e-14:
                        best_h[i] -= d
                        best_h[j] += d
                        c += delta_c
                        best_c = new_max
                        update_efficacy(efficacy, i, -d)
                        update_efficacy(efficacy, j, d)
                        improved = True
    
    return best_h, best_c

def subgradient_refine(h, n, deadline, n_iter=10000):
    """Subgradient method with adaptive step size"""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    eta = 0.01
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_c5_vec(best_h, n)
        c_max = c.max()
        # Active set: top shifts
        tol = max(1e-8, 0.001 * c_max)
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx_m in active:
            k = idx_m - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.01, 0.05)
        else:
            eta *= 0.95
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Load parent solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: L-BFGS at original resolution (150s)
    p1_end = t0 + 150
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.01,
        maxiters=[1500, 1500, 2000, 2000, 2500, 3000, 3000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active set at original resolution (100s)
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 100, deadline - 400)
        h2, c2 = run_active_set(best_h, best_n, p2_end, top_k=10,
            alphas=[50000, 200000, 1000000, 5000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to n=1000 and optimize (400s)
    n_fine = 1000
    if time.time() < deadline - 400:
        x_fine = np.linspace(0, 2, n_fine + 1)[:-1] + 1.0/n_fine  # cell centers
        x_coarse = np.linspace(0, 2, best_n + 1)[:-1] + 1.0/best_n
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        
        p3_end = min(time.time() + 350, deadline - 250)
