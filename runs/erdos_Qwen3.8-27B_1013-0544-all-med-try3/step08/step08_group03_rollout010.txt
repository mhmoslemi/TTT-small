import numpy as np
from scipy.optimize import minimize
import time


# ----------------------------- core objective -----------------------------
def c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def c5(h, n):
    return float(np.max(c5_vec(h, n)))


# ----------------------------- feasibility projection ---------------------
def project_feasible(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    h[h > 1.0] = 1.0
    h[h < 0.0] = 0.0
    for _ in range(6):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            lo = 0.0
            hi = float(room.max())
            for _ in range(60):
                mid = 0.5 * (lo + hi)
                if np.minimum(room, mid).sum() > d:
                    hi = mid
                else:
                    lo = mid
            h = h + np.minimum(room, hi)
        else:
            rem = h
            tot = rem.sum()
            if tot < 1e-14:
                break
            lo = 0.0
            hi = float(rem.max())
            for _ in range(60):
                mid = 0.5 * (lo + hi)
                if np.minimum(rem, mid).sum() > (-d):
                    hi = mid
                else:
                    lo = mid
            h = h - np.minimum(rem, hi)
    h[h > 1.0] = 1.0
    h[h < 0.0] = 0.0
    for _ in range(4):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0.0:
            j = int(np.argmax(1.0 - h))
            h[j] = min(1.0, h[j] + d)
        else:
            j = int(np.argmax(h))
            h[j] = max(0.0, h[j] + d)
    return h


# ----------------------------- per-lag gradient (correct) ------------------
def lag_grad(m, h, n):
    # gradient of O_m = corr[m]*dx with corr = np.correlate(h,1-h,'full')
    dx = 2.0 / n
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[k:n] += (1.0 - h[0:nv])
        g[0:nv] += (-h[k:n])
    else:
        kk = -k
        nv = n - kk
        g[0:nv] += (1.0 - h[kk:n])
        g[kk:n] += (-h[0:nv])
    return g * dx


# ----------------------------- smooth-max machinery ------------------------
def smooth_obj_core(x, n, a):
    c = c5_vec(x, n)
    m = c.max()
    zz = a * (c - m)
    zz = zz - zz.max()
    return float(m + np.log(np.exp(zz).sum()) / a)


def smooth_grad_core(x, n, a):
    c = c5_vec(x, n)
    m = c.max()
    zz = a * (c - m)
    zz = zz - zz.max()
    w = np.exp(zz)
    w /= w.sum()
    g = np.zeros(n)
    L = len(c)
    for mi in range(L):
        wi = w[mi]
        if wi <= 1e-14:
            continue
        g += wi * lag_grad(mi, x, n)
    return g


def smooth_obj(x, n, a, mu, target):
    c = c5_vec(x, n)
    m = c.max()
    zz = a * (c - m)
    zz = zz - zz.max()
    val = m + np.log(np.exp(zz).sum()) / a
    val += mu * (x.sum() - target) ** 2
    return float(val)


def smooth_grad(x, n, a, mu, target):
    c = c5_vec(x, n)
    m = c.max()
    zz = a * (c - m)
    zz = zz - zz.max()
    w = np.exp(zz)
    w /= w.sum()
    g = np.zeros(n)
    L = len(c)
    for mi in range(L):
        wi = w[mi]
        if wi <= 1e-14:
            continue
        g += wi * lag_grad(mi, x, n)
    g += 2.0 * mu * (x.sum() - target)
    return g


def smooth_obj_A(x, n, A, a, mu, target):
    c = c5_vec(x, n)
    cA = c[A]
    m = cA.max()
    zz = a * (cA - m)
    zz = zz - zz.max()
    val = m + np.log(np.exp(zz).sum()) / a
    val += mu * (x.sum() - target) ** 2
    return float(val)


def smooth_grad_A(x, n, A, a, mu, target):
    c = c5_vec(x, n)
    cA = c[A]
    m = cA.max()
    zz = a * (cA - m)
    zz = zz - zz.max()
    w = np.exp(zz)

    g = np.zeros(n)
    for ai in range(len(A)):
        wi = w[ai]
        if wi <= 1e-14:
            continue
        g += wi * lag_grad(int(A[ai]), x, n)
    g += 2.0 * mu * (x.sum() - target)
    return g


# ----------------------------- gradient self test --------------------------
def check_gradient_ok():
    try:
        n = 20
        rng = np.random.RandomState(1)
        x = 0.2 + 0.6 * rng.rand(n)
        x = project_feasible(x, n)
        a = 6.0
        f0 = smooth_obj_core(x, n, a)
        g = smooth_grad_core(x, n, a)
        eps = 1e-5
        fd = np.zeros(n)
        for i in range(n):
            xp = x.copy()
            xp[i] += eps
            fd[i] = (smooth_obj_core(xp, n, a) - f0) / eps
        num = float(np.max(np.abs(g - fd)))
        den = float(np.max(np.abs(fd))) + 1e-12
        return (num / den) < 5e-4
    except Exception:
        return True


# ----------------------------- exact subgradient polish ---------------------
def polish_exact(h, n, deadline):
    margin = 1.0
    h = project_feasible(h, n)
    cval = c5(h, n)
    best = h.copy()
    bestc = cval
    lr = 0.4
    for _ in range(2500):
        if time.time() > deadline - margin:
            break
        cvec = c5_vec(h, n)
        m = cvec.max()
        A = np.where(cvec >= m - 1e-4)[0]
        if len(A) == 0:
            break
        g = np.zeros(n)
        for mi in A:
            g += lag_grad(int(mi), h, n)
        g /= float(len(A))
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        d = -g / gn
        step = lr
        accepted = False
        for _ in range(18):
            hn = project_feasible(h + step * d, n)
            cn = c5(hn, n)
            if cn < cval - 1e-15:
                h = hn
                cval = cn
                best = hn.copy()
                bestc = cn
                accepted = True
                break
            step *= 0.5
        if not accepted:
            lr *= 0.6
            if lr < 1e-7:
                break
    return best, float(bestc)

        except Exception:

        hn = project_feasible(res.x, n)

    # active-set finishing: converge the current top lags

        except Exception:
            continue
        hn = project_feasible(res.x, n)

        if cn < bestc:

    if time.time() < deadline - margin:
