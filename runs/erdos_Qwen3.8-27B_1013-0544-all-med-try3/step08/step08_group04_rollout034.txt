def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):

        if abs(diff) < 1e-14:
            break
        if diff > 0:

            if tot < 1e-15:

            h += (diff / tot) * h

def coordinate_descent(h, n, deadline, alpha=10000):

    best_c = compute_c5(h, n)
    dx = 2.0 / n
    nn = n - 1

    # I_i[k] = (1-h[i+k-nn])_valid - h[i-k+nn]_valid
    # This changes when h changes, so we need to recompute periodically
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for delta in deltas:
        while time.time() < deadline:
            improved = False

            g = 1.0 - h
            I = np.zeros((n, 2*n-1))
            for i in range(n):

                k_hi = min(2*n-2, 2*nn - i)
                if k_lo <= k_hi:

                        else:
                            h[i] -= delta
                            h[j] += delta
                
                # Move mass from i to best j (decrease h[i])
                if h[i] - delta >= 0.0:
                    # We want S_j < S_i, so best j is argmin S
                    j = int(np.argmin(S))
                    if j != i and h[j] + delta <= 1.0:
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
            
            if not improved:
                break
    
    return h, best_c

Wait, there's a subtlety. The "best j" depends on i (it's argmax of S, not argmax of S - S_i = argmax of S, since S_i is constant for fixed i). So the best j for any i (for increasing h[i]) is always argmax(S). And the best j for decreasing h[i] is always argmin(S). This means I only need to identify the top and bottom elements of S, not check all pairs.

But this greedy approach might not find all useful moves. For example, the second-best j might be useful if the best j is already at a boundary. Let me try top-k and bottom-k.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    return m + (s + np.log(np.sum(np.exp(sh - s)))) / alpha

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return smooth_max_val(c, alpha)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() > deadline:
            return project_feasible(h, n)
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    exp_sh = np.exp(sh - s)
    w = exp_sh / exp_sh.sum()
    obj = m + (s + np.log(exp_sh.sum())) / alpha
    
    grad = np.zeros(n)
    nn = n - 1
    for j in range(n):
        k_lo = max(0, nn - j)
        k_hi = min(2 * n - 2, 2 * nn - j)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            idx = j + ks - nn
            grad[j] += np.dot(w[ks], g[idx])
        k_lo2 = max(0, j)
        k_hi2 = min(2 * n - 2, j + nn)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            idx2 = j - ks2 + nn
            grad[j] -= np.dot(w[ks2], h[idx2])
    grad *= dx
    return obj, grad

def projected_gradient(h, n, alpha, n_iter, deadline):
    h_cur = project_feasible(h.copy(), n)
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    
    lr = 1.0 / (alpha * 1.5)
    stale = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = smooth_max_grad(h_cur, n, alpha)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        for trial in range(15):
            h_new = h_cur - lr * grad / (grad_norm + 1e-15)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-6:
                h_cur = h_new
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = h_cur.copy()
                    stale = 0
                break
            lr *= 0.5
        
        lr = min(lr * 1.05, 1.0 / (alpha * 0.5))
        stale += 1
        if stale > 200:
            break
    
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_sweeps=500):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    nn = n - 1
    
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    
    for delta in deltas:
        sweeps = 0
        while time.time() < deadline and sweeps < max_sweeps:
            sweeps += 1
            improved = False
            g = 1.0 - h
            c = np.correlate(h, g, mode='full') * dx
            m = c.max()
            sh = 50000.0 * (c - m)
            s = sh.max()
            exp_sh = np.exp(sh - s)
            w = exp_sh / exp_sh.sum()
            
            I = np.zeros((n, 2 * n - 1))
            for i in range(n):
                k_lo = max(0, nn - i)
                k_hi = min(2 * n - 2, 2 * nn - i)
                if k_lo <= k_hi:
                    ks = np.arange(k_lo, k_hi + 1)
                    I[i, ks] += g[i + ks - nn]
                k_lo2 = max(0, i)
                k_hi2 = min(2 * n - 2, i + nn)
                if k_lo2 <= k_hi2:
                    ks2 = np.arange(k_lo2, k_hi2 + 1)
                    I[i, ks2] -= h[i - ks2 + nn]
            
            S = I @ w
            
            order_inc = np.argsort(-S)

                    return h, best_c

                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)

                            best_c = c_new
                            improved = True
                            S = I @ w
                            break

                            h[j] += delta
                for j in order_dec[:top_k]:
                    j = int(j)
                    if j == i:
                        continue
                    if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)

def near_pair_search(h, n, deadline, deltas, radius=50):
