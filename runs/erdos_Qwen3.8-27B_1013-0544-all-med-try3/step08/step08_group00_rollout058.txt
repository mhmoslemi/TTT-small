import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    s = np.sum(np.exp(sh - mx))
    return float(m + (mx + np.log(s)) / alpha)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w_sum = w.sum()
    w = w / w_sum
    dx = 2.0 / n
    nn1 = n - 1
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx] * dx
        if wi < 1e-14:
            continue
        k = idx - nn1
        if k >= 0:
            nv = n - k
            grad[:nv] += wi * (1.0 - x[k:])
            grad[k:] += -wi * x[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wi * (1.0 - x[:nv])
            grad[:nv] += -wi * x[kk:]
    return grad

def lbfgs_optimize(h, n, deadline, alphas=(200, 1000, 5000, 20000, 100000)):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_val, h, jac=smooth_max_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 20000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 200000, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                h = project_feasible(res.x, n)
        except Exception:
            pass
    return best_h, best_c

def active_set_qp_step(h, n, top_m=15, rho=0.01):
    """SQP-style step: minimize linearization of top active + regularization."""
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    order = np.argsort(-c_vec)
    active = order[:min(top_m, len(order))]
    active = list(active)
    
    dx = 2.0 / n
    nn1 = n - 1
    
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t
    
    grads = []
    c_vals = []
    for idx in active:
        k = idx - nn1
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] += (1.0 - h[k:])
            g[k:] += -h[:nv]
        else:
            kk = -k
            nv = n - kk
            g[kk:] += (1.0 - h[:nv])
            g[:nv] += -h[kk:]
        g *= dx
        grads.append(g)
        c_vals.append(c_vec[idx])
    
    n_act = len(active)
    total_ineq = n_act + 2 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        A_ub[row, :n] = grads[j]
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds_list = [(-rho, rho)] * n + [(0.0, 2.0)]
    
    try:

def coord_descent(h, n, deadline, n_passes=None):

            max_val = c_vec.max()

                    nv = n - kk

            if len(active) > 0:

                            improved = True
                        else:
                            h[i] -= dt
                            h[j] += dt
                    # Move mass from i to j

        thresh = np.partition(np.sort(c_vec)[::-1], n_act - 1)[n_act - 1] if 2*n-1 > n_act else c_vec.min()
        active = np.where(c_vec >= thresh)[0]
        
        # Weights: proportional to (c_i - min_active)^2
        w = c_vec[active] - c_vec[active].min()
        w = (w * w + 1e-10)
        w /= w.sum()
        
        dx = 2.0 / n
        nn1 = n - 1
        grad = np.zeros(n)
        for j, idx in enumerate(active):
            wi = w[j] * dx
            k = idx - nn1
            if k >= 0:
                nv = n - k
                grad[:nv] += wi * (1.0 - h[k:])
                grad[k:] += -wi * h[:nv]
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wi * (1.0 - h[:nv])
                grad[:nv] += -wi * h[kk:]
        
        # Step size (Polyak-style)
        step = alpha_init / (it + 1)
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            lb = max(lb, best_c - 0.02)
        h = h_new
    return best_h, best_c

def multi_start_refine(h, n, deadline, n_starts=3, perturb=0.01):
    """Try small random perturbations and re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    
    for s in range(n_starts):
        if time.time() > deadline - 30:
            break
        h_pert = best_h + rng.randn(n) * perturb / np.sqrt(n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Quick SQP from perturbed point
        h_opt, c_opt = sqp_loop(h_pert, n, time.time() + 15, max_iter=100, rho_init=0.02, top_m=10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    """Interpolate from n_old to n_new points."""
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, n, a):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = a * (c - m)
        mx = sh.max()
        s = np.sum(np.exp(sh - mx))
        return m + (mx + np.log(s)) / a
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_overall_h = project_feasible(h_init, n_orig)
    best_overall_c = compute_c5(best_overall_h, n_orig)
    best_overall_n = n_orig
    
    # Phase 1: Quick L-BFGS at original resolution (30s)
    t1 = min(t0 + 30, deadline - 900)
    if t1 > time.time() + 3:
        h1, c1 = lbfgs_optimize(best_overall_h, n_orig, t1, alphas=(500, 5000, 30000))
        if c1 < best_overall_c:
            best_overall_c = c1
            best_overall_h = h1.copy()
    
    # Phase 2: SQP refinement at original resolution (40s)
    t2 = min(t0 + 70, deadline - 850)
    if t2 > time.time() + 5:
        h2, c2 = sqp_loop(best_overall_h, n_orig, t2, max_iter=200, rho_init=0.04, top_m=12)
        if c2 < best_overall_c:
            best_overall_c = c2
            best_overall_h = h2.copy()
    
    # Phase 3: Upsample to n=1000
    n_hi = 1000
    h_hi = upsample(best_overall_h, len(best_overall_h), n_hi)
    c_hi = compute_c5(h_hi, n_hi)
    
    # Phase 4: L-BFGS at n=1000 (80s)
    t4 = min(time.time() + 80, deadline - 550)
    if t4 > time.time() + 5:
        h_hi, c_hi = lbfgs_optimize(h_hi, n_hi, t4, alphas=(500, 2000, 10000, 50000, 200000))
    
    # Phase 5: SQP at n=1000 (60s)
    t5 = min(time.time() + 60, deadline - 450)
    if t5 > time.time() + 5:
        h_hi, c_hi = sqp_loop(h_hi, n_hi, t5, max_iter=300, rho_init=0.03, top_m=15)
    
    # Phase 6: Subgradient method (40s)
    t6 = min(time.time() + 40, deadline - 350)
    if t6 > time.time() + 5:
        h_sg, c_sg = subgradient_method(h_hi, n_hi, t6, n_iter=3000)
        if c_sg < c_hi:
            h_hi, c_hi = h_sg, c_sg
    
    # Phase 7: Coordinate descent (200s)
    t7 = min(time.time() + 200, deadline - 180)
    if t7 > time.time() + 5:
        h_hi, c_hi = coord_descent(h_hi, n_hi, t7)
    
    # Phase 8: Pair search (150s)
    t8 = min(time.time() + 150, deadline - 100)
    if t8 > time.time() + 5:
        h_hi, c_hi = pair_search(h_hi, n_hi, t8, radius=60)
    
    # Phase 9: SQP after local search (40s)
    t9 = min(time.time() + 40, deadline - 60)
    if t9 > time.time() + 5:
        h_hi, c_hi = sqp_loop(h_hi, n_hi, t9, max_iter=200, rho_init=0.02, top_m=10)
    
    # Phase 10: Multi-start refinement (30s)
    t10 = min(time.time() + 30, deadline - 30)
    if t10 > time.time() + 5:
        h_hi, c_hi = multi_start_refine(h_hi, n_hi, t10, n_starts=2, perturb=0.005)
    
    # Phase 11: Final SLSQP polish
    if time.time() < deadline - 10:
        t11 = deadline - 10
        h_final, c_final = slqp_polish(h_hi, n_hi, t11, alpha=1000000, maxiter=5000)
        if c_final < c_hi:
            h_hi, c_hi = h_final, c_final
    
    # Track best overall
    if c_hi < best_overall_c:
        best_overall_c = c_hi
        best_overall_h = h_hi.copy()
        best_overall_n = n_hi
    else:
        h_hi = best_overall_h
        c_hi = best_overall_c
        n_hi = best_overall_n
    
    # Final feasibility check
    n_final = len(best_overall_h)
    best_overall_h = project_feasible(best_overall_h, n_final)
    best_overall_c = compute_c5(best_overall_h, n_final)
    
    return best_overall_h, best_overall_c, n_final
