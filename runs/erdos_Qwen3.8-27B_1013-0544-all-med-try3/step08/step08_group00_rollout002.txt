                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.04, active_frac=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        n_keep = max(15, int(len(active) * active_frac * 3))
        if len(active) > n_keep:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:n_keep]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.06:
                rho = min(rho * 1.2, 0.08)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if no_improve > 30:
                break
    return best_h, best_c

def binary_refine(h, n, deadline):
    """Push values toward 0 or 1 while maintaining feasibility."""
    best_c = compute_c5(h, n)
    h = h.copy()
    eps_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for eps in eps_list:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            h_copy = h.copy()
            changed = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if h_copy[i] < eps:
                    h_copy[i] = 0.0
                    changed = True
                elif h_copy[i] > 1.0 - eps:
                    h_copy[i] = 1.0
                    changed = True
            if changed:
                h_new = project_feasible(h_copy, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    h = h_new
                    improved = True
                else:
                    # Try partial push
                    h2 = h.copy()
                    for i in range(n):
                        if h2[i] < eps * 0.5:
                            h2[i] = 0.0
                        elif h2[i] > 1.0 - eps * 0.5:
                            h2[i] = 1.0
                    h2 = project_feasible(h2, n)
                    c2 = compute_c5(h2, n)
                    if c2 < best_c - 1e-14:
                        best_c = c2
                        h = h2
                        improved = True
    return h, best_c

def fast_local_search(h, n, deadline, radius=60):
    """Targeted local search on top gradient regions."""
    best_c = compute_c5(h, n)
    h = h.copy()
    steps = [0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for idx in active[:30]:
                grad += np.abs(shift_grad_single(h, n, idx))
            grad /= min(len(active), 30)
            top_pos = np.argsort(-grad)[:min(radius * 2, n)]
            for i_val in top_pos:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += step, h[j] -= step
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Try h[i] -= step, h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: L-BFGS warmup with increasing alpha (60s)
    t1 = min(t0 + 60, deadline - 900)
    if t1 > time.time() + 5:
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            alphas=[300, 1500, 8000, 30000, 100000],
            mus=[100.0, 1000.0, 10000.0, 100000.0, 500000.0])

    # Phase 2: SQP optimization (120s)
    t2 = min(t0 + 180, deadline - 700)
    if t2 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=600, rho_init=0.04, active_frac=0.06)

    # Phase 3: Binary refinement (30s)
    t3 = min(t0 + 210, deadline - 650)
    if t3 > time.time() + 5:
        best_h, best_c = binary_refine(best_h, n_orig, t3)

    # Phase 4: SQP again after binary push (80s)
    t4 = min(t0 + 290, deadline - 550)
    if t4 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t4, max_iter=400, rho_init=0.02, active_frac=0.05)

    # Phase 5: Local search (80s)
    t5 = min(t0 + 370, deadline - 450)
    if t5 > time.time() + 5:
        best_h, best_c = fast_local_search(best_h, n_orig, t5, radius=60)

    # Phase 6: L-BFGS at high alpha (60s)
    t6 = min(t0 + 430, deadline - 380)
    if t6 > time.time() + 5:
        best_h, best_c = run_lbfgs(best_h, n_orig, t6,
            alphas=[50000, 200000], mus=[200000.0, 1000000.0])

    # Phase 7: SQP final (50s)
    t7 = min(t0 + 480, deadline - 330)
    if t7 > time.time() + 5:
        best_h, best_c = sqp_optimize(best_h, n_orig, t7, max_iter=300, rho_init=0.015, active_frac=0.04)

    # Phase 8: Binary refine again (20s)
    t8 = min(t0 + 500, deadline - 310)
    if t8 > time.time() + 3:
        best_h, best_c = binary_refine(best_h, n_orig, t8)

    # Phase 9: Upsample to n=1000
    n_high = 1000
    if time.time() < deadline - 250:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)

        # L-BFGS on high res (60s)
        t9 = min(time.time() + 60, deadline - 150)
        if t9 > time.time() + 5:
            h_high, c_high = run_lbfgs(h_high, n_high, t9,
                alphas=[1000, 5000, 20000],
                mus=[500.0, 5000.0, 50000.0])

        # SQP on high res (60s)
        t10 = min(time.time() + 60, deadline - 70)
        if t10 > time.time() + 5:
            h_high, c_high = sqp_optimize(h_high, n_high, t10, max_iter=300, rho_init=0.03, active_frac=0.05)

        # Local search on high res (40s)
        t11 = min(time.time() + 40, deadline - 30)
        if t11 > time.time() + 5:
            h_high, c_high = fast_local_search(h_high, n_high, t11, radius=40)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_orig = n_high

    # Final SLSQP polish
    if time.time() < deadline - 15:
        h_final, c_final = slqp_polish(best_h, len(best_h), deadline - 3, alpha=2000000, maxiter=5000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
