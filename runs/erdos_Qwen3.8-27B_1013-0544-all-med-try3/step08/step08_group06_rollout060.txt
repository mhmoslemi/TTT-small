def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr_array(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

def simulated_annealing(h, n, t_start, t_end, n_iter, deadline, rng):
    """Simulated annealing to escape local optima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    cur_energy = cur_c
    
    # Precompute correlation indices
    # For a move (i, j, d): h[i] += d, h[j] -= d
    # The change in correlation can be computed incrementally
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Temperature schedule
        progress = it / max(1, n_iter)
        temp = t_start * (t_end / t_start) ** progress
        
        # Choose a random move
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        d = rng.choice([-1, 1]) * rng.uniform(0.001, 0.05)
        
        hi = cur_h[i] + d
        hj = cur_h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        
        cur_h[i] = hi
        cur_h[j] = hj
        
        new_c = compute_c5(cur_h, n)
        
        if new_c < cur_energy:
            cur_energy = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # Accept with probability exp(-dE/T)
            delta = new_c - cur_energy
            if delta / max(temp, 1e-15) < -50:
                # Too unlikely to accept
                cur_h[i] = cur_h[i] - d
                cur_h[j] = cur_h[j] + d
            elif rng.random() < np.exp(-delta / max(temp, 1e-15)):
                cur_energy = new_c
            else:
                cur_h[i] = cur_h[i] - d
                cur_h[j] = cur_h[j] + d
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def topk_smooth_obj(x, n, k, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    idx = np.argsort(c)[-k:]
    topk_vals = c[idx]
    m = topk_vals.max()
    shifted = alpha * (topk_vals - m)
    mx = shifted.max()
    if mx > 500:
        base = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        base = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(base)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15, use_topk=False, topk=5):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if use_topk:
        obj = topk_smooth_obj
        args = (n, topk, alpha)
    else:
        obj = smooth_max_obj
        args = (n, alpha)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, restricted_radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        max_passes = 80 if n <= 200 else 40
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if restricted_radius is not None:
                    j_lo = max(0, i - restricted_radius)
                    j_hi = min(n, i + restricted_radius + 1)
                else:
                    j_lo = 0
                    j_hi = n
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def subgradient_descent(h, n, deadline, n_steps=5000):
    """Subgradient method for the max correlation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur = h.copy()
    cur_c = best_c
    dx = 2.0 / n

    for it in range(n_steps):
        if time.time() > deadline:
            break
        c_arr = compute_corr_array(cur, n)
        c_max = c_arr.max()
        if c_max < best_c:
            best_c = c_max
            best_h = cur.copy()

        # Step size
        lr = 0.05 / (1.0 + it * 0.001)

        # Subgradient of the max (use average of top few active lags)
        top_idx = np.argsort(c_arr)[-min(10, 2*n-1):]
        wg = np.zeros(2*n-1)
        wg[top_idx] = 1.0 / len(top_idx)

        # Compute subgradient: for each j, sum_m wg[m] * dc[m]/dx[j]
        # c[m] = sum_i x[i] * (1-x[i+m-n+1]) * dx (for valid indices)
        # dc[m]/dx[j] = dx * [(1-x[a]) * I(a valid) - x[b] * I(b valid)]
        # where a = j+m-n+1, b = j-m+n-1
        grad = np.zeros(n)
        for m in range(2*n-1):
            w = wg[m]
            if w == 0:
                continue
            # For lag m: c[m] = sum_i x[i+m-n+1] * (1-x[i]) * dx
            # x[j] appears as first factor when i = j - m + n - 1
            # and as second factor (inside 1-x[i]) when i = j and first factor is x[j+m-n+1]
            # Actually let me index differently:
            # c[m] = sum_i x[i] * (1 - x[i + m - n + 1]) * dx
            # valid i: 0 <= i < n and 0 <= i+m-n+1 < n
            # x[j] as first factor: i=j, need 0 <= j+m-n+1 < n
            # x[j] as (1-x) factor: i+m-n+1=j, i=j-m+n-1, need 0 <= j-m+n-1 < n
            offset = m - n + 1
            for j in range(n):
                a = j + offset  # index in (1-x) part
                b = j - offset  # index in x part (for the (1-x) contribution)
                g = 0.0
                if 0 <= a < n:
                    g += (1.0 - cur[a]) * dx
                if 0 <= b < n:
                    g += (-cur[b]) * dx
                grad[j] += w * g

        # Projected step
        cur -= lr * grad
        cur = project_feasible(cur, n)
        cur_c = compute_c5(cur, n)

    return best_h, best_c

def make_structural_init(n, variant):
    """Create structured initial functions to explore different regions."""
    x = np.linspace(0, 2, n)
    if variant == 0:
        # 3-bump pattern
        h = np.maximum(0, np.cos(3 * np.pi * x)) ** 2
    elif variant == 1:
        # Comb pattern
        h = (np.sin(np.pi * 5 * x) ** 2 > 0.5).astype(float)
    elif variant == 2:
        # Asymmetric two-bump
        h = np.maximum(0, np.cos(2 * np.pi * x)) ** 3 + 0.3 * np.maximum(0, np.cos(4 * np.pi * x))
    elif variant == 3:
        # Sine wave
        h = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x)
    elif variant == 4:
        # Step pattern: 1 on [0,0.4] U [0.8,1.2] U [1.6,2]
        h = np.zeros(n)
        h[x < 0.4] = 1.0
        h[(x >= 0.8) & (x < 1.2)] = 1.0
        h[x >= 1.6] = 1.0
    else:
        h = np.full(n, 0.5 + 0.3 * np.sin(2 * np.pi * 7 * x / 2))
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def optimize_single(h_start, n, deadline, rng, alphas=None, do_pair=True):
