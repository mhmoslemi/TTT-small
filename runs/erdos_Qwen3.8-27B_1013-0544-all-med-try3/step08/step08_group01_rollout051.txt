import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(c.max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_active_shifts(h, n, frac=0.001, max_k=10):
    """Get shifts within frac of the max value."""
    c = compute_c5_vec(h, n)
    m = c.max()
    thresh = m - max(1e-8, frac * m)
    active = np.where(c >= thresh)[0]
    if len(active) > max_k:
        # Keep only the top max_k
        idx = np.argsort(-c[active])[:max_k]
        active = active[idx]
    return active, m

def softmax_obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    # Add variance penalty among active shifts
    val += 0.1 * np.var(vals)  # encourage equalization
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_active(h, n, active, alpha, mu, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj_active, h, jac=softmax_grad_active,
                  args=(n, active, alpha, mu), method='L-BFGS-B',
                  bounds=bounds,
                  options={'maxiter': maxiter, 'ftol': ftol,
                           'gtol': 1e-12, 'maxls': 50})
    h_new = project_feasible(res.x, n)
    return h_new

def fine_pair_search(h, n, deadline, radius=30):

    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]

            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, 2):  # step by 2 for speed

                    return h, best_c

                hi = min(n, i + radius + 1)

                    if i == j:
                        continue

    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(c.max())

def compute_c_at_shifts(h, n, shift_indices):
    dx = 2.0 / n
    vals = np.zeros(len(shift_indices))
    for i, s in enumerate(shift_indices):
        k = s - (n - 1)
        if k >= 0:
            nv = n - k
            if nv > 0:
                vals[i] = np.sum(h[0:nv] * (1.0 - h[k:n])) * dx
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                vals[i] = np.sum(h[kk:n] * (1.0 - h[0:nv])) * dx
    return vals

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_active_shifts(h, n, frac=0.001, max_k=6):
    c = compute_c5_vec(h, n)
    m = c.max()
    thresh = m - max(1e-8, frac * m)
    active = np.where(c >= thresh)[0]
    if len(active) > max_k:
        order = np.argsort(-c[active])[:max_k]
        active = active[order]
    return active, m

def active_softmax_obj(x, n, active_idx, alpha, mu):
    vals = compute_c_at_shifts(x, n, active_idx)
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_softmax_grad(x, n, active_idx, alpha, mu):
    vals = compute_c_at_shifts(x, n, active_idx)
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, s in zip(w, active_idx):
        k = s - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def full_softmax_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_softmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()

    grad = np.zeros(n)

        improved = True

            step = 2 if n > 1000 else 1
            for i in range(0, n, step):

                lo = max(0, i - radius)

        h_r, c_r = random_perturb_search(best_h, n_best, t9, n_trials=3)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
    
    # Final: ensure feasibility and compute exact value
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    
    return best_h, best_c, n_best
