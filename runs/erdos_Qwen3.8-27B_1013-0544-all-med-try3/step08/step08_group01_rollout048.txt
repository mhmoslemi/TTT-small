    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_val(vals, alpha):
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        return mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return mx + np.log(np.sum(np.exp(sh))) / alpha

    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_val(vals, alpha):
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        return mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return mx + np.log(np.sum(np.exp(sh))) / alpha

def soft_max_weights(vals, alpha):
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def obj_full(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    val = soft_max_val(c, alpha)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def grad_full(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    w = soft_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def obj_active(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    val = soft_max_val(vals, alpha)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def grad_active(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    w = soft_max_weights(vals, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def eq_sum(x, n):
    return float(x.sum() - n / 2.0)

def eq_sum_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alpha_list, mu_list=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if mu_list is None:
        mu_list = [10.0, 100.0, 1000.0, 10000.0]
    for alpha in alpha_list:
        if time.time() > deadline - 3:
            break
        for mu in mu_list:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(obj_full, h.copy(), jac=grad_full,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxfun': 20000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def active_set_optimize(h, n, deadline, top_k=5, rederive_every=5):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alpha_list = [500, 2000, 10000, 50000, 200000]
    it = 0
    while time.time() < deadline - 3:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])][:top_k]
        if len(active) == 0:
            break
        for alpha in alpha_list:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(obj_active, h.copy(), jac=grad_active,
                              args=(n, active, alpha, 1.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxfun': 10000, 'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
        it += 1
    return h, best_c

def slqp_optimize(h, n, deadline, alpha=100000, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_full, h.copy(), args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'eps': 1e-8})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
    except Exception:
        pass
    return h, best_c

def projected_subgradient(h, n, deadline, alpha=50000, n_iter=2000):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    zk = h.copy()
    tk = 1.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        gk = grad_full(zk, n, alpha, 0.0)
        norm_g = np.linalg.norm(gk)
        if norm_g < 1e-14:
            break
        step = 1.0 / (alpha * 4.0 / n + 1e-8)
        step = min(step, 0.01 / max(norm_g, 1e-10))
        yk = zk - step * gk
        xk = project_feasible(yk, n)
        c = compute_c5(xk, n)
        if c < best_c:
            best_c = c
            best_h = xk.copy()
        tk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        zk = xk + ((tk - 1.0) / tk_new) * (xk - zk)
        zk = np.clip(zk, 0.0, 1.0)
        zk = project_feasible(zk, n)
        tk = tk_new
    return best_h, best_c

def coordinate_descent(h, n, deadline, n_sweeps=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            hi = h[i]
            for delta in [0.001, -0.001, 0.0005, -0.0005, 0.0001, -0.0001]:
                if hi + delta < -1e-12 or hi + delta > 1 + 1e-12:
                    continue
                h[i] = hi + delta
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    h[i] = hi
            if not improved and sweep > 20:
                h[i] = hi
        if not improved:
            break
    return h, best_c

def pair_exchange(h, n, deadline, radius=40):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved:
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(0, n, max(1, n // 500)):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def perturbed_restart(h, n, deadline, rng, n_attempts=3):

def multires_refine(h, n, deadline):
