```
def weighted_grad(h, w, n):
    dx = 2.0/n
    if n <= 1:
       return np.zeros(n)
    w0 = w[n-1]

       P[1:] = w[n:]  # length n-1

    Q = np.zeros(n)
```

```python
import time
import numpy as np
from scipy.optimize import minimize

INTERRUPT_TIME = [0.0]


def interrupt_callback(x):
    if time.time() > INTERRUPT_TIME[0]:
        raise RuntimeError("time-limit")


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h

    lo = -1.0
    hi = 1.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        smid = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if smid > target:
            lo = mid
        else:
            hi = mid

    c = 0.5 * (lo + hi)
    out = np.clip(h - c, 0.0, 1.0)

    for _ in range(8):
        diff = target - float(out.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            i = int(np.argmin(out))
            d = min(diff, 1.0 - out[i])
            out[i] += d
        else:
            i = int(np.argmax(out))
            d = min(-diff, out[i])
            out[i] -= d

    return np.clip(out, 0.0, 1.0)


def compute_c_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(np.max(compute_c_vec(h, n)))


def compute_c_at_indices(x, n, idxs):
    idxs = np.asarray(idxs, dtype=int)
    out = np.empty(len(idxs), dtype=float)
    if len(idxs) == 0:
        return out

    dx = 2.0 / float(n)
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    prefix[1:] = np.cumsum(x)
    total = prefix[n]

    for t, idx in enumerate(idxs):
        k = int(idx) - (n - 1)
        if k == 0:
            out[t] = dx * (total - float(np.dot(x, x)))
        elif k > 0:
            a = k
            m = n - a
            out[t] = dx * (prefix[m] - float(np.dot(x[:m], x[a:n])))
        else:
            a = -k
            m = n - a
            out[t] = dx * ((total - prefix[a]) - float(np.dot(x[a:n], x[:m])))
    return out


def weighted_grad(x, w, n):
    dx = 2.0 / float(n)
    if n <= 1:
        if n == 1:
            return dx * (np.asarray(w, dtype=float)[0] * (1.0 - 2.0 * x))
        return np.zeros(n)

    w = np.asarray(w, dtype=float)
    P = np.zeros(n, dtype=float)
    Neg = np.zeros(n, dtype=float)
    P[1:] = w[n:]
    Neg[1:] = w[:n - 1][::-1]
    s0 = float(w[n - 1])

    tol = 1e-15
    active = np.nonzero((P[:-1] > tol) | (Neg[:-1] > tol))[0]

    if len(active) < max(24, n // 8):
        inner = np.zeros(n, dtype=float)
        if s0 != 0.0:
            inner += s0 * (1.0 - 2.0 * x)
        for a in active:
            p = float(P[a])
            ng = float(Neg[a])
            s = p + ng
            m = n - int(a)
            inner[:m] += p - s * x[a:]
            inner[a:] += ng - s * x[:m]
        return dx * inner

    Q = np.zeros(n, dtype=float)
    Q[1:] = P[1:] + Neg[1:]

    convL = np.convolve(x, Q, mode="full")[:n]
    convR = np.convolve(x[::-1], Q, mode="full")[:n][::-1]

    totalP = float(P[1:].sum())
    cumP = np.cumsum(P)
    cumNeg = np.cumsum(Neg)
    const = totalP - cumP + cumNeg

    inner = s0 * (1.0 - 2.0 * x) + const - (convL + convR)
    return dx * inner


def soft_obj(x, n, alpha, mu):
    c = compute_c_vec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    shm = float(sh.max())
    sh2 = sh - shm
    lse = shm + float(np.log(np.sum(np.exp(sh2))))
    penalty = mu * (float(np.sum(x)) - float(n) / 2.0) ** 2
    return float(m + lse / alpha + penalty)


def soft_grad(x, n, alpha, mu):
    c = compute_c_vec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    shm = float(sh.max())
    sh2 = sh - shm
    w = np.exp(sh2)
    w = w / float(w.sum())
    g = weighted_grad(x, w, n)
    g = g + 2.0 * mu * (float(np.sum(x)) - float(n) / 2.0)
    return g


def active_obj(x, n, act, alpha, mu):
    vals = compute_c_at_indices(x, n, act)
    m = float(vals.max())
    sh = alpha * (vals - m)
    shm = float(sh.max())
    sh2 = sh - shm
    lse = shm + float(np.log(np.sum(np.exp(sh2))))
    penalty = mu * (float(np.sum(x)) - float(n) / 2.0) ** 2
    return float(m + lse / alpha + penalty)


def active_grad(x, n, act, alpha, mu):
    vals = compute_c_at_indices(x, n, act)
    m = float(vals.max())
    sh = alpha * (vals - m)
    shm = float(sh.max())
    sh2 = sh - shm
    subw = np.exp(sh2)
    subw = subw / float(subw.sum())

    wfull = np.zeros(2 * n - 1, dtype=float)
    wfull[np.asarray(act, dtype=int)] = subw
    g = weighted_grad(x, wfull, n)
    g = g + 2.0 * mu * (float(np.sum(x)) - float(n) / 2.0)
    return g


def active_fixed_obj(x, n, idxs, weights, mu):
    vals = compute_c_at_indices(x, n, idxs)
    penalty = mu * (float(np.sum(x)) - float(n) / 2.0) ** 2
    return float(float(np.dot(weights, vals)) + penalty)


def active_fixed_grad(x, n, idxs, weights, mu):
    wfull = np.zeros(2 * n - 1, dtype=float)
    wfull[np.asarray(idxs, dtype=int)] = np.asarray(weights, dtype=float)
    g = weighted_grad(x, wfull, n)
    g = g + 2.0 * mu * (float(np.sum(x)) - float(n) / 2.0)
    return g


def get_active_indices(h, n, top_k, frac):
    c = compute_c_vec(h, n)
    m = float(c.max())
    threshold = m - max(1e-11, frac * max(m, 1e-12))
    idxs = np.where(c >= threshold)[0]
    if len(idxs) > top_k:
        vals = c[idxs]
        idxs = idxs[np.argsort(-vals, kind="stable")[:top_k]]
    if len(idxs) < max(3, top_k // 5):
        idxs = np.argsort(-c, kind="stable")[:top_k]
    return np.asarray(idxs, dtype=int)


def get_active_weights(h, n, top_k, frac, weight_alpha):
    idxs = get_active_indices(h, n, top_k, frac)
    c = compute_c_vec(h, n)
    vals = c[idxs]
    sh = weight_alpha * (vals - float(vals.max()))
    shm = float(sh.max())
    sh2 = sh - shm
    w = np.exp(sh2)
    w = w / float(w.sum())
    return idxs, w


def resize_to(h, n_old, n_new):
    h = np.asarray(h, dtype=float)
    if int(n_old) == int(n_new):
        return h.copy()
    old_x = np.linspace(0.0, 2.0, int(n_old))
    new_x = np.linspace(0.0, 2.0, int(n_new))
    return np.interp(new_x, old_x, h)


def deterministic_candidates(n):
    x = np.linspace(0.0, 2.0, int(n))
    tri = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    blk = (np.abs(x - 1.0) < 0.5).astype(float)
    mix = np.clip(0.5 * tri + 0.5 * blk, 0.0, 1.0)
    cos_like = np.clip(0.5 - 0.35 * np.cos(np.pi * x), 0.0, 1.0)
    return [project_feasible(y, int(n)) for y in (tri, blk, mix, cos_like)]


def configs_for_n(n):
    n = int(n)
    if n <= 600:
        return [
            (50.0, 1e-3, 600),
            (200.0, 1e-3, 500),
            (1000.0, 1e-2, 500),
            (5000.0, 1e-2, 400),
            (20000.0, 0.1, 300),
            (100000.0, 1.0, 200),
        ]
    if n <= 1200:
        return [
            (50.0, 1e-3, 350),
            (200.0, 1e-3, 350),
            (1000.0, 1e-2, 350),
            (5000.0, 1e-2, 300),
            (20000.0, 0.1, 250),
            (100000.0, 1.0, 180),
        ]
    return [
        (100.0, 1e-3, 250),
        (1000.0, 1e-2, 220),
        (10000.0, 0.1, 180),
        (100000.0, 1.0, 120),
    ]


def run_lbfgs(h, n, phase_end, configs):
    h = project_feasible(h, int(n))
    n = int(n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if not configs or (phase_end - time.time()) < 12.0:
        return best_h, best_c

    bounds = [(0.0, 1.0)] * n
    total = len(configs)

    for idx, (alpha, mu, maxiter) in enumerate(configs):
        rem = phase_end - time.time()
        if rem < 12.0:
            break

        left = total - idx
        if left > 1:
            alloc = rem / float(left)
            alloc = min(alloc, 75.0)
            alloc = max(alloc, 8.0)
        else:
            alloc = rem - 4.0

        if alloc <= 6.0:
            break

        end = time.time() + alloc
        INTERRUPT_TIME[0] = end

        res = None
        try:
            res = minimize(
                soft_obj,
                best_h,
                args=(n, alpha, mu),
                jac=soft_grad,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-18,
                    "gtol": 1e-13,
                    "maxfun": 120000,
                    "maxls": 30,
                    "callback": interrupt_callback,
                },
            )
        except Exception:
            res = None

        if res is not None and getattr(res, "x", None) is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    return best_h, best_c


def active_refine(h, n, phase_end):
    h = project_feasible(h, int(n))
    n = int(n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if phase_end - time.time() < 15.0:
        return best_h, best_c

    bounds = [(0.0, 1.0)] * n

    tasks = []
    for top_k, frac in [(100, 0.015), (60, 0.008), (35, 0.004), (20, 0.002), (10, 0.001)]:
        tasks.append(("fixed", int(top_k), float(frac), 1000.0, 0.0, 0.1, 180))
    for alpha in [5000.0, 20000.0, 100000.0, 200000.0]:
        tasks.append(("lse", 40, 0.003, 0.0, alpha, 0.1, 140))

    total = len(tasks)
    for idx, (mode, top_k, frac, w_alpha, opt_alpha, mu, maxiter) in enumerate(tasks):
        rem = phase_end - time.time()
        if rem < 12.0:
            break

        left = total - idx
        if left > 1:
            alloc = rem / float(left)
            alloc = min(alloc, 45.0)
            alloc = max(alloc, 7.0)
        else:
            alloc = rem - 4.0

        if alloc <= 5.0:
            break

        end = time.time() + alloc
        res = None

        if mode == "fixed":
            idxs, wts = get_active_weights(best_h, n, top_k, frac, w_alpha)
            if len(idxs) == 0:
                continue
            INTERRUPT_TIME[0] = end
            try:
                res = minimize(
                    active_fixed_obj,
                    best_h,
                    args=(n, idxs, wts, mu),
                    jac=active_fixed_grad,
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={
                        "maxiter": int(maxiter),
                        "ftol": 1e-18,
                        "gtol": 1e-13,
                        "maxfun": 120000,
                        "maxls": 30,
                        "callback": interrupt_callback,
                    },
                )
            except Exception:
                res = None
        else:
            idxs = get_active_indices(best_h, n, top_k, frac)
            if len(idxs) == 0:
                continue
            INTERRUPT_TIME[0] = end
            try:
                res = minimize(
                    active_obj,
                    best_h,
                    args=(n, idxs, opt_alpha, mu),
                    jac=active_grad,
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={
                        "maxiter": int(maxiter),
                        "ftol": 1e-18,
                        "gtol": 1e-13,
                        "maxfun": 120000,
                        "maxls": 30,
                        "callback": interrupt_callback,
                    },
                )
            except Exception:
                res = None

        if res is not None and getattr(res, "x", None) is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    return best_h, best_c


def line_polish(h, n, phase_end):
    h = project_feasible(h, int(n))
    n = int(n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    for alpha in (2000.0, 20000.0, 200000.0):
        for _ in range(8):
            if time.time() > phase_end - 2.0:
                break
            c = compute_c_vec(best_h, n)
            m = float(c.max())
            sh = alpha * (c - m)
            shm = float(sh.max())
            sh2 = sh - shm
            w = np.exp(sh2)
            w = w / float(w.sum())

            g = weighted_grad(best_h, w, n)
            g = g - float(g.mean())
            norm = float(np.linalg.norm(g))
            if norm < 1e-11:
                break

            d = (g / norm) * 0.035
            improved = False
            for scale in (1.0, 0.5, 0.2, 0.1, 0.05, -1.0, -0.5, -0.2):
                v = project_feasible(best_h + scale * d, n)
                cc = compute_c5(v, n)
                if cc < best_c - 1e-14:
                    best_c = cc
                    best_h = v
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_c


def pair_polish(h, n, phase_end, max_evals=220):
    h = project_feasible(h, int(n))
    n = int(n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    evals = 0
    alpha = 20000.0

    for _ in range(6):
        if time.time() > phase_end - 3.0 or evals >= max_evals:
            break

        c = compute_c_vec(best_h, n)
        m = float(c.max())
        sh = alpha * (c - m)
        shm = float(sh.max())
        sh2 = sh - shm
        w = np.exp(sh2)
        w = w / float(w.sum())

        g = weighted_grad(best_h, w, n)
        g = g - float(g.mean())

        hi = np.argsort(g)[-8:]
        lo = np.argsort(g)[:8]

        pairs = []
        for i in hi:
            for j in lo:
                if i == j:
                    continue
                if g[i] > g[j]:
                    pairs.append((float(g[i] - g[j]), int(i), int(j)))
                else:
                    pairs.append((float(g[j] - g[i]), int(j), int(i)))
        pairs.sort(key=lambda z: z[0], reverse=True)

        improved = False
        for diff, sidx, tidx in pairs:
            if evals >= max_evals or time.time() > phase_end - 3.0:
                break
            if lebest_h[sidx] <= 1e-13 or best_h[tidx] >= 1.0 - 1e-13:
                continue
            for delta in (0.025, 0.01, 0.004):
                d = min(delta, float(best_h[sidx]), 1.0 - float(best_h[tidx]))
                if d <= 1e-13:
                    continue
                best_h[sidx] -= d
                best_h[tidx] += d
                evals += 1
                cc = compute_c5(best_h, n)
                if cc < best_c:
                    best_c = cc
                    improved = True
                    break
                else:
                    best_h[sidx] += d
                    best_h[tidx] -= d
            if improved:
                break

        if not improved:
            break

    return project_feasible(best_h, n), best_c


def try_high_resolution(src_h, src_n, src_c, phase_end):
    best_h = np.asarray(src_h, dtype=float).copy()
    best_n = int(src_n)
    best_c = float(src_c)

    for N in (1500, 2000):
        rem = phase_end - time.time()
        if rem < 100.0 or N <= best_n:
            continue

        hN = project_feasible(resize_to(best_h, best_n, N), N)
        cN = compute_c5(hN, N)

        lb_end = time.time() + min(0.40 * rem, 95.0)
        if lb_end - time.time() > 35.0:
            hN, cN = run_lbfgs(hN, N, lb_end, configs_for_n(N))

        rem2 = phase_end - time.time()
        if rem2 > 70.0:
            act_end = min(phase_end - 15.0, time.time() + min(0.45 * rem2, 70.0))
            hN, cN = active_refine(hN, N, act_end)

        if cN < best_c:
            best_c = cN
            best_h = hN.copy()
            best_n = N

    return best_h, best_c, best_n


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    limit = t0 + max(1.0, float(budget_s) - 5.0)

    have_parent = False
    h0 = None
    n0 = 500
    try:
        h_candidate = np.asarray(initial_h_values, dtype=float).copy()
        if h_candidate.size > 0:
            h0 = h_candidate
            n0 = int(h_candidate.size)
            have_parent = True
    except Exception:
        have_parent = False

    if not have_parent or h0 is None:
        n0 = 500
        cands = deterministic_candidates(n0)
        best_can = None
        best_cc = np.inf
        for cnd in cands:
            vv = compute_c5(cnd, n0)
            if vv < best_cc:
                best_cc = vv
                best_can = cnd
        h0 = best_can if best_can is not None else deterministic_candidates(n0)[0]

    global_best_h = project_feasible(h0, n0)
    global_best_n = int(n0)
    global_best_c = compute_c5(global_best_h, global_best_n)

    sym = project_feasible(0.5 * (global_best_h + global_best_h[::-1]), global_best_n)
    c_sym = compute_c5(sym, global_best_n)
    if c_sym < global_best_c:
        global_best_c = c_sym
        global_best_h = sym

    remaining = limit - time.time()
    if remaining > 750.0:
        main_planned = 1000
    elif remaining > 450.0:
        main_planned = 800
    else:
        main_planned = global_best_n

    if main_planned > global_best_n and remaining > 450.0:
        small_n = min(global_best_n, 500)
        if small_n < 120:
            small_n = 120
        h_s = project_feasible(resize_to(global_best_h, global_best_n, small_n), small_n)
        c_s = compute_c5(h_s, small_n)
        if c_s < global_best_c:
            global_best_c = c_s
            global_best_h = h_s

        rem = limit - time.time()
        if rem > 220.0:
            small_end = min(time.time() + min(85.0, rem - 260.0), limit - 260.0)
            h_s, c_s = run_lbfgs(h_s, small_n, small_end, configs_for_n(small_n)[:4])
            if c_s < global_best_c:
                global_best_c = c_s
                global_best_h = h_s

    remaining = limit - time.time()
    main_n = int(global_best_n)
    h_main = global_best_h.copy()
    c_main = global_best_c

    if main_planned > global_best_n and remaining > 180.0:
        main_n = int(main_planned)
        h_main = project_feasible(resize_to(global_best_h, global_best_n, main_n), main_n)
        c_main = compute_c5(h_main, main_n)
        if c_main < global_best_c:
            global_best_c = c_main
            global_best_h = h_main.copy()
            global_best_n = main_n

        main_budget = min(450.0, max(50.0, remaining - 300.0))
        if main_budget > 30.0:
            h_main, c_main = run_lbfgs(
                h_main,
                main_n,
                time.time() + main_budget,
                configs_for_n(main_n),
            )
            if c_main < global_best_c:
                global_best_c = c_main
                global_best_h = h_main.copy()
                global_best_n = main_n
    elif remaining > 140.0:
        main_n = int(global_best_n)
        h_main = global_best_h.copy()
        main_budget = min(260.0, max(30.0, remaining - 170.0))
        if main_budget > 20.0:
            h_main, c_main = run_lbfgs(
                h_main,
                main_n,
                time.time() + main_budget,
                configs_for_n(main_n),
            )
            if c_main < global_best_c:
                global_best_c = c_main
                global_best_h = h_main.copy()
                global_best_n = main_n

    c_work = compute_c5(h_main, main_n)
    if c_work < global_best_c + 0.005:
        work_h, work_n = h_main, main_n
    else:
        work_h, work_n = global_best_h.copy(), int(global_best_n)

    remaining = limit - time.time()
    if remaining > 140.0:
        active_budget = min(160.0, max(35.0, remaining - 130.0))
        if active_budget > 25.0:
            work_h, c_work = active_refine(work_h, work_n, time.time() + active_budget)
            if c_work < global_best_c:
                global_best_c = c_work
                global_best_h = work_h.copy()
                global_best_n = int(work_n)

    remaining = limit - time.time()
    if remaining > 120.0:
        pol_budget = min(85.0, max(25.0, remaining - 80.0))
        if pol_budget > 18.0:
            work_h, c_work = line_polish(work_h, work_n, time.time() + pol_budget)
            if c_work < global_best_c:
                global_best_c = c_work
                global_best_h = work_h.copy()
                global_best_n = int(work_n)

        remaining = limit - time.time()
        if remaining > 60.0:
            pair_budget = min(45.0, remaining - 25.0)
            if pair_budget > 15.0:
                work_h, c_work = pair_polish(work_h, work_n, time.time() + pair_budget)
                if c_work < global_best_c:
                    global_best_c = c_work
                    global_best_h = work_h.copy()
                    global_best_n = int(work_n)

    remaining = limit - time.time()
    if remaining > 220.0 and global_best_n < 2000:
        hh, cc, nn = try_high_resolution(global_best_h, global_best_n, global_best_c, limit - 8.0)
        if cc < global_best_c:
            global_best_c = cc
            global_best_h = hh
            global_best_n = int(nn)

    global_best_h = project_feasible(global_best_h, global_best_n)
    global_best_c = compute_c5(global_best_h, global_best_n)

    return global_best_h, float(global_best_c), int(global_best_n)
```
