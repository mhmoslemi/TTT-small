    """Project h onto {0<=h<=1, sum(h) = n/2}"""
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine adjustment
    diff = target - proj.sum()
    if abs(diff) < 1e-12:
        return proj
    # Adjust non-bound entries
    mask = (proj > 1e-10) & (proj < 1 - 1e-10)
    if mask.sum() > 0:
        proj[mask] += diff / mask.sum()
        proj = np.clip(proj, 0, 1)

def project_simplex_box(h, n):
    """Project onto {0<=h<=1, sum(h) = n/2} using bisection"""
    target = n / 2.0
    # Bisection on lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0

        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid

def compute_gradient_smooth(h, n, weights):
    """Compute gradient of smooth max approximation.
    weights: array of shape (2n-1,), softmax weights for each lag.
    Returns gradient of shape (n,).
    """
    # weights[t] corresponds to lag tau = t - (n-1)
    # grad[t][j] = (1-h[j+tau]) * [0<=j+tau<n] - h[j-tau] * [0<=j-tau<n]
    grad = np.zeros(n)
    for t in range(len(weights)):
        w = weights[t]
        if w < 1e-15:
            continue
        tau = t - (n - 1)
        # Term 1: (1-h[j+tau]) for 0<=j<n AND 0<=j+tau<n
        if tau >= 0:
            # j ranges from 0 to n-1-tau
            j_range = np.arange(n - tau)
            grad[j_range] += w * (1.0 - h[j_range + tau])
            # Term 2: -h[j-tau] for 0<=j-tau<n AND 0<=j<n
            # j ranges from tau to n-1
            j_range2 = np.arange(tau, n)
            grad[j_range2] += w * (-h[j_range2 - tau])
        else:
            tau_abs = -tau
            # Term 1: (1-h[j+tau]) = (1-h[j-tau_abs]) for 0<=j-tau_abs<n
            # j ranges from tau_abs to n-1
            j_range = np.arange(tau_abs, n)
            grad[j_range] += w * (1.0 - h[j_range - tau_abs])
            # Term 2: -h[j-tau] = -h[j+tau_abs] for 0<=j+tau_abs<n
            # j ranges from 0 to n-1-tau_abs
            j_range2 = np.arange(n - tau_abs)
            grad[j_range2] += w * (-h[j_range2 + tau_abs])
    return grad

                        c = float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                no_improve += 1
            else:
                no_improve = 0
    return h, best_c

def exhaustive_pair_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(n - 1, i, -1):
                    for s in (1.0, -1.0):
                        hi = h[i] + s * d
                        hj = h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)
                            if c < best_c - 1e-15:
                                best_c = c
                                prog = True
                            else:
                                h[i], h[j] = h[i] - s * d, h[j] + s * d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    # Phase 1: Smooth projected gradient (40% budget)
    p1 = t0 + 0.40 * (budget_s - 3)
    h1, c1 = projected_smooth_descent(best_h.copy(), n, p1,
                                       alpha_start=300, alpha_factor=1.15,
                                       alpha_max=5e7, max_iter=5000)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()
    # Phase 2: Perturbed restart (10% budget)
    p2 = t0 + 0.50 * (budget_s - 3)
    if time.time() < p2 - 5:
        h2 = project_simplex_box(best_h + 0.08 * rng.randn(n), n)
        h2, c2 = projected_smooth_descent(h2, n, p2,
                                           alpha_start=5000, alpha_factor=1.25,
                                           alpha_max=5e8, max_iter=2000)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
    # Phase 3: Gradient-guided fine pair refinement (35% budget)
    p3 = t0 + 0.85 * (budget_s - 3)
    if time.time() < p3 - 5:
        h3, c3 = fine_pair_refine(best_h.copy(), n, p3, seed_val=seed)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    # Phase 4: Final smooth polish + pair (15% budget)
    p4 = dl - 2
    if time.time() < p4 - 20:
        h4, c4 = projected_smooth_descent(best_h.copy(), n, p4 - 20,
                                           alpha_start=5e7, alpha_factor=1.4,
                                           alpha_max=5e9, max_iter=1000)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
    if time.time() < p4 - 1:
        h5, c5 = fine_pair_refine(best_h.copy(), n, p4 - 1, seed_val=seed + 7)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
    # Final projection and evaluation
    best_h = project_simplex_box(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
