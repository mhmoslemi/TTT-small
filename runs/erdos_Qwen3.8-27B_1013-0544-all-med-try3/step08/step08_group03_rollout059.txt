    def con_fun(x):
        return float(x.sum() - n / 2.0)

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_hs(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def projected_gradient(h, n, deadline, alpha=50000, lr_init=0.005, max_iters=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = _grad_smax(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        step = lr
        improved = False
        for _ in range(30):
            h_try = project_hs(h - step * g, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            alpha *= 1.3
        if lr < 1e-14:
            break
    return best_h, best_c

def fine_pair_search(h, n, deadline, steps, radius=None):
    best_c = compute_c5(h, n)
    for d in steps:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def triple_search(h, n, deadline, step=0.001, radius=30):
    best_c = compute_c5(h, n)
    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                for j2 in range(j + 1, hi):
                    if j2 == i or j2 == j:
                        continue
                    if h[i] - step >= 0 and h[j] + step * 0.5 <= 1 and h[j2] + step * 0.5 <= 1:
                        h[i] -= step
                        h[j] += step * 0.5
                        h[j2] += step * 0.5
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += step
                            h[j] -= step * 0.5
                            h[j2] -= step * 0.5
                    if h[i] + step <= 1 and h[j] - step * 0.5 >= 0 and h[j2] - step * 0.5 >= 0:
                        h[i] += step
                        h[j] -= step * 0.5
                        h[j2] -= step * 0.5
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= step
                            h[j] += step * 0.5
                            h[j2] += step * 0.5
    return h, best_c

def critical_shift_descent(h, n, deadline, top_k=10, lr=0.002, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vals = compute_c5_vals(best_h, n)
        idx = np.argsort(c_vals)[-top_k:]
        c_crit = c_vals[idx]
        c_max = c_crit.max()
        dx = 2.0 / n
        grad = np.zeros(n)
        for mi in idx:
            k = mi - (n - 1)
            wm = dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += wm * (1.0 - best_h[k:n])
                if k > 0:
                    grad[k:n] += wm * (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += wm * (1.0 - best_h[0:nv])
                if kk > 0:
                    grad[0:nv] += wm * (-best_h[kk:n])
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        grad = grad / g_norm
        step = lr
        improved = False
        for _ in range(20):
            h_try = project_hs(best_h - step * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.7
        if lr < 1e-12:
            break
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h, right=h[-1])
    return project_hs(h_new, n_new)

def binary_snap(h, n, deadline):
    best_c = compute_c5(h, n)
    for _ in range(5):
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] > 0.95:
                    dh = 1.0 - h[i]
                    h[i] = 1.0
                    rest = (h > 1e-8) & (h < 1.0 - 1e-8) & (np.arange(n) != i)
                    nr = rest.sum()
                    if nr > 0:
                        h[rest] -= dh / nr
                        h = np.clip(h, 0.0, 1.0)
                        h = project_hs(h, n)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= 1.0
                            h[rest] += dh / nr
                            h = project_hs(h, n)
                elif h[i] < 0.05:
                    dh = h[i]
                    h[i] = 0.0
                    rest = (h < 1.0 - 1e-8) & (h > 1e-8) & (np.arange(n) != i)
                    nr = rest.sum()
                    if nr > 0:
                        h[rest] += dh / nr
                        h = np.clip(h, 0.0, 1.0)
                        h = project_hs(h, n)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += dh
                            h[rest] -= dh / nr
                            h = project_hs(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_hs(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Quick polish at original resolution
    t_phase1 = min(t0 + 80, deadline - 400)
    if t_phase1 > t0 + 10:
        best_h, best_c = lbfgs_phase(best_h, n_init, t_phase1,
                                     alphas=[10000, 100000, 500000], max_iters=200)

    # Phase 2: Upsample to higher resolutions and optimize
    # Try n=600 as primary
    n_high = 600
    h_high = upsample(best_h, n_init, n_high)
    c_high = compute_c5(h_high, n_high)

    t_phase2 = min(t0 + 400, deadline - 250)
    if t_phase2 > time.time() + 30:
        h_high, c_high = lbfgs_phase(h_high, n_high, t_phase2,
                                     alphas=[10000, 50000, 200000, 1000000, 5000000],
                                     max_iters=200)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high

    # Phase 3: SLSQP refinement at high resolution
    if time.time() < deadline - 150 and len(best_h) == n_high:
        t3 = min(time.time() + 80, deadline - 150)
        h_s, c_s = slsqp_phase(best_h, n_high, t3, alpha=1000000, maxiter=3000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 4: Projected gradient descent
    if time.time() < deadline - 120:
        t4 = min(time.time() + 80, deadline - 120)
        h_pg, c_pg = projected_gradient(best_h, len(best_h), t4,
                                        alpha=200000, lr_init=0.003, max_iters=3000)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg

    # Phase 5: Critical shift focused descent
    if time.time() < deadline - 100:
        t5 = min(time.time() + 60, deadline - 100)
        h_cs, c_cs = critical_shift_descent(best_h, len(best_h), t5,
                                            top_k=8, lr=0.003, max_rounds=100)
        if c_cs < best_c:
            best_c = c_cs
            best_h = h_cs

    # Phase 6: Binary snap
    if time.time() < deadline - 80:
        t6 = min(time.time() + 40, deadline - 80)
        h_bs, c_bs = binary_snap(best_h, len(best_h), t6)
        if c_bs < best_c:
            best_c = c_bs
            best_h = h_bs

    # Phase 7: Fine pair search
    if time.time() < deadline - 40:
        t7 = min(time.time() + 80, deadline - 40)
        best_h, best_c = fine_pair_search(best_h, len(best_h), t7,
                                          [0.01, 0.005, 0.002, 0.001, 0.0005,
                                           0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                          radius=50)

    # Phase 8: Try n=450 and n=500 if time permits
    for n_try in [450, 500, 750]:
        if time.time() > deadline - 60:
            break
        h_interp = upsample(best_h, len(best_h), n_try)
        alloc = min(50, deadline - time.time() - 20)
        if alloc < 20:
            break
        h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5,
                               alphas=[50000, 500000, 2000000], max_iters=150)
        if time.time() < deadline - 15:
            h_t, c_t = critical_shift_descent(h_t, n_try, time.time() + min(20, deadline - time.time() - 10),
                                              top_k=8, lr=0.002, max_rounds=50)
        if time.time() < deadline - 10:
            h_t, c_t = fine_pair_search(h_t, n_try, time.time() + min(10, deadline - time.time() - 5),
                                        [0.001, 0.0005, 0.0002, 0.0001], radius=40)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 9: Final ultra-fine polish
    if time.time() < deadline - 5:
        best_h, best_c = fine_pair_search(best_h, len(best_h), deadline - 2,
                                          [0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002],
                                          radius=30)

    # Final projection and evaluation
    best_h = project_hs(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
