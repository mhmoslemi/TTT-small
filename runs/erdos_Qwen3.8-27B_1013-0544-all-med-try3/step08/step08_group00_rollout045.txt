            no_improve += 1
            rho *= 0.5
            if rho < 1e-11 or no_improve > 30:
                break
    return best_h, best_c

def binary_push(h, n):
    """Push values toward 0 or 1 while maintaining feasibility"""
    h_new = h.copy()
    # Push values > 0.7 toward 1, values < 0.3 toward 0
    # But we need to maintain sum = n/2
    frac_ones = np.sum(h_new > 0.5)
    frac_zeros = np.sum(h_new <= 0.5)
    
    # Try hard binary
    h_bin = np.where(h_new > 0.5, 1.0, 0.0)
    diff = n/2.0 - h_bin.sum()
    if abs(diff) < 1e-6:
        return h_bin
    # Adjust: find values closest to 0.5 and flip them
    if diff > 0:
        # Need more 1s
        zeros = np.where(h_new <= 0.5)[0]
        # Sort by h_new (closest to 0.5 first)
        zeros_sorted = zeros[np.argsort(-h_new[zeros])]
        for idx in zeros_sorted[:int(diff)+1]:
            h_bin[idx] = 1.0
            if h_bin.sum() >= n/2:
                break
    else:
        ones = np.where(h_new > 0.5)[0]
        ones_sorted = ones[np.argsort(h_new[ones])]
        for idx in ones_sorted[:int(-diff)+1]:
            h_bin[idx] = 0.0
            if h_bin.sum() <= n/2:
                break
    return h_bin

def local_search_fine(h, n, deadline, radius=20, steps=None):
    """Very fine local search on the grid"""
    if steps is None:
        steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_c = compute_c5(h, n)
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            # Focus on regions contributing to max
            thresh = max_val - 0.001 * max_val
            active = np.where(c_vec >= thresh)[0]
            
            # Compute which h indices are most "responsible" for active terms
            influence = np.zeros(n)
            for idx in active[:min(len(active), 30)]:
                k = idx - (n-1)
                if k >= 0:
                    nv = n - k
                    influence[0:nv] += 1.0
                    influence[k:n] += 1.0
                else:
                    kk = -k
                    nv = n - kk
                    influence[kk:n] += 1.0
                    influence[0:nv] += 1.0
            
            top_idx = np.argsort(-influence)[:min(radius*2, n)]
            
            for ii in top_idx:
                if time.time() > deadline:
                    return h, best_c
                i = int(ii)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if h[i] + step <= 1.0 + 1e-10 and h[j] - step >= -1e-10:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Try moving mass from i to j
                    if h[i] - step >= -1e-10 and h[j] + step <= 1.0 + 1e-10:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def run_lsmultiple(h, n, deadline, alphas, mus):
    """Run L-BFGS with multiple alpha/mu settings"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    for alpha, mu in zip(alphas, mus):
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(lse_objective, best_h, jac=lse_gradient,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)
    
    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400
    
    h = project_feasible(h_init, n_init)
    c5 = compute_c5(h, n_init)
    best_c = c5
    best_h = h.copy()
    
    # Upsample to n=2000
    n_high = 2000
    h_high = upsample(h, n_init, n_high)
    c_high = compute_c5(h_high, n_high)
    
    if c_high < best_c:
        best_c = c_high
        best_h = h_high.copy()
    
    h_work = h_high.copy()
    
    # Phase 1: L-BFGS warmup (60s)
    t1 = min(t0 + 60, deadline - 700)
    if time.time() < t1 - 3:
        h_work, c1 = run_lsmultiple(h_work, n_high, t1,
            alphas=[500, 2000, 8000, 20000, 50000],
            mus=[10, 100, 1000, 10000, 50000])
        if c1 < best_c:
            best_c = c1
            best_h = h_work.copy()
    
    # Phase 2: Momentum gradient (80s)
    t2 = min(time.time() + 80, deadline - 500)
    if time.time() < t2 - 5:
        h_work, c2 = projected_momentum(h_work, n_high, t2,
            alpha=10000, mu=1000, momentum=0.85, max_iter=8000)
        if c2 < best_c:
            best_c = c2
            best_h = h_work.copy()
    
    # Phase 3: SQP refinement (120s)
    t3 = min(time.time() + 120, deadline - 350)
    if time.time() < t3 - 10:
        h_work, c3 = sqp_refine(h_work, n_high, t3, max_iter=300, rho_init=0.04, top_k=20)
        if c3 < best_c:
            best_c = c3
            best_h = h_work.copy()
    
    # Phase 4: Binary push + local search (80s)
    h_bin = binary_push(best_h, n_high)
    c_bin = compute_c5(h_bin, n_high)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    
    t4 = min(time.time() + 80, deadline - 250)
    if time.time() < t4 - 5:
        h_work, c4 = local_search_fine(best_h, n_high, t4, radius=25)
        if c4 < best_c:
            best_c = c4
            best_h = h_work.copy()
    
    # Phase 5: Another SQP round (100s)
    t5 = min(time.time() + 100, deadline - 150)
    if time.time() < t5 - 10:
        h_work, c5_val = sqp_refine(best_h, n_high, t5, max_iter=200, rho_init=0.02, top_k=15)
        if c5_val < best_c:
            best_c = c5_val
            best_h = h_work
