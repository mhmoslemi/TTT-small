```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
```

Let me be more careful about the coordinate descent. The key insight is:

For coordinate i, the change in c_vals when h[i] goes from old to old+delta is:
c_new[s] = c_old[s] + slope_i[s] * delta

where slope_i[s] = ∂c[s]/∂h[i]

The new max is max_s(c_old[s] + slope_i[s] * delta).

I want to minimize this over delta ∈ [-old, 1-old].

Since each c_old[s] + slope_i[s]*delta is linear in delta, the max is a convex, piecewise-linear function of delta. Its minimum is at a "vertex" where two lines cross, or at a boundary.

```python
def find_exact_min(c_vals, slopes, lo, hi):
    """Find min of max_s(c[s] + slopes[s]*delta) over delta in [lo, hi]."""
    # At delta=0: f(0) = max(c_vals)
    # The function is piecewise linear and convex
    # Find the minimum exactly
    
    # Method: binary search on the derivative
    # f'(delta) = slope of the active line (the one achieving the max)
    # At the minimum, f'(delta) changes from negative to positive (passes through 0)
    
    # Alternative: find all breakpoints and check each
    # Breakpoint between s1 and s2: c1 + s1*d = c2 + s2*d → d = (c2-c1)/(s1-s2)
    
    # Efficient method: since f is convex and piecewise linear,
    # we can do binary search on derivative
    # f(delta) = max_s(c[s] + slopes[s]*delta)
    # Subgradient at delta: {slopes[s] : c[s] + slopes[s]*delta = f(delta)}
    # Minimum where 0 is in subgradient, i.e., some active slope < 0 and some > 0
    
    # Binary search:
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = c_vals + slopes * mid
        mx = vals.max()
        active = vals >= mx - 1e-15
        active_slopes = slopes[active]
        if active_slopes.min() >= 0:
            # Derivative ≥ 0, minimum is to the left
            hi = mid
        elif active_slopes.max() <= 0:
            # Derivative ≤ 0, minimum is to the right
            lo = mid
        else:
            # 0 is in subgradient, we're at the minimum
            break
    
    return (lo + hi) / 2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_sum(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def _slopes_for_i(h, n, i):
    m = 2 * n - 1
    slopes = np.zeros(m)
    for s in range(m):
        r = s - (n - 1)
        sl = 0.0
        if r >= 0:
            if r <= i:
                sl += (1.0 - h[i - r])
            if i <= n - 1 - r:
                sl -= h[i + r]
        else:
            mv = -r
            if i <= n - 1 - mv:
                sl += (1.0 - h[i + mv])
            if mv <= i:
                sl -= h[i - mv]
        slopes[s] = sl
    return slopes

def _find_opt_delta(c_vals, slopes, old):
    lo = -old
    hi = 1.0 - old
    if hi - lo < 1e-15:
        return 0.0
    for _ in range(60):
        if hi - lo < 1e-16:
            break
        mid = 0.5 * (lo + hi)
        vals = c_vals + slopes * mid
        mx = vals.max()
        active = vals >= mx - 1e-14
        if not active.any():
            active = vals >= mx - 1e-8
        asl = slopes[active]
        if asl.size == 0:
            break
        if asl.min() >= -1e-15:
            hi = mid
        elif asl.max() <= 1e-15:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, deadline=None, max_passes=200, seed=None):
    h = project_feasible(h.copy(), n)
    c_vals = compute_c5_sum(h, n)
    m = 2 * n - 1
    dx = 2.0 / n
    best_c = float(np.max(c_vals) * dx)
    rng = np.random.default_rng(seed if seed is not None else 0)
    max_passes_actual = max_passes

    for p in range(max_passes_actual):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)
        changed = 0
        for i in order:
            if deadline is not None and time.time() > deadline:
                return h, best_c
            old = h[i]
            if old < 1e-14 or old > 1.0 - 1e-14:
                continue
            slopes = _slopes_for_i(h, n, i)
            delta = _find_opt_delta(c_vals, slopes, old)
            new_val = old + delta
            if new_val < 0:
                new_val = 0.0
            elif new_val > 1:
                new_val = 1.0
            actual_delta = new_val - old
            if abs(actual_delta) < 1e-15:
                continue
            c_vals = c_vals + slopes * actual_delta
            h[i] = new_val
            new_max = float(np.max(c_vals))
            nc = new_max * dx
            if nc < best_c:
                best_c = nc
                changed += 1

        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - h
                tot = room.sum()
                if tot > 1e-15:
                    h += (diff / tot) * room
                    h = np.clip(h, 0.0, 1.0)
            else:
                tot = h.sum()
                if tot > 1e-15:
                    h += (diff / tot) * h
                    h = np.clip(h, 0.0, 1.0)
            c_vals = compute_c5_sum(h, n)
            best_c = float(np.max(c_vals) * dx)

        if changed == 0:
            break

    h = project_feasible(h, n)
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def _smooth_obj(x, n, alpha, mu, top_idx=None):
    c = compute_c5_sum(x, n)
    if top_idx is not None:
        csub = c[top_idx]
    else:
        csub = c
    mx = csub.max()
    sh = alpha * (csub - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val) / (2.0 / n)

def _smooth_grad(x, n, alpha, mu, top_idx=None):
    c = compute_c5_sum(x, n)
    if top_idx is not None:
        csub = c[top_idx]
    else:
        csub = c
    mx = csub.max()
    sh = alpha * (csub - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    if top_idx is None:
        idxs = np.arange(2 * n - 1)
    else:
        idxs = top_idx
    for mm in idxs:
        wi = w[mm - (top_idx[0] if top_idx is not None else 0)] if top_idx is not None else w[mm]
        if abs(wi) < 1e-16:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, configs=None):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 50.0), (2000, 200.0), (10000, 1000.0),
                   (50000, 5000.0), (200000, 50000.0), (1000000, 500000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_smooth_obj, h, jac=_smooth_grad,
                          args=(n, alpha, mu, None), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def lbfgs_top(h, n, deadline, top_k=5):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for _ in range(8):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_sum(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])[:top_k]]
        if len(active) == 0:
            break
        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(_smooth_obj, h, jac=_smooth_grad,
                              args=(n, alpha, 1.0, active), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': 20000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def basin_hopping(best_h, n, deadline, seed=42):
    rng = np.random.default_rng(seed)
    best_h = best_h.copy()
    best_c = compute_c5(best_h, n)

    count = 0
    while time.time() < deadline:

        h_p = best_h + rng.normal(0, sigma, n)

        if count % 10 == 0:
```
