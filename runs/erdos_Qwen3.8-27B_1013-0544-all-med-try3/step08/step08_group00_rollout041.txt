```python
def mps_iteration(s, n_iter, alpha):
    n = len(s)
    N = 2*n  # oversampling
    for it in range(n_iter):
        S = np.fft.fft(s, N)
        # Shape spectrum: reduce peaks
        mag = np.abs(S)
        phase = np.angle(S)
        # New magnitude: flatten peaks
        # Target: flat spectrum (all equal magnitude except DC)
        target_mag = np.ones(N)
        target_mag[0] = 0  # DC (balanced sequence)
        # Use a soft version: new_mag = target + alpha * old_mag
        new_mag = (target_mag + alpha * mag) / (1 + alpha)
        # Or: new_mag = mag^beta for beta < 1 to reduce peaks
        new_mag = mag ** 0.5  # square root compresses peaks
        S_new = new_mag * np.exp(1j * phase)
        s_new = np.fft.ifft(S_new).real[:n]
        # Binarize
        s_new = np.sign(s_new)
        # Fix balance
        n_pos = (s_new > 0).sum()
        if n_pos != n//2:
            # Flip the signs of the least confident entries
            magnitudes = np.abs(np.fft.ifft(S_new).real[:n])
            if n_pos > n//2:
                pos_indices = np.where(s_new > 0)[0]
                flip_indices = pos_indices[np.argsort(magnitudes[pos_indices])[:n_pos - n//2]]
                s_new[flip_indices] = -1
            else:
                neg_indices = np.where(s_new < 0)[0]
                flip_indices = neg_indices[np.argsort(magnitudes[neg_indices])[:n//2 - n_pos]]
                s_new[flip_indices] = 1
        s = s_new
    return s
```

```python
    c = np.correlate(h, 1-h, mode='full')  # c[m] for m=0..2n-2

    zeros = np.where(h == 0)[0]

        # Pick a random 1-position and a random 0-position
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        # Update correlation in O(n)
        old_c = c.copy()
        # ... compute new c ...

            c = new_c
            current_max = new_max
```

Where:

    new contribution = 0*(1-h[i+m]) = 0
    Δ = -(1-h[i+m]) = h[i+m]-1 (if 0≤i+m<n)
  - k=j: old contribution = 0*(1-h[j+m]) = 0
    new contribution = 1*(1-h[j+m]) = 1-h[j+m]

  - k+m=i, i.e., k=i-m: old contribution = h[i-m]*(1-h[i]) = h[i-m]*0 = 0

def swap_corr_update(c, h, i, j):
    """
    Compute new correlation after swapping h[i](1→0) and h[j](0→1).
    c is the full correlation array (length 2n-1).
    Returns new_c array.
    """
    n = len(h)
    new_c = c.copy()
    # For each shift m (0 to 2n-2), shift = m - (n-1)
    # delta_c[m] = f(i, j, shift)
    # We can vectorize this!
    shifts = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    # idx1 = i + shift, valid when 0 <= idx1 < n
    idx1 = i + shifts
    valid1 = (idx1 >= 0) & (idx1 < n)
    
    # idx2 = j + shift, valid when 0 <= idx2 < n
    idx2 = j + shifts
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    # idx3 = i - shift, valid when 0 <= idx3 < n
    idx3 = i - shifts
    valid3 = (idx3 >= 0) & (idx3 < n)
    
    # idx4 = j - shift, valid when 0 <= idx4 < n
    idx4 = j - shifts
    valid4 = (idx4 >= 0) & (idx4 < n)
    
    delta = np.zeros(2*n-1)

    delta[valid2] += 1.0 - h[idx2[valid2]]
    delta[valid3] += h[idx3[valid3]]
    delta[valid4] -= h[idx4[valid4]]

        if n_pos > n // 2:

        else:
