def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_solve(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, rho)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, top_k=30, rho_init=0.05, max_iter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:top_k]]
        delta, t_opt = sqp_lp_solve(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.06:
                rho = min(rho * 1.2, 0.08)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if no_improve > 30:
                break
    return best_h, best_c

def lbfgs_phase(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(150, 50.0), (500, 200.0), (1500, 2000.0), (5000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fourier_kick(h, n, kick_idx, seed_val):
    rng = np.random.RandomState(seed_val)
    H = fft(h)
    k_freqs = min(n // 4, 30)
    num_perturb = 3 + kick_idx
    idxs = rng.choice(range(1, k_freqs), size=min(num_perturb, k_freqs - 1), replace=False)
    for idx in idxs:
        amp = rng.uniform(0.005, 0.02) * (1.0 / (1.0 + idx * 0.5))
        phase = rng.uniform(0, 2 * np.pi)
        H[idx] += amp * np.exp(1j * phase)
        H[n - idx] += amp * np.exp(-1j * phase)
    h_new = np.real(ifft(H))
    h_new = project_feasible(h_new, n)
    return h_new

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    h_new = project_feasible(h_new, n_new)
    return h_new

def final_gradient_polish(h, n, deadline, max_iter=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in step_sizes:
        if time.time() > deadline - 5:
            break
        for it in range(max_iter):
            if time.time() > deadline - 5:
                break
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-12, 1e-5 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for idx in active:
                grad += shift_grad_single(h, n, idx)
            grad /= len(active)
            # Project gradient: remove sum component
            grad -= grad.mean()
            # Find steepest descent direction in feasible set
            # Move in direction -grad
            direction = -grad
            # Clip to stay in [0,1]
            max_up = np.where(direction > 0, direction, 0).max()
            min_down = np.where(direction < 0, direction, 0).min()
            if max_up > 0:
                max_up_step = min(step, (1.0 - h.max()) / max_up) if max_up > 1e-15 else step
            else:
                max_up_step = step
            if min_down < 0:
                min_down_step = min(step, h.min() / (-min_down)) if min_down < -1e-15 else step
            else:
                min_down_step = step
            t = min(max_up_step, min_down_step)
            if t < 1e-15:
                break
            h_try = h + t * direction
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-16:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
            else:
                break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    def obj_fn(x, n_val, alpha_val):
        c = compute_c5_vec(x, n_val)
        m = c.max()
        sh = alpha_val * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha_val)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha_val)
    def sq_fn(x, n_val):
        return float(x.sum() - n_val / 2.0)
    def sq_jac_fn(x, n_val):
        return np.ones(n_val)
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS warmup
    t1 = min(t0 + 40, deadline - 600)
    if time.time() < t1:
        h, c = lbfgs_phase(h, n, t1,
            configs=[(200, 50.0), (800, 500.0), (2500, 5000.0), (8000, 20000.0)])
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: SQP main optimization
    t2 = min(time.time() + 100, deadline - 500)
    if time.time() < t2:
        h2, c2 = sqp_optimize(best_h, n, t2, top_k=30, rho_init=0.05, max_iter=400)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: Fourier basin hopping (3 kicks)
    t3_end = min(time.time() + 90, deadline - 250)
    for kick_i in range(3):
        if time.time() > t3_end:
            break
        h_k = fourier_kick(best_h, n, kick_i + 1, seed + 100 * kick_i)
        t_kick_end = min(time.time() + 28, t3_end)
        if time.time() < t_kick_end:
            h_k, c_k = sqp_optimize(h_k, n, t_kick_end, top_k=30, rho_init=0.03, max_iter=200)
            if c_k < best_c:
                best_c = c_k
                best_h = h_k.copy()

    # Phase 4: Second SQP with larger active set
    t4 = min(time.time() + 60, deadline - 180)
    if time.time() < t4:
        h4, c4 = sqp_optimize(best_h, n, t4, top_k=50, rho_init=0.02, max_iter=300)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()

    # Phase 5: Try n=600 resolution
    if time.time() < deadline - 120:
        n_alt = 600
        h_alt = interpolate_to(best_h, n, n_alt)
        t5 = min(time.time() + 55, deadline - 60)
        if time.time() < t5:
            h_alt, c_alt = sqp_optimize(h_alt, n_alt, t5, top_k=30, rho_init=0.03, max_iter=200)
            if c_alt < best_c:
                best_c = c_alt
                best_h = h_alt.copy()
                n = n_alt

    # Phase 6: Try n=400 resolution
    if time.time() < deadline - 80:
        n_alt2 = 400
        h_alt2 = interpolate_to(best_h, len(best_h), n_alt2)
        t6 = min(time.time() + 35, deadline - 40)
        if time.time() < t6:
            h_alt2, c_alt2 = sqp_optimize(h_alt2, n_alt2, t6, top_k=25, rho_init=0.03, max_iter=150)
            if c_alt2 < best_c:
                best_c = c_alt2
                best_h = h_alt2.copy()
                n = n_alt2

    # Phase 7: Gradient polish
    t7 = min(time.time() + 50, deadline - 20)
    if time.time() < t7:
        h7, c7 = final_gradient_polish(best_h, len(best_h), t7)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()

    # Phase 8: Final SLSQP micro-polish
    if time.time() < deadline - 3:
        n_fin = len(best_h)
        h8, c8 = slqp_polish(best_h, n_fin, deadline, alpha=2000000, maxiter=1000)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
