    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx  # length 2n-1
    
    # Softmax weights
    amax = c.max()
    exp_c = np.exp(alpha * (c - amax))
    w = exp_c / exp_c.sum()
    
    # Gradient: for each k, grad_k = (1-shift(h,lag_k)) - shift(h,-lag_k) (with bounds)
    # Total grad = dx * sum_k w_k * grad_k

        if w[k] < 1e-15:

            grad[:n-lag] += w[k] * (1.0 - h[lag:])  # (1-h[j+lag]) for j=0..n-1-lag

                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp_with_grad(best_h, n, a, maxiter=3000, ftol=1e-14)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    if n <= 250 and time.time() < deadline - 10:
        best_h, best_c = pair_search(best_h, n, deadline - 10,
                                     [0.02, 0.01, 0.005, 0.002, 0.001], max_passes=80)
    elif time.time() < deadline - 10:
        best_h, best_c = near_pair_search(best_h, n, deadline - 10,
                                          [0.02, 0.01, 0.005, 0.002], radius=40, max_passes=50)

    if time.time() < deadline - 5:
        ht = slsqp_with_grad(best_h, n, 1000000, maxiter=4000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    if n <= 250 and time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, n, deadline - 1,
                                     [0.005, 0.002, 0.001, 0.0005], max_passes=60)
    elif time.time() < deadline - 2:
        best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                          [0.005, 0.002, 0.001], radius=30, max_passes=40)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at parent resolution
    alloc1 = min(300, (deadline - t0) * 0.3)
    if alloc1 > 60:
        h_opt, c_opt = optimize_at_resolution(best_h, n_init, t0 + alloc1, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Try different resolutions
    n_candidates = [300, 400, 150, 100]
    if n_init != 200:
        n_candidates = [n_init, int(n_init * 1.5), int(n_init * 0.7), int(n_init * 2)]
        n_candidates = [x for x in n_candidates if 50 <= x <= 600 and x != n_init]

    for n_try in n_candidates:
        if time.time() > deadline - 60:
            break
        remaining = deadline - time.time()
        alloc = min(120, remaining * 0.18)
        if alloc < 40:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Random restarts at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 40
        n_restarts = max(4, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.03 + 0.15 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 40) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Try structured initializations
    if time.time() < deadline - 60:
        # Try step-function-like initializations
        structures = []
        for half_frac in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            ns = max(5, int(n_best * half_frac))
            h_struct = np.zeros(n_best)
            h_struct[:ns] = 1.0
            h_struct = project_feasible(h_struct, n_best)
            structures.append(h_struct)

        for h_struct in structures:
            if time.time() > deadline - 40:
                break
            alloc = min(35, (deadline - time.time() - 35) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_resolution(h_struct, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final fine-tuning with pair search
    if time.time() < deadline - 3:
        n_f = len(best_h)
        if n_f <= 250:
            best_h, best_c = pair_search(best_h, n_f, deadline - 1,
                                         [0.002, 0.001, 0.0005, 0.0002, 0.0001], max_passes=100)
        else:
            best_h, best_c = near_pair_search(best_h, n_f, deadline - 1,
                                              [0.002, 0.001, 0.0005, 0.0002], radius=25, max_passes=60)
        best_h = project_feasible(best_h, n_f)
        best_c = compute_c5(best_h, n_f)

    # Final SLSQP polish
    if time.time() < deadline - 2:
        n_f = len(best_h)
        ht = slsqp_with_grad(best_h, n_f, 5000000, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n_f)
        if c < best_c:
            best_c = c
            best_h = ht
        best_h = project_feasible(best_h, n_f)
        best_c = compute_c5(best_h, n_f)

    return best_h, best_c, len(best_h)
