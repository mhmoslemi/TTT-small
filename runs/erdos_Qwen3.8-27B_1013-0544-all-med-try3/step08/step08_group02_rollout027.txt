import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()
    sh = alpha * (c - m)

def make_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    n_ones = n // 2
    idx = np.argsort(-h)[:n_ones]
    h_b = np.zeros(n)

def fast_correlate_max(h, n):
    """Compute max of correlate(h, 1-h, 'full') * dx without computing all values."""
    # Actually, let's just compute all values since numpy is fast
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n), c * 2.0 / n

def swap_delta(c_prev, h, n, a, b):
    """
    Given current correlation c (unnormalized, before *dx), and a swap h[a]: 1->0, h[b]: 0->1,
    compute the change in correlation.
    """
    # Delta at shift s: -(1-h[a+s]) + h[a-s] + (1-h[b+s]) - h[b-s]
    # where out-of-range terms are 0
    # s ranges from -(n-1) to n-1, stored at index s + n - 1 in c
    delta = np.zeros(2*n-1)
    
    # Term 1: -(1-h[a+s]) for s where 0 <= a+s < n, i.e., -a <= s < n-a
    s_min = max(-a, -(n-1))
    s_max = min(n-1-a, n-1)
    if s_min <= s_max:
        p = a + np.arange(s_min, s_max+1)
        delta[np.arange(s_min, s_max+1) + n - 1] += -(1.0 - h[p])
    
    # Term 2: +h[a-s] for s where 0 <= a-s < n, i.e., a-n+1 < s <= a
    s_min = max(a-n+1, -(n-1))
    s_max = min(a, n-1)
    if s_min <= s_max:
        p = a - np.arange(s_min, s_max+1)
        delta[np.arange(s_min, s_max+1) + n - 1] += h[p]
    
    # Term 3: +(1-h[b+s]) for s where 0 <= b+s < n
    s_min = max(-b, -(n-1))
    s_max = min(n-1-b, n-1)
    if s_min <= s_max:
        p = b + np.arange(s_min, s_max+1)
        delta[np.arange(s_min, s_max+1) + n - 1] += (1.0 - h[p])
    
    # Term 4: -h[b-s] for s where 0 <= b-s < n
    s_min = max(b-n+1, -(n-1))
    s_max = min(b, n-1)
    if s_min <= s_max:
        p = b - np.arange(s_min, s_max+1)
        delta[np.arange(s_min, s_max+1) + n - 1] += -h[p]
    
    return delta

def binary_sa(h, n, deadline, t_start=1.0, t_end=0.001, max_iter=500000):

    h_bin = make_binary(h, n)
    best_h = h_bin.copy()
    best_c, _ = fast_correlate_max(h_bin, n)
    cur_c, c_vec = fast_correlate_max(h_bin, n)

    t = t_start
    cooling = (t_end / t_start) ** (1.0 / max(1, max_iter))

        if time.time() > deadline:
            break
        t *= cooling
        
        # Random swap

        h_bin[j] = 1.0
        
        new_c, _ = fast_correlate_max(h_bin, n)
        delta = new_c - cur_c
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(t, 1e-10)):

        else:
            # Revert
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            # Sets unchanged

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(-h)[:n_ones]
    h_b = np.zeros(n)
    h_b[idx] = 1.0
    return h_b

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def greedy_binary_swap(h, n, deadline, max_passes=200):
    """Greedy swap optimization on binary h. Returns (h_bin, c5)."""
    h_b = make_binary(h, n)
    dx = 2.0 / n
    c_raw = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    cur_val = float(c_raw.max() * dx)
    best_h = h_b.copy()
    best_val = cur_val

    ones = np.where(h_b == 1.0)[0].copy()
    zeros = np.where(h_b == 0.0)[0].copy()

    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for ai in range(len(ones)):
            if time.time() > deadline:
                return best_h, best_val
            i = ones[ai]
            for aj in range(len(zeros)):
                j = zeros[aj]
                # Try swap: h[i] 1->0, h[j] 0->1
                h_b[i] = 0.0
                h_b[j] = 1.0
                c_raw = np.correlate(h_b, 1.0 - h_b, mode='full')
                new_val = float(c_raw.max() * dx)
                if new_val < cur_val - 1e-15:
                    cur_val = new_val
                    improved = True
                    if new_val < best_val:
                        best_val = new_val
                        best_h = h_b.copy()
                    # Update sets
                    ones = np.delete(ones, ai)
                    zeros = np.delete(zeros, aj)
                    ai -= 1
                    aj -= 1
                    if ai >= 0:
                        ones = np.insert(ones, ai, j)
                    else:
                        ones = np.append(ones, j)
                    if aj >= 0:
                        zeros = np.insert(zeros, aj, i)
                    else:
                        zeros = np.append(zeros, i)
                else:
                    h_b[i] = 1.0
                    h_b[j] = 0.0
        if not improved:
            break
    return best_h, best_val

def sa_binary(h, n, deadline, t_start=2.0, t_end=0.05, max_iter=300000):
    """Simulated annealing on binary configurations."""
    h_b = make_binary(h, n)
    dx = 2.0 / n
    c_raw = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    cur_val = float(c_raw.max() * dx)
    best_h = h_b.copy()
    best_val = cur_val

    rng = np.random.default_rng(123)
    ones = np.where(h_b == 1.0)[0].copy()
    zeros = np.where(h_b == 0.0)[0].copy()

    t = t_start
    cool = (t_end / t_start) ** (1.0 / max(1, max_iter))

    for it in range(max_iter):
        if time.time() > deadline:
            break
        t *= cool
        if t < 1e-12:
            break

        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        h_b[i] = 0.0
        h_b[j] = 1.0
        c_raw = np.correlate(h_b, 1.0 - h_b, mode='full')
        new_val = float(c_raw.max() * dx)
        delta = new_val - cur_val

        accept = delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t))
        if accept:
            cur_val = new_val
            # Update sets
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_val < best_val:
                best_val = new_val
                best_h = h_b.copy()
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0

    return best_h, best_val

def lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxiter_per=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 4))

        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': maxit * 8,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_expand = 0.002

    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)

        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.97
        h = h_new

    return best_h, best_c
