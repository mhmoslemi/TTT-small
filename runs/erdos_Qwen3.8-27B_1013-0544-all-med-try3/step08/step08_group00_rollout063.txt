def softmax_gradient(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    # Softmax weights
    sh = alpha * (c - m)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(2*n-1)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2*n-1):

            continue

        if k >= 0:

            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            kk = -k
            nv = n - kk

            grad[0:nv] += wm * (-h[kk:n])

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_smooth(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    lse = smax + np.log(np.sum(np.exp(sh - smax)))
    return float(m + lse / alpha + mu * (h.sum() - n/2.0)**2)

def soft_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    ws = w.sum()
    if ws > 1e-30:
        w /= ws
    else:
        w = np.zeros(2*n-1)
        w[np.argmax(c)] = 1.0
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2*n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    grad += 2.0 * mu * (h.sum() - n/2.0)
    return grad

def lbfgs_opt(h, n, deadline, alphas=None, mu=500.0):
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(soft_smooth, h, jac=soft_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 8000, 'ftol': 1e-19,
                                  'gtol': 1e-16, 'maxfun': 80000, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def sqp_step(h, n, c_vec, active, rho):
    n_act = len(active)
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t
    
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    dx = 2.0 / n
    for idx in active:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] = (1.0 - h[k:]) * dx
            g[k:] += (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:] = (1.0 - h[:nv]) * dx
            g[:nv] += (-h[kk:]) * dx
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds_var = [(None, None)] * n + [(0.0, 5.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_var, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def sqp_opt(h, n, deadline, max_iter=500, rho_init=0.03, active_frac=0.005):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline - 1:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, active_frac * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        # Sort by value, keep top 40
        if len(active) > 40:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:40]]
        delta = sqp_step(h, n, c_vec, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            if rho < 0.06:
                rho = min(rho * 1.2, 0.08)
        else:
            stall += 1
            rho *= 0.5
            if rho < 1e-11 or stall > 30:
                break
    return best_h, best_c

def local_search(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            # Compute aggregated gradient magnitude
            dx = 2.0 / n
            grad_mag = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                g = np.zeros(n)
                if k >= 0:
                    nv = n - k
                    g[:nv] = (1.0 - h[k:]) * dx
                    g[k:] += (-h[:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:] = (1.0 - h[:nv]) * dx
                    g[:nv] += (-h[kk:]) * dx
                grad_mag += np.abs(g)
            if len(active) > 0:
                grad_mag /= len(active)
            top = np.argsort(-grad_mag)[:min(2*radius, n)]
            for i_val in top:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try +step at i, -step at j
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Try -step at i, +step at j
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def binary_snap_attempt(h, n):
    """Try snapping values to 0 or 1 and see if it improves."""
    best_c = compute_c5(h, n)
    h_snapped = np.where(h > 0.5, 1.0, 0.0)
    h_snapped = project_feasible(h_snapped, n)
    c_snapped = compute_c5(h_snapped, n)
    if c_snapped < best_c:
        return h_snapped, c_snapped
    # Try soft snap: push toward 0 or 1 by 0.1
    h_pushed = np.clip(h + np.sign(h - 0.5) * 0.05, 0, 1)
    h_pushed = project_feasible(h_pushed, n)
    c_pushed = compute_c5(h_pushed, n)
    if c_pushed < best_c:
        return h_pushed, c_pushed
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000):
    def obj(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return float(m + (sm + np.log(np.sum(np.exp(sh - sm)))) / alpha)
    def confun(x, n):
        return float(x.sum() - n / 2.0)
    def conjac(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': confun, 'jac': conjac, 'args': (n,)}]
    try:
        res = minimize(obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-19})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

Actually, let me try something different: **Alternating minimization with projection**.

The idea: alternate between:
1. For each coordinate i (in random order), find the best value of h[i] (given all others fixed and the sum constraint)
2. Project back to feasible set

For step 1, given h[j] for j≠i fixed, and the sum constraint ∑h[j] = n/2, we have h[i] = n/2 - ∑_{j≠i} h[j]. So h[i] is DETERMINED. We can't change it independently.

Unless we do PAIRED updates: change h[i] by +δ and h[j] by -δ.

For a pair (i,j), the change in correlation at shift k is:
Δc_k = δ * (∂c_k/∂h[i] - ∂c_k/∂h[j])

We want to minimize max_k (c_k + Δc_k) over δ. This is a 1D problem:
minimize max_k (a_k * δ + c_k) where a_k = ∂c_k/∂h[i] - ∂c_k/∂h[j]

The optimal δ* is found by the intersection of the two "most constraining" lines.

For the top active shifts, this is a small LP:
minimize t
subject to: a_k * δ + c_k ≤ t for k in active
           max(0, h[i]) - 1 ≤ δ ≤ min(1-h[i], h[j])  (or reversed)
           t ≥ 0

This is a 2-variable LP with n_act constraints. Very fast to solve.

    n_act = len(active)
    # Build LP: min t s.t. (a_k)*d + c_k - t <= 0 for each k; 0 <= d <= dmax
    # Variables: [d, t]
    # Objective: minimize t → c = [0, 1]
    c_obj = np.array([0.0, 1.0])
    # Constraints: a_k * d - t <= -c_k
    A_ub = np.zeros((n_act + 2, 2))
    b_ub = np.zeros(n_act + 2)
    for idx, mm in enumerate(active):
        k = mm - (n-1)
        # Compute grad at i and j for this shift
        gi = 0.0
        gj = 0.0
        if k >= 0:
            if i < n - k: gi += (1.0 - h[i+k]) * dx
            if i >= k: gi += (-h[i-k]) * dx
            if j < n - k: gj += (1.0 - h[j+k]) * dx
            if j >= k: gj += (-h[j-k]) * dx
        else:
            kk = -k
            if i < n - kk: gi += (1.0 - h[i+kk]) * dx
            if i >= kk: gi += (-h[i-kk]) * dx
            if j < n - kk: gj += (1.0 - h[j+kk]) * dx
            if j >= kk: gj += (-h[j-kk]) * dx
        a_k = sign * (gi - gj)
        A_ub[idx, 0] = a_k
        A_ub[idx, 1] = -1.0
        b_ub[idx] = -c_vec[mm]
    # 0 <= d <= dmax
    dmax = 0.1 * sign  # cap the step
    if sign > 0:
        dmax = min(1.0 - h[i], h[j]) * 0.5
    else:
        dmax = min(h[i], 1.0 - h[j]) * 0.5
    A_ub[-2, 0] = -1.0
    b_ub[-2] = 0.0
    A_ub[-1, 0] = 1.0
    b_ub[-1] = dmax
    bounds_var = [(0, None), (None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_var, method='highs')
        if res.success and res.x[0] > 1e-12:
            d = res.x[0]
            h[i] += sign * d
            h[j] -= sign * d
            return True
    except Exception:
        pass
    return False

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad


def soft_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    lse = sm + np.log(np.sum(np.exp(sh - sm)))
    return float(m + lse / alpha + mu * (x.sum() - n / 2.0) ** 2)


def soft_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    ws = w.sum()
    if ws > 1e-30:
        w /= ws
    else:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def lbfgs_opt(h, n, deadline, alphas=None, mu=1000.0):
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 500000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(soft_obj, h, jac=soft_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-20,
                                   'gtol': 1e-17, 'maxfun': 100000, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c


def sqp_step(h, n, c_vec, active, rho):
    n_act = len(active)
    if n_act == 0:
        return None
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    dx = 2.0 / n
    for idx in active:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] = (1.0 - h[k:]) * dx
            g[k:] += (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:] = (1.0 - h[:nv]) * dx
            g[:nv] += (-h[kk:]) * dx
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds_var = [(None, None)] * n + [(0.0, 5.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            return res.x[:n]
    except Exception:
        pass
    return None

    rho = rho_init

def grad_mag_lookup(h, n, active, dx):
    gmag = np.zeros(n)
    for idx in active:
        mm = int(idx)
        k = mm - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] = (1.0 - h[k:]) * dx
            g[k:] += (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:] = (1.0 - h[:nv]) * dx
            g[:nv] += (-h[kk:]) * dx
        gmag += np.abs(g)
    if len(active) > 0:
        gmag /= len(active)
    return gmag


def local_search(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            gmag = grad_mag_lookup(h, n, active, 2.0 / n)
            top = np.argsort(-gmag)[:min(2 * radius, n)]
            for i_val in top:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)

def slqp_polish(h, n, deadline, alpha=2000000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return float(m + (sm + np.log(np.sum(np.exp(sh - sm)))) / alpha)

    def confun(x, n):
        return float(x.sum() - n / 2.0)

    def conjac(x, n):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': confun, 'jac': conjac, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 5000, 'ftol': 1e-19})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)


def binary_snap_and_restore(h, n):
    best_c = compute_c5(h, n)
    h_b = np.where(h > 0.5, 1.0, 0.0)
    h_b = project_feasible(h_b, n)
    c_b = compute_c5(h_b, n)
    if c_b < best_c:
        return h_b, c_b
    for snap_frac in [0.7, 0.6, 0.8, 0.4]:
        h_s = np.where(h > snap_frac, 1.0, np.where(h < 1.0 - snap_frac, 0.0, h))
        h_s = project_feasible(h_s, n)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            best_c = c_s
            h = h_s
    return h, best_c
