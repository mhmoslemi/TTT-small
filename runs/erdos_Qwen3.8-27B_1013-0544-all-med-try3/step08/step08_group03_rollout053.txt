import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.exp(sh - mx).sum())) / alpha)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    sh = alpha * c
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wi = w[m_idx]
        if wi < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha)

    return float(x.sum() - n / 2.0)

def slsqp_con_jac(x, n):
    return np.ones(n)

        res = minimize(penalty_obj, h, jac=penalty_grad,

                               'maxls': 100})

        return h.copy(), compute_c5(h, n)

        best_h, best_c = full_pair_search(best_h, nf, deadline - 3,
                                           [0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
                                           radius=80)

    # Final projection and evaluation
    nf = len(best_h)
    best_h = project_feasible(best_h, nf)
    best_c = compute_c5(best_h, nf)
    return best_h, best_c, nf
