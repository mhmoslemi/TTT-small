    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sm = alpha * (c - m)
    weights = np.exp(sm - sm.max())
    weights /= weights.sum()
    # weights[k] = p_k = exp(alpha * c[k]) / sum exp(alpha * c[m])
    
    # For each shift k, compute ∂c[k]/∂h[i]
    # c[k] = sum_{i: j=i+k-(n-1) in [0,n-1]} h[i]*(1-h[j])

        # Second term: -h[j2] where j2 = i-k+(n-1), i.e., i is the j in some pair

        i_min2 = max(0, k-(n-1))
        i_max2 = min(n-1, k)
        for i in range(i_min2, i_max2+1):
            j2 = i - k + (n-1)
            grad[i] -= w * x[j2]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_eval(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    return m + (shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())) / alpha

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_shifted = shifted.max()
    weights = np.exp(shifted - max_shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        # First term: (1 - x[j]) where j = i + k - (n-1)
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, n - 1 + k)
        if i_hi >= i_lo:
            j_lo = i_lo + k - (n - 1)
            j_hi = i_hi + k - (n - 1)
            grad[i_lo:i_hi + 1] += (1.0 - x[j_lo:j_hi + 1]) * w
        # Second term: -x[jp] where jp = i - k + (n-1)
        i_lo2 = max(0, k - (n - 1))
        i_hi2 = min(n - 1, k)
        if i_hi2 >= i_lo2:
            jp_lo = i_lo2 - k + (n - 1)
            jp_hi = i_hi2 - k + (n - 1)
            grad[i_lo2:i_hi2 + 1] -= x[jp_lo:jp_hi + 1] * w
    return grad

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        if time.time() > deadline:
            return float('inf')
        return smooth_max_eval(x, n, alpha)
    
    def jac(x):
        if time.time() > deadline:
            return np.zeros(n)
        return smooth_max_grad(x, n, alpha)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_exact(h, n, deadline, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.05
    stuck_count = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = np.correlate(h, 1 - h, mode='full') * dx
        m = c_all.max()
        # Find all argmax (there might be ties)
        ks = np.where(np.abs(c_all - m) < 1e-14)[0]
        
        # Compute exact subgradient
        grad = np.zeros(n)
        for k in ks:
            w = 1.0 / len(ks)
            # First term
            i_lo = max(0, n - 1 - k)
            i_hi = min(n - 1, n - 1 + k)
            if i_hi >= i_lo:
                j_lo = i_lo + k - (n - 1)
                j_hi = i_hi + k - (n - 1)
                grad[i_lo:i_hi + 1] += w * (1.0 - h[j_lo:j_hi + 1])
            # Second term
            i_lo2 = max(0, k - (n - 1))
            i_hi2 = min(n - 1, k)
            if i_hi2 >= i_lo2:
                jp_lo = i_lo2 - k + (n - 1)
                jp_hi = i_hi2 - k + (n - 1)
                grad[i_lo2:i_hi2 + 1] -= w * h[jp_lo:jp_hi + 1]
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        # Projected gradient step with backtracking
        improved = False
        for f in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                h = ht
                improved = True
                break
        if not improved:
            lr *= 0.9
            stuck_count += 1
            if stuck_count > 200:
                break
        else:
            stuck_count = 0
            if lr < 0.05:
                lr += 0.001
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def binary_swap_search(h, n, deadline, max_sweeps=500):

            grad[i_lo:i_hi + 1] += wk * (1.0 - x[j_lo:j_hi + 1])
        i_lo2 = max(0, k - (n - 1))
        i_hi2 = min(n - 1, k)
        if i_hi2 >= i_lo2:
            jp_lo = i_lo2 - k + (n - 1)
            jp_hi = i_hi2 - k + (n - 1)
            grad[i_lo2:i_hi2 + 1] -= wk * x[jp_lo:jp_hi + 1]
    return grad

def slsqp_optimize(h, n, alpha, maxiter, dl):
    def f(x):
        return smooth_max_eval(x, n, alpha)
    def g(x):
        return smooth_max_grad(x, n, alpha)
    def cf(x):
        return x.sum() - n / 2.0
    def cg(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cg}]
    try:
        res = minimize(f, h, method='SLSQP', jac=g, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_refine(h, n, dl, max_it=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.08
    stuck = 0
    for it in range(max_it):
        if time.time() > dl:
            break
        c = np.correlate(h, 1 - h, 'full') * dx
        m = c.max()
        ks = np.where(c >= m - 1e-14)[0]
        grad = np.zeros(n)
        for k in ks:
            w = 1.0 / len(ks)
            i_lo = max(0, n - 1 - k)
            i_hi = min(n - 1, n - 1 + k)
            if i_hi >= i_lo:
                j_lo = i_lo + k - (n - 1)
                j_hi = i_hi + k - (n - 1)
                grad[i_lo:i_hi + 1] += w * (1.0 - h[j_lo:j_hi + 1])
            i_lo2 = max(0, k - (n - 1))
            i_hi2 = min(n - 1, k)
            if i_hi2 >= i_lo2:
                jp_lo = i_lo2 - k + (n - 1)
                jp_hi = i_hi2 - k + (n - 1)
                grad[i_lo2:i_hi2 + 1] -= w * h[jp_lo:jp_hi + 1]
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        ok = False
        for fct in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01, 0.005]:
            step = lr * fct / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                ok = True
                break
        if not ok:
            lr *= 0.88
            stuck += 1
            if stuck > 500 or lr < 1e-14:
                break
        else:
            stuck = 0
            if lr < 0.08:
                lr = min(lr + 0.002, 0.08)
            cnew = compute_c5(h, n)
            if cnew < best_c:
                best_c = cnew
                best_h = h.copy()
    return best_h, best_c

def binary_greedy(h, n, dl, max_sweeps=300, max_swaps=800):
    tgt = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:tgt]] = 1.0
    best_c = compute_c5(hb, n)
    ones_set = set(np.where(hb == 1)[0].tolist())
    zeros_set = set(np.where(hb == 0)[0].tolist())
    for sweep in range(max_sweeps):
        if time.time() > dl:
            break
        found = False
        swaps_done = 0
        for i in sorted(zeros_set):
            if time.time() > dl or swaps_done >= max_swaps:
                break
            for j in sorted(ones_set):
                if swaps_done >= max_swaps:
                    break
                hb[i], hb[j] = 1.0, 0.0
                c = compute_c5(hb, n)
                if c < best_c - 1e-15:
                    best_c = c
                    swaps_done += 1
                    ones_set.remove(j)
                    ones_set.add(i)
                    zeros_set.add(j)
                    zeros_set.remove(i)
                    found = True
                    break
                else:
                    hb[i], hb[j] = 0.0, 1.0
                    swaps_done += 1
            if found:
                break
        if not found:
            break
    return hb, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP from parent with increasing alpha
    p1 = t0 + 0.30 * (budget_s - 5)
    h = best_h.copy()
    schedule = [(200, 150), (1000, 200), (5000, 250), (20000, 300),
                (100000, 300), (500000, 400), (2000000, 400), (10000000, 400)]
    for a, mi in schedule:
        if time.time() > p1 - 10:
            break
        h = slsqp_optimize(h, n, a, mi, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 1b: Random restarts
    if time.time() < p1 - 20:
        for rs in range(3):
            hp = project_feasible(best_h + 0.15 * rng.randn(n), n)
            hp = slsqp_optimize(hp, n, 5000, 200, p1)
            hp = slsqp_optimize(hp, n, 200000, 300, p1)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 2: Subgradient refinement
    p2 = t0 + 0.45 * (budget_s - 5)
    h2, c2 = subgradient_refine(best_h.copy(), n, p2, max_it=200000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Try different resolutions
    p3 = t0 + 0.70 * (budget_s - 5)
    for nh in [200, 300, 400]:
        if time.time() > p3 - 50:
            break
        if nh == n:
            continue
        xo = (np.arange(n) + 0.5) * (2.0 / n)
        xn = (np.arange(nh) + 0.5) * (2.0 / nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        for a, mi in [(2000, 200), (20000, 250), (200000, 300), (2000000, 300)]:
            if time.time() > p3 - 30:
                break
            hh = slsqp_optimize(hh, nh, a, mi, p3 - 5)
        hh, ch = subgradient_refine(hh, nh, p3 - 15, max_it=20000)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh

    # Phase 4: Binary greedy search
    p4 = t0 + 0.87 * (budget_s - 5)
    if n <= 400:
        hb, cb = binary_greedy(best_h.copy(), n, p4, max_sweeps=200, max_swaps=600)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 5: Final SLSQP polish
    if time.time() < dl - 20:
        hf = slsqp_optimize(best_h.copy(), n, 50000000, 300, dl - 10)
        cf = compute_c5(hf, n)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    # Phase 6: Brief subgradient final
    if time.time() < dl - 3:
        best_h, best_c = subgradient_refine(best_h, n, dl, max_it=50000)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
