from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def binary_ls(h_bin, n, deadline, max_sweeps=500):
    """Local search: accept any 2-swap that reduces C5."""
    best_c = compute_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(zeros)):
            if time.time() > deadline:
                return h_bin, best_c
            i = zeros[ii]
            for jj in range(len(ones)):
                j = ones[jj]
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    zeros.pop(ii)
                    ones.pop(jj)
                    zeros.append(i)
                    ones.append(j)
                    break
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def sa_binary(h_bin, n, deadline, max_iter=200000):
    """Simulated annealing on binary h."""
    cur_c = compute_c5(h_bin, n)
    best_c = cur_c

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    """SLSQP with smooth max approximation."""
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        diff = g - m
        sh = alpha * diff
        sm = sh.max()
        ess = np.exp(sh - sm).sum()
        return m + np.log(ess) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(-h)[:n_ones]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def binary_ls(h_bin, n, deadline, max_sweeps=500, rng=None):
    best_c = compute_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    no_improve_count = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(zeros)):
            if time.time() > deadline:
                return h_bin, best_c
            i = zeros[ii]
            for jj in range(len(ones)):
                j = ones[jj]
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    zeros.pop(ii)
                    ones.pop(jj)
                    zeros.append(i)
                    ones.append(j)
                    no_improve_count = 0
                    break
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 3:
                break
    return h_bin, best_c

def sa_binary(h_bin, n, deadline, max_iter=100000, T_start=0.015, T_end=0.0001, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    cur_c = compute_c5(h_bin, n)
    best_c = cur_c
    best_h = h_bin.copy()
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    n_ones = n // 2
    cool_rate = (T_end / T_start) ** (1.0 / max(max_iter, 1))
    T = T_start
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = max(T * cool_rate, T_end)
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i], h_bin[j] = 1.0, 0.0
        new_c = compute_c5(h_bin, n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            zeros.remove(i)
            ones.remove(j)
            ones.append(i)
            zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = 0.0, 1.0
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + np.log(np.exp(sh - sm).sum()) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_polish(h, n, deadline, max_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        m = g.max()
        ks = int(np.argmax(g))
        grad = np.zeros(n)
        eps = 1e-8
        for i in range(n):
            old = h[i]
            h[i] = old + eps
            grad[i] = (np.correlate(h, 1 - h, 'full')[ks] * dx - g[ks]) / eps
            h[i] = old
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                found = True
                break
        if not found:
            lr *= 0.9
            if lr < 1e-12:
                break
    return best_h, best_c

def multi_swap_refine(h_bin, n, deadline, max_rounds=50):
    """Try 3-swaps: move one 1 to a 0 position and swap two other positions for balance."""
    best_c = compute_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try moving a 1 from position a to position b (both must be valid)
        # Actually just do random 2-swaps more aggressively
        for _ in range(200):
            if time.time() > deadline:
                return h_bin, best_c
            i = zeros[np.random.randint(len(zeros))]
            j = ones[np.random.randint(len(ones))]
            h_bin[i], h_bin[j] = 1.0, 0.0
            c = compute_c5(h_bin, n)
            if c < best_c - 1e-14:
                best_c = c
                improved = True
                zeros.remove(i)
                ones.remove(j)
                ones.append(i)
                zeros.append(j)
            else:
                h_bin[i], h_bin[j] = 0.0, 1.0
        if not improved:
            break
    return h_bin, best_c
