import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_top_shifts(h, n, top_k=5, threshold_frac=0.005):
    c = compute_c5_vec(h, n)
    m = c.max()
    threshold = m - max(1e-8, threshold_frac * m)
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c[active])
        active = active[order[:top_k]]
    return active.astype(int)

def smax_obj(x, n, alpha, top_idx):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh - smx) + 1e-300)) / alpha
    return float(val)

def smax_grad(x, n, alpha, top_idx):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-16:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def smax_obj_with_mu(x, n, alpha, top_idx, mu):

    m = vals.max()

    smx = sh.max()

    else:
        val = m + np.log(np.sum(np.exp(sh - smx) + 1e-300)) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def smax_grad_with_mu(x, n, alpha, top_idx, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)

def slqp_direct(h, n, deadline, top_idx, maxiter=2000):
    """SLSQP: minimize t s.t. corr_k <= t for top shifts, sum constraint, bounds."""
    x0 = np.concatenate([h, [compute_c5(h, n) + 0.01]])
    
    def obj(x):
        return x[n]
    
    def obj_jac(x):
        j = np.zeros(n + 1)
        j[n] = 1.0
        return j
    
    cons = []
    for idx in top_idx:
        k = idx - (n - 1)
        cons.append({
            'type': 'ineq',
            'fun': lambda x, idx=idx, k=k: slqp_corr_diff(x, n, idx, k),
            'jac': lambda x, idx=idx, k=k: slqp_corr_jac(x, n, idx, k)
        })
    cons.append({
        'type': 'eq',
        'fun': lambda x: x[:n].sum() - n / 2.0,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    })
    
    bounds = [(0.0, 1.0)] * n + [(0.01, 2.0)]
    
    try:
        res = minimize(obj, x0, jac=obj_jac, args=(), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': -1})
        h_new = project_feasible(res.x[:n], n)
        return h_new
    except Exception:
        return h.copy()

def slqp_corr_diff(x, n, idx, k):
    h = x[:n]
    t = x[n]
    c = compute_c5_vec(h, n)
    return t - c[idx]

def slqp_corr_jac(x, n, idx, k):
    h = x[:n]
    dx = 2.0 / n
    j = np.zeros(n + 1)
    j[n] = 1.0
    if k >= 0:
        nv = n - k
        j[:nv] -= dx * (1.0 - h[k:])
        j[k:] -= dx * h[:nv]
    else:
        kk = -k
        nv = n - kk
        j[kk:] -= dx * (1.0 - h[:nv])
        j[:nv] -= dx * h[kk:]
    return j

                return h_work, best_c

def _coord_obj(h, n, i, t, top_idx, mu):

    s = h.sum() - n / 2.0
    h[i] = t  # keep it

def binary_push(h, n, threshold=0.01):
    """Push values near 0 or 1 to exactly 0 or 1, adjust rest."""
    h = h.copy()
    mask_low = h < threshold
    mask_high = h > 1.0 - threshold
    n_free = n - mask_low.sum() - mask_high.sum()
    if n_free <= 0:
        return h
    old_sum = h.sum()
    h[mask_low] = 0.0
    h[mask_high] = 1.0
    target_sum = n / 2.0
    remaining_target = target_sum - h.sum()
    # Distribute remaining over free coords
    free_idx = np.where((~mask_low) & (~mask_high))[0]
    if len(free_idx) > 0:
        h[free_idx] += remaining_target / len(free_idx)
    return project_feasible(h, n)

    if time.time() < t7:
        best_h, best_c = multistart_explore(best_h, n, t7, n_starts=2)
    
    # Phase 8: Final SLSQP polish (30s)
    if time.time() < deadline - 5:
        h = best_h.copy()
        top_idx = get_top_shifts(h, n, top_k=3)
        h_new = slqp_direct(h, n, deadline - 3, top_idx, maxiter=1000)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
    
    # Final: ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
