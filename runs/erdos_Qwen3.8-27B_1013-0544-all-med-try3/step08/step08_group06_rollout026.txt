    # gamma: change coefficients (2n-1,)
    # Find dh in [dh_min, dh_max] that minimizes max_m(c[m] + dh*gamma[m])
    
    # Evaluate at endpoints
    f_min = c + dh_min * gamma
    f_min_val = f_min.max()
    best_dh = dh_min
    
    f_max = c + dh_max * gamma
    f_max_val = f_max.max()
    if f_max_val < f_min_val:
        f_min_val = f_max_val
        best_dh = dh_max
    
    # Check if dh=0 is better
    f_0 = c.max()
    if f_0 < f_min_val:
        f_min_val = f_0
        best_dh = 0.0
    
    # Subgradient bisection
    # At dh, the subgradient is {gamma[m] : c[m] + dh*gamma[m] == f(dh)}
    lo = dh_min
    hi = dh_max
    
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = c + mid * gamma
        f_val = vals.max()
        # Active set
        tol = 1e-12
        active = np.abs(vals - f_val) < tol
        if not active.any():
            active = (vals == f_val)
        
        min_g = gamma[active].min()
        max_g = gamma[active].max()
        
        if min_g < -tol:
            # f is decreasing at mid, minimum is to the left
            hi = mid
        elif max_g > tol:
            # f is increasing at mid, minimum is to the right
            lo = mid

        best_dh = best_dh

def compute_A(h, n):
    """Compute the sensitivity matrix A[i, m] = (1-h[i+delta_m]) - h[i-delta_m]"""
    A = np.zeros((n, 2*n-1))
    deltas = np.arange(2*n-1) - (n-1)
    for i in range(n):
        idx1 = i + deltas
        mask1 = (idx1 >= 0) & (idx1 < n)
        A[i, mask1] += (1.0 - h[idx1[mask1]])
        idx2 = i - deltas
        mask2 = (idx2 >= 0) & (idx2 < n)
        A[i, mask2] -= h[idx2[mask2]]
    return A
