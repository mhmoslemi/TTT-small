            if tot < 1e-14:

            h += (diff / tot) * room

            if tot < 1e-14:

    shifted = alpha * (c - m)
    mx = shifted.max()

        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:

def pair_optimize(h, n, i, j, hi, hj):
    # Optimize h[i] and h[j] with h[i]+h[j] = hi+hj = const
    const = hi + hj
    # t = h[i], h[j] = const - t
    # t must satisfy: 0 <= t <= 1, 0 <= const-t <= 1
    t_lo = max(0.0, const - 1.0)
    t_hi = min(1.0, const)
    if t_lo >= t_hi:
        return 0.0
    
    best_t = t_lo
    best_c = compute_c5(h, n)  # current
    
    # Sampling-based search
    ts = np.linspace(t_lo, t_hi, 20)
    for t in ts:
        h[i] = t
        h[j] = const - t
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_t = t
    
    # Refine around best_t
    if t_hi - t_lo > 1e-6:
        window = (t_hi - t_lo) / 20
        ts2 = np.linspace(max(t_lo, best_t - window), min(t_hi, best_t + window), 10)
        for t in ts2:
            h[i] = t
            h[j] = const - t
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_t = t
    
    h[i] = best_t
    h[j] = const - best_t
    return best_c
