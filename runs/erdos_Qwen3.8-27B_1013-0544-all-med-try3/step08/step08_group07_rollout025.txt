def g_array(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5_value(h):
    return float(np.max(g_array(h)))


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    n = len(v)
    if n == 0:
        return v.copy()
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    lam = 0.5 * (lo + hi)
    p = np.clip(v - lam, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        free = (p > 1e-12) & (p < 1.0 - 1e-12)
        nf = int(free.sum())
        if nf > 0:
            p[free] += diff / float(nf)
        else:
            p += diff / float(n)
        p = np.clip(p, 0.0, 1.0)
        diff = target - p.sum()
        if abs(diff) > 1e-12:
            free = (p > 1e-12) & (p < 1.0 - 1e-12)
            nf = int(free.sum())
            if nf > 0:
                p[free] += diff / float(nf)
    return p


def finalize_feasible(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) <= 1e-12:
        return h
    free = (h > 1e-12) & (h < 1.0 - 1e-12)
    nf = int(free.sum())
    if nf > 0:
        h[free] += diff / float(nf)
    else:
        h += diff / float(n)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        project_feas = project_box_sum(h, target)
        return project_feas
    return h


def top_active_indices(c, L):
    m = len(c)
    if L <= 0:
        return np.empty(0, dtype=int)
    if L >= m:
        return np.argsort(c)[::-1]
    part = np.argpartition(c, m - L)[-L:]
    return part[np.argsort(c[part])[::-1]]


def softmax_weights(vals, temp):
    if vals.size == 0:
        return np.empty(0, dtype=float)
    if temp <= 1e-9:
        w = np.zeros(vals.shape, dtype=float)
        w[int(np.argmax(vals))] = 1.0
        return w
    v = (vals - float(vals[0])) / temp
    v = np.clip(v, -50.0, 0.0)
    e = np.exp(v)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros(vals.shape, dtype=float)
        w[0] = 1.0
        return w
    return e / s


def fixed_surrogate(ref, vals, temp):
    if vals.size == 0:
        return float(ref)
    if temp <= 1e-9:
        return float(max(ref, np.max(vals)))
    v = (vals - ref) / temp
    vmax = float(np.max(v))
    s = float(np.exp(np.clip(v - vmax, -50.0, 0.0)).sum())
    return float(ref + temp * (math.log(s) + vmax))


def classify_lag(h, n, c, idx):
    center = n - 1
    t = abs(int(idx) - center)
    if t == 0:
        return 0, True
    dx = 2.0 / float(n)
    seg = n - t
    pv = dx * float(np.dot(h[:seg], 1.0 - h[t:]))
    nv = dx * float(np.dot(h[t:], 1.0 - h[:seg]))
    cv = float(c[idx])
    if abs(cv - pv) <= abs(cv - nv):
        return t, True
    return t, False


def add_grad_type(h, n, dx, g, t, is_pos, w):
    if t == 0:
        g += float(w) * dx * (1.0 - 2.0 * h)
        return
    seg = n - t
    if is_pos:
        g[:seg] += float(w) * dx * (1.0 - h[t:])
        g[t:] -= float(w) * dx * h[:seg]
    else:
        g[t:] += float(w) * dx * (1.0 - h[:seg])
        g[:seg] -= float(w) * dx * h[t:]


def smooth_projected(h, n, t_end, schedule, rng=None):
    target = 0.5 * float(n)
    best = project_box_sum(h, target).copy()
    best_c = c5_value(best)

    for temp, L, itmax in schedule:
        if time.time() >= t_end:
            break
        h = best.copy()
        c = g_array(h)
        cur = float(c.max())
        if cur < best_c:
            best_c = cur
            best = h.copy()

        base = 1.0
        stall = 0
        drift = max(1e-6, 0.5 * temp)

        for _ in range(int(itmax)):
            if time.time() >= t_end:
                break
            if cur > best_c + drift:
                h = best.copy()
                c = g_array(h)
                cur = float(c.max())

            L_eff = min(int(L), len(c))
            idxs = top_active_indices(c, L_eff)
            if idxs.size == 0:
                break

            if rng is not None and rng.random_sample() < 0.18:
                r = 1 + int(rng.randint(0, min(int(idxs.size), 6)))
                idxs_use = idxs[:r]
                w = np.full(r, 1.0 / float(r), dtype=float)
            else:
                idxs_use = idxs
                w = softmax_weights(c[idxs], temp)

            g = np.zeros(n, dtype=float)
            dx = 2.0 / float(n)
            for p in range(len(idxs_use)):
                wi = float(w[p])
                if wi <= 1e-15:
                    continue
                t, is_pos = classify_lag(h, n, c, idxs_use[p])
                add_grad_type(h, n, dx, g, t, is_pos, wi)

            g -= g.mean()
            gn = float(np.linalg.norm(g))
            if gn < 1e-12:
                base *= 0.6
                stall += 1
                if stall >= 45 or base < 1e-3:
                    break
                continue

            oldF = fixed_surrogate(cur, c[idxs_use], temp)
            allow = max(1e-8, 0.5 * temp)
            step0 = min(20.0, 0.25 / max(gn, 1e-6))
            accepted = False
            accept_h = None
            accept_c = None
            accept_val = None
            accept_exact = False

            for factor in STEP_FACTORS:
                if time.time() >= t_end:
                    break
                step = base * step0 * factor
                if step <= 0.0:
                    continue
                y = project_box_sum(h - step * g, target)
                cny = g_array(y)
                cy = float(cny.max())
                if cy < cur - 1e-12:
                    accepted = True
                    accept_h = y
                    accept_c = cny
                    accept_val = cy
                    accept_exact = True
                    break
                if temp > 1e-9 and cy <= cur + allow:
                    fn = fixed_surrogate(cur, cny[idxs_use], temp)
                    if fn < oldF - 1e-11:
                        accepted = True
                        accept_h = y
                        accept_c = cny
                        accept_val = cy
                        accept_exact = False
                        break

            if accepted:
                h = accept_h
                c = accept_c
                cur = float(accept_val)
                if cur < best_c - 1e-14:
                    best_c = cur
                    best = h.copy()
                stall = 0
                if accept_exact:
                    base = min(2.0, base * 1.15)
                else:
                    base = min(2.0, base * 1.01)
            else:
                base *= 0.6
                stall += 1
                if stall >= 20 and rng is not None:
                    sig = 0.015 + 0.02 * min(1.0, temp)
                    h = project_box_sum(h + sig * rng.normal(size=n), target)
                    c = g_array(h)
                    cur = float(c.max())
                    base = 1.0
                    stall = 0
                if stall >= 60 or base < 1e-3:
                    break

    return finalize_feasible(best), best_c


def pair_exchange(h, n, t_end, active_count=14, top_edges=96, rng=None):
    target = 0.5 * float(n)
    h = project_box_sum(h, target).copy()
    c = g_array(h)
    cur = float(c.max())
    best = h.copy()
    best_c = cur
    sweeps = 0

    while time.time() < t_end and sweeps < 1000000:
        sweeps += 1
        if cur < best_c:
            best_c = cur
            best = h.copy()

        idxs = top_active_indices(c, min(int(active_count), len(c)))
        if idxs.size == 0:
            break

        if rng is not None and rng.random_sample() < 0.15:
            r = 1 + int(rng.randint(0, min(int(idxs.size), 6)))
            idxs_use = idxs[:r]
            wt = np.full(r, 1.0 / float(r), dtype=float)
        else:
            idxs_use = idxs
            wt = softmax_weights(c[idxs], 0.02)

        src_arrays = []
        dst_arrays = []
        score_arrays = []
        center = n - 1

        for p in range(len(idxs_use)):
            w = float(wt[p])
            if w <= 1e-12:
                continue
            idx = int(idxs_use[p])
            t, is_pos = classify_lag(h, n, c, idx)
            if t == 0 or t >= n:
                continue
            seg = n - t
            if is_pos:
                s = w * h[:seg] * (1.0 - h[t:])
                iv = np.nonzero(s > 1e-10)[0]
                if iv.size:
                    src_arrays.append(iv)
                    dst_arrays.append(iv + t)
                    score_arrays.append(s[iv])
            else:
                s = w * h[t:] * (1.0 - h[:seg])
                iv = np.nonzero(s > 1e-10)[0]
                if iv.size:
                    src_arrays.append(iv + t)
                    dst_arrays.append(iv)
                    score_arrays.append(s[iv])

        if not src_arrays:
            if rng is None:
                break
            random_done = False
            for _ in range(120):
                if time.time() >= t_end:
                    break
                idx = int(idxs_use[rng.randint(0, len(idxs_use))])
                t, is_pos = classify_lag(h, n, c, idx)
                if t == 0:
                    continue
                seg = n - t
                if is_pos:
                    i = int(rng.randint(0, seg))
                    j = i + t
                else:
                    j = int(rng.randint(0, seg))
                    i = j + t
                maxd = min(float(h[i]), float(1.0 - h[j]))
                if maxd <= 1e-9:
                    continue
                hi = float(h[i])
                hj = float(h[j])
                for frac in (0.25, 0.1, 0.05, 0.02):
                    d = frac * maxd
                    h[i] = hi - d
                    h[j] = hj + d
                    cny = g_array(h)
                    ny = float(cny.max())
                    if ny < cur - 1e-13:
                        c = cny
                        cur = ny
                        random_done = True
                        if cur < best_c:
                            best_c = cur
                            best = h.copy()
                        break
                    else:
                        h[i] = hi
                        h[j] = hj
            if not random_done:
                break
            continue

        src = np.concatenate(src_arrays)
        dst = np.concatenate(dst_arrays)
        scores = np.concatenate(score_arrays)

        if src.size > top_edges:
            part = np.argpartition(scores, -top_edges)[-top_edges:]
            part = part[np.argsort(scores[part])[::-1]]
        else:
            part = np.argsort(scores)[::-1]

        improved = False
        for e in part:
            if time.time() >= t_end:
                break
            i = int(src[e])
            j = int(dst[e])
            if i == j:
                continue
            maxd = min(float(h[i]), float(1.0 - h[j]))
            if maxd <= 1e-12:
                continue
            hi = float(h[i])
            hj = float(h[j])
            for frac in PAIR_FRACTIONS:
                d = frac * maxd
                h[i] = hi - d
                h[j] = hj + d
                cny = g_array(h)
                ny = float(cny.max())
                if ny < cur - 1e-13:
                    c = cny
                    cur = ny
                    improved = True
                    if cur < best_c:
                        best_c = cur
                        best = h.copy()
                    break
                else:
                    h[i] = hi
                    h[j] = hj

    return finalize_feasible(best), best_c


def binary_from_values(h, n):
    target = n // 2
    out = np.zeros(n, dtype=float)
    if target <= 0:
        return out
    if target >= n:
        out[:] = 1.0
        return out
    idx = np.argpartition(h, -target)[-target:]
    out[idx] = 1.0
    return out


def binary_state(h):
    n = len(h)
    ones = []
    zeros = []
    for i, v in enumerate(h):
        if v > 0.5:
            ones.append(i)
        else:
            zeros.append(i)
    pos_o = np.full(n, -1, dtype=int)
    pos_z = np.full(n, -1, dtype=int)
    for k, i in enumerate(ones):
        pos_o[i] = k
    for k, i in enumerate(zeros):
        pos_z[i] = k
    return ones, zeros, pos_o, pos_z


def swap_binary_state(h, a, b, ones, zeros, pos_o, pos_z):
    pa = int(pos_z[a])
    pb = int(pos_o[b])
    if pa < 0 or pb < 0:
        return
    h[a] = 1.0
    h[b] = 0.0
    zeros[pa] = b
    pos_z[b] = pa
    pos_z[a] = -1
    ones[pb] = a
    pos_o[a] = pb
    pos_o[b] = -1


def binary_anneal(start, n, t_end, rng):
    if n <= 2:
        h = binary_from_values(start, n) if len(start) == n else project_box_sum(start, 0.5 * n)
        return finalize_feasible(h), c5_value(h)
    if n % 2 != 0:
        n = 2 * (n // 2)
        start = start[:n] if len(start) >= n else interpolate_h(start, len(start), n)
    h = binary_from_values(start, n)
    old_c = c5_value(h)
    best = h.copy()
    best_c = old_c
    ones, zeros, pos_o, pos_z = binary_state(h)
    T = 0.025
    evals = 0
    stuck = 0

    while time.time() < t_end and len(zeros) > 0 and len(ones) > 0:
        a = zeros[rng.randint(0, len(zeros))]
        b = ones[rng.randint(0, len(ones))]
        h[a] = 1.0
        h[b] = 0.0
        cny = g_array(h)
        new_c = float(cny.max())
        evals += 1

        if new_c <= old_c + 1e-15:
            swap_binary_state(h, a, b, ones, zeros, pos_o, pos_z)
            old_c = new_c
            stuck = 0
            if new_c < best_c - 1e-15:
                best_c = new_c
                best = h.copy()
        else:
            accepted = False
            if T > 1e-6:
                logp = (old_c - new_c) / T
                if logp > -50.0 and rng.random_sample() < math.exp(logp):
                    swap_binary_state(h, a, b, ones, zeros, pos_o, pos_z)
                    old_c = new_c
                    accepted = True
            if not accepted:
                h[a] = 0.0
                h[b] = 1.0
            stuck += 1

        if evals % 1000 == 0:
            T *= 0.985
            if T < 1e-5:
                T = 1e-5
        if stuck > 6000:
            T *= 1.25
            if T > 0.08:
                T = 0.08
            stuck = 0
        if evals % 4000 == 0 and (best_c - old_c > 0.05 or stuck > 40000):
            h = best.copy()
            old_c = best_c
            ones, zeros, pos_o, pos_z = binary_state(h)
            T = 0.03
            stuck = 0

    return finalize_feasible(best), best_c


def upsample_h(h, n_old, n_new):
    if n_new == n_old:
        return project_box_sum(h, 0.5 * float(n_new)).copy()
    old_x = np.linspace(0.0, 2.0, n_old)
    new_x = np.linspace(0.0, 2.0, n_new)
    lin = np.interp(new_x, old_x, h)
    idx = np.clip(np.rint(np.linspace(0.0, float(n_old - 1), n_new)).astype(int), 0, n_old - 1)
    near = h[idx]
    li = project_box_sum(lin, 0.5 * float(n_new))
    ni = project_box_sum(near, 0.5 * float(n_new))
    cli = c5_value(li)
    cni = c5_value(ni)
    return li if cli <= cni else ni


def random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
    return finalize_feasible(h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 12.0)
    rng = np.random.RandomState(seed)

    parent = kwargs.get("initial_h_values", None)
    if parent is None:
        parent = globals().get("initial_h_values", None)

    if parent is not None:
        parent = np.asarray(parent, dtype=float).copy()
        if parent.size == 0:
            parent = np.full(256, 0.5, dtype=float)
    else:
        parent = random_binary(256, rng)

    n0 = int(parent.size)
    if n0 < 2:
        parent = np.full(256, 0.5, dtype=float)
        n0 = 256
    if n0 % 2 != 0:
        parent = np.interp(
            np.linspace(0.0, 2.0, n0 + 1),
            np.linspace(0.0, 2.0, n0),
            parent,
        )
        n0 += 1

    best_h = finalize_feasible(parent)
    best_n = int(best_h.size)
    best_c = c5_value(best_h)

    work = max(10.0, min(float(budget_s), 1100.0) - 20.0)
    if budget_s > 600:
        t_n0 = t0 + min(0.38 * work, 450.0)
        if t_n0 < t0 + 60.0:
            t_n0 = t0 + 60.0
        t_n0 = min(t_n0, deadline - 170.0)
    elif budget_s > 350:
        t_n0 = t0 + 0.50 * work
        t_n0 = min(t_n0, deadline - 80.0)
    else:
        t_n0 = deadline - 5.0

    n0_span = max(1.0, t_n0 - t0)
    e1 = min(t_n0, t0 + 0.50 * n0_span)
    e2 = min(t_n0, t0 + 0.72 * n0_span)
    e3 = min(t_n0, t0 + 0.88 * n0_span)

    if time.time() < e1:
        h_work, c_work = smooth_projected(best_h, n0, e1, schedule_for_n(n0), rng)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
            best_n = n0

    if time.time() < e2:
        h_work, c_work = pair_exchange(best_h, n0, e2, active_count=14, top_edges=96, rng=rng)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
            best_n = n0

    if time.time() < e3 and n0 <= 600:
        h_work = binary_from_values(best_h, n0)
        h_work, c_work = binary_anneal(h_work, n0, e3, rng)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
            best_n = n0

    restarts = 0
    while time.time() < t_n0 - 5.0 and restarts < 3:
        sig = 0.025 + 0.02 * restarts
        y = project_box_sum(best_h + sig * rng.normal(size=best_n), 0.5 * float(best_n))
        hard_end = min(t_n0 - 2.0, time.time() + max(10.0, 0.12 * n0_span))
        y, cy = smooth_projected(y, best_n, hard_end, SCHEDULE_SHORT, rng)
        if cy < best_c:
            best_c = cy
            best_h = y.copy()
            best_n = int(y.size)
        if time.time() < min(t_n0 - 2.0, time.time() + max(5.0, 0.05 * n0_span)):
            y, cy = pair_exchange(best_h, best_n, min(t_n0 - 2.0, time.time() + 15.0),
                                  active_count=10, top_edges=48, rng=rng)
            if cy < best_c:
                best_c = cy
                best_h = y.copy()
                best_n = int(y.size)
        restarts += 1

    if time.time() < deadline - 120.0:
        if best_n <= 340:
            nH = 512
        elif best_n <= 700:
            nH = 768
        else:
            nH = min(1024, 2 * ((int(best_n) + 128) // 2))
        if nH % 2 != 0:
            nH += 1
        if nH <= best_n:
            nH = int(best_n) + 128
            if nH % 2 != 0:
                nH += 1
        nH = min(nH, 1024)

        if nH > best_n and nH >= 2:
            hH = upsample_h(best_h, best_n, nH)
            if time.time() < deadline - 60.0:
                hH, chH = smooth_projected(hH, nH, deadline - 60.0, schedule_for_n(nH), rng)
            else:
                chH = c5_value(hH)
            if chH < best_c:
                best_c = chH
                best_h = hH.copy()
                best_n = nH

            if time.time() < deadline - 30.0:
                hH, chH = pair_exchange(hH, nH, deadline - 25.0,
                                        active_count=12, top_edges=80, rng=rng)
                if chH < best_c:
                    best_c = chH
                    best_h = hH.copy()
                    best_n = nH

    if time.time() < deadline - 8.0:
        h_final, c_final = smooth_projected(best_h, best_n, deadline - 6.0, SCHEDULE_FINAL, rng)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
            best_n = int(h_final.size)

    if time.time() < deadline - 3.0:
        h_final, c_final = pair_exchange(best_h, best_n, deadline - 2.0,
                                         active_count=10, top_edges=48, rng=rng)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
            best_n = int(h_final.size)

    best_h = finalize_feasible(best_h)
    best_c = c5_value(best_h)
    return best_h, float(best_c), int(best_h.size)
