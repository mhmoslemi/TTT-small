    """Compute gradient of smoothed max, focusing on top shifts."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    # Get weights for top shifts
    weights = np.exp(alpha * (c[top_indices] - m))
    weights /= weights.sum()
    
    # Compute gradient as weighted sum of shifted (1-h) and h
    grad = np.zeros(n)
    for w, idx in zip(weights, top_indices):
        k = idx - (n-1)
        if k >= 0:
            nv = n - k
            grad[:nv] += w * (1 - h[k:]) * dx
            grad[k:] += w * (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += w * (1 - h[:nv]) * dx
            grad[:nv] += w * (-h[kk:]) * dx
    return grad

                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def targeted_local_search(h, n, deadline, top_k=10, max_radius=60, steps=None):
    """Local search that focuses on pairs affecting the top shifts."""
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_c = compute_c5(h, n)
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            top_idx = np.argsort(-c_vec)[:top_k]
            # Determine which indices are "active" for top shifts
            active_set = set()
            for idx in top_idx:
                k = int(idx) - (n - 1)
                if k >= 0:
                    for i in range(n - k):
                        active_set.add(i)
                        if i + k < n:
                            active_set.add(i + k)
                else:
                    kk = -k
                    for i in range(n - kk):
                        active_set.add(i)
                        if i + kk < n:
                            active_set.add(i + kk)
            active_list = list(active_set)
            np.random.shuffle(active_list)
            for i in active_list:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def perturbation_restart(h, n, best_c, bounds, deadline, scale=0.02, n_trials=3):
    """Try perturbing the solution and re-optimizing."""
    best_h = h.copy()
    best_c_local = best_c
    for trial in range(n_trials):
        if time.time() > deadline - 30:
            break
        h_pert = h + scale * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c_local:
            best_h = h_pert
            best_c_local = c_pert
            continue
        # Quick L-BFGS from perturbed starting point
        try:
            res = minimize(lse_objective_top, h_pert, jac=lse_gradient_top,
                         args=(n, 100000, np.argsort(-compute_c5_vec(h_pert, n))[:8]),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c_local:
                best_h = h_new
                best_c_local = c_new
        except Exception:
            pass
        scale *= 0.5
    return best_h, best_c_local

def slqp_smooth_objective(x, n, alpha):
    return lse_objective_full(x, n, alpha)

def slqp_smooth_gradient(x, n, alpha):
    return lse_gradient_full(x, n, alpha, top_k=30)

def sum_constraint(x, n):
    return float(x.sum() - n / 2.0)

def sum_constraint_jac(x, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 8

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    # Choose working resolution
    n = 1000
    x_init = np.linspace(0, 2, n_init)
    x = np.linspace(0, 2, n)
    h = np.interp(x, x_init, h_init)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with increasing alpha (full LSE)
    alpha_configs = [
        (50, 300, 40), (200, 400, 40), (1000, 500, 30),
        (5000, 500, 25), (20000, 400, 20), (100000, 300, 15), (500000, 200, 10)
    ]
    for alpha, maxiter, tk in alpha_configs:
        if time.time() > deadline - 350:
            break
        try:
            res = minimize(lse_objective_full, h, jac=lse_gradient_full,
                         args=(n, alpha, tk), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-20,
                                  'gtol': 1e-12, 'maxfun': maxiter * 10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    # Phase 2: Top-shifts focused optimization (multiple rounds)
    for round_num in range(6):
        if time.time() > deadline - 250:
            break
        c_vec = compute_c5_vec(h, n)
        top_idx = np.argsort(-c_vec)[:12]
        for alpha in [20000, 200000, 2000000]:
            if time.time() > deadline - 240:
                break
            try:
                res = minimize(lse_objective_top, h, jac=lse_gradient_top,
                             args=(n, alpha, top_idx), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 3000, 'ftol': 1e-20,
                                      'gtol': 1e-13, 'maxfun': 30000, 'maxls': 100})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
            pass

    # Phase 3: Top-k average objective optimization
    if time.time() < deadline - 200:
        for _ in range(3):
            if time.time() > deadline - 190:
                break
            c_vec = compute_c5_vec(h, n)
            top_idx3 = np.argsort(-c_vec)[:8]
            try:
                res = minimize(topk_avg_objective, h, jac=topk_avg_gradient,
                             args=(n, top_idx3), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 3000, 'ftol': 1e-18,
                                      'gtol': 1e-12, 'maxfun': 30000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass

    # Phase 4: Local pair search (coarse)
    if time.time() < deadline - 150:
        h, c_val = local_pair_search(h, n, deadline - 150, max_radius=40,
                                     steps=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

    # Phase 5: Targeted local search (focusing on active shifts)
    if time.time() < deadline - 100:
        h, c_val = targeted_local_search(h, n, deadline - 100, top_k=10, max_radius=50,
                                         steps=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

    # Phase 6: Another L-BFGS round with very high alpha
    if time.time() < deadline - 80:
        c_vec = compute_c5_vec(h, n)
        top_idx6 = np.argsort(-c_vec)[:6]
        for alpha in [1000000, 5000000, 20000000]:
            if time.time() > deadline - 70:
                break
            try:
                res = minimize(lse_objective_top, h, jac=lse_gradient_top,
                             args=(n, alpha, top_idx6), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 2000, 'ftol': 1e-20,
                                      'gtol': 1e-13, 'maxfun': 20000, 'maxls': 100})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass

    # Phase 7: Fine local search
    if time.time() < deadline - 50:
        h, c_val = local_pair_search(h, n, deadline - 50, max_radius=30,
                                     steps=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

    # Phase 8: Perturbation restarts
    if time.time() < deadline - 40:
        h, c_val = perturbation_restart(h, n, best_c, bounds, deadline - 40,
                                         scale=0.01, n_trials=3)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

    # Phase 9: SLSQP polish
    if time.time() < deadline - 20:
        cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
        for alpha in [500000, 5000000]:
            if time.time() > deadline - 15:
                break
            try:
                res = minimize(slqp_smooth_objective, h, jac=slqp_smooth_gradient,
                             args=(n, alpha), method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 2000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 10: Final fine local search
    if time.time() < deadline - 5:
        h, c_val = local_pair_search(best_h, n, deadline - 5, max_radius=20,
                                     steps=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002])
        if c_val < best_c:
            best_c = c_val
            best_h = h

    # Final verification
    best_h = project_feasible(best_h, n)
    final_c = compute_c5(best_h, n)

    return best_h, final_c, n
