def subgrad_step(h, n, step_size, num_active=5):
    c = compute_c5_vec(h, n)
    max_val = c.max()
    # Get top num_active shifts
    top_idx = np.argsort(-c)[:num_active]
    # Weight by how close to max
    weights = c[top_idx] - (max_val - 0.01)  # or just uniform
    weights = np.maximum(weights, 0.01)
    weights /= weights.sum()
    
    # Compute gradient as weighted combination
    dx = 2.0 / n
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        k = idx - (n - 1)
        wm = w * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    
    return h - step_size * grad, c[top_idx]

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_grad_single(h, n, idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def grad_smooth_c5(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        g = get_grad_single(h, n, idx)
        grad += w[idx] * g
    return grad

def smooth_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def fista_optimize(h, n, alpha_start, alpha_end, deadline, step_scale=1.0):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    base_step = step_scale * dx * 50.0
    y = h.copy()
    t = 1.0
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 500.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        step = base_step / (1.0 + 0.1 * alpha / alpha_end)
        grad = grad_smooth_c5(y, n, alpha)
        z = y - step * grad
        h_new = project_feasible(z, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        if np.any(np.abs(y) > 10.0):
            y = h_new.copy()
            t = 1.0
        h = h_new
        t = t_new
        if it % 5 == 0 or time.time() > deadline - 2:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if it > 50:
            y = h + 0.5 * (h - h)  # reset momentum every 50 to avoid divergence
            if it % 50 == 0:
                t = 1.0
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def subgrad_optimize(h, n, deadline, step_init=0.3, n_active=10):
    h = project_feasible(h).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = step_init
    n_iters = 0
    stagnant = 0
    while time.time() < deadline:
        n_iters += 1
        c = compute_c5_vec(h, n)
        max_val = c.max()
        top_idx = np.argsort(-c)[:n_active]
        weights = c[top_idx] - (max_val - 0.005)
        weights = np.maximum(weights, 0.05)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            grad += w * get_grad_single(h, n, idx)
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
            if step < 0.5:
                step *= 1.05
        else:
            h = h_new
            stagnant += 1
            step *= 0.9
            if stagnant > 100:
                step = step_init * 0.5
                stagnant = 0
    return best_h, best_c

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.05, n_active_max=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    consecutive_no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > n_active_max:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:n_active_max]]
        n_act = len(active)
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0
        c_vals = c_vec[active]
        total_ineq = n_act + 4 * n
        A_ub = np.zeros((total_ineq, n_var))
        b_ub = np.zeros(total_ineq)
        row = 0
        for j in range(n_act):
            g = get_grad_single(h, n, active[j])
            A_ub[row, 0:n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = -c_vals[j]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = rho
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = rho
            row += 1
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(None, None)] * n + [(0.0, 4.0)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                delta = res.x[0:n]
                h_new = project_feasible(h + delta, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    consecutive_no_improve = 0
                    if rho < 0.1:
                        rho = min(rho * 1.2, 0.15)
                else:
                    consecutive_no_improve += 1
                    rho *= 0.5
                    if rho < 1e-10:
                        break
                    if consecutive_no_improve > 30:
                        break
            else:
                rho *= 0.5
                if rho < 1e-10:
                    break
        except Exception:
            rho *= 0.5
            if rho < 1e-10:
                break
    return best_h, best_c

def fine_local_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            # Gradient-based: find where to add/remove mass
            grad = np.zeros(n)
            for idx in active:
                grad += get_grad_single(h, n, idx)
            grad /= len(active)
            # Try moving mass from high-grad to low-grad positions
            n_try = min(n, 200)
            top_pos = np.argsort(grad)[:n_try]
            top_neg = np.argsort(-grad)[:n_try]
            for i_val in top_pos[:50]:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                for j_val in top_neg[:50]:
                    j = int(j_val)
                    if i == j:
                        continue
                    # Move mass from j to i (increase h[i], decrease h[j])
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Move mass from i to j
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return best_h, best_c

def pairwise_random_search(h, n, deadline, n_trials=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        amt = rng.uniform(0.001, 0.01)
        # Try both directions
        for sign in [1, -1]:
            di = sign * amt
            dj = -sign * amt
            if h[i] + di < -1e-12 or h[i] + di > 1.0 + 1e-12:
                continue
            if h[j] + dj < -1e-12 or h[j] + dj > 1.0 + 1e-12:
                continue
            h[i] += di
            h[j] += dj
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
            else:
                h[i] -= di
                h[j] -= dj
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
