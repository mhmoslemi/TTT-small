```python
import numpy as np
import time
from scipy.optimize import minimize, linprog

FFT_SIDE = -1
FFT_ORDER = -1
EXACT_LIMIT = 700


def project_target(h, n, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    for _ in range(100):
        s = float(h.sum())
        if s <= 1e-15:
            if target <= 0.0:
                break
            add = min(target / float(n), 1.0)
            h += add
            continue

        diff = target - s
        if abs(diff) <= 1e-13:
            break

        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot <= 1e-15:
                break
            h += (diff / tot) * h

    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    for _ in range(12):
        if abs(diff) <= 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            i = int(np.argmax(mask))
            h[i] += diff
            h[i] = min(1.0, max(0.0, h[i]))
        else:
            h += diff / float(n)
            h = np.clip(h, 0.0, 1.0)
        diff = target - float(h.sum())

    return h


def project_feasible(h, n):
    return project_target(h, n, float(n) / 2.0)


def eval_c5(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * (2.0 / float(n)))


def choose_fft_mode():
    global FFT_SIDE, FFT_ORDER
    if FFT_SIDE >= 0:
        return

    m = 64
    x = 0.5 + 0.25 * np.sin(np.linspace(0.0, 3.0, m)) + 0.10 * np.cos(np.linspace(0.0, 5.0, m))
    x = np.clip(x, 0.0, 1.0)
    exact = np.correlate(x, 1.0 - x, mode="full")

    N = 1
    while N < 2 * m - 1:
        N <<= 1

    Fx = np.fft.rfft(x, N)
    Fy = np.fft.rfft(1.0 - x, N)
    z1 = np.fft.irfft(Fx * np.conj(Fy), N)
    z2 = np.fft.irfft(Fy * np.conj(Fx), N)

    best_err = 1e100
    best_side = -1
    best_order = -1

    for side, z in ((0, z1), (1, z2)):
        a0 = np.concatenate((z[N - m + 1:N], z[0:m]))
        a1 = np.concatenate((z[0:m], z[N - m + 1:N]))
        cands = (a0, a1, a0[::-1], a1[::-1])
        for order, a in enumerate(cands):
            err = float(np.linalg.norm(np.real(a) - exact))
            if err < best_err:
                best_err = err
                best_side = side
                best_order = order

    if best_err < 1e-4:
        FFT_SIDE = best_side
        FFT_ORDER = best_order
    else:
        FFT_SIDE = -1
        FFT_ORDER = -1


def c_vec(h, n):
    h = np.asarray(h, dtype=float)
    if n <= EXACT_LIMIT or FFT_SIDE < 0:
        return np.correlate(h, 1.0 - h, mode="full")

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    try:
        Fh = np.fft.rfft(h, N)
        Fone = np.fft.rfft(1.0 - h, N)
        if FFT_SIDE == 0:
            z = np.fft.irfft(Fh * np.conj(Fone), N)
        elif FFT_SIDE == 1:
            z = np.fft.irfft(Fone * np.conj(Fh), N)
        else:
            return np.correlate(h, 1.0 - h, mode="full")

        o = FFT_ORDER
        if o == 0:
            out = np.concatenate((z[N - n + 1:N], z[0:n]))
        elif o == 1:
            out = np.concatenate((z[0:n], z[N - n + 1:N]))
        elif o == 2:
            out = np.concatenate((z[N - n + 1:N], z[0:n]))[::-1]
        else:
            out = np.concatenate((z[0:n], z[N - n + 1:N]))[::-1]

        out = np.real(out)
        if not np.all(np.isfinite(out)) or out.min() < -0.1:
            return np.correlate(h, 1.0 - h, mode="full")
        if out.min() < 0.0:
            out = np.where(out > 0.0, out, 0.0)
        return out
    except Exception:
        return np.correlate(h, 1.0 - h, mode="full")


def fast_cmax(h, n):
    return float(c_vec(h, n).max())


def fast_c5(h, n):
    return fast_cmax(h, n) * (2.0 / float(n))


def consider(h, n, best_h, best_c, best_n):
    if h is None:
        return best_h, best_c, best_n
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n <= 0:
        return best_h, best_c, best_n
    h = project_feasible(h, n)
    c = eval_c5(h, n)
    if c < best_c - 1e-14:
        return h.copy(), c, n
    return best_h, best_c, best_n


def consider_rolls(h, n, best_h, best_c, best_n):
    if h is None or n <= 1:
        return best_h, best_c, best_n
    shifts = (0, 1, -1, n // 8, -n // 8, n // 4, -n // 4)
    for s in shifts:
        hh = np.roll(h, int(s))
        best_h, best_c, best_n = consider(hh, n, best_h, best_c, best_n)
    return best_h, best_c, best_n


def make_alphas(n):
    base = max(100.0, float(n) / 3.0)
    out = []
    for mult in (1, 5, 20, 80, 320):
        a = base * float(mult)
        if a < 1e7:
            out.append(float(a))
    if not out:
        out = [1000.0]
    return out


def add_lag_grad(h, n, idx, grad, weight):
    if weight == 0.0:
        return
    k = int(idx) - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += weight * (1.0 - h[k:n])
            grad[k:n] += weight * (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += weight * (1.0 - h[0:nv])
            grad[0:nv] += weight * (-h[kk:n])


def sg_obj(x, n, alpha, mu, kcap):
    c = c_vec(x, n)
    if not np.all(np.isfinite(c)):
        return 1e100
    m = float(c.max())
    a = alpha * (c - m)
    s = float(np.sum(np.exp(a)))
    val = m + np.log(s) / alpha
    if mu > 0.0:
        e = float(x.sum() - n / 2.0)
        val += mu * e * e
    return float(val)


def sg_grad(x, n, alpha, mu, kcap):
    c = c_vec(x, n)
    m = float(c.max())
    a = alpha * (c - m)
    w = np.exp(a)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(n)
    w = w / s

    km = int(kcap)
    if km <= 0:
        km = 24 if n > 1500 else 60
    if n > 1500:
        km = max(km, 16)
    elif n > 800:
        km = max(km, 32)

    wm = float(w.max())
    thr = max(1e-14, wm * 1e-8)
    idxs = np.nonzero(w > thr)[0]
    if idxs.size == 0:
        idxs = np.array([int(np.argmax(w))])
    elif idxs.size > km:
        idxs = idxs[np.argsort(-w[idxs])[:km]]

    ww = w[idxs]
    t = float(ww.sum())
    if t <= 0.0:
        idxs = np.array([int(np.argmax(w))])
        ww = np.array([float(w[int(np.argmax(w))])])
        t = float(ww.sum())
    if t > 0.0:
        ww = ww / t

    g = np.zeros(n)
    for wi, idx in zip(ww, idxs):
        add_lag_grad(x, n, int(idx), g, float(wi))

    if mu > 0.0:
        e = float(x.sum() - n / 2.0)
        g += 2.0 * mu * e
    return g


def run_global_lbfgs(h, n, deadline, alphas=None, mu=0.01):
    h = project_feasible(h, n)
    cur_h = h.copy()
    best_h = h.copy()
    cur_c = fast_c5(cur_h, n)
    best_c = cur_c

    bounds = [(0.0, 1.0) for _ in range(n)]
    kcap = 24 if n > 1500 else 60
    if alphas is None:
        alphas = make_alphas(n)

    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 25.0:
            break

        if n > 2000:
            maxit = int(max(80, min(900, rem * 2.0)))
        elif n > 1000:
            maxit = int(max(120, min(1600, rem * 2.5)))
        else:
            maxit = int(max(150, min(3000, rem * 4.0)))

        try:
            res = minimize(
                sg_obj,
                cur_h,
                jac=sg_grad,
                args=(n, alpha, mu, kcap),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxit,
                    "ftol": 1e-18,
                    "gtol": 1e-12,
                    "maxfun": maxit * 25,
                    "maxls": 50,
                    "maxcor": 50,
                },
            )
            h_new = project_feasible(res.x, n)
            c_new = fast_c5(h_new, n)

            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new.copy()

            if c_new < best_c + 0.02:
                cur_h = h_new
                cur_c = c_new
            else:
                cur_h = best_h
                cur_c = best_c
        except Exception:
            pass

    return best_h, best_c


def sum_fun(x, n):
    return float(x.sum() - n / 2.0)


def sum_jac(x, n):
    return np.ones(n)


def run_slqp(h, n, deadline, alpha=10000.0, kcap=32):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = fast_c5(h, n)
    rem = deadline - time.time()
    if rem < 12.0:
        return best_h, best_c

    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [{"type": "eq", "fun": sum_fun, "jac": sum_jac, "args": (n,)}]
    maxit = int(max(50, min(700, rem * 4.0)))

    try:
        res = minimize(
            sg_obj,
            h,
            jac=sg_grad,
            args=(n, alpha, 0.0, kcap),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxit, "ftol": 1e-18},
        )
        h_new = project_feasible(res.x, n)
        c_new = fast_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
    except Exception:
        pass

    return best_h, best_c


def run_lp_phase(h, n, deadline, active_limit=None):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_fast = fast_cmax(h, n)

    if active_limit is None:
        active_limit = 55 if n > 1500 else 110

    no_improve = 0
    solve_count = 0
    max_solves = 220 if n <= 1000 else 110

    while time.time() < deadline and no_improve < 14 and solve_count < max_solves:
        cvec = c_vec(h, n)
        if not np.all(np.isfinite(cvec)):
            break
        cmax = float(cvec.max())
        if cmax <= 0.0:
            break

        active = np.argsort(-cvec)
        m = min(int(active_limit), len(active), 2 * n - 1)
        if m <= 0:
            break
        active = active[:m]

        G = np.zeros((m, n))
        b = np.empty(m)
        for r, idx in enumerate(active):
            g = G[r]
            add_lag_grad(h, n, int(idx), g, 1.0)
            b[r] = float(cvec[int(idx)] - np.dot(g, h))

        A = np.empty((m, n + 1))
        A[:, :n] = G
        A[:, n] = -1.0

        cobj = np.zeros(n + 1)
        cobj[n] = 1.0

        Aeq = np.ones(n + 1)
        Aeq[n] = 0.0
        beq = float(n) / 2.0

        t_hi = max(cmax * 1.5 + 10.0, 1.0)
        bounds = [(0.0, 1.0) for _ in range(n)] + [(0.0, t_hi)]

        solve_count += 1
        res = None
        try:
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b,
                A_eq=Aeq,
                b_eq=beq,
                bounds=bounds,
                method="HiGHS",
                options={"disp": False},
            )
        except Exception:
            try:
                res = linprog(
                    cobj,
                    A_ub=A,
                    b_ub=b,
                    A_eq=Aeq,
                    b_eq=beq,
                    bounds=bounds,
                    options={"disp": False},
                )
            except Exception:
                res = None

        if res is None or not getattr(res, "success", False) or res.x is None:
            no_improve += 2
            active_limit = max(25, active_limit // 2)
            continue

        if not np.all(np.isfinite(res.x)):
            no_improve += 2
            continue

        x_lp = project_feasible(res.x[:n], n)
        d = x_lp - h
        sc = float(np.linalg.norm(d))
        if sc < 1e-7:
            no_improve += 2
            continue

        improved = False
        for step in (1.0, 0.6, 0.35, 0.20, 0.10, 0.05, 0.02):
            cand = project_feasible(h + step * d, n)
            cm = fast_cmax(cand, n)
            if cm < best_fast - 1e-9:
                if cm < best_fast:
                    best_fast = cm
                    best_h = cand.copy()
                h = cand
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1

    return best_h, best_fast


def make_binary(h, n):
    h = project_feasible(h, n)
    b = np.zeros(n)
    m = n // 2
    if m <= 0:
        return b
    order = np.argsort(-h)
    b[order[:m]] = 1.0
    return b


def binary_propose(b, n, idx, rs):
    k = int(idx) - (n - 1)
    if k == 0:
        return None

    if k > 0:
        nv = n - k
        if nv <= 0:
            return None
        bad = (b[k:n] > 0.5) & (b[0:nv] < 0.5)
    else:
        kk = -k
        nv = n - kk
        if nv <= 0:
            return None
        bad = (b[0:nv] > 0.5) & (b[kk:n] < 0.5)

    if np.any(bad):
        nz = np.flatnonzero(bad)
        ii = int(nz[int(rs.randint(len(nz)))])
        if k > 0:
            return (k + ii, ii)
        return (ii, kk + ii)

    ones = np.flatnonzerob if False else np.flatnonzero(b > 0.5)
    zeros = np.flatnonzero(b < 0.5)
    if ones.size == 0 or zeros.size == 0:
        return None
    a = int(ones[int(rs.randint(len(ones)))])
    z = int(zeros[int(rs.randint(len(zeros)))])
    return (a, z)


def binary_local_search(h, n, deadline, rs, max_iter=12000):
    if n % 2 != 0:
        return h, fast_c5(h, n)

    b = make_binary(h, n)
    cvec = c_vec(b, n)
    cur_max = float(cvec.max())
    best_b = b.copy()
    best_max = cur_max

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if cur_max <= 0.0:
            break

        active = np.argsort(-cvec)
        top = min(20, len(active))
        if it % 5 == 0 or top == 0:
            idx = int(active[0])
        else:
            idx = int(active[int(rs.randint(top))])

        prop = binary_propose(b, n, idx, rs)
        if prop is None:
            continue
        a, j = prop
        if not (0 <= a < n and 0 <= j < n and a != j):
            continue
        if b[a] < 0.5 or b[j] > 0.5:
            continue

        b[a] = 0.0
        b[j] = 1.0
        cvec_new = c_vec(b, n)
        cm_new = float(cvec_new.max())

        if cm_new < cur_max - 1e-12 or (cm_new == cur_max and float(rs.random()) < 0.05):
            cur_max = cm_new
            cvec = cvec_new
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_b = b.copy()
        else:
            b[a] = 1.0
            b[j] = 0.0

    return best_b, best_max * (2.0 / float(n))


def upsample_h(h, ns, nt):
    h = np.asarray(h, dtype=float)
    if ns == nt:
        return h.copy()
    xs = np.arange(ns, dtype=float) * (2.0 / float(ns))
    xt = np.arange(nt, dtype=float) * (2.0 / float(nt))
    return np.interp(xt, xs, h)


def downsample_pair(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n % 2 != 0:
        return None
    return 0.5 * (h[0::2] + h[1::2])


def half_complement_h(h, n, rev=False):
    if n % 2 != 0:
        return None
    m = n // 2
    a = project_target(np.asarray(h, dtype=float)[:m].copy(), m, float(m) / 2.0)
    if rev:
        b = 1.0 - a[::-1]
    else:
        b = 1.0 - a
    hh = np.concatenate((a, b))
    return project_feasible(hh, n)


def run_active_line(h, n, deadline, top=8, max_rounds=24):
    h = project_feasible(h, n)
    cur_h = h.copy()
    cur_c = fast_c5(cur_h, n)
    best_h = h.copy()
    best_c = cur_c

    etas = (0.06, 0.03, 0.018, 0.010, 0.005, 0.0025, 0.0012, 0.0006)

    for _ in range(max_rounds):
        if time.time() > deadline:
            break

        cvec = c_vec(cur_h, n)
        if not np.all(np.isfinite(cvec)):
            break

        idxs = np.argsort(-cvec)
        top = min(int(top), len(idxs), 2 * n - 1)
        if top <= 0:
            break
        idxs = idxs[:top]

        g = np.zeros(n)
        inv = 1.0 / float(top)
        for idx in idxs:
            add_lag_grad(cur_h, n, int(idx), g, inv)

        d = -g
        d -= np.mean(d)
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-13:
            break
        d /= nrm

        improved = False
        for eta in etas:
            cand = project_feasible(cur_h + eta * d, n)
            cf = fast_c5(cand, n)
            if cf < cur_c - 1e-12:
                cur_h = cand
                cur_c = cf
                best_h = cand.copy()
                best_c = cur_c
                improved = True
                break

        if not improved:
            break

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    global FFT_SIDE, FFT_ORDER

    t0 = time.time()
    deadline = t0 + max(60.0, float(budget_s)) - 8.0
    choose_fft_mode()

    rs = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = int(h0.size)
    except Exception:
        n0 = 500
        x = np.linspace(0.0, 1.0, n0)
        h0 = np.clip(0.5 + 0.35 * np.sin(np.pi * x) + 0.08 * np.sin(3.0 * np.pi * x), 0.0, 1.0)

    if n0 <= 0:
        n0 = 500
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)

    best_h, best_c, best_n = consider(h0, n0, None, 1e100, 0)
    cur_h = best_h.copy()
    cur_n = best_n

    hs = 0.5 * (h0 + h0[::-1])
    best_h, best_c, best_n = consider(hs, n0, best_h, best_c, best_n)
    best_h, best_c, best_n = consider(1.0 - h0, n0, best_h, best_c, best_n)

    hd = downsample_pair(h0)
    if hd is not None:
        best_h, best_c, best_n = consider(hd, len(hd), best_h, best_c, best_n)

    half1 = half_complement_h(h0, n0, False)
    if half1 is not None:
        best_h, best_c, best_n = consider(half1, n0, best_h, best_c, best_n)
    half2 = half_complement_h(h0, n0, True)
    if half2 is not None:
        best_h, best_c, best_n = consider(half2, n0, best_h, best_c, best_n)

    if time.time() + 20.0 < deadline:
        best_h, best_c, best_n = consider_rolls(h0, n0, best_h, best_c, best_n)

    if best_h is not None:
        cur_h = best_h.copy()
        cur_n = best_n

    # Polish the given/current resolution first.
    if time.time() + 40.0 < deadline:
        end1 = min(time.time() + 180.0, deadline - 520.0)
        hh, cc = run_global_lbfgs(cur_h, cur_n, end1, mu=0.01)
        cur_h = project_feasible(hh, cur_n)
        best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

    if time.time() + 20.0 < deadline:
        best_h, best_c, best_n = consider_rolls(cur_h, cur_n, best_h, best_c, best_n)

    if time.time() + 30.0 < deadline:
        end2 = min(time.time() + 80.0, deadline - 380.0)
        hh, _ = run_lp_phase(cur_h, cur_n, end2)
        cur_h = project_feasible(hh, cur_n)
        best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

    if time.time() + 30.0 < deadline and cur_n % 2 == 0 and cur_n <= 2048:
        end3 = min(time.time() + 60.0, deadline - 300.0)
        hh, _ = binary_local_search(cur_h, cur_n, end3, rs)
        cur_h = project_feasible(hh, cur_n)
        best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

    # Raise resolution to 1024 if that helps.
    n_hi1 = 1024
    use_hi1 = (
        (time.time() + 120.0 < deadline)
        and (cur_n <= n_hi1)
        and (FFT_SIDE >= 0 or n_hi1 <= 1500)
    )

    if use_hi1:
        if cur_n == n_hi1:
            hi1 = cur_h.copy()
        else:
            hi1 = upsample_h(cur_h, cur_n, n_hi1)
        best_h, best_c, best_n = consider(hi1, n_hi1, best_h, best_c, best_n)
        cur_h = hi1.copy()
        cur_n = n_hi1

        end4 = min(time.time() + 220.0, deadline - 230.0)
        hh, _ = run_global_lbfgs(cur_h, cur_n, end4, mu=0.01)
        cur_h = project_feasible(hh, cur_n)
        best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

        if time.time() + 40.0 < deadline:
            best_h, best_c, best_n = consider_rolls(cur_h, cur_n, best_h, best_c, best_n)

        if time.time() + 40.0 < deadline:
            end4b = min(time.time() + 80.0, deadline - 150.0)
            hh, _ = run_lp_phase(cur_h, cur_n, end4b)
            cur_h = project_feasible(hh, cur_n)
            best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

        if time.time() + 35.0 < deadline and cur_n <= 1000:
            end4c = min(time.time() + 35.0, deadline - 120.0)
            hh, _ = run_slqp(cur_h, cur_n, end4c, alpha=80000.0, kcap=64)
            cur_h = project_feasible(hh, cur_n)
            best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

        if time.time() + 40.0 < deadline and cur_n % 2 == 0:
            end4d = min(time.time() + 45.0, deadline - 100.0)
            hh, _ = binary_local_search(cur_h, cur_n, end4d, rs)
            cur_h = project_feasible(hh, cur_n)
            best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

    # Raise resolution to 2048 if the FFT route is available and time remains.
    n_hi2 = 2048
    use_hi2 = (
        (time.time() + 90.0 < deadline)
        and (cur_n < n_hi2)
        and (FFT_SIDE >= 0)
    )

    if use_hi2:
        hi2 = upsample_h(cur_h, cur_n, n_hi2)
        best_h, best_c, best_n = consider(hi2, n_hi2, best_h, best_c, best_n)
        cur_h = hi2.copy()
        cur_n = n_hi2

        end5 = min(time.time() + 150.0, deadline - 100.0)
        hh, _ = run_global_lbfgs(cur_h, cur_n, end5, mu=0.01)
        cur_h = project_feasible(hh, cur_n)
        best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

        if time.time() + 30.0 < deadline:
            best_h, best_c, best_n = consider_rolls(cur_h, cur_n, best_h, best_c, best_n)

        if time.time() + 30.0 < deadline:
            end5b = min(time.time() + 40.0, deadline - 50.0)
            hh, _ = run_lp_phase(cur_h, cur_n, end5b, active_limit=45)
            cur_h = project_feasible(hh, cur_n)
            best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

        if time.time() + 35.0 < deadline and cur_n % 2 == 0:
            end5c = min(time.time() + 35.0, deadline - 35.0)
            hh, _ = binary_local_search(cur_h, cur_n, end5c, rs)
            cur_h = project_feasible(hh, cur_n)
            best_h, best_c, best_n = consider(cur_h, cur_n, best_h, best_c, best_n)

    # Final small improvement routines on the best/fast current point.
    if best_h is not None:
        if time.time() + 25.0 < deadline:
            hh, _ = run_active_line(best_h, best_n, min(time.time() + 50.0, deadline - 20.0), top=10, max_rounds=28)
            best_h, best_c, best_n = consider(hh, best_n, best_h, best_c, best_n)

        if time.time() + 20.0 < deadline:
            hh, _ = run_lp_phase(best_h, best_n, min(time.time() + 25.0, deadline - 12.0), active_limit=45)
            best_h, best_c, best_n = consider(hh, best_n, best_h, best_c, best_n)

        if time.time() + 15.0 < deadline and best_n <= 1000:
            hh, _ = run_slqp(best_h, best_n, min(time.time() + 15.0, deadline - 6.0), alpha=200000.0, kcap=64)
            best_h, best_c, best_n = consider(hh, best_n, best_h, best_c, best_n)

    if best_h is None:
        best_h = project_feasible(h0, n0)
        best_c = eval_c5(best_h, n0)
        best_n = n0

    best_h = project_feasible(best_h, best_n)
    best_c = eval_c5(best_h, best_n)

    return best_h, float(best_c), int(best_n)
```
