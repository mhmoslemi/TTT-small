```python
    c = np.correlate(h, 1-h, mode='full') * dx  # length 2n-1

        if w[idx] < 1e-15:
            continue
        wi = w[idx]

            grad += wi * (1 - 2*h)
        elif ell > 0:

            grad[e:] += wi * (1 - h[:n-e])
            # -h[j+e] for j=0..n-1-e
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _ones_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _ones_j, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_smooth_max(h, n, alpha=100):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-12:
            continue
        ell = k - (n - 1)
        if ell == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif ell > 0:
            grad[:n-ell] += wi * (1.0 - h[ell:])
            grad[ell:] += -wi * h[:n-ell]
        else:
            e = -ell
            grad[e:] += wi * (1.0 - h[:n-e])
            grad[:n-e] += -wi * h[e:]
    return grad

def proj_grad_step(h, n, grad, lr):
    h_new = h - lr * grad
    return project_feasible(h_new, n)

def proj_grad_descent(h, n, deadline, alpha=100, lr_init=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    
    while time.time() < deadline:
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        grad = grad_smooth_max(h, n, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:

        # Backtracking line search
        accepted = False
        for _ in range(30):
            h_try = proj_grad_step(h, n, grad, lr)
            c_try = compute_c5(h_try, n)
            if c_try < c - 1e-15:

    return best_h, best_c

def grad_guided_search(h, n, deadline, delta=0.002, top_k=100):

    best_h = h.copy()
    best_c = compute_c5(h, n)

        # Sort by gradient: we want to move from high grad to low grad

        # Source: high gradient indices (end of sorted array)
        # Destination: low gradient indices (start of sorted array)
        sources = order[n-top_k:]
        dests = order[:top_k]

            c = compute_c5(h, n)

                if c < cur_c:
                    improved = True
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i] -= sign * delta
                h[j] += sign * delta
        
        T *= 0.95
        if not improved and T < 0.0001:
            break
    
    return best_h, best_c
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_obj_grad(x, n, alpha):
    """Gradient of smooth objective."""
    dx = 2.0 / n
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    # d/dh_j of sum_k w_k * c_k
    # = sum_k w_k * (1-h_{j+ell_k}) - h_{j-ell_k} (for lag ell_k = k-(n-1))
    # This is the same as grad_smooth_max above
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-12:
            continue
        ell = k - (n - 1)
        if ell == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif ell > 0:
            grad[:n-ell] += wi * (1.0 - h[ell:])
            grad[ell:] += -wi * h[:n-ell]
        else:
            e = -ell
            grad[e:] += wi * (1.0 - h[:n-e])
            grad[:n-e] += -wi * h[e:]
    return grad
