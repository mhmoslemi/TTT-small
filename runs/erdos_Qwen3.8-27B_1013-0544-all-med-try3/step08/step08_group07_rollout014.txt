        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h, c = subgrad_polish(best_h.copy(), n, p6 - 3, max_iter=10000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 7: Surgical + pair refine (remaining)
    if time.time() < dl - 30:
        h, c = surgical_refine(best_h.copy(), n, dl - 30)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    if time.time() < dl - 5:
        best_h, best_c = pair_refine(best_h.copy(), n, dl - 2,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    # Final SLSQP micro-polish
    if time.time() < dl - 2:
        h = slsqp_step(best_h.copy(), n, 100000000, 300)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
