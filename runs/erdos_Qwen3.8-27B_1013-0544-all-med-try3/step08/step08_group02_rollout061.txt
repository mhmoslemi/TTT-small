```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    sh -= sh.max()
    val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active_idx:
        if w[idx - active_idx[0]] < 1e-15:
            continue
        k = idx - (n - 1)
        wm = w[idx - active_idx[0]] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h0, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        rem = deadline - time.time()
        if rem < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = max(200, maxiters[i])
        else:
            maxit = max(500, int(rem * 4))
        
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 50,
                                   'iprint': -1})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def run_active_focus(h0, n, deadline, tol_frac=0.002, max_active=12, alphas=None):
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000]
    
    for rnd in range(15):
        rem = deadline - time.time()
        if rem < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, tol_frac * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > max_active:
            active = np.argsort(-c_vec[active])[:max_active]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

                                       'gtol': 1e-18, 'maxfun': maxit * 10,
                                       'maxls': 40, 'maxcor': 40,
                                       'iprint': -1})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new <= best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def coord_descent(h0, n, deadline, n_sweeps=50):
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    target = n / 2.0
    
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for ss in step_sizes:
        if time.time() > deadline:
            break
        improved_any = True
        while improved_any:
            if time.time() > deadline:
                break
            improved_any = False
            for j in range(n):
                if time.time() > deadline - 5:
                    break
                for direction in [1, -1]:
                    delta = direction * ss
                    new_val = best_h[j] + delta
                    if new_val < 0 or new_val > 1:
                        continue
                    # Check sum constraint approximately
                    new_sum = best_h.sum() + delta
                    if abs(new_sum - target) > 0.02:
                        continue
                    best_h[j] = new_val
                    h_proj = project_feasible(best_h, n)
                    c_new = compute_c5(h_proj, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_proj.copy()
                        improved_any = True
                    else:
                        best_h[j] = new_val - delta
    
    return best_h, best_c

def pair_search(h0, n, deadline, radius=25, steps=None):
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for ss in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    for direction in [1, -1]:
                        d = direction * ss
                        if 0 <= best_h[i] + d <= 1 and 0 <= best_h[j] - d <= 1:
                            best_h[i] += d
                            best_h[j] -= d
                            h_proj = project_feasible(best_h, n)
                            c_new = compute_c5(h_proj, n)
                            if c_new < best_c - 1e-16:
                                best_c = c_new
                                best_h = h_proj.copy()
                                improved = True
                            else:
                                best_h[i] -= d
                                best_h[j] += d
    
    return best_h, best_c

def run_slqp_polish(h0, n, deadline, alpha=1000000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    try:
        res = minimize(lse_obj, h0, jac=lse_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h0.copy(), n), compute_c5(h0, n)

def make_structured_start(n, variant='hilbert'):
    h = np.zeros(n)
    target = n / 2.0
    if variant == 'hilbert':
        # Hilbert curve-like 1D pattern: use van der Corput sequence
        def vdc(k, base=2):
            d = 1.0
            r = 0.0
            while k > 0:
                d /= base
                r += d * (k % base)
                k //= base
            return r
        idx = np.array([int(vdc(k) * n) % n for k in range(n)])
        idx = np.sort(idx)
        h[idx[:int(target)]] = 1.0
        remaining = target - h.sum()
        if abs(remaining) > 1e-10:
            h[int(target)] += remaining
    elif variant == 'blocks':
        # Alternating blocks of decreasing size
        pos = 0
        block_size = n // 8
        val = 1.0
        while pos < n:
            bs = min(block_size, n - pos)
            h[pos:pos+bs] = val
            pos += bs
            val = 1.0 - val
            block_size = max(1, block_size // 2)
        h = project_feasible(h, n)
        return h
    elif variant == 'golden':
        phi = (1 + np.sqrt(5)) / 2
        positions = np.sort((np.arange(n) * (1.0/phi)) % n).astype(int)
        h[positions[:int(target)]] = 1.0
        remaining = target - h.sum()
        if abs(remaining) > 1e-10:
            h[positions[int(target)]] += remaining
    else:
        h[:] = 0.5
    return project_feasible(h, n)

def upsample(h, n_old, n_new):

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
```
