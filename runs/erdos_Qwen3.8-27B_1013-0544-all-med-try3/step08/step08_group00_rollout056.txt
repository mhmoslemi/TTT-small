def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def get_subgradient(h, n, c_vec, max_active):

def smooth_lbfgs(h, n, deadline, alpha, mu, maxiter):
    def smooth_obj(x, nn, al, muu):

        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al
        else:
            val = m + np.log(np.sum(np.exp(sh))) / al
        val += muu * (x.sum() - nn / 2.0) ** 2
        return float(val)
    def smooth_grad(x, nn, al, muu):
        c = compute_c5_vec(x, nn)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)

            k = mm - (nn - 1)

            if k >= 0:

                grad[0:nv] += wm * (-x[kk:nn])
        grad += 2.0 * muu * (x.sum() - nn / 2.0)
        return grad

    try:

        h5, c5 = smooth_lbfgs(best_h, n, t5, alpha=20000, mu=5000.0, maxiter=3000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Another SQP round (80s)
    t6 = min(t0 + 550, deadline - 350)
    if time.time() < t6 - 10:
        h6, c6 = sqp_optimize(best_h, n, t6, max_iter=500, rho_init=0.02, max_active=100)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Binary swap search (50s)
    t7 = min(t0 + 600, deadline - 300)
    if time.time() < t7 - 5:
        h7, c7 = binary_swap_search(best_h, n, t7, max_swaps_per_pass=5000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: SLSQP polish (60s)
    t8 = min(t0 + 660, deadline - 250)
    if time.time() < t8 - 10:
        h8, c8 = slqp_polish(best_h, n, t8, alpha=5000000, maxiter=2000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Try higher resolution n=800 (120s)
    n_high = 800
    if time.time() < deadline - 180:
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)

        t9 = min(time.time() + 50, deadline - 120)
        if time.time() < t9 - 5:
            h_high, c_high = sqp_optimize(h_high, n_high, t9, max_iter=200, rho_init=0.04, max_active=60)

        t10 = min(time.time() + 40, deadline - 60)
        if time.time() < t10 - 5:
            h_high, c_high = fine_local_search(h_high, n_high, t10)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n = n_high

    # Phase 10: Final SQP + local search (remaining)
    if time.time() < deadline - 20:
        tf1 = min(time.time() + 35, deadline - 12)
        hf1, cf1 = sqp_optimize(best_h, len(best_h), tf1, max_iter=200, rho_init=0.015, max_active=80)
        if cf1 < best_c:
            best_c = cf1
            best_h = hf1

    if time.time() < deadline - 8:
        hf2, cf2 = fine_local_search(best_h, len(best_h), deadline - 5)
        if cf2 < best_c:
            best_c = cf2
            best_h = hf2

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
