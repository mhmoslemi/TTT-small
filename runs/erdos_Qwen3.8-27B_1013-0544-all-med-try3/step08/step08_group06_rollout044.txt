import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_arr(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        val = np.log(np.sum(np.exp(sh)))
    return float(m + val / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new, n) <= compute_c5(h, n):
            return h_new
    except Exception:
        pass
    return h.copy()

def targeted_search(h, n, deadline, steps):
    """Local search targeting worst shifts."""
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for d in steps:
        stall = 0
        max_stall = 30
        while time.time() < deadline and stall < max_stall:
            c_arr = np.correlate(h, 1 - h, mode='full')
            max_c = c_arr.max()
            # Active shifts: those within 5% of max
            tol = 0.05 * max_c + 0.1
            active = np.where(c_arr >= max_c - tol)[0]
            
            pass_improved = False
            for shift_idx in active:

                    return h, best_c
                s = abs(int(shift_idx - (n - 1)))
                if s == 0 or s >= n:
                    continue

                        return h, best_c
                    j = i + s
                    # Try both directions
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:

                        else:

                    if h[j] >= d and h[i] <= 1.0 - d:

                            pass_improved = True
                        else:

    return h, best_c
