```python
def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)
```

The algorithm:

2. Take a step in the negative gradient direction
3. Project onto the feasible set {0 ≤ h ≤ 1, sum(h) = n/2}

The projection onto {0 ≤ h ≤ 1, sum(h) = n/2} can be done efficiently:
- It's the Euclidean projection onto the intersection of a hypercube and a hyperplane
- This can be solved by bisection on a Lagrange multiplier

```python
def project_onto_simplex_box(v, n):
    """Project v onto {0 ≤ h ≤ 1, sum(h) = n/2}."""
    target = n / 2.0
    # Find λ such that sum(clip(v - λ, 0, 1)) = target
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    # Fix any numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute the difference
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff / s * room
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff / s * h
    return np.clip(h, 0, 1)
```

```python
def projected_gradient(h, n, alpha, deadline, max_iter=10000):
    """Projected gradient on smoothed min-max objective."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        g = compute_all_g(h, n)
        m = g.max()
        
        # Compute gradient of smoothed objective
        shifted = (g - m) / alpha
        sm = shifted.max()
        weights = np.exp(shifted - sm)
        weights /= weights.sum()
        
        # Gradient: ∂obj/∂h[i] = (1/|S|)*sum_{k in S} ∂g(k)/∂h[i] + (1/alpha)*sum_k weights[k]*∂g(k)/∂h[i]
        # For the first term, if we approximate argmax by the weighted argmax, we get:
        # ∂obj/∂h[i] ≈ sum_k (weights[k]/alpha + indicator(k in S)/|S|) * ∂g(k)/∂h[i]
        # With large alpha, the first term dominates, so:
        # ∂obj/∂h[i] ≈ (1/alpha) * sum_k weights[k] * ∂g(k)/∂h[i]
        
        # For efficiency, compute: grad[i] = (1/alpha) * sum_k weights[k] * grad_at(h, n, i)[k]
        # grad_at(h, n, i)[k] = dx * (1 - h[i+k-(n-1)] - h[i+n-1-k])
        
        grad = np.zeros(n)
        for i in range(n):
            gi = grad_at(h, n, i)
            grad[i] = np.sum(weights * gi) / alpha
        
        # Add the "m" gradient (for the active set)
        max_ks = np.where(np.abs(g - m) < m * 1e-10 + 1e-14)[0]
        if len(max_ks) > 0:
            for i in range(n):
                gi = grad_at(h, n, i)
                grad[i] += gi[max_ks].mean()
        
        # Step
        h_new = project_onto_simplex_box(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        if c_new < compute_c5(h, n) + lr * 0.5 * np.sum(grad**2):
            h = h_new
            lr = min(lr * 1.1, 10.0)
        else:
            lr *= 0.5
            if lr < 1e-12:
                break
    
    return best_h, best_c
```
