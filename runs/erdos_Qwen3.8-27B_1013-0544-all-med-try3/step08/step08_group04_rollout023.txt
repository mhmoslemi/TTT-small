                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] = max(0.0, h[i] - d)
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        if alpha < 1e6:
            shifted = alpha * (c - m)
            mx = shifted.max()
            logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
            return float(m + logsum / alpha)
        else:
            # For very large alpha, use direct max with softening
            c_soft = c + 1e-10
            return float(c.max())
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        top_idx = np.argsort(c)[-3:]
        m = c[top_idx[0]]
        if alpha < 1e6:
            shifted = alpha * (c - m)
            weights = np.exp(shifted[top_idx] - shifted[top_idx].max())
            weights = weights / weights.sum()
            grad = np.zeros(n)
            for w, k in zip(weights, top_idx):
                grad += w * grad_corr_at(x, int(k), n)
            return grad
        else:
            # Subgradient of max: gradient of worst component
            k = top_idx[0]
            return grad_corr_at(x, int(k), n)
    
    def sum_con(x):
        return x.sum() - n / 2.0
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj, h, args=(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h, n)

def optimize_resolution(h_start, n, deadline, use_full_pair):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    # Phase A: Subgradient
    t_sub_end = min(deadline, time.time() + max(20, (deadline - time.time()) * 0.30))
    if time.time() < t_sub_end - 5:
        h, bc = subgradient_optimize(h, n, t_sub_end, top_m=5, max_iter=30000)
        if bc < best_c:
            best_c = bc
            h = h
    
    # Phase B: Pair search
    t_pair_end = min(deadline, time.time() + max(15, (deadline - time.time()) * 0.35))
    if time.time() < t_pair_end - 3:
        if use_full_pair and n <= 160:
            h, bc = pair_search_fast(h, n, t_pair_end,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            h, bc = near_pair_search(h, n, t_pair_end,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=max(30, n // 5))
        if bc < best_c:
            best_c = bc
            h = h
    
    # Phase C: SLSQP with increasing alpha
    if time.time() < deadline - 3:
        for a in [1000, 10000, 100000]:
            if time.time() > deadline - 3:
                break
            ht = slsqp_optimize(h, n, a, maxiter=1500)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                h = ht
    
    # Phase D: Final micro pair search
    if time.time() < deadline - 1:
        if use_full_pair and n <= 160:
            h, bc = pair_search_fast(h, n, deadline - 0.5,
                                     [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        else:
            h, bc = near_pair_search(h, n, deadline - 0.5,
                                     [0.0005, 0.0002, 0.0001, 0.00005],
                                     radius=max(20, n // 8))
        if bc < best_c:
            best_c = bc
            h = h
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    total_deadline = t0 + budget_s - 3
    
    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    
    # === Phase 1: Optimize initial resolution with subgradient + pair search ===
    alloc1 = (total_deadline - t0) * 0.30
    d1_end = t0 + alloc1
    h_opt, c_opt = optimize_resolution(h_init, n_init, d1_end, use_full_pair=(n_init <= 160))
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    
    # === Phase 2: Try different resolutions ===
    # Interpolate to other grid sizes
    x_init = np.linspace(0, 2, n_init)
    resolutions_tried = {n_init}
    
    n_candidates = []
    for n_try in [200, 300, 350, 400, 150, 120]:
        if n_try == n_init:
            continue
        if n_try < 40 or n_try > 500:
            continue
        n_candidates.append(n_try)
    
    # Sort by proximity to initial (to interpolate well)
    n_candidates.sort(key=lambda x: abs(x - n_init))
    
    remaining = total_deadline - time.time()
    n_res = len(n_candidates)
    if n_res > 0:
        per_res = remaining * 0.35 / n_res
        for n_try in n_candidates:
            if time.time() > total_deadline - 60:
                break
            if per_res < 10:
                per_res = 10
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_opt)
            h_interp = project_feasible(h_interp, n_try)
            
            alloc = min(per_res, total_deadline - time.time() - 50)
            if alloc < 15:
                break
            
            h_r, c_r = optimize_resolution(h_interp, n_try, time.time() + alloc,
                                           use_full_pair=(n_try <= 150))
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    
    # === Phase 3: Multi-restart perturbation on best resolution ===
    n_best = len(best_h)
    remaining = total_deadline - time.time()
    if remaining > 80:
        n_restarts = max(2, int(remaining * 0.20 / 40))
        for r in range(n_restarts):
            if time.time() > total_deadline - 40:
                break
            scales = [0.02, 0.04, 0.06, 0.08, 0.1]
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            
            alloc = min(50, (total_deadline - time.time() - 20) * 0.25)
            if alloc < 20:
                break
            
            h_r, c_r = optimize_resolution(hp, n_best, time.time() + alloc,
                                           use_full_pair=(n_best <= 150))
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    
    # === Phase 4: Aggressive final optimization on best ===
    n_best = len(best_h)
    remaining = total_deadline - time.time()
    if remaining > 30:
        alloc = min(remaining - 5, 120)
        # Subgradient again with different parameters
        h_r, c_r = subgradient_optimize(best_h, n_best, time.time() + alloc * 0.5,
                                        top_m=8, max_iter=20000)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
        
        # Pair search with many small steps
        if time.time() < total_deadline - 5:
            if n_best <= 160:
                h_r, c_r = pair_search_fast(best_h, n_best, total_deadline - 3,
                                            [0.005, 0.002, 0.001, 0.0005, 0.0002,
                                             0.0001, 0.00005, 0.00002, 0.00001])
            else:
                h_r, c_r = near_pair_search(best_h, n_best, total_deadline - 3,
                                            [0.005, 0.002, 0.001, 0.0005, 0.0002,
                                             0.0001, 0.00005, 0.00002],
                                            radius=max(40, n_best // 5))
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    
    # === Phase 5: Final SLSQP polish ===
    if time.time() < total_deadline - 2 and len(best_h) > 0:
        n_best = len(best_h)
        ht = slsqp_optimize(best_h, n_best, 500000, maxiter=3000, ftol=1e-15)
        c = compute_c5(ht, n_best)
        if c < best_c:
            best_c = c
            best_h = ht
        # One more micro pair search
        if time.time() < total_deadline - 0.5:
            if n_best <= 160:
                best_h, best_c = pair_search_fast(best_h, n_best, total_deadline - 0.2,
                                                  [0.0001, 0.00005, 0.00002, 0.00001])
            else:
                best_h, best_c = near_pair_search(best_h, n_best, total_deadline - 0.2,
                                                  [0.0001, 0.00005, 0.00002],
                                                  radius=30)
    
    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
