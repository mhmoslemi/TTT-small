        if n_c < 50:

        if time.time() > deadline - 10:
            break
        x_f = np.linspace(0, 2, n)
        x_c = np.linspace(0, 2, n_c)
        h_c = np.interp(x_c, x_f, h)
        h_c = project_feasible(h_c, n_c)
        t_alloc = min(15.0, (deadline - time.time()) * 0.2)
        if t_alloc > 5:
            h_c, c_c = smooth_max_lbfgs(h_c, n_c, time.time() + t_alloc, 10000, 0.0)
        h_f = np.interp(x_f, x_c, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_orig = 400

    # Choose working resolution
    N = 1000
    x_orig = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, N)
    h = np.interp(x_new, x_orig, h_init)
    h = project_feasible(h, N)
    best_c = compute_c5(h, N)
    best_h = h.copy()

    # Phase 1: Quick L-BFGS warmup at N=1000
    t1 = min(t0 + 60, deadline - 600)
    h, best_c = smooth_max_lbfgs(h, N, t1, alpha=5000, mu=10.0, maxiter=3000)
    best_h = h.copy()

    # Phase 2: Active-set refinement
    t2 = deadline - 500
    if time.time() < t2:
        h, best_c = run_active_set(h, N, t2, n_iter=40, top_k=5)
        best_h = h.copy()

    # Phase 3: Smooth max with higher alpha
    t3 = deadline - 380
    if time.time() < t3:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > t3:
                break
            t_sub = min(30.0, (t3 - time.time()) * 0.4)
            if t_sub < 5:
                break
            h, c = smooth_max_lbfgs(h, N, time.time() + t_sub, alpha, 0.0)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 4: More active-set iterations
    t4 = deadline - 280
    if time.time() < t4:
        h, best_c = run_active_set(best_h, N, t4, n_iter=30, top_k=4)
        best_h = h.copy()

    # Phase 5: Random perturbations and re-optimization (multiple attempts)
    t5 = deadline - 150
    n_perturb = 6
    per_perturb = max(15, (t5 - time.time()) / n_perturb)
    for pi in range(n_perturb):
        if time.time() > t5 - 2:
            break
        t_pi = time.time() + per_perturb
        # Different perturbation strategies
        if pi % 3 == 0:
            # Gaussian noise
            h_pert = best_h + rng.normal(0, 0.01, N)
        elif pi % 3 == 1:
            # Structured perturbation (shift-like)
            h_pert = best_h.copy()
            shift_amt = rng.randint(3, 15)
            h_pert[shift_amt:] += best_h[:-shift_amt] * 0.05
            h_pert[:-shift_amt] -= best_h[shift_amt:] * 0.05
        else:
            # Frequency-domain perturbation
            hf = np.fft.rfft(best_h)
            hf[1:-1] *= (1.0 + 0.02 * rng.randn(len(hf[1:-1])))
            h_pert = np.fft.irfft(hf, N)
        h_pert = project_feasible(h_pert, N)
        h_new, c_new = smooth_max_lbfgs(h_pert, N, t_pi, alpha=100000, mu=0.0, maxiter=3000)
        if c_new < best_c - 1e-12:
            # Refine with active set
            remaining = t5 - time.time()
            if remaining > 10:
                h_new, c_new = run_active_set(h_new, N, time.time() + remaining * 0.6, n_iter=15, top_k=4)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

    # Phase 6: Multi-resolution escape
    t6 = deadline - 100
    if time.time() < t6:
        h, c = multi_resolution(best_h, N, t6)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 7: Final active-set + smooth max polish
    t7 = deadline - 40
    if time.time() < t7:
        h, best_c = run_active_set(best_h, N, t7, n_iter=25, top_k=3)
        best_h = h.copy()
        if time.time() < deadline - 10:
            h, c = smooth_max_lbfgs(best_h, N, deadline - 5, 5000000, 0.0, maxiter=3000)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 8: Try upscale to N=2000
    N2 = 2000
    if time.time() < deadline - 30:
        x1 = np.linspace(0, 2, N)
        x2 = np.linspace(0, 2, N2)
        h2 = np.interp(x2, x1, best_h)
        h2 = project_feasible(h2, N2)
        c2 = compute_c5(h2, N2)
        if c2 < best_c:
            h2, c2 = smooth_max_lbfgs(h2, N2, deadline - 5, 50000, 0.0, maxiter=3000)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                N = N2

    # Phase 9: Final active set on best
    if time.time() < deadline - 3:
        h, c = run_active_set(best_h, len(best_h), deadline - 2, n_iter=10, top_k=3)
        if c < best_c:
            best_c = c
            best_h = h

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
