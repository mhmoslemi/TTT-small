def _obj_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _con_fun(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx_v = sh.max()
    if mx_v > 400:
        return float(m + (mx_v + np.log(np.sum(np.exp(sh - mx_v)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def lbfgs_opt(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 5.0), (1000, 10.0), (5000, 20.0), (20000, 50.0), (100000, 200.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_obj, h, jac=_obj_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-17,
                                   'gtol': 1e-14, 'maxfun': max_iters * 8,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_opt(h, n, deadline, alpha=200000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_fine_search(h, n, deadline):
    """Gradient-guided pair moves with adaptive step sizes."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        if time.time() > deadline - 4:
            break
        for alpha in [500, 5000, 50000]:
            stall = 0
            while stall < 4 and time.time() < deadline - 3:
                improved = False
                g = grad_smooth_max(h, n, alpha)
                order = np.argsort(g)[::-1]
                n_try = min(n, 200)
                for ia in range(n_try):
                    if time.time() > deadline - 2:
                        break
                    i = order[ia]
                    for ib in range(ia + 1, min(ia + 30, n_try)):
                        j = order[ib]
                        d = g[i] - g[j]
                        if d <= 0:
                            continue
                        move = step * min(1.0, abs(d) * 8.0)
                        if move < 1e-10:
                            continue
                        if h[i] - move >= 0.0 and h[j] + move <= 1.0:
                            h[i] -= move; h[j] += move
                            c = compute_c5(h, n)
                            if c < best_c - 1e-16:
                                best_c = c; best_h = h.copy(); improved = True
                            else:
                                h[i] += move; h[j] -= move
                if not improved:
                    stall += 1
                else:
                    stall = 0
    return best_h, best_c

def random_pair_search(h, n, deadline, base_step=0.003):
    """Random pair moves - effective for fine polishing."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [base_step, base_step * 0.4, base_step * 0.15, base_step * 0.05]
    
    for step in steps:
        if time.time() > deadline - 4:
            break
        np.random.seed(int(time.time() * 1000) % 100000)
        while time.time() < deadline - 3:
            improved = False
            for _ in range(8000):
                if time.time() > deadline - 2:
                    break
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                d = np.random.choice([-step, step])
                if h[i] + d >= 0.0 and h[i] + d <= 1.0 and h[j] - d >= 0.0 and h[j] - d <= 1.0:
                    h[i] += d; h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c; best_h = h.copy(); improved = True
                    else:
                        h[i] -= d; h[j] += d
        h = best_h
    return best_h, best_c

def full_pair_search(h, n, deadline, step):
    """Exhaustive pair search at a given step size."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    h = best_h.copy()
    
    stall = 0
    max_stall = 3
    while stall < max_stall and time.time() < deadline - 3:
        improved = False
        for i in range(n):
            if time.time() > deadline - 2:
                break
            for j in range(i + 1, n):
                if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                    h[i] -= step; h[j] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c; best_h = h.copy(); improved = True
                    else:
                        h[i] += step; h[j] -= step
                if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                    h[i] += step; h[j] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c; best_h = h.copy(); improved = True
                    else:
                        h[i] -= step; h[j] += step
        if not improved:
            stall += 1
        else:
            stall = 0
            h = best_h.copy()
    return best_h, best_c

def restart_optimize(h, n, deadline, rng, n_restarts=5):
    """Perturb and re-optimize to escape local minima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alloc_per = max(10, (deadline - time.time() - 5) / max(1, n_restarts))
    
    for r in range(n_restarts):
        if time.time() > deadline - 5:
            break
        h_pert = h + rng.randn(n) * 0.01
        h_pert = project_feasible(h_pert, n)
        t_sub = time.time() + alloc_per * 0.5
        h_r, c_r = lbfgs_opt(h_pert, n, t_sub, max_iters=100)
        if time.time() < deadline - 10:
            t_sub2 = min(time.time() + alloc_per * 0.4, deadline - 8)
            h_r, c_r = gradient_fine_search(h_r, n, t_sub2)
        if c_r < best_c - 1e-10:
            best_c = c_r
            best_h = h_r.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    
    # Phase 1: L-BFGS-B from parent (fast convergence)
    t1 = min(t0 + 80, deadline - 500)
    best_h, best_c = lbfgs_opt(best_h, best_n, t1, max_iters=250)
    
    # Phase 2: SLSQP polish
    if time.time() < deadline - 400:
        t2 = min(time.time() + 60, deadline - 400)
        h2, c2 = slsqp_opt(best_h, best_n, t2, alpha=500000, maxiter=800)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Gradient fine search (main workhorse)
    if time.time() < deadline - 200:
        t3 = min(time.time() + 250, deadline - 200)
        best_h, best_c = gradient_fine_search(best_h, best_n, t3)
    
    # Phase 4: Exhaustive pair search at fine steps
    if time.time() < deadline - 150:
        for step in [0.001, 0.0005, 0.0002, 0.0001]:
            if time.time() > deadline - 100:
                break
            t4 = min(time.time() + 40, deadline - 100)
            best_h, best_c = full_pair_search(best_h, best_n, t4, step)
    
    # Phase 5: Random pair search for additional polish
    if time.time() < deadline - 80:
        t5 = min(time.time() + 80, deadline - 80)
        best_h, best_c = random_pair_search(best_h, best_n, t5, base_step=0.002)
    
    # Phase 6: Try higher resolution (n=400)
    if time.time() < deadline - 120:
        n_try = 400
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = max(30, min(100, deadline - time.time() - 60))
        if alloc > 25:
            t6 = time.time() + alloc * 0.4
            h_t, c_t = lbfgs_opt(h_interp, n_try, t6, max_iters=200)
            if time.time() < deadline - 30:
                t6b = min(time.time() + alloc * 0.35, deadline - 25)
                h_t, c_t = gradient_fine_search(h_t, n_try, t6b)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
    
    # Phase 7: Try n=500
    if time.time() < deadline - 100 and best_n != 500:
        n_try = 500
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = max(25, min(80, deadline - time.time() - 50))
        if alloc > 20:
            t7 = time.time() + alloc * 0.5
            h_t, c_t = lbfgs_opt(h_interp, n_try, t7, max_iters=150)
            if time.time() < deadline - 20:
                t7b = min(time.time() + alloc * 0.35, deadline - 15)
                h_t, c_t = gradient_fine_search(h_t, n_try, t7b)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
    
    # Phase 8: Restart strategy (escape local optima)
    if time.time() < deadline - 60:
        t8 = min(time.time() + 60, deadline - 30)
        best_h, best_c = restart_optimize(best_h, best_n, t8, rng, n_restarts=3)
    
    # Phase 9: Final fine polish
    if time.time() < deadline - 5:
        nf = len(best_h)
        t9 = deadline - 3
        best_h, best_c = gradient_fine_search(best_h, nf, t9)
    
    # Phase 10: Ultra-fine random search at the very end
    if time.time() < deadline - 2:
        nf = len(best_h)
        best_h, best_c = random_pair_search(best_h, nf, deadline - 1, base_step=0.001)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
