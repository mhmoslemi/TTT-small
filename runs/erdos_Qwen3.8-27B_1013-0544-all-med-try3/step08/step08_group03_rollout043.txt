import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad_h = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad_h[0:nv] += wm * (1.0 - h[k:n])
            grad_h[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad_h[kk:n] += wm * (1.0 - h[0:nv])
            grad_h[0:nv] += wm * (-h[kk:n])
    return grad_h

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def adam_optimize(h, n, deadline, alpha, lr=0.005, max_iter=30000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    no_improve = 0
    lr_cur = lr
    while time.time() < deadline and t < max_iter:
        t += 1
        g = grad_smooth_max(h, n, alpha)
        m_buf = beta1 * m_buf + (1 - beta1) * g
        v_buf = beta2 * v_buf + (1 - beta2) * (g * g)
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)
        h_new = h - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve % 300 == 0 and no_improve > 0:
            lr_cur *= 0.8
            if lr_cur < 1e-11:
                break
    return best_h, best_c

def adam_logit_optimize(h, n, deadline, alpha, lr=0.01, mu=100.0, max_iter=30000):
    """Adam optimization in logit space to avoid projection overhead."""
    h = project_feasible(h, n)
    h = np.clip(h, 1e-10, 1 - 1e-10)
    z = np.log(h / (1 - h))
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    no_improve = 0
    lr_cur = lr
    while time.time() < deadline and t < max_iter:
        t += 1
        h_cur = 1.0 / (1.0 + np.exp(-np.clip(z, -50, 50)))
        c = compute_c5_vec(h_cur, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad_h = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-14:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad_h[0:nv] += wm * (1.0 - h_cur[k:n])
                grad_h[k:n] += wm * (-h_cur[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad_h[kk:n] += wm * (1.0 - h_cur[0:nv])
                grad_h[0:nv] += wm * (-h_cur[kk:n])
        grad_z = grad_h * h_cur * (1 - h_cur)
        constraint = h_cur.sum() - n / 2.0
        grad_z += 2.0 * mu * constraint * h_cur * (1 - h_cur)
        m_buf = beta1 * m_buf + (1 - beta1) * grad_z
        v_buf = beta2 * v_buf + (1 - beta2) * (grad_z * grad_z)
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)
        z = z - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -30, 30)
        h_new = 1.0 / (1.0 + np.exp(-z))
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve % 400 == 0 and no_improve > 0:
            lr_cur *= 0.85
            if lr_cur < 1e-10:
                break
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, jac=grad_smooth_max,
                      args=(n, alpha), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 60,
                               'gtol': 1e-12, 'maxfun': maxiter * 20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, deltas=None, radius=30):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if improved:
                        break
            if not improved:
                break
    return h, best_c

def spectral_init(n, rng):
    phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    spectrum = np.exp(1j * phases)
    spectrum[0] = 0
    h = np.fft.irfft(spectrum, n=n)
    h += 0.5
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def random_bernstein_init(n, rng):
    """Create a structured initialization with blocks of high/low values."""
    n_blocks = rng.randint(20, 60)
    block_size = max(1, n // n_blocks)
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = min(n, start + block_size)
        val = rng.choice([0.1, 0.2, 0.8, 0.9, rng.uniform(0, 1)])
        h[start:end] = val
    return project_feasible(h, n)
