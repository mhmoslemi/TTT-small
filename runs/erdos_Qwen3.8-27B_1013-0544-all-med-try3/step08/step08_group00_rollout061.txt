            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_lse(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only compute for significant weights
    thresh = 1e-12
    for mm in range(len(w)):
        if w[mm] < thresh:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.08, max_active=50):
    """SQP with more active constraints and adaptive trust region."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0

        for j in active:

            A_ub[row, 0:n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = -c_vec[j]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = rho
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = rho

        A_eq[0, 0:n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(None, None)] * n + [(0.0, 3.0)]

def gradient_local_search(h, n, deadline, max_iter=5000):
    """Gradient descent with line search on the smooth max."""
    best_h = h.copy()

    alpha = 10000.0
    mu = 100.0
    
    for it in range(max_iter):

            break

        # projected gradient step
        step_size = 0.5 / (np.max(np.abs(grad)) + 1e-10)
        h_new = h - step_size * grad
        # Line search
        for ls in range(10):
            h_try = project_feasible(h_new, n)
            c_try = compute_c5(h_try, n)

                best_h = h_try.copy()
                h = h_try
                break

            step_size *= 0.5

def swap_local_search(h, n, deadline, max_iter=10000):
    """Swap-based local search: move mass from one position to another."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    # Precompute which shifts are active
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    rng = np.random.default_rng(12345)
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            
            # Compute gradient from active constraints
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad_single(h, n, idx))
            if len(active) > 0:
                grad /= len(active)
            
            # Find best pair to swap
            # We want to increase h where gradient is negative (reduces overlap)
            # and decrease where gradient is positive
            # Try top candidates
            n_try = min(200, n)
            candidates = np.argsort(np.abs(grad))[:n_try]
            
            for i_val in candidates:
                i = int(i_val)
                # Try transferring to/from random positions
                for _ in range(20):
                    if time.time() > deadline:
                        return best_h, best_c
                    j = int(rng.integers(n))
                    if i == j:
                        continue
                    # Move mass from i to j
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                    # Move mass from j to i
                    if h[j] - step >= -1e-12 and h[i] + step <= 1.0 + 1e-12:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[j] += step
                            h[i] -= step

def make_structured_init(n, rng):
    """Create a structured initialization."""
    h = np.zeros(n)
    # Try different patterns
    pattern = rng.integers(4)
    if pattern == 0:
        # Block: first half 1s, second half 0s
        h[:n//2] = 1.0
    elif pattern == 1:
        # Alternating pairs
        h[0::4] = 1.0
        h[1::4] = 1.0
    elif pattern == 2:
        # Three blocks
        h[:n//3] = 1.0
        h[2*n//3:] = 1.0
    else:
        # Random
        h[rng.choice(n, n//2, replace=False)] = 1.0
    
    return project_feasible(h, n)

            else:
                rho *= 0.5
                if rho < 1e-11:
                    break
        except Exception:
            rho *= 0.5
            if rho < 1e-11:
                break
    return best_h, best_c

def gradient_descent_h(h, n, deadline, max_iter=8000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 5000.0
    mu = 50.0
    step_scale = 1.0
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_lse_grad(h, n, alpha, mu)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step_size = step_scale / (gnorm + 1e-10)
        improved_this_iter = False
        for ls in range(12):
            h_try = project_feasible(h - step_size * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                improved_this_iter = True
                stagnant = 0
                if step_size < 0.5:
                    step_scale = min(step_scale * 1.2, 2.0)
                break
            step_size *= 0.5
        if not improved_this_iter:
            stagnant += 1
            step_scale *= 0.7
            if stagnant > 30 and step_scale < 0.01:
                break
    return best_h, best_c

def swap_local_search(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        max_inner = 20000
        inner_count = 0
        while improved and inner_count < max_inner and time.time() < deadline:
            improved = False
            inner_count += 1
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad_single(h, n, idx))
            if len(active) > 0:
                grad /= len(active)
            top_k = min(100, n)
            candidates_h = np.argsort(grad)[:top_k]
            candidates_l = np.argsort(-grad)[:top_k]
            for i_val in candidates_h:
                i = int(i_val)
                for j_val in candidates_l:
                    j = int(j_val)
                    if i == j:
                        continue
                    if time.time() > deadline:
                        return best_h, best_c
                    if h[j] - step >= -1e-12 and h[i] + step <= 1.0 + 1e-12:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[j] += step
                            h[i] -= step
                        inner_count += 1
    return best_h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start, n)
    best_c = compute_c5(h, n)
    t1 = min(time.time() + 18, deadline - 80)
    if t1 > time.time() + 3:
        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(smooth_lse, h, jac=smooth_lse_grad,
                          args=(n, 400, 50.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 15000})
            h_try = project_feasible(res.x, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                h = h_try
        except Exception:
            pass
    t2 = min(time.time() + 40, deadline - 50)
    if t2 > time.time() + 5:
        h, best_c = sqp_optimize(h, n, t2, max_iter=400, rho_init=0.10)
    t3 = min(time.time() + 12, deadline - 20)
    if t3 > time.time() + 3:
        h, c3 = gradient_descent_h(h, n, t3, max_iter=5000)
        if c3 < best_c:
            best_c = c3
    t4 = min(time.time() + 15, deadline - 8)
    if t4 > time.time() + 3:
        rng_lc = np.random.default_rng(99)
        h, c4 = swap_local_search(h, n, t4, rng_lc)
        if c4 < best_c:
            best_c = c4
    t5 = min(time.time() + 20, deadline - 3)
    if t5 > time.time() + 5:
        h, c5 = sqp_optimize(h, n, t5, max_iter=200, rho_init=0.04)
        if c5 < best_c:
            best_c = c5
    return h, best_c

def make_random_binary_init(n, rng):
    h = np.zeros(n)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1.0
    return project_feasible(h, n)

def make_periodic_init(n, rng):
    period = rng.integers(max(4, n // 20), max(5, n // 8))
    duty = rng.uniform(0.35, 0.65)
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        if phase < duty:
            h[i] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_deadline = t0 + budget_s - 3
    rng = np.random.default_rng(seed)
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(400, 0.5)
        n_parent = 400
    h_parent = project_feasible(h_parent, n_parent)
    best_c = compute_c5(h_parent, n_parent)
    best_h = h_parent.copy()
    best_n = n_parent
    scan_n_list = sorted(set([150, 200, 250, 300, 350, 400, 500, 600, 700, 800, 1000, 1200]))
    scan_deadline = t0 + 100
    candidates = {}
    x_old = np.linspace(0, 2, n_parent)
    for n_try in scan_n_list:
        if time.time() > scan_deadline - 8:
            break
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, h_parent)
        h_try = project_feasible(h_try, n_try)
        c_try = compute_c5(h_try, n_try)
        if time.time() < scan_deadline - 12:
            try:
                bounds_sc = [(0.0, 1.0)] * n_try
                res_sc = minimize(smooth_lse, h_try, jac=smooth_lse_grad,
                                args=(n_try, 300, 100.0), method='L-BFGS-B',
                                bounds=bounds_sc,
                                options={'maxiter': 400, 'ftol': 1e-17,
                                         'gtol': 1e-13})
                h_try = project_feasible(res_sc.x, n_try)
                c_try = compute_c5(h_try, n_try)
            except Exception:
                pass
        candidates[n_try] = (h_try, c_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
            best_n = n_try
    sorted_cands = sorted(candidates.items(), key=lambda kv: kv[1][1])
    top_ns = [n for n, _ in sorted_cands[:5]]
    deep_deadline = t0 + 900
    for n_deep in top_ns:
        if time.time() > deep_deadline - 60:
            break
        h_deep = candidates[n_deep][0].copy()
        phase_deadline = min(time.time() + 180, deep_deadline)
        h_deep, c_deep = optimize_at_n(h_deep, n_deep, phase_deadline)
        if c_deep < best_c:
            best_c = c_deep
            best_h = h_deep.copy()
            best_n = n_deep
    if time.time() < total_deadline - 40:
        n_cur = best_n
        alt_deadline = t0 + 1020
        for trial in range(4):
            if time.time() > alt_deadline:
                break
            if trial % 2 == 0:
                h_alt = make_random_binary_init(n_cur, rng)
            else:
                h_alt = make_periodic_init(n_cur, rng)
            t_alt = min(time.time() + 12, alt_deadline)
            if t_alt > time.time() + 3:
                try:
                    bounds_alt = [(0.0, 1.0)] * n_cur
                    res_alt = minimize(smooth_lse, h_alt, jac=smooth_lse_grad,
                                      args=(n_cur, 500, 100.0), method='L-BFGS-B',
                                      bounds=bounds_alt,
                                      options={'maxiter': 800, 'ftol': 1e-17})
                    h_alt = project_feasible(res_alt.x, n_cur)
                    c_alt = compute_c5(h_alt, n_cur)
                    if c_alt < best_c:
                        best_c = c_alt
                        best_h = h_alt.copy()
                except Exception:
                    pass
    if time.time() < total_deadline - 10:
        h_fin, c_fin = sqp_optimize(best_h, best_n, total_deadline - 2,
                                     max_iter=300, rho_init=0.03)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
