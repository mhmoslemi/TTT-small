import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj_fn(x, n, alpha, K, mu):
    c_all = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    if K < len(c_all):
        idx = np.argpartition(c_all, -K)[-K:]
    else:
        idx = np.arange(len(c_all))
    c_top = c_all[idx]
    c_max = c_top.max()
    shifted = np.clip(alpha * (c_top - c_max), -500, 500)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return val

def _grad_fn(x, n, alpha, K, mu):
    c_all = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    if K < len(c_all):
        idx = np.argpartition(c_all, -K)[-K:]
    else:
        idx = np.arange(len(c_all))
    c_top = c_all[idx]
    c_max = c_top.max()
    shifted = np.clip(alpha * (c_top - c_max), -500, 500)
    w = np.exp(shifted)
    ws = w.sum()
    w /= ws
    grad = np.zeros(n)
    for ii in range(len(idx)):
        s_val = idx[ii] - (n - 1)
        wi = w[ii]
        for j in range(n):
            g = 0.0
            a = j - s_val
            if 0 <= a < n:
                g += (1.0 - x[a])
            b = j + s_val
            if 0 <= b < n:
                g -= x[b]
            grad[j] += wi * g
    s_pen = x.sum() - n / 2.0
    grad += 2.0 * mu * s_pen
    return grad

def optimize_smooth(h, n, deadline, stages):
    h = h.copy()
    for alpha, K, mu, maxiter in stages:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_obj_fn, h, args=(n, alpha, K, mu),
                          jac=_grad_fn, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter, 'gtol': 1e-12, 'maxls': 30})
            h = project_feasible(res.x, n)
        except Exception:
            pass
    return h

def subgradient_refine(h, n, deadline, n_max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    for it in range(n_max_iter):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        c_all = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = c_all.max()
        active = c_all > c_max - 1e-4
        na = active.sum()
        if na == 0:
            active = c_all > c_max - 1e-3
            na = active.sum()
        if na == 0:
            break
        
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            s_val = k - (n - 1)
            for j in range(n):
                g = 0.0
                a = j - s_val
                if 0 <= a < n:
                    g += (1.0 - h[a])
                b = j + s_val
                if 0 <= b < n:
                    g -= h[b]
                grad[j] += g
        grad /= na
        
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.9
            if step < 1e-8:
                break
        if it % 100 == 0:
            step = max(step, 0.005)
    return best_h, best_c
