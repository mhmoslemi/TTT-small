```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

# Soft-max objective and gradient
def softmax_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def softmax_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad
```

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            h += diff * room / room.sum()
        else:
            h += diff * h / h.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def shift_gradients_all(h, n, active_indices, n_full):
    """Compute sum of w[i]*grad for active shifts. active_indices are indices in the full correlation array."""
    dx = 2.0 / n
    weights = active_indices[:, 0]
    idxs = active_indices[:, 1].astype(int)
    grad = np.zeros(n)
    for ii in range(len(idxs)):
        k = idxs[ii] - (n - 1)
        wdx = weights[ii] * dx
        if abs(wdx) < 1e-16:
            continue
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[k:] += wdx * (1.0 - h[:nv])
                grad[:nv] += -wdx * h[k:]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[:nv] += wdx * (1.0 - h[kk:])
                grad[kk:] += -wdx * h[:nv]
    return grad

def softmax_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 700:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def softmax_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    # Only keep significant weights
    active = np.where(w > 1e-14)[0]
    if len(active) == 0:
        return np.zeros(n)
    active = np.column_stack([w[active], active])
    return shift_gradients_all(h, n, active, len(c))

def linsrch_obj(h, n, active_idx, mu):
    """Objective focusing on a specific set of active shifts (average, not max)."""
    c = compute_c5_vec(h, n)
    vals = c[active_idx]
    val = np.mean(vals)
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def linsrch_grad(h, n, active_idx, mu):
    c = compute_c5_vec(h, n)
    # Average gradient over active shifts
    grad = np.zeros(n)
    for idx in active_idx:
        k = idx - (n - 1)

            grad[kk:] += -dx * h[:nv]
    grad /= len(active_idx)

        improved = True

            improved = False
            for _ in range(200):  # limit iterations per step size
                if time.time() > deadline - 1:
                    return best_h, best_c
                # Pick a random pair
                i = np.random.randint(0, n)
                j = np.random.randint(0, n)
                if i == j:
                    continue
                if abs(i - j) > max_radius * steps.index(d) + max_radius:
                    # Allow wider pairs for smaller steps
                    if np.random.rand() < 0.3:
                        j = np.random.randint(0, n)

                # Try both directions
                for direction in [1, -1]:
