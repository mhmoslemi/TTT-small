    grad *= dx
    C5 = f.max()

    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_lse(h, n, alpha):
    """Compute log-sum-exp smoothed max and its gradient (vectorized)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    f = corr * dx
    m = f.max()
    sh = np.clip(alpha * (f - m), -500.0, 500.0)

    sum_exp = exp_sh.sum()
    w = exp_sh / sum_exp

    s_vec = np.arange(-(n-1), n)  # shape (2n-1,)
    j_idx = np.arange(n)[:, None]  # shape (n, 1)
    
    diffs = j_idx - s_vec[None, :]  # j - s, shape (n, 2n-1)
    sums_ = j_idx + s_vec[None, :]  # j + s, shape (n, 2n-1)

    """Adam optimization with log-sum-exp smoothing."""
    if alpha_starts is None:
        alpha_starts = [100, 500, 2000, 10000, 50000, 200000]
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    
    for alpha in alpha_starts:
        if time.time() > deadline:
            break
        # Reset moments for new alpha
        m_t = np.zeros(n)
        v_t = np.zeros(n)
        
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            t += 1
            C5s, grad, f = grad_lse(h, n, alpha)
            
            # Adam update
            m_t = 0.9 * m_t + 0.1 * grad
            v_t = 0.999 * v_t + 0.001 * grad**2
            m_hat = m_t / (1 - 0.9**t)
            v_hat = v_t / (1 - 0.999**t)
            lr_t = lr * np.sqrt(1 - 0.999**t) / (1 - 0.9**t)

            c = compute_c5(h, n)
            if c < best_c:

                best_h = h.copy()

            best_h = h.copy()

        j_idx = np.arange(n)[:, None]

            g2 = np.where(mask2, h[np.clip(sums_, 0, n-1)], 0.0)
            grad += g1 - g2

        lr_t = lr_start * (lr_end / lr_start) ** frac

def update_corr_after_perturb(corr, h, j1, j2, eps, n):
    """Update correlation after h[j1]+=eps, h[j2]-=eps."""
    delta = np.zeros(2*n-1)
    n1 = n - 1  # offset for shift indexing
    
    for j, sign in [(j1, eps), (j2, -eps)]:
        s_vec = np.arange(-(n-1), n)  # shift for each m
        # term1: sign * (1-h[j-s]) where 0<=j-s<n
        diffs = j - s_vec  # shape (2n-1,)
        mask1 = (diffs >= 0) & (diffs < n)
        delta[mask1] += sign * (1 - h[diffs[mask1]])
        # term2: -sign * h[j+s] where 0<=j+s<n
        sums_ = j + s_vec
        mask2 = (sums_ >= 0) & (sums_ < n)
        delta[mask2] -= sign * h[sums_[mask2]]
    
    return corr + delta

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_lse(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    f = corr * dx
    m = f.max()
    sh = np.clip(alpha * (f - m), -500.0, 0.0)
    exp_sh = np.exp(sh)
    sum_exp = exp_sh.sum()
    w = exp_sh / sum_exp
    C5s = m + np.log(sum_exp) / alpha

    n1 = n - 1
    s_vec = np.arange(-n1, n)  # (2n-1,)
    j_idx = np.arange(n)[:, None]  # (n, 1)
    
    diffs = j_idx - s_vec[None, :]  # (n, 2n-1): j - s
    sums_ = j_idx + s_vec[None, :]  # (n, 2n-1): j + s
    
    mask1 = (diffs >= 0) & (diffs < n)
    mask2 = (sums_ >= 0) & (sums_ < n)
    
    d_clip = np.clip(diffs, 0, n-1)
    s_clip = np.clip(sums_, 0, n-1)
    
    term1 = np.where(mask1, 1.0 - h[d_clip], 0.0)
    term2 = np.where(mask2, h[s_clip], 0.0)
    
    w_row = w[None, :]
    g1 = (w_row * term1).sum(axis=1)
    g2 = (w_row * term2).sum(axis=1)
    
    grad = dx * (g1 - g2)
    return C5s, grad, f, corr

def adam_phase(h, n, deadline, alphas=None, iters_per_alpha=200, lr_base=0.003):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        m_t = np.zeros(n)
        v_t = np.zeros(n)
        t = 0
        # Decrease lr for larger alpha
        lr = lr_base / np.sqrt(alpha / 50.0)
        
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            t += 1
            C5s, grad, f, corr = grad_lse(h, n, alpha)
            
            # Adam
            betay1 = 0.9
            betay2 = 0.999
            m_t = betay1 * m_t + (1-betay1) * grad
            v_t = betay2 * v_t + (1-betay2) * grad**2
            m_hat = m_t / (1 - betay1**t)
            v_hat = v_t / (1 - betay2**t)
            
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            # Scale step to not be too large
            max_step = np.abs(step).max()
            if max_step > 0.05:
                step *= 0.05 / max_step
            
            h_new = h - step
            h = project_feasible(h_new, n)
            
            c = compute_c5(h, n)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

def subgrad_phase(h, n, deadline, n_iter=20000, topk=5, lr_start=0.002):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n1 = n - 1
    s_vec = np.arange(-n1, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1-h, mode='full')
        f = corr * dx
        c = f.max()
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        
        topk_idx = np.argsort(f)[::-1][:topk]
        
        j_idx = np.arange(n)[:, None]
        grad = np.zeros(n)
        
        for mk in topk_idx:
            s = mk - n1
            diffs = j_idx - s
            sums_ = j_idx + s
            mask1 = (diffs >= 0) & (diffs < n)
            mask2 = (sums_ >= 0) & (sums_ < n)
            d_clip = np.clip(diffs, 0, n-1)
            s_clip = np.clip(sums_, 0, n-1)
            g1 = np.where(mask1, 1.0 - h[d_clip], 0.0)
            g2 = np.where(mask2, h[s_clip], 0.0)
            grad += (g1 - g2).sum(axis=1)
        
        grad /= topk
        grad *= dx
        
        # Polya-Shorin step size
        lr_t = lr_start / (1.0 + 0.0001 * it)
        
        h = project_feasible(h - lr_t * grad, n)
    
    return best_h, best_c
