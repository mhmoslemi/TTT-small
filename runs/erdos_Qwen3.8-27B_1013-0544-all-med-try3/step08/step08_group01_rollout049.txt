def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

# Custom Adam optimizer
def adam_optimize(h, n, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, 
                  p=50, max_iter=100000):
    """Adam optimizer with p-norm approximation of max."""
    h = h.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    dx = 2.0 / n
    
    while t < max_iter and time.time() < deadline:
        t += 1
        
        # Compute correlation vector
        c = np.correlate(h, 1 - h, mode='full') * dx
        K = len(c)  # 2n-1
        
        # p-norm approximation of max
        if p > 100:
            c_max = np.abs(c).max()
            grad_obj = np.sign(c) * np.abs(c)**(p-1) / (K * c_max**(p-1)) if c_max > 0 else np.zeros_like(c)
        else:
            c_max = np.abs(c).max() + 1e-10
            c_norm = np.abs(c) / c_max
            c_pow = c_norm ** p
            grad_obj = np.sign(c) * (c_norm ** (p-1)) / (K * c_max)
        
        # Gradient w.r.t. h
        grad = np.zeros(n)
        for mm in range(2 * n - 1):
            gi = grad_obj[mm]
            if abs(gi) < 1e-15:
                continue
            k = mm - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += gi * (1.0 - h[k:n])
                grad[k:n] += gi * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += gi * (1.0 - h[0:nv])
                grad[0:nv] += gi * (-h[kk:n])
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_value(c, alpha):
    c_shifted = c - c.max()
    sh = alpha * c_shifted
    mx = sh.max()
    if mx > 500:
        return c.max() + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return c.max() + np.log(np.sum(np.exp(sh))) / alpha

def soft_max_grad_h(h, n, alpha):
    """Gradient of smooth max of correlation w.r.t. h."""
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    c_shifted = c - c.max()
    sh = alpha * c_shifted
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
            if k > 0:
                grad[k:n] += wm * (-h[0:nv])
            else:
                grad[0:nv] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def adam_smooth(h, n, deadline, alpha=5000, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on smooth max objective."""
    h = h.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    stuck_count = 0
    
    while time.time() < deadline:
        t += 1
        grad = soft_max_grad_h(h, n, alpha)
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * grad ** 2
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        h = h_new
        if stuck_count > 5000:
            lr *= 0.5
            stuck_count = 0
            if lr < 1e-6:
                break
    return best_h, best_c

def slqp_active_obj(x, n, active_idx):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    vals = c[active_idx]
    return float(vals.max())

def slqp_active_cons(x, n, active_idx, base_val):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    vals = c[active_idx]
    cons = []
    for i in range(1, len(active_idx)):
        cons.append({
            'type': 'eq',
            'fun': lambda x, i=i, ai=active_idx, aj=active_idx[0]: 
                float(np.correlate(x, 1-x, mode='full')[ai] - np.correlate(x, 1-x, mode='full')[aj]) * (2.0/len(x)),
            'jac': None
        })
    return cons

def slqp_smooth_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return soft_max_value(c, alpha)

def slqp_smooth_jac(x, n, alpha):
    return soft_max_grad_h(x, n, alpha)

def constraint_sum(x, n):
    return x.sum() - n / 2.0

def constraint_sum_jac(x, n):
    return np.ones(n)

def run_slqp_alpha(h, n, deadline, alpha=50000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_sum_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_smooth_obj, h, jac=slqp_smooth_jac,
                      args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'maxfun': maxiter * 2})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, radius=40, base_delta=0.001):
    """Local pair swap search."""
    best_c = compute_c5(h, n)
    deltas = [base_delta, base_delta * 0.5, base_delta * 0.25, base_delta * 0.1]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def gradient_descent_2point(h, n, deadline, active_k=8):
    """Use gradient to find best 2-point moves on active shifts."""
    best_c = compute_c5(h, n)
    base_step = 0.01
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()

        if len(active) > active_k:
            active = active[np.argsort(-c_vec[active])][:active_k]
        if len(active) == 0:
            break
        
        # Compute average gradient over active shifts
        grad = np.zeros(n)

            g = soft_max_grad_h(h, n, 100000)  # approximates max gradient
            break  # just use one for direction
        # Actually compute per-shift gradients

                if k > 0:
                    grad[k:n] += (-h[0:nv]) * dx
                else:
                    grad[0:nv] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        # Find best 2-point move
        g_abs = np.abs(grad)
        top_idx = np.argsort(-g_abs)[:30]
        found = False
        for ii in range(len(top_idx)):
            if time.time() > deadline:
                return h, best_c
            i = top_idx[ii]
            for jj in range(len(top_idx)):
                if ii == jj:
                    continue
                j = top_idx[jj]
                # Try moving mass from i to j (or j to i)
                for direction in [1, -1]:
                    for mult in [1.0, 0.5, 0.25]:
                        step = base_step * mult
                        if direction == 1:
                            new_i, new_j = h[i] - step, h[j] + step
                        else:
                            new_i, new_j = h[i] + step, h[j] - step
                        if 0 <= new_i <= 1 and 0 <= new_j <= 1:
                            h[i] = new_i
                            h[j] = new_j

                                found = True

                                # undo

                                    h[j] -= step
                                else:
                                    h[i] -= step
                                    h[j] += step

def multi_start_perturb(h, n, deadline, n_starts=5, pert_scale=0.05):
    """Try multiple perturbed starting points."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(123)
    
    t_per = (deadline - time.time()) / max(1, n_starts)
    for s in range(n_starts):
        if time.time() > deadline - 5:
            break
        t_end = min(time.time() + t_per, deadline - 5)
        h_p = h + rng.normal(0, pert_scale, size=n)
        h_p = project_feasible(h_p, n)
        h_p, c_p = adam_smooth(h_p, n, t_end, alpha=2000, lr=0.03)
