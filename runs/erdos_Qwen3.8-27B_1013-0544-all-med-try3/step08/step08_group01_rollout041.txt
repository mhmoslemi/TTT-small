import numpy as np
from scipy.optimize import linprog, minimize
from scipy.linalg import null_space
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def correlation_gradients(h, n, shift_indices):
    """Compute gradients of correlation values at given shift indices."""
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    grads = np.zeros((len(shift_indices), n))
    for a, si in enumerate(shift_indices):
        k = si - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[0:nv] = (1.0 - h[k:n]) * dx
            g[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:n] = (1.0 - h[0:nv]) * dx
            g[0:nv] += (-h[kk:n]) * dx
        grads[a] = g
    return grads

def solve_slp(h, n, deadline, tol=1e-10, max_iter=2000):
    """Sequential Linear Programming approach."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Find active set: shifts within a small fraction of max
        threshold = max_val - max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        n_active = min(len(active), 20)  # Limit active set size
        if n_active == 0:
            break
        
        # Get the most important active shifts
        active_sorted = np.argsort(-c_vec)
        active = active_sorted[:n_active]
        
        # Compute gradients
        grads = correlation_gradients(h, n, active)  # shape: (n_active, n)
        
        # Current correlation values at active shifts
        c_active = c_vec[active]
        
        # Solve LP: minimize t
        # s.t. c_active[a] + grads[a] @ (x - h) <= t  for all a
        #      0 <= x[j] <= 1
        #      sum(x) = n/2
        #
        # Variables: x[0..n-1], t
        # Total: n+1 variables
        # 
        # Objective: minimize t -> c_obj = [0, ..., 0, 1]
        #
        # Constraints:
        # For each a: grads[a] @ x + t >= c_active[a] + grads[a] @ h
        #   -> -grads[a] @ x - t <= -c_active[a] - grads[a] @ h
        #
        # sum(x) = n/2 -> Ax_eq @ var = b_eq
        # 0 <= x <= 1, t free (or t >= 0)
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        # Inequality constraints: for each active shift a
        # -grads[a] @ x - t <= -(c_active[a] + grads[a] @ h)
        A_ub = np.zeros((n_active, n + 1))
        b_ub = np.zeros(n_active)
        for a in range(n_active):
            A_ub[a, :n] = -grads[a]
            A_ub[a, n] = -1.0
            b_ub[a] = -(c_active[a] + grads[a] @ h)
        
        # Equality: sum(x) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds: 0 <= x[j] <= 1, t >= 0 (t can be any value but we bound it)
        bounds = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        x_new = res.x[:n]
        t_new = res.x[n]
        
        # Take a step towards the LP solution
        # Use a damping factor to ensure we stay in a region where linearization is valid
        step = x_new - h
        # Check step size
        step_norm = np.linalg.norm(step)
        
        if step_norm < 1e-15:
            break
        
        # Try full step first, then damped
        for damp in [1.0, 0.5, 0.25, 0.1]:
            h_try = h + damp * step
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                h = h_try
                best_c = c_try

            elif c_try <= max_val + 1e-8:
                # Even if not improving, move towards the solution
                h = h_try
                break
        else:
            # No improvement even with damping
            # Try a smaller step
            break
    
    return best_h if compute_c5(best_h, n) < best_c else h, best_c
