    if cb < best_c:

        best_h = hb.copy()

    # Phase 2: More SA with diverse starts (120s)
    p2_dl = t0 + 240
    starts = [
        to_binary(best_h, n),
        to_binary(project_feasible(best_h + 0.3 * rng.randn(n), n), n),
        to_binary(project_feasible(best_h + 0.5 * rng.rand(n) - 0.25, n), n),
        np.zeros(n),
    ]
    starts[3][rng.choice(n, size=n//2, replace=False)] = 1.0
    n_starts = len(starts)
    per_start = (p2_dl - time.time()) / max(n_starts, 1)
    for s in starts:
        sub_dl = time.time() + per_start
        if sub_dl > p2_dl:
            sub_dl = p2_dl
        if time.time() >= sub_dl:
            break
        for T0v, Tiv in [(0.05, 0.00005), (0.2, 0.0005)]:
            if time.time() >= sub_dl:
                break
            hb2, cb2 = sa_binary(s.copy(), n, sub_dl, rng, T0=T0v, Tend=Tiv, max_stale=50000)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()

    # Phase 3: Greedy local search (80s)
    p3_dl = t0 + 320
    hb, cb = greedy_local_search(best_h.copy(), n, p3_dl, max_sweeps=500)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 4: SLSQP with softmax (200s)
    p4_dl = t0 + 520
    best_h, best_c = slsqp_soft(best_h.copy(), n, p4_dl,
                                alpha_list=[500, 2000, 10000, 50000, 200000, 500000],
                                maxiter=400)

    # Phase 5: Try different resolutions (180s)
    p5_dl = t0 + 700
    for nh in [200, 300, 350]:
        if time.time() > p5_dl - 40:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        sub_dl = time.time() + (p5_dl - time.time()) / 3.0
        if sub_dl > p5_dl:
            sub_dl = p5_dl
        hb5, cb5 = sa_binary(to_binary(hh, nh), nh, sub_dl, rng, T0=0.05, Tend=0.0001)
        hb5, cb5 = greedy_local_search(hb5, nh, min(p5_dl - 10, time.time() + 30), max_sweeps=200)
        hb5, cb5 = slsqp_soft(hb5, nh, min(p5_dl - 5, time.time() + 20),
                               alpha_list=[2000, 20000, 100000], maxiter=300)
        if cb5 < best_c:
            best_c = cb5
            best_h = hb5.copy()
            n = nh

    # Phase 6: Targeted continuous refinement (150s)
    p6_dl = t0 + 850
    if time.time() < p6_dl - 10:
        best_h, best_c = targeted_continuous(best_h, n, p6_dl, top_k=6)

    # Phase 7: Final SLSQP + coordinate descent (180s)
    if time.time() < dl - 10:
        best_h, best_c = slsqp_soft(best_h.copy(), n, dl - 5,
                                     alpha_list=[100000, 500000, 2000000, 5000000],
                                     maxiter=600)
    if time.time() < dl - 3:
        best_h, best_c = coordinate_descent(best_h, n, dl - 2)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
