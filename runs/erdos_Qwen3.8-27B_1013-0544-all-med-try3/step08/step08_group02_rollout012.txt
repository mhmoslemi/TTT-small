```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            if s < 1e-14:
                break
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def soft_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def soft_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def active_obj(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    sh -= sh.max()
    val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def active_grad(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_phase(h, n, deadline, alphas, maxiter_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_list is not None and i < len(maxiter_list):
            maxit = maxiter_list[i]
        else:
            maxit = max(500, int(remaining * 2))
        try:
            res = minimize(soft_obj, best_h, jac=soft_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 6,
                                   'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            else:
                best_h = project_feasible(0.5 * best_h + 0.5 * res.x, n)
        except Exception:
            pass
    return best_h, best_c

def active_shifts_phase(h, n, deadline, top_k=8):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [2000, 10000, 50000, 200000, 1000000]
    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def coord_descent(h, n, deadline, sweeps_max=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    order = np.random.RandomState(123).permutation(n)
    for sweep in range(sweeps_max):
        if time.time() > deadline - 5:
            break
        improved = False
        for i in order:
            if time.time() > deadline - 5:
                break
            h_orig = best_h[i]
            best_local = np.inf
            best_x = h_orig
            # Evaluate at a few points
            for t in [max(0, h_orig - 0.05), max(0, h_orig - 0.02), h_orig,
                      min(1, h_orig + 0.02), min(1, h_orig + 0.05),
                      0.0, 0.5, 1.0]:
                if t < 0 or t > 1:
                    continue
                best_h[i] = t
                h_p = project_feasible(best_h, n)
                c = compute_c5(h_p, n)
                if c < best_local:
                    best_local = c
                    best_x = h_p.copy()
            if best_local < best_c - 1e-14:
                best_c = best_local
                best_h = best_x
                improved = True
            else:
                best_h[i] = h_orig
        if not improved:
            break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 2:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline - 2:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.0005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:]) * dx
                grad[k:] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[kk:]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def slqp_phase(h, n, deadline, alpha=200000, maxiter=1000):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        sh -= sh.max()
        return m + np.log(np.sum(np.exp(sh))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def random_perturb(h, n, scale=0.05, seed=0):
    rng = np.random.RandomState(seed)
    h_new = h + rng.randn(n) * scale
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    print(f"Initial: n={n0}, C5={best_c:.6f}, t={time.time()-t0:.1f}s")

    # Phase 1: Optimize at original resolution (n0=500)
    p1_end = t0 + 120
    h1, c1 = lbfgs_phase(best_h, best_n, p1_end,
                         alphas=[50, 200, 1000, 5000, 20000, 100000, 500000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    print(f"Phase 1 (LBFGS n{best_n}): C5={best_c:.6f}, t={time.time()-t0:.1f}s")

    # Phase 2: Active shifts at n0
    if time.time() < deadline - 300:
        p2_end = min(time.time() + 80, deadline - 280)
        h2, c2 = active_shifts_phase(best_h, best_n, p2_end, top_k=6)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        print(f"Phase 2 (Active n{best_n}): C5={best_c:.6f}, t={time.time()-t0:.1f}s")

    # Phase 3: Upsample to n=1500 and optimize
    n_fine = 1500
    if time.time() < deadline - 250:
        h_fine = upsample(best_h, best_n, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine
            best_n = n_fine
        p3_end = min(time.time() + 250, deadline - 200)
        h3, c3 = lbfgs_phase(h_fine, n_fine, p3_end,
                             alphas=[30, 100, 500, 2000, 10000, 50000, 200000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_fine
        print(f"Phase 3 (LBFGS n{n_fine}): C5={best_c:.6f}, t={time.time()-t0:.1f}s")

    # Phase 4: Active shifts at fine resolution
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 120, deadline - 100)
        h4, c4 = active_shifts_phase(best_h, best_n, p4_end, top_k=5)
        if c4 < best_c:
            best_c = c4
            best_h = h4
        print(f"Phase 4 (Active n{best_n}): C5={best_c:.6f}, t={time.time()-t0:.1f}s")

    # Phase 5: Subgradient refinement
    if time.time() < deadline - 60:
        p5_end = min(time.time() + 40, deadline - 40)
        h5, c5 = subgrad_refine(best_h, best_n, p5_end, n_iter=5000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        print(f"Phase 5 (Subgrad n{best_n}): C5={best_c:.6f}, t={time.time()-t0:.1f}s")

    # Phase 6: Coordinate descent polish
    if time.time() < deadline - 30:
        p6_end = min(time.time() + 25, deadline - 15)
        h6, c6 = coord_descent(best_h, best_n, p6_end, sweeps_max=20)
        if c6 < best_c:
            best_c = c6
            best_h = h6
        print(f"Phase 6 (CoordDescent n{best_n}): C5={best_c:.6f}, t={time.time()-t0:.1f}s")

    # Phase 7: Fine pair search
    if time.time() < deadline - 5:
        p7_end = deadline - 2
        h7, c7 = fine_pair_search(best_h, best_n, p7_end, radius=15)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        print(f"Phase 7 (PairSearch n{best_n}): C5={best_c:.6f}, t={time.time()-t0:.1f}s")

    # Phase 8: Try SLSQP polish
    if time.time() < deadline - 2:
        h8, c8 = slqp_phase(best_h, best_n, deadline - 1, alpha=500000, maxiter=500)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        print(f"Phase 8 (SLSQP n{best_n}): C5={best_c:.6f}, t={time.time()-t0:.1f}s")

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    print(f"Final: n={final_n}, C5={best_c:.6f}")
    return best_h, best_c, final_n
```
