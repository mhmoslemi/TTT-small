For the discrete problem with n points, the feasible set is [0,1]^n ∩ {sum = n/2}. The objective is max of n² linear functions (each c[k] is linear in h). Wait, is it?

This is NOT linear in h! It's quadratic: c[k] = sum_i (h[i] - h[i]*h[i+s]) * dx.

So the objective is max of quadratic functions. The optimum might be interior (non-binary).

The vertices of [0,1]^n ∩ {sum = n/2} are the binary vectors with exactly n/2 ones (and the "near-binary" vectors with n/2 - 1 ones, one 0.5, and n/2 - 1 ones... no, the vertices are exactly the 0/1 vectors with sum n/2 when n/2 is integer).

However, each individual c[k] is bilinear (not convex), so the max of them is not convex. The problem is non-convex.

In practice, for this type of problem, binary solutions are often near-optimal. Let me focus on finding the best binary solution, and then try small perturbations.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_ls_targeted(h, n, deadline, max_sweeps=2000):
    """Fast targeted binary local search."""
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    c = np.correlate(h, 1 - h, mode='full')
    best_c = float(c.max())
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        c = np.correlate(h, 1 - h, mode='full')
        current_max = c.max()
        
        # Identify "hot" shifts (near max)
        threshold = current_max - max(1, 2)  # within 2 of max
        hot_positions = np.where(c >= threshold)[0]
        
        # Limit to at most 30 hot positions
        if len(hot_positions) > 30:
            # Keep the top 30
            order = np.argsort(c[hot_positions])[::-1][:30]
            hot_positions = hot_positions[order]
        
        for pos in hot_positions:
            if time.time() > deadline:
                break
            s = pos - (n - 1)
            
            # Find candidate swaps
            candidates = []
            zeros_list = list(zeros)
            for a in zeros_list:
                # Compute term_a = (1-h[a+s]) - h[a-s]
                term_a = 0.0
                if 0 <= a + s < n:
                    term_a += (1.0 - h[a + s])
                if 0 <= a - s < n:
                    term_a -= h[a - s]
                
                if term_a >= 2:  # Even in best case, won't help
                    continue
                
                for b in ones:
                    term_b = 0.0
                    if 0 <= b + s < n:
                        term_b -= (1.0 - h[b + s])
                    if 0 <= b - s < n:
                        term_b += h[b - s]
                    
                    delta = term_a + term_b
                    if delta < 0:
                        candidates.append((delta, a, b))
            
            # Sort by delta (most negative first) and try top candidates
            if candidates:
                candidates.sort()
                for delta, a, b in candidates[:10]:
                    h[a], h[b] = 1.0, 0.0
                    c_new = np.correlate(h, 1 - h, mode='full')
                    new_max = c_new.max()
                    if new_max < current_max - 1e-14:
                        improved = True
                        current_max = new_max
                        c = c_new
                        zeros.discard(a)
                        zeros.add(b)
                        ones.discard(b)
                        ones.add(a)
                        if current_max < best_c:
                            best_c = current_max
                        break
                    else:
                        h[a], h[b] = 0.0, 1.0
                if improved:
                    break
        
        if not improved:
            break
    
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha_list=None, maxiter=2000):
    """SLSQP with smooth max approximation."""
    if alpha_list is None:
        alpha_list = [5000, 50000, 500000]
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        def cons_fun(x):
            return x.sum() - n / 2.0
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]
        
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

            n_best = n_best
    
    # Final check
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    
    return best_h, best_c, n_best
```

Actually, let me reconsider. The `for a in zeros_list: for b in ones:` loop is the bottleneck. For n=250, that's 125*125 = 15625 iterations per shift. With Python overhead (~1μs per iteration), that's ~15ms per shift. With 30 shifts, 450ms per sweep. Over 200s, that's ~440 sweeps. That's reasonable.

But I can vectorize this! Instead of looping over all (a,b) pairs, I can compute the delta for all swaps at once using numpy.

For a fixed shift s:
- For each a in zeros: term_a[a] = (1-h[a+s])·[valid] - h[a-s]·[valid]
- For each b in ones: term_b[b] = -(1-h[b+s])·[valid] + h[b-s]·[valid]
- delta[a,b] = term_a[a] + term_b[b]

This is a rank-1 outer sum! I can compute it as:
- term_a vector of length |zeros|
- term_b vector of length |ones|
- delta matrix = term_a[:,None] + term_b[None,:]
- Find entry with min delta < 0

This is O(n) vectorized operations instead of O(n²) Python loop!

```python
def binary_ls_vectorized(h, n, deadline, max_sweeps=5000):
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    c = np.correlate(h, 1 - h, mode='full')
    best_c = float(c.max())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        c = np.correlate(h, 1 - h, mode='full')
        current_max = c.max()
        
        # Hot positions
        threshold = current_max - 2
        hot = np.where(c >= threshold)[0]
        if len(hot) > 30:
            hot = hot[np.argsort(c[hot])[::-1][:30]]
        
        for pos in hot:
            if time.time() > deadline:
                break
            s = pos - (n - 1)
            
            # Vectorized computation of delta for all swaps
            a = zeros_arr  # positions to set to 1
            b = ones_arr   # positions to set to 0
            
            # term_a[i] = (1-h[a[i]+s]) - h[a[i]-s] (with bounds)
            ta = np.zeros(len(a))
            valid1 = (a + s >= 0) & (a + s < n)
            ta[valid1] += (1.0 - h[a[valid1] + s])
            valid2 = (a - s >= 0) & (a - s < n)
            ta[valid2] -= h[a[valid2] - s]
            
            # term_b[j] = -(1-h[b[j]+s]) + h[b[j]-s] (with bounds)
            tb = np.zeros(len(b))
            valid3 = (b + s >= 0) & (b + s < n)
            tb[valid3] -= (1.0 - h[b[valid3] + s])
            valid4 = (b - s >= 0) & (b - s < n)
            tb[valid4] += h[b[valid4] - s]
            
            # delta[i,j] = ta[i] + tb[j]
            # Find min delta
            min_idx = np.argmin(ta[:, None] + tb[None, :])
            min_val = (ta + tb[np.unravel_index(min_idx, (len(ta), len(tb)))[:][0]]).min()  # hmm
            
            # Actually: min of ta[:,None] + tb[None,:] = min(ta) + min(tb)
            min_val = ta.min() + tb.min()
            
            if min_val < 0:
                i_star = int(np.argmin(ta))
                j_star = int(np.argmin(tb))
                a_star = a[i_star]
                b_star = b[j_star]
                
                # Try the swap
                h[a_star], h[b_star] = 1.0, 0.0
                c_new = np.correlate(h, 1 - h, mode='full')
                new_max = c_new.max()
                if new_max < current_max - 1e-14:
                    improved = True
                    current_max = new_max
                    c = c_new
                    # Update sets
                    zeros_arr = zeros_arr[zeros_arr != a_star]
                    zeros_arr = np.append(zeros_arr, b_star)
                    ones_arr = ones_arr[ones_arr != b_star]
                    ones_arr = np.append(ones_arr, a_star)
                    if current_max < best_c:
                        best_c = current_max
                    break
                else:
                    h[a_star], h[b_star] = 0.0, 1.0
        
        if not improved:
            break
    
    return h, best_c
```
