def project_sum_box(v, target):
    v = np.asarray(v, dtype=float).ravel()
    n = v.size
    if n == 0:
        return v
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    # 80 iterations enough
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    res = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(np.sum(res))
    if diff > 1e-13:
        room = 1.0 - res
        if float(np.sum(room)) > 0:
            order = np.argsort(-room)
            rem = diff
            for idx in order:
                idx = int(idx)
                if rem <= 1e-15:
                    break
                add = min(rem, room[idx])
                if add > 0:
                    res[idx] += add
                    rem -= add
    elif diff < -1e-13:
        rem = -diff
        order = np.argsort(-res)
        for idx in order:
            idx = int(idx)
            if rem <= 1e-15:
                break
            sub = min(rem, res[idx])
            if sub > 0:
                res[idx] -= sub
                rem -= sub
    return res
```

def project_sum_box(v, target):
    v = np.asarray(v, dtype=float).ravel()
    n = v.size
    if n == 0:
        return v

    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)

    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    res = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(res))

    if diff > 1e-13:
        room = 1.0 - res
        if float(np.sum(room)) > 0.0:
            order = np.argsort(-room)
            rem = diff
            for idx in order:
                idx = int(idx)
                if rem <= 1e-15:
                    break
                add = min(rem, float(room[idx]))
                if add > 0.0:
                    res[idx] += add
                    rem -= add
    elif diff < -1e-13:
        rem = -diff
        order = np.argsort(-res)
        for idx in order:
            idx = int(idx)
            if rem <= 1e-15:
                break
            sub = min(rem, float(res[idx]))
            if sub > 0.0:
                res[idx] -= sub
                rem -= sub

    return res


def compute_g_single(h, d):
    n = h.size
    d = int(d)
    if d >= 0:
        if d >= n:
            return 0.0
        L = n - d
        return float(np.dot(h[:L], 1.0 - h[d:]))
    else:
        e = -d
        if e >= n:
            return 0.0
        L = n - e
        return float(np.dot(h[e:], 1.0 - h[:L]))


def compute_grad_g_single(h, d):
    n = h.size
    d = int(d)
    grad = np.zeros(n, dtype=float)

    if d >= 0:
        L = n - d
        if L > 0:
            grad[:L] += 1.0 - h[d:]
            grad[d:] += -h[:L]
    else:
        e = -d
        L = n - e
        if L > 0:
            grad[e:] += 1.0 - h[:L]
            grad[:L] += -h[e:]

    return grad


def compute_all_values(h):
    n = h.size
    dx = 2.0 / float(n)
    omh = 1.0 - h
    vals = np.empty(2 * n - 1, dtype=float)
    vals[n - 1] = float(np.dot(h, omh)) * dx

    for d in range(1, n):
        L = n - d
        vals[n - 1 + d] = float(np.dot(h[:L], omh[d:])) * dx
        vals[n - 1 - d] = float(np.dot(h[d:], omh[:L])) * dx

    return vals


def compute_c5_np(h):
    n = h.size
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def compute_c5_loop(h):
    return float(np.max(compute_all_values(h)))


def active_obj(x):
    return float(x[-1])


def active_obj_jac(x):
    g = np.zeros(len(x), dtype=float)
    g[-1] = 1.0
    return g


def active_ineq(x):
    h = np.asarray(x[:-1], dtype=float)
    t = float(x[-1])
    dx = ACTIVE_DX
    lags = ACTIVE_LAGS
    out = np.empty(len(lags), dtype=float)
    for i, d in enumerate(lags):
        out[i] = t - compute_g_single(h, int(d)) * dx
    return out


def active_ineq_jac(x):
    h = np.asarray(x[:-1], dtype=float)
    n = ACTIVE_N
    dx = ACTIVE_DX
    lags = ACTIVE_LAGS
    m = len(lags)
    J = np.empty((m, n + 1), dtype=float)
    J[:, -1] = 1.0
    for i, d in enumerate(lags):
        J[i, :n] = -compute_grad_g_single(h, int(d)) * dx
    return J


def active_eq(x):
    return float(np.sum(x[:-1]) - ACTIVE_TARGET)


def active_eq_jac(x):
    J = np.zeros((1, len(x)), dtype=float)
    J[0, :-1] = 1.0
    return J


def active_solve(h, t, lags, maxiter=50):
    global ACTIVE_LAGS, ACTIVE_DX, ACTIVE_N, ACTIVE_TARGET

    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return h, float(t)

    lags_list = []
    seen = set()
    for off in lags:
        off = int(off)
        if -n + 1 <= off <= n - 1 and off not in seen:
            seen.add(off)
            lags_list.append(off)

    if not lags_list:
        lags_list = [0]

    ACTIVE_LAGS = lags_list
    ACTIVE_N = n
    ACTIVE_DX = 2.0 / float(n)
    ACTIVE_TARGET = float(n) / 2.0

    x = np.empty(n + 1, dtype=float)
    x[:-1] = h
    x[-1] = float(t)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {"type": "ineq", "fun": active_ineq, "jac": active_ineq_jac},
        {"type": "eq", "fun": active_eq, "jac": active_eq_jac},
    ]

    try:
        res = minimize(
            active_obj,
            x,
            method="SLSQP",
            jac=active_obj_jac,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-10, "disp": 0},
        )
    except Exception:
        return h, float(t)

    if res is None:
        return h, float(t)

    x2 = np.asarray(res.x, dtype=float)
    if not np.all(np.isfinite(x2)):
        return h, float(t)

    h2 = project_sum_box(x2[:-1], ACTIVE_TARGET)
    t2 = float(np.clip(x2[-1], 0.0, 1.0))
    return h2, t2


def make_active_set(vals, n, old_offsets, current_max, t_used, max_active=70, delta=None):
    base = n - 1
    if delta is None:
        scale = max(float(abs(current_max)), 1e-12)
        delta = max(1e-7, 1e-5 * scale)

    combined = []
    seen = set()

    top_k = min(10, len(vals))
    top_idx = np.argsort(-vals)[:top_k]
    for idx in top_idx:
        off = int(idx - base)
        if off not in seen:
            seen.add(off)
            combined.append(off)

    viol_idx = np.flatnonzero(vals > t_used + 1e-7)
    for idx in viol_idx:
        off = int(idx - base)
        if off not in seen:
            seen.add(off)
            combined.append(off)

    near_idx = np.flatnonzero(vals >= current_max - delta)
    for idx in near_idx:
        off = int(idx - base)
        if off not in seen:
            seen.add(off)
            combined.append(off)

    for off in old_offsets:
        off = int(off)
        if off not in seen:
            seen.add(off)
            combined.append(off)

    if not combined:
        combined = [int(np.argmax(vals) - base)]

    vcombo = np.array([float(vals[off + base]) for off in combined], dtype=float)
    order = np.argsort(-vcombo)
    if len(combined) > max_active:
        ordered = [int(combined[i]) for i in order[:max_active]]
    else:
        ordered = [int(combined[i]) for i in order]

    return ordered


def active_refine(h_in, n, maxiter=50, max_outer=10, max_active=70, deadline=None):
    n = int(n)
    h = project_sum_box(np.asarray(h_in, dtype=float).ravel(), float(n) / 2.0)
    vals = compute_all_values(h)
    current = float(vals.max())
    t = current

    delta_init = max(1e-6, 1e-3 * current)
    old = make_active_set(vals, n, [], current, t, max_active, delta_init)

    best_h = h.copy()
    best_c = compute_c5_loop(h)

    for it in range(int(max_outer)):
        if deadline is not None and time.time() > deadline:
            break

        mi = max(15, int(maxiter - 2 * it))
        h_sol, t_sol = active_solve(h, t, old, maxiter=mi)
        h_sol = project_sum_box(h_sol, float(n) / 2.0)

        vals_sol = compute_all_values(h_sol)
        new_max = float(vals_sol.max())

        if new_max < best_c - 1e-12:
            best_c = new_max
            best_h = h_sol.copy()

        if new_max > best_c + 1e-6:
            h = best_h.copy()
            vals = compute_all_values(h)
            current = float(vals.max())
            t = current
            old = make_active_set(vals, n, old, current, t, max_active)
            continue

        current_after = new_max
        next_old = make_active_set(vals_sol, n, old, current_after, t_sol, max_active)

        if set(next_old) == set(old) and abs(current - current_after) < 1e-11:
            break

        current = current_after
        t = max(current, float(t_sol))
        old = next_old
        h = h_sol

    return project_sum_box(best_h, float(n) / 2.0), float(best_c)


def smooth_obj(x):
    h = np.asarray(x, dtype=float)
    lags = SMOOTH_LAGS
    dx = SMOOTH_DX
    alpha = SMOOTH_ALPHA
    m = len(lags)

    if m == 0:
        return 0.0

    vals = np.empty(m, dtype=float)
    for i, d in enumerate(lags):
        vals[i] = compute_g_single(h, int(d)) * dx

    mm = float(np.max(vals))
    sh = alpha * (vals - mm)
    sm = float(np.max(sh))
    e = np.exp(sh - sm)
    s = float(np.sum(e))
    if s <= 0.0:
        return mm
    return float(mm + (sm + float(np.log(s))) / alpha)


def smooth_obj_jac(x):
    h = np.asarray(x, dtype=float)
    lags = SMOOTH_LAGS
    dx = SMOOTH_DX
    alpha = SMOOTH_ALPHA
    m = len(lags)

    if m == 0:
        return np.zeros_like(h, dtype=float)

    vals = np.empty(m, dtype=float)
    for i, d in enumerate(lags):
        vals[i] = compute_g_single(h, int(d)) * dx

    mm = float(np.max(vals))
    sh = alpha * (vals - mm)
    sm = float(np.max(sh))
    e = np.exp(sh - sm)
    s = float(np.sum(e))

    if s > 0.0:
        w = e / s
    else:
        w = np.full(m, 1.0 / float(m), dtype=float)

    grad = np.zeros_like(h, dtype=float)
    for i, d in enumerate(lags):
        wi = float(w[i])
        if abs(wi) > 0.0:
            grad += wi * compute_grad_g_single(h, int(d))

    return grad * dx


def smooth_eq(x):
    return float(np.sum(x) - SMOOTH_TARGET)


def smooth_eq_jac(x):
    J = np.zeros((1, len(x)), dtype=float)
    J[0, :] = 1.0
    return J


def smooth_solve(h_in, n, offsets, alpha, maxiter=40):
    global SMOOTH_LAGS, SMOOTH_DX, SMOOTH_N, SMOOTH_TARGET, SMOOTH_ALPHA

    h = project_sum_box(np.asarray(h_in, dtype=float).ravel(), float(n) / 2.0)
    n = int(n)

    offs = []
    seen = set()
    for off in offsets:
        off = int(off)
        if -n + 1 <= off <= n - 1 and off not in seen:
            seen.add(off)
            offs.append(off)

    if not offs:
        return h

    SMOOTH_LAGS = offs
    SMOOTH_DX = 2.0 / float(n)
    SMOOTH_N = n
    SMOOTH_TARGET = float(n) / 2.0
    SMOOTH_ALPHA = float(alpha)

    x = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": smooth_eq, "jac": smooth_eq_jac}]

    try:
        res = minimize(
            smooth_obj,
            x,
            method="SLSQP",
            jac=smooth_obj_jac,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-10, "disp": 0},
        )
    except Exception:
        return h

    if res is None:
        return h

    h2 = np.asarray(res.x, dtype=float)
    if not np.all(np.isfinite(h2)):
        return h

    return project_sum_box(h2, float(n) / 2.0)


def select_top_offsets(vals, n, max_active):
    base = n - 1
    if max_active <= 0:
        return []
    k = min(int(max_active), len(vals))
    idx = np.argsort(-vals)[:k]
    return [int(i - base) for i in idx]


def smooth_polish(h_in, n, deadline, top_k=50, alphas=(100.0, 400.0, 1500.0, 5000.0)):
    h = project_sum_box(np.asarray(h_in, dtype=float).ravel(), float(n) / 2.0)
    n = int(n)
    best_h = h.copy()
    best_c = compute_c5_loop(h)

    for alpha in alphas:
        if time.time() > deadline:
            break
        vals = compute_all_values(best_h)
        offs = select_top_offsets(vals, n, top_k)
        mi = int(max(15, min(45, int(9000.0 / float(max(1, n))))))
        h2 = smooth_solve(best_h, n, offs, alpha, maxiter=mi)
        h2 = project_sum_box(h2, float(n) / 2.0)
        c2 = compute_c5_loop(h2)
        if c2 < best_c - 1e-12:
            best_c = c2
            best_h = h2.copy()

    return project_sum_box(best_h, float(n) / 2.0), float(best_c)


def interpolate_h(h, n):
    h = np.asarray(h, dtype=float).ravel()
    m = h.size
    n = int(n)
    if m == 0:
        return project_sum_box(np.full(n, 0.5, dtype=float), float(n) / 2.0)

    x0 = np.linspace(0.0, 2.0, m, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    vals = np.interp(x1, x0, h)
    return project_sum_box(vals, float(n) / 2.0)


def perturb_h(h, rng, scale):
    n = h.size
    return project_sum_box(h + float(scale) * rng.randn(n), float(n) / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_budget = float(budget_s)
    deadline = t0 + total_budget - 5.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
        n0 = h0.size
    except Exception:
        h0 = None
        n0 = 0

    if h0 is None or n0 < 10:
        n0 = 250
        x = np.linspace(0.0, 2.0, n0, endpoint=False)
        h0 = (
            0.5
            + 0.45 * np.cos(np.pi * x)
            + 0.10 * np.sin(2.0 * np.pi * x)
            + 0.02 * rng.standard_normal(n0)
        )

    h0 = np.nan_to_num(h0, nan=0.5, posinf=1.0, neginf=0.0)
    h0 = project_sum_box(h0, float(n0) / 2.0)

    initial_ref_h = project_sum_box(h0, float(n0) / 2.0)
    initial_ref_c = compute_c5_np(initial_ref_h)

    best_h = initial_ref_h.copy()
    best_n = int(n0)
    best_c = float(initial_ref_c)

    # Base active-set refinement at the parent resolution.
    base_limit = min(460.0, max(110.0, 0.44 * total_budget))
    dl1 = min(deadline - 240.0, t0 + base_limit)
    if time.time() + 80.0 < dl1:
        h1, c1 = active_refine(
            best_h,
            best_n,
            maxiter=50,
            max_outer=10,
            max_active=80,
            deadline=dl1,
        )
        c1_exact = compute_c5_np(h1)
        if c1_exact < best_c:
            best_c = float(c1_exact)
            best_h = h1.copy()
            best_n = len(best_h)

    # Smooth top-lag polishing.
    if time.time() + 80.0 < deadline - 90.0:
        dl2 = min(deadline - 160.0, time.time() + min(220.0, 0.20 * total_budget))
        h2, c2 = smooth_polish(
            best_h,
            best_n,
            deadline=dl2,
            top_k=60,
            alphas=(200.0, 700.0, 2500.0, 8000.0),
        )
        c2_exact = compute_c5_np(h2)
        if c2_exact < best_c:
            best_c = float(c2_exact)
            best_h = h2.copy()
            best_n = len(best_h)

    # Final active-set solve at the current resolution.
    if time.time() + 70.0 < deadline - 50.0:
        dl3 = min(deadline - 35.0, time.time() + min(150.0, 0.13 * total_budget))
        h3, c3 = active_refine(
            best_h,
            best_n,
            maxiter=35,
            max_outer=6,
            max_active=80,
            deadline=dl3,
        )
        c3_exact = compute_c5_np(h3)
        if c3_exact < best_c:
            best_c = float(c3_exact)
            best_h = h3.copy()
            best_n = len(best_h)

    # Optional perturbation restart, only when there is meaningful time left.
    if best_c > 0.38080 and time.time() + 180.0 < deadline - 160.0:
        dl_r = min(deadline - 150.0, time.time() + 150.0)
        h_p = perturb_h(best_h, rng, 0.025)
        h_r, c_r = active_refine(
            h_p,
            best_n,
            maxiter=35,
            max_outer=6,
            max_active=60,
            deadline=dl_r,
        )
        c_r_exact = compute_c5_np(h_r)
        if c_r_exact < best_c:
            best_c = float(c_r_exact)
            best_h = h_r.copy()
            best_n = len(best_h)

    # Optional higher-resolution refinements.
    if time.time() + 300.0 < deadline:
        candidates = []
        if best_n <= 260:
            candidates = [300, 380, 480]
        elif best_n <= 420:
            candidates = [best_n + 80, best_n + 180]
        else:
            candidates = [best_n + 180]

        tried = set()
        for nh in candidates:
            if time.time() + 260.0 >= deadline:
                break

            nh = int(nh)
            if nh <= best_n or nh in tried:
                continue
            if nh % 2 == 1:
                nh += 1
            tried.add(nh)

            rem = deadline - time.time()
            phase_time = min(250.0, max(90.0, rem - 170.0))
            if phase_time < 70.0:
                break

            phase_deadline = time.time() + phase_time
            h_up = interpolate_h(best_h, nh)

            h4, c4 = active_refine(
                h_up,
                nh,
                maxiter=30,
                max_outer=4,
                max_active=45,
                deadline=phase_deadline,
            )

            if time.time() + 40.0 < phase_deadline:
                h4, c4b = smooth_polish(
                    h4,
                    nh,
                    deadline=phase_deadline,
                    top_k=35,
                    alphas=(500.0, 2000.0),
                )
                c4 = c4b

            c4_exact = compute_c5_np(h4)
            if c4_exact < best_c:
                best_c = float(c4_exact)
                best_h = h4.copy()
                best_n = len(best_h)

    # Tiny final active polish, guarded by time.
    if time.time() + 70.0 < deadline:
        n_fin = len(best_h)
        vals_fin = compute_all_values(best_h)
        cur_fin = float(vals_fin.max())
        act_fin = make_active_set(vals_fin, n_fin, [], cur_fin, cur_fin, 50, 1e-6)
        h_fin, t_fin = active_solve(best_h, cur_fin, act_fin, maxiter=25)
        h_fin = project_sum_box(h_fin, float(n_fin) / 2.0)
        c_fin = compute_c5_np(h_fin)
        if c_fin < best_c:
            best_c = float(c_fin)
            best_h = h_fin.copy()
            best_n = len(best_h)

    # Tiny final smooth polish, guarded by time.
    if time.time() + 40.0 < deadline:
        n_fin = len(best_h)
        vals_fin = compute_all_values(best_h)
        offs_fin = select_top_offsets(vals_fin, n_fin, 25)
        h_fin = smooth_solve(best_h, n_fin, offs_fin, 3000.0, maxiter=18)
        h_fin = project_sum_box(h_fin, float(n_fin) / 2.0)
        c_fin = compute_c5_np(h_fin)
        if c_fin < best_c:
            best_c = float(c_fin)
            best_h = h_fin.copy()
            best_n = len(best_h)

    # Final feasible projection. If projection made things slightly worse than the
    # saved feasible parent/reference, fall back to that one.
    best_h = project_sum_box(best_h, float(len(best_h)) / 2.0)
    best_c = compute_c5_np(best_h)

    if best_c > initial_ref_c:
        best_h = initial_ref_h.copy()
        best_c = float(initial_ref_c)
        best_n = int(n0)

    return best_h, float(best_c), int(len(best_h))
