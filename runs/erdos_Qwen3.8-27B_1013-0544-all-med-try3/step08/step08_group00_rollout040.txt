import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, mm):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = mm - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj_fn(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_fn(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 5.0), (500, 50.0), (2000, 500.0), (8000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj_fn, h, jac=smooth_grad_fn,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.05, active_tol=0.002, max_active=25):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, active_tol * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > max_active:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:max_active]]
        
        # Build LP: min t
        # s.t. grad_k . d - t <= -c_k for active k
        #      sum(d) = 0
        #      lb[i] <= d[i] <= ub[i]
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0
        
        n_act = len(active)
        A_ub = np.zeros((n_act, n_var))
        b_ub = np.zeros(n_act)
        for j, idx in enumerate(active):
            g = shift_grad_single(h, n, idx)
            A_ub[j, 0:n] = g
            A_ub[j, n] = -1.0
            b_ub[j] = -c_vec[idx]
        
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        b_eq = np.zeros(1)
        
        lb = np.maximum(-h, -rho)
        ub = np.minimum(1.0 - h, rho)
        bounds = [(lb[i], ub[i]) for i in range(n)] + [(0.0, 1.0)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                delta = res.x[0:n]
                h_new = h + delta
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    no_improve = 0
                    if rho < 0.08:
                        rho = min(rho * 1.2, 0.1)
                else:
                    no_improve += 1
                    rho *= 0.5
                    if rho < 1e-10:
                        break
                    if no_improve > 30:
                        break
            else:
                rho *= 0.5
                if rho < 1e-10:
                    break
        except Exception:
            rho *= 0.5
            if rho < 1e-10:
                break
    return best_h, best_c

def local_search(h, n, deadline, radius=30, steps=None):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-10, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            # Compute priority gradient
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad_single(h, n, idx))
            grad /= len(active)
            # Focus on high-gradient regions
            top_idx = np.argsort(-grad)[:min(radius * 2, n)]

                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)

                        h[i] += step
                        h[j] -= step

                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Move mass from i to j
                    if h[i] >= step and h[j] + step <= 1.0 + 1e-12:

                        h[j] += step
                        c = compute_c5(h, n)

                            best_c = c
                            improved = True

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def structured_init(n, seed):

    # Create a pattern with some structure

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        remaining = deadline - time.time()
        maxiter_use = min(maxiter, int(remaining * 2))
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max(maxiter_use, 100), 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
