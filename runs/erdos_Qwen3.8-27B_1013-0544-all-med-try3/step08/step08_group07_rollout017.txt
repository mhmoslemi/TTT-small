def project_feasible(h, n):
    target = n / 2.0

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

        max_lags = np.where(max_mask)[0]
        
        if len(max_lags) == 0:
            max_lags = [np.argmax(g)]
        
        grad = np.zeros(n)
        for lag in max_lags[:5]:  # Limit to top 5
            lag_offset = int(lag)
            grad_temp = np.zeros(n)
            # For correlation at this lag, gradient wrt h[i]
            # c[lag] = sum_i h[i] * (1-h[i+lag-(n-1)])
            # dc/dh[i] involves: (1-h[i+lag-(n-1)]) - h[i] from the term where h is first
            # and -h[i-lag+(n-1)] from where h is second
            center = n - 1
            for i in range(n):
                j = i + lag - center
                if 0 <= j < n:
                    grad_temp[i] += (1 - h[j])
                j2 = i - lag + center
                if 0 <= j2 < n:
                    grad_temp[i] -= h[j2]
            grad += grad_temp
        grad /= len(max_lags[:5])
        
        # Project gradient to constraint (sum = n/2)
        grad -= grad.mean()
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        # Line search
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                h = ht
                found = True
                break
            elif ct < best_c:
                h = ht

                break
        else:
            c = compute_c5(h, n)

    return best_h, best_c

            best_c = c
            best_h = h_new.copy()
    
    return best_h, best_c

def pair_refine_fine(h, n, deadline, deltas=None):
    """Fine pair refinement for continuous solutions."""
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    best_c = compute_c5(h, n)
    
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n - 1, i, -1):
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s * d, h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                prog = True
                                break
                            else:
                                h[i], h[j] = hi - s * d, hj + s * d
                    if prog:
                        break
                if prog:
                    break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    n = n0
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Binary initialization + SA at original resolution (35% budget)
    p1_end = t0 + 0.35 * (budget_s - 5)
    h_bin = binarize(best_h, n)
    c_bin = compute_c5(h_bin, n)
    
    if time.time() < p1_end:
        h_bin, c_bin = sa_binary(h_bin, n, time.time(), p1_end, rng, n_restarts=4)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
    
    # Phase 2: Best-improvement local search on binary (12% budget)
    p2_end = t0 + 0.47 * (budget_s - 5)
    if time.time() < p2_end:
        h_bin2, c_bin2 = best_improvement_binary(best_h.copy(), n, p2_end, max_rounds=200)
        if c_bin2 < best_c:
            best_c = c_bin2
            best_h = h_bin2.copy()
    
    # Phase 3: Block swap search (8% budget)
    p3_end = t0 + 0.55 * (budget_s - 5)
    if time.time() < p3_end:
        h_blk, c_blk = block_swap_search(best_h.copy(), n, p3_end)
        if c_blk < best_c:
            best_c = c_blk
            best_h = h_blk.copy()
    
    # Phase 4: Continuous SLSQP optimization (20% budget)
    p4_end = t0 + 0.75 * (budget_s - 5)
    if time.time() < p4_end:
        h_cont = best_h.copy()
        # Add small perturbation to escape binary
        h_cont = project_feasible(h_cont + 0.05 * rng.randn(n), n)
        h_cont, c_cont = multi_alpha_slsqp(h_cont, n, p4_end,
                                            alphas=[200, 1000, 5000, 20000, 100000, 500000, 2000000],
                                            maxiter_each=600)
        if c_cont < best_c:
            best_c = c_cont
            best_h = h_cont.copy()
        
        # Also try SLSQP from the binary directly
        if time.time() < p4_end - 30:
            h_b2 = best_h.copy()
            if (h_b2 == 0).any() and (h_b2 == 1).any():
                h_b2 = project_feasible(h_b2 + 0.02 * rng.randn(n), n)
            h_b2, c_b2 = multi_alpha_slsqp(h_b2, n, p4_end - 10,
                                            alphas=[5000, 50000, 500000, 5000000],
                                            maxiter_each=400)
            if c_b2 < best_c:
                best_c = c_b2
                best_h = h_b2.copy()
    
    # Phase 5: Subgradient polish (8% budget)
    p5_end = t0 + 0.83 * (budget_s - 5)
    if time.time() < p5_end:
        h_sg, c_sg = subgradient_polish(best_h.copy(), n, p5_end, max_iter=20000)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg.copy()
    
    # Phase 6: Try higher resolution (12% budget)
    p6_end = t0 + 0.95 * (budget_s - 5)
    if time.time() < p6_end - 60:
        nh = n * 2  # Double resolution
        if nh <= 500:
            xo = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, nh, endpoint=False)
            h_hi = np.interp(xn, xo, best_h)
            h_hi = project_feasible(h_hi, nh)
            
            # Binarize and SA at higher res
            h_hi_bin = binarize(h_hi, nh)
            if time.time() < p6_end - 30:
                h_hi_bin, c_hi = sa_binary(h_hi_bin, nh, time.time(), p6_end - 15, rng, n_restarts=2)
                if c_hi < best_c:
                    best_c = c_hi
                    best_h = h_hi_bin.copy()
                    n = nh
    
    # Phase 7: Final polish
    if time.time() < dl - 5:
        # Final SLSQP
        h_final = slsqp_no_grad(best_h.copy(), n, 5000000, 500, dl - 5)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
        
        # Final pair refine
        if time.time() < dl - 2:
            best_h, best_c = pair_refine_fine(best_h, n, dl,
                                               deltas=[0.005, 0.002, 0.001, 0.0005, 0.0001])
    
    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
