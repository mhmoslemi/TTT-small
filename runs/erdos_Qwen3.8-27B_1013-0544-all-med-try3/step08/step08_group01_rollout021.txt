def smooth_grad(x, n, active, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    vals = c[active]
    alpha_vals = alpha * vals
    mx = alpha_vals.max()
    w = np.exp(alpha_vals - mx)
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, ak in zip(w, active):
        k = ak - (n - 1)
        wdx = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wdx * (1.0 - x[k:n])
            grad[k:n] += wdx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wdx * (1.0 - x[0:nv])
            grad[0:nv] += wdx * (-x[kk:n])
    
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    if abs(h.sum() - target) < 1e-14:
        return h
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h_proj
            tot = room.sum()
            if tot > 1e-15:
                h_proj += (diff / tot) * room
        else:
            tot = h_proj.sum()
            if tot > 1e-15:
                h_proj += (diff / tot) * h_proj
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_obj(x, n, active, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    vals = c[active]
    avals = alpha * vals
    mx = avals.max()
    log_sum = mx + np.log(np.sum(np.exp(avals - mx)))
    val = log_sum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, active, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    vals = c[active]
    avals = alpha * vals
    mx = avals.max()
    w = np.exp(avals - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ak in zip(w, active):
        if wi < 1e-16:
            continue
        k = ak - (n - 1)
        wdx = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wdx * (1.0 - x[k:n])
            grad[k:n] += wdx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wdx * (1.0 - x[0:nv])
            grad[0:nv] += wdx * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_active(h, n, active, alpha, mu, maxiter, deadline, bounds):
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad,
                      args=(n, active, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 30, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def find_active(c_vec, n, mode='top_k', k=10, frac=0.005):
    max_val = c_vec.max()
    if mode == 'top_k':
        idx = np.argsort(-c_vec)
        return idx[:min(k, 2 * n - 1)]
    else:
        threshold = max_val * (1.0 - frac)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 50:
            active = active[np.argsort(-c_vec[active])[:50]]
        if len(active) == 0:
            active = np.argsort(-c_vec)[:5]
        return active
