import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        if alpha < 1e10:
            sh = alpha * (g - m)
            sm = sh.max()
            smooth_term = (sm + np.log(np.exp(sh - sm).sum())) / alpha
        else:
            smooth_term = 0
        return m + smooth_term
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

        # Gradient polish at new resolution
        if time.time() < deadline - 10:
            hh, c = projected_gradient_inf(hh, nh, deadline - 10, max_iter=5000)
            if c < c_best:
                c_best = c
                best_h = hh.copy()
                n_best = nh
                results[-1] = (best_h.copy(), c_best, n_best)
    return best_h, c_best, n_best

def perturbed_restart(h, n, c, deadline, rng, max_restarts=5):
    """Try escaping local minima with random perturbations."""
    best_h = h.copy()
    best_c = c
    for _ in range(max_restarts):
        if time.time() > deadline:
            break
        scale = np.random.uniform(0.05, 0.2)
        h_new = project_feasible(h + scale * rng.randn(n), n)
        # Quick SLSQP
        h_new = slsqp_optimize(h_new, n, 100000, 500, 1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: SLSQP with very large alpha (25% budget)
    p1_end = t0 + 0.25 * (budget_s - 3)
    h = best_h.copy()
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000, 50000000, 500000000]
    for alpha in alphas:
        if time.time() > p1_end:
            break
        h = slsqp_optimize(h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Projected gradient with analytic gradient (15% budget)
    p2_end = t0 + 0.40 * (budget_s - 3)
    if time.time() < p2_end:
        h, c = projected_gradient_inf(best_h.copy(), n, p2_end, max_iter=30000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Simulated annealing on binary (15% budget)
    p3_end = t0 + 0.55 * (budget_s - 3)
    if time.time() < p3_end:
        hb, cb = simulated_annealing_binary(best_h.copy(), n, p3_end, target_iters=100000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        # Second SA from perturbed start
        if time.time() < p3_end:
            h_pert = project_feasible(best_h + 0.15 * rng.randn(n), n)
            hb2, cb2 = simulated_annealing_binary(h_pert, n, p3_end, target_iters=50000)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()

    # Phase 4: Higher resolution (20% budget)
    p4_end = t0 + 0.75 * (budget_s - 3)
    if time.time() < p4_end:
        best_h, best_c, n = multi_resolution_attempt(best_h, n, best_c, p4_end, rng)

    # Phase 5: Coordinate descent (10% budget)
    p5_end = t0 + 0.85 * (budget_s - 3)
    if time.time() < p5_end:
        h, c = coordinate_descent_fast(best_h.copy(), n, p5_end, max_sweeps=60)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Perturbed restarts (5% budget)
    p6_end = t0 + 0.90 * (budget_s - 3)
    if time.time() < p6_end:
        best_h, best_c = perturbed_restart(best_h, n, best_c, p6_end, rng, max_restarts=4)

    # Phase 7: Final SLSQP + gradient polish (5% budget)
    if time.time() < dl - 10:
        h = slsqp_optimize(best_h.copy(), n, 1000000000, 5000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    if time.time() < dl - 5:
        h, c = projected_gradient_inf(best_h.copy(), n, dl - 3, max_iter=10000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    assert abs(best_h.sum() - n / 2.0) < 1e-8, f"Sum constraint violated: {best_h.sum()} vs {n/2}"
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12), "Bounds violated"
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
