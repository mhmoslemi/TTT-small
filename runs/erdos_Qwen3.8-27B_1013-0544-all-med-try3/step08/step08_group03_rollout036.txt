import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += (diff / tr) * room
        else:
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_correlation(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        w = np.exp(c_shift - mx)
        w /= w.sum()
    else:
        w = np.exp(c_shift)
        w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:k+nv])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:kk+nv])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_correlation(x, n)
    c_max = c.max()
    sh = alpha * (c - c_max)
    mx = sh.max()
    if mx > 500:
        val = c_max + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = c_max + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alpha_list = [2000, 10000, 50000, 200000, 1000000, 5000000]

def binary_flip_delta(h, n, i):
    """Compute delta in corr (without dx) when flipping h[i] from 1 to 0. h must be binary."""
    m_lo = i
    m_hi = min(i + n - 1, 2 * n - 2)
    delta = np.zeros(2 * n - 1)
    if m_lo > m_hi:
        return delta
    indices = np.arange(m_lo, m_hi + 1)

    if 2 * i >= m_lo and 2 * i <= m_hi:

    delta[m_lo:m_hi + 1] = vals
    return delta
