def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):

import numpy as np
from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

def _run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):

    best_c = compute_c5(best_h, n)

            pass
    return best_h, best_c

def _run_active(h, n, deadline, top_k=5, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000]
    for rnd in range(12):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(200, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.002), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def _run_slqp(h, n, deadline, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sq_eq, 'jac': _sq_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slqp_max_obj, h, jac=_slqp_max_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def _coord_descent(h, n, deadline, max_rounds=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stuck = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 5e-5 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        idx_sorted = np.argsort(grad)
        targets = idx_sorted[:8]
        sources = idx_sorted[-8:]
        improved = False
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        for i in targets:
            if time.time() > deadline:
                break
            for j in sources:
                if i == j:
                    continue
                for d in deltas:
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c_new = compute_c5(best_h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c_new = compute_c5(best_h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
        if not improved:
            stuck += 1
            if stuck > 30:
                break
        else:
            stuck = 0
    return best_h, best_c

def _subgrad(h, n, deadline, n_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
    return best_h, best_c

def _tug_of_war(h, n, deadline, n_iter=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t = best_c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 0.002 / max(gnorm, 1e-10)
        step = min(step, 0.01)
        h_new = best_h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick L-BFGS at original resolution
    p1_end = t0 + 180
    h1, c1 = _run_lbfgs(best_h, best_n, p1_end,
        alphas=[100, 1000, 10000, 50000, 200000, 1000000, 5000000],
        mu=0.01, maxiters=[1500, 2000, 2000, 2500, 2500, 2500, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set at original resolution
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 100, deadline - 350)
        h2, c2 = _run_active(best_h, best_n, p2_end, top_k=4,
            alphas=[20000, 100000, 500000, 2000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: SLSQP at original resolution
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 50, deadline - 300)
        h3, c3 = _run_slqp(best_h, best_n, p3_end, maxiter=2000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to n=1000 and optimize
    n_fine = 1000
    if time.time() < deadline - 350:
        x_fine = np.linspace(0, 2, n_fine)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)
        c_fine = compute_c5(h_fine, n_fine)

        p4_end = min(time.time() + 250, deadline - 250)
        h4, c4 = _run_lbfgs(h_fine, n_fine, p4_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.005, maxiters=[2000, 2000, 2500, 2500, 2500, 2500, 2000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_fine

    # Phase 5: Active set at best resolution
    if time.time() < deadline - 250:
        p5_end = min(time.time() + 100, deadline - 230)
        h5, c5 = _run_active(best_h, best_n, p5_end, top_k=4,
            alphas=[50000, 200000, 1000000, 5000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: SLSQP at best resolution
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 60, deadline - 140)
        h6, c6 = _run_slqp(best_h, best_n, p6_end, maxiter=2000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Tug-of-war
    if time.time() < deadline - 100:
        p7_end = min(time.time() + 50, deadline - 90)
        h7, c7 = _tug_of_war(best_h, best_n, p7_end, n_iter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Coordinate descent
    if time.time() < deadline - 60:
        p8_end = min(time.time() + 50, deadline - 50)
        h8, c8 = _coord_descent(best_h, best_n, p8_end, max_rounds=400)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final SLSQP
    if time.time() < deadline - 15:
        p9_end = deadline - 5
        h9, c9 = _run_slqp(best_h, best_n, p9_end, maxiter=500)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Subgradient micro-refinement
    if time.time() < deadline - 3:
        h10, c10 = _subgrad(best_h, best_n, deadline - 1, n_iter=1000)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final projection
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
