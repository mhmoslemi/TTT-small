def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

def overlap_at_shift(h, n, k):

        nv = n - k
        if nv <= 0:

        return float(np.dot(h[:nv], 1 - h[k:k+nv])) * dx

        nv = n - kk
        if nv <= 0:
            return 0.0
        return float(np.dot(h[kk:kk+nv], 1 - h[:nv])) * dx

def overlap_at_shifts(h, n, shift_indices):
    """shift_indices are full-correlate indices (0 to 2n-2)."""
    dx = 2.0 / n
    results = np.zeros(len(shift_indices))
    for ii, mm in enumerate(shift_indices):
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            results[ii] = float(np.dot(h[:nv], 1 - h[k:k+nv])) * dx
        else:
            kk = -k
            nv = n - kk
            results[ii] = float(np.dot(h[kk:kk+nv], 1 - h[:nv])) * dx

def grad_overlap_at_shift(x, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[:nv] += (1.0 - x[k:k+nv]) * dx
            grad[k:k+nv] += (-x[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:kk+nv] += (1.0 - x[:nv]) * dx
            grad[:nv] += (-x[kk:kk+nv]) * dx
    return grad

def top_k_smoothed_obj(x, n, shift_indices, alpha, mu):
    c_vals = overlap_at_shifts(x, n, shift_indices)
    mx = c_vals.max()
    sh = alpha * (c_vals - mx)
    smx = sh.max()
    if smx > 500:
        logsumexp = smx + np.log(np.sum(np.exp(sh - smx)))
    else:
        logsumexp = np.log(np.sum(np.exp(sh)))
    val = mx + logsumexp / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def top_k_smoothed_grad(x, n, shift_indices, alpha, mu):
    c_vals = overlap_at_shifts(x, n, shift_indices)
    mx = c_vals.max()
    sh = alpha * (c_vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n

        g = grad_overlap_at_shift(x, n, k)
        grad += wi * g

    return grad


def topk_obj(x, n, si, alpha, mu):
    vals = overlap_at_shifts(x, n, si)
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        v = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        v = mx + np.log(np.sum(np.exp(sh))) / alpha
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)


def topk_grad(x, n, si, alpha, mu):
    vals = overlap_at_shifts(x, n, si)
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(len(x))
    for wi, mm in zip(w, si):
        if wi < 1e-14:
            continue
        grad += wi * grad_one_shift(x, n, int(mm))
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def run_lbfgs_topk(h, n, deadline, k=5, alphas=None):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    cv = compute_c5_vec(h, n)
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        top_idx = np.argsort(-cv)[:k]
        try:
            res = minimize(topk_obj, h, jac=topk_grad,
                           args=(n, top_idx, alpha, 1.0),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': 10000,
                                    'maxls': 50})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c = cn
                h = hn
                cv = compute_c5_vec(h, n)
        except Exception:
            pass
    return h, best_c


def asl_obj(x, n, si):
    return float(x[n])


def asl_obj_jac(x, n, si):
    v = np.zeros(n + 1)
    v[n] = 1.0
    return v


def asl_ineq(x, n, si):
    h = x[:n]
    t = x[n]
    cv = overlap_at_shifts(h, n, si)
    return t - cv


def asl_ineq_jac(x, n, si):
    h = x[:n]
    m = len(si)
    J = np.zeros((m, n + 1))
    for j, mm in enumerate(si):
        J[j, :n] = -grad_one_shift(h, n, int(mm))
        J[j, n] = 1.0
    return J


def asl_eq(x, n):
    return float(x[:n].sum() - n / 2.0)


def asl_eq_jac(x, n):
    v = np.zeros(n + 1)
    v[:n] = 1.0
    return v


def run_slsqp_active(h, n, deadline, top_k=5):
    cv = compute_c5_vec(h, n)
    mx = cv.max()
    thr = mx - max(1e-7, 0.003 * mx)
    active = np.where(cv >= thr)[0]
    if len(active) > top_k:
        order = np.argsort(-cv[active])
        active = active[order[:top_k]]
    
    x0 = np.concatenate([h, [mx + 0.005]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': asl_ineq, 'jac': asl_ineq_jac, 'args': (n, active)},
        {'type': 'eq', 'fun': asl_eq, 'jac': asl_eq_jac, 'args': (n,)}
    ]
    try:
        res = minimize(asl_obj, x0, jac=asl_obj_jac, args=(n, active),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-16})
        hn = project_feasible(res.x[:n], n)
        cn = compute_c5(hn, n)
        return hn, cn
    except Exception:
        return h.copy(), compute_c5(h, n)


def resample(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    xo = np.linspace(0, 2, n_old)
    xn = np.linspace(0, 2, n_new)
    hr = np.interp(xn, xo, h)
    return project_feasible(hr, n_new)


def gradient_local_search(h, n, deadline):
    best_c = compute_c5(h, n)
    base_steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for base in base_steps:
        if time.time() > deadline:
            break
        steps_remaining = base
        for _ in range(500):
            if time.time() > deadline:
                return h, best_c
            cv = compute_c5_vec(h, n)
            mx = cv.max()
            thr = mx - max(1e-7, 0.002 * mx)
            active = np.where(cv >= thr)[0]
            if len(active) == 0:
                break
            
            # Average gradient of active shifts
            g = np.zeros(n)
            for mm in active:
                g += grad_one_shift(h, n, int(mm))
            g /= len(active)
            
            # Find best 2-point move along -g
            gi = np.argmax(-g)  # Where g is most negative → increase h
            gj = np.argmin(-g)  # Where g is most positive → decrease h
            # Actually: we want to move h in -g direction
            # h_new = h - step * g
            # So at points where g > 0, h decreases; where g < 0, h increases
            
            # Try moving mass from high-g to low-g
            i_pos = np.argmax(g)  # h[i] should decrease
            j_pos = np.argmin(g)  # h[j] should increase
            
            improved = False
            for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
                step = steps_remaining * frac
                if h[i_pos] - step >= 0 and h[j_pos] + step <= 1:
                    h[i_pos] -= step
                    h[j_pos] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i_pos] += step
                        h[j_pos] -= step
                if improved:
                    break
            if not improved:
                steps_remaining *= 0.5
                if steps_remaining < 1e-8:
                    break
    return h, best_c


def pair_search(h, n, deadline, radius=20):
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c
