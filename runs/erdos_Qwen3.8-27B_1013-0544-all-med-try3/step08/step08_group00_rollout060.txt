    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(cobj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_loop(h, n, deadline, max_iter=3000, rho_init=0.04, act_frac=0.004, act_max=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        cvec = compute_c5_vec(h, n)
        mx = cvec.max()
        thr = mx - max(1e-12, act_frac * mx)
        act = np.where(cvec >= thr)[0]
        if len(act) > act_max:
            idx_order = np.argsort(-cvec[act])
            act = act[idx_order[:act_max]]
        if len(act) == 0:
            break
        delta, _ = sqp_lp_solve(h, n, act, rho)
        if delta is None:
            rho *= 0.4
            if rho < 1e-12:
                break
            continue
        if np.max(np.abs(delta)) < 1e-14:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            rho = min(rho * 1.6, 0.12)
        else:
            stall += 1
            rho *= 0.3
            if rho < 1e-13 or stall > 60:
                break
    return best_h, best_c

def lbfgs_pass(h, n, deadline, alphas=None, mus=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [300, 1000, 5000, 20000, 80000]
        mus = [3.0, 10.0, 50.0, 200.0, 800.0]
    for a, mu in zip(alphas, mus):
        if time.time() > deadline - 4:
            break
        try:
            r = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                        args=(n, a, mu), method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 30000, 'ftol': 1e-21,
                                 'gtol': 1e-18, 'maxfun': 150000, 'maxls': 60})
            h_new = project_feasible(np.asarray(r.x), n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def local_pair_search(h, n, deadline, steps=None):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            cvec = compute_c5_vec(h, n)
            mx = cvec.max()
            thr = mx - max(1e-9, 0.003 * mx)
            act = np.where(cvec >= thr)[0]
            if len(act) == 0:
                break
            grad = np.zeros(n)
            for idx in act:
                grad += shift_grad(h, n, idx)
            grad /= len(act)
            top = np.argsort(-np.abs(grad))[:min(80, n)]
            for i in top:
                if time.time() > deadline:
                    return h, best_c
                i = int(i)
                for dj in [1, -1, 2, -2, 3, -3, 5, -5, 10, -10, 20, -20, 30, -30]:
                    j = i + dj
                    if j < 0 or j >= n:
                        continue
                    for sgn in [1, -1]:
                        si = h[i] + sgn * step
                        sj = h[j] - sgn * step
                        if si < -1e-9 or si > 1.0 + 1e-9 or sj < -1e-9 or sj > 1.0 + 1e-9:
                            continue
                        h[i] = si
                        h[j] = sj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= sgn * step
                            h[j] += sgn * step
    return h, best_c

def subgradient_polish(h, n, deadline, n_iter=50000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    h = h.copy()
    lr = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        cvec = compute_c5_vec(h, n)
        k_star = int(np.argmax(cvec))
        g = shift_grad(h, n, k_star)
        h_cand = h - lr * g
        h_cand = project_feasible(h_cand, n)
        c_cand = compute_c5(h_cand, n)
        if c_cand < best_c - 1e-15:
            best_c = c_cand
            best_h = h_cand.copy()
            h = h_cand
            lr = min(lr * 1.08, 0.02)
        else:
            lr *= 0.85
            if lr < 1e-7:
                break
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_orig = 400

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: L-BFGS warmup at original resolution
    t1 = min(t0 + 35, deadline - 900)
    if t1 > time.time() + 3:
        best_h, best_c = lbfgs_pass(best_h, n_orig, t1,
                                     alphas=[200, 800, 3000, 12000],
                                     mus=[2.0, 8.0, 30.0, 120.0])

    # Phase 2: Main SQP at original resolution
    t2 = min(t0 + 160, deadline - 650)
    if t2 > time.time() + 5:
        best_h, best_c = sqp_loop(best_h, n_orig, t2,
                                   max_iter=4000, rho_init=0.04,
                                   act_frac=0.004, act_max=28)

    # Phase 3: L-BFGS with high alpha (tight approximation)
    t3 = min(t0 + 220, deadline - 550)
    if t3 > time.time() + 3:
        best_h, best_c = lbfgs_pass(best_h, n_orig, t3,
                                     alphas=[20000, 60000, 150000],
                                     mus=[200.0, 600.0, 1500.0])

    # Phase 4: SQP again with tighter params
    t4 = min(t0 + 320, deadline - 450)
    if t4 > time.time() + 5:
        best_h, best_c = sqp_loop(best_h, n_orig, t4,
                                   max_iter=3000, rho_init=0.02,
                                   act_frac=0.003, act_max=20)

    # Phase 5: Local pair search
    t5 = min(t0 + 420, deadline - 350)
    if t5 > time.time() + 3:
        best_h, best_c = local_pair_search(best_h, n_orig, t5)

    # Phase 6: Subgradient polish
    t6 = min(t0 + 500, deadline - 250)
    if t6 > time.time() + 3:
        h_sub, c_sub = subgradient_polish(best_h, n_orig, t6, n_iter=80000)
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub

    # Phase 7: Upsample to n=1000 and optimize
    n_hi = 1000
    if time.time() < deadline - 200:
        h_hi = upsample(best_h, n_orig, n_hi)
        c_hi = compute_c5(h_hi, n_hi)

        t7 = min(time.time() + 50, deadline - 120)
        if t7 > time.time() + 3 and c_hi < best_c + 0.001:
            h_hi, c_hi = lbfgs_pass(h_hi, n_hi, t7,
                                     alphas=[500, 2000, 8000],
                                     mus=[5.0, 20.0, 80.0])

        t8 = min(time.time() + 50, deadline - 40)
        if t8 > time.time() + 5 and c_hi < best_c + 0.001:
            h_hi, c_hi = sqp_loop(h_hi, n_hi, t8,
                                   max_iter=1500, rho_init=0.03,
                                   act_frac=0.004, act_max=25)

        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
            n_orig = n_hi

    # Phase 8: Final SQP polish at best resolution
    if time.time() < deadline - 10:
        final_deadline = deadline - 2
        h_fin, c_fin = sqp_loop(best_h, n_orig, final_deadline,
                                 max_iter=2000, rho_init=0.015,
                                 act_frac=0.002, act_max=15)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Final feasibility verification
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
