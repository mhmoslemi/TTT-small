        n_ones_sample = min(len(ones), 80)
        n_zeros_sample = min(len(zeros), 80)
        one_sample = np.random.choice(ones, size=n_ones_sample, replace=False) if len(ones) > n_ones_sample else ones
        zero_sample = np.random.choice(zeros, size=n_zeros_sample, replace=False) if len(zeros) > n_zeros_sample else zeros
        
        for i in one_sample:
            if time.time() > deadline - 10:
                return best_h, best_c
            for j in zero_sample:
                if i == j:
                    continue
                # Swap: set h[i]=0, h[j]=1
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                h[i] = 1.0
                h[j] = 0.0
                if c < max_val - 1e-14:
                    improvement = max_val - c
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            # Try smaller neighborhood
            break
    
    # If 0/1 is better, return it; otherwise return best
    if best_c <= compute_c5(project_feasible(np.clip(h, 0, 1), n), n):
        return best_h, best_c
    return h, compute_c5(h, n)

def push_toward_01(h, n, push_strength, n_iters=500, deadline=None):
    """Iteratively push h values toward 0 or 1 while maintaining feasibility."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(n_iters):
        if deadline and time.time() > deadline - 5:
            break
        # Identify values closest to 0 or 1, push them further
        close_to_0 = np.where(h < 0.1)[0]
        close_to_1 = np.where(h > 0.9)[0]
        
        if len(close_to_0) < 5 or len(close_to_1) < 5:
            break
        
        # Pick a random pair: one near 0, one near 1
        i = np.random.choice(close_to_0)
        j = np.random.choice(close_to_1)
        if i == j:
            continue
        
        # Push i toward 0, j toward 1
        amt = min(h[i], 1.0 - h[j], push_strength)
        if amt < 1e-10:
            continue
        h[i] -= amt
        h[j] += amt
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c > best_c + 1e-8:
            h[i] += amt
            h[j] -= amt
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, n)
        else:
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def slsqp_refine(h, n, deadline, alpha=100000, maxiter=2000):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        return float(val)
    
    def obj_jac(x):
        c = compute_c5_vec(x, n)
        w = lse_weights(c, alpha)
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in range(len(w)):
            if w[idx] < 1e-14:
                continue
            k = idx - (n - 1)
            wm = w[idx] * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
        return grad
    
    def sq(x):
        return float(x.sum() - n / 2.0)
    
    def sq_j(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq, 'jac': sq_j}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def local_pair_search_global(h, n, deadline, n_attempts=50000):
    """Global pair search with random sampling."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.default_rng(12345)
    
    for _ in range(n_attempts):
        if time.time() > deadline - 5:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        step = rng.uniform(0.001, 0.01)
        
        # Direction 1: h[i] += step, h[j] -= step
        if h[i] + step <= 1.0 and h[j] - step >= 0.0:
            h[i] += step
            h[j] -= step
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
            else:
                h[i] -= step
                h[j] += step
        # Direction 2: h[i] -= step, h[j] += step
        if h[i] - step >= 0.0 and h[j] + step <= 1.0:
            h[i] -= step
            h[j] += step
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
            else:
                h[i] += step
                h[j] -= step
    
    return best_h, best_c

def top_shifts_lbfgs(h, n, deadline, top_k=5, alpha_seq=None):
    """L-BFGS focused on top-k shifts."""
    if alpha_seq is None:
        alpha_seq = [500, 5000, 50000, 200000]
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    n_half = n / 2.0
    
    def topk_obj(x, alpha):
        c = compute_c5_vec(x, n)
        W = lse_weights(c, alpha)
        # Only use top-k weighted
        c_max = c.max()
        thresh = c_max - 0.005 * c_max
        mask = c >= thresh
        w_ak = W.copy()
        w_ak[~mask] = 0
        if w_ak.sum() < 1e-15:
            w_ak = W.copy()
        w_ak /= w_ak.sum()
        val = float(np.dot(w_ak, c))
        val += 0.1 * (x.sum() - n_half) ** 2
        return val
    
    def topk_jac(x, alpha):
        c = compute_c5_vec(x, n)
        W = lse_weights(c, alpha)
        c_max = c.max()
        thresh = c_max - 0.005 * c_max
        mask = c >= thresh
        w_ak = W.copy()
        w_ak[~mask] = 0
        if w_ak.sum() < 1e-15:
            w_ak = W.copy()
        w_ak /= w_ak.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in range(len(w_ak)):
            if w_ak[idx] < 1e-14:
                continue
            k = idx - (n - 1)
            wm = w_ak[idx] * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
        grad += 0.2 * (x.sum() - n_half)
        return grad
    
    for alpha in alpha_seq:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(topk_obj, best_h, jac=topk_jac, args=(alpha,),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': 20000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def multistart_optimize(h_init, n, deadline, n_starts=3, seed_base=777):
    """Try multiple perturbations of the initial solution."""
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    rng = np.random.default_rng(seed_base)
    
    for s in range(n_starts):
        if time.time() > deadline - 60:
            break
        # Create perturbation
        scale = 0.05 / (s + 1)
        h_pert = h_init + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Short Adam optimization
        h_opt, c_opt = run_adam_cycles(h_pert, n, min(time.time() + 80, deadline - 30),
                                       n_cycles=2, top_k=5, alpha_start=500, alpha_end=50000,
                                       lr_start=0.005, lr_end=0.001)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)
    
    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    n = n_orig
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Adam optimization with cycling (main workhorse)
    t1_end = min(t0 + 180, deadline - 400)
    best_h, best_c = run_adam_cycles(best_h, n, t1_end,
        n_cycles=5, top_k=7, alpha_start=100, alpha_end=50000,
        lr_start=0.02, lr_end=0.002, tol_frac=0.004)
    
    # Phase 2: L-BFGS top-shifts refinement
    if time.time() < deadline - 250:
        t2_end = min(time.time() + 120, deadline - 250)
        best_h, best_c = top_shifts_lbfgs(best_h, n, t2_end, top_k=5,
                                          alpha_seq=[1000, 10000, 50000, 200000])
    
    # Phase 3: Push toward 0/1 and combinatorial search
    if time.time() < deadline - 180:
        h_push, c_push = push_toward_01(best_h, n, push_strength=0.005,
                                         n_iters=800, deadline=deadline - 100)
        if c_push < best_c:
            best_c = c_push
            best_h = h_push
        
        # Combinatorial swap search on the pushed solution
        if time.time() < deadline - 80:
            t3_end = min(time.time() + 60, deadline - 80)
            h_comb, c_comb = combinatorial_swap_search(best_h, n, t3_end, n_rounds=30)
            if c_comb < best_c:
                best_c = c_comb
                best_h = h_comb
    
    # Phase 4: SLSQP polish
    if time.time() < deadline - 150:
        t4_end = min(time.time() + 60, deadline - 150)
        h_slq, c_slq = slsqp_refine(best_h, n, t4_end, alpha=200000, maxiter=2000)
        if c_slq < best_c:
            best_c = c_slq
            best_h = h_slq
    
    # Phase 5: Multi-start with perturbations
    if time.time() < deadline - 120:
        t5_end = min(time.time() + 100, deadline - 120)
        h_ms, c_ms = multistart_optimize(best_h, n, t5_end, n_starts=4, seed_base=999)
        if c_ms < best_c:
            best_c = c_ms
            best_h = h_ms
    
    # Phase 6: Another Adam round with different params
    if time.time() < deadline - 80:
        t6_end = min(time.time() + 50, deadline - 80)
        h_a2, c_a2 = run_adam_cycles(best_h, n, t6_end,
            n_cycles=3, top_k=4, alpha_start=5000, alpha_end=200000,
            lr_start=0.003, lr_end=0.0005, tol_frac=0.002)
        if c_a2 < best_c:
            best_c = c_a2
            best_h = h_a2
    
    # Phase 7: Global local pair search
    if time.time() < deadline - 40:
        t7_end = min(time.time() + 30, deadline - 40)
        h_pair, c_pair = local_pair_search_global(best_h, n, t7_end, n_attempts=30000)
        if c_pair < best_c:
            best_c = c_pair
            best_h = h_pair
    
    # Phase 8: Final SLSQP
    if time.time() < deadline - 15:
        t8_end = min(time.time() + 12, deadline - 15)
        h_fin, c_fin = slsqp_refine(best_h, n, t8_end, alpha=500000, maxiter=1500)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    
    # Phase 9: Try at slightly different resolution
    if time.time() < deadline - 25:
        n_alt = n + 100  # e.g., 600
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_alt)
        h_alt = np.interp(x_new, x_old, best_h)
        h_alt = project_feasible(h_alt, n_alt)
        c_alt = compute_c5(h_alt, n_alt)
        if c_alt < best_c + 0.001:
            t9_end = min(time.time() + 15, deadline - 10)
            h_alt, c_alt = run_adam_cycles(h_alt, n_alt, t9_end,
                n_cycles=2, top_k=5, alpha_start=1000, alpha_end=20000,
                lr_start=0.005, lr_end=0.001)
            if c_alt < best_c:
                best_c = c_alt
                best_h = h_alt
                n = n_alt
    
    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
