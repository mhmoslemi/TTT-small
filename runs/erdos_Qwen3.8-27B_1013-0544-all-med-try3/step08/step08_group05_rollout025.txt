    dx = 2.0 / n
    h = x

    m = corr.max()

    sm = shifted.max()

    obj = m + np.log(weight_sum) / alpha

    s = h.sum() - n / 2.0

                             options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-12})
                h = project_feasible(res.x, n)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            except Exception:
                pass
    return best_h, best_c

def binary_local_search(h, n, deadline, n_restarts=3, seed=42):
    target = n // 2
    h = make_binary(h, n)
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    s_arr = np.arange(-(n-1), n)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            n_perturb = rng.randint(3, 12)
            for _ in range(n_perturb):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) == 0 or len(zi) == 0:
                    break
                a = rng.choice(oi)
                b = rng.choice(zi)
                h[a], h[b] = 0.0, 1.0
        
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        
        for sweep in range(2000):
            if time.time() > deadline:
                break
            improved = False
            s_star = int(np.argmax(corr) - (n-1))
            
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            z_order = zeros[rng.permutation(len(zeros))]
            o_order = ones[rng.permutation(len(ones))]
            
            for i in z_order:
                if time.time() > deadline:
                    break
                delta_i = np.zeros(2*n-1)
                v1 = (i + s_arr >= 0) & (i + s_arr < n)
                delta_i[v1] = (1.0 - h[i + s_arr[v1]])
                v2 = (i - s_arr >= 0) & (i - s_arr < n)
                delta_i[v2] = delta_i[v2] - h[i - s_arr[v2]]
                
                di_ss = delta_i[s_star + n - 1]
                
                found = False
                for j in o_order:
                    if time.time() > deadline:
                        break
                    dj_ss = 0.0
                    if 0 <= j + s_star < n:
                        dj_ss -= (1.0 - h[j + s_star])
                    if 0 <= j - s_star < n:
                        dj_ss += h[j - s_star]
                    
                    if di_ss + dj_ss >= -1e-15:
                        continue
                    
                    delta_j = np.zeros(2*n-1)
                    v3 = (j + s_arr >= 0) & (j + s_arr < n)
                    delta_j[v3] = -(1.0 - h[j + s_arr[v3]])
                    v4 = (j - s_arr >= 0) & (j - s_arr < n)
                    delta_j[v4] = delta_j[v4] + h[j - s_arr[v4]]
                    
                    delta = delta_i + delta_j
                    new_max = np.max(corr + delta)
                    if new_max < cur_max - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = corr + delta
                        cur_max = new_max
                        improved = True
                        found = True
                        if cur_max * dx < best_c:
                            best_c = cur_max * dx
                            best_h = h.copy()
                        break
                if found:
                    break
            if not improved:
                break
    
    return best_h, best_c

def subgrad_step(h, n, lr, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for m in idx:
        s = m - (n-1)
        if s == 0:
            grad[:] += (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += (-h[:n-s])
        else:
            sa = -s
            grad[sa:] += (1.0 - h[:n-sa])
            grad[:n-sa] += (-h[sa:])
    grad *= dx / len(idx)
    h_new = project_feasible(h - lr * grad, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0
    
    # Phase 1: Try multiple resolutions from parent
    p1 = t0 + 0.08 * (budget_s - 5)
    resolutions = [n0]
    for delta_n in [-8, -4, 4, 8, 12]:
        nc = n0 + delta_n
        if nc % 2 == 0 and nc >= 100 and nc <= 400:
            resolutions.append(nc)
    
    candidates = []
    for nh in resolutions:
        if time.time() > p1:
            break
        if nh == n0:
            h_test = best_h.copy()
        else:
            xo = np.linspace(0, 2, n0)
            xn = np.linspace(0, 2, nh)
            h_test = project_feasible(np.interp(xn, xo, h0), nh)
        c_test = compute_c5(h_test, nh)
        candidates.append((c_test, h_test, nh))
    
    candidates.sort(key=lambda x: x[0])
    for c_test, h_test, nh in candidates:
        if c_test < best_c:
            best_c = c_test
            best_h = h_test.copy()
            best_n = nh
    
    n = best_n
    h = best_h.copy()
    
    # Phase 2: L-BFGS smooth optimization
    p2 = t0 + 0.30 * (budget_s - 5)
    h, c = lbfgs_smooth(h, n, p2, alphas=[500, 2000, 10000, 50000], lams=[1000, 10000])
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 3: Binary local search (main phase)
    p3 = t0 + 0.65 * (budget_s - 5)
    hb, cb = binary_local_search(best_h, best_n, p3, n_restarts=8, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 4: Sell - alternate smooth and binary
    p4 = t0 + 0.80 * (budget_s - 5)
    h_cur = best_h.copy()
    for it in range(6):
        if time.time() > p4:
            break
        # Smooth step
        h_cur = project_feasible(h_cur + 0.02 * rng.randn(best_n), best_n)
        h_s, c_s = lbfgs_smooth(h_cur, best_n, min(time.time() + 20, p4), 
                                  alphas=[20000, 100000], lams=[10000, 100000])
        if c_s < best_c:
            best_c = c_s
            best_h = h_s.copy()
            h_cur = h_s.copy()
        # Binary step
        h_b, c_b = binary_local_search(h_cur, best_n, min(time.time() + 20, p4), 
                                         n_restarts=3, seed=seed + it + 10)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b.copy()
            h_cur = h_b.copy()
    
    # Phase 5: Try other resolutions with best solution
    p5 = t0 + 0.92 * (budget_s - 5)
    for dh in [-6, -3, 3, 6, 10, -10, 15, -15]:
        if time.time() > p5:
            break
        nh = best_n + dh
        if nh % 2 != 0 or nh < 100 or nh > 400:
            continue
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        h_new = project_feasible(np.interp(xn, xo, best_h), nh)
        h_new, c_new = lbfgs_smooth(h_new, nh, min(time.time() + 15, p5),
                                      alphas=[10000, 50000], lams=[10000])
        h_new, c_new2 = binary_local_search(h_new, nh, min(time.time() + 10, p5),
                                             n_restarts=2, seed=seed + 100 + dh)
        c_final = min(c_new, c_new2)
        if c_final < best_c:
            best_c = c_final
            if c_new2 < c_new:
                best_h = h_new.copy()
            else:
                best_h = project_feasible(h_new, nh)
            best_n = nh
    
    # Phase 6: Final polish
    if time.time() < dl - 3:
        hb, cb = binary_local_search(best_h, best_n, dl - 1, n_restarts=5, seed=seed + 200)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
