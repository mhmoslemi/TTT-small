```
def c_full_fft(h,n,size):
    dx=2.0/n
    hp=np.zeros(size, dtype=float); hp[:n]=h
    qp=np.empty(size); qp[:n]=1.0-h; qp[n:]=0.0
    Fh=np.fft.rfft(hp)
    Fq=np.fft.rfft(qp)
    r=np.fft.irfft(Fh*np.conj(Fq), size)
    # may have tiny imaginary discarded
    return np.concatenate((r[size-n+1:size], r[0:n])) * dx
```

`project_feasible`:
```
def project_feasible(h, n, exact=False):
    h = np.array(h, dtype=float, copy=True)
    if h.size != n: h = np.resize(h, n) # safeguard
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(4):
        s = h.sum()
        diff = target - s
        if abs(diff) <= 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15: break
            h += (diff / tot) * room
        else:
            if s <= 1e-15: break
            h *= (1.0 + diff / s)
    np.clip(h,0,1,out=h)
    if exact:
       diff=target-h.sum()
       if diff>1e-13:
          idxs=np.argsort(1-h)[::-1]
          for idx in idxs:
             room=1-h[idx]
             add=min(room, diff)
             if add<=0: break
             h[idx]+=add; diff-=add
             if abs(diff)<1e-14: break
       elif diff<-1e-13:
          idxs=np.argsort(h)[::-1]
          for idx in idxs:
             sub=min(h[idx], -diff)
             if sub<=0: break
             h[idx]-=sub; diff += sub
             if abs(diff)<1e-14: break
       np.clip(h,0,1,out=h)
    return h
```

```
def project_feasible(h,n,exact=False):
    h=np.array(h,dtype=float,copy=True)
    if h.shape[0]!=n: h=np.resize(h,n)
    h=np.where(np.isfinite(h), h, 0.5)
    np.clip(h,0.0,1.0,out=h)
    target=n/2.0
    for _ in range(5):
        s=float(np.sum(h))
        diff=target-s
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            room=1.0-h; tot=float(np.sum(room))
            if tot <= 1e-15: break
            h += (diff/tot)*room
        else:
            if s <= 1e-15: break
            h *= (1.0 + diff/s)
    np.clip(h,0.0,1.0,out=h)
    if exact:
       diff=target-float(np.sum(h))
       if diff > 1e-12:
          idxs=np.argsort(1.0-h)[::-1]
          for idx in idxs:
             room=1.0-h[idx]
             if room <= 0 or diff <= 1e-15: break
             add=min(room,diff)
             h[idx]+=add
             diff-=add
       elif diff < -1e-12:
          idxs=np.argsort(h)[::-1]
          for idx in idxs:
             if h[idx] <=0 or -diff <=1e-15: break
             sub=min(h[idx], -diff)
             h[idx]-=sub
             diff += sub
    return h
```

```
def c_full_fft(h,n,N):
    dx=2.0/float(n)
    hp=np.zeros(N); hp[:n]=h
    qp=np.zeros(N); qp[:n]=1.0-h

    return np.concatenate((r[N-n+1:N], r[0:n]))*dx
```

```python
import time
import numpy as np
from scipy.optimize import minimize

_FFT_CACHE = {}


def make_even(n):
    try:
        n = int(round(float(n)))
    except Exception:
        n = 300
    if n < 100:
        n = 100
    if n % 2 != 0:
        n += 1
    return n


def get_fft_size(n):
    v = _FFT_CACHE.get(n)
    if v is not None:
        return v
    size = 1
    target = 2 * n - 1
    while size < target:
        size <<= 1
    _FFT_CACHE[n] = size
    return size


def project_feasible(h, n, exact=False):
    h = np.array(h, dtype=float, copy=True)
    if h.shape[0] != n:
        h = np.resize(h, n)
    h = np.where(np.isfinite(h), h, 0.5)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0

    for _ in range(5):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(np.sum(room))
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            if s <= 1e-15:
                break
            h *= (1.0 + diff / s)

    np.clip(h, 0.0, 1.0, out=h)

    if exact:
        diff = target - float(np.sum(h))
        if diff > 1e-12:
            idxs = np.argsort(1.0 - h)[::-1]
            for idx in idxs:
                room = 1.0 - h[idx]
                if room <= 0.0 or diff <= 1e-15:
                    break
                add = min(room, diff)
                h[idx] += add
                diff -= add
        elif diff < -1e-12:
            idxs = np.argsort(h)[::-1]
            for idx in idxs:
                if h[idx] <= 0.0 or -diff <= 1e-15:
                    break
                sub = min(h[idx], -diff)
                h[idx] -= sub
                diff += sub
    return h


def compute_c_direct(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))


def c_full_fft(h, n, N):
    dx = 2.0 / float(n)
    hp = np.zeros(N, dtype=float)
    hp[:n] = h
    qp = np.zeros(N, dtype=float)
    qp[:n] = 1.0 - h
    Fh = np.fft.rfft(hp)
    Fq = np.fft.rfft(qp)
    r = np.fft.irfft(Fh * np.conj(Fq), N)
    return np.concatenate((r[N - n + 1:N], r[0:n])) * dx


def eval_fft_max(h, n, N):
    c = c_full_fft(h, n, N)
    return float(np.max(c))


def softmax_lag_weights(c, beta):
    m = float(np.max(c))
    z = beta * (c - m)
    z = np.where(z > 0.0, 0.0, z)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def build_w_arrays(w, n, N):
    W = np.zeros(N, dtype=float)
    Wrev = np.zeros(N, dtype=float)
    W[0] = w[n - 1]
    Wrev[0] = w[n - 1]
    if n > 1:
        W[1:n] = w[n:]
        W[N - n + 1:N] = w[:n - 1]
        Wrev[1:n] = w[:n - 1][::-1]
        Wrev[N - n + 1:N] = w[n:][::-1]
    return W, Wrev


def weighted_gradient_pre(h, n, FW, FWr, N, wsum=1.0, scl=1.0):
    hp = np.zeros(N, dtype=float)
    hp[:n] = h
    Fh = np.fft.rfft(hp)
    c1 = np.fft.irfft(Fh * FW, N)
    c2 = np.fft.irfft(Fh * FWr, N)
    g = (wsum - (c1 + c2))[:n]
    return scl * (2.0 / float(n)) * g


def grad_smooth_fft(h, n, alpha, N, scl=1.0):
    c = c_full_fft(h, n, N)
    w = softmax_lag_weights(c, alpha)
    W, Wrev = build_w_arrays(w, n, N)
    FW = np.fft.rfft(W)
    FWr = np.fft.rfft(Wrev)
    return weighted_gradient_pre(h, n, FW, FWr, N, float(np.sum(w)), scl)


def fw_obj_unified(x, n, w, W, FW, FWr, N, scl, mu):
    c = c_full_fft(x, n, N)
    val = float(scl * np.dot(w, c))
    s = float(np.sum(x))
    val += mu * (s - n * 0.5) ** 2
    return val


def fw_grad_unified(x, n, w, W, FW, FWr, N, scl, mu):
    g = weighted_gradient_pre(x, n, FW, FWr, N, float(np.sum(w)), scl)
    diff = float(np.sum(x) - n * 0.5)
    g += 2.0 * mu * diff
    return g


def fw_lbfgs(h, n, w, N, deadline, mu=100.0, maxiter=200):
    h = project_feasible(h, n)
    if maxiter < 10 or time.time() > deadline - 2.0:
        return h

    W, Wrev = build_w_arrays(w, n, N)
    FW = np.fft.rfft(W)
    FWr = np.fft.rfft(Wrev)
    scl = 0.5 * float(n)
    bounds = [(0.0, 1.0)] * n
    options = {
        'maxiter': int(maxiter),
        'ftol': 1e-13,
        'maxfun': int(max(maxiter * 5, 20)),
        'maxls': 30
    }
    try:
        res = minimize(
            fw_obj_unified,
            h,
            jac=fw_grad_unified,
            args=(n, w, W, FW, FWr, N, scl, mu),
            method='L-BFGS-B',
            bounds=bounds,
            options=options
        )
        x = res.x
        if np.all(np.isfinite(x)):
            return project_feasible(x, n)
    except Exception:
        pass
    return h


def alternating_phase(h, n, deadline, N, mu=100.0):

    best_h = h.copy()
    best_c = compute_c_direct(h, n)

    if n <= 500:
        maxit = 250
    elif n <= 1000:
        maxit = 180
    else:
        maxit = 120

    betas = (10, 50, 200, 1000, 5000, 20000, 100000, 500000)

        c2 = compute_c_direct(h2, n)

        if c2 < best_c - 1e-13:

def interpolate_to_n(h, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)


def make_perturbed(h, n, sigma, rng):
    if rng is None:
        rng = np.random.RandomState(7)
    h2 = h + sigma * (2.0 * rng.rand(n) - 1.0)
    return project_feasible(h2, n)


def make_threshold(h, n):
    b = np.zeros(n, dtype=float)
    idx = np.argsort(h)[::-1]
    k = n // 2
    if k == 0:
        k = 1
    b[idx[:k]] = 1.0
    return project_feasible(b, n)


def make_random(n, rng):
    if rng is None:
        rng = np.random.RandomState(11)
    idx = rng.permutation(n)
    b = np.zeros(n, dtype=float)
    k = n // 2
    if k == 0:
        k = 1
    b[idx[:k]] = 1.0
    if n > 10:
        ker = np.array([1.0, 2.0, 1.0])
        b = np.convolve(b, ker, mode='same') / 6.0
    b += 0.20 * (2.0 * rng.rand(n) - 1.0)
    return project_feasible(b, n)


def build_candidate_start(kind, n, ref_h, ref_n, rng):
    if kind == 'parent':
        if n == ref_n:
            return project_feasible(ref_h, n)
        return interpolate_to_n(ref_h, ref_n, n)

    base = interpolate_to_n(ref_h, ref_n, n) if n != ref_n else ref_h

    if kind == 'perturb':
        sigma = 0.12 if n <= 900 else 0.18
        return make_perturbed(base, n, sigma, rng)
    if kind == 'threshold':
        return make_threshold(base, n)
    if kind == 'random':
        return make_random(n, rng)
    return base


def optimize_candidate(h, n, deadline, N, rng):
    h = project_feasible(h, n)
    c = compute_c_direct(h, n)

    rem = deadline - time.time()
    if rem > 30.0:
        alt_deadline = time.time() + rem * 0.65
        h, c = alternating_phase(h, n, alt_deadline, N)

    if time.time() < deadline - 25.0:
        rem2 = deadline - time.time()
        if rem2 > 50.0:
            pg_deadline = time.time() + rem2 * 0.40
            h, c = projected_grad_refine(h, n, pg_deadline, N)

    if time.time() < deadline - 1.0:
        h, c = local_pair_refine(h, n, deadline - 1.0, N, rng)

    h = project_feasible(h, n, exact=True)
    c = compute_c_direct(h, n)
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(300, 0.5)
        n_parent = 300

    if n_parent < 100:
        h_parent = project_feasible(h_parent, n_parent, exact=True)
        best_h = h_parent.copy()
        best_n = n_parent
        best_c = compute_c_direct(best_h, best_n)
        n_parent = 300
        h_parent = make_random(n_parent, rng)
        h_parent = project_feasible(h_parent, n_parent, exact=True)

    h_parent = project_feasible(h_parent, n_parent, exact=True)
    best_h = h_parent.copy()
    best_n = n_parent
    best_c = compute_c_direct(best_h, best_n)

    base = make_even(n_parent)
    max_n = 2000
    cand = []

    cand.append((n_parent, 'parent', 0.30))
    if n_parent <= 1200:
        cand.append((n_parent, 'perturb', 0.12))
        cand.append((n_parent, 'threshold', 0.05))
    else:
        cand.append((n_parent, 'perturb', 0.17))

    def_interp_weights = [
        (2, 0.22),
        (3, 0.18),
        (4, 0.08)
    ]
    for mult, wt in def_interp_weights:
        m = make_even(base * mult)
        if m <= max_n and m != n_parent:
            dup = False
            for item in cand:
                if item[0] == m and item[1] == 'interp':
                    dup = True
                    break
            if not dup:
                cand.append((m, 'interp', wt))

    m15 = make_even(1500)
    if m15 <= max_n and m15 != n_parent:
        dup = False
        for item in cand:
            if item[0] == m15 and item[1] == 'interp':
                dup = True
                break
        if not dup:
            cand.append((m15, 'interp', 0.04))

    mr = make_even(2 * base)
    if mr <= max_n:
        cand.append((mr, 'random', 0.05))

    total_w = 0.0
    for item in cand:
        total_w += item[2]

    if total_w <= 0.0:
        scale_time = 0.0
    else:
        scale_time = max(30.0, deadline - t0 - 80.0) / total_w

    ref_h = best_h
    ref_n = best_n

    for idx, (n, kind, wt) in enumerate(cand):
        if time.time() > deadline - 60.0:
            break
        alloc = wt * scale_time
        if idx > 0 and alloc < 20.0:
            break
        max_alloc = deadline - time.time() - 60.0
        if alloc > max_alloc:
            alloc = max_alloc
        if alloc <= 5.0:
            break

        N = get_fft_size(n)
        start = build_candidate_start(kind, n, ref_h, ref_n, rng)
        c_deadline = time.time() + alloc

        try:
            h_opt, c_opt = optimize_candidate(start, n, c_deadline, N, rng)
        except Exception:
            h_opt = project_feasible(start, n, exact=True)
            c_opt = compute_c_direct(h_opt, n)

        if c_opt < best_c - 1e-12:
            best_c = c_opt
            best_h = h_opt
            best_n = n
            ref_h = best_h
            ref_n = best_n

    rem = deadline - time.time() - 2.0
    if rem > 15.0:
        N = get_fft_size(best_n)
        fin_deadline = time.time() + min(rem, 180.0)
        try:
            h_f, c_f = optimize_candidate(best_h, best_n, fin_deadline, N, rng)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
                best_n = best_n
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n, exact=True)
    best_c = compute_c_direct(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
