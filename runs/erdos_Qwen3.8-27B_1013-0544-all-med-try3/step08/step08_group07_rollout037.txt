def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:

        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
