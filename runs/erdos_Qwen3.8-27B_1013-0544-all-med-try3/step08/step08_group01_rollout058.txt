def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_topK_A(h_tilde, n, top_idx):
    """Build constraint rows for selected shifts."""
    K = len(top_idx)
    A = np.zeros((K, n))
    for ii, r in enumerate(top_idx):
        k = r - (n - 1)
        j_start = max(k, 0)
        j_end = min(n - 1 + k, n - 1)
        if j_start <= j_end:
            idx_start = j_start - k
            idx_end = j_end - k
            A[ii, j_start:j_end+1] = 1.0 - h_tilde[idx_start:idx_end+1]
    return A

    """Solve LP: min t s.t. c[k] <= t (linearized), 0<=h<=1, sum(h)=n/2."""
    K = len(active_idx)
    A = build_topK_A(h_tilde, n, active_idx)

    c_obj[-1] = 1.0

    A_ub[:K, N] = -1.0
    A_ub[K:K+N, :N] = -np.eye(N)
    A_ub[K+N:K+2*N, :N] = np.eye(N)
    b_ub = np.zeros(K + 2*N)
    b_ub[K+N:] = 1.0

    return None

        h_new = solve_lp(h_cur, n, top_idx)
        
        if h_new is not None:
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            
            # Damped update
            h_cur = damping * h_new + (1.0 - damping) * h_cur
            h_cur = project_feasible(h_cur, n)
        else:
            # LP failed, small perturbation
            break
        
        # Convergence check
        if it > 10 and max_val - compute_c5_vec(h_cur, n).max() < 1e-10:
            c_cur = compute_c5(h_cur, n)
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
            break
    
    return best_h, best_c

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_polish(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(50, 5.0), (200, 50.0), (500, 200.0),
                   (2000, 1000.0), (5000, 5000.0), (10000, 50000.0),
                   (20000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def topK_smooth_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topK_smooth_grad(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topK_lbfgs(h, n, deadline, top_k=10, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(1000, 10.0), (5000, 100.0), (20000, 1000.0), (50000, 5000.0)]
    
    for _ in range(6):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h, n)
        active_idx = np.argsort(-c_vec)[:top_k]
        
        for alpha, mu in configs:
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(topK_smooth_obj, h, jac=topK_smooth_grad,
                              args=(n, active_idx, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 20000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def fine_pair_search(h, n, deadline, center_radius=40):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def stochastic_restarts(h_start, n, deadline, n_restarts=5, restart_frac=0.3):
    """Try random perturbations of the solution to escape local optima."""
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    
    rng = np.random.RandomState(123)
    spread = (deadline - time.time()) / (n_restarts + 1)
    
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        
        # Random perturbation
        eps = 0.05
        h_pert = h_start + rng.uniform(-eps, eps, size=n)
        h_pert = project_feasible(h_pert, n)
        
        # Quick alternating LP from perturbed start
        t_sub = min(time.time() + spread * restart_frac, deadline - 15)
        h_new, c_new = alternating_lp(h_pert, n, t_sub, max_iter=50, topK_start=20, topK_end=40, damping=0.5)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    return best_h, best_c

def try_different_n(h_best, c_best, n_target, deadline):
    """Upsample/downsample and re-optimize at different n."""
    x_old = np.linspace(0, 2, len(h_best))
    x_new = np.linspace(0, 2, n_target)
    h_new = np.interp(x_new, x_old, h_best)
    h_new = project_feasible(h_new, n_target)
    c_new = compute_c5(h_new, n_target)
    
    if c_new <= c_best or c_new < c_best + 0.001:
        # Worth optimizing at this resolution
        h_new, c_new = alternating_lp(h_new, n_target, min(time.time() + 60, deadline - 30),
                                       max_iter=80, topK_start=30, topK_end=60, damping=0.6)
        if c_new < c_best:
            return h_new, c_new, n_target
    
    return h_best, c_best, len(h_best)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    np.random.seed(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        n_orig = 500
        h_init = np.full(n_orig, 0.5)
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Alternating LP at original resolution (primary driver)
    t1 = min(t0 + 200, deadline - 500)
    best_h, best_c = alternating_lp(best_h, n_orig, t1,
                                     max_iter=150, topK_start=20, topK_end=80, damping=0.6)
    
    # Phase 2: L-BFGS-B polish
    t2 = min(time.time() + 120, deadline - 350)
    best_h, best_c = lbfgs_polish(best_h, n_orig, t2,
                                   configs=[(100, 10.0), (500, 100.0), (2000, 500.0),
                                            (5000, 5000.0), (10000, 50000.0)])
    
    # Phase 3: Top-K L-BFGS focused on worst shifts
    t3 = min(time.time() + 80, deadline - 250)
    best_h, best_c = topK_lbfgs(best_h, n_orig, t3, top_k=8,
                                 configs=[(2000, 50.0), (10000, 500.0), (50000, 5000.0)])
    
    # Phase 4: Another alternating LP round (reconverge after L-BFGS)
    t4 = min(time.time() + 100, deadline - 180)
    h_alt, c_alt = alternating_lp(best_h, n_orig, t4,
                                   max_iter=100, topK_start=40, topK_end=100, damping=0.7)
    if c_alt < best_c:
        best_c = c_alt
        best_h = h_alt
    
    # Phase 5: Stochastic restarts to escape local optima
    if time.time() < deadline - 120:
        t5 = min(time.time() + 80, deadline - 120)
        h_sr, c_sr = stochastic_restarts(best_h, n_orig, t5, n_restarts=4, restart_frac=0.5)
        if c_sr < best_c:
            best_c = c_sr
            best_h = h_sr
    
    # Phase 6: Try n=1000 (upsample)
    if time.time() < deadline - 100:
        h_up, c_up, n_up = try_different_n(best_h, best_c, 1000, deadline)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            n_orig = n_up
    
    # Phase 7: Final L-BFGS polish
    if time.time() < deadline - 40:
        t7 = min(time.time() + 30, deadline - 40)
        best_h, best_c = lbfgs_polish(best_h, n_orig, t7,
                                       configs=[(5000, 5000.0), (20000, 50000.0)])
    
    # Phase 8: Top-K focused final
    if time.time() < deadline - 30:
        t8 = min(time.time() + 20, deadline - 30)
        best_h, best_c = topK_lbfgs(best_h, n_orig, t8, top_k=5,
                                     configs=[(10000, 1000.0), (50000, 10000.0)])
    
    # Phase 9: Fine pair search
    if time.time() < deadline - 10:
        t9 = min(time.time() + 15, deadline - 5)
        best_h, best_c = fine_pair_search(best_h, n_orig, t9, center_radius=30)
    
    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
