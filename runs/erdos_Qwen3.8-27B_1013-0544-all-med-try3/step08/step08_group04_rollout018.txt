import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_ls(h, n, alpha):
    """Log-sum-exp objective for SLSQP."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    out = shifted.max()
    if out > 500:
        return float(m + (out + np.log(np.sum(np.exp(shifted - out)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_constraint(h, n):
    return h.sum() - n / 2.0

def _sum_jac(h, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n,)}]
    res = minimize(obj_ls, h, args=(n, alpha), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)

def subgradient_step(h, n, step_size, top_k=5):
    """One projected subgradient step."""
    c = compute_corr(h)
    max_val = c.max()
    
    # Find top_k active shifts
    indices = np.argsort(c)[-top_k:]
    
    # Compute subgradient (average of gradients of active shifts)
    m = np.arange(-(n-1), n)  # length 2n-1
    grad = np.zeros(n)
    for k in indices:
        mk = k - (n-1)
        # grad[i] = 1 - h[i+mk] (if valid) - h[i-mk] (if valid)
        idx_p = np.arange(n) + mk
        valid_p = (idx_p >= 0) & (idx_p < n)
        idx_m = np.arange(n) - mk
        valid_m = (idx_m >= 0) & (idx_m < n)
        g_k = np.ones(n)
        g_k[valid_p] -= h[idx_p[valid_p]]
        g_k[valid_m] -= h[idx_m[valid_m]]
        grad += g_k
    grad /= top_k
    
    # Projected step
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new, max_val

def projected_subgradient(h, n, deadline, top_k=5, initial_step=0.01, decay=0.999):
    """Projected subgradient with decreasing step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = initial_step
    
    while time.time() < deadline:
        h_new, max_val = subgradient_step(h, n, step, top_k)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
        else:
            h = h_new  # still move, but don't update best
        step *= decay
        if step < 1e-8:
            break
    
    return best_h, best_c
