                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None, max_rounds=100):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    best_c = compute_c5(h, n)
    for d in deltas:
        rounds = 0
        while time.time() < deadline and rounds < max_rounds:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - (radius if radius else n))
                hi = min(n, i + (radius if radius else n) + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] += 0.0 - d + d  # revert
                            h[i] += 0
                            h[i] = min(1.0, max(0.0, h[i]))
                            h[j] = min(1.0, max(0.0, h[j]))
            rounds += 1
            if not improved:
                break
    return h, best_c

def gradient_pair_search(h, n, deadline, base_step=0.003):
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000, 20000]
    for alpha in alphas:
        improved_round = 0
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-13:
                break
            g = g / g_norm
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Find best j to swap with based on gradient
                # We want to decrease objective: move mass from high grad to low grad
                best_j = -1
                best_improve = 0
                radius = min(80, n//4)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    move = -base_step * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    if h[i] + move >= 0.0 and h[i] + move <= 1.0 and \
                       h[j] - move >= 0.0 and h[j] - move <= 1.0:
                        h[i] += move; h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            best_improve = 1
                            break
                        else:
                            h[i] -= move; h[j] += move
                if best_improve > 0:
                    break  # restart with new gradient
            if not improved:
                break
    return h, best_c

def interpolate_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial values
    try:

        n_init = len(h_init)

    best_c = compute_c5(best_h, n_init)

    # Phase 1: Quick optimization of parent (n=300)
    t1 = min(t0 + 120, deadline - 500)
    if t1 > t0:
        best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=100)
        best_h, best_c = active_set_cycle(best_h, n_init, t1 + 30, top_k=3)

    # Phase 2: Try higher resolutions
    resolutions = [400, 500, 600]
    candidates = [(best_h, best_c, n_init)]
    
    for n_try in resolutions:
        if time.time() > deadline - 300:
            break
        # Interpolate
        h_try = interpolate_to_resolution(best_h, best_n, n_try)
        # Add small perturbation for diversity
        if time.time() > deadline - 280:
            break
        n_cand = len(h_try)
        
        # Optimize at this resolution
        alloc = min(150, deadline - time.time() - 200)
        if alloc < 40:
            break
        t_phase = time.time() + alloc * 0.5
        h_try, c_try = lbfgs_phase(h_try, n_try, t_phase, max_iters=80)
        
        if time.time() < deadline - 180:
            t_as = time.time() + min(50, deadline - time.time() - 150)
            h_try, c_try = active_set_cycle(h_try, n_try, t_as, top_k=4)
        
        if time.time() < deadline - 160:
            t_sl = time.time() + min(40, deadline - time.time() - 130)
            h_try, c_try = slsqp_phase(h_try, n_try, t_sl, alpha=500000, maxiter=800)
        
        candidates.append((h_try, c_try, n_try))
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
            best_n = n_try

    # Phase 3: Also try a perturbed version at n_init
    if time.time() < deadline - 250:
        h_pert = best_h + rng.normal(0, 0.02, n_init)
        h_pert = project_feasible(h_pert, n_init)
        c_pert = compute_c5(h_pert, n_init)
        if c_pert < 0.40:  # only if perturbation didn't destroy too much
            t_pert = time.time() + min(80, deadline - time() - 200)
            h_pert, c_pert2 = lbfgs_phase(h_pert, n_init, t_pert, max_iters=80)
            if c_pert2 < best_c:
                best_c = c_pert2
                best_h = h_pert
                best_n = n_init

    # Phase 4: Focus on best solution - aggressive optimization
    if time.time() < deadline - 100:
        nf = len(best_h)
        t4 = min(time.time() + 100, deadline - 100)
        best_h, best_c = lbfgs_phase(best_h, nf, t4, max_iters=200)
    
    if time.time() < deadline - 80:
        nf = len(best_h)
        t4b = min(time.time() + 60, deadline - 80)
        best_h, best_c = active_set_cycle(best_h, nf, t4b, top_k=3)
    
    if time.time() < deadline - 60:
        nf = len(best_h)
        t4c = min(time.time() + 50, deadline - 60)
        for a in [200000, 1000000, 5000000]:
            if time.time() > t4c:
                break
            h_t, c_t = slsqp_phase(best_h, nf, time.time() + min(30, deadline - time() - 30),
                                   alpha=a, maxiter=800)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t

    # Phase 5: Gradient descent + steepest descent
    if time.time() < deadline - 80:
        nf = len(best_h)
        t5 = min(time.time() + 50, deadline - 80)
        best_h, best_c = gradient_descent(best_h, nf, t5, lr_init=0.003, alpha_init=500.0)
    
    if time.time() < deadline - 50:
        nf = len(best_h)
        t5b = min(time.time() + 40, deadline - 50)
        best_h, best_c = steep_steepest_descent(best_h, nf, t5b)

    # Phase 6: Gradient-guided pair search
    if time.time() < deadline - 60:
        nf = len(best_h)
        t6 = min(time.time() + 50, deadline - 60)
        best_h, best_c = gradient_pair_search(best_h, nf, t6, base_step=0.002)

    # Phase 7: Standard pair search (very fine steps)
    if time.time() < deadline - 30:
        nf = len(best_h)
        t7 = min(time.time() + 35, deadline - 30)
        best_h, best_c = pair_search(best_h, nf, t7,
                                     deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                     radius=50, max_rounds=200)

    # Phase 8: Final micro-optimization
    if time.time() < deadline - 5:
        nf = len(best_h)
        best_h, best_c = active_set_cycle(best_h, nf, deadline - 3, top_k=2)
        
        # Tiny pair search
        if time.time() < deadline - 2:
            best_h, best_c = pair_search(best_h, nf, deadline - 1,
                                         deltas=[0.0001, 0.00005, 0.00002, 0.00001],
                                         radius=30, max_rounds=500)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
