            dk += h_jk2

    #   Change: -h_old[i+k] [using new h[i+k] if i+k==j]
    # Change due to h[j]: 1->0 (j as second index h[t], t=j)
    #   Valid if 0 <= j <= n-1-k, i.e., k <= n-1-j
    #   Change: +h_new[j+k] where h_new[j+k] = 1 if j+k==i else h[j+k]

            h_jk2 = 1.0 if (j + k == i) else h[j + k]
            dk += h_jk2
        delta[n - 1 - k] += dk
    
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def greedy_search(h, n, deadline, max_sweeps=5000):
    h = to_binary(h, n).copy()
    dx = 2.0 / n
    cur_c = float(np.correlate(h, 1 - h, mode='full').max())
    best_c = cur_c * dx
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                nc = float(np.correlate(h, 1 - h, mode='full').max())
                if nc < cur_c - 1e-13:
                    cur_c = nc
                    improved = True
                    ones[ones.index(j)] = i
                    break
                else:
                    h[i] = 0.0
                    h[j] = 1.0
            if improved:
                break
        if not improved:
            break
        if cur_c * dx < best_c:
            best_c = cur_c * dx
            best_h = h.copy()
    return best_h, best_c

def steepest_search(h, n, deadline, max_steps=500):
    h = to_binary(h, n).copy()
    dx = 2.0 / n
    cur_c = float(np.correlate(h, 1 - h, mode='full').max())
    best_c = cur_c * dx
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        best_delta = 0.0
        best_i = -1
        best_j = -1
        for i in zeros:
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                nc = float(np.correlate(h, 1 - h, mode='full').max())
                d = cur_c - nc
                if d > best_delta:
                    best_delta = d
                    best_i = i
                    best_j = j
                h[i] = 0.0
                h[j] = 1.0
        if best_delta < 1e-13:
            break
        h[best_i] = 1.0
        h[best_j] = 0.0
        cur_c = float(np.correlate(h, 1 - h, mode='full').max())
        ones[ones.index(best_j)] = best_i
        if cur_c * dx < best_c:
            best_c = cur_c * dx
            best_h = h.copy()
    return best_h, best_c

def simulated_annealing(h, n, deadline, T0, T_floor, seed):
    rng = np.random.RandomState(seed)
    h = to_binary(h, n).copy()
    dx = 2.0 / n
    cur_c = float(np.correlate(h, 1 - h, mode='full').max()) * dx
    best_c = cur_c
    best_h = h.copy()
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    n_iters = 800000
    gamma = (T_floor / T0) ** (1.0 / n_iters)
    T = T0
    greedy_every = 20000
    it = 0
    while it < n_iters:
        if time.time() > deadline:
            break
        T = max(T * gamma, T_floor)
        i = zeros_list[rng.randint(len(zeros_list))]

        h[j] = 0.0

        delta = new_c - cur_c

            cur_c = new_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
