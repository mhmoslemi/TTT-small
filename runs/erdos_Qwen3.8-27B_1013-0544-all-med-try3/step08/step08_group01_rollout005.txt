import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, shift_idx):
    """Gradient of correlation at a specific shift index in 'full' mode."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def topk_obj(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def topk_grad(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-12:

        k = idx - (n - 1)
        wm = wi * dx

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def topk_obj(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def topk_grad(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_lbfgs(h, n, deadline, alpha_list, maxiter_list, topk_list=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alpha_list):
        if time.time() > deadline - 5:
            break
        maxiter = maxiter_list[i] if i < len(maxiter_list) else 3000
        try:
            if topk_list is not None and i < len(topk_list) and topk_list[i] is not None:
                tk = topk_list[i]
                c_vec = compute_c5_vec(best_h, n)
                active = np.argsort(-c_vec)[:tk]
                res = minimize(topk_obj, best_h, jac=topk_grad,
                              args=(n, active, alpha), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxiter * 2, 'maxls': 40})
            else:
                res = minimize(smooth_obj, best_h, jac=smooth_grad,
                              args=(n, alpha), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxiter * 2, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def topk_focused_opt(h, n, deadline):

def block_local_search(h, n, deadline, block_size=10):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    n_blocks = n // block_size
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for bi in range(n_blocks):
                if time.time() > deadline:
                    return best_h, best_c
                base_i = bi * block_size
                end_i = min(base_i + block_size, n)
                block_i = slice(base_i, end_i)
                for bj in range(bi + 1, n_blocks):
                    if time.time() > deadline:
                        return best_h, best_c
                    base_j = bj * block_size
                    end_j = min(base_j + block_size, n)
                    block_j = slice(base_j, end_j)
                    # Move mass from block_i to block_j
                    if np.all(best_h[block_i] - d >= 0) and np.all(best_h[block_j] + d <= 1):
                        best_h[block_i] -= d
                        best_h[block_j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[block_i] += d
                            best_h[block_j] -= d
    return best_h, best_c

def gradient_descent_momentum(h, n, deadline, lr=0.001, momentum=0.9, topk=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    v = np.zeros(n)

        new_h = best_h + v

                momentum *= 0.95
    return best_h, best_c

def perturbation_restart(h, n, deadline, seed_rng, magnitude=0.02):
    h_pert = h + seed_rng.random(n) * magnitude - magnitude / 2
    h_pert = project_feasible(h_pert, n)
    c = compute_c5(h_pert, n)
    h_pert, c = run_lbfgs(h_pert, n, deadline, [5000, 50000], [3000, 3000])

def make_structural_init(n, pattern='sine1'):
    x = np.linspace(0, 2, n)
    if pattern == 'sine1':
        h = 0.5 + 0.25 * np.sin(np.pi * x)
    elif pattern == 'sine2':
        h = 0.5 + 0.25 * np.sin(2 * np.pi * x)
    elif pattern == 'sine3':
        h = 0.5 + 0.2 * np.sin(3 * np.pi * x)
    elif pattern == 'triangle':
        h = 0.5 + 0.3 * np.sinc(x - 1)
    elif pattern == 'blocks':
        h = 0.5 + 0.3 * np.sign(np.sin(4 * np.pi * x))
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)
