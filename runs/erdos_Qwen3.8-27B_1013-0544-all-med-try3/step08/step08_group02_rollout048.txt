        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
    val = m + logsum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Efficient gradient computation
    for mm in range(2 * n - 1):
        wi = w[mm]
        if wi < 1e-14:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    if smx > 500:
        logsum = smx + np.log(np.sum(np.exp(shifted - smx)))
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
    val = mx + logsum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    w = np.exp(shifted - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
    return float(m + logsum / alpha)

def integral_fun(x, n):
    return x.sum() - n / 2.0

def integral_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiters and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(500, int(remaining * 3))
        try:
            res = minimize(soft_max_obj, best_h, jac=soft_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxit*10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_k=10, n_rounds=15):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [5000, 20000, 100000, 500000, 2000000]
    for rnd in range(n_rounds):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Adaptive threshold
        thr = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= thr)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxit*10,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': integral_fun, 'jac': integral_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def local_search(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in step_sizes:
        rounds = 0
        while rounds < max_rounds and time.time() < deadline:
            improved = False
            # Random pairs
            for _ in range(5000):
                if time.time() > deadline:
                    break
                i, j = np.random.randint(0, n, 2)
                if i == j:
                    continue
                # Try both directions
                for d in [step, -step]:
                    hi, hj = best_h[i] + d, best_h[j] - d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        best_h[i], best_h[j] = hi, hj
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i], best_h[j] = hi - d, hj + d
            rounds += 1
            if not improved:
                break
    return best_h, best_c

def local_search_deterministic(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in step_sizes:
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                for d in [step, -step]:
                    hi_val, hj_val = best_h[i] + d, best_h[j] - d
                    if 0 <= hi_val <= 1 and 0 <= hj_val <= 1:
                        best_h[i], best_h[j] = hi_val, hj_val
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                        else:
                            best_h[i], best_h[j] = hi_val - d, hj_val + d
    return best_h, best_c

def make_periodic_start(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        if phase < duty:
            h[i] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        h[rng.randint(n)] += 0.5
        h = project_feasible(h, n)
    return project_feasible(h, n)

def subgradient_hinge(h, n, deadline, n_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        # Use all constraints above threshold
        thr = c_max - max(1e-6, 0.005 * c_max)
        active = np.where(c_vec >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:]) * dx_val
                grad[k:] += (-h[:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * dx_val
                grad[:nv] += (-h[kk:]) * dx_val
        grad /= len(active)
        h_new = np.clip(h - eta * grad, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.05)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c
