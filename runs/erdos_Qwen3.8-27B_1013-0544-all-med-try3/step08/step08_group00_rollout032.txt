def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot <= 1e-15:

            h += (diff / tot) * h

            h[idx] = min(1.0, h[idx] + diff)
        elif diff < 0:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

```python
import time
import math
import numpy as np
from scipy.optimize import linprog

TARGET_C = 0.3808
SEARCH_TARGET_C = 0.3760


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(250):
        diff = target - h.sum()
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot <= 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h


def compute_c5_values(h, n):
    hf = np.asarray(h, dtype=float)
    c = np.correlate(hf, 1.0 - hf, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def compute_raw_max(h, n):
    hf = np.asarray(h, dtype=float)
    c = np.correlate(hf, 1.0 - hf, mode="full")
    return float(np.max(c))


def compute_corr_own_float(h, n):
    hf = np.asarray(h, dtype=float)
    return np.correlate(hf, 1.0 - hf, mode="full")[::-1].copy()


def compute_corr_own_int(bits, n):
    b = np.asarray(bits, dtype=np.int32)
    v = 1 - b
    return np.correlate(b, v, mode="full")[::-1].astype(np.int32)


def bits_from_values(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if len(h) != n:
        if len(h) < 2:
            val = float(h[0]) if len(h) > 0 else 0.5
            h = np.full(n, val, dtype=float)
        else:
            x_old = np.linspace(0.0, 2.0, len(h), dtype=float)
            x_new = np.linspace(0.0, 2.0, n, dtype=float)
            h = np.interp(x_new, x_old, h)
    h = np.clip(h, 0.0, 1.0)
    target = n // 2
    bits = (h >= 0.5).astype(np.int8)
    s = int(bits.sum())
    if s > target:
        ones = np.nonzero(bits == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[: s - target]:
            bits[idx] = 0
    elif s < target:
        zeros = np.nonzero(bits == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[: target - s]:
            bits[idx] = 1
    return bits


def markov_bits(n, q, seed):
    rng = np.random.default_rng(seed)
    bits = np.empty(n, dtype=np.int8)
    b = int(rng.random() < 0.5)
    for i in range(n):
        bits[i] = b
        if rng.random() < q:
            b ^= 1
    return bits


def adjust_bits_sum_score(bits, n, rng):
    bits = np.array(bits, dtype=np.int8).copy()
    target = n // 2
    s = int(bits.sum())
    if s == target:
        return bits
    radius = min(max(1, n // 20), 60)
    ker = np.ones(2 * radius + 1, dtype=np.int32)
    while s > target:
        ones = np.nonzero(bits == 1)[0]
        if len(ones) == 0:
            break
        ws = np.convolve(bits.astype(np.int32), ker, mode="same")
        scores = ws[ones].astype(float) + rng.random(len(ones)) * 0.1
        i = int(ones[int(np.argmin(scores))])
        bits[i] = 0
        s -= 1
    while s < target:
        zeros = np.nonzero(bits == 0)[0]
        if len(zeros) == 0:
            break
        ws = np.convolve(bits.astype(np.int32), ker, mode="same")
        scores = -ws[zeros].astype(float) + rng.random(len(zeros)) * 0.1
        i = int(zeros[int(np.argmin(scores))])
        bits[i] = 1
        s += 1
    return bits


def markov_exact_bits(n, q, seed):
    rng = np.random.default_rng(seed)
    bits = markov_bits(n, q, seed)
    bits = adjust_bits_sum_score(bits, n, rng)
    return bits


def perturb_bits(bits, n, rng, steps):
    bits = np.array(bits, dtype=np.int8).copy()
    for _ in range(int(steps)):
        ones = np.nonzero(bits == 1)[0]
        zeros = np.nonzero(bits == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = int(rng.choice(ones))
        b = int(rng.choice(zeros))
        bits[a] = 0
        bits[b] = 1
    return bits


def precompute_flip_deltas(bits, n):
    L = 2 * n - 1
    FP = np.zeros((n, L), dtype=np.int16)
    FM = np.zeros((n, L), dtype=np.int16)
    b16 = bits.astype(np.int16)
    for p in range(n):
        if p > 0:
            left = b16[:p]
            oml = 1 - left
            sl_plus = slice(n - 1 + p, n - 1, -1)
            sl_minus = slice(n - 1 - p, n - 1, 1)
            FP[p, sl_plus] -= left
            FP[p, sl_minus] += oml
            FM[p, sl_plus] += left
            FM[p, sl_minus] -= oml
        if p < n - 1:
            right = b16[p + 1:]
            omr = 1 - right
            length = n - p - 1
            sl_plus = slice(n, n + length, 1)
            sl_minus = slice(n - 2, p - 1, -1)
            FP[p, sl_plus] += omr
            FP[p, sl_minus] -= right
            FM[p, sl_plus] -= omr
            FM[p, sl_minus] += right
    return FP, FM


def bit_near_score(corr, M):
    if M <= 0:
        return 0
    tol = max(2.0, 0.02 * float(M))
    return int(np.sum(corr[corr >= M - tol]))


def binary_build_candidates(bits, corr, n, M, target, rng, max_pairs=250):
    cands = []
    seen = set()
    order = np.argsort(-corr)
    lags = []
    for ii in order:
        idx = int(ii)
        if len(lags) >= 8:
            break
        if corr[idx] < max(target, M - 4):
            break
        lags.append(idx)
    if not lags:
        lags = [int(np.argmax(corr))]
    per_lag = max(10, max_pairs // max(1, len(lags)))
    for idx in lags:
        l = idx - (n - 1)
        if l >= 0:
            d = l
            m = n - d
            pos = np.nonzero((bits[:m] == 1) & (bits[d:] == 0))[0]
            if len(pos):
                if len(pos) > per_lag:
                    pos = rng.choice(pos, per_lag, replace=False)
                for pp in pos:
                    pp = int(pp)
                    a = pp
                    b = pp + d
                    if 0 <= a < n and 0 <= b < n and bits[a] == 1 and bits[b] == 0:
                        key = a * n + b
                        if key not in seen:
                            seen.add(key)
                            cands.append((a, b))
        else:
            d = -l
            m = n - d
            pos = np.nonzero((bits[d:] == 1) & (bits[:m] == 0))[0] + d
            if len(pos):
                if len(pos) > per_lag:
                    pos = rng.choice(pos, per_lag, replace=False)
                for pp in pos:
                    pp = int(pp)
                    a = pp
                    b = pp - d
                    if 0 <= a < n and 0 <= b < n and bits[a] == 1 and bits[b] == 0:
                        key = a * n + b
                        if key not in seen:
                            seen.add(key)
                            cands.append((a, b))
    i = 0
    while i < n - 1:
        if bits[i] != bits[i + 1]:
            if bits[i] == 1:
                a, b = i, i + 1
            else:
                a, b = i + 1, i
            key = a * n + b
            if key not in seen:
                seen.add(key)
                cands.append((a, b))
        i += 1
    if len(cands) < 80:
        ones = np.nonzero(bits == 1)[0]
        zeros = np.nonzero(bits == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            need = max(0, 120 - len(cands))
            for _ in range(need):
                a = int(rng.choice(ones))
                b = int(rng.choice(zeros))
                key = a * n + b
                if key not in seen:
                    seen.add(key)
                    cands.append((a, b))
    return cands


def bits_local_search(bits, n, target_raw, seed, max_time, iters_limit=400):
    bits = np.array(bits, dtype=np.int8).copy()
    rng = np.random.default_rng(seed)
    bits = adjust_bits_sum_score(bits, n, rng)
    corr = compute_corr_own_int(bits, n)
    M = int(corr.max())
    best_bits = bits.copy()
    best_corr = corr.copy()
    best_M = M
    best_score = bit_near_score(corr, M)

    if M <= target_raw:
        return best_bits, best_corr

    deadline = time.time() + max(1.0, float(max_time))
    L = 2 * n - 1
    work = np.zeros(L, dtype=np.int32)
    iters = 0
    no_improve = 0
    kicks = 0

    while time.time() < deadline and M > target_raw and iters < iters_limit:
        if time.time() > deadline - 1.5:
            break
        iters += 1
        FP, FM = precompute_flip_deltas(bits, n)
        near_tol = max(2.0, 0.02 * M)
        near_mask = corr >= M - near_tol
        old_score = int(np.sum(corr[near_mask]))
        cands = binary_build_candidates(bits, corr, n, M, target_raw, rng, 260)
        best_key = None
        best_move = None
        best_nm = M + 1
        best_sc = old_score + 1

        for a, b in cands:
            if a == b or not (0 <= a < n and 0 <= b < n):
                continue
            if bits[a] != 1 or bits[b] != 0:
                continue
            d = abs(a - b)
            np.add(corr, FM[a], out=work)
            work += FP[b]
            work[n - 1 + d] += 1
            work[n - 1 - d] += 1
            nm = int(work.max())
            if nm > M:
                continue
            sc = int(work[near_mask].sum())
            key = (nm, sc)
            if best_key is None or key < best_key:
                best_key = key
                best_move = (a, b)
                best_nm = nm
                best_sc = sc

        if (
            best_move is not None
            and (best_nm < M or (best_nm == M and best_sc < old_score))
        ):
            a, b = best_move
            if bits[a] == 1 and bits[b] == 0:
                bits[a] = 0
                bits[b] = 1
                corr = compute_corr_own_int(bits, n)
                M = int(corr.max())
                cur_score = bit_near_score(corr, M)
                if M < best_M or (M == best_M and cur_score < best_score):
                    best_bits = bits.copy()
                    best_corr = corr.copy()
                    best_M = M
                    best_score = cur_score
                no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8 and kicks < 3:
                ones = np.nonzero(bits == 1)[0]
                zeros = np.nonzero(bits == 0)[0]
                kick_made = 0
                for _ in range(30):
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    a = int(rng.choice(ones))
                    b = int(rng.choice(zeros))
                    bits[a] = 0
                    bits[b] = 1
                    corr2 = compute_corr_own_int(bits, n)
                    M2 = int(corr2.max())
                    if M2 <= M + 1:
                        corr = corr2
                        M = M2
                        cur_score = bit_near_score(corr, M)
                        if M < best_M or (M == best_M and cur_score < best_score):
                            best_bits = bits.copy()
                            best_corr = corr.copy()
                            best_M = M
                            best_score = cur_score
                        kicks += 1
                        kick_made = 1
                        break
                    else:
                        bits[a] = 1
                        bits[b] = 0
                if kick_made:
                    no_improve = 0
            elif M > best_M and no_improve > 20:
                bits = best_bits.copy()
                corr = best_corr.copy()
                M = best_M
                no_improve = 0
            elif no_improve > 45:
                break

    return best_bits, best_corr


def allocate_start_times(count, total_time):
    base = [80, 60, 45, 30, 25, 20, 20, 15, 15, 15, 15, 15, 15, 15, 15]
    if count <= len(base):
        arr = base[:count]
    else:
        arr = base + [15.0] * (count - len(base))
    arr = [float(x) for x in arr]
    s = sum(arr)
    total_time = max(1.0, float(total_time))
    if s <= 0.0:
        return [max(2.0, total_time / max(1, count))] * count
    if s > total_time:
        factor = total_time / s
        arr = [max(3.0, x * factor) for x in arr]
        s2 = sum(arr)
        if s2 > total_time:
            factor = total_time / s2
            arr = [max(1.0, x * factor) for x in arr]
    return arr


def try_binary(parent, n, seed, deadline, target_C):
    n = int(n)
    if n % 2 == 1:
        n -= 1
    if n < 20:
        return None, None
    target_raw = int(math.floor(target_C * n / 2.0 - 1e-9))
    rng = np.random.default_rng(seed)

    starts = []
    starts.append(bits_from_values(parent, n))
    lambdas = [5, 7, 9, 12, 15, 20, 30, 45]
    for lam in lambdas:
        q = lam / (2.0 * float(n))
        for sidx in range(2):
            starts.append(markov_exact_bits(n, q, seed + 97 * lam + 131 * sidx))

    best_bits = None
    best_raw = 10**12

    corr0 = compute_corr_own_int(starts[0], n)
    m0 = int(corr0.max())
    best_bits = starts[0].copy()
    best_raw = m0
    if best_raw <= target_raw:
        return best_bits.astype(float), float(best_raw) * (2.0 / float(n))

    total_time = max(1.0, float(deadline - time.time()))
    max_starts = max(2, min(len(starts), int(total_time // 8) + 1))
    starts = starts[:max_starts]
    times = allocate_start_times(len(starts), total_time)

    idx = 0
    for start_bits, t_alloc in zip(starts, times):
        if time.time() >= deadline - 2.0:
            break
        t_alloc = max(2.0, min(float(t_alloc), max(1.0, deadline - time.time())))
        if t_alloc <= 1.0:
            break
        bits, corr = bits_local_search(
            start_bits, n, target_raw, seed + 19 * idx, t_alloc, iters_limit=350
        )
        M = int(corr.max())
        if M < best_raw:
            best_raw = M
            best_bits = bits.copy()
        if best_raw <= target_raw:
            return best_bits.astype(float), float(best_raw) * (2.0 / float(n))

        if best_bits is not None and time.time() < deadline - 25.0:
            for k in range(2):
                if time.time() >= deadline - 12.0:
                    break
                pert = perturb_bits(best_bits, n, rng, 70 + 35 * k)
                pert = adjust_bits_sum_score(pert, n, rng)
                t2 = max(4.0, min(35.0, float(deadline - time.time() - 8.0)))
                bits2, corr2 = bits_local_search(
                    pert, n, target_raw, seed + 500 + 17 * idx + k, t2, iters_limit=220
                )
                M2 = int(corr2.max())
                if M2 < best_raw:
                    best_raw = M2
                    best_bits = bits2.copy()
                if best_raw <= target_raw:
                    return best_bits.astype(float), float(best_raw) * (2.0 / float(n))
        idx += 1

    if best_bits is None:
        best_bits = starts[0]
        best_raw = int(compute_corr_own_int(best_bits, n).max())
    return best_bits.astype(float), float(best_raw) * (2.0 / float(n))


def shift_grad_own(h, n, l):
    g = np.zeros(n, dtype=float)
    d = abs(int(l))
    if d == 0:
        return 1.0 - 2.0 * h
    m = n - d
    if l > 0:
        g[0:m] += 1.0 - h[d:]
        g[d:n] += -h[0:m]
    else:
        g[d:n] += 1.0 - h[0:m]
        g[0:m] += -h[d:]
    return g


def lp_polish(h, n, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_raw = compute_raw_max(h, n)
    n = int(n)
    max_solves = 90 if n <= 600 else 60
    solves = 0

    base_rhos = [0.05, 0.10, 0.20, 0.40, 0.80, 1.00]
    for base_rho in base_rhos:
        if time.time() > deadline - 4.0 or solves >= max_solves:
            break
        h = best_h.copy()
        current_raw = best_raw
        rho = float(base_rho)
        for _ in range(120):
            if time.time() > deadline - 4.0 or solves >= max_solves:
                break
            S = compute_corr_own_float(h, n)
            M = float(S.max())
            tol = max(0.25, 0.0075 * M)
            active = np.where(S >= M - tol)[0]
            if len(active) == 0:
                active = np.argsort(-S)[:10]
            cap = 90 if n > 600 else 130
            if len(active) > cap:
                active = active[np.argsort(-S[active])[:cap]]
            if len(active) == 0:
                break

            m = len(active)
            A_ub = np.zeros((m, n + 1), dtype=float)
            b_ub = np.empty(m, dtype=float)
            for r, idx in enumerate(active):
                l = int(idx) - (n - 1)
                g = shift_grad_own(h, n, l)
                A_ub[r, 0:n] = g
                A_ub[r, n] = -1.0
                b_ub[r] = -float(S[idx])

            A_eq = np.zeros((1, n + 1), dtype=float)
            A_eq[0, 0:n] = 1.0
            b_eq = np.zeros(1, dtype=float)

            bounds = []
            for i in range(n):
                lo = -float(h[i])
                hi = 1.0 - float(h[i])
                lo = max(lo, -rho)
                hi = min(hi, rho)
                if lo > hi:
                    mid = min(max(h[i], 0.0), 1.0)
                    lo = hi = mid
                bounds.append((lo, hi))
            bounds.append((-10.0, 3.0 * float(n) + 10.0))

            c_obj = np.zeros(n + 1, dtype=float)
            c_obj[-1] = 1.0

            try:
                res = linprog(
                    c_obj,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method="highs",
                )
            except Exception:
                res = None
            solves += 1

            if res is None or not res.success:
                rho *= 0.5
                if rho < 0.005:
                    break
                continue

            delta = np.asarray(res.x[0:n], dtype=float)
            if np.linalg.norm(delta) < 1e-8:
                break

            cand_h = None
            cand_raw = current_raw + 1.0
            for theta in (1.0, 0.5, 0.25, 0.10, 0.03, 0.005):
                h_try = project_feasible(h + theta * delta, n)
                r_try = compute_raw_max(h_try, n)
                if r_try < cand_raw:
                    cand_raw = r_try
                    cand_h = h_try
            if cand_h is not None and cand_raw < current_raw - 1e-9:
                h = cand_h
                current_raw = cand_raw
                if current_raw < best_raw - 1e-10:
                    best_h = h.copy()
                    best_raw = current_raw
                if M - current_raw < 1e-6:
                    rho = min(rho, 0.5 * base_rho)
            else:
                rho *= 0.5
                if rho < 0.005:
                    break
    return best_h, best_raw


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s)) - 5.0
    np.random.seed(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = markov_exact_bits(500, 0.01, int(seed)).astype(float)

    n_initial = int(len(parent))
    if n_initial % 2 == 1:
        n_initial -= 1
    if n_initial < 20:
        n_initial = 50

    parent = project_feasible(parent, n_initial)
    best_h = parent.copy()
    best_n = n_initial
    best_c = compute_c5_values(best_h, best_n)

    cont_reserve = max(70.0, min(520.0, 0.42 * float(budget_s)))
    discrete_deadline = max(t0 + 20.0, deadline - cont_reserve)

    n_trials = []
    candidate_ns = [
        n_initial,
        500,
        1000,
        max(50, min(1000, n_initial // 2)),
        max(50, min(1000, n_initial * 2)),
    ]
    for nn in candidate_ns:
        nn = int(nn)
        nn = max(50, min(1000, nn))
        if nn % 2 == 1:
            nn -= 1
        if nn >= 20 and nn not in n_trials:
            n_trials.append(nn)
    if n_initial in n_trials:
        n_trials.remove(n_initial)
        n_trials.insert(0, n_initial)
    else:
        n_trials.sort()

    idx_trial = 0
    for n_tri in n_trials:
        if time.time() >= discrete_deadline - 5.0:
            break
        rem = max(1.0, discrete_deadline - time.time())
        if rem < 20.0:
            break
        per_trial_deadline = time.time() + max(
            18.0,
            rem / max(1, len(n_trials) - idx_trial),
        )
        per_trial_deadline = min(per_trial_deadline, discrete_deadline - 2.0)
        try:
            h_bin, c_bin = try_binary(
                parent, n_tri, int(seed) + 1000 * idx_trial, per_trial_deadline,
                SEARCH_TARGET_C,
            )
        except Exception:
            h_bin, c_bin = None, None
        if h_bin is not None:
            h_bin = project_feasible(h_bin, n_tri)
            c_bin = compute_c5_values(h_bin, n_tri)
            if c_bin < best_c - 1e-12:
                best_h = h_bin
                best_n = n_tri
                best_c = c_bin
        idx_trial += 1

    if time.time() < deadline - 10.0 and best_n > 0:
        try:
            h_lp, c_lp_raw = lp_polish(best_h, best_n, deadline - 5.0)
            h_lp = project_feasible(h_lp, best_n)
            c_lp = compute_c5_values(h_lp, best_n)
            if c_lp < best_c - 1e-12:
                best_h = h_lp
                best_c = c_lp
        except Exception:
            pass

    if time.time() < deadline - 10.0 and best_c > SEARCH_TARGET_C and best_n > 0:
        try:
            base_bits = bits_from_values(best_h, best_n)
            rng = np.random.default_rng(int(seed) + 77777)
            base_bits = adjust_bits_sum_score(base_bits, best_n, rng)
            rem = max(1.0, deadline - time.time() - 3.0)
            for k in range(3):
                if time.time() >= deadline - 3.0:
                    break
                pert = perturb_bits(best_bits if False else base_bits, best_n, rng, 50 + 20 * k)
                pert = adjust_bits_sum_score(pert, best_n, rng)
                bits2, corr2 = bits_local_search(
                    pert, best_n,
                    int(math.floor(SEARCH_TARGET_C * best_n / 2.0 - 1e-9)),
                    int(seed) + 9000 + k,
                    max(4.0, (deadline - time.time()) // 2.0 - 2.0),
                    iters_limit=120,
                )
                h2 = project_feasible(bits2.astype(float), best_n)
                c2 = compute_c5_values(h2, best_n)
                if c2 < best_c - 1e-12:
                    best_h = h2
                    best_c = c2
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_n = int(len(best_h))
    best_c = compute_c5_values(best_h, best_n)
    return best_h, best_c, best_n
```
