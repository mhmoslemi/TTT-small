        for k in top_ks:
            grad += vectorized_grad_at_k(h, n, k)
        grad /= len(top_ks)
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:

                found = True
                no_improve = 0
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
        
        if not found:
            lr *= 0.95
            no_improve += 1
            if lr < 1e-10 or no_improve > 2000:

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_sa(h, n, deadline, max_iter=500000):
    """Simulated annealing for binary solutions"""
    target = n // 2
    ones = set(np.where(h >= 0.5)[0][:target].tolist())
    if len(ones) < target:
        # Fill remaining
        remaining = set(range(n)) - ones
        for i in range(target - len(ones)):
            ones.add(remaining.pop())
    
    h_bin = np.zeros(n)
    for i in ones:
        h_bin[i] = 1.0
    
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    
    T = 0.1
    T_min = 1e-10
    cooling = 0.9999
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if T < T_min:
            break
        
        # Random swap
        a = np.random.randint(n)
        b = np.random.randint(n)
        while b == a:
            b = np.random.randint(n)
        
        if (a in ones) == (b in ones):
            continue
        
        # Perform swap
        if a in ones:
            ones.remove(a)
            ones.add(b)
        else:
            ones.remove(b)
            ones.add(a)
        
        h_bin[a], h_bin[b] = 1.0 - h_bin[a], 1.0 - h_bin[b]
        c = compute_c5(h_bin, n)
        
        if c < best_c:
            best_c = c
            best_h = h_bin.copy()
        elif c < best_c or np.random.random() < np.exp(-(c - best_c) / T):
            # Accept (keep)
            pass
        else:
            # Reject
            if a in ones:
                ones.remove(a)
                ones.add(b)
            else:
                ones.remove(b)
                ones.add(a)
            h_bin[a], h_bin[b] = 1.0 - h_bin[a], 1.0 - h_bin[b]
        
        T *= cooling
    
    return best_h, best_c

The subgradient at δ is {sum of d[k] * p_k : p_k ≥ 0, sum p_k = 1, p_k > 0 only for k where g[k]+δ*d[k] = max}.

For a simple bisection: if at the current max, d[k_max] > 0, then increasing δ increases the max, so the minimum is to the left. If d[k_max] < 0, the minimum is to the right.

This is clean and correct. But I need `vectorized_grad_at_k_all` which computes the gradient of ALL g[k] with respect to h[i]:

```python
def grad_all_k(h, n, i):
    """Compute ∂g[k]/∂h[i] for all k (0 to 2n-2)"""
    dx = 2.0 / n
    result = np.zeros(2*n-1)
    for k in range(2*n-1):
        s = k - (n-1)
        coeff = 0.0
        if 0 <= i - s < n:
            coeff += (1.0 - h[i-s])
        if 0 <= i + s < n:
            coeff -= h[i+s]
        result[k] = coeff * dx
    return result
```

This is O(n) for a single i (since 2n-1 ≈ 2n iterations, each O(1)). For all n coordinates, it's O(n²) to precompute the full Jacobian.

Actually, I can precompute the full Jacobian J[i,k] = ∂g[k]/∂h[i] in a structured way:

        i_cand = int(np.argmin(grad_worst))
