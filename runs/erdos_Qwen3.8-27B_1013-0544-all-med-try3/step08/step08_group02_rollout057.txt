    c_vec = compute_c5_vec(h, n)
    m = c_vec.max()

    mx = sh.max()

    for mi in range(2 * n - 1):

        if k >= 0:

            grad[k:n] += wm * (-h[0:nv])

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def frank_wolfe_step(h, n, grad):
    """Frank-Wolfe linear minimization oracle: min grad^T h s.t. 0<=h<=1, sum(h)=n/2."""
    n_ones = int(n / 2)
    h_lp = np.zeros(n)
    sorted_idx = np.argsort(grad)

        # Actually for odd n, n/2 is not integer, so we need one fractional
        h_lp[sorted_idx[n_ones]] = 2.0 * (n / 2.0 - n_ones)
    return h_lp

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):

def frank_wolfe(h, n, deadline, alpha_schedule=None, max_iter=None):
    """Frank-Wolfe method for the smooth max problem."""
    if alpha_schedule is None:
        alpha_schedule = [500, 2000, 10000, 50000, 200000, 1000000]
    if max_iter is None:
        max_iter = 20000
    
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for alpha_idx, alpha in enumerate(alpha_schedule):
        if time.time() > deadline - 5:
            break
        
        # Determine iterations for this alpha
        remaining_budget = deadline - time.time()
        alphas_left = len(alpha_schedule) - alpha_idx
        iters = max(500, int(remaining_budget * 8 * (1.0 / alphas_left)))
        iters = min(iters, max_iter)
        
        for it in range(iters):
            if time.time() > deadline - 5:

            grad = grad_smooth_max(h, n, alpha)
            
            # FW linear oracle: min grad^T h s.t. 0<=h<=1, sum(h)=n/2
            h_lp = np.zeros(n)
            sorted_idx = np.argsort(grad)

def pair_search(h, n, deadline, radius=50):
    """Fine pair swap search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if abs(i - j) > radius:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c

                        else:

def slqp_polish(h, n, deadline, alpha=200000, maxiter=1000):
    """SLSQP with equality constraint."""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def obj_grad(x):
        return grad_smooth_max(x, n, alpha)
    
    def con_fun(x):
        return x.sum() - target
    
    def con_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)

    n_hi = 1000
    h_hi = upsample(best_h, best_n, n_hi)
    c_hi = compute_c5(h_hi, n_hi)
    
    p4_end = min(time.time() + 200, deadline - 250)
    h4, c4 = frank_wolfe(h_hi, n_hi, p4_end,
        alpha_schedule=[500, 2000, 10000, 50000, 200000, 1000000])
    if c4 < best_c:
        best_c, best_h, best_n = c4, h4, n_hi
    elif c4 < c_hi:
        # Keep the improved hi-res even if not better than overall best
        pass
    
    # Phase 5: Active set at best resolution (100s)
    if time.time() < deadline - 150:
        p5_end = min(time.time() + 100, deadline - 130)
        h5, c5 = active_set_opt(best_h, best_n, p5_end, top_k=4, alpha=500000)
        if c5 < best_c:
            best_c, best_h, best_n = c5, h5, best_n
    
    # Phase 6: L-BFGS polish at best resolution (80s)
    if time.time() < deadline - 100:
        p6_end = min(time.time() + 80, deadline - 80)
        h6, c6 = lbfgs_smooth(best_h, best_n, p6_end,
            alphas=[100000, 500000, 2000000], mu=0.005)
        if c6 < best_c:
            best_c, best_h, best_n = c6, h6, best_n
    
    # Phase 7: Try n=2000 if time permits (150s)
    n_vhi = 2000
    if time.time() < deadline - 200:
        h_vhi = upsample(best_h, best_n, n_vhi)
        c_vhi = compute_c5(h_vhi, n_vhi)
        p7_end = min(time.time() + 150, deadline - 120)
        h7, c7 = frank_wolfe(h_vhi, n_vhi, p7_end,
            alpha_schedule=[1000, 5000, 20000, 100000, 500000])
        if c7 < best_c:
            best_c, best_h, best_n = c7, h7, n_vhi
        # Also try active set at n_vhi
        if time.time() < deadline - 100:
            h7b, c7b = active_set_opt(h_vhi if c7 >= best_c else h7, n_vhi,
                deadline - 60, top_k=4, alpha=200000)
            if c7b < best_c:
                best_c, best_h, best_n = c7b, h7b, n_vhi
    
    # Phase 8: Random restarts from best (80s)
    if time.time() < deadline - 80:
        p8_end = min(time.time() + 80, deadline - 60)
        h8, c8 = random_restarts(best_h, best_n, p8_end, n_restarts=6, pert_scale=0.03)
        if c8 < best_c:
            best_c, best_h, best_n = c8, h8, best_n
    
    # Phase 9: Subgradient refinement (60s)
    if time.time() < deadline - 50:
        p9_end = min(time.time() + 60, deadline - 40)
        h9, c9 = subgrad_descent(best_h, best_n, p9_end, n_iter=5000)
        if c9 < best_c:
            best_c, best_h, best_n = c9, h9, best_n
    
    # Phase 10: Pair search (80s)
    if time.time() < deadline - 40:
        p10_end = min(time.time() + 80, deadline - 20)
        h10, c10 = pair_search(best_h, best_n, p10_end, radius=40)
        if c10 < best_c:
            best_c, best_h, best_n = c10, h10, best_n
    
    # Phase 11: SLSQP polish (40s)
    if time.time() < deadline - 15:
        p11_end = min(time.time() + 40, deadline - 5)
        h11, c11 = slqp_polish(best_h, best_n, p11_end, alpha=500000, maxiter=800)
        if c11 < best_c:
            best_c, best_h, best_n = c11, h11, best_n
    
    # Phase 12: Final FW polish
    if time.time() < deadline - 3:
        p12_end = deadline - 1
        h12, c12 = frank_wolfe(best_h, best_n, p12_end,
            alpha_schedule=[100000, 500000, 2000000])
        if c12 < best_c:
            best_c, best_h, best_n = c12, h12, best_n
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
