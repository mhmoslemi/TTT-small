```python
import numpy as np
from scipy.optimize import minimize
import time

def correlate_full(a, b):
    """Compute np.correlate(a, b, mode='full') using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = np.fft.rfft(a[::-1], fsize)
    B = np.fft.rfft(b, fsize)
    result = np.fft.irfft(A * B, fsize)[:size]
    return result[::-1]

def compute_c5_vec(h, n):
    """Compute all correlation values scaled by dx."""
    dx = 2.0 / n
    v = 1.0 - h
    corr = correlate_full(h, v)
    return corr * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            scale = diff / tot
            h += scale * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            scale = diff / tot
            h += scale * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu):
    """Smooth max objective."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    log_sum = mx + np.log(np.sum(np.exp(sh - mx)))
    val = m + log_sum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    """Gradient of smooth max objective."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Only process shifts with significant weight
    threshold = 1e-12
    for mm in range(2 * n - 1):
        if w[mm] < threshold:
            continue
        tau = mm - (n - 1)
        wm = w[mm] * dx
        if tau >= 0:
            nv = n - tau
            grad[tau:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[tau:n])
        else:
            t = -tau
            nv = n - t
            grad[0:nv] += wm * (1.0 - x[t:n])
            grad[t:n] += wm * (-x[0:nv])
    
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    """Objective focused on active shifts."""
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    log_sum = smx + np.log(np.sum(np.exp(sh - smx)))
    val = mx + log_sum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    """Gradient of active-set objective."""
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        tau = idx - (n - 1)
```

```python
        for idx in active:
            tau = idx - (n - 1)
            if tau >= 0:
                nv = n - tau
                grad[tau:n] += dx * (1.0 - h[0:nv])
                grad[0:nv] += dx * (-h[tau:n])
            else:
                t = -tau
                nv = n - t
                grad[0:nv] += dx * (1.0 - h[t:n])
                grad[t:n] += dx * (-h[0:nv])
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.008, 0.03)
        else:
            eta *= 0.985
        h = h_new
    return best_h, best_c

def pair_search(h, n, deadline, radius=15):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + step <= 1.0 and best_h[j] - step >= 0.0:
                        oi = best_h[i]
                        oj = best_h[j]
                        best_h[i] += step
                        best_h[j] -= step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = oi
                            best_h[j] = oj
                    if best_h[i] - step >= 0.0 and best_h[j] + step <= 1.0:
                        oi = best_h[i]
                        oj = best_h[j]
                        best_h[i] -= step
                        best_h[j] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = oi
                            best_h[j] = oj
    return best_h, best_c

def slqp_run(h, n, deadline, alpha=200000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_con_fun, 'jac': slqp_con_jac, 'args': (n,)}]
    maxit = max(200, int((deadline - time.time()) * 1.5))
    try:
        res = minimize(slqp_fun, h, jac=slqp_jac_fn,
                      args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def upsamples_to(h_coarse, n_coarse, n_fine):
    x_c = np.linspace(0, 2, n_coarse, dtype=float)
    x_f = np.linspace(0, 2, n_fine, dtype=float)
    h_fine = np.interp(x_f, x_c, h_coarse)
    return project_feasible(h_fine, n_fine)

def random_start(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.random(n)
    return project_feasible(h, n)

def periodic_start(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(min(ones_per_period, period)):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            for i in range(n):
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            for i in range(n - 1, -1, -1):
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Brief L-BFGS at original resolution
    if time.time() < deadline - 400:
        p1_end = min(t0 + 80, deadline - 400)
        h1, c1 = run_lbfgs(best_h, best_n, p1_end,
            alphas=[50, 200, 1000, 5000, 50000],
            mu=0.01, maxiters=[1500, 1500, 2000, 2000, 2000])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upscale to n=2048 and optimize
    n_h = 2048
    if n_h != best_n:
        h_high = upsamples_to(best_h, best_n if best_n == n0 else len(best_h), n_h)
    else:
        h_high = best_h.copy()
    c_high = compute_c5(h_high, n_h)
    if c_high < best_c:
        best_c = c_high
        best_h = h_high
        best_n = n_h

    # Phase 3: L-BFGS at n=2048 with alpha annealing
    if time.time() < deadline - 300:
        p3_end = min(t0 + 350, deadline - 300)
        h3, c3 = run_lbfgs(h_high, n_h, p3_end,
            alphas=[20, 50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.005, maxiters=[1000, 1500, 2000, 2500, 3000, 3000, 3000, 3000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_h

    # Phase 4: Active-set optimization at n=2048
    if time.time() < deadline - 200:
        p4_end = min(t0 + 550, deadline - 200)
        h4, c4 = run_active(h3, n_h, p4_end, top_k=5, rounds=15)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_h

    # Phase 5: Try n=4096
    n_xh = 4096
    if time.time() < deadline - 180:
        h_xh = upsamples_to(best_h, best_n, n_xh)
        c_xh = compute_c5(h_xh, n_xh)
        p5_end = min(t0 + 700, deadline - 150)
        h5, c5 = run_lbfgs(h_xh, n_xh, p5_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000],
            mu=0.003, maxiters=[800, 1000, 1500, 2000, 2000, 2000, 2000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n_xh

    # Phase 6: Active-set at best resolution
    if time.time() < deadline - 120:
        bs = len(best_h)
        p6_end = min(t0 + 850, deadline - 100)
        h6, c6 = run_active(best_h, bs, p6_end, top_k=4, rounds=10)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = bs

    # Phase 7: Subgradient refinement
    if time.time() < deadline - 60:
        bs = len(best_h)
        p7_end = min(t0 + 950, deadline - 50)
        h7, c7 = subgrad_refine(best_h, bs, p7_end, n_iter=3000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = bs

    # Phase 8: Pair exchange (use lower res if too large)
    if time.time() < deadline - 30:
        bs = len(best_h)
        pe_n = bs if bs <= 2048 else 2048
        h_pe = best_h if bs == pe_n else upsamples_to(best_h, bs, pe_n)
        if pe_n < bs and compute_c5(h_pe, pe_n) < best_c:
            best_c = compute_c5(h_pe, pe_n)
            best_h = h_pe
            best_n = pe_n
        p8_end = min(t0 + 1050, deadline - 10)
        h8, c8 = pair_search(best_h, best_n, p8_end, radius=12)
        if c8 < best_c:
            best_c = c8
            best_h = h8
            best_n = best_n

    # Phase 9: SLSQP polish
    if time.time() < deadline - 5:
        bs = len(best_h)
        h9, c9 = slqp_run(best_h, bs, deadline - 2, alpha=500000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
            best_n = bs

    # Final: try a few random restarts if time permits
    if time.time() < deadline - 20:
        rng = np.random.RandomState(seed + 1)
        for trial in range(3):
            if time.time() > deadline - 8:
                break
            tr_n = 1024
            h_tr = random_start(tr_n, seed + 100 + trial)
            # Quick L-BFGS
            maxit = max(50, int((deadline - time.time() - 10) * 1.5))
            try:
                bounds_tr = [(0.0, 1.0)] * tr_n
                res = minimize(sm_obj, h_tr, jac=sm_grad,
                              args=(tr_n, 10000, 0.01), method='L-BFGS-B',
                              bounds=bounds_tr,
                              options={'maxiter': maxit, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxfun': maxit * 3,
                                       'maxls': 20, 'maxcor': 30})
                h_tr = project_feasible(res.x, tr_n)
                c_tr = compute_c5(h_tr, tr_n)
                if c_tr < best_c:
                    best_c = c_tr
                    best_h = h_tr
                    best_n = tr_n
            except Exception:
                pass

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
```
