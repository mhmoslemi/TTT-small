    """Find the steepest descent direction in the subdifferential of max."""
    # Compute gradients for active shifts
    grads = []
    for mm in active_k:
        g = shift_grad_single(h, nn, mm)
        grads.append(g)
    grads = np.array(grads)  # shape (K, n)
    
    # Find w that minimizes ||sum w_k g_k||^2 subject to w>=0, sum w=1
    # This is: min 1/2 w^T G w subject to w>=0, sum w=1
    # where G_{ij} = g_i · g_j
    K = len(active_k)
    G = grads @ grads.T
    # Solve QP: min 1/2 w^T G w s.t. w>=0, 1^T w=1
    # KKT: Gw + λ*1 - μ = 0, w>=0, μ>=0, w_i*μ_i=0
    # This is a small QP (K is usually 2-5)
    
    # Simple approach: if K is small, enumerate subsets
    best_dir = None
    best_norm = np.inf
    # Try all non-empty subsets of active_k
    for mask in range(1, 2**K):
        idx = [i for i in range(K) if mask & (1<<i)]
        G_sub = G[np.ix_(idx, idx)]
        # Solve: G_sub * w_sub = -λ * 1 (for the equality constraint)
        # Actually: minimize 1/2 w^T G w s.t. sum w = 1, w >= 0
        # Unconstrained: G w = λ * 1, w = (1/sum(G^{-1}1)) * G^{-1}1...

        w_sub = res.x
        direction = (w_sub @ grads[idx])
        norm = np.linalg.norm(direction)
        if norm < best_norm:

            best_dir = direction

def active_shifts(h, n, frac=0.005, max_k=20):
    c = compute_c5_vec(h, n)
    mv = c.max()
    thresh = mv - max(1e-8, frac * mv)
    active = np.where(c >= thresh)[0]
    if len(active) > max_k:
        active = active[np.argsort(-c[active])[:max_k]]
    return active

def fourier_init(M, n, rng):
    x = np.arange(n) * 2.0 / n
    a = rng.uniform(-0.4, 0.4, size=M)
    h = np.full(n, 0.5)
    for k in range(1, M + 1):
        h += a[k - 1] * np.sin(k * np.pi * x)
    return project_feasible(h, n)

def lbfgs_refine(h, n, deadline, alphas=None, bounds=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    if bounds is None:
        bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(
                softmax_obj, h, jac=softmax_grad,
                args=(n, alpha, None), method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 20000, 'maxls': 50}
            )
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def top_shifts_refine(h, n, deadline, top_k=5, alphas=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 50000, 200000, 1000000, 5000000]
    for _ in range(20):
        if time.time() > deadline - 3:
            break
        act = active_shifts(h, n, frac=0.003, max_k=top_k)
        if len(act) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(
                    softmax_obj, h, jac=softmax_grad,
                    args=(n, alpha, act), method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 10000, 'maxls': 40}
                )
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def perturbed_restart(h, n, n_restarts, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        scale = 0.02 / (r + 1)
        h_pert = h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        h_cur = h_pert if c_pert < best_c else best_h
        c_cur = min(c_pert, best_c)
        try:
            res = minimize(
                softmax_obj, h_cur, jac=softmax_grad,
                args=(n, 10000, None), method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-15, 'gtol': 1e-11, 'maxfun': 5000, 'maxls': 30}
            )
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def gradient_refine(h, n, deadline, n_iter=5000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    step = 0.01
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        mv = c.max()
        act = active_shifts(h, n, frac=0.002, max_k=5)
        if len(act) == 0:
            break
        g = softmax_grad(h, n, 10000, act.tolist())
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        for mult in [1.0, 0.5, 0.25, 0.1]:
            s = step * mult
            h_try = h - s * g
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                h = h_try
                break
        else:
            step *= 0.5
            if step < 1e-8:
                break
    return h, best_c

def pair_search(h, n, deadline, radius=20):
    best_c = compute_c5(h, n)
    for delta in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if delta > 0.001 and abs(i - j) > radius:
                        continue
                    if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                        h[i] += delta
                        h[j] -= delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                    if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                        h[i] -= delta
                        h[j] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    except Exception:
        h_parent = np.full(250, 0.5)
        n_base = 250

    n = n_base
    best_h = project_feasible(h_parent, n)
    best_c = compute_c5(best_h, n)

    # Try Fourier inits as alternatives
    for M in [15, 25, 40]:
        h_f = fourier_init(M, n, rng)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 1: Main L-BFGS at base resolution (~25% budget)
    t1 = min(t0 + 0.25 * budget_s, deadline - 0.55 * budget_s)
    best_h, best_c = lbfgs_refine(best_h, n, t1,
        alphas=[200, 1000, 5000, 20000, 100000, 500000, 2000000])

    # Phase 2: Top-shifts focused (~15% budget)
    t2 = min(time.time() + 0.15 * budget_s, deadline - 0.35 * budget_s)
    best_h, best_c = top_shifts_refine(best_h, n, t2, top_k=5,
        alphas=[10000, 100000, 500000, 5000000])

    # Phase 3: Perturbation restarts (~12% budget)
    t3 = min(time.time() + 0.12 * budget_s, deadline - 0.2 * budget_s)
    best_h, best_c = perturbed_restart(best_h, n, 5, t3, rng)

    # Phase 4: Another top-shifts round
    t4 = min(time.time() + 0.1 * budget_s, deadline - 0.15 * budget_s)
    best_h, best_c = top_shifts_refine(best_h, n, t4, top_k=3,
        alphas=[100000, 1000000, 10000000])

    # Phase 5: Upsample to n=1000 for finer solution
    n_fine = min(1000, max(n, 1000))
    if time.time() < deadline - 0.2 * budget_s and n_fine > n:
        h_fine = upsample(best_h, n, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        t5 = min(time.time() + 0.1 * budget_s, deadline - 0.1 * budget_s)
        h_fine, c_fine = lbfgs_refine(h_fine, n_fine, t5,
            alphas=[5000, 50000, 500000, 5000000])
        t5b = min(time.time() + 0.05 * budget_s, deadline - 0.08 * budget_s)
        h_fine, c_fine = top_shifts_refine(h_fine, n_fine, t5b, top_k=3,
            alphas=[100000, 1000000])
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine
            n = n_fine

    # Phase 6: Gradient-based refinement
    t6 = min(time.time() + 0.05 * budget_s, deadline - 0.05 * budget_s)
    best_h, best_c = gradient_refine(best_h, n, t6, n_iter=2000)

    # Phase 7: Final top-shifts polish
    t7 = min(time.time() + 0.04 * budget_s, deadline - 0.03 * budget_s)
    best_h, best_c = top_shifts_refine(best_h, n, t7, top_k=3,
        alphas=[500000, 5000000, 50000000])

    # Phase 8: Pair search
    t8 = min(time.time() + 0.03 * budget_s, deadline - 3)
    best_h, best_c = pair_search(best_h, n, t8, radius=15)

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
