import numpy as np
import time
import math
from scipy.optimize import minimize


def compute_c5_np(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_corr_full(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    need = 2 * n - 1
    m = next_pow2(need)
    H = np.fft.rfft(h, m)
    V = np.fft.rfft(1.0 - h, m)
    C = np.fft.irfft(H * np.conj(V), m)
    out = np.empty(need, dtype=float)
    out[n - 1:] = C[:n]
    out[:n - 1] = C[m - n + 1:m]
    return out


def fft_c5(h, n):
    return float(np.max(fft_corr_full(h)) * (2.0 / float(n)))


def project_box_sum(y, target):
    y = np.asarray(y, dtype=float)
    z = np.clip(y, 0.0, 1.0)
    diff = target - float(z.sum())
    if abs(diff) < 1e-12:
        return z

    if diff > 0.0:
        mask = z < 1.0 - 1e-14
        cnt = int(mask.sum())
        if cnt > 0:
            add = diff / cnt
            if add <= (1.0 - float(np.max(z[mask]))) + 1e-14:
                z = z.copy()
                z[mask] += add
                z = np.clip(z, 0.0, 1.0)
                return z
    else:
        mask = z > 1e-14
        cnt = int(mask.sum())
        if cnt > 0:
            sub = -diff / cnt
            if sub <= (float(np.min(z[mask]))) + 1e-14:
                z = z.copy()
                z[mask] -= sub
                z = np.clip(z, 0.0, 1.0)
                return z

    lo = min(-10.0, -float(np.max(y)) - 1.0)
    hi = max(10.0, 1.0 - float(np.min(y)) + 1.0)
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(z.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = z < 1.0 - 1e-14
            if mask.any():
                z = z.copy()
                z[mask] += diff / float(mask.sum())
        else:
            mask = z > 1e-14
            if mask.any():
                z = z.copy()
                z[mask] += diff / float(mask.sum())
    return np.clip(z, 0.0, 1.0)


def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    h = project_box_sum(h, target)

    for _ in range(12):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            idx = np.flatnonzero(h < 1.0 - 1e-12)
            if idx.size == 0:
                break
            i = int(idx[int(np.argmax(1.0 - h[idx]))])
            add = min(diff, 1.0 - h[i])
            if add > 0.0:
                h[i] += add
        else:
            idx = np.flatnonzero(h > 1e-12)
            if idx.size == 0:
                break
            i = int(idx[int(np.argmax(h[idx]))])
            sub = min(-diff, h[i])
            if sub > 0.0:
                h[i] -= sub
    return h


def upsample_to(h, n_new):
    h = np.asarray(h, dtype=float).copy()
    old = len(h)
    if old == n_new:
        return make_feasible(h, n_new)
    if old == 1:
        hn = np.full(n_new, float(h[0]), dtype=float)
    else:
        xo = np.linspace(0.0, 2.0, old)
        xn = np.linspace(0.0, 2.0, n_new)
        hn = np.interp(xn, xo, h)
    return make_feasible(hn, n_new)


def select_top_indices(corr, k):
    L = len(corr)
    k = int(min(k, L))
    if k <= 0:
        return np.array([], dtype=int)
    if k >= L:
        return np.arange(L, dtype=int)
    return np.argpartition(corr, L - k)[L - k:].astype(int, copy=False)


def grad_lag_vec(h, lag):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if lag >= 0:
        if lag < n:
            g[lag:n] += 1.0 - h[0:n - lag]
            g[0:n - lag] -= h[lag:n]
    else:
        q = -lag
        if q < n:
            g[0:n - q] += 1.0 - h[q:n]
            g[q:n] -= h[0:n - q]
    return g


def gradient_for_indices(h, corr, idx_set, alpha, dx):
    L = len(idx_set)
    if L == 0:
        return np.zeros(len(h), dtype=float)
    vals = np.asarray(corr[idx_set], dtype=float)
    m = float(np.max(vals))
    shift = vals - m
    sm = float(np.max(shift))
    w = np.exp(shift - sm)
    sw = float(w.sum())
    if sw <= 0.0:
        w = np.zeros(L, dtype=float)
        w[0] = 1.0
        sw = 1.0
    else:
        w = w / sw
    g = np.zeros(len(h), dtype=float)
    for ii in range(L):
        if abs(w[ii]) > 1e-15:
            lag = int(idx_set[ii]) - (len(h) - 1)
            g += w[ii] * grad_lag_vec(h, lag)
    return g * dx


def smooth_obj_fixed(x, n, alpha, idx_set):
    dx = 2.0 / float(n)
    corr = fft_corr_full(x)
    vals = corr[idx_set]
    m = float(np.max(vals))
    shift = vals - m
    sm = float(np.max(shift))
    lse = sm + math.log(float(np.exp(shift - sm).sum()))
    return float(dx * (m + lse / alpha))


def smooth_grad_fixed(x, n, alpha, idx_set):
    dx = 2.0 / float(n)
    corr = fft_corr_full(x)
    return gradient_for_indices(x, corr, idx_set, alpha, dx)


def eq_sum(x, target):
    return float(x.sum()) - float(target)


def slsqp_smooth(h, n, alpha, deadline, maxiter=12, idx_set=None):
    h = make_feasible(h, n)
    if time.time() > deadline - 5:
        return h
    if idx_set is None:
        corr = fft_corr_full(h)
        idx_set = select_top_indices(corr, 64)
    if len(idx_set) == 0:
        return h
    target = 0.5 * float(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": eq_sum, "args": (target,)}]
    try:
        res = minimize(
            smooth_obj_fixed,
            h,
            args=(n, alpha, idx_set),
            method="SLSQP",
            jac=smooth_grad_fixed,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-9, "disp": False},
        )
        h2 = res.x
    except Exception:
        h2 = h
    h2 = make_feasible(h2, n)
    return h2


def projected_phase(h, n, alpha, deadline, top_k=64, base_step=0.05):
    h = make_feasible(h, n)
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    c = fft_c5(h, n)
    best = h.copy()
    best_c = c
    no_improve = 0

    for _ in range(200000):
        if time.time() > deadline:
            break
        corr = fft_corr_full(h)
        m = float(np.max(corr))
        c = m * dx
        if c < best_c - 1e-12:
            best_c = c
            best = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 35:
            break

        top_k_eff = int(min(top_k, 2 * n - 1))
        idx = select_top_indices(corr, top_k_eff)
        g = gradient_for_indices(h, corr, idx, alpha, dx)
        g = g - float(np.mean(g))
        rms = float(np.sqrt(np.mean(g * g)))
        if rms < 1e-13:
            break
        direction = g / rms

        accepted = False
        steps = [base_step * 1.2, base_step, base_step * 0.6, base_step * 0.3, base_step * 0.15, base_step * 0.075]
        for base in steps:

            cand = project_box_sum(cand, target)

                c = cc
                if c < best_c - 1e-12:
                    best_c = c
                    best = h.copy()
                accepted = True
                break

            no_improve += 8
            if no_improve > 25:

    return best, best_c

def binary_threshold(h, n):
    h = upsample_to(h, n)
    target = int(round(0.5 * float(n)))
    if target <= 0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)
    order = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=float)
    hb[order[:target]] = 1.0
    return hb


def corr_binary_init(h):

def binary_propose_lag(h, lag, rng):

            p = int(rng.randint(q0, n))

    k = int(pos[int(rng.randint(pos.size))])

            reject = 0
            if d <= 0:
                T = max(Tmin, T * 0.99996)
            else:
                T = min(3.0, T * 0.99998)
        else:
            reject += 1
            if reject > 3000:
                reject = 0
                if not mode_det:
                    T = min(4.0, T * 1.3)
            if mode_det and reject > 6000:
                mode_det = False
                T = 1.0
                reject = 0

    return best_h, float(best_count) * dx


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(45.0, float(budget_s) - 6.0)
    dl = t0 + budget_s - 5.0

    try:
        n_best = len(initial_h_values)
        best_h = make_feasible(np.asarray(initial_h_values, dtype=float), n_best)
    except Exception:
        n_best = 250
        x = np.arange(n_best, dtype=float)
        phase = (x * 0.6180339887498949) % 1.0
        best_h = make_feasible((phase < 0.5).astype(float), n_best)

    best_c = compute_c5_np(best_h, n_best)

    # Phase 0: polish the parent resolution.
    if n_best <= 350:
        ph_end = t0 + 0.06 * total
        if time.time() < ph_end - 10:
            h2, c2 = continuous_optimize(best_h, n_best, ph_end, use_slsqp=True, seed=seed)
            ce = compute_c5_np(h2, n_best)
            if ce < best_c:
                best_c = ce
                best_h = h2
                n_best = n_best

    stages = [
        (500, 0.06, 0.12, 0.19, True),
        (1000, 0.19, 0.27, 0.36, True),
        (2000, 0.36, 0.50, None, False),
        (3000, 0.50, 0.63, None, False),
        (4000, 0.63, 0.76, None, False),
    ]

    for n, cstart, cend, bend, do_bin in stages:
        if time.time() > dl - 35:
            break
        if n <= 600 and n <= n_best and n_best > 300:
            continue

        cont_end = t0 + cend * total
        if cont_end > dl - 25:
            cont_end = dl - 25
        if time.time() < cont_end - 10:
            h2, c2 = continuous_optimize(best_h, n, cont_end, use_slsqp=False, seed=seed + 17 * n)
            ce = compute_c5_np(h2, n)
            if ce < best_c:
                best_c = ce
                best_h = h2
                n_best = n

        if do_bin and bend is not None:
            bstart = t0 + cend * total
            bin_end = t0 + bend * total
            if bin_end > dl - 50:
                bin_end = dl - 50
            if time.time() < bin_end - 20:
                hb, cb = binary_anneal(best_h, n, bin_end, seed=seed + 1000 + n, target_c=0.3808)
                cbe = compute_c5_np(hb, n)
                if cbe < best_c:
                    best_c = cbe
                    best_h = hb
                    n_best = n

                if n <= 1200 and time.time() < dl - 40:
                    pol_end = min(dl - 25, time.time() + 25.0)
                    hp = hb + 0.025 * np.random.RandomState(seed + 3000 + n).randn(n)
                    hp, cp = continuous_optimize(hp, n, pol_end, use_slsqp=False, seed=seed + 4000 + n)
                    cpe = compute_c5_np(hp, n)
                    if cpe < best_c:
                        best_c = cpe
                        best_h = hp
                        n_best = n

    # Final polish at the best resolution, or a larger one if there is time.
    if time.time() < dl - 70:
        n_final = max(int(n_best), 2000)
        if n_final > 4000:
            n_final = 4000
        if n_final % 2:
            n_final += 1
        final_end = dl - 15
        h2, c2 = continuous_optimize(best_h, n_final, final_end, use_slsqp=False, seed=seed + 99999)
        ce = compute_c5_np(h2, n_final)
        if ce < best_c:
            best_c = ce
            best_h = h2
            n_best = n_final

    best_h = make_feasible(best_h, int(n_best))
    best_c = compute_c5_np(best_h, int(n_best))
    return best_h, float(best_c), int(n_best)
