```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()

    shifted = alpha * (corr - m)
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()
    
    # Gradient
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
            h = np.clip(h, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
            h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1.0 - h)
            for i in range(int(abs(diff))):
                if i < len(idx) and h[idx[i]] < 1.0:
                    h[idx[i]] += 1.0
        else:
            idx = np.argsort(h)
            for i in range(int(abs(diff))):
                if i < len(idx) and h[idx[i]] > 0.0:
                    h[idx[i]] -= 1.0
        h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_grad_softmax(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    
    shifted = alpha * (corr - m)
    shifted -= shifted.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(2*n-1)
        p[np.argmax(corr)] = 1.0
    else:
        p /= p_sum
    
    grad = np.zeros(n)
    for j in range(n):
        lo = max(0, n - 1 - j)
        hi = min(2*n-2, 2*n-2 - j)
        t1 = 0.0
        if hi >= lo:
            idx = j + np.arange(lo, hi+1) - (n-1)
            t1 = np.dot(p[lo:hi+1], 1.0 - h[idx])
        
        lo2 = max(0, j)
        hi2 = min(2*n-2, j + n - 1)
        t2 = 0.0
        if hi2 >= lo2:
            idx2 = j + (n-1) - np.arange(lo2, hi2+1)
            t2 = np.dot(p[lo2:hi2+1], h[idx2])
        
        grad[j] = t1 - t2
    
    return grad * dx, float(m), corr

def projected_gradient_descent(h, n, deadline, alpha_start=100, alpha_end=100000, max_iter=50000):
    """Custom projected gradient with soft-max."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    step = 0
    for alpha in np.geomspace(alpha_start, alpha_end, 10):
        if time.time() > deadline:
            break
        lr = 0.01
        v = np.zeros(n)  # momentum
        momentum = 0.9
        
        for it in range(max_iter // 10):
            if time.time() > deadline:
                break
            grad, m, corr = compute_grad_softmax(h, n, alpha)
            
            # Momentum step
            v = momentum * v - lr * grad
            h_new = h + v
            
            # Project
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            
            # Line search
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                if c_new < best_c - 1e-8:
                    best_c = c_new
                    best_h = h_new.copy()
                h = h_new
            else:
                # Reduce step
                lr *= 0.5
                if lr < 1e-6:
                    lr = 0.01
                    v = np.zeros(n)
                    h_new = h - lr * grad
                    h_new = np.clip(h_new, 0.0, 1.0)
                    h_new = project_feasible(h_new, n)
                    h = h_new
        step += 1
    
    return best_h, best_c
```

For the binary case, the local search is:

- This changes the autocorrelation at every lag
- We want to reduce the maximum "disagreement"

The efficient way to do this:
1. Precompute for each position i and each lag k: does changing h[i] affect c_k?
2. For a candidate swap (i: 0→1, j: 1→0), the change in c_k is:
   - Adding 1 at i: c_k increases by (1-h[i+k]) for each valid k
   - Removing 1 at j: c_k decreases by (1-h[j+k]) for each valid k
   - Cross terms

For binary h, this is just counting:
- Adding 1 at i: for each lag k, c_k increases by 1 if h[i+k]=0 (and i+k is in range)
- Removing 1 at j: for each lag k, c_k decreases by 1 if h[j+k]=0 (and j+k is in range)

So the delta for swap (i, j) is:
Δc_k = (1 if h[i+k]=0 and i+k in range, else 0) - (1 if h[j+k]=0 and j+k in range, else 0)

This is O(n) to compute for each swap. And we have O(n²) candidate swaps. Total: O(n³) per full sweep.

For n=250: 250³ = 15.6M per sweep. In numpy, this is maybe 50ms per sweep. So in 30 seconds we can do about 600 sweeps. That's reasonable.

But we can be smarter: instead of checking all swaps, use a **greedy** approach:
1. Find the worst lag k*
2. Only consider swaps that reduce c_{k*}
3. Among those, pick the one that minimizes the new max

```python
def binary_local_search(h, n, deadline, max_sweeps=1000):
    """Efficient binary local search for the overlap problem."""
    dx = 2.0 / n
    h = h.copy()
    # Ensure binary
    # Sort and make top n/2 be 1
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        
        # Find worst lag
        worst_lags = np.where(corr == corr.max())[0]
        
        # Try random swaps, prioritizing those that reduce worst lags
        n_tries = 200
        for _ in range(n_tries):
            i = zeros[np.random.randint(len(zeros))]
            j = ones[np.random.randint(len(ones))]
            
            # Compute delta
            shift = np.arange(-(n-1), n)  # lags
            # For position i (0->1): contributes +1 to c where h[i+lag]=0
            i_offset = i + shift  # positions relative to i
            j_offset = j + shift  # positions relative to j
            
            delta = np.zeros(2*n-1)
            msk_i = (i_offset >= 0) & (i_offset < n)
            delta[msk_i] += 1.0 - h[i_offset[msk_i]]
            
            msk_j = (j_offset >= 0) & (j_offset < n)
            delta[msk_j] -= 1.0 - h[j_offset[msk_j]]
            
            # Cross terms: the (i,j) and (j,i) interactions
            # When we set i to 1 and j to 0 simultaneously:
            # At lag (j-i): i contributes (1-h[j]) = 1, but j is being set to 0, so it's 1
            # Actually the cross term is: at the lag where i and j interact
            # When computing c_lag for the new h:
            # The term h[i]*(1-h[i+lag]) at lag (j-i): h[i]=1, h[i+(j-i)]=h[j]=0, so contributes 1
            # But in old h: h[i]=0, so contributed 0. In new: contributes 1.
            # The term h[j]*(1-h[j+lag]) at lag (i-j): h[j]=0 in new, contributes 0
            # But in old: h[j]=1, h[j+(i-j)]=h[i]=0, contributed 1. In new: 0.

            # For position i (0->1): I add (1-h[i+lag]) which includes the case where i+lag=j
            #   At lag (j-i): adds (1-h[j]) = 1 (since h[j] is still 1 at this point in computation)
            # For position j (1->0): I subtract (1-h[j+lag]) which includes the case where j+lag=i
            #   At lag (i-j): subtracts (1-h[i]) = 1 (since h[i] is still 0 at this point)
            
            # But wait, in the new h, h[i]=1 and h[j]=0. So:
            # At lag (j-i): the term is h[i]*(1-h[j]) = 1*1 = 1. In old: 0*(1-1) = 0. Delta = +1. ✓
            # At lag (i-j): the term is h[j]*(1-h[i]) = 0*1 = 0. In old: 1*(1-0) = 1. Delta = -1. ✓
            # At lag 0 (self): term is h[i]*(1-h[i]) + h[j]*(1-h[j]) = 1*0 + 0*1 = 0. Both old and new have 0. ✓
            
            # Actually wait, the self-term: in the full correlation, lag 0 corresponds to
            # ∑ h[k]*(1-h[k]) = 0 always for binary. So no issue.
            
            # But what about the case where i+lag = j AND j+lag' = i for the same lag?
            # That's lag = j-i and lag' = i-j, which are different. So no double counting.
            
            new_corr = corr + delta
            nc = new_corr.max() * dx

                h[j] = 0.0
                corr = new_corr
                cur_c = nc
                improved = True
                # Update ones and zeros

                zeros = np.where(h == 0)[0]
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        
        if not improved:
            # Try more diverse swaps or break

                break
    
    return best_h, best_c
```

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softgrad(h, n, alpha):
    """Compute gradient of soft-max objective. Returns (grad, c_max, corr)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    shifted -= shifted.max()
    p = np.exp(shifted)
    ps = p.sum()
    if ps > 1e-30:
        p /= ps
    else:
        p = np.zeros(2*n-1)
        p[np.argmax(corr)] = 1.0
    
    grad = np.zeros(n)
    for j in range(n):
        # ∑_l p[l] * (1 - h[j + l - (n-1)])  [valid range]
        lo = max(0, n - 1 - j)
        hi = min(2*n-2, 2*n-2 - j)
        t1 = 0.0
        if hi >= lo:
            idx = j + np.arange(lo, hi+1) - (n-1)
            t1 = np.dot(p[lo:hi+1], 1.0 - h[idx])
        
        # ∑_l p[l] * h[j + (n-1) - l]  [valid range]
        lo2 = max(0, j)
        hi2 = min(2*n-2, j + n - 1)
        t2 = 0.0
        if hi2 >= lo2:
            idx2 = j + (n-1) - np.arange(lo2, hi2+1)
            t2 = np.dot(p[lo2:hi2+1], h[idx2])
        
        grad[j] = t1 - t2
    
    return grad * dx, float(m), corr

def proj_grad(h0, n, deadline, alpha_list=None, lr_init=0.02):
    """Projected gradient descent with Nesterov momentum."""
    if alpha_list is None:
        alpha_list = [100, 1000, 5000, 20000, 100000, 500000]
    
    h = h0.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        lr = lr_init * (1.0 / (1.0 + 0.01 * alpha / 100))
        v = np.zeros(n)
        mu = 0.9
        
        for it in range(20000):
            if time.time() > deadline:
                break
            y = h + mu * v  # Nesterov lookahead
            grad, m, _ = softgrad(y, n, alpha)
            v = mu * v - lr * grad
            h_new = h + v
            h_new = project_feasible(h_new, n)
            
            c_new = compute_c5(h_new, n)
            if c_new > best_c * 1.01 + 0.01:
                # Diverging, reduce lr
                lr *= 0.5
                v *= 0.5
                if lr < 1e-8:
                    lr = lr_init
                    v = np.zeros(n)
                continue
            
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if it % 100 == 0:
                    lr *= 1.05  # slightly increase step
            else:
                if it % 500 == 0:
                    lr *= 0.95  # slowly decrease
    
    return best_h, best_c

def binarize(h, n):
    """Convert to binary: top n/2 values become 1."""
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def binary_search(h_bin, n, deadline, n_restarts=10, swaps_per_iter=500, seed=42):

    dx = 2.0 / n
    target = n // 2

    best_c = cur_c
    best_h = h.copy()

            break
        
        if restart > 0:
            # Perturb: swap random pairs

                    a = rng.choice(ones)
                    b = rng.choice(zeros)

def make_structured_init(n, seed):
    """Create a structured initial binary pattern."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    
    # Try a "block" pattern: alternating blocks of 1s and 0s
    # Block size chosen to be roughly n/2 / (2*nb) for nb blocks

        sizes = np.random.dirichlet(np.ones(nb * 2)) * n
        sizes = sizes.astype(int)
        # Adjust to sum to n
        diff = n - sizes.sum()
        for d in range(abs(diff)):
            sizes[rng.randint(0, len(sizes))] += 1 if diff > 0 else -1
        
        pos = 0
        for bi in range(len(sizes)):
            val = 1.0 if bi % 2 == 0 else 0.0
            h[pos:pos+sizes[bi]] = val
            pos += sizes[bi]
        
        # Verify count
        c1 = h.sum()
        if abs(c1 - target) > n * 0.05:
            continue
        
        # Fine-tune
        if c1 > target:
            excess = int(c1 - target)
            ones = np.where(h == 1)[0]
            to_zero = ones[rng.choice(len(ones), size=excess, replace=False)]
            h[to_zero] = 0.0
        elif c1 < target:
            deficit = int(target - c1)
            zeros = np.where(h == 0)[0]
            to_one = zeros[rng.choice(len(zeros), size=deficit, replace=False)]
            h[to_one] = 1.0
        
        break
    
    return h
