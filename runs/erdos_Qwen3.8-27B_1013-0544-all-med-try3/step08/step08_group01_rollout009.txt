import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm]
        if wm < 1e-12:
            continue
        k = mm - (n - 1)
        wm_dx = wm * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm_dx * (1.0 - x[k:n])
            grad[k:n] += wm_dx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm_dx * (1.0 - x[0:nv])
            grad[0:nv] += wm_dx * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-12, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def stochastic_local_search(h, n, deadline, rng, step_decay=0.999):
    """Local search with random 2-point transfers."""
    best_c = compute_c5(h, n)
    step = 0.01
    dx = 2.0 / n
    target = n / 2.0
    no_improve = 0
    max_no_improve = 5000
    
    while time.time() < deadline:
        # Decay step
        step *= step_decay
        if step < 1e-5:
            break
        
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try transfer: h[i] += d, h[j] -= d (or vice versa)
        d = step * (0.5 + rng.random())
        
        for direction in [1, -1]:
            if direction == 1:
                if h[i] + d > 1 or h[j] - d < 0:
                    continue
                h[i] += d
                h[j] -= d
            else:
                if h[i] - d < 0 or h[j] + d > 1:
                    continue
                h[i] -= d
                h[j] += d
            
            # Check constraint
            s = h.sum()
            if abs(s - target) < 1e-10:
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    no_improve = 0
                else:
                    # Reject
                    if direction == 1:
                        h[i] -= d
                        h[j] += d
                    else:
                        h[i] += d
                        h[j] -= d
                    no_improve += 1
            else:
                # Reject (constraint violation)
                if direction == 1:
                    h[i] -= d
                    h[j] += d
                else:
                    h[i] += d
                    h[j] -= d
                no_improve += 1
        
        if no_improve > max_no_improve:
            break
    
    return h, best_c

def gradient_descent_local(h, n, deadline, alpha=50000):
    """Subgradient descent on the max function with projection."""
    best_c = compute_c5(h, n)
    lr = 0.01
    target = n / 2.0
    no_improve = 0
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - 0.001
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        # Average gradient over active shifts
        dx = 2.0 / n
        grad = np.zeros(n)
        n_act = len(active)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= n_act
        
        # Gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
            no_improve = 0
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.9
            no_improve += 1
        
        if lr < 1e-8 or no_improve > 500:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm]
        if wm < 1e-13:
            continue
        k = mm - (n - 1)
        wm_dx = wm * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm_dx * (1.0 - x[k:n])
            grad[k:n] += wm_dx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm_dx * (1.0 - x[0:nv])
            grad[0:nv] += wm_dx * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        t_start = time.time()
        maxiter = max(100, min(5000, int((deadline - 2 - t_start) * 15)))
        maxfun = max(500, int((deadline - 2 - t_start) * 60))
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-12, 'maxfun': maxfun, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def active_shifts_search(h, n, deadline, top_frac=0.02, lr_init=0.005):
    """Subgradient descent focusing on active (near-max) shifts."""
    best_c = compute_c5(h, n)
    h = h.copy()
    lr = lr_init
    target = n / 2.0
    no_improve = 0
    max_no_improve = 300
    dx = 2.0 / n

    while time.time() < deadline and no_improve < max_no_improve:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, top_frac * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c_vec))])

        n_act = len(active)
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= n_act

        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            h = h_new
            no_improve = 0
            lr = min(lr * 1.1, 0.05)
        else:
            no_improve += 1
            lr *= 0.92

        if lr < 1e-9:
            break
    return h, best_c

def stochastic_search(h, n, deadline, rng, step0=0.004, decay=0.999998):

    target = n / 2.0
    no_improve = 0

def block_descent(h, n, deadline, block_size=20, n_restarts=5):
    """Block coordinate descent: optimize small blocks while fixing others."""
    best_c = compute_c5(h, n)
    h = h.copy()
    rng = np.random.default_rng(12345)
    bounds_block = [(0.0, 1.0)] * block_size

    for restart in range(n_restarts):
        if time.time() > deadline - 5:
            break
        # Shuffle blocks
        perm = rng.permutation(n)
        for chunk_start in range(0, n - block_size + 1, block_size):
            if time.time() > deadline - 3:
                break
            idx = perm[chunk_start:chunk_start + block_size]
            h_sub = h[idx].copy()

            def block_obj(x_sub, idx=idx):
                h_tmp = h.copy()
                h_tmp[idx] = x_sub
                return compute_c5(h_tmp, n)

            def block_grad(x_sub, idx=idx):
                h_tmp = h.copy()
                h_tmp[idx] = x_sub
                c_vec = compute_c5_vec(h_tmp, n)
                m = c_vec.max()
                alpha_b = 10000.0
                sh = alpha_b * (c_vec - m)
                mx = sh.max()
                w = np.exp(sh - mx)
                w /= w.sum()
                dx_b = 2.0 / n
                full_grad = np.zeros(n)
                for mm in range(2 * n - 1):
                    wm = w[mm]
                    if wm < 1e-12:
                        continue
                    k = mm - (n - 1)
                    wm_dx = wm * dx_b
                    if k >= 0:
                        nv = n - k
                        full_grad[0:nv] += wm_dx * (1.0 - h_tmp[k:n])
                        full_grad[k:n] += wm_dx * (-h_tmp[0:nv])
                    else:
                        kk = -k
                        nv = n - kk
                        full_grad[kk:n] += wm_dx * (1.0 - h_tmp[0:nv])
                        full_grad[0:nv] += wm_dx * (-h_tmp[kk:n])
                return full_grad[idx]

            try:
                res = minimize(block_obj, h_sub, jac=block_grad,
                              method='L-BFGS-B', bounds=bounds_block,
                              options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10})
                h_new = h.copy()
                h_new[idx] = res.x
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c
