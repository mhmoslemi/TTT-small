def subgradient_step(h, n, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)  # index in [0, 2n-2]
    k = k_star - (n - 1)      # actual lag in [-(n-1), n-1]
    
    grad = np.zeros(n)
    if k >= 0:
        # corr = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        for i in range(n - k):
            grad[i + k] += (1 - h[i]) * dx  # derivative w.r.t. h[i+k]
            grad[i] += (-h[i+k]) * dx        # derivative w.r.t. h[i]
    else:
        al = -k
        # corr = sum_{i=0}^{n-1-al} h[i] * (1-h[i+al])
        for i in range(n - al):
            grad[i] += (1 - h[i+al]) * dx   # derivative w.r.t. h[i]
            grad[i + al] += (-h[i]) * dx     # derivative w.r.t. h[i+al]
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new

And: **a "steepest pair" search** where instead of trying all pairs, I compute the gradient and find the pair that gives the steepest descent.

For a pair (i, j) with change h[i] += d, h[j] -= d:
∂c[k*]/∂(this change) = d * (∂c[k*]/∂h[i] - ∂c[k*]/∂h[j])

The steepest descent pair is the one that maximizes this (for the lag k* that achieves the max).

This is O(n²) to compute but gives a much better direction than random pair search.

```python
def steepest_pair_step(h, n, d):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    k = k_star - (n - 1)
    
    # Compute gradient of c[k_star] w.r.t. each h[i]
    grad = np.zeros(n)
    if k >= 0:
        idx = np.arange(n - k)
        grad[idx + k] += (1.0 - h[idx]) * dx
        grad[idx] += (-h[idx + k]) * dx
    else:
        al = -k
        idx = np.arange(n - al)
        grad[idx] += (1.0 - h[idx + al]) * dx
        grad[idx + al] += (-h[idx]) * dx
    
    # Find steepest pair: maximize grad[i] - grad[j]
    # (we want to increase h[i] where grad is high and decrease h[j] where grad is low)
    # subject to h[i] + d <= 1 and h[j] - d >= 0
    
    best_diff = -np.inf
    best_i, best_j = -1, -1
    for i in range(n):
        if h[i] + d > 1.0:
            continue
        for j in range(n):
            if j == i:
                continue
            if h[j] - d < 0.0:
                continue
            diff = grad[i] - grad[j]
            if diff > best_diff:
                best_diff = diff
                best_i, best_j = i, j
    
    if best_i >= 0 and best_diff > 0:
        h[best_i] += d
        h[best_j] -= d
    return h, best_diff
```

This is O(n²) per step, same as the full pair search, but it finds the BEST pair in one shot instead of trying all pairs.
