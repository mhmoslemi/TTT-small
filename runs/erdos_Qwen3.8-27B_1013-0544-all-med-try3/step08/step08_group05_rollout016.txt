def subgradient_step(corr, k_star, h, n):
    """Compute gradient of f_{k_star} at h."""
    dx = 2.0 / n
    k = k_star - (n - 1)  # Convert from correlation index to actual lag
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        ik = i + k
        if 0 <= ik < n:
            val += 1.0 - h[ik]
        ik2 = i - k
        if 0 <= ik2 < n:
            val -= h[ik2]
        grad[i] = dx * val
    return grad

    dx = 2.0 / n
    best_h = h.copy()

                    oi = np.union1d(oi, [b])
