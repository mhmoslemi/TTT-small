        grad[:n-jj] -= h[jj:]

            h_opt2, c_opt2 = optimize_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2
    
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    if time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.01 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 25) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        best_h, best_c = full_pair_search(best_h, n, deadline - 2,
                                           [0.001, 0.0005, 0.0002, 0.0001, 0.00005], top_k=80)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        ht = slsqp_optimize(best_h, n, 20000000, maxiter=5000, ftol=1e-17)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
