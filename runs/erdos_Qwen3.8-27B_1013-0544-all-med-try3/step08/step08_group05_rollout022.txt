import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h * (1.0 + diff / s)
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute among non-boundary elements
        idx = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, n, alpha):
    h = x.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    log_sum = max_c + np.log(np.exp(shifted).sum()) / alpha
    f = log_sum * dx
    
    p = np.exp(shifted)
    p /= p.sum()

                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]

                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)

                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
            
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += (corr + (np.correlate(h, 1.0 - h, mode='full').astype(float) - corr))  # recompute? No, use delta

                cur_c = best_nc
                ones.discard(best_j)
                ones.add(i)
                zeros.discard(i)

        best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = corr.max()
    sh = alpha * (corr - mc)
    f = (mc + np.log(np.exp(sh).sum()) / alpha) * dx
    p = np.exp(sh)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros_like(corr)
        p[np.argmax(corr)] = 1.0
    else:
        p /= ps
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        s = k - (n - 1)
        jlo = max(0, -s)
        jhi = min(n - 1, n - 1 - s)
        if jlo <= jhi:
            grad[jlo:jhi+1] += pk * (1.0 - x[jlo+s:jhi+s+1])
        jlo2 = max(0, s)
        jhi2 = min(n - 1, n - 1 + s)
        if jlo2 <= jhi2:
            grad[jlo2:jhi2+1] -= pk * x[jlo2-s:jhi2-s+1]
    grad *= dx
    return f, grad

def lbfgs_run(h0, n, alpha, maxiter, deadline, pen=1000.0):
    t = n / 2.0
    def f(x):
        x2 = np.clip(x, 0.0, 1.0)
        val, _ = smooth_obj_grad(x2, n, alpha)
        cv = x2.sum() - t
        return val + pen * cv * cv
    def g(x):
        x2 = np.clip(x, 0.0, 1.0)
        _, gr = smooth_obj_grad(x2, n, alpha)
        cv = x2.sum() - t
        gr += 2.0 * pen * cv
        return gr
    bd = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0, jac=g, method='L-BFGS-B', bounds=bd,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        if time.time() > deadline:
            return h0
        return project_feasible(r.x, n)
    except Exception:
        return h0

def binary_ls(h, n, deadline, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    cur_c = compute_c5(h, n)
    best_c = cur_c
    best_h = h.copy()
    mr = np.arange(-(n-1), n)
    
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        # Shuffle for diversity
        rng.shuffle(zeros)
        rng.shuffle(ones)
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones:
                im = i + mr
                imn = i - mr
                jm = j + mr
                jmn = j - mr
                d = np.zeros(2 * n - 1)
                m1 = (im >= 0) & (im < n)
                d[m1] += 1.0 - h[im[m1]]
                m2 = (imn >= 0) & (imn < n)
                d[m2] -= h[imn[m2]]
                m3 = (jm >= 0) & (jm < n)
                d[m3] -= 1.0 - h[jm[m3]]
                m4 = (jmn >= 0) & (jmn < n)
                d[m4] += h[jmn[m4]]
                d[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    d[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    d[k2] += 1.0
                nc = (corr + d).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                cur_c = best_nc
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: L-BFGS-B on parent resolution
    p1 = t0 + 0.12 * budget_s
    h = best_h.copy()
    for alpha in [200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > p1:
            break
        h = lbfgs_run(h, n, alpha, 400, p1, pen=2000.0)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Binary local search
    p2 = t0 + 0.45 * budget_s
    hb, cb = binary_ls(best_h, n, p2, max_sweeps=5000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 3: L-BFGS-B from binary
    p3 = t0 + 0.58 * budget_s
    if time.time() < p3:
        h = project_feasible(best_h + 0.08 * rng.randn(n), n)
        for alpha in [2000, 20000, 200000, 1000000]:
            if time.time() > p3:
                break
            h = lbfgs_run(h, n, alpha, 300, p3, pen=5000.0)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
