import numpy as np
from scipy.optimize import minimize
import time

_n_g = 250
_dx_g = 0.008
_alpha_g = 1000.0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _eq_fun(x):
    return x.sum() - _n_g / 2.0

def _smooth_obj(x):
    g = np.correlate(x, 1 - x, mode='full') * _dx_g
    m = g.max()
    sh = _alpha_g * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / _alpha_g

def slsqp_step(h, n, alpha, maxiter, deadline):
    global _n_g, _dx_g, _alpha_g
    _n_g = n
    _dx_g = 2.0 / n
    _alpha_g = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun}]
    try:
        res = minimize(_smooth_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_nesterov(h, n, deadline, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    y = h.copy()
    dx = 2.0 / n
    
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        g = np.correlate(y, 1 - y, mode='full') * dx
        m = g.max()
        k_star = int(np.argmax(g))
        
        # Subgradient of c[k_star] w.r.t. y
        # c[k] = sum_i y[i] * (1-y[i+k-(n-1)])
        # dc[k]/dy[j] = (1-y[j+k-(n-1)]) - y[j] (for valid indices)
        grad = np.zeros(n)
        offset = k_star - (n - 1)
        for j in range(n):
            # j as first index: j + offset in [0, n-1]
            j2 = j + offset
            if 0 <= j2 < n:
                grad[j] += (1 - y[j2])
            # j as second index: j = i + offset, so i = j - offset
            j1 = j - offset
            if 0 <= j1 < n:
                grad[j] += (-y[j1])
        
        # Nesterov step
        alpha_t = 2.0 / (t + 1)
        lr = 0.5 / (t + 1)  # decreasing step size
        x_new = y - lr * grad
        x_new = project_feasible(x_new, n)
        
        beta_t = t / (t + 2.0)
        y = x_new + beta_t * (x_new - y)
        y = project_feasible(y, n)
        
        c = compute_c5(y, n)
        if c < best_c:
            best_c = c
            best_h = y.copy()
    
    return best_h, best_c

def binary_full_search(h_start, n, deadline, max_sweeps=200):
    target = int(n / 2)
    order = np.argsort(-h_start)
    h = np.zeros(n)
    h[order[:target]] = 1.0
    
    best_c = compute_c5(h, n)
    best_h = h.copy()

        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in zeros:
            for j in ones:
                if time.time() > deadline:
                    return best_h, best_c
                h[i], h[j] = 1.0, 0.0
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                else:
                    h[i], h[j] = 0.0, 1.0
            if improved:
                break
    
    return best_h, best_c

def sim_anneal_binary(h_start, n, deadline):
    rng = np.random.RandomState(12345)
    target = int(n / 2)
    order = np.argsort(-h_start)
    h = np.zeros(n)
    h[order[:target]] = 1.0
    
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    
    temps = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    steps_per_temp = 5000
    
    for temp in temps:
        if time.time() > deadline:
            break
        for step in range(steps_per_temp):
            if time.time() > deadline:

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            
            h[i], h[j] = 1.0, 0.0
            new_c = compute_c5(h, n)
            delta = new_c - cur_c
            
            if delta < 0 or rng.random() < np.exp(-delta / temp):
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i], h[j] = 0.0, 1.0
    
    return best_h, best_c

```python
def greedy_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    positions = list(range(n))
    
    for _ in range(target):
        best_pos = -1
        best_c = float('inf')
        for p in positions:
            h[p] = 1
            c = compute_c5(h, n)
            h[p] = 0
            if c < best_c:
                best_c = c
                best_pos = p
        h[best_pos] = 1
        positions.remove(best_pos)
    
    return h
```

This is O(target * n * n^2) = O(n^4) which for n=250 is 250^4 = 3.9 * 10^9. Too slow.

A faster version: instead of computing the full C5 each time, just compute the max correlation incrementally. But that's complex.

Alternatively, use a simpler criterion: minimize the sum of correlations (not the max), which is easier to compute incrementally.

Actually, let me just skip the greedy and focus on what's proven to work.

Let me write the final version now. I'll keep it focused and efficient:

```python
import numpy as np
from scipy.optimize import minimize
import time

_n_g = 250
_dx_g = 0.008
_alpha_g = 1000.0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _eq_fun(x):
    return x.sum() - _n_g / 2.0

def _smooth_obj(x):
    g = np.correlate(x, 1 - x, mode='full') * _dx_g
    m = g.max()
    sh = _alpha_g * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / _alpha_g

def slsqp_step(h, n, alpha, maxiter, deadline):
    global _n_g, _dx_g, _alpha_g
    _n_g = n
    _dx_g = 2.0 / n
    _alpha_g = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun}]
    try:
        res = minimize(_smooth_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_search(h_start, n, deadline, max_sweeps=500):
    target = int(n / 2)
    order = np.argsort(-h_start)
    h = np.zeros(n)
    h[order[:target]] = 1.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones:
                h[i], h[j] = 1.0, 0.0
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                else:
                    h[i], h[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def sa_binary(h_start, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    target = int(n / 2)
    order = np.argsort(-h_start)
    h = np.zeros(n)
    h[order[:target]] = 1.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    temps = [0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for temp in temps:
        if time.time() > deadline:
            break
        steps = int(100000 * (1.0 / (time.time() - (deadline - 300)) + 0.01)) if (deadline - 300) > time.time() else 5000
        # Simpler: just do a fixed number of steps per temp
        steps = 8000
        for _ in range(steps):
            if time.time() > deadline:
                return best_h, best_c
            i = zeros_list[rng.randint(len(zeros_list))]
            j = ones_list[rng.randint(len(ones_list))]
            h[i], h[j] = 1.0, 0.0
            new_c = compute_c5(h, n)
            delta = new_c - cur_c
            if delta < 0 or rng.random() < np.exp(min(0, -delta / temp)):
                cur_c = new_c
                ones_list.remove(j)
                ones_list.append(i)
                zeros_list.remove(i)
                zeros_list.append(j)
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i], h[j] = 0.0, 1.0
    return best_h, best_c
