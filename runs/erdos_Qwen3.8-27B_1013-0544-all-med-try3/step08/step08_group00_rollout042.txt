import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project onto {0 <= h <= 1, sum(h) = n/2} using bisection on lambda."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    
    # Bisection on lambda: find lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h + mid, 0.0, 1.0)
        s = h_try.sum()
        if abs(s - target) < 1e-15:
            break
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final fix for numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-14:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_subgradient(h, n, c_vec, alpha=10000.0):
    """Compute smoothed subgradient of max_k c_k(h)."""
    m = c_vec.max()
    sh = alpha * (c_vec - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-16:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    h = project_feasible(h, n)

def lp_active_set_step(h, n, deadline, max_active=80, step_size=0.05):
    """LP step using active constraints."""
    c_vec = compute_c5_vec(h, n)
    c_max = c_vec.max()
    
    # Identify active set
    thresh = c_max - max(1e-10, 0.002 * c_max)
    active = np.where(c_vec >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    
    # Limit to top max_active
    if len(active) > max_active:
        top_order = np.argsort(-c_vec[active])
        active = active[top_order[:max_active]]
    
    # Build LP: minimize t
    # s.t. g_k^T delta - t <= -c_k for k in active
    #      sum(delta) = 0
    #      -step_size <= delta[i] <= step_size
    #      0 <= h[i] + delta[i] <= 1
    
    n_var = n + 1  # delta[n] = t
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((len(active) + n * 4, n_var))
    b_ub = np.zeros(len(active) + n * 4)
    row = 0
    
    dx = 2.0 / n
    for idx in active:
        k = int(idx) - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[0:nv] = dx * (1.0 - h[k:n])
            g[k:n] += dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:n] = dx * (1.0 - h[0:nv])
            g[0:nv] += dx * (-h[kk:n])
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    
    for i in range(n):
        # delta[i] <= step_size
        A_ub[row, i] = 1.0
        b_ub[row] = step_size
        row += 1
        # -delta[i] <= step_size  =>  delta[i] >= -step_size
        A_ub[row, i] = -1.0
        b_ub[row] = step_size
        row += 1
        # h[i] + delta[i] <= 1  =>  delta[i] <= 1 - h[i]
        A_ub[row, i] = 1.0
        b_ub[row] = min(step_size, 1.0 - h[i] + 1e-12)
        row += 1
        # -(h[i] + delta[i]) <= -0  =>  delta[i] >= -h[i]
        A_ub[row, i] = -1.0
        b_ub[row] = min(step_size, h[i] + 1e-12)
        row += 1
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and c_vec.max() + res.fun > 0:
            delta = res.x[0:n]
            h_new = project_feasible(h + delta, n)
            return h_new
    except Exception:
        pass
    return None

def local_search_refined(h, n, deadline, radius=30):

            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            c_max = c_vec.max()
            thresh = c_max - max(1e-8, 0.005 * c_max)
            active = np.where(c_vec >= thresh)[0]
            
            # Compute aggregate gradient magnitude per coordinate

            for idx in active:
                k = int(idx) - (n - 1)
                if k >= 0:
                    nv = n - k
                    g = np.zeros(n)

                    nv = n - kk
                    g = np.zeros(n)
                    g[kk:n] = (1.0 - h[0:nv])
                    g[0:nv] -= h[kk:n]
                    grad_mag += np.abs(g)
            
            # Focus on coordinates with high gradient magnitude
            top_coords = np.argsort(-grad_mag)[:min(radius * 2, n)]
            
            for i_val in top_coords:

                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:

                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

def smooth_obj_full(x, n, alpha, mu):

    else:

def smooth_grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - x[k:n])

        mid = (lo + hi) / 2.0
        h_try = np.clip(h + mid, 0.0, 1.0)
        s = h_try.sum()
        if abs(s - target) < 1e-15:
            lo = hi = mid
            break
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-14:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    active_mask = w > 1e-14
    active_idx = np.where(active_mask)[0]
    for mm in active_idx:
        k = int(mm) - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def solve_qp_lp(h, n, active_list, step_size):
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_list)
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    n_ub = n_act + 2 * n
    A_ub = np.zeros((n_ub, n_var))
    b_ub = np.zeros(n_ub)
    row = 0
    for idx in active_list:
        idx = int(idx)
        g = grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        ub_i = min(step_size, 1.0 - h[i] + 1e-10)
        A_ub[row, i] = 1.0
        b_ub[row] = ub_i
        row += 1
        lb_i = min(step_size, h[i] + 1e-10)
        A_ub[row, i] = -1.0
        b_ub[row] = lb_i
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub[:row], b_ub=b_ub[:row],
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n]
    except Exception:
        pass
    return None

def sqp_refined(h, n, deadline, max_active=80, rho_init=0.04, max_iter=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rho = rho_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        thresh = c_max - max(1e-10, 0.003 * c_max)
        active = np.where(c_vec >= thresh)[0].tolist()
        if not active:
            active = [int(np.argmax(c_vec))]
        if len(active) > max_active:
            order = np.argsort(-c_vec[np.array(active)])
            active = [active[i] for i in order[:max_active]]
        delta = solve_qp_lp(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-9:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            if rho < 0.06:
                rho = min(rho * 1.15, 0.06)
        else:
            stall += 1
            rho *= 0.5
            if rho < 1e-10 or stall > 30:
                break
    return best_h, best_c

def fw_optimize(h, n, deadline, alpha=100000.0, max_iter=20000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    n_ones = n // 2
    for t in range(max_iter):
        if time.time() > deadline - 2:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = float(c_vec.max())
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        sh = alpha * (c_vec - c_max)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = np.zeros(n)
        active_mask = w > 1e-14
        active_idx = np.where(active_mask)[0]
        dx = 2.0 / n
        for mm in active_idx:
            k = int(mm) - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += wm * (1.0 - h[k:n])
                    grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += wm * (1.0 - h[0:nv])
                    grad[0:nv] += wm * (-h[kk:n])
        sorted_idx = np.argsort(grad)
        v = np.zeros(n)
        v[sorted_idx[:n_ones]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h = project_feasible(h_new, n)
    return h, best_c

def subgrad_optimize(h, n, deadline, alpha=100000.0, max_iter=5000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for t in range(max_iter):
        if time.time() > deadline - 2:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = float(c_vec.max())
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        sh = alpha * (c_vec - c_max)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = np.zeros(n)
        active_mask = w > 1e-14
        active_idx = np.where(active_mask)[0]
        dx = 2.0 / n
        for mm in active_idx:
            k = int(mm) - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += wm * (1.0 - h[k:n])
                    grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += wm * (1.0 - h[0:nv])
                    grad[0:nv] += wm * (-h[kk:n])
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        eta = 0.05 / (1.0 + 0.005 * t)
        h_new = h - eta * grad / gnorm
        h = project_feasible(h_new, n)
    return h, best_h, best_c

def local_search(h, n, deadline, radius=35):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            c_max = c_vec.max()
            thresh = c_max - max(1e-8, 0.004 * c_max)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad_mag = np.zeros(n)
            for idx in active:
                k = int(idx) - (n - 1)
                if k >= 0:
                    nv = n - k
                    if nv > 0:
                        grad_mag[0:nv] += np.abs(1.0 - h[k:n])
                        grad_mag[k:n] += np.abs(h[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    if nv > 0:
                        grad_mag[kk:n] += np.abs(1.0 - h[0:nv])
                        grad_mag[0:nv] += np.abs(h[kk:n])
            top = np.argsort(-grad_mag)[:min(radius * 2, n)]
            for i_val in top:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return best_h, best_c

def lbfgs_phase(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj_fn, h, jac=smooth_grad_fn,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 15000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def perturb_and_retry(h, n, deadline, n_perturb=5, max_active=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    for p in range(n_perturb):
        if time.time() > deadline - 30:
            break
        h_p = h + rng.randn(n) * 0.02
        h_p = project_feasible(h_p, n)
        h_p, c_p = sqp_refined(h_p, n, min(time.time() + 60, deadline - 20),
                               max_active=max_active, rho_init=0.03, max_iter=100)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS warmup with aggressive smoothing (70s)
    t1 = min(t0 + 70, deadline - 850)
    if t1 > time.time() + 5:
        h, best_c = lbfgs_phase(h, n, t1,
            configs=[(300, 50.0), (1000, 200.0), (3000, 1000.0),
                     (8000, 5000.0), (15000, 15000.0), (30000, 30000.0)])
        best_h = h.copy()

    # Phase 2: SQP with many active constraints (100s)
    t2 = min(t0 + 170, deadline - 750)
    if t2 > time.time() + 10:
        h, best_c = sqp_refined(h, n, t2, max_active=100, rho_init=0.05, max_iter=300)
        best_h = h.copy()

    # Phase 3: Frank-Wolfe (60s)
    t3 = min(t0 + 230, deadline - 650)
    if t3 > time.time() + 10:
        h_fw = project_feasible(best_h, n)
        h_fw, c_fw = fw_optimize(h_fw, n, t3, alpha=200000.0, max_iter=30000)
        if c_fw < best_c:
            best_c = c_fw
            best_h = h_fw.copy()
            h = h_fw
        else:
            h = best_h.copy()

    # Phase 4: SQP again after FW (80s)
    t4 = min(t0 + 310, deadline - 550)
    if t4 > time.time() + 10:
        h, best_c = sqp_refined(h, n, t4, max_active=100, rho_init=0.03, max_iter=250)
        best_h = h.copy()

    # Phase 5: Subgradient (50s)
    t5 = min(t0 + 360, deadline - 500)
    if t5 > time.time() + 10:
        h_sub, best_sub_h, c_sub = subgrad_optimize(h, n, t5, alpha=200000.0, max_iter=5000)
        if c_sub < best_c:
            best_c = c_sub
            best_h = best_sub_h
            h = h_sub

    # Phase 6: SQP polish (70s)
    t6 = min(t0 + 430, deadline - 420)
    if t6 > time.time() + 10:
        h, best_c = sqp_refined(h, n, t6, max_active=120, rho_init=0.02, max_iter=200)
        best_h = h.copy()

    # Phase 7: Local search (120s)
    t7 = min(t0 + 550, deadline - 300)
    if t7 > time.time() + 5:
        h, best_c = local_search(h, n, t7, radius=40)
        best_h = h.copy()

    # Phase 8: Perturb and retry (80s)
    t8 = min(t0 + 630, deadline - 200)
    if t8 > time.time() + 20:
        h, best_c = perturb_and_retry(h, n, t8, n_perturb=4, max_active=80)
        best_h = h.copy()

    # Phase 9: Final SQP (40s)
    t9 = min(t0 + 670, deadline - 150)
    if t9 > time.time() + 5:
        h, best_c = sqp_refined(h, n, t9, max_active=100, rho_init=0.015, max_iter=150)
        best_h = h.copy()

    # Phase 10: Upsample to n=1000 and optimize
    n_high = 1000
    if time.time() < deadline - 200:
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n_high, endpoint=False) + 1.0 / n_high
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)

        t10 = min(time.time() + 70, deadline - 100)
        if t10 > time.time() + 5:
            h_high, c_high = lbfgs_phase(h_high, n_high, t10,
                configs=[(2000, 200.0), (5000, 1000.0), (10000, 5000.0), (20000, 20000.0)])

        t11 = min(time.time() + 50, deadline - 40)
        if t11 > time.time() + 5:
            h_high, c_high = sqp_refined(h_high, n_high, t11, max_active=80, rho_init=0.03, max_iter=100)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n = n_high
            h = h_high

    # Final local search on best
    if time.time() < deadline - 15:
        t12 = deadline - 10
        h, best_c = local_search(h, len(h), t12, radius=25)
        best_h = h

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
