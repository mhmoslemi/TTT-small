    a = max(-h[i], h[j] - 1.0)
    b = min(1.0 - h[i], h[j])
    if a >= b:
        return h, compute_c5(h, n)
    
    def obj(t):
        h_new = h.copy()
        h_new[i] = h[i] + t
        h_new[j] = h[j] - t

            c1 = c2
            f1 = f2
