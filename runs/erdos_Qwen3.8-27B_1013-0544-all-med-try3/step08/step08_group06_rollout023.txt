    # Clip indices to valid range for indexing
    fwd_clipped = np.clip(fwd, 0, n-1)

    bwd_vals = x[bwd_clipped] * bwd_valid  # n x (2n-1)

def smooth_obj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    s = c.max()
    shifted = alpha * (c - s)
    logsum = s + np.log(np.sum(np.exp(shifted))) / alpha
    sum_diff = x.sum() - n / 2.0
    return float(logsum + mu * sum_diff * sum_diff)

def smooth_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    s = c.max()
    shifted = alpha * (c - s)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()

    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    fwd = i_idx + m_idx - (n - 1)
    bwd = i_idx - m_idx + (n - 1)
    fwd_valid = (fwd >= 0) & (fwd < n)
    bwd_valid = (bwd >= 0) & (bwd < n)
    fwd_clipped = np.clip(fwd, 0, n - 1)
    bwd_clipped = np.clip(bwd, 0, n - 1)
    fwd_vals = (1 - x[fwd_clipped]) * fwd_valid
    bwd_vals = x[bwd_clipped] * bwd_valid
    grad_corr = dx * (fwd_vals @ w) - dx * (bwd_vals @ w)

    sum_diff = x.sum() - n / 2.0
    grad = grad_corr + 2 * mu * sum_diff * np.ones(n)
    return grad

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    def obj_only(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        s = c.max()
        shifted = alpha * (c - s)
        return float(s + np.log(np.sum(np.exp(shifted))) / alpha)

    def obj_jac(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        s = c.max()
        shifted = alpha * (c - s)
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        i_idx = np.arange(n)[:, None]
        m_idx = np.arange(2 * n - 1)[None, :]
        fwd = i_idx + m_idx - (n - 1)
        bwd = i_idx - m_idx + (n - 1)
        fwd_valid = (fwd >= 0) & (fwd < n)
        bwd_valid = (bwd >= 0) & (bwd < n)
        fwd_clipped = np.clip(fwd, 0, n - 1)
        bwd_clipped = np.clip(bwd, 0, n - 1)
        fwd_vals = (1 - x[fwd_clipped]) * fwd_valid
        bwd_vals = x[bwd_clipped] * bwd_valid
        grad = dx * (fwd_vals @ w) - dx * (bwd_vals @ w)
        val = obj_only(x)
        return val, grad

    try:
        res = minimize(obj_jac, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lbfgs_optimize(h, n, alpha, mu=1e4, maxiter=2000, gtol=1e-14):

    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': gtol,
                                'maxls': 50})
        return project_feasible(res.x, n)

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        while time.time() < deadline:
            improved = False
            passes = 0
            while improved and passes < 200 and time.time() < deadline:
                improved = False
                passes += 1
                for i in range(n):
                    if time.time() > deadline:
                        return h, best_c
                    if radius is None:
                        j_range = range(i + 1, n)
                    else:
                        j_lo = max(0, i - radius)
                        j_hi = min(n, i + radius + 1)
                        j_range = range(j_lo, j_hi)
                    for j in j_range:
                        if j == i:
                            continue
                        for sign in [1, -1]:
                            hi = h[i] + sign * d
                            hj = h[j] - sign * d
                            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

                                if c < best_c - 1e-15:
                                    best_c = c
                                    improved = True
                                else:
                                    h[i], h[j] = hi - sign * d, hj + sign * d
    return h, best_c

def local_1d_optimize(h, n, i, j, deadline):
    """Try to optimize the pair (i,j) by 1D search over h[i]+h[j]=const."""
    original_i, original_j = h[i], h[j]
    s = original_i + original_j
    lo = max(0.0, s - 1.0)
    hi_b = min(1.0, s)
    if lo >= hi_b:
        return h

    best_t = original_i
    best_c = compute_c5(h, n)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi_b
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    h[i], h[j] = c_pt, s - c_pt
    fc = compute_c5(h, n)
    h[i], h[j] = d_pt, s - d_pt
    fd = compute_c5(h, n)

    for _ in range(60):
        if time.time() > deadline:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            h[i], h[j] = c_pt, s - c_pt
            fc = compute_c5(h, n)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            h[i], h[j] = d_pt, s - d_pt
            fd = compute_c5(h, n)

                if (i % period) < period // 2:

            pat = project_feasible(pat, n)
            structured_starts.append(pat)
        # Smooth sinusoidal
        for freq in [2, 3, 4, 5, 7]:
            pat = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0)
            pat = np.clip(pat, 0, 1)
            pat = project_feasible(pat, n)
            structured_starts.append(pat)

        for pat in structured_starts:
            if time.time() > deadline - 30:
                break
            remaining = deadline - time.time() - 25
            alloc = min(35, remaining * 0.25)
            if alloc < 15:
                break
            # Blend with best to stay in the right region
            blend = 0.5 * best_h + 0.5 * pat
            blend = project_feasible(blend, n)
            h_opt, c_opt = optimize_resolution(blend, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final intensive polish on best
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Final SLSQP with very high alpha
        if n <= 300:
            ht = slsqp_optimize(best_h, n, 5000000, maxiter=3000, ftol=1e-16)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
        # Final pair search with tiny steps
        if time.time() < deadline - 1:
            dset = [0.0005, 0.0002, 0.0001, 0.00005]
            if n <= 300:
                best_h, best_c = pair_search(best_h, n, deadline - 0.3, dset)
            else:
                best_h, best_c = pair_search(best_h, n, deadline - 0.3, dset[1:], radius=50)

    # Final validation
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
