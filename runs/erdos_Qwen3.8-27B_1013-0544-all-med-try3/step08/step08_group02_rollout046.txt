        if count % 500 == 0 and eta > 0.001:
            eta *= 0.995
    return best_h, best_c

def directed_pair_search(h, n, deadline, max_rounds=200):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(99)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for di, d in enumerate(deltas):
        improved = True
        rounds = 0
        while improved and rounds < max_rounds and time.time() < deadline:
            improved = False
            rounds += 1
            # Focus on indices near the "action" (where h is between 0 and 1)
            mid_mask = (best_h > 0.01) & (best_h < 0.99)
            mid_idx = np.where(mid_mask)[0]
            if len(mid_idx) == 0:
                break
            # Sample pairs from mid_idx and high/low regions
            for _ in range(min(2000, len(mid_idx) * 10)):
                if time.time() > deadline:
                    break
                if len(mid_idx) >= 2:
                    idx_a = mid_idx[rng.randint(len(mid_idx))]
                else:
                    idx_a = mid_idx[0]
                idx_b = rng.randint(n)
                if idx_a == idx_b:
                    continue
                # Try transfers
                d2 = d * (0.5 + rng.random())
                if d2 > 0.02:
                    d2 = 0.02
                if best_h[idx_a] + d2 <= 1.0 and best_h[idx_b] - d2 >= 0.0:
                    best_h[idx_a] += d2
                    best_h[idx_b] -= d2
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        best_h[idx_a] -= d2
                        best_h[idx_b] += d2
                if best_h[idx_a] - d2 >= 0.0 and best_h[idx_b] + d2 <= 1.0:
                    best_h[idx_a] -= d2
                    best_h[idx_b] += d2
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        best_h[idx_a] += d2
                        best_h[idx_b] -= d2
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=8000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    eta = 0.003
    dx_val = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    c0 = compute_c5(h0, n_orig)
    
    # Target resolution: use n=1000 for more freedom
    n_use = 1000
    if n_orig != n_use:
        x_fine = np.linspace(0, 2, n_use)
        x_coarse = np.linspace(0, 2, n_orig)
        h_start = np.interp(x_fine, x_coarse, h0)
    else:
        h_start = h0.copy()
    h_start = project_feasible(h_start, n_use)
    c_start = compute_c5(h_start, n_use)
    
    best_h = h_start.copy()
    best_c = c_start
    best_n = n_use
    
    # Also track the original resolution solution
    best_h_orig = h0.copy()
    best_c_orig = c0
    
    # Phase 1: L-BFGS warmup at n=1000 (fast convergence)
    p1_end = min(t0 + 250, deadline - 500)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[30, 100, 500, 2000, 10000, 50000],
        mu=0.01, maxiters=[1500, 1500, 2000, 2000, 2500, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active-set focused optimization (top 3 shifts)
    p2_end = min(time.time() + 250, deadline - 300)
    h2, c2 = run_active(best_h, best_n, p2_end, top_k=3,
        alphas=[50000, 200000, 1000000, 5000000], mu=0.003)
    if c2 < best_c:
        best_c = c2
        best_h = h2
    
    # Phase 3: Stochastic local search
    p3_end = min(time.time() + 150, deadline - 200)
    h3, c3 = stochastic_local_search(best_h, best_n, p3_end, n_iter=80000)
    if c3 < best_c:
        best_c = c3
        best_h = h3
    
    # Phase 4: Directed pair search
    p4_end = min(time.time() + 150, deadline - 120)
    h4, c4 = directed_pair_search(best_h, best_n, p4_end, max_rounds=300)
    if c4 < best_c:
        best_c = c4
        best_h = h4
    
    # Phase 5: Another active-set round with even more focus
    p5_end = min(time.time() + 100, deadline - 80)
    h5, c5_val = run_active(best_h, best_n, p5_end, top_k=2,
        alphas=[500000, 2000000, 10000000], mu=0.001)
    if c5_val < best_c:
        best_c = c5_val
        best_h = h5
    
    # Phase 6: Subgradient refinement
    p6_end = min(time.time() + 80, deadline - 40)
    h6, c6 = subgrad_refine(best_h, best_n, p6_end, n_iter=10000)
    if c6 < best_c:
        best_c = c6
        best_h = h6
    
    # Phase 7: SLSQP polish
    p7_end = min(time.time() + 60, deadline - 15)
    for alpha in [200000, 1000000]:
        if time.time() > p7_end:
            break
        h7, c7 = slqp_polish(best_h, best_n, p7_end, alpha, maxiter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Final tiny moves
    if time.time() < deadline - 3:
        h8, c8 = directed_pair_search(best_h, best_n, deadline - 2, max_rounds=50)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    # Compare with original resolution
    if c0 < best_c:
        best_h = project_feasible(best_h_orig, n_orig)
        best_c = compute_c5(best_h, n_orig)
        best_n = n_orig
    
    return best_h, best_c, best_n
