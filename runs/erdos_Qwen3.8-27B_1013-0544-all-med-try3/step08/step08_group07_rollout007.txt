            eta = (t_mom - 1.0) / t_new
            y = h + eta * (h - h_prev) - lr * grad
            h_prev = h.copy()
            h = project_simplex(y, n)
            h = project_simplex(h, n)
            t_mom = t_new
            if lr < 1e-10:
                break
        h = best_h.copy()
    return best_h, best_c

def subgrad_descent(h, n, deadline, max_iter, init_lr=0.5):
    best_h = h.copy()
    best_c = compute_c5(h)
    h = h.copy()
    lr = init_lr
    stall = 0
    rng = np.random.RandomState(12345)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h)
        m = float(g.max())
        if m < best_c:
            best_c = m
            best_h = h.copy()
        k_star = int(np.argmax(g))
        dx = 2.0 / n
        s = k_star - (n - 1)
        grad = np.zeros(n)
        one_mh = 1.0 - h
        if s >= 0:
            grad[:n - s] += one_mh[s:]
            grad[s:] -= h[:n - s]
        else:
            grad[-s:] += one_mh[:n + s]
            grad[:n + s] -= h[-s:]
        grad *= dx
        h_new = project_simplex(h - lr * grad, n)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-14:
            h = h_new
            stall = 0
            lr = min(lr * 1.02, 2.0)
        else:
            stall += 1
            lr *= 0.97
            if lr < 1e-9:
                break
        if stall > 200:
            h = project_simplex(h + 0.02 * rng.randn(n), n)
            lr = 0.1
            stall = 0
    return best_h, best_c

def top_k_focus_descent(h, n, deadline, max_iter, n_top=5):
    best_h = h.copy()
    best_c = compute_c5(h)
    h = h.copy()
    lr = 0.3
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h)
        m = float(g.max())
        # Get top n_top lags
        top_idx = np.argsort(g)[::-1][:n_top]
        if g[top_idx[0]] - g[top_idx[-1]] < 1e-6:
            top_idx = top_idx
        else:
            thresh = g.max() - 0.01
            top_idx = np.where(g >= thresh)[0][:n_top]
        dx = 2.0 / n
        grad = np.zeros(n)
        one_mh = 1.0 - h
        for k in top_idx:
            s = k - (n - 1)
            if s >= 0:
                grad[:n - s] += one_mh[s:]
                grad[s:] -= h[:n - s]
            else:
                grad[-s:] += one_mh[:n + s]
                grad[:n + s] -= h[-s:]
        grad *= dx / len(top_idx)
        h_new = project_simplex(h - lr * grad, n)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.01, 1.0)
        else:
            stall += 1
            lr *= 0.96
            h = h_new
            if lr < 1e-8:
                break
        if stall > 150:
            h = project_simplex(h + 0.01 * np.random.randn(n), n)
            lr = 0.05
            stall = 0
    return best_h, best_c

def coordinate_polish(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h)
    h = h.copy()
    target = n / 2.0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                continue
            g = compute_g(h)
            k_star = int(np.argmax(g))
            s = k_star - (n - 1)
            # Compute effect of changing h[i]
            # Try small steps
            for delta in [0.05, -0.05, 0.01, -0.01, 0.001, -0.001]:
                new_val = h[i] + delta
                if new_val < 0 or new_val > 1:
                    continue
                # Adjust another index to maintain sum
                h[i] = new_val
                diff = target - h.sum()
                if abs(diff) < 1e-10:
                    pass
                else:
                    # Find best index to adjust
                    if diff > 0:
                        candidates = np.where(h < 0.99)[0]
                    else:
                        candidates = np.where(h > 0.01)[0]
                    if len(candidates) == 0:
                        h[i] = new_val - delta
                        break
                    j = candidates[np.argmax(h) if diff < 0 else np.argmin(h)]
                    h[j] = np.clip(h[j] + diff, 0, 1)
                    # Fix remaining
                    rem = target - h.sum()
                    if abs(rem) > 1e-10:
                        h = project_simplex(h, n)
                        h[i] = np.clip(new_val, 0, 1)
                        h = project_simplex(h, n)
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] = new_val - delta
            if not improved and time.time() > deadline:
                break
        if not improved:
            break
    return best_h, best_c

def slsqp_final(h, n, deadline, alpha=50000, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        gm = g - m
        sm = gm.max()
        return m + (sm + np.log(np.exp(gm - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < compute_c5(h):
            return project_simplex(res.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    n = n0
    
    # Phase 1: Nesterov at original resolution (20% budget)
    p1_dl = t0 + 0.20 * (budget_s - 5)
    if time.time() < p1_dl:
        alphas = [30, 100, 300, 1000, 3000]
        h, c = nesterov_optimize(best_h.copy(), n, p1_dl, alphas, 800, lr_scale=1.0)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Subgradient + top-k focus at original resolution (15% budget)
    p2_dl = t0 + 0.35 * (budget_s - 5)
    if time.time() < p2_dl:
        h, c = subgrad_descent(best_h.copy(), n, p2_dl, 20000, init_lr=0.4)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h, c = top_k_focus_descent(best_h.copy(), n, p2_dl, 15000, n_top=5)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 3: Higher resolution (45% budget)
    p3_dl = t0 + 0.80 * (budget_s - 5)
    resolutions = [350, 450, 550]
    for nh in resolutions:
        if time.time() > p3_dl - 80:
            break
        if nh <= n:
            continue
        # Interpolate
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_simplex(np.interp(xn, xo, best_h), nh)
        
        # Nesterov at new resolution
        alphas = [50, 200, 800, 3000]
        hh, c = nesterov_optimize(hh, nh, p3_dl - 30, alphas, 600, lr_scale=0.8)
        if c < best_c:
            best_c = c
            best_h = hh.copy()
            n = nh
        
        # Subgradient polish
        if time.time() < p3_dl - 20:
            hh2, c2 = subgrad_descent(best_h.copy(), n, p3_dl - 10, 10000, init_lr=0.3)
            if c2 < best_c:
                best_c = c2
                best_h = hh2.copy()
    
    # Phase 4: Final high-alpha Nesterov (8% budget)
    p4_dl = t0 + 0.88 * (budget_s - 5)
    if time.time() < p4_dl:
        alphas = [5000, 15000, 40000]
        h, c = nesterov_optimize(best_h.copy(), n, p4_dl, alphas, 1500, lr_scale=0.5)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 5: SLSQP final polish (7% budget)
    p5_dl = t0 + 0.95 * (budget_s - 5)
    if time.time() < p5_dl:
        h2 = slsqp_final(best_h.copy(), n, p5_dl, alpha=80000, maxiter=300)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
    
    # Phase 6: Final subgradient micro-polish
    if time.time() < deadline:
        h3, c3 = subgrad_descent(best_h.copy(), n, deadline, 20000, init_lr=0.1)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    
    best_h = project_simplex(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c, n
