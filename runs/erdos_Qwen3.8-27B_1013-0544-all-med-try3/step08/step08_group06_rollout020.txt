```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact sum fix
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            if room.sum() > 1e-14:
                h += (diff / room.sum()) * room
        else:
            if h.sum() > 1e-14:
                h += (diff / h.sum()) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted_max = shifted.max()
    if shifted_max > 500:
        val = m + (shifted_max + np.log(np.sum(np.exp(shifted - shifted_max)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def slsqp_step(h, n, alpha, maxiter=2000, ftol=1e-15):
    target_sum = n / 2.0
    def obj(x):
        return smooth_obj(x, n, alpha)
    def con_fun(x):
        return x.sum() - target_sum
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < smooth_obj(h, n, alpha):
            return project_feasible(res.x, n)
        return h.copy()
    except Exception:
        return h.copy()

def compute_D(h, n):
    """Compute the sensitivity matrix D[p, k] where k=0,...,2n-2
    D[p,k] = change in corr[k] when h[p] increases by 1 (before scaling by dx)
    Actually, D[p,k] = dc[k]/dh[p] where corr = correlate(h, 1-h, 'full') * dx
    """
    # corr[k] = sum_j h[j] * (1-h[j-r]) * dx where r = k-n+1
    # dc[k]/dh[p] = (1-h[p-r])*[0<=p-r<n] - h[p+r]*[0<=p+r<n] (times dx)
    dx = 2.0 / n
    r = np.arange(-(n-1), n)  # length 2n-1
    p_idx = np.arange(n)[:, None]  # n x 1
    pr_minus = p_idx - r[None, :]  # n x (2n-1): p - r
    pr_plus = p_idx + r[None, :]   # n x (2n-1): p + r
    mask_fm = (pr_minus >= 0) & (pr_minus < n)  # p - r in [0, n)
    mask_fp = (pr_plus >= 0) & (pr_plus < n)    # p + r in [0, n)
    D = np.zeros((n, 2*n-1))
    D[mask_fm] += (1.0 - h[pr_minus[mask_fm]])
    D[mask_fp] -= h[pr_plus[mask_fp]]
    D *= dx
    return D

def fast_local_search(h_in, n, deadline, deltas=None, max_passes=300):
    """Fast pair-move local search with O(n) per candidate move."""
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    h = np.array(h_in, dtype=float).copy()
    c = compute_corr(h, n)
    best_c = float(c.max())
    D = compute_D(h, n)
    
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i+1, n):
                    # Direction 1: h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        delta_c = d * (D[i] - D[j])
                        new_max = float((c + delta_c).max())
                        if new_max < best_c - 1e-15:
                            best_c = new_max
                            c += delta_c
                            h[i] += d
                            h[j] -= d
                            D = compute_D(h, n)
                            improved = True
                    # Direction 2: h[i] -= d, h[j] += d
                    if h[i] - d >= -1e-15 and h[j] + d <= 1.0 + 1e-15:
                        delta_c = d * (D[j] - D[i])
                        new_max = float((c
