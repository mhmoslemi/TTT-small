```python
def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha, lam):
    dx = 2.0 / n
    xc = np.clip(x, 0, 1)
    c = np.correlate(xc, 1 - xc, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    val = m + np.log(np.sum(np.exp(shifted))) / alpha
    val += lam * (xc.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, lam):
    dx = 2.0 / n
    xc = np.clip(x, 0, 1)

    w = np.exp(shifted)
    ws = w / w.sum()  # length 2n-1

    first = np.convolve(1 - xc, ws)[:n]
    second = np.correlate(xc, ws, 'full')[:n]
    grad = dx * (first - second)
```

```python
def precompute_grads(h, n):
    """Precompute ∂(c[k]*dx)/∂h[i] for all i,k. Returns (n, 2n-1) array."""
    dx = 2.0 / n
    ncorr = 2 * n - 1
    grads = np.zeros((n, ncorr))
    for i in range(n):
        for k in range(ncorr):
            val = 0.0
            if 0 <= i - k < n:
                val += (1 - h[i - k])
            if 0 <= i + k < n:
                val -= h[i + k]
            grads[i, k] = val
    return grads * dx
```

```python
    # params: [h1, h2, ..., hM, b1, b2, ..., b_{M-1}]
    # Build h on grid of n points
    M_h = M
    M_b = M - 1
    heights = params[:M_h]
    breakpoints = params[M_h:]  # M-1 breakpoints in [0, 2]

    breakpoints = np.sort(breakpoints)

    full_bp = np.concatenate([[0.0], breakpoints, [2.0]])

    if np.any(np.diff(full_bp) < 1e-8):

    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # midpoints
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smo_val(x, n, alpha, lam):
    dx = 2.0 / n
    xc = np.clip(x, 0, 1)
    c = np.correlate(xc, 1 - xc, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    val = float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    val += lam * (xc.sum() - n / 2.0) ** 2
    return val

def _smo_grad(x, n, alpha, lam):
    dx = 2.0 / n
    xc = np.clip(x, 0, 1)
    c = np.correlate(xc, 1 - xc, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    w = np.exp(shifted)
    ws = w / w.sum()
    first = np.convolve(1 - xc, ws)[:n]
    second = np.correlate(xc, ws, 'full')[:n]
    grad = dx * (first - second)
    grad += 2.0 * lam * (xc.sum() - n / 2.0)
    return grad

def lbfgs_opt(h0, n, alpha, lam, maxiter=400):
    x0 = project_feasible(h0.copy(), n)
    bounds = [(0.0, 1.0)] * n
    opts = {'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7, 'maxfun': maxiter * 5}
    try:
        res = minimize(_smo_val, x0, jac=_smo_grad, args=(n, alpha, lam),
                       method='L-BFGS-B', bounds=bounds, options=opts)
        return project_feasible(res.x, n)
    except Exception:
        return x0

def _slsqp_obj(x, n, alpha, dummy):
    dx = 2.0 / n
    xc = np.clip(x, 0, 1)
    c = np.correlate(xc, 1 - xc, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _slsqp_grad(x, n, alpha, dummy):
    dx = 2.0 / n
    xc = np.clip(x, 0, 1)
    c = np.correlate(xc, 1 - xc, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    w = np.exp(shifted)
    ws = w / w.sum()
    first = np.convolve(1 - xc, ws)[:n]
    second = np.correlate(xc, ws, 'full')[:n]
    return dx * (first - second)

def _cons_fun(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def slsqp_polish(h0, n, alpha, maxiter=2000):
    x0 = project_feasible(h0.copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, x0, jac=_slsqp_grad, args=(n, alpha, 0),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return x0

def precompute_corr_grads(h, n):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    grads = np.zeros((n, ncorr))
    for i in range(n):
        idx1 = np.arange(max(0, i - (n - 1)), min(n, i + n))
        k1 = i - idx1
        valid1 = (k1 >= 0) & (k1 < ncorr)
        grads[i, k1[valid1]] += (1.0 - h[idx1[valid1]]) * dx
        idx2 = np.arange(max(0, -i), min(n, ncorr - i))
        k2 = i + idx2
        valid2 = (k2 >= 0) & (k2 < ncorr)
        grads[i, k2[valid2]] -= h[idx2[valid2]] * dx
    return grads

def fast_pair_search(h, n, deadline, deltas, radius=40, max_passes=40):
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    best_c = float(c.max())
    ncorr = 2 * n - 1
    for d in deltas:
        if time.time() > deadline:
            break
        grads = precompute_corr_grads(h, n)
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        dc = d * (grads[i] - grads[j])
                        nm = (c + dc).max()
                        if nm < best_c - 1e-13:
                            h[i] = min(1.0, h[i] + d)
                            h[j] = max(0.0, h[j] - d)
                            c = c + dc
                            best_c = nm
                            improved = True
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        dc = -d * (grads[i] - grads[j])
                        nm = (c + dc).max()
                        if nm < best_c - 1e-13:
                            h[i] = max(0.0, h[i] - d)
                            h[j] = min(1.0, h[j] + d)
                            c = c + dc
                            best_c = nm
                            improved = True
            if not improved:
                break
        c = compute_c5_vec(h, n)
        best_c = float(c.max())
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [500, 2000, 10000, 50000, 200000]
    lams = [10.0, 100.0, 1000.0, 5000.0, 10000.0]
    for idx, a in enumerate(alphas):
        if time.time() > deadline - 10:
            break
        lam = lams[min(idx, len(lams) - 1)]
        ht = lbfgs_opt(best_h, n, a, lam, maxiter=300)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    if time.time() < deadline - 15:
        ht = slsqp_polish(best_h, n, 100000, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    if time.time() < deadline - 5:
        best_h, best_c = fast_pair_search(best_h, n, deadline - 2,
                                          [0.01, 0.005, 0.002, 0.001], radius=35)
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def diverse_start(base_h, n, rng, scale=0.05):
    h = base_h + rng.normal(0, scale, n)
    return project_feasible(h, n)

def step_init(n, rng, n_steps):
    h = np.zeros(n)
    edges = rng.choice(np.arange(1, n), n_steps - 1, replace=False)
    edges = np.sort(edges)
    all_edges = np.concatenate([[0], edges, [n]])
    for s in range(n_steps):
        h[all_edges[s]:all_edges[s + 1]] = 0.5
    heights = []
    for s in range(n_steps):
        heights.append(0.3 + 0.4 * rng.random())
    for s in range(n_steps):
        h[all_edges[s]:all_edges[s + 1]] = heights[s]
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 200
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at parent resolution
    alloc1 = min(180, (deadline - time.time()) * 0.3)
    if alloc1 > 20:
        h1, c1 = optimize_at_n(best_h, n_init, t0 + alloc1, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Try higher resolution (n=400)
    n_high = 400
    if time.time() < deadline - 120:
        h_high = interpolate_h(best_h, len(best_h), n_high)
        alloc2 = min(200, (deadline - time.time()) * 0.3)
        if alloc2 > 30:
            h2, c2 = optimize_at_n(h_high, n_high, time.time() + alloc2, rng)
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 3: Try n=600
    n_higher = 600
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        h_higher = interpolate_h(best_h, n_cur, n_higher)
        alloc3 = min(150, (deadline - time.time()) * 0.25)
        if alloc3 > 30:
            h3, c3 = optimize_at_n(h_higher, n_higher, time.time() + alloc3, rng)
            if c3 < best_c:
                best_h, best_c = h3, c3

    # Phase 4: Diverse restarts at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 60:
        n_restarts = max(4, int((deadline - time.time() - 20) / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            if r < 3:
                hp = diverse_start(best_h, n_best, rng, scale=0.03 + 0.04 * r)
            else:
                hp = step_init(n_best, rng, n_steps=4 + r)
                if hp.sum() * (2.0 / n_best) < 0.1:
                    hp = best_h + rng.normal(0, 0.08, n_best)
                    hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 10) * 0.3)
            if alloc < 10:
                break
            hr, cr = optimize_at_n(hp, n_best, time.time() + alloc, rng)
            if cr < best_c:
                best_h, best_c = hr, cr

    # Phase 5: Interpolate best to higher res and try once more
    if time.time() < deadline - 50:
        n_final = max(len(best_h) + 100, 500)
        h_fin = interpolate_h(best_h, len(best_h), n_final)
        alloc5 = min(60, (deadline - time.time() - 10) * 0.5)
        if alloc5 > 15:
            hf, cf = optimize_at_n(h_fin, n_final, time.time() + alloc5, rng)
            if cf < best_c:
                best_h, best_c = hf, cf

    # Phase 6: Final SLSQP polish + pair search
    n_fin = len(best_h)
    if time.time() < deadline - 8:
        ht = slsqp_polish(best_h, n_fin, 200000, maxiter=1000)
        c = compute_c5(ht, n_fin)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
        best_h, best_c = fast_pair_search(best_h, n_fin, deadline - 1,
                                          [0.005, 0.002, 0.001, 0.0005], radius=25)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
