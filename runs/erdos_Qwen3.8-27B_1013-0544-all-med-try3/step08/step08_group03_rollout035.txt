import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    """Log-sum-exp for smooth max."""
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_i in range(2 * n - 1):
        if w[m_i] < 1e-13:
            continue
        k = m_i - (n - 1)
        wm = w[m_i] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    top_idx = np.argsort(c)[::-1][:K]
    w = np.zeros(2*n-1)

    dx = 2.0 / n

        if w[m_i] < 1e-15:
            continue
        k = m_i - (n - 1)
        wm = w[m_i] * dx
        if k >= 0:
            nv = n - k

        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad, c.max()

def obj_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_value(c, alpha)
    val += mu * (x.sum() - n / 2.0)**2
    return float(val)

def grad_smooth_obj(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_topk(x, n, K):
    c = compute_c5_vec(x, n)
    top_idx = np.argsort(c)[::-1][:K]
    val = c[top_idx].sum() / K + mu_penalty(x, n)
    return float(val)

def mu_penalty(x, n, mu=1000.0):
    return mu * (x.sum() - n/2.0)**2

def lbfgs_block(h, n, deadline, max_iters=200):
    """L-BFGS-B with increasing alpha."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]

def nesterov_descent(h, n, deadline, alpha=5000, lr=0.005, momentum=0.9):
    """Nesterov-accelerated projected gradient descent."""
    h = project_feasible(h, n)
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    stuck = 0
    for it in range(50000):
        if time.time() > deadline:
            break
        # Nesterov lookahead
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)
        g = smooth_max_grad(y, n, alpha)
        # Update
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
        h = h_new
        if stuck > 200:
            # Reset momentum, reduce lr
            v = np.zeros(n)
            lr *= 0.5
            stuck = 0
            h = best_h.copy()
        if lr < 1e-10:
            break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.exp(sh).sum()) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nm1 = n - 1
    for m_i in range(2 * n - 1):
        if w[m_i] < 1e-13:
            continue
        k = m_i - nm1
        wm = w[m_i] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_smooth_obj(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [(100, 5.0), (500, 10.0), (2000, 10.0), (5000, 20.0),
              (20000, 50.0), (100000, 100.0), (500000, 200.0)]
    for alpha, mu in alphas:
        if time.time() > deadline:
            break
        t_alloc = min(40, deadline - time.time() - 3)
        if t_alloc < 10:
            break
        try:
            res = minimize(obj_smooth, h, jac=grad_smooth_obj,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 2000,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def nesterov_descent(h, n, deadline):
    h = project_feasible(h, n)
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 5000.0
    lr = 0.004
    momentum = 0.85
    stuck = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)
        g = smooth_max_grad(y, n, alpha)
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
        h = h_new
        if stuck > 300:
            v = np.zeros(n)
            lr *= 0.5
            stuck = 0
            h = best_h.copy()
        if lr < 1e-10:
            break
    return best_h, best_c

def coord_minimax_1d(h, n, j, c_vec):
    """Solve 1D minimax for coordinate j. Returns optimal x for h[j]."""
    # Compute coefficients a_k = dc_k/dh[j]
    a = np.zeros(2 * n - 1)
    nm1 = n - 1
    for m_i in range(2 * n - 1):
        k = m_i - nm1
        if k >= 0:
            # lag k >= 0
            coeff = 0.0
            if k <= j <= n - 1:
                coeff += 1.0
                if j - k >= 0 and j - k <= n - 1 - k:
                    coeff -= h[j - k]
            if j + k <= n - 1:
                coeff -= h[j + k]
            a[m_i] = coeff
        else:
            kp = -k
            coeff = 0.0
            if 0 <= j <= n - 1 - kp:
                coeff += 1.0
                coeff -= h[j + kp]
            if kp <= j <= n - 1:
                coeff -= h[j - kp]
            a[m_i] = coeff
    # base_k = c_k - a_k * h[j]
    x_old = h[j]
    b = c_vec - a * x_old
    # Bisection on t to find min_x max_k (b_k + a_k * x)
    # Lower bound
    lb = 0.0
    for k_i in range(len(b)):
        if a[k_i] >= 0:
            lb = max(lb, b[k_i])
        else:
            lb = max(lb, b[k_i] + a[k_i])
    # Upper bound
    f0 = b.max()
    f1 = (b + a).max()
    ub = min(f0, f1)
    # Bisection
    best_x = 0.0
    for _ in range(60):
        mid = (lb + ub) / 2.0
        # Check feasibility: exists x in [0,1] with b_k + a_k*x <= mid for all k
        lo_x = 0.0
        hi_x = 1.0
        feasible = True
        for k_i in range(len(b)):
            if a[k_i] > 1e-15:
                bound = (mid - b[k_i]) / a[k_i]
                if bound < hi_x:
                    hi_x = bound
            elif a[k_i] < -1e-15:
                bound = (mid - b[k_i]) / a[k_i]
                if bound > lo_x:
                    lo_x = bound
            else:
                if b[k_i] > mid + 1e-15:
                    feasible = False
                    break
        if lo_x <= hi_x and feasible:
            ub = mid
            best_x = (lo_x + hi_x) / 2.0
        else:
            lb = mid
    # Verify
    c_test = (b + a * best_x).max()
    x_at_0 = b.max()
    x_at_1 = (b + a).max()
    if x_at_0 <= c_test:
        best_x = 0.0
    if x_at_1 <= c_test:
        best_x = 1.0
    return best_x

def coord_descent_minimax(h, n, deadline, max_sweeps=2000):
    """Cyclic coordinate descent with 1D minimax per coordinate."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    sweeps_no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        sweep_improved = False
        for j in range(n):
            if time.time() > deadline:
                return h, best_c
            x_opt = coord_minimax_1d(h, n, j, c_vec)
            if abs(x_opt - h[j]) < 1e-10:
                continue
            # Apply change
            delta = x_opt - h[j]
            h[j] = x_opt
            # Fix constraint by adjusting another coordinate
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                # Find the coordinate with most room
                if diff > 0:
                    room = 1.0 - h
                    j2 = np.argmax(room)
                    adj = min(diff, room[j2])
                    h[j2] += adj
                else:
                    j2 = np.argmin(h)
                    adj = min(-diff, h[j2])
                    h[j2] -= adj
            # Recompute and check
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                sweep_improved = True
                c_vec = compute_c5_vec(h, n)
            else:
                # Revert
                h[j] -= delta
                # Redo constraint fix (reverse)
                diff2 = target - h.sum()
                if abs(diff2) > 1e-12 and diff2 < 0:
                    # We need to add back
                    room = 1.0 - h
                    j3 = np.argmax(room)
                    h[j3] += min(abs(diff2), room[j3])
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=70):

                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:

                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:

def block_swap_search(h, n, deadline, block_size=3):
    """Try moving a small block of mass to another position."""
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n - block_size, 2):
                if time.time() > deadline:
                    return h, best_c
                block = h[i:i+block_size]
                total_move = d
                if block.sum() < total_move:
                    continue
                for j in range(0, n - block_size, 2):
                    if abs(i - j) < 3:
                        continue
                    if time.time() > deadline:
                        return h, best_c
                    # Try moving mass from block at i to block at j
                    src_room = block.copy()
                    dst_room = 1.0 - h[j:j+block_size]
                    # Move d total from src to dst
                    moves = []
                    remaining = total_move
                    for b in range(block_size):
                        mv = min(remaining, src_room[b], dst_room[b])
                        if mv > 1e-12:
                            moves.append((b, mv))
                            remaining -= mv
                        if remaining <= 1e-12:
                            break
                    if remaining > 1e-12:
                        continue
                    old_vals = h.copy()
                    for (b, mv) in moves:
                        h[i+b] -= mv
                        h[j+b] += mv
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h = old_vals
    return h, best_c
