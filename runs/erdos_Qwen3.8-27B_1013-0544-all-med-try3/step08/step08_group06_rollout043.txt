```python
def fast_pair_search(h, n, d, max_passes=50):
    """Try all pairs with step size d, keep only improving moves."""
    best_c = compute_c5(h, n)
    dx = 2.0/n
    c = np.correlate(h, 1-h, mode='full') * dx
    # Precompute G: G[p, m] = influence of h[p] on c[m]
    # G[p, m] = (1-h[p+m-(n-1)]) * [valid] - h[p-m+(n-1)] * [valid]
    # where c[m] corresponds to shift (m-(n-1))*dx
    
    improved = True
    passes = 0
    while improved and passes < max_passes:
        improved = False
        passes += 1
        c = np.correlate(h, 1-h, mode='full') * dx
        for i in range(n):
            for j in range(n):
                if i == j: continue
                h_new_i = h[i] + d
                h_new_j = h[j] - d
                if h_new_i < 0 or h_new_i > 1 or h_new_j < 0 or h_new_j > 1:
                    continue
                h[i] = h_new_i
                h[j] = h_new_j
                c_new = np.correlate(h, 1-h, mode='full') * dx
                c_new_max = c_new.max()
                if c_new_max < best_c:
                    best_c = c_new_max
                    improved = True
                else:
                    h[i] = h[i] - d
                    h[j] = h[j] + d
    return h, best_c
```

```python
    s = alpha * (x - m)

    return m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, alpha):
    m = x.max()
    s = alpha * (x - m)
    ms = s.max()
    return m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha

def compute_smooth_max_grad(x, alpha):

    m = x.max()

    # smooth_max = m + (ms + log(sum(exp(s - ms)))) / alpha
    val = m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha
    return val, w  # w = softmax(s) = exp(s-ms)/sum(exp(s-ms))

    # g[p] = dx * sum_m J_unscaled[p,m] * w[m] + 2*lam*sv
    # where J_unscaled[p,m] = [(1-h[p+a])*1{valid} - h[p-a])*1{valid}]

    A = np.arange(-(n-1), n)[None, :]

    M[mask2] -= h[PA2[mask2]]

    g += 2.0 * lam * sv
    
    return val, g

        v, _ = obj_and_grad(x, n, alpha, lam)
        return v
    def grad(x):

        return g
```
