import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    sm = np.max(shifted)
    exps = np.exp(shifted - sm)
    w = exps / exps.sum()
    val = m + (sm + np.log(exps.sum())) / alpha
    
    # Gradient
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for idx in range(len(lags)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        l = lags[idx]
        # term1[j] = (1-x[j+l]) if 0<=j+l<n else 0
        # term2[j] = x[j-l] if 0<=j-l<n else 0
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if l >= 0:
            t1[:n-l] = 1 - x[l:]
            t2[l:] = x[:n-l]
        else:
            il = -l
            t1[il:] = 1 - x[:n-il]
            t2[:n-il] = x[il:]
        grad += wi * (t1 - t2)
    grad *= dx
    return val, grad

def slsqp_with_grad(h, n, alpha, maxiter, deadline):
    def objective(x):
        val, _ = smooth_max_obj_and_grad(x, n, alpha)
        return val
    def gradient(x):
        _, g = smooth_max_obj_and_grad(x, n, alpha)
        return g
    def constraint_fun(x):
        return x.sum() - n / 2.0
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        res = minimize(objective, h, jac=gradient, method='SLSQP', 
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def interpolate_to_resolution(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

    """Try perturbation + SLSQP polish."""
    best_c = compute_c5(h, n)
    best_h = h.copy()

        if c < best_c - 1e-10:

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(x, n, alpha):
    """Compute smooth max approximation and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    sm = np.max(shifted)
    exps = np.exp(shifted - sm)
    w = exps / exps.sum()
    val = m + (sm + np.log(exps.sum())) / alpha
    
    # Gradient computation
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        l = idx - (n - 1)  # lag
        # term1: (1-x[j+l]) for valid j where 0<=j+l<n
        # term2: x[j-l] for valid j where 0<=j-l<n
        if l >= 0:
            # term1: j from 0 to n-1-l, value = 1-x[l+n-1-j]... 
            # Actually: term1[j] = 1-x[j+l] for j in [0, n-1-l]
            # term2[j] = x[j-l] for j in [l, n-1]
            grad[:n-l] += wi * (1 - x[l:])
            grad[l:] += wi * (-x[:n-l])
        else:
            # l < 0, let il = -l
            il = -l
            # term1[j] = 1-x[j+l] = 1-x[j-il] for j in [il, n-1]

            grad[il:] += wi * (1 - x[:n-il])

            # j+il ranges from il to n-1
            grad[:n-il] += wi * (-x[il:])
    grad *= dx
    
    return val, grad
