def project_feasible(h,n):
    h=np.asarray(h,dtype=float)
    target=n/2.0
    if n==0: return h
    # if sum very close and in bounds, adjust with free vars
    if h.min() >= -1e-12 and h.max() <=1+1e-12:
       s=h.sum()

          return np.clip(h,0,1)
    # Use bisection. If h sum target but out of bounds, waterfill from adjusted?
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    # if sum at lo > target? at lo= min-1 => all >=1 clipped 1 sum n; yes > target. hi sum 0.
    for _ in range(80):

       if s>target: lo=mid
       else: hi=mid
    lam=(lo+hi)/2

   c,C=eval_vec(h,n)
   best_c=C; best_h=h.copy()
   step=0.05
   for it in range(max_iter):
      if time.time()>=deadline: break
      c,C=eval_vec(h,n)
      if C < best_c: best_c=C; best_h=h.copy()
      z=beta*(c-C)
      zm=float(np.max(z))
      e=np.exp(z-zm)
      s=float(e.sum())
      if s<=0 or not finite: break
      w=e/s
      g=(2.0/n)*grad_from_weights(w,h,n)
      g=g-g.mean()
      gn=float(np.linalg.norm(g))
      if gn<1e-12: break
      base=step/gn
      accepted=False
      for fac in (1.0,0.5,0.25,0.1,0.05,0.025,0.01):
         if time.time()>=deadline: break
         hh=h - base*fac*g
         hh=project_sum_fast(hh,n)
         cc=eval_vec(hh,n)[1]
         if cc < C - 1e-13:
            h=hh; C=cc; accepted=True
            if C < best_c: best_c=C; best_h=h.copy()
            step=min(2.0*base*fac, 0.5)
            break
      if not accepted:
         step *= 0.65
         if step < 1e-8: break
   return best_h,best_c

def project_sum_fast(h,n):
   h=np.asarray(h,float)
   if h.size != n: return project_feasible(h,n)
   if h.min() < -1e-12 or h.max() > 1+1e-12:
       return project_feasible(h,n)
   diff=n/2.0 - h.sum()
   if abs(diff) < 1e-13:
       return np.clip(h,0,1)
   p=h+diff/n
   if p.min() >= -1e-12 and p.max() <= 1+1e-12:
       p=np.clip(p,0,1)
       if abs(p.sum()-n/2.0) < 1e-10:
          return p
   return project_feasible(h,n)

def transfer_polish(h,n,deadline,max_evals=4000,topm=20,beta=200.0):
   h=np.asarray(h,float).copy()
   c,C=eval_vec(h,n)
   best_c=C; best_h=h.copy()
   evals=0
   base_deltas=(0.002,0.001,0.0005,0.0002,0.0001,0.00005)
   while time.time()<deadline and evals<max_evals:
      c,C=eval_vec(h,n)
      if C < best_c: best_c=C; best_h=h.copy()
      z=beta*(c-C)
      zm=float(np.max(z)); e=np.exp(z-zm); s=float(e.sum())
      if s<=0 or not np.isfinite(s): break
      w=e/s
      g=(2.0/n)*grad_from_weights(w,h,n)
      g=g-g.mean()
      if n <= 2*topm:
         high=np.argsort(g)[::-1][:max(1,n)]
         low=np.argsort(g)[:max(1,n)]
      else:
         high=np.argsort(g)[-topm:][::-1]
         low=np.argsort(g)[:topm]
      accepts=0
      for ii in high:
         if accepts >= 25: break
         for jj in low:
            if accepts >=25 or evals>=max_evals or time.time()>=deadline: break
            if ii==jj: continue
            hi=h[ii]; hj=h[jj]
            maxd=min(hi,1.0-hj)
            if maxd <= 1e-8: continue
            cands=[]
            for d in base_deltas:
               if d <= maxd: cands.append(d)
            for rel in (0.5,0.25,0.1):
               d=maxd*rel
               if d > 1e-8: cands.append(d)
            # unique sorted descending
            uniq=[]; 
            seen=set()
            for d in sorted(set(cands), reverse=True):
                key=round(d,12)
                if key not in seen:
                   seen.add(key); uniq.append(d)
            for d in uniq:
               if evals>=max_evals or time.time()>=deadline: break
               hi=h[ii]; hj=h[jj]
               if hi< d or hj + d >1: continue
               evals += 1
               h[ii]=hi-d
               h[jj]=hj+d
               cc=eval_vec(h,n)[1]
               if cc < C - 1e-12:
                  C=cc; accepts +=1
                  if C<best_c: best_c=C; best_h=h.copy()
               else:
                  h[ii]=hi; h[jj]=hj
               # if improved, break candidates for this pair
               break
         if accepts >=25: break
      if accepts ==0:
         break
   return best_h,best_c

def clean_array(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return h
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    return np.clip(h, 0.0, 1.0)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if n <= 0:
        return h

    target = n / 2.0
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)

    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            hp = np.zeros(n, dtype=float)
            hp[:h.size] = h
            h = hp

    if h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12:
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            return np.clip(h, 0.0, 1.0)
        p = h + diff / float(n)
        if p.min() >= -1e-12 and p.max() <= 1.0 + 1e-12:
            p = np.clip(p, 0.0, 1.0)
            if abs(p.sum() - target) < 1e-10:
                return p

    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    p = np.clip(h - lam, 0.0, 1.0)
    diff = target - p.sum()

    if abs(diff) > 1e-10:
        free = (p > 1e-12) & (p < 1.0 - 1e-12)
        if free.any():
            p[free] += diff / float(int(free.sum()))
            p = np.clip(p, 0.0, 1.0)
        else:
            if diff > 0.0:
                room = 1.0 - p
                sroom = float(room.sum())
                if sroom > 1e-14:
                    p += (diff / sroom) * room
            else:
                sp = float(p.sum())
                if sp > 1e-14:
                    p += (diff / sp) * p

    return np.clip(p, 0.0, 1.0)


def project_sum_fast(h, n):

    if h.size != n or n <= 0:
        return project_feasible(h, n)

    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:

        return np.clip(h, 0.0, 1.0)

def fix_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    if n <= 0:
        return h

    target = n / 2.0
    for _ in range(80):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break

        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, max(0.0, 1.0 - h[idx]))
        else:
            idx = int(np.argmin(h))
            add = -min(-diff, max(0.0, h[idx]))

        if abs(add) < 1e-16:
            break
        h[idx] += add

    return np.clip(h, 0.0, 1.0)

def eval_vec(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if c.size == 0:
        return c, 0.0
    return c, float(np.max(c))


def eval_c5(h, n):
    return eval_vec(h, n)[1]


def grad_from_weights(w, h, n):
    w = np.asarray(w, dtype=float)
    h = np.asarray(h, dtype=float)
    off = n - 1
    c1 = np.convolve(w, 1.0 - h, mode='full')
    c2 = np.convolve(w[::-1], h, mode='full')
    return c1[off:off + n] - c2[off:off + n]


def soft_obj_grad(h, n, alpha):
    c, C = eval_vec(h, n)
    dx = 2.0 / float(n)

    if alpha <= 0.0:
        active = c >= (C - 1e-12)
        if not active.any():
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = active.astype(float)
        s = float(w.sum())
        if s <= 0.0:
            w = np.zeros_like(c, dtype=float)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= s
        return C, dx * grad_from_weights(w, h, n)

    z = alpha * (c - C)
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    s = float(np.sum(e))

    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return C, dx * grad_from_weights(w, h, n)

    w = e / s
    F = C + (zmax + math.log(s)) / alpha
    g = dx * grad_from_weights(w, h, n)
    return float(F), g

def expand_symmetric(x, n):
    x = np.asarray(x, dtype=float)
    h = np.empty(n, dtype=float)
    M = n // 2

    if n % 2 == 0:
        h[:M] = x[:M]

    else:
        h[:M] = x[:M]
        if x.size > M:
            h[M] = x[M]
        else:
            h[M] = 0.5
        if M > 0:
            h[M + 1:] = x[:M][::-1]

    return np.clip(h, 0.0, 1.0)


def symmetric_variables(h, n):

    if n % 2 == 0:
        return h[:M].copy()
    if M == 0:
        return np.array([float(h[0])], dtype=float)

def _slsqp_fun(x):
    st = OPT_STATE
    n = int(st['n'])
    alpha = float(st['alpha'])

    if st['mode'] == 'full':
        F, g = soft_obj_grad(x, n, alpha)
    else:

        F, g = soft_obj_grad(h, n, alpha)

    st['cache_x'] = np.asarray(x, dtype=float).copy()
    st['cache_F'] = float(F)
    st['cache_g'] = np.asarray(g, dtype=float).copy()
    return float(F)


def _slsqp_jac(x):
    st = OPT_STATE
    n = int(st['n'])
    alpha = float(st['alpha'])

    cx = st.get('cache_x')
    if cx is not None:
        try:
            if np.array_equal(np.asarray(x, dtype=float), cx):
                return st['cache_g']
        except Exception:
            pass

    if st['mode'] == 'full':
        _, g = soft_obj_grad(x, n, alpha)
        return np.asarray(g, dtype=float)

    h = expand_symmetric(x, n)
    _, gfull = soft_obj_grad(h, n, alpha)
    M = n // 2

    if n % 2 == 0:
        gy = gfull[:M] + gfull[-M:][::-1]
    else:
        gy = np.empty(M + 1, dtype=float)
        if M > 0:
            gy[:M] = gfull[:M] + gfull[M + 1:][::-1]
        gy[M] = gfull[M]

    return gy


def _cons_fun(x):
    st = OPT_STATE
    if st['mode'] == 'full':
        return float(np.sum(x) - st['target'])

    n = int(st['n'])
    M = n // 2
    if n % 2 == 0:
        return float(np.sum(x) - M / 2.0)
    return float(2.0 * np.sum(x[:M]) + float(x[M]) - (M + 0.5))


def _cons_jac(x):
    st = OPT_STATE
    if st['mode'] == 'full':
        return st['ones'].copy()

    n = int(st['n'])
    M = n // 2
    if n % 2 == 0:
        return np.full(M, 1.0, dtype=float)

    if M == 0:
        return np.array([1.0], dtype=float)
    j = np.full(M, 2.0, dtype=float)
    return np.append(j, 1.0)


def slsqp_smooth(x, n, alpha, maxiter, mode='full'):
    start = np.asarray(x, dtype=float).copy()
    st = OPT_STATE
    st['n'] = int(n)
    st['alpha'] = float(alpha)
    st['mode'] = mode
    st['target'] = n / 2.0
    st['ones'] = np.ones(int(n), dtype=float)
    st['cache_x'] = None
    st['cache_F'] = None
    st['cache_g'] = None

    try:
        if mode == 'full':
            start = project_feasible(start, n)
            nv = int(n)
        else:
            nv = int(n) // 2 + (1 if int(n) % 2 else 0)
            start = start[:nv].copy()

        var_bounds = [(0.0, 1.0)] * nv
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        res = minimize(
            _slsqp_fun,
            start,
            jac=_slsqp_jac,
            bounds=var_bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': int(maxiter), 'ftol': 1e-11},
        )

        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            if mode == 'full':
                return project_feasible(res.x, n)
            return np.clip(np.asarray(res.x, dtype=float), 0.0, 1.0)
    except Exception:
        pass

    return start


def resample(h, n, nn, mode='linear'):
    h = np.asarray(h, dtype=float)
    if nn < 1:
        return np.zeros(0, dtype=float)
    if nn == n:
        return h.copy()

    if mode == 'repeat' and n > 0 and nn % n == 0:
        return np.repeat(h, nn // n)

    if n == 1:
        return np.full(nn, float(h[0]))

    x_old = np.linspace(0.0, 2.0, int(n), dtype=float)
    x_new = np.linspace(0.0, 2.0, int(nn), dtype=float)

    if mode == 'nearest':
        idx = np.searchsorted(x_old, x_new)
        idx = np.clip(idx, 0, n - 1)
        return h[idx].astype(float, copy=True)

    return np.interp(x_new, x_old, h)

def exact_polish(h, n, deadline, beta=100.0, max_iter=20000):
    h = np.asarray(h, dtype=float).copy()
    c, C = eval_vec(h, n)
    best_c = float(C)
    best_h = h.copy()
    step = 0.05

    for _ in range(int(max_iter)):
        if time.time() >= deadline:
            break

        c, C = eval_vec(h, n)
        if C < best_c:
            best_c = float(C)
            best_h = h.copy()

        z = beta * (c - C)
        zm = float(np.max(z))
        e = np.exp(z - zm)
        s = float(np.sum(e))
        if s <= 0.0 or not np.isfinite(s):
            break

        w = e / s
        g = (2.0 / float(n)) * grad_from_weights(w, h, n)
        g = g - np.mean(g)
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break

        base = step / gn
        accepted = False

        for fac in (1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01):
            if time.time() >= deadline:
                break
            hh = h - base * fac * g
            hh = project_sum_fast(hh, n)
            cc = eval_c5(hh, n)

            if cc < C - 1e-13:
                h = hh
                C = cc
                accepted = True
                if C < best_c:
                    best_c = float(C)
                    best_h = h.copy()
                step = min(2.0 * base * fac, 0.5)
                break

        if not accepted:
            step *= 0.65
            if step < 1e-8:
                break

    return best_h, float(best_c)


def transfer_polish(h, n, deadline, max_evals=4000, topm=20, beta=200.0):
    h = np.asarray(h, dtype=float).copy()
    c, C = eval_vec(h, n)
    best_c = float(C)
    best_h = h.copy()
    evals = 0

    base_deltas = (0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005)
    max_accepts_per_sweep = 25

    while time.time() < deadline and evals < max_evals:
        c, C = eval_vec(h, n)
        if C < best_c:
            best_c = float(C)
            best_h = h.copy()

        z = beta * (c - C)
        zm = float(np.max(z))
        e = np.exp(z - zm)
        s = float(np.sum(e))
        if s <= 0.0 or not np.isfinite(s):
            break

        w = e / s
        g = (2.0 / float(n)) * grad_from_weights(w, h, n)
        g = g - np.mean(g)

        if n <= max(2, 2 * int(topm)):
            high = np.argsort(g)[::-1][:max(1, n)]
            low = np.argsort(g)[:max(1, n)]
        else:
            high = np.argsort(g)[-int(topm):][::-1]
            low = np.argsort(g)[:int(topm)]

        accepts = 0
        for ii in high:
            if accepts >= max_accepts_per_sweep:
                break
            for jj in low:
                if accepts >= max_accepts_per_sweep:
                    break
                if evals >= max_evals or time.time() >= deadline:
                    break
                if ii == jj:
                    continue

                hi = float(h[ii])
                hj = float(h[jj])
                maxd = min(hi, 1.0 - hj)
                if maxd <= 1e-8:
                    continue

                cands = []
                for d in base_deltas:
                    if d <= maxd:
                        cands.append(float(d))
                for rel in (0.5, 0.25, 0.1):
                    d = maxd * rel
                    if d > 1e-8:
                        cands.append(float(d))

                seen = set()
                uniq = []
                for d in sorted(set(cands), reverse=True):
                    key = round(float(d), 12)
                    if key not in seen:
                        seen.add(key)
                        uniq.append(float(d))

                for d in uniq:
                    if evals >= max_evals or time.time() >= deadline:
                        break

                    hi = float(h[ii])
                    hj = float(h[jj])
                    if hi < d or hj + d > 1.0 + 1e-14:
                        continue

                    evals += 1
                    h[ii] = hi - d
                    h[jj] = hj + d
                    cc = eval_c5(h, n)

                    if cc < C - 1e-12:
                        C = cc
                        accepts += 1
                        if C < best_c:
                            best_c = float(C)
                            best_h = h.copy()
                    else:
                        h[ii] = hi
                        h[jj] = hj

                    break

        if accepts == 0:
            break

    return best_h, float(best_c)

    for alpha, mx in alpha_full:
        if time.time() >= full_dl:
            break
        if best_n > 400:
            mx = int(0.6 * mx + 20)
        h2 = slsqp_smooth(best_h, best_n, alpha, mx, 'full')
        h2 = make_good(h2, best_n)
        c2 = eval_c5(h2, best_n)
        if c2 < best_c - 1e-14:
            best_c = float(c2)
            best_h = h2.copy()
            no_improve = 0
        else:
            no_improve += 1
            if alpha > 10000 and no_improve >= 2:
                break

    pol_dl = t0 + 0.62 * total
    if time.time() < pol_dl:
        best_h, best_c = exact_polish(best_h, best_n, pol_dl, beta=100.0, max_iter=10000)

    tr_dl = t0 + 0.66 * total
    if time.time() < tr_dl:
        max_evals = 3000 if best_n <= 300 else 1200
        best_h, best_c = transfer_polish(best_h, best_n, tr_dl, max_evals=max_evals)

    sym_dl = t0 + 0.78 * total
    if time.time() < sym_dl:
        hsym = 0.5 * (best_h + best_h[::-1])
        hsym = make_good_sym(hsym, best_n)
        csym = eval_c5(hsym, best_n)

        if csym <= best_c + 0.003:
            y = symmetric_variables(hsym, best_n)
            alpha_sym = (
                (1000, 80),
                (5000, 100),
                (20000, 100),
            )
            no_improv_sym = 0
            for alpha, mx in alpha_sym:
                if time.time() >= sym_dl:
                    break
                if best_n > 400:
                    mx = int(0.6 * mx + 20)
                y2 = slsqp_smooth(y, best_n, alpha, mx, 'sym')
                h2 = expand_symmetric(y2, best_n)
                h2 = make_good_sym(h2, best_n)
                c2 = eval_c5(h2, best_n)

                if c2 < best_c - 1e-14:
                    best_c = float(c2)
                    best_h = h2.copy()
                    no_improv_sym = 0
                else:
                    no_improv_sym += 1
                    if alpha > 5000 and no_improv_sym >= 2:
                        break

    sym_pol_dl = t0 + 0.82 * total
    if time.time() < sym_pol_dl:
        best_h, best_c = exact_polish(best_h, best_n, sym_pol_dl, beta=100.0, max_iter=6000)

    high_dl = t0 + 0.92 * total
    if time.time() < high_dl - 60.0:
        for nh in (400, 500):
            if nh % 2 != 0:
                nh -= 1
            if nh <= best_n:
                continue
            if time.time() >= high_dl - 60.0:
                break

            starts = []
            lin = resample(best_h, best_n, nh, 'linear')
            starts.append(lin)
            if nh % best_n == 0:
                rep = resample(best_h, best_n, nh, 'repeat')
                if rep.size == nh:
                    starts.append(rep)

            hstart = None
            cstart = None
            for arr in starts:
                hn = make_good(arr, nh)
                cn = eval_c5(hn, nh)
                if hstart is None or cn < cstart:
                    hstart = hn
                    cstart = cn

            if hstart is None:
                continue

            if cstart < best_c + 0.002:
                if cstart < best_c:
                    best_c = float(cstart)
                    best_h = hstart.copy()
                    best_n = int(nh)

                high_alphas = (
                    (800, 50),
                    (5000, 70),
                    (20000, 70),
                )
                if nh > 400:
                    high_alphas = (
                        (800, 40),
                        (5000, 50),
                        (10000, 50),
                    )

                hh = hstart.copy()
                ch = float(cstart)

                for alpha, mx in high_alphas:
                    if time.time() >= high_dl - 25.0:
                        break
                    h2 = slsqp_smooth(hh, nh, alpha, mx, 'full')
                    h2 = make_good(h2, nh)
                    c2 = eval_c5(h2, nh)
                    if c2 < ch - 1e-14:
                        ch = float(c2)
                        hh = h2.copy()
                        if ch < best_c:
                            best_c = ch
                            best_h = hh.copy()
                            best_n = int(nh)
                    elif c2 < best_c - 1e-14:
                        best_c = float(c2)
                        best_h = h2.copy()
                        best_n = int(nh)

                if time.time() < high_dl - 5.0:
                    hh, ch = exact_polish(hh, nh, high_dl - 2.0, beta=100.0, max_iter=4000)
                    if ch < best_c:
                        best_c = float(ch)
                        best_h = hh.copy()
                        best_n = int(nh)

    if time.time() < final_dl - 25.0:
        best_h, best_c = exact_polish(best_h, best_n, final_dl - 12.0, beta=120.0, max_iter=8000)

    if time.time() < final_dl - 20.0:
        max_evals = 1800 if best_n <= 350 else 900
        best_h, best_c = transfer_polish(best_h, best_n, final_dl - 8.0, max_evals=max_evals)

    if time.time() < final_dl - 90.0:
        h2 = slsqp_smooth(best_h, best_n, 100000, 40, 'full')
        h2 = make_good(h2, best_n)
        c2 = eval_c5(h2, best_n)
        if c2 < best_c - 1e-14:
            best_c = float(c2)
            best_h = h2.copy()

    best_h = make_good(best_h, best_n)
    _, final_c = eval_vec(best_h, best_n)
    return best_h, float(final_c), int(best_n)
