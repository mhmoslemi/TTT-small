def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def greedy_binary_search(h, n, deadline, max_sweeps=2000):
    dx = 2.0 / n
    target = n // 2
    # Ensure exactly target ones
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    list_ones = list(ones)
    list_zeros = list(zeros)
    
    h_flat = h  # work with flat array
    g = np.correlate(h_flat, 1 - h_flat, mode='full')
    best_c = g.max()
    g_scaled = g * dx
    
    improved = True
    sweeps = 0
    while improved and sweeps < max_sweeps:
        if time.time() > deadline:
            break
        improved = False
        sweeps += 1
        best_c_new = best_c
        best_pair = None
        
        for zi in list_zeros:
            for oj in list_ones:
                # Try swap
                h_flat[zi] = 1.0
                h_flat[oj] = 0.0
                g_new = np.correlate(h_flat, 1 - h_flat, mode='full')
                c_new = g_new.max()
                if c_new < best_c_new - 1e-10:
                    best_c_new = c_new
                    best_pair = (zi, oj)
                else:
                    h_flat[zi] = 0.0
                    h_flat[oj] = 1.0
        
        if best_pair is not None:
            zi, oj = best_pair
            h_flat[zi] = 1.0
            h_flat[oj] = 0.0
            ones.discard(oj)
            ones.add(zi)
            zeros.discard(zi)
            zeros.add(oj)
            list_ones = list(ones)
            list_zeros = list(zeros)
            g = np.correlate(h_flat, 1 - h_flat, mode='full')
            best_c = g.max()
            improved = True
    
    return h_flat, float(best_c * dx)
