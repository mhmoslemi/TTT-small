def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, target_sum):
    n = len(h)
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.clip(h * (target_sum / tot), 0.0, 1.0)
    return h

def smooth_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        val = (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        val = np.log(np.sum(np.exp(s))) / alpha
    return m + val

def smooth_max_grad(c, h, n, alpha):
    """Analytical gradient of smooth max of correlations."""
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        e = np.exp(s - mx)
    else:
        e = np.exp(s)
    w = e / e.sum()  # softmax weights
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-10:
            continue
        shift = k_idx - (n - 1)
        wk = w[k_idx] * dx
        # dc_k/dh_j = (1-h_j) if j-shift in [0,n-1], plus -h_j if j+shift in [0,n-1]
        # j - shift in [0, n-1] → j in [shift, shift + n - 1] ∩ [0, n-1]

        hi2 = min(n, n - shift)
        if lo2 < hi2:

        return smooth_max_grad(c, x, n, alpha)

        return x.sum() - target_sum

        return np.ones(n)

    step_size = 0.1

            if lo1 < hi1:

            lo2 = max(0, -shift)
            hi2 = min(n, n - shift)
            if lo2 < hi2:
                grad[lo2:hi2] -= best_h[lo2:hi2]

        else:
            step_size *= 0.7

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def get_corr_scaled(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, target_sum):
    n = len(h)
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.clip(h * (target_sum / tot), 0.0, 1.0)
    return h


def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)


def smooth_max_grad(c, h, n, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        e = np.exp(s - mx)
    else:
        e = np.exp(s)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-10:
            continue
        shift = k_idx - (n - 1)
        wk = w[k_idx] * dx
        lo1 = max(0, shift)
        hi1 = min(n, shift + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1:hi1])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2:hi2]
    return grad

    c = np.correlate(x, 1 - x, mode='full')
    return smooth_max_val(c * (2.0 / n), alpha)


def slsqp_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return smooth_max_grad(c, x, n, alpha)


def slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)


def slsqp_eq_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, jac=lambda x: slsqp_grad(x, n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return h.copy()


def compute_f_vectors(h, n):
    """Compute f_p[k_idx] for all positions p.
    f_p[k] = (1-h[p-shift])*1[p-shift valid] - h[p+shift]*1[p+shift valid]
    where shift = k - (n-1), k in [0, 2n-2]."""
    f = np.zeros((n, 2 * n - 1))

                val += (1.0 - h[p - shift])
            if 0 <= p + shift < n:
                val -= h[p + shift]
            f[p, k_idx] = val
    return f

def subgradient_descent(h, n, deadline, n_steps=300):

        if time.time() > deadline:
            break
        c = get_corr_scaled(best_h, n)
        top_k = max(3, n // 15)
        top_idx = np.argsort(c)[-top_k:]

        grad = np.zeros(n)
        for k_idx in top_idx:
            shift = k_idx - (n - 1)
            lo1 = max(0, shift)
            hi1 = min(n, shift + n)
            if lo1 < hi1:
                grad[lo1:hi1] += (1.0 - best_h[lo1:hi1])
            lo2 = max(0, -shift)
            hi2 = min(n, n - shift)
            if lo2 < hi2:
                grad[lo2:hi2] -= best_h[lo2:hi2]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:

                best_c = c_new
                best_h = h_new

        # Ultra-fine pair search
        if time.time() < deadline - 3:
            best_h, best_c = fast_pair_search(best_h, n, deadline - 2,
                                              [0.0005, 0.0002, 0.0001])

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
