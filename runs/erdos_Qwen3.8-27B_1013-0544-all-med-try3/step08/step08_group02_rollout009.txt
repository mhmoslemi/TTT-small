    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (target / tot) if tot > 0 else h, 0, 1)
    return h

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    n_c = len(c)
    cp = c**p
    s = np.sum(cp) / n_c
    w = s**(1.0/p - 1.0) * c**(p-1) / n_c
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-20:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k

            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:

    return grad

def subgrad_step(h, n, eta, tol_expand):

    c_max = c_vec.max()
    tol = max(1e-9, tol_expand * c_max)
    active = np.where(c_vec >= c_max - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active:

        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    grad /= len(active)
    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)

    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_work = h.copy()

            break

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        d = rng.uniform(0.001, 0.05)

        if h_work[i] + d > 1.0 or h_work[j] - d < 0.0:
            # Try reverse
            if h_work[i] - d < 0.0 or h_work[j] + d > 1.0:
                continue
            h_work[i] -= d
            h_work[j] += d
        else:
            h_work[i] += d
            h_work[j] -= d

            best_c = c_new
            best_h = h_work.copy()
            c_work = c_new
        elif c_new < c_work or (sa_temp > 0 and rng.random() < np.exp(-(c_new - c_work) / max(t_current, 1e-10))):
            c_work = c_new
            # Accept (even if worse than best)
        else:

            if h_work[i] > h_work[j]:  # We added to i
                h_work[i] -= d
                h_work[j] += d
            else:
                h_work[i] -= d
                h_work[j] += d
        
        # Cool down
        if it % 100 == 0:
            t_current *= 0.95
    
    return best_h, best_c

def block_search(h, n, deadline, rng):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_work = h.copy()

        end = min(n, start + b)

        d_start = rng.randint(0, max(1, n - b))

        if np.max(h_work[end_start:end]) + d > 1.0 if end_start < n else True:

        if block_room.sum() < d:

        h_work[i] -= d
        h_work[j_start:j_end] += d / b * (block_room / block_room.sum() if block_room.sum() > 0 else 1.0/b)

            best_c = c_new
            best_h = h_work.copy()
        else:
            # Revert - need to recompute from best_h
            h_work = best_h.copy()
    
    return best_h, best_c

    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def pm_obj(x, n, p, mu):
    c = compute_c5_vec(x, n)
    val = float((np.sum(c**p) / len(c))**(1.0/p))
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def pm_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    n_c = len(c)
    cp = c**p
    s = np.sum(cp) / n_c
    w = s**(1.0/p - 1.0) * c**(p-1) / n_c
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-20:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
