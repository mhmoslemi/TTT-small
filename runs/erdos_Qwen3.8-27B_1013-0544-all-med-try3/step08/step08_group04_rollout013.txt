def resample_h_area(hold, n_new):
    hold = np.asarray(hold, float).ravel()
    n_new = int(n_new)
    if n_new <= 0: return np.zeros(0)
    if hold.size == 0: return project_box_sum(np.full(n_new,0.5),0.5*float(n_new))
    if hold.size == n_new: return project_box_sum(hold,0.5*float(n_new))
    if not np.all(np.isfinite(hold)): hold = np.where(np.isfinite(hold), hold, 0.5)
    no = hold.size
    cum = np.concatenate(([0.0], np.cumsum(hold) / float(no)))
    edges_old = np.linspace(0.0, 1.0, no + 1)
    edges_new = np.linspace(0.0, 1.0, n_new + 1)
    cum_new = np.interp(edges_new, edges_old, cum)
    vals = (cum_new[1:] - cum_new[:-1]) * float(n_new)

def get_ncache(n):

        return _NCACHE[n]

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).ravel().copy()
    target = float(target)
    if x.size == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.5)
    if target <= 0.0:
        return np.zeros_like(x)
    if target >= float(x.size):
        return np.ones_like(x)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        s = float(np.sum(h))
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    hold = np.asarray(hold, dtype=float).ravel()
    n_new = int(n_new)
    if n_new <= 0:
        return np.zeros(0, dtype=float)
    if hold.size == 0:

        return project_box_sum(hold, 0.5 * float(n_new))

def softmax_weights(c, alpha):
    c = np.asarray(c, dtype=float).ravel()
    L = int(c.size)
    if L == 0:
        return np.zeros(0, dtype=float)
    if not np.all(np.isfinite(c)):
        c = np.where(np.isfinite(c), c, -np.inf)
    if not np.any(np.isfinite(c)):
        return np.full(L, 1.0 / float(L), dtype=float)
    if alpha <= 1e-12:
        return np.full(L, 1.0 / float(L), dtype=float)

    m = float(np.max(c))
    z = float(alpha) * (c - m)
    zmax = float(np.max(z))
    z = z - zmax
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        w = np.full(L, 1.0 / float(L), dtype=float)
        idx = int(np.argmax(c))
        w[idx] = 1.0
        return w
    return e / s

    x = np.asarray(x, dtype=float).ravel()

    if n <= 0:
        return 0.0
    c = c_s_raw(x, n)
    m = float(np.max(c))
    dx = 2.0 / float(n)
    if alpha <= 1e-12:
        return float(dx * m)
    z = float(alpha) * (c - m)
    zmax = float(np.max(z))
    z = z - zmax
    s = float(np.sum(np.exp(z)))
    if not np.isfinite(s) or s <= 0.0:
        s = 1.0
    return float(dx * (m + (zmax + float(np.log(s))) / float(alpha)))

def _w_and_P_from_weights(weights, n, cache):
    L, dx, r_min, r_max, Lfft, offset = cache
    pref = np.concatenate(([0.0], np.cumsum(weights)))
    lin_pref = pref[r_max + 1] - pref[r_min]

    W = np.zeros(n, dtype=float)
    W[0] = 2.0 * weights[n - 1]
    if n > 1:
        W[1:] = weights[n:] + weights[n - 2::-1]

    Kpad = np.zeros(Lfft, dtype=float)
    Kpad[offset] = W[0]
    if n > 1:
        Kpad[offset + 1:offset + n] = W[1:]
        Kpad[offset - 1::-1] = W[1:]
    return lin_pref, W, Kpad

def smooth_grad(x, n, alpha):
    x = np.asarray(x, dtype=float).ravel()
    n = int(x.size)
    if n <= 0:
        return np.zeros_like(x)
    c = c_s_raw(x, n)
    w = softmax_weights(c, alpha)

    cache = get_ncache(n)
    L, dx, r_min, r_max, Lfft, offset = cache
    lin_pref, W, Kpad = _w_and_P_from_weights(w, n, cache)

    Hpad = np.zeros(Lfft, dtype=float)
    Hpad[:n] = x
    P = np.fft.irfft(np.fft.rfft(Hpad) * np.fft.rfft(Kpad), Lfft)
    P = P[offset:offset + n]
    grad = (lin_pref - P) * dx
    return grad

def make_quad_precomp(weights, n):
    n = int(n)
    cache = get_ncache(n)
    L, dx, r_min, r_max, Lfft, offset = cache
    weights = np.asarray(weights, dtype=float).ravel()
    if weights.size != L:
        weights = np.full(L, 1.0 / float(L), dtype=float)
    if not np.all(np.isfinite(weights)):
        weights = np.where(np.isfinite(weights), weights, 0.0)
    s = float(np.sum(weights))
    if s <= 0.0 or not np.isfinite(s):
        weights = np.full(L, 1.0 / float(L), dtype=float)
    else:
        weights = weights / s

    lin_pref, W, Kpad = _w_and_P_from_weights(weights, n, cache)
    Kfft = np.fft.rfft(Kpad)
    return (n, dx, lin_pref, Kfft, Lfft, offset)

def quad_obj(x, precomp):
    n, dx, lin_pref, Kfft, Lfft, offset = precomp
    x = np.asarray(x, dtype=float).ravel()
    if x.size != n:
        x = resample_h_area(x, n)
    Hpad = np.zeros(Lfft, dtype=float)
    Hpad[:n] = x
    P = np.fft.irfft(np.fft.rfft(Hpad) * Kfft, Lfft)
    P = P[offset:offset + n]
    raw = float(np.dot(lin_pref, x) - 0.5 * float(np.dot(x, P)))
    return float(dx * raw)

def quad_grad(x, precomp):
    n, dx, lin_pref, Kfft, Lfft, offset = precomp
    x = np.asarray(x, dtype=float).ravel()
    if x.size != n:
        x = resample_h_area(x, n)
    Hpad = np.zeros(Lfft, dtype=float)
    Hpad[:n] = x
    P = np.fft.irfft(np.fft.rfft(Hpad) * Kfft, Lfft)
    P = P[offset:offset + n]
    return (lin_pref - P) * dx

def slsqp_solve(fun, jac, x0, n, args, maxiter):
    n = int(n)
    x0 = np.asarray(x0, dtype=float).ravel()
    if x0.size != n:
        x0 = resample_h_area(x0, n)
    x0 = project_box_sum(x0, 0.5 * float(n))

    bounds = [(0.0, 1.0)] * n
    cons = ({
        "type": "eq",
        "fun": sum_eq_fun,
        "jac": sum_eq_jac,
        "args": (n,)
    },)
    options = {
        "maxiter": int(max(1, int(maxiter))),
        "ftol": 1e-12,
        "disp": False
    }
    try:
        res = minimize(
            fun, x0, method="SLSQP", jac=jac,
            bounds=bounds, constraints=cons,
            args=args, options=options
        )
        y = np.asarray(res.x, dtype=float).ravel()
        if y.size != n or not np.all(np.isfinite(y)):
            y = x0
    except Exception:
        y = x0
    return project_box_sum(y, 0.5 * float(n))

def derivative_matrix(h, n):
    h = np.asarray(h, dtype=float).ravel()
    n = int(n)
    if n <= 0:
        return np.zeros((0, 0), dtype=float)
    L = 2 * n - 1
    mid = n - 1
    lin = np.zeros((n, L), dtype=float)
    lin[:, mid] = 1.0 - 2.0 * h
    if n > 1:
        for s in range(1, n):
            length = n - s
            idxp = mid + s
            lin[0:length, idxp] += 1.0 - h[s:]
            lin[s:n, idxp] += -h[0:length]

            idxn = mid - s
            lin[s:n, idxn] += 1.0 - h[0:length]
            lin[0:length, idxn] += -h[s:]
    return lin

    return random_smooth_start(n, rng), n

def local_refine(h_start, n, deadline, rng):
    n = int(n)
    deadline = float(deadline)
    h_start = np.asarray(h_start, dtype=float).ravel()
    if h_start.size != n:
        h_start = resample_h_area(h_start, n)
    h = project_box_sum(h_start, 0.5 * float(n))

    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c

    smooth_alphas = [5.0, 20.0, 80.0, 300.0, 1000.0, 4000.0, 12000.0, 40000.0]
    for alpha in smooth_alphas:
        if time.time() >= deadline - 30.0:
            break
        maxit = slsqp_maxiter(n, deadline)
        if maxit < 3:
            break
        try:
            h2 = slsqp_smooth(cur_h, n, alpha, maxit)
            c2 = compute_c5(h2, n)
        except Exception:
            continue
        if not np.isfinite(c2):
            continue
        if c2 < best_c - 1e-14:
            best_c = float(c2)
            best_h = np.asarray(h2, dtype=float).ravel().copy()
            cur_h = best_h.copy()
            cur_c = best_c
        else:
            if c2 <= cur_c + 0.02:
                cur_h = np.asarray(h2, dtype=float).ravel().copy()
                cur_c = float(c2)
            if cur_c > best_c + 0.008:
                cur_h = best_h.copy()
                cur_c = best_c

    for alpha_a in [100.0, 500.0, 2000.0]:
        for _ in range(2):
            if time.time() >= deadline - 25.0:
                break
            maxit = max(10, slsqp_maxiter(n, deadline) // 2)
            if maxit < 4:
                break
            try:
                c_raw = c_s_raw(cur_h, n)
                w = softmax_weights(c_raw, alpha_a)
                pre = make_quad_precomp(w, n)
                h2 = slsqp_quad(cur_h, n, pre, maxit)
                c2 = compute_c5(h2, n)
            except Exception:
                continue
            if not np.isfinite(c2):
                continue
            if c2 < best_c - 1e-14:
                best_c = float(c2)
                best_h = np.asarray(h2, dtype=float).ravel().copy()
                cur_h = best_h.copy()
                cur_c = best_c
            else:
                if c2 <= cur_c + 0.02:
                    cur_h = np.asarray(h2, dtype=float).ravel().copy()
                    cur_c = float(c2)
                if cur_c > best_c + 0.008:
                    cur_h = best_h.copy()
                    cur_c = best_c

    if time.time() < deadline - 35.0:
        try:
            hs = symmetrize(best_h, n)
            cs = compute_c5(hs, n)
            if np.isfinite(cs) and cs < best_c - 1e-14:
                best_c = float(cs)
                best_h = hs
            hs = symmetrize(cur_h, n)
            cs = compute_c5(hs, n)
            if np.isfinite(cs) and cs < best_c - 1e-14:
                best_c = float(cs)
                best_h = hs
                cur_h = hs
                cur_c = best_c
        except Exception:
            pass

    if time.time() < deadline - 60.0:
        try:
            h2 = slsqp_smooth(best_h, n, 50000.0, max(12, slsqp_maxiter(n, deadline)))
            c2 = compute_c5(h2, n)
            if np.isfinite(c2) and c2 < best_c - 1e-14:
                best_c = float(c2)
                best_h = h2
        except Exception:
            pass

    if n <= 700 and time.time() < deadline - 25.0:
        try:
            hp, cp = pair_polish(best_h, n, deadline - 5.0, rng)
            if np.isfinite(cp) and cp < best_c - 1e-15:
                best_h = hp
                best_c = float(cp)
        except Exception:
            pass

    best_h = project_box_sum(best_h, 0.5 * float(n))
    best_c = compute_c5(best_h, n)
    return best_h, float(best_c)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget_s = float(budget_s)
        if budget_s < 30.0:
            budget_s = 30.0
    except Exception:
        budget_s = 1100.0
    hard_deadline = t0 + max(60.0, budget_s - 45.0)
    rng = np.random.RandomState(int(seed) if seed is not None else 42)

    try:
        h0, n0 = get_initial_h(seed)
        n0 = int(n0)
        best_h, best_c = consider_candidate(h0, n0, h0.copy(), compute_c5(h0, n0))

        try:
            hs = symmetrize(h0, n0)
            cs = compute_c5(hs, n0)
            if np.isfinite(cs) and cs < best_c:
                best_h = hs
                best_c = float(cs)
        except Exception:
            pass

        # 1) Refine the parent at its natural resolution.
        remaining = hard_deadline - time.time()
        if remaining > 180.0:
            alloc = min(320.0, remaining - 180.0)
            if alloc > 30.0:
                task_deadline = min(hard_deadline - 20.0, time.time() + alloc)
                hl, cl = local_refine(best_h, n0, task_deadline, rng)
                best_h, best_c = consider_candidate(hl, n0, best_h, best_c)

        # 2) Multi-resolution transfer: coarse -> fine.
        n_coarse = max(90, n0 // 2)
        remaining = hard_deadline - time.time()
        if remaining > 320.0 and n_coarse != n0 and n_coarse <= max(80, n0 // 2):
            try:
                hc = resample_h_area(best_h, n_coarse)
                alloc = min(170.0, remaining - 320.0)
                if alloc > 35.0:
                    task_deadline = min(hard_deadline - 70.0, time.time() + alloc)
                    hc2, cc = local_refine(hc, n_coarse, task_deadline, rng)
                    ht = resample_h_area(hc2, n0)
                    remaining = hard_deadline - time.time()
                    if remaining > 180.0:
                        alloc2 = min(170.0, remaining - 180.0)
                        if alloc2 > 35.0:
                            task_deadline = min(hard_deadline - 70.0, time.time() + alloc2)
                            ht2, ct = local_refine(ht, n0, task_deadline, rng)
                            best_h, best_c = consider_candidate(ht2, n0, best_h, best_c)
            except Exception:
                pass

        # 3) Higher resolution from the best found shape.
        def add_high_resolution_task(n_target):
            nonlocal_t0 = t0
            return False

        nbest = int(len(best_h))
        target_highs = []
        for cand in [
            int(round(n0 * 1.25)),
            int(round(n0 * 1.6)),
            int(round(n0 * 2.0)),
            max(n0 + 50, 400)
        ]:
            cand = int(max(80, min(750, cand)))
            if cand != nbest and cand not in target_highs:
                target_highs.append(cand)

        for nh in target_highs[:2]:
            remaining = hard_deadline - time.time()
            if remaining < 170.0:
                break
            try:
                hh = resample_h_area(best_h, nh)
                alloc = min(170.0, remaining - 170.0)
                if alloc < 35.0:
                    break
                task_deadline = min(hard_deadline - 70.0, time.time() + alloc)
                hloc, cloc = local_refine(hh, nh, task_deadline, rng)
                best_h, best_c = consider_candidate(hloc, nh, best_h, best_c)
            except Exception:
                break

        # 4) Controlled perturbations of the current best basin.
        nbest = int(len(best_h))
        scales = [0.03, 0.06, 0.10, 0.16]
        for scale in scales:
            remaining = hard_deadline - time.time()
            if remaining < 95.0:
                break
            try:
                hp = best_h + smooth_noise(nbest, rng, scale)
                hp = project_box_sum(hp, 0.5 * float(nbest))
                hp = project_box_sum(symmetrize(hp, nbest), 0.5 * float(nbest))
                alloc = min(85.0, remaining - 95.0)
                if alloc < 20.0:
                    break
                task_deadline = min(hard_deadline - 50.0, time.time() + alloc)
                hloc, cloc = local_refine(hp, nbest, task_deadline, rng)
                old_c = best_c
                best_h, best_c = consider_candidate(hloc, nbest, best_h, best_c)
                if not np.isfinite(best_c):
                    best_c = old_c
            except Exception:
                break

        # 5) Final polish.
        nbest = int(len(best_h))
        remaining = hard_deadline - time.time()
        if remaining > 170.0:
            try:
                hloc, cloc = local_refine(best_h, nbest, hard_deadline - 10.0, rng)
                best_h, best_c = consider_candidate(hloc, nbest, best_h, best_c)
            except Exception:
                pass
        elif remaining > 45.0:
            try:
                hp, cp = pair_polish(best_h, nbest, hard_deadline - 5.0, rng)
                if np.isfinite(cp) and cp < best_c:
                    best_h = hp
                    best_c = float(cp)
            except Exception:
                pass

        try:
            hs = symmetrize(best_h, nbest)
            cs = compute_c5(hs, nbest)
            if np.isfinite(cs) and cs < best_c:
                best_h = hs
                best_c = float(cs)
        except Exception:
            pass

        best_h = project_box_sum(best_h, 0.5 * float(nbest))
        best_c = compute_c5(best_h, nbest)

        if not np.isfinite(best_c):
            hfb = project_box_sum(np.full(n0 if n0 > 0 else 250, 0.5), 0.5)
            return hfb, compute_c5(hfb, len(hfb)), len(hfb)

        return best_h, float(best_c), int(len(best_h))

    except Exception:
        try:
            h, n = get_initial_h(seed)
        except Exception:
            n = 250
            h = project_box_sum(np.full(n, 0.5), 0.5)
        h = project_box_sum(h, 0.5 * float(len(h)))
        return h, compute_c5(h, len(h)), int(len(h))
