def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def active_set_obj(x, n, active_idx, alpha, mu):
    """Smooth max over active set of shifts with log-sum-exp."""
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_set_grad(x, n, active_idx, alpha, mu):

    vals = c[active_idx]
    mx = vals.max()

    dx = 2.0 / n
    grad = np.zeros(n)
    nw = len(active_idx)

        wm = wi * dx

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def shift_grad_single(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv <= 0:
            return grad
        grad[0:nv] += dx
        grad[0:nv] -= dx * h[k:n]
        grad[k:n] -= dx * h[0:nv]
    else:
        kk = -k
        nv = n - kk
        if nv <= 0:
            return grad
        grad[kk:n] += dx
        grad[kk:n] -= dx * h[0:nv]
        grad[0:nv] -= dx * h[kk:n]
    return grad


def active_set_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def active_set_grad(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        g = shift_grad_single(x, n, k)
        grad += wi * g
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def find_active(c_vec, rel_tol=0.001, top_k=None):
    max_val = c_vec.max()
    threshold = max_val - max(1e-9, rel_tol * max_val)
    active = np.where(c_vec >= threshold)[0]
    if top_k is not None and len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    return active


def lbfgs_active(h, n, active, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(active_set_obj, h, jac=active_set_grad,
                       args=(n, active, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 12, 'maxls': 50})
        h_out = project_feasible(res.x, n)
    except Exception:
        h_out = h.copy()
    return h_out


def iterative_active(h, n, deadline, max_rounds=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for rnd in range(max_rounds):
        if time.time() > deadline - 10:
            break

        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()

        active = find_active(c_vec, rel_tol=0.003, top_k=15)
        if len(active) < 2:
            active = np.argsort(-c_vec)[:5]

        for alpha in [5000, 30000, 100000, 500000]:
            if time.time() > deadline - 10:

            h_new = lbfgs_active(best_h, n, active, alpha, 5.0, 3000, deadline)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                best_h = h_new

def coarse_refine(h_fine, n_fine, n_coarse, deadline):
    x_fine = np.linspace(0, 2, n_fine)
    x_coarse = np.linspace(0, 2, n_coarse)
    h_c = np.interp(x_coarse, x_fine, h_fine)
    h_c = project_feasible(h_c, n_coarse)

    t_budget = min(50, (deadline - time.time()) * 0.5)
    if t_budget > 20:
        h_c, _ = iterative_active(h_c, n_coarse, time.time() + t_budget, max_rounds=8)

    h_up = np.interp(x_fine, x_coarse, h_c)
    h_up = project_feasible(h_up, n_fine)
    return h_up


def slqp_constr(x, n):
    return float(x.sum() - n / 2.0)


def slqp_constr_jac(x, n):
    return np.ones(n)


def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)


def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_constr, 'jac': slqp_constr_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)


def gradient_descent(h, n, deadline, base_step=0.003, n_iters=400):
    best_c = compute_c5(h, n)
    h = h.copy()
    alpha = 100000
    mu = 5.0

    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()

        if max_val < best_c:
            best_c = max_val

        active = find_active(c_vec, rel_tol=0.002, top_k=8)
        if len(active) < 1:
            break

        g = active_set_grad(h, n, active, alpha, mu)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break

        improved = False
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            step = base_step * step_mult
            h_trial = h - step * g
            h_trial = project_feasible(h_trial, n)
            if np.allclose(h_trial, h, atol=1e-15):
                continue
            c_trial = compute_c5(h_trial, n)
            if c_trial < max_val - 1e-14:
                h = h_trial
                improved = True
                break

        if not improved:
            base_step *= 0.5
            if base_step < 1e-9:
                break

    return h, best_c


def pair_search(h, n, deadline, radius=60):

                for (i, j) in pairs:
                    if time.time() > deadline:
                        return h, best_c
                    if h[i] - d >= 0 and h[j] + d <= 1:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d

                    if h[i] + d <= 1 and h[j] - d >= 0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d

    return h, best_c


def random_perturb(h, n, n_perturbations=5, magnitude=0.05):
    """Try random perturbations to escape local optima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for _ in range(n_perturbations):
        h_pert = h.copy()
        # Random small perturbation
        mask = np.random.random(n) < 0.3
        h_pert[mask] += np.random.uniform(-magnitude, magnitude, mask.sum())
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)

        # Quick local optimization from perturbed point
        c_vec = compute_c5_vec(h_pert, n)
        active = find_active(c_vec, rel_tol=0.005, top_k=10)
        h_opt = lbfgs_active(h_pert, n, active, 50000, 5.0, 1000, time.time() + 8)
        c_opt = compute_c5(h_opt, n)

        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    return best_h, best_c


def full_smooth_lbfgs(h, n, deadline, alphas=None):
    """L-BFGS-B with full smooth max over all shifts."""
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000]

    def full_obj(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        return float(val)

    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(full_obj, best_h, args=(n, alpha),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-18,
                                    'gtol': 1e-13, 'maxfun': 20000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Full smooth L-BFGS for initial convergence (0-100s)
    t1 = min(t0 + 100, deadline - 500)
    best_h, best_c = full_smooth_lbfgs(best_h, n, t1,
                                       alphas=[50, 200, 1000, 5000, 20000])

    # Phase 2: Iterative active-set optimization (100-300s)
    t2 = min(time.time() + 200, deadline - 300)
    best_h, best_c = iterative_active(best_h, n, t2, max_rounds=15)

    # Phase 3: Coarse-to-fine perturbation (300-380s)
    if time.time() < deadline - 250:
        n_coarse = max(80, n // 5)
        t3 = time.time() + 50
        h_coarse = coarse_refine(best_h, n, n_coarse, t3)
        c_coarse = compute_c5(h_coarse, n)
        if c_coarse < best_c:
            best_h = h_coarse
            best_c = c_coarse
        else:
            # Try blends
            for t_blend in [0.2, 0.4, 0.6]:
                h_b = (1 - t_blend) * best_h + t_blend * h_coarse
                h_b = project_feasible(h_b, n)
                c_b = compute_c5(h_b, n)
                if c_b < best_c:
                    best_c = c_b
                    best_h = h_b

    # Phase 4: Random perturbations to escape local optima (380-450s)
    if time.time() < deadline - 200:
        t4 = min(time.time() + 50, deadline - 200)
        h_p, c_p = random_perturb(best_h, n, n_perturbations=4, magnitude=0.04)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p

    # Phase 5: Another round of iterative active optimization (450-650s)
    t5 = min(time.time() + 180, deadline - 180)
    best_h, best_c = iterative_active(best_h, n, t5, max_rounds=20)

    # Phase 6: Gradient descent refinement (650-730s)
    if time.time() < deadline - 100:
        t6 = min(time.time() + 70, deadline - 100)
        h_g, c_g = gradient_descent(best_h, n, t6, base_step=0.002, n_iters=300)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g

    # Phase 7: Tight active-set optimization (730-830s)
    if time.time() < deadline - 60:
        c_vec = compute_c5_vec(best_h, n)
        active_tight = find_active(c_vec, rel_tol=0.0002, top_k=5)
        if len(active_tight) < 2:
            active_tight = np.argsort(-c_vec)[:3]
        t7 = min(time.time() + 80, deadline - 60)
        for alpha in [500000, 2000000, 10000000]:
            if time.time() > t7 - 5:
                break
            h_t = lbfgs_active(best_h, n, active_tight, alpha, 5.0, 3000, t7)
            c_t = compute_c5(h_t, n)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                # Update active set
                c_vec = compute_c5_vec(best_h, n)
                active_tight = find_active(c_vec, rel_tol=0.0002, top_k=5)
                if len(active_tight) < 2:
                    active_tight = np.argsort(-c_vec)[:3]

    # Phase 8: SLSQP polish (830-950s)
    if time.time() < deadline - 50:
        for alpha in [200000, 2000000, 10000000]:
            if time.time() > deadline - 40:
                break
            t8 = min(time.time() + 35, deadline - 40)
            h_s, c_s = slqp_polish(best_h, n, t8, alpha=alpha, maxiter=2000)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 9: Fine pair search (950-1080s)
    if time.time() < deadline - 15:
        t9 = deadline - 10
        best_h, best_c = pair_search(best_h, n, t9, radius=50)

    # Phase 10: Final SLSQP micro-polish
    if time.time() < deadline - 3:
        h_f, c_f = slqp_polish(best_h, n, deadline - 2, alpha=5000000, maxiter=1000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
