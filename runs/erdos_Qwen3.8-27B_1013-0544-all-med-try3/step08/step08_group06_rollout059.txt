    shifted = alpha * (c - m)
    result = shifted - shifted.max()

This is essentially a cross-correlation between h and the indicator vectors. Can I compute all gradients at once?

For a fixed lag d:
grad_d[j] = (1-h[j-d]) * [max(0,d) ≤ j < min(n, n+d)] - h[j+d] * [max(0,-d) ≤ j < min(n, n-d)]

The first term is (1-h) correlated with a shifted indicator. The second term is h correlated with a shifted indicator.

Actually, for the first term when d ≥ 0:
(1-h[j-d]) for j in [d, n-1]
This is just (1-h) shifted right by d, zero-padded.

For the second term when d ≥ 0:
-h[j+d] for j in [0, n-d-1]
This is -h shifted left by d, zero-padded.

So for each lag d, grad_d is:
grad_d = (1-h) shifted right by d (zero-padded, only positions [d, n-1]) - h shifted left by d (zero-padded, only positions [0, n-d-1])

Equivalently:
grad_d = correlate(1-h, [0...0, 1, 0...0]_{len n, 1 at position d}) - correlate(h, [0...0, 1, 0...0]_{len n, 1 at position d})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_f(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def active_grad(h, n, lag):
    i_lo = max(0, lag)
    i_hi = min(n, n + lag)
    grad = np.zeros(n)
    if i_lo < i_hi:
        idx1 = np.arange(i_lo, i_hi)
        j_lag = idx1 - lag
        valid1 = (j_lag >= 0) & (j_lag < n)
        grad[idx1[valid1]] += (1.0 - h[j_lag[valid1]])
        j_lo2 = max(0, i_lo - lag)
        j_hi2 = min(n, i_hi - lag)
        if j_lo2 < j_hi2:
            idx2 = np.arange(j_lo2, j_hi2)
            j_lag2 = idx2 + lag
            valid2 = (j_lag2 >= 0) & (j_lag2 < n)
            grad[idx2[valid2]] -= h[j_lag2[valid2]]
    return grad

def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    e = np.exp(shifted - mx)
    return float(m + (mx + np.log(np.sum(e))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def do_slsqp(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() < deadline:
            return project_f(res.x, n)
    except Exception:
        pass
    return h.copy()

def gradient_step(h, n, top_lags, lr):
    """Projected gradient step targeting top lags."""
    dx = 2.0 / n
    corr = compute_corr(h, n)
    scores = corr * dx
    max_s = scores.max()
    active_mask = scores >= max_s - 0.002
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        active_indices = [int(np.argmax(scores))]
    n_active = min(len(active_indices), top_lags)
    active_subset = np.argsort(scores[active_indices])[-n_active:]
    active_subset = active_indices[active_subset]
    
    weights = scores[active_subset] - (scores[active_subset].min())
    if weights.sum() < 1e-15:
        weights = np.ones(len(active_subset))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for idx, k in enumerate(active_subset):
        lag = k - (n - 1)
        grad += weights[idx] * active_grad(h, n, lag)
    grad *= dx
    grad /= grad.sum()  # normalize for step size control
    grad -= grad.mean()
    
    h_new = h - lr * grad
    return project_f(h_new, n), compute_c5(project_f(h_new, n), n)

def targeted_search(h, n, deadline, deltas):
    """Efficient search targeting active lags."""
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        stall = 0
        for _pass in range(500):

            improved = False
            corr = compute_corr(h, n)
            dx = 2.0 / n
            scores = corr * dx
            max_s = scores.max()
            
            # Get top active lags
            top_n = min(5, len(scores))
            top_k = np.argsort(scores)[-top_n:][::-1]
            
            for k in top_k:
                if time.time() > deadline:
                    return h, best_c
                lag = k - (n - 1)
                grad = active_grad(h, n, lag)
                sorted_idx = np.argsort(grad)
                
                # Try top candidate pairs
                for attempt in range(15):
                    if attempt % 2 == 0:
                        j_inc = sorted_idx[attempt % n]
                        j_dec = sorted_idx[(n - 1 - attempt) % n]
                    else:
                        j_inc = sorted_idx[(attempt * 7 + 3) % n]
                        j_dec = sorted_idx[(n - 1 - attempt * 5 - 2) % n]
                    
                    if j_inc == j_dec:
                        continue
                    
                    if h[j_inc] + d <= 1.0 + 1e-10 and h[j_dec] - d >= -1e-10:
                        h[j_inc] += d
                        h[j_dec] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            stall = 0
                        else:
                            h[j_inc] -= d
                            h[j_dec] += d
                    
                    if time.time() > deadline:
                        return h, best_c
            
            if not improved:
                stall += 1
                if stall > 5:
                    break
    
    return h, best_c

def pair_search_small(h, n, deadline, deltas):

    best_c = compute_c5(h, n)

        if time.time() > deadline:

                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

    return h, best_c

def make_antipodal(h, n):
    """Create antipodally symmetric version: h[i] = 1 - h[n-1-i]"""
    ha = h.copy()
    for i in range(n // 2):
        ha[i] = h[i]
        ha[n - 1 - i] = 1.0 - h[i]
    if n % 2 == 1:
        ha[n // 2] = 0.5
    return project_f(ha, n)

def make_symmetric(h, n):
    """Create centrally symmetric version: h[i] = h[n-1-i]"""
    hs = h.copy()
    for i in range(n // 2):
        hs[i] = (h[i] + h[n - 1 - i]) / 2.0
        hs[n - 1 - i] = hs[i]
    return project_f(hs, n)
