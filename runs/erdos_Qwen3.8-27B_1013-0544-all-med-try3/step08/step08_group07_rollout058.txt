import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slope_ascent_step(h, n, alpha, maxiter, deadline):
    """SLSQP with log-sum-exp smoothing"""
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        # Stable log-sum-exp
        sh_max = sh.max()
        if sh_max > 700:
            sh = sh - sh_max
            log_sum = sh_max + np.log(np.exp(sh).sum())
        else:
            log_sum = np.log(np.exp(sh).sum())
        return (m + log_sum / alpha)
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

    c_all = compute_all_c(h, n)  # (2n-1,) array

    golden = (5**0.5 - 1) / 2  # ~0.618
    
    def eval_h_i(x, i, c_all, h):
        """Evaluate max c_k after setting h[i]=x (without updating c_all)."""
        # c_k changes:
        # For d=0 (index n-1 in c_all): delta = (x*(1-x) - h[i]*(1-h[i]))*dx
        # For d≠0:
        #   if 0<=i-d<n: c[n-1+d] += (x - h[i])*(1-h[i-d])*dx
        #   if 0<=i+d<n: c[n-1+d] += h[i+d]*(h[i]-x)*dx = c[n-1+d] + (x-h[i])*(-h[i+d])*dx
        
        delta = x - h[i]
        if abs(delta) < 1e-15:
            return c_all.max()
        
        c_new = c_all.copy()
        # d=0: index n-1
        c_new[n-1] += (x*(1-x) - h[i]*(1-h[i])) * dx
        # d≠0:
        for d in range(-(n-1), n):
            if d == 0:
                continue
            idx = n - 1 + d
            contrib = 0.0
            if 0 <= i - d < n:
                contrib += (1 - h[i-d]) * dx
            if 0 <= i + d < n:
                contrib += (-h[i+d]) * dx
            c_new[idx] += delta * contrib
        return c_new.max()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            # Golden section search for h[i] in [0, 1]
            a, b = 0.0, 1.0
            c_val = a + (1 - golden) * (b - a)
            d_val = a + golden * (b - a)
            fc = eval_h_i(c_val, i, c_all, best_h)
            fd = eval_h_i(d_val, i, c_all, best_h)
            for _ in range(25):
                if fc < fd:
                    b = d_val; d_val = c_val; fd = fc
                    c_val = a + (1 - golden) * (b - a)
                    fc = eval_h_i(c_val, i, c_all, best_h)
                else:
                    a = c_val; c_val = d_val; fc = fd
                    d_val = a + golden * (b - a)
                    fd = eval_h_i(d_val, i, c_all, best_h)
            x_new = (a + b) / 2
            if abs(x_new - best_h[i]) > 1e-10:
                best_h[i] = x_new
                # Update c_all
                delta = x_new - best_h[i]  # wait, best_h[i] was already changed
                # Need to recompute or update incrementally
                # Let's just recompute c_all
                c_all = compute_all_c(best_h, n)
                imp = True
        
        # Project
        best_h = project_feasible(best_h, n)
        c_all = compute_all_c(best_h, n)
        c = c_all.max()
        if c < best_c:
            best_c = c
        c_all = compute_all_c(best_h, n)  # ensure consistency
    
    return best_h, best_c
