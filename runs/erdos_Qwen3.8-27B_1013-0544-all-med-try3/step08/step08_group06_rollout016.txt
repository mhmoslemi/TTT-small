import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            tot = mask.sum()
            if tot < 1e-15:
                break
            h[mask] += diff / tot
        else:
            mask = h > 0.0
            tot = mask.sum()
            if tot < 1e-15:
                break
            h[mask] += diff / tot
    return np.clip(h, 0.0, 1.0)

def weighted_gradient(h, n, weights):
    """Compute sum_k w_k * dc_k/dh where c_k are correlation lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    indices = np.where(weights > 1e-14)[0]
    for idx in indices:
        k = idx - (n - 1)
        w = weights[idx]
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            ka = -k
            grad[ka:] += w * (1.0 - h[:n - ka])
            grad[:n - ka] -= w * h[ka:]
    grad *= dx
    return grad

def lse_value_and_weights(h, n, alpha):
    """Compute log-sum-exp value and softmax weights."""
    c = compute_c5_array(h, n)
    m = c.max()
    if alpha > 1e6:
        w = np.zeros(2 * n - 1)
        active = np.where(c >= m - 1e-10)[0]
        w[active] = 1.0 / len(active)
        return m, w
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    log_sum = max_sh + np.log(exp_sh.sum())
    value = m + log_sum / alpha
    w = exp_sh / exp_sh.sum()
    return value, w

def projected_gradient_lse(h, n, alpha, max_iter, deadline, step_init=None):
    """Projected gradient with LSE smoothing."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_true = compute_c5(best_h, n)
    
    val, w = lse_value_and_weights(h, n, alpha)
    if step_init is None:
        step = 2.0 / (alpha * (2.0 / n) * (2 * n - 1) ** 0.5 + 1e-8)
        step = min(step, 0.5)
    else:
        step = step_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad = weighted_gradient(h, n, w)
        
        # Gradient step with backtracking
        lr = step
        h_old = h.copy()
        val_old = val
        found = False
        for _ls in range(15):
            h_try = project_feasible(h_old - lr * grad, n)
            if np.allclose(h_try, h_old, atol=1e-15):
                break
            val_try, w_try = lse_value_and_weights(h_try, n, alpha)
            if val_try < val_old - 1e-14:
                h = h_try
                val = val_try
                w = w_try
                found = True
                break
            lr *= 0.5
        step = max(lr, 1e-10) if not found else min(step * 1.1, 5.0)
        
        if it % 100 == 0:
            tc = compute_c5(h, n)
            if tc < best_true - 1e-15:
                best_true = tc
                best_h = h.copy()
    
    tc = compute_c5(h, n)
    if tc < best_true:
        best_true = tc
        best_h = h.copy()
    return best_h, best_true

def coordinate_polish(h, n, deadline, deltas=None):
    """Coordinate descent polish."""
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

    best_c = compute_c5(h, n)

        if time.time() > deadline:
            break

        passes = 0

                return h, best_c
            improved = False
            passes += 1

                if time.time() > deadline:
                    return h, best_c

                    hi = h[i] + dj

                        h[i] = hi
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] = hi - dj

    """Pair update polish."""
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005]

        radius = min(n, 60)
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(30):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi, 2):
                    if i >= j:
                        if j <= i + radius:
                            pass
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

    """Run LSE optimization with increasing alphas."""
    if alphas is None:
