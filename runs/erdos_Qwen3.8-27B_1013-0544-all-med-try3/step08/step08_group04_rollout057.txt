    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

            if tot < 1e-15:

    """Compute gradient of c_{k_shift} w.r.t. h, where k_shift is the lag index in full correlation.

    Actual shift = index - (n-1).

    return float(corr.max())

def project_feasible(h, n):

            if tot < 1e-15:

            h += (diff / tot) * h

    """Compute gradient matrix G of shape (2n-1, n) where G[k,j] = dc_k/dh[j]."""
    dx = 2.0 / n

                val -= h[j-s]

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c = np.maximum(c, 1e-15)
    cp = c ** p
    Sp = cp.sum()
    val = Sp ** (1.0 / p)
    # gradient
    coeff = Sp ** (1.0 / p - 1.0)
    # sum_k p * c_k^(p-1) * G[k,j]
    # Compute weighted sum without forming full G
    weighted = np.zeros(n)
    for k in range(2 * n - 1):
        w = p * c[k] ** (p - 1)
        s = k - (n - 1)
        for j in range(n):
            if 0 <= j + s < n:
                weighted[j] += w * dx * (1.0 - x[j + s])
            if 0 <= j - s < n:
                weighted[j] -= w * dx * x[j - s]
    grad = coeff * weighted
    return val, grad

def pnorm_obj(x, n, p):
    v, _ = pnorm_obj_and_grad(x, n, p)
    return v

def _sum_c(x, n):
    return x.sum() - n / 2.0

def _sum_j(x, n):
    return np.ones(n)

def pnorm_slsqp(h, n, p, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(pnorm_obj, h, args=(n, p), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_optimize(h, n, deadline, n_top=5):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    alpha = 0.02
    stall = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        cmax = c.max()
        # Find top n_top shifts
        top_idx = np.argsort(c)[-n_top:]
        # Average gradient of top shifts
        G_avg = np.zeros(n)
        for k in top_idx:
            G_avg += gradient_single(h, n, int(k))
        G_avg /= len(top_idx)
        gnorm = np.linalg.norm(G_avg)
        if gnorm < 1e-14:
            break
        G_avg /= gnorm
        # Line search
        found = False
        for a in [alpha, alpha * 2, alpha / 2, alpha / 4, alpha / 8]:
            h_try = project_feasible(h - a * G_avg * gnorm * 0.1, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                stall = 0
                found = True
                break
        if not found:
            # Try smaller steps
            for a in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
                h_try = project_feasible(h - a * G_avg, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-15:
                    h = h_try
                    best_c = c_try
                    alpha = a
                    stall = 0
                    found = True
                    break
        if not found:
            stall += 1
            alpha *= 0.7
            if stall > 30 or alpha < 1e-8:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=60):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if i + 1 > j:
                        continue
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[j] += d
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[j] -= d
                            h[i] += d
            if not improved:
                break
    return h, best_c

def random_pair_search(h, n, deadline, deltas, n_pairs_per_delta=3000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(int(time.time()) % 100000)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for _ in range(n_pairs_per_delta):
                if time.time() > deadline:
                    return h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
                if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                    h[j] += d
                    h[i] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[j] -= d
                        h[i] += d
            if not improved:
                break
    return h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    if n_new >= n_old:
        return upsample(h, n_old, n_new)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    # Multi-p SLSQP
    p_list = [20, 50, 100, 200]
    for p in p_list:
        if time.time() > deadline - 20:
            break
        h_try = pnorm_slsqp(h, n, p, maxiter=1500)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            h = h_try

    # Subgradient polish
    if time.time() < deadline - 15:
        sg_deadline = min(time.time() + 60, deadline - 10)
        h_sg, c_sg = subgradient_optimize(h, n, sg_deadline, n_top=5)
        if c_sg < best_c - 1e-14:
            best_c = c_sg
            h = h_sg

    # Pair search
    if time.time() < deadline - 5:
        if n <= 300:
            h_ps, c_ps = near_pair_search(h, n, deadline - 3,
                                          [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=80)
        else:
            h_ps, c_ps = near_pair_search(h, n, deadline - 3,
                                          [0.005, 0.002, 0.001, 0.0005], radius=50)
        if c_ps < best_c - 1e-14:
            best_c = c_ps
            h = h_ps

    # Random pair search
    if time.time() < deadline - 3:
        h_rs, c_rs = random_pair_search(h, n, deadline - 1,
                                        [0.002, 0.001, 0.0005, 0.0002], n_pairs_per_delta=2000)
        if c_rs < best_c - 1e-14:
            best_c = c_rs
            h = h_rs

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at initial resolution
    alloc1 = min(180, (deadline - t0) * 0.18)
    h1, c1 = optimize_at_resolution(h_init, n_init, t0 + alloc1, rng)
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Optimize at 1.5x resolution
    n2 = int(round(n_init * 1.5))
    if time.time() < deadline - 120:
        h2_start = upsample(best_h, len(best_h), n2)
        alloc2 = min(180, (deadline - time.time()) * 0.18)
        h2, c2 = optimize_at_resolution(h2_start, n2, time.time() + alloc2, rng)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Optimize at 2x resolution
    n3 = int(round(n_init * 2))
    if time.time() < deadline - 120:
        h3_start = upsample(best_h, len(best_h), n3)
        alloc3 = min(200, (deadline - time.time()) * 0.20)
        h3, c3 = optimize_at_resolution(h3_start, n3, time.time() + alloc3, rng)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Multi-restart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 40
        n_restarts = max(3, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 + 0.04 * (r + 1) / (n_restarts + 1)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc_r = min(55, (deadline - time.time() - 25) * 0.4)
            if alloc_r < 20:
                break
            hr, cr = optimize_at_resolution(hp, n_best, time.time() + alloc_r, rng)
            if cr < best_c:
                best_c, best_h = cr, hr

    # Phase 5: Try a completely different starting point
    if time.time() < deadline - 60:
        # Structured initialization: alternate high/low
        n_alt = n_best
        h_alt = np.zeros(n_alt)
        for i in range(n_alt):
            if i % 4 < 2:
                h_alt[i] = 0.7
            else:
                h_alt[i] = 0.3
        h_alt = project_feasible(h_alt, n_alt)
        # Add small random perturbation
        h_alt = project_feasible(h_alt + rng.normal(0, 0.03, n_alt), n_alt)
        alloc_alt = min(60, (deadline - time.time()) * 0.15)
        if alloc_alt > 20:
            h_alt_opt, c_alt = optimize_at_resolution(h_alt, n_alt, time.time() + alloc_alt, rng)
            if c_alt < best_c:
                best_c, best_h = c_alt, h_alt_opt

    # Phase 6: Final intensive polishing
    if time.time() < deadline - 5:
        n_f = len(best_h)
        # Subgradient
        h_f, c_f = subgradient_optimize(best_h.copy(), n_f, min(time.time() + 40, deadline - 5), n_top=8)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

        # Very fine pair search
        if time.time() < deadline - 2:
            if n_f <= 300:
                h_f, c_f = near_pair_search(best_h.copy(), n_f, deadline - 1,
                                            [0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=100)
            else:
                h_f, c_f = near_pair_search(best_h.copy(), n_f, deadline - 1,
                                            [0.001, 0.0005, 0.0002, 0.0001], radius=50)
            if c_f < best_c:
                best_c, best_h = c_f, h_f

        # One more SLSQP at high p
        if time.time() < deadline - 2:
            h_f = pnorm_slsqp(best_h, n_f, 500, maxiter=2000)
            c_f = compute_c5(h_f, n_f)
            if c_f < best_c:
                best_c, best_h = c_f, h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
