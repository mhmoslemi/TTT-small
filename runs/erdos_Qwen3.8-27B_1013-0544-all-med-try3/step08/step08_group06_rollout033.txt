import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smax_and_grad(h, n, alpha):
    dx = 2.0 / n
    g = 1.0 - h
    c_raw = np.correlate(h, g, mode='full')  # length 2n-1
    c = c_raw * dx
    
    # Smooth max
    sh = alpha * c
    mx = sh.max()
    exp_sh = np.exp(sh - mx)
    sum_exp = exp_sh.sum()
    val = (mx + np.log(sum_exp)) / alpha
    
    # Softmax weights
    w = exp_sh / sum_exp  # length 2n-1, sums to 1
    
    # Gradient
    # grad[i] = dx * sum_j w[j] * d(correlate(h,g)[j])/dh[i]
    # where correlate(h,g)[j] = sum_n h[n+l] * g[n], l = j-(n-1)
    # d/dh[i] = (1-h[i-l]) * [0<=i-l<n] - h[i+l] * [0<=i+l<n]
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-17:
            continue
        l = j - (n - 1)  # lag, ranges -(n-1) to (n-1)
        # Term 1: wj * (1-h[i-l]) for valid i: 0<=i-l<n AND 0<=i<n
        # i range: max(0, l) <= i < min(n, n+l)
        i_lo1 = max(0, l)
        i_hi1 = min(n, n + l)
        if i_lo1 < i_hi1:
            grad[i_lo1:i_hi1] += wj * (1.0 - h[i_lo1 - l:i_hi1 - l])
        # Term 2: -wj * h[i+l] for valid i: 0<=i+l<n AND 0<=i<n
        # i range: max(0, -l) <= i < min(n, n-l)
        i_lo2 = max(0, -l)
        i_hi2 = min(n, n - l)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= wj * h[i_lo2 + l:i_hi2 + l]
    
    grad *= dx
    return val, grad

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_optimize(h, n, alpha, maxiter=1000, factr=1e12):
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        v, _ = smax_and_grad(x, n, alpha)
        return v
    
    def grad(x):
        _, g = smax_and_grad(x, n, alpha)
        return g
    
    try:
        res = fmin_l_bfgs_b(fun, h, fprime=grad, pgtol=1e-14, maxiter=maxiter, factr=factr)
        x_opt = res  # fmin_l_bfgs_b returns the x directly
    except Exception:
        x_opt = h
    
    # Now enforce sum constraint by projection
    x_opt = project_feasible(x_opt, n)
    return x_opt

def smax_and_grad(h, n, alpha, penalty_M=10000.0):
    """Smooth max objective with sum constraint penalty, and analytical gradient."""
    dx = 2.0 / n
    g = 1.0 - h
    c_raw = np.correlate(h, g, mode='full')
    c = c_raw * dx
    
    # Smooth max
    sh = alpha * c
    mx = sh.max()
    exp_sh = np.exp(sh - mx)
    sum_exp = exp_sh.sum()
    val_smax = (mx + np.log(sum_exp)) / alpha
    
    # Sum constraint penalty
    sum_h = h.sum()
    target = n / 2.0
    diff_sum = sum_h - target
    val_penalty = penalty_M * diff_sum ** 2
    val = val_smax + val_penalty
    
    # Softmax weights
    w = exp_sh / sum_exp
    
    # Gradient of smooth max part
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-17:

            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment: tweak one element
    residual = target_sum - h.sum()
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + residual)
    return h


def smax_and_grad(h, n, alpha, penalty_M):
    dx = 2.0 / n
    g = 1.0 - h
    c_raw = np.correlate(h, g, mode='full')
    c = c_raw * dx

    sh = alpha * c
    mx = sh.max()
    exp_sh = np.exp(sh - mx)
    sum_exp = exp_sh.sum()
    val_smax = (mx + np.log(sum_exp)) / alpha

    sum_h = h.sum()
    target = n / 2.0
    diff_sum = sum_h - target
    val_penalty = penalty_M * diff_sum * diff_sum
    val = val_smax + val_penalty

    w = exp_sh / sum_exp

    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-17:
            continue
        l = j - (n - 1)
        i_lo1 = max(0, l)
        i_hi1 = min(n, n + l)
        if i_lo1 < i_hi1:
            grad[i_lo1:i_hi1] += wj * (1.0 - h[i_lo1 - l:i_hi1 - l])
        i_lo2 = max(0, -l)
        i_hi2 = min(n, n - l)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= wj * h[i_lo2 + l:i_hi2 + l]
    grad *= dx

    grad += 2.0 * penalty_M * diff_sum

    return val, grad


def _smax_fun(x, n, alpha, M):
    v, _ = smax_and_grad(x, n, alpha, M)
    return v


def _smax_grad(x, n, alpha, M):
    _, g = smax_and_grad(x, n, alpha, M)
    return g


def lbfgs_step(h, n, alpha, penalty_M, maxiter):
    try:
        x_opt = fmin_l_bfgs_b(
            _smax_fun, h, fprime=_smax_grad,
            args=(n, alpha, penalty_M),
            pgtol=1e-15, maxiter=maxiter, factr=1e15, iprint=0
        )
    except Exception:
        x_opt = h.copy()
    return project_feasible(x_opt, n)


def pair_search(h, n, deadline, deltas, max_passes=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        old_i, old_j = h[i], h[j]
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = old_i, old_j
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        old_i, old_j = h[i], h[j]
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = old_i, old_j
            if not improved:
                break
    return h, best_c


def near_pair_search(h, n, deadline, deltas, radius=30, max_passes=15):
    best_c = compute_c5(h, n)
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    for sign in [1, -1]:
                        hi = h[i] + sign * d
                        hj = h[j] - sign * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            old_i, old_j = h[i], h[j]
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = old_i, old_j
            if not improved:
                break
    return h, best_c


def binary_flip_search(h, n, deadline):
    """Try setting individual coordinates to 0 or 1, adjusting partner to maintain sum."""
    best_c = compute_c5(h, n)
    target = n / 2.0
    for _ in range(50):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for target_val in [0.0, 1.0]:
                diff = target_val - h[i]
                if abs(diff) < 1e-12:
                    continue
                h[i] = target_val
                adj_needed = -diff
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                else:
                    h[i] = target_val - (target_val - (target_val - diff))
                    h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c


def optimize_one_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with increasing alpha (annealing)
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        M = 5000.0
        x_opt = lbfgs_step(best_h, n, a, M, maxiter=400)
        c = compute_c5(x_opt, n)
        if c < best_c - 1e-13:
            best_c = c
            best_h = x_opt

    # Phase 2: Pair search
    if n <= 220:
        if time.time() < deadline - 8:
            best_h, best_c = pair_search(best_h, n, deadline - 5,
                                         [0.02, 0.01, 0.005, 0.002, 0.001], max_passes=20)
    elif n <= 400:
        if time.time() < deadline - 8:
            best_h, best_c = near_pair_search(best_h, n, deadline - 5,
                                              [0.02, 0.01, 0.005, 0.002], radius=35, max_passes=10)
    else:
        if time.time() < deadline - 8:
            best_h, best_c = near_pair_search(best_h, n, deadline - 5,
                                              [0.02, 0.01, 0.005], radius=25, max_passes=6)

    # Phase 3: High-alpha L-BFGS-B polish
    if time.time() < deadline - 5:
        x_opt = lbfgs_step(best_h, n, 5000000, 20000.0, maxiter=300)
        c = compute_c5(x_opt, n)
        if c < best_c:
            best_c = c
            best_h = x_opt

    # Phase 4: Fine pair search
    if time.time() < deadline - 2:
        if n <= 220:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.005, 0.002, 0.001, 0.0005], max_passes=15)
        elif n <= 400:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.005, 0.002, 0.001], radius=30, max_passes=8)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.005, 0.002], radius=25, max_passes=5)

    # Phase 5: Final L-BFGS-B
    if time.time() < deadline - 1:
        x_opt = lbfgs_step(best_h, n, 10000000, 50000.0, maxiter=200)
        c = compute_c5(x_opt, n)
        if c < best_c:
            best_c = c
            best_h = x_opt

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(200, 0.5)
        n_parent = 200

    best_h = None
    best_c = float('inf')

    x_parent = np.linspace(0, 2, n_parent)

    # Multi-resolution strategy
    # Primary: parent resolution
    # Secondary: slightly different resolutions to escape local optima
    resolutions_to_try = [n_parent]
    for n_try in [180, 220, 150, 250, 300, 400]:
        if n_try != n_parent and n_try not in resolutions_to_try:
            resolutions_to_try.append(n_try)

    # Time allocation: parent gets 40%, others split 45%, save 15% for restarts/polish
    total_avail = deadline - t0
    parent_time = total_avail * 0.40
    other_total = total_avail * 0.45
    n_others = max(1, len(resolutions_to_try) - 1)
    per_other = other_total / n_others

    for n in resolutions_to_try:
        elapsed = time.time() - t0
        if time.time() > deadline - 25:
            break

        if n == n_parent:
            alloc = parent_time
            h_start = h_parent.copy()
        else:
            remaining = deadline - time.time() - 25
            alloc = min(per_other, remaining * 0.25)
            if alloc < 20:
                continue
            h_start = np.interp(np.linspace(0, 2, n), x_parent, h_parent)

        h_start = project_feasible(h_start, n)
        h_opt, c_opt = optimize_one_n(h_start, n, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Random restarts from best solution
    if best_h is not None and time.time() < deadline - 35:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(2, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.02 + 0.08 * (r + 1) / max(1, n_restarts)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(35, (deadline - time.time() - 20) * 0.35)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_one_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Try a completely different structure: piecewise constant with few levels
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        # Try: h = [0]*a + [1]*b + [0.5]*(n-a-b) with sum constraint
        # sum = b + 0.5*(n-a-b) = n/2 => b + n/2 - a/2 - b/2 = n/2 => b/2 = a/2 => a = b
        # So a = b, n - 2a >= 0
        for a_frac in [0.2, 0.25, 0.3, 0.35, 0.4]:
            if time.time() > deadline - 25:
                break
            a = int(n * a_frac)
            b = a
            h_try = np.zeros(n)
            # Spread zeros and ones symmetrically
            h_try[:a//2] = 0.0
            h_try[a//2:a//2+b//2] = 1.0
            h_try[a//2+b//2:n-a//2-b//2] = 0.5
            h_try[n-a//2-b//2:n-a//2] = 1.0
            h_try[n-a//2:] = 0.0
            # Fix sum
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                h_opt, c_opt = optimize_one_n(h_try, n, time.time() + min(30, deadline - time.time() - 20))
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Final intensive polishing on best solution
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        # One more L-BFGS-B push
        x_opt = lbfgs_step(best_h, n, 20000000, 100000.0, maxiter=200)
        c = compute_c5(x_opt, n)
        if c < best_c:
            best_c = c
            best_h = x_opt

        if n <= 220:
            best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                         [0.002, 0.001, 0.0005, 0.0002, 0.0001], max_passes=20)
        elif n <= 400:
            best_h, best_c = near_pair_search(best_h, n, deadline - 0.5,
                                              [0.002, 0.001, 0.0005], radius=30, max_passes=10)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 0.5,
                                              [0.002, 0.001], radius=25, max_passes=6)

        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_parent, 0.5), n_parent)
        best_c = compute_c5(best_h, n_parent)

    return best_h, best_c, len(best_h)
