    steps = [0.002, 0.001, 0.0005, 0.0002]
    rng = np.random.default_rng(99999)
    n_pairs = 3
    radius = 40
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad_single(h, n, idx))
            if len(active) > 0:
                grad /= len(active)
            
            top_pos = np.argsort(-grad)[:min(radius * 2, n)]
            for _ in range(20):
                if time.time() > deadline:
                    return h, best_c
                # Pick n_pairs donors (high h, positive grad) and receivers
                donors = rng.choice(top_pos, size=min(n_pairs, len(top_pos)), replace=False)
                receivers = rng.integers(n, size=n_pairs)
                
                feasible = True
                orig_h = h.copy()
                for d in donors:
                    for r in receivers:
                        if d == r:
                            feasible = False
                            break
                    if not feasible:
                        break
                
                if not feasible:
                    continue
                
                # Try move
                for d in donors:
                    for r in receivers:
                        if d == r:
                            feasible = False
                            break
                        h[d] -= step / n_pairs
                        h[r] += step / n_pairs
                    
                if feasible:
                    if np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12):
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[:] = orig_h
                    else:
                        h[:] = orig_h
    return h, best_c

def slqp_polish(h, n, deadline):
    def obj_fn(x):
        return smooth_max_val(compute_c5_vec(x, n), 3000000)
    
    def grad_fn(x):
        c = compute_c5_vec(x, n)
        w = smooth_max_weights(c, 3000000)
        return combined_gradient(x, n, w)
    
    def sum_fn(x):
        return x.sum() - n / 2.0
    
    def sum_jac_fn(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_jac_fn}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    best_n = n_orig
    
    # Phase 1: Focused gradient descent at original resolution (100s)
    t1 = min(t0 + 100, deadline - 800)
    if t1 > time.time() + 5:
        h1, c1 = gradient_descent_focused(best_h, best_n, t1, alpha=3000, max_iter=100000, n_top=6)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: SQP optimization (130s)
    t2 = min(t0 + 230, deadline - 550)
    if t2 > time.time() + 10:
        h2, c2 = sqp_optimize(best_h, best_n, t2, max_iter=400, rho_init=0.03, n_active=5)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Multi-pair local search (70s)
    t3 = min(t0 + 300, deadline - 400)
    if t3 > time.time() + 5:
        h3, c3 = multi_pair_local_search(best_h, best_n, t3, radius=40)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Second SQP round (60s)
    t4 = min(t0 + 360, deadline - 350)
    if t4 > time.time() + 5:
        h4, c4 = sqp_optimize(best_h, best_n, t4, max_iter=300, rho_init=0.02, n_active=4)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Upsample to n=1000 (300s)
    n_high = 1000
    t5_start = max(time.time(), t0 + 370)
    t5_end = min(t5_start + 280, deadline - 130)
    if t5_end > t5_start + 40:
        h_high = upsample(best_h, best_n, n_high)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)
        
        t5a = min(t5_start + 90, t5_end - 150)
        if t5a > time.time() + 10:
            h_high, c_high = gradient_descent_focused(h_high, n_high, t5a, alpha=8000, max_iter=100000, n_top=8)
        
        t5b = min(t5_start + 180, t5_end - 60)
        if t5b > time.time() + 10:
            h_high, c_high = sqp_optimize(h_high, n_high, t5b, max_iter=250, rho_init=0.03, n_active=6)
        
        t5c = min(t5_start + 250, t5_end)
        if t5c > time.time() + 5:
            h_high, c_high = multi_pair_local_search(h_high, n_high, t5c, radius=50)
        
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            best_n = n_high
    
    # Phase 6: SLSQP polish (100s)
    t6 = min(time.time() + 100, deadline - 20)
    if t6 > time.time() + 15:
        h6, c6 = slqp_polish(best_h, best_n, t6)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Final local search squeeze
    if time.time() < deadline - 10:
        h7, c7 = multi_pair_local_search(best_h, best_n, deadline - 3, radius=20)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Final feasibility check
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
