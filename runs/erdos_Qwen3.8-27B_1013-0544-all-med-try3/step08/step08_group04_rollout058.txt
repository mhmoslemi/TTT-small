```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lbx(x, n, alpha, penalty):
    """Objective for L-BFGS-B: smooth max + penalty for sum constraint"""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    smooth = m + logsum / alpha
    sum_penalty = penalty * (x.sum() - n/2.0)**2
    return smooth + sum_penalty

def grad_lbx(x, n, alpha, penalty):
    """Gradient of obj_lbx"""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    # Compute gradient of smooth max
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for j in range(n):
        idx1 = j + lags  # j + l
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j - lags  # j - l
        v2 = (idx2 >= 0) & (idx2 < n)
        s1 = np.dot(w[v1], 1 - x[idx1[v1]])
        s2 = np.dot(w[v2], x[idx2[v2]])
        grad[j] = (s1 - s2) * dx
    
    # Add penalty gradient
    grad += 2 * penalty * (x.sum() - n/2.0)
    
    return grad

def lbfgs_optimize(h, n, alpha, penalty, maxiter, deadline):
    """Run L-BFGS-B optimization"""
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()
    try:
        res = minimize(
            obj_lbx, h, args=(n, alpha, penalty),
            jac=grad_lbx,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-12, 'maxls': 50,
                     'ftol': 1e-16}
        )
        return res.x
    except Exception:

def pair_search(h, n, deadline, deltas):
    """Local search: try swapping mass between pairs"""
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    # Try h[i]+=d, h[j]-=d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i]-=d, h[j]+=d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
```

```python
def grad_lbx(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    # Build index arrays
    # For each j in 0..n-1 and each lag l in -(n-1)..n-1:
    # idx1 = j + l (valid if 0 <= j+l < n)
    # idx2 = j - l (valid if 0 <= j-l < n)
    
    # First term: grad[j] += dx * sum_l w[n-1+l] * (1-x[j+l]) * [0<=j+l<n]
    # This is a "masked cross-correlation"
    
    # I'll use a 2D approach:
    # J = np.tile(np.arange(n), (2n-1, 1))  # n x (2n-1)
    # L = np.tile(np.arange(-(n-1), n), (n, 1))
    # IDX1 = J + L  # (n, 2n-1): j + l
    # IDX2 = J - L  # (n, 2n-1): j - l
    # V1 = (IDX1 >= 0) & (IDX1 < n)
    # V2 = (IDX2 >= 0) & (IDX2 < n)
    # FIRST = w[None, :] * (1 - x[np.where(V1, IDX1, 0)]) * V1  # (n, 2n-1)
    # SECOND = w[None, :] * x[np.where(V2, IDX2, 0)] * V2
    # grad = (FIRST.sum(axis=1) - SECOND.sum(axis=1)) * dx
    
    # This uses (n × (2n-1)) = 250 × 499 = 125K elements. Memory is fine.
    
    J, L = np.meshgrid(np.arange(n), np.arange(-(n-1), n), indexing='ij')
    IDX1 = J + L
    IDX2 = J - L
    V1 = (IDX1 >= 0) & (IDX1 < n)
    V2 = (IDX2 >= 0) & (IDX2 < n)
    
    # Safe indexing
    IDX1_safe = np.where(V1, IDX1, 0)
    IDX2_safe = np.where(V2, IDX2, 0)
    
    FIRST = (w[None, :] * (1 - x[IDX1_safe]) * V1).sum(axis=1)
    SECOND = (w[None, :] * x[IDX2_safe] * V2).sum(axis=1)
    grad = (FIRST - SECOND) * dx
    
    grad += 2 * penalty * (x.sum() - n/2.0)
    return grad
```

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
