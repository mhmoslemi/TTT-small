def corr_array(h,n):
    dx = 2.0 / float(n)
    m = 2*n - 1
    c = np.empty(m, dtype=float)
    ih = 1.0 - h
    c[n-1] = dx * float(np.dot(h, ih))
    for t in range(1, n):
        l = n - t
        c[n-1+t] = dx * float(np.dot(h[:l], ih[t:]))
        c[n-1-t] = dx * float(np.dot(h[t:], ih[:l]))
    return c
```

def softmax_weights(c, alpha):
    m = float(np.max(c))
    if not np.isfinite(m):
        z=np.zeros(c.shape); z[np.argmax(c)]=1; return z
    with np.errstate(under='ignore'):
        sh = alpha * (c - m)
    sh = np.where(sh > 0.0, 0.0, sh)

def weighted_gradient(c,w,h,n,center=False):
    dx=2.0/float(n)
    g=np.zeros(n,dtype=float)
    ih=1.0-h
    offset=n-1
    for idx in range(c.size):
       wt=w[idx]
       if wt == 0.0: continue
       t=idx-offset
       val=wt*dx
       if t >= 0:
          l=n-t
          if l <=0: continue
          g[:l] += val*ih[t:t+l]
          g[t:t+l] -= val*h[:l]

          l=n-d
          if l <=0: continue
          g[d:d+l] += val*ih[:l]
          g[:l] -= val*h[d:d+l]
    if center:
       g -= g.mean()
    return g

import numpy as np
import time
from scipy.optimize import minimize


def corr_array(h, n):
    dx = 2.0 / float(n)
    c = np.empty(2 * n - 1, dtype=float)
    ih = 1.0 - h
    c[n - 1] = dx * float(np.dot(h, ih))
    for t in range(1, n):
        l = n - t
        c[n - 1 + t] = dx * float(np.dot(h[:l], ih[t:]))
        c[n - 1 - t] = dx * float(np.dot(h[t:], ih[:l]))
    return c


def exact_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            tmp = np.zeros(n, dtype=float)
            tmp[:h.size] = h
            h = tmp
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n

    for _ in range(10000):
        s = h.sum()
        d = target - s
        if abs(d) <= 1e-14:
            break
        if d > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (d / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h += (d / tot) * h
        h = np.clip(h, 0.0, 1.0)

    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0.0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            rooms = 1.0 - h[idx]
            i = idx[int(np.argmax(rooms))]
            add = min(d, 1.0 - h[i])
            h[i] += add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            vals = h[idx]
            i = idx[int(np.argmax(vals))]
            sub = min(-d, h[i])
            h[i] -= sub
    return np.clip(h, 0.0, 1.0)


def symmetrize(h, n):
    h = (h + h[::-1]) * 0.5
    return project_feasible(h, n)


def resample(h, n_old, n):
    if n == n_old:
        return project_feasible(h.copy(), n)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n) + 0.5) * (2.0 / float(n))
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n)


def softmax_weights(c, alpha):
    m = float(np.max(c))
    if not np.isfinite(m):
        w = np.zeros(c.shape, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return w
    sh = alpha * (c - m)
    sh = np.minimum(sh, 0.0)
    sh = np.maximum(sh, -700.0)
    with np.errstate(under="ignore"):
        p = np.exp(sh)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(c.shape, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return w
    return p / s


def weighted_gradient(c, w, h, n, center=False):
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    ih = 1.0 - h
    offset = n - 1
    for idx in range(c.size):
        wt = w[idx]
        if wt == 0.0:
            continue
        t = idx - offset
        val = wt * dx
        if t >= 0:
            l = n - t
            if l <= 0:
                continue
            g[:l] += val * ih[t:t + l]
            g[t:t + l] -= val * h[:l]
        else:
            d = -t
            l = n - d
            if l <= 0:
                continue
            g[d:d + l] += val * ih[:l]
            g[:l] -= val * h[d:d + l]
    if center:
        g -= g.mean()
    return g


def lse_fun(x, n, alpha):
    c = corr_array(x, n)
    m = float(np.max(c))
    sh = alpha * (c - m)
    sh = np.minimum(sh, 0.0)
    sh = np.maximum(sh, -700.0)
    with np.errstate(under="ignore"):
        s = float(np.sum(np.exp(sh)))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + np.log(s) / alpha)


def lse_jac(x, n, alpha):
    c = corr_array(x, n)
    w = softmax_weights(c, alpha)
    return weighted_gradient(c, w, x, n, False)


def sum_con(x, n):
    return float(np.sum(x)) - 0.5 * n


def sum_con_jac(x, n):
    return np.ones_like(x)


def penalty_fun(x, n, alpha, mu):
    perr = float(np.sum(x)) - 0.5 * n
    return lse_fun(x, n, alpha) + mu * perr * perr


def penalty_jac(x, n, alpha, mu):
    g = lse_jac(x, n, alpha)
    perr = float(np.sum(x)) - 0.5 * n
    g += 2.0 * mu * perr
    return g


def run_slsqp(x, n, alpha, maxiter):
    x = project_feasible(x, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    try:
        res = minimize(
            lse_fun, x, args=(n, alpha), jac=lse_jac, method="SLSQP",
            bounds=bounds, constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-11, "disp": False}
        )
        return project_feasible(res.x, n)
    except Exception:
        return x


def run_lbfgsb(x, n, alpha, mu, maxiter):
    x = project_feasible(x, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            penalty_fun, x, args=(n, alpha, mu), jac=penalty_jac,
            method="L-BFGS-B", bounds=bounds,
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-12,
                "gtol": 1e-8,
                "maxfun": int(2 * maxiter) + 10,
                "maxls": 20
            }
        )
        return project_feasible(res.x, n)
    except Exception:
        return x


def pair_trial(h, c, n, tmp, buf, hh, i, j, delta):
    dx = 2.0 / float(n)
    dh = delta * dx
    d2 = delta * delta * dx

    tmp.fill(0.0)
    np.multiply(h, dh, out=hh)
    hhr = hh[::-1]

    a = n - 1 - i
    b = 2 * n - 1 - i
    tmp[a:b] += dh
    tmp[a:b] -= hh
    a2 = i
    b2 = i + n
    tmp[a2:b2] -= hhr

    a = n - 1 - j
    b = 2 * n - 1 - j
    tmp[a:b] -= dh
    tmp[a:b] += hh
    a2 = j
    b2 = j + n
    tmp[a2:b2] += hhr

    tmp[n - 1] -= 2.0 * d2
    d = abs(i - j)
    tmp[n - 1 + d] += d2
    tmp[n - 1 - d] += d2

    np.add(c, tmp, out=buf)
    return float(buf.max())


def fast_local_search(h, n, rng, deadline, deltas=None):
    h = project_feasible(h.astype(float).copy(), n)
    c = corr_array(h, n)
    best = float(c.max())
    if time.time() > deadline - 2.0:
        return h, c, best

    if deltas is None:
        deltas = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    m = 2 * n - 1
    tmp = np.zeros(m, dtype=float)
    buf = np.empty(m, dtype=float)
    hh = np.empty(n, dtype=float)

    alpha_g = 5000.0
    K = int(min(50, max(12, n // 4)))
    rnd_pairs = int(min(700, max(300, 2 * n)))

    tol = 1e-13

    for delta in deltas:
        if time.time() > deadline - 2.0:
            break
        for _pass in range(3):
            if time.time() > deadline - 2.0:
                break

            w = softmax_weights(c, alpha_g)
            g = weighted_gradient(c, w, h, n, True)
            order = np.argsort(g)
            low = order[:K]
            high = order[-K:][::-1]

            pass_improved = 0
            stop = False

            for p in low:
                if stop:
                    break
                if time.time() > deadline - 2.0:
                    stop = True
                    break
                p = int(p)
                if h[p] + delta > 1.0:
                    continue
                for q in high:
                    q = int(q)
                    if q == p:
                        continue
                    if h[q] - delta < 0.0:
                        continue
                    newmax = pair_trial(h, c, n, tmp, buf, hh, p, q, delta)
                    if newmax < best - tol:
                        c[:] = buf
                        h[p] += delta
                        h[q] -= delta
                        best = newmax
                        pass_improved += 1
                        if pass_improved % 25 == 0:
                            c = corr_array(h, n)
                            best = float(c.max())
                            tmp = np.zeros(m, dtype=float)
                    if time.time() > deadline - 2.0:
                        stop = True
                        break

            if time.time() < deadline - 2.0:
                for _ in range(rnd_pairs):
                    if time.time() > deadline - 2.0:
                        break
                    p = int(rng.randint(0, n))
                    q = int(rng.randint(0, n))
                    if p == q:
                        continue
                    if g[p] <= g[q]:
                        inc, dec = p, q
                    else:
                        inc, dec = q, p

                    tried = False
                    if 0.0 <= h[dec] - delta and h[inc] + delta <= 1.0:
                        newmax = pair_trial(h, c, n, tmp, buf, hh, inc, dec, delta)
                        if newmax < best - tol:
                            c[:] = buf
                            h[inc] += delta
                            h[dec] -= delta
                            best = newmax
                            pass_improved += 1
                            tried = True
                            if pass_improved % 25 == 0:
                                c = corr_array(h, n)
                                best = float(c.max())
                                tmp = np.zeros(m, dtype=float)

                    if not tried and 0.0 <= h[inc] - delta and h[dec] + delta <= 1.0:
                        newmax = pair_trial(h, c, n, tmp, buf, hh, dec, inc, delta)
                        if newmax < best - tol:
                            c[:] = buf
                            h[dec] += delta
                            h[inc] -= delta
                            best = newmax
                            pass_improved += 1
                            if pass_improved % 25 == 0:
                                c = corr_array(h, n)
                                best = float(c.max())
                                tmp = np.zeros(m, dtype=float)

            c = corr_array(h, n)
            best = float(c.max())
            if pass_improved == 0:
                break

    c = corr_array(h, n)
    best = float(c.max())
    return h, c, best


def random_sine_start(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    a = int(rng.randint(1, 5))
    b = int(rng.randint(2, 8))
    p1 = float(rng.uniform(0.0, 2.0 * np.pi))
    p2 = float(rng.uniform(0.0, 2.0 * np.pi))
    y = 0.5 + 0.45 * np.sin(a * np.pi * x + p1) + 0.25 * np.sin(b * np.pi * x + p2)
    h = np.clip(y, 0.0, 1.0)
    return project_feasible(h, n)


def random_binary_start(n, rng):
    h = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = rng.choice(n, k, replace=False)
        h[idx] = 1.0
    return project_feasible(h, n)


def random_block_start(n, rng):
    h = np.zeros(n, dtype=float)
    need = n // 2
    i = 0
    cur = 1 if rng.randint(0, 2) == 1 else 0
    max_run = max(2, n // 12)

    while i < n:
        run = int(rng.randint(1, max_run))
        if i + run > n:
            run = n - i
        if cur == 1:
            take = min(run, need)
            if take > 0:
                h[i:i + take] = 1.0
                need -= take
        i += run
        cur = 1 - cur

    if need > 0:
        idx = np.where(h < 1.0)[0]
        if idx.size >= need:
            take = idx[:need]
            h[take] = 1.0
        else:
            h[:] = 1.0
    return project_feasible(h, n)


def generate_start(ref, ref_n, n, rng, mode):
    if n == ref_n:
        base = ref.copy()
    else:
        base = resample(ref, ref_n, n)

    if mode == 0:
        h = base + rng.normal(0.0, 0.05, n)
    elif mode == 1:
        h = base + rng.normal(0.0, 0.12, n)
    elif mode == 2:
        h = random_binary_start(n, rng)
    elif mode == 3:
        h = random_sine_start(n, rng)
    elif mode == 4:
        h = symmetrize(base + rng.normal(0.0, 0.08, n), n)
    else:
        h = random_block_start(n, rng)
    return project_feasible(h, n)


def make_candidates(n0):
    raw = [
        n0 - 96, n0 - 64, n0 - 32,
        n0 + 32, n0 + 64, n0 + 96, n0 + 160,
        100, 120, 160, 200, 240, 300, 360, 400
    ]
    out = []
    seen = set()
    for x in raw:
        x = int(x)
        if x % 2 != 0:
            x += 1
        if x < 60 or x > 520:
            continue
        if x == n0:
            continue
        if x in seen:
            continue
        seen.add(x)
        out.append(x)
    return out


def optimize_one(h, n, deadline, rng, quick=False):
    h = project_feasible(h.astype(float).copy(), n)
    best = exact_c5(h, n)

    if n <= 120:
        lb_params = [
            (100.0, 1e3, 400),
            (2000.0, 1e3, 400),
            (20000.0, 1e4, 400),
            (100000.0, 1e4, 300)
        ]
    elif n <= 240:
        lb_params = [
            (100.0, 1e3, 400),
            (5000.0, 1e3, 350),
            (50000.0, 1e4, 300)
        ]
    else:
        lb_params = [
            (100.0, 1e3, 200),
            (20000.0, 1e3, 200),
            (200000.0, 1e4, 150)
        ]

    if quick:
        lb_params = lb_params[-2:] if len(lb_params) > 2 else lb_params

    for al, mu, base in lb_params:
        rem = deadline - time.time()
        if rem < 25.0:
            break
        if rem < 60.0:
            it = max(30, int(base // 3))
        else:
            it = base
        try:
            hx = run_lbfgsb(h, n, al, mu, it)
            val = exact_c5(hx, n)
            if val < best - 1e-12:
                best = val
                h = project_feasible(hx, n)
        except Exception:
            pass

    if n <= 140:
        sl_params = [
            (2000.0, 350),
            (20000.0, 300),
            (200000.0, 200)
        ]
    elif n <= 260:
        sl_params = [
            (5000.0, 250),
            (50000.0, 200),
            (300000.0, 120)
        ]
    else:
        sl_params = [
            (20000.0, 100),
            (300000.0, 80)
        ]

    if quick:
        sl_params = sl_params[-1:] if len(sl_params) > 1 else sl_params

    for al, base in sl_params:
        rem = deadline - time.time()
        if rem < 20.0:
            break
        if rem < 60.0:
            it = max(30, int(base // 3))
        elif rem < 120.0:
            it = max(40, int(2 * base // 3))
        else:
            it = base
        try:
            hx = run_slsqp(h, n, al, it)
            val = exact_c5(hx, n)
            if val < best - 1e-12:
                best = val
                h = project_feasible(hx, n)
        except Exception:
            pass

    if time.time() < deadline - 12.0:
        try:
            h, c, best = fast_local_search(h, n, rng, deadline - 2.0)
        except Exception:
            pass

    if time.time() < deadline - 5.0:
        try:
            hs = symmetrize(h, n)
            vs = exact_c5(hs, n)
            if vs < best - 1e-12:
                h = hs
                best = vs
        except Exception:
            pass

    h = project_feasible(h, n)
    val = exact_c5(h, n)
    if val < best:
        best = val
    return h, best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + max(10.0, total - 5.0)
    rng = np.random.RandomState(seed)

    h0 = None
    n0 = 200
    try:
        parent = np.array(initial_h_values, dtype=float).reshape(-1).copy()
        n0 = int(parent.size)
        if n0 < 4:
            raise ValueError("bad parent")
    except Exception:
        parent = None

    if parent is not None:
        h0 = project_feasible(parent, n0)
        if n0 > 600:
            h0 = resample(h0, n0, 400)
            n0 = 400
    else:
        n0 = 200
        h1 = random_sine_start(n0, rng)
        h2 = random_binary_start(n0, rng)
        h3 = np.full(n0, 0.5, dtype=float)
        best0 = exact_c5(h1, n0)
        best_h0 = h1
        v = exact_c5(h2, n0)
        if v < best0:
            best0 = v
            best_h0 = h2
        v = exact_c5(h3, n0)
        if v < best0:
            best0 = v
            best_h0 = h3
        h0 = project_feasible(best_h0, n0)

    c0 = exact_c5(h0, n0)
    try:
        hs0 = symmetrize(h0, n0)
        cs0 = exact_c5(hs0, n0)
    except Exception:
        hs0 = h0.copy()
        cs0 = c0

    if cs0 < c0 - 1e-12:
        best_h = hs0
        best_c = cs0
    else:
        best_h = h0
        best_c = c0

    source_h = h0.copy()
    source_n = int(n0)

    if deadline - time.time() > 45.0:
        alloc = min(500.0, (deadline - time.time()) * 0.5)
        try:
            hopt, copt = optimize_one(best_h, int(best_h.size), time.time() + alloc, rng, quick=False)
            if copt < best_c - 1e-12:
                best_h = hopt
                best_c = float(copt)
                source_h = hopt.copy()
                source_n = int(hopt.size)
        except Exception:
            pass

    if deadline - time.time() > 80.0:
        try:
            hs = symmetrize(best_h, int(best_h.size))
            cs = exact_c5(hs, int(best_h.size))
            if cs < best_c - 1e-12:
                best_h = hs
                best_c = float(cs)
        except Exception:
            hs = best_h.copy()
        alloc = min(150.0, (deadline - time.time()) * 0.25)
        if alloc > 20.0:
            try:
                hopt, copt = optimize_one(hs, int(hs.size), time.time() + alloc, rng, quick=True)
                if copt < best_c - 1e-12:
                    best_h = hopt
                    best_c = float(copt)
                    source_h = hopt.copy()
                    source_n = int(hopt.size)
            except Exception:
                pass

    cand_list = make_candidates(source_n)
    for n_try in cand_list:
        if deadline - time.time() < 60.0:
            break
        remaining = deadline - time.time()
        alloc = min(110.0, remaining * 0.25)
        if alloc < 18.0:
            break
        try:
            hinterp = resample(source_h, source_n, n_try)
            quick = (n_try > 260 or n_try < 80)
            hopt, copt = optimize_one(hinterp, n_try, time.time() + alloc, rng, quick=quick)
            if copt < best_c - 1e-12:
                best_h = hopt
                best_c = float(copt)
                if n_try >= source_n:
                    source_h = hopt.copy()
                    source_n = int(hopt.size)
        except Exception:
            pass

    if deadline - time.time() > 80.0:
        restart_n = int(best_h.size) if int(best_h.size) <= 240 else 200
        n_rest = int((deadline - time.time() - 40.0) / 50.0)
        n_rest = max(1, min(5, n_rest))
        for r in range(n_rest):
            if time.time() > deadline - 60.0:
                break
            remaining = deadline - time.time() - 40.0
            alloc = min(100.0, remaining / float(n_rest - r))
            if alloc < 18.0:
                break
            try:
                hstart = generate_start(best_h, int(best_h.size), restart_n, rng, r % 6)
                hopt, copt = optimize_one(hstart, restart_n, time.time() + alloc, rng, quick=True)
                if copt < best_c - 1e-12:
                    best_h = hopt
                    best_c = float(copt)
                    if restart_n >= source_n:
                        source_h = hopt.copy()
                        source_n = int(hopt.size)
            except Exception:
                pass

    if deadline - time.time() > 20.0:
        try:
            h2, c2, b2 = fast_local_search(
                best_h, int(best_h.size), rng, deadline - 2.0,
                deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001]
            )
            if b2 < best_c - 1e-12:
                best_h = h2
                best_c = float(b2)
        except Exception:
            pass

    if deadline - time.time() > 3.0:
        try:
            hs = symmetrize(best_h, int(best_h.size))
            vs = exact_c5(hs, int(best_h.size))
            if vs < best_c - 1e-12:
                best_h = hs
                best_c = float(vs)
        except Exception:
            pass

    best_h = project_feasible(best_h, int(best_h.size))
    best_c = exact_c5(best_h, int(best_h.size))
    return best_h, float(best_c), int(best_h.size)
