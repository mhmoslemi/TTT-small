    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 25:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(topK_obj, best_h, jac=topK_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def grad_guided_search(h, n, deadline, step_sizes=None, max_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    steps_done = 0
    for step in step_sizes:
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-10) & (best_h > 1e-10)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:20]:
                gi = grad[i]
                for j in neg_idx[:20]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=50, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def binary_coord_descent(h, n, deadline, max_rounds=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones_set = set(np.where(best_h > 0.5)[0].tolist())
    zeros_set = set(np.where(best_h < 0.5)[0].tolist())
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones_list = list(ones_set)
        zeros_list = list(zeros_set)
        np.random.shuffle(ones_list)
        np.random.shuffle(zeros_list)
        for i in ones_list:
            if time.time() > deadline:
                break
            for j in zeros_list[:len(zeros_list)//4]:
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones_set.remove(i)
                    ones_set.add(j)
                    zeros_set.remove(j)
                    zeros_set.add(i)
                    improved = True
                    break
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
                    best_h = project_feasible(best_h, n)
        if not improved:
            break
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 2 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        rem = max(10, int(deadline - time.time()))
        maxit = min(maxiter, max(200, rem * 10))
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def multi_active_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (4, 100000, 5),
        (6, 500000, 5),
        (8, 2000000, 4),
        (10, 5000000, 4),
        (16, 10000000, 3),
        (24, 20000000, 3),
    ]
    for top_k, alpha, rounds in configs:
        if time.time() > deadline - 15:
            break
        for rnd in range(rounds):
            if time.time() > deadline - 10:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.0005 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                break
            maxit = max(200, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(topK_obj, best_h, jac=topK_grad,
                              args=(n, active, alpha, 0.0005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Generate alternative binary starts to compare
    alt_starts = []
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts.append(('beatty_golden', make_beatty_start(n_orig, golden / 2.0)))
    alt_starts.append(('beatty_sqrt2', make_beatty_start(n_orig, np.sqrt(2.0) / 2.0)))
    alt_starts.append(('random1', make_random_binary(n_orig, rng)))
    alt_starts.append(('random2', make_random_binary(n_orig, rng)))

    for name, h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS warmup at n_orig
    p1_end = t0 + 250
    h1, c1 = phase_lbfgs(best_h, best_n, p1_end,
        alphas=[20, 50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.01, maxit_list=[1000, 1500, 2000, 2000, 2500, 3000, 3500, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Multi-active set optimization
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 200, deadline - 300)
        h2, c2 = multi_active_optimize(best_h, best_n, p2_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Gradient-guided coordinate descent
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 180, deadline - 200)
        h3, c3 = grad_guided_search(best_h, best_n, p3_end,
            step_sizes=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_steps=100000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Fine pair search
    if time.time() < deadline - 180:
        p4_end = min(time.time() + 130, deadline - 150)
        h4, c4 = fine_pair_search(best_h, best_n, p4_end, radius=40,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Active set + grad-guided again (refined)
    if time.time() < deadline - 120:
        p5_end = min(time.time() + 60, deadline - 100)
        h5, c5 = multi_active_optimize(best_h, best_n, p5_end)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        if time.time() < deadline - 50:
            p5b_end = min(time.time() + 50, deadline - 45)
            h5b, c5b = grad_guided_search(best_h, best_n, p5b_end,
                step_sizes=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                max_steps=50000)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b

    # Phase 6: Very fine pair search
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 45, deadline - 40)
        h6, c6 = fine_pair_search(best_h, best_n, p6_end, radius=30,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: SLSQP polish
    if time.time() < deadline - 30:
        for alpha in [2000000, 10000000, 50000000]:
            if time.time() > deadline - 20:
                break
            h7, c7 = run_slqp(best_h, best_n, deadline - 5, alpha=alpha, maxiter=800)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Final very fine adjustments
    if time.time() < deadline - 5:
        h8, c8 = fine_pair_search(best_h, best_n, deadline - 2, radius=20,
            deltas=[0.00001, 0.000005, 0.000002])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
