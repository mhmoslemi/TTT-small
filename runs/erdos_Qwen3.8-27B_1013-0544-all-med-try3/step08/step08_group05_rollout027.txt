A better continuous approach: **subgradient methods**. At each iteration:
1. Find k* = argmax_k g_k(x) (the worst shift)
2. Compute the subgradient of g_{k*} w.r.t. x
3. Take a step in the negative subgradient direction
4. Project onto the constraint set

∂g_k/∂x[i] = (1-x[i+k]) - x[i-k] (for valid indices)

This is a bilinear function, so the subgradient is just the gradient of the specific g_k.

def subgradient_search(h, n, deadline, lr=0.01, max_iters=100000):
    h = h.copy().astype(float)
    # Project to feasible set
    h = np.clip(h, 0, 1)
    diff = n/2 - h.sum()
    h += diff / n  # simple projection (not exact but close)
    h = np.clip(h, 0, 1)
    # Exact projection
    for _ in range(100):
        diff = n/2 - h.sum()
        if abs(diff) < 1e-12: break
        h += diff / n
        h = np.clip(h, 0, 1)
    
    for it in range(max_iters):
        if time.time() > deadline: break
        corr = np.correlate(h, 1-h, mode='full')
        k_best = np.argmax(corr)
        k_shift = k_best - (n-1)  # actual shift
        
        # Subgradient
        m = np.arange(-(n-1), n)
        sg = np.zeros(n)
        # ∂g_k/∂x[p] = (1-x[p+k]) - x[p-k] (for valid indices)
        pk = p + k_shift  # for each p
        # Need to sum over valid p
        
        # Actually: g_k = sum_p x[p] * (1-x[p+k]) for valid p, p+k
        # ∂/∂x[p]: (1-x[p+k]) + x[p] * (-∂x[p+k]/∂x[p])
        # If p+k ≠ p (i.e., k≠0): ∂/∂x[p] = (1-x[p+k]) - x[p-k] (the second term is from the p' where p'+k = p, i.e., p' = p-k)
        # If k=0: ∂/∂x[p] = (1-2x[p])
        
        if k_shift == 0:
            sg = 1 - 2*h
        else:
            sg = np.zeros(n)
            for p in range(n):
                grad = 0.0
                pk = p + k_shift
                if 0 <= pk < n:
                    grad += (1 - h[pk])
                pmk = p - k_shift
                if 0 <= pmk < n:
                    grad -= h[pmk]
                sg[p] = grad
        
        # Projected gradient step
        h_new = h - lr * sg
        h_new = np.clip(h_new, 0, 1)
        # Project to sum constraint
        diff = n/2 - h_new.sum()
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        
        # Simple projection (iterate)
        for _ in range(50):
            diff = n/2 - h_new.sum()
            if abs(diff) < 1e-14: break
            if diff > 0:
                mask = h_new < 1
                h_new[mask] += diff / mask.sum()
            else:
                mask = h_new > 0
                h_new[mask] += diff / mask.sum()
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
        lr *= 0.999  # slow decay
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            s = mask.sum()
            if s < 1e-14:
                break
            h[mask] += diff / s
        else:
            mask = h > 0.0
            s = mask.sum()
            if s < 1e-14:
                break
            h[mask] += diff / s
    return np.clip(h, 0.0, 1.0)

def swap_delta_binary(h, i, j, n):
    m = np.arange(-(n - 1), n)
    im = i + m
    im_v = (im >= 0) & (im < n)
    h_im = np.zeros(2 * n - 1)
    h_im[im_v] = h[im[im_v]]
    imn = i - m
    imn_v = (imn >= 0) & (imn < n)
    h_imn = np.zeros(2 * n - 1)
    h_imn[imn_v] = h[imn[imn_v]]
    jm = j + m
    jm_v = (jm >= 0) & (jm < n)
    h_jm = np.zeros(2 * n - 1)
    h_jm[jm_v] = h[jm[jm_v]]
    jmn = j - m
    jmn_v = (jmn >= 0) & (jmn < n)
    h_jmn = np.zeros(2 * n - 1)
    h_jmn[jmn_v] = h[jmn[jmn_v]]
    return h_im + h_imn - h_jm - h_jmn

def binary_local_search(h, n, deadline, max_rounds=50, seed=42):
    rng = np.random.RandomState(seed)
    h = make_binary(h, n)
    corr = compute_corr(h).astype(np.float64)
    max_c = corr.max()
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    for round_idx in range(max_rounds):
        if time.time() > deadline:
            break
        rng.shuffle(ones_list)
        rng.shuffle(zeros_list)
        improved = False
        for i in zeros_list:
            if improved:
                break
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = swap_delta_binary(h, i, j, n)
                new_corr_max = np.max(corr + delta)
                if new_corr_max < max_c - 1e-12:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    max_c = new_corr_max
                    if i in zeros_list:
                        zeros_list.remove(i)
                    zeros_list.append(j)
                    if j in ones_list:
                        ones_list.remove(j)
                    ones_list.append(i)
                    improved = True
                    break
        if not improved:
            break
    return h, float(max_c * 2.0 / n)

def binary_sa_search(h, n, deadline, T0=2.0, T_end=0.1, max_iters=200000, seed=42):
    rng = np.random.RandomState(seed)
    h = make_binary(h, n)
    corr = compute_corr(h).astype(np.float64)
    max_c = corr.max()
    best_h = h.copy()
    best_max_c = max_c
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    ones_arr = np.array(list(ones_set))
    zeros_arr = np.array(list(zeros_set))
    it = 0
    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max_iters)
        T = max(T, T_end * 0.01)
        if len(zeros_arr) == 0 or len(ones_arr) == 0:
            break
        i = zeros_arr[rng.randint(len(zeros_arr))]
        j = ones_arr[rng.randint(len(ones_arr))]
        delta = swap_delta_binary(h, i, j, n)
        new_corr_max = np.max(corr + delta)
        d = new_corr_max - max_c
        if d < 0 or rng.rand() < np.exp(-d / max(T, 1e-8)):
            h[i], h[j] = 1.0, 0.0
            corr = corr + delta
            max_c = new_corr_max
            ones_set.discard(j)
            ones_set.add(i)
            zeros_set.discard(i)
            zeros_set.add(j)
            ones_arr = np.array(list(ones_set))
            zeros_arr = np.array(list(zeros_set))
            if max_c < best_max_c:
                best_max_c = max_c
                best_h = h.copy()
    return best_h, float(best_max_c * 2.0 / n)

def subgradient_optimize(h_init, n, deadline, lr0=0.02, max_iters=500000, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(np.float64)
    h = project_feasible(h, n)
    lr = lr0
    best_h = h.copy()
    best_val = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = np.argmax(corr)
        k_shift = k_idx - (n - 1)
        sg = np.zeros(n)
        if k_shift == 0:
            sg = 1.0 - 2.0 * h
        else:
            for p in range(n):
                g = 0.0
                pk = p + k_shift
                if 0 <= pk < n:
                    g += (1.0 - h[pk])
                pmk = p - k_shift
                if 0 <= pmk < n:
                    g -= h[pmk]
                sg[p] = g
        h = h - lr * sg
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)
        lr *= 0.9995
        if it % 500 == 0:
            val = compute_c5(h, n)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    val = compute_c5(h, n)
    if val < best_val:
        best_val = val
        best_h = h.copy()
    return best_h, best_val

def slsqp_smooth(h_init, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def interpolate_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)
