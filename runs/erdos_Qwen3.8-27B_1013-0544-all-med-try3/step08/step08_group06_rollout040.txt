def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def compute_c5_array(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]

    h = np.zeros(n)
    if pattern == 'alternating':
        h[::2] = 1.0
        # Ensure exactly n/2 ones
        if h.sum() > n / 2:
            # Remove excess
            excess = h.sum() - n / 2
            ones = np.where(h == 1)[0]
            np.random.shuffle(ones)
            h[ones[:int(excess)]] = 0
        elif h.sum() < n / 2:
            deficit = n / 2 - h.sum()
            zeros = np.where(h == 0)[0]
            np.random.shuffle(zeros)
            h[zeros[:int(deficit)]] = 1
    elif pattern == 'block':
        # Blocks of ones and zeros
        k = n // 4
        for i in range(0, n, 2 * k):
            h[i:i+k] = 1.0
        if h.sum() != n / 2:
            # Fix
            order = np.argsort(-h)
            h = np.zeros(n)
            h[order[:n//2]] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)
    deadline = t0 + budget_s - 3
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200
    
    # Phase 1: Optimize the initial construction (continuous)
    if h_init is not None:
        alloc = min(120, (deadline - t0) * 0.15)
        if alloc > 20:
            h_cont, c_cont = optimize_n_continuous(h_init, n_init, t0 + alloc, rng)
            if c_cont < best_c:
                best_c = c_cont
                best_h = h_cont
                best_n = n_init
    
    # Phase 2: Binary optimization on initial construction
    if h_init is not None:
        alloc = min(150, (deadline - t0) * 0.25)
        if alloc > 30:
            start_time = time.time()
            h_bin, c_bin = optimize_n_binary(h_init, n_init, start_time + alloc, rng)
            if c_bin < best_c:
                best_c = c_bin
                best_h = h_bin
                best_n = n_init
    
    # Phase 3: Try different n values with binary optimization
    n_candidates = [100, 120, 150, 200, 250, 300, 400, 500]
    
    for n_try in n_candidates:
        if time.time() > deadline - 30:
            break
        if n_try == n_init and h_init is not None:
            continue
        
        remaining = deadline - time.time() - 15
        alloc = min(80, remaining * 0.15)
        if alloc < 20:
            break
        
        # Create initial pattern
        if h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
        else:
            h_interp = make_structured_init(n_try, 'alternating')
        
        h_interp = project_feasible(h_interp, n_try)
        
        start_time = time.time()
        h_bin, c_bin = optimize_n_binary(h_interp, n_try, start_time + alloc, rng)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
            best_n = n_try
    
    # Phase 4: Random restarts on best n
    if best_h is not None and time.time() < deadline - 30:
        n_best = best_n
        remaining = deadline - time.time() - 10
        n_restarts = max(2, int(remaining / 40))
        
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            alloc = min(50, (deadline - time.time() - 15) * 0.35)
            if alloc < 15:
                break
            
            # Create perturbed binary solution
            if np.all((best_h == 0) | (best_h == 1)):
                h_start = best_h.copy()
                # Randomly swap a few 1s and 0s
                n_swaps = max(1, n_best // 20)
                for _ in range(n_swaps):
                    ones = np.where(h_start == 1)[0]
                    zeros = np.where(h_start == 0)[0]
                    i = ones[np.random.randint(len(ones))]
                    j = zeros[np.random.randint(len(zeros))]
                    h_start[i], h_start[j] = h_start[j], h_start[i]
            else:
                h_start = best_h + rng.normal(0, 0.05, n_best)
                h_start = project_feasible(h_start, n_best)
            
            h_opt, c_opt = optimize_n_binary(h_start, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                best_n = n_best
    
    # Phase 5: Final refinement
    if best_h is not None and time.time() < deadline - 5:
        n_final = best_n
        # If binary, do final local search
        if np.all((best_h == 0) | (best_h == 1)):
            best_h, best_c = combinatorial_local_search(best_h, n_final, deadline - 2, strategy='first')
        else:
            # Continuous refinement
            ht = slsqp_optimize(best_h, n_final, 3000000, maxiter=3000, ftol=1e-16)
            c = compute_c5(ht, n_final)
            if c < best_c:
                best_h = project_feasible(ht, n_final)
                best_c = compute_c5(best_h, n_final)
            if n_final <= 150:
                best_h, best_c = pair_search(best_h, n_final, deadline - 2, [0.005, 0.002, 0.001, 0.0005])
    
    # Final check
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
