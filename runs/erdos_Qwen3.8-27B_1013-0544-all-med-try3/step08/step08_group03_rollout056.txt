                        old_i, old_j = h[i], h[j]
                        h[i] = old_i + move
                        h[j] = old_j - move
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] = old_i
                            h[j] = old_j
            if not improved:
                break
    return h, best_c

def upscale_interpolate(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, n_new)

def make_block_pattern(n, rng, block_sizes=None):
    h = np.zeros(n)
    target = n / 2.0
    pos = 0
    while pos < n:
        if block_sizes is None:
            bsz = rng.randint(3, 12)
        else:
            bsz = block_sizes[rng.randint(len(block_sizes))]
        bsz = min(bsz, n - pos)
        if rng.random() < 0.5:
            h[pos:pos+bsz] = 1.0
        pos += bsz
    h = project_feasible(h, n)
    return h

def smooth_pattern(h, n, kernel_size=3):
    kernel = np.ones(kernel_size) / kernel_size
    h_s = np.convolve(h, kernel, mode='same')
    return project_feasible(h_s, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS on initial resolution (quick pass)
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1)

    # Phase 2: Try higher resolution (2x)
    n_hi = n_init * 2
    if time.time() < deadline - 350:
        h_hi = upscale_interpolate(best_h, n_init, n_hi)
        alloc = min(180, deadline - time.time() - 350)
        if alloc > 60:
            h_hi, c_hi = lbfgs_phase(h_hi, n_hi, time.time() + alloc)
            if c_hi < best_c:
                best_c = c_hi
                best_h = h_hi
                n_init = n_hi

    # Phase 3: Gradient descent for further improvement
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 200)
        h3, c3 = gradient_step(best_h, n_init, t3, alpha=500.0, lr=0.008)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: SLSQP refinement
    if time.time() < deadline - 120:
        for a in [200000, 1000000, 5000000]:
            if time.time() > deadline - 80:
                break
            t4 = min(time.time() + 40, deadline - 80)
            h4, c4 = slsqp_phase(best_h, n_init, t4, alpha=a, maxiter=1500)
            if c4 < best_c:
                best_c = c4
                best_h = h4

    # Phase 5: Gradient-guided pair search
    if time.time() < deadline - 80:
        t5 = min(time.time() + 80, deadline - 80)
        best_h, best_c = gradient_pair_search(best_h, n_init, t5, step_size=0.002)

    # Phase 6: Standard pair search (main local optimization)
    if time.time() < deadline - 30:
        t6 = min(time.time() + 200, deadline - 10)
        best_h, best_c = pair_search(best_h, n_init, t6,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=80)

    # Phase 7: Try a different resolution if time permits
    if time.time() < deadline - 80:
        for n_try in [n_init + 150, n_init - 100, n_init * 3 // 2]:
            if time.time() > deadline - 50:
                break
            if n_try < 50:
                continue
            h_try = upscale_interpolate(best_h, n_init, n_try)
            alloc = min(60, deadline - time.time() - 30)
            if alloc < 30:
                break
            h_t, c_t = lbfgs_phase(h_try, n_try, time.time() + alloc)
            if time.time() < deadline - 15:
                h_t, c_t = pair_search(h_t, n_try, time.time() + min(20, deadline - 15),
                                       [0.005, 0.002, 0.001, 0.0005], radius=50)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n_init = n_try

    # Phase 8: Final L-BFGS + pair search polish
    if time.time() < deadline - 8:
        nf = len(best_h)
        if time.time() < deadline - 5:
            h_f, c_f = lbfgs_phase(best_h, nf, deadline - 5)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, len(best_h), deadline - 1,
                                     [0.001, 0.0005, 0.0002, 0.0001],
                                     radius=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
