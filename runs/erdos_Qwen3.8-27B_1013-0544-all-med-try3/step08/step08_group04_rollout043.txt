def compute_gradient_smooth_vec(h, n, alpha, top_k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    nc = len(c)  # 2n-1
    
    if top_k is not None and top_k < nc:
        idx = np.argsort(c)[-top_k:]
        c_active = c[idx]
        w = np.zeros(nc)
        w[idx] = softmax(alpha * c_active)
    else:
        w = softmax(alpha * c)
    
    # w[k] corresponds to lag m = k - (n-1)
    # For each j, grad[j] = dx * (sum_m w[m] * (1-h[j-m]) * 1{valid} - sum_m w[m] * h[j+m] * 1{valid})
    
    # First term: for each lag m, it contributes w[m] * (1-h[j-m]) to positions j where 0<=j-m<n
    # i.e., j in [m, m+n-1] (clipped to [0, n-1])
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)  # m values
    w_lags = w  # w[m + (n-1)]
    
    for idx_m, m in enumerate(lags):
        wm = w_lags[idx_m]
        if wm < 1e-15:
            continue
        # First term: (1-h[j-m]) for j where 0<=j-m<n, i.e., max(0,m) <= j <= min(n-1, n-1+m)
        j_lo = max(0, m)
        j_hi = min(n-1, n-1+m)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wm * (1 - h[j_lo-m:j_hi-m+1])
        # Second term: -h[j+m] for j where 0<=j+m<n, i.e., max(0,-m) <= j <= min(n-1, n-1-m)
        j_lo2 = max(0, -m)
        j_hi2 = min(n-1, n-1-m)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= wm * h[j_lo2+m:j_hi2+m+1]
    
    grad *= dx
    return grad

def active_set_lp_step(h, n, top_k=5, budget=0.01):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    nc = len(c)
    
    # Get top k lags
    top_idx = np.argsort(c)[-top_k:]
    c_max = c.max()
    
    # Compute gradients of the top k lags
    # ∂c_m/∂h[j] = dx * [(1-h[j-m]) * 1{0<=j-m<n} - h[j+m] * 1{0<=j+m<n}]
    
    grads = np.zeros((top_k, n))
    for idx, m_global in enumerate(top_idx):
        m = m_global - (n-1)  # actual lag
        for j in range(n):
            g = 0.0
            if 0 <= j - m < n:
                g += (1 - h[j-m])
            if 0 <= j + m < n:
                g -= h[j+m]
            grads[idx, j] = dx * g
    
    # Solve LP: min t
    # s.t. grads[i, :] @ d <= -t for all i (we want each active lag to decrease by at least t)
    #      sum(d) = 0
    #      -h[j] <= d[j] <= 1-h[j] for all j
    #      d[j] >= -h[j], d[j] <= 1-h[j]
    
    # Using scipy.optimize.linprog
    from scipy.optimize import linprog
    
    # Variables: d[0], ..., d[n-1], t
    # Objective: minimize t => c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ [d, t] <= b_ub
    # For each active lag i: grads[i] @ d + t <= 0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = grads  # grads[i, :] @ d
    A_ub[:, -1] = 1.0    # + t
    b_ub = np.zeros(top_k)
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[j] <= d[j] <= 1-h[j], t >= 0
    bounds = [(-h[j], 1-h[j]) for j in range(n)] + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        d = res.x[:n]
        step = res.x[-1]  # this is t, the minimum decrease
        if step > 1e-10:
            h_new = h + d
            h_new = np.clip(h_new, 0, 1)
            h_new = project_feasible(h_new, n)
            return h_new, step
    return h, 0.0

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        # Distribute proportionally to available room
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax(x):
    mx = x.max()
    e = np.exp(x - mx)
    return e / e.sum()

def compute_smooth_gradient(h, n, w):
    """Compute gradient of smooth max using precomputed weights w (length 2n-1)."""
    dx = 2.0 / n
    nc = 2 * n - 1
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for idx in range(nc):
        wm = w[idx]
        if wm < 1e-12:
            continue
        m = lags[idx]
        # First term: sum over j where 0 <= j-m < n: w[m] * (1-h[j-m])
        j_lo = max(0, m)
        j_hi = min(n-1, n-1+m)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wm * (1.0 - h[j_lo-m:j_hi-m+1])
        # Second term: sum over j where 0 <= j+m < n: -w[m] * h[j+m]
        j_lo2 = max(0, -m)
        j_hi2 = min(n-1, n-1-m)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= wm * h[j_lo2+m:j_hi2+m+1]
    grad *= dx
    return grad

def compute_lag_gradient(h, n, m):
    """Compute gradient of a single lag correlation."""
    dx = 2.0 / n
    grad = np.zeros(n)
    j_lo = max(0, m)
    j_hi = min(n-1, n-1+m)
    if j_lo <= j_hi:
        grad[j_lo:j_hi+1] += (1.0 - h[j_lo-m:j_hi-m+1])
    j_lo2 = max(0, -m)
    j_hi2 = min(n-1, n-1-m)
    if j_lo2 <= j_hi2:
        grad[j_lo2:j_hi2+1] -= h[j_lo2+m:j_hi2+m+1]
    grad *= dx
    return grad

def active_set_lp(h, n, top_k=5, box_bound=0.05):
    """Solve LP to find direction that minimizes max of top-k lags (linearized)."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    nc = len(c)
    
    top_idx = np.argsort(c)[-top_k:]
    
    # Compute gradients for top k lags
    grads = np.zeros((top_k, n))
    for idx, gi in enumerate(top_idx):
        m = gi - (n-1)
        grads[idx] = compute_lag_gradient(h, n, m)
    
    # LP: min t
    # s.t. grads[i] @ d - t <= -c[top_idx[i]] for all i
    #      sum(d) = 0
    #      -box <= d[j] <= box
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = -c[top_idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-box_bound, box_bound)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            d = res.x[:n]
            t_new = res.x[-1]
            return d, t_new, c[top_idx].max()
    except Exception:
        pass
    return None, None, c[top_idx].max()

def smooth_gradient_descent(h, n, deadline, alpha_start=20.0, alpha_end=5000.0, 
                            top_k=10, lr=0.005, momentum=0.9):
    """Smooth gradient descent with adaptive alpha and momentum."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    t_start = time.time()
    total_time = max(1e-6, deadline - t_start)
    vel = np.zeros(n)
    
    iteration = 0
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total_time)
        alpha = alpha_start * (alpha_end / alpha_start) ** (frac ** 0.7)
        
        dx = 2.0 / n
        c = np.correlate(h, 1-h, mode='full') * dx
        
        if top_k < len(c):
            top_idx = np.argsort(c)[-top_k:]
            w = np.zeros(len(c))
            w[top_idx] = softmax(alpha * c[top_idx])
        else:
            w = softmax(alpha * c)
        
        grad = compute_smooth_gradient(h, n, w)
        # Project gradient to maintain sum constraint
        grad = grad - np.mean(grad)
        
        # Momentum update
        vel = momentum * vel + (1 - momentum) * grad
        h_new = h - lr * vel
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        elif c_new < best_c:
            h = h_new
        # If no improvement for a while, reduce learning rate
        if c_new >= best_c:
            lr *= 0.995
        
        h = h_new
        iteration += 1
    
    return best_h, best_c
