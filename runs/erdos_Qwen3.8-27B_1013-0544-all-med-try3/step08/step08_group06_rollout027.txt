from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # log-sum-exp
    mx = shifted.max()
    if mx > 500:
        lse = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        lse = np.log(np.sum(np.exp(shifted))) / alpha
    return float(m + lse)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        lse = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        lse = np.log(np.sum(np.exp(shifted))) / alpha
    return float(m + lse)


def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_con_jac(x, n):
    return np.ones(n)


def slsqp_h(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def pair_search(h, n, deadline, deltas, max_passes=300):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            improved = True
                        else:
                            h[i], h[j] = hi2 - d, hj2 + d
            if not improved:
                break
    return h, best_c


def near_pair_search(h, n, deadline, deltas, radius=40, max_passes=200):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False

                    return h, best_c

                j_hi = min(n, i + radius + 1)

    return h, best_c


def coord_descent_max(h, n, deadline, max_iters=5000):
    """Direct coordinate descent on the non-smooth max objective."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    c_arr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = int(np.argmax(c_arr))
    
    for it in range(max_iters):

            return h, best_c
        # Find the worst k
        c_arr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(c_arr))
        cur_max = c_arr[k_star]
        
        # Try small adjustments to individual coordinates
        step = 0.01
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            # Try decreasing h[i] (if it's contributing to overlap at k_star)
            # and compensating elsewhere via projection
            for delta in [step, -step, step/2, -step/2]:
                h_new = h.copy()
                h_new[i] += delta
                h_new = project_feasible(h_new, n)

                    h = h_new
                    best_c = c_new
                    improved = True
                    break
        if not improved:
            step /= 2
            if step < 0.0001:
                break
    return h, best_c

    best_c = compute_c5(best_h, n)

    # SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp_h(best_h, n, a, maxiter=4000, ftol=1e-16)
        c = compute_c5(ht, n)

            best_c = c
            best_h = ht

    # Pair search
    if n <= 200:
        if time.time() < deadline - 15:
            best_h, best_c = pair_search(best_h, n, min(deadline - 30, time.time() + 120),

        best_h, best_c = near_pair_search(best_h, n, deadline - 1,

    """Enforce reflection symmetry h[i] = h[n-1-i]."""
    hs = 0.5 * (h + h[::-1])
