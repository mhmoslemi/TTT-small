                        improved = True
                        if c < best_c:
                            best_c = c
                            best_h = cur.copy()
                    else:
                        cur[j] = cur[j] - direction * frac * (1.0 - cur[j] if direction > 0 else cur[j])

def pair_swap(h, n, deadline, max_sweeps=100):
    """Try swapping small amounts between pairs of coordinates."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try all pairs (or a random subset for large n)
        if n <= 300:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            rng = np.random.RandomState(42 + sweep)
            pairs = [(int(rng.randint(n)), int(rng.randint(n))) for _ in range(n * 20)]
        
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            if i == j:
                continue
            for delta in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]:
                ni, nj = cur[i] + delta, cur[j] - delta
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    cur[i], cur[j] = ni, nj
                    c = compute_c5(cur, n)
                    if c < best_c:
                        best_c = c
                        best_h = cur.copy()
                        improved = True
                    else:
                        cur[i], cur[j] = ni - delta, nj + delta
        if not improved:
            break
    return best_h, best_c
