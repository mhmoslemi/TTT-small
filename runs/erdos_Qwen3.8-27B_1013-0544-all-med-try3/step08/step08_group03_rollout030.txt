import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue

        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-14, 'maxfun': max_iters * 15,

            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:

                h = h_new

def gradient_directed_pair_search(h, n, deadline):
    """Pair search guided by gradient to focus on most promising pairs."""
    best_c = compute_c5(h, n)
    for alpha in [1000, 10000, 100000]:
        while time.time() < deadline:
            g = grad_smooth_max(h, n, alpha)
            # Score for pair (i,j): directional derivative
            # Moving h[i] up by d, h[j] down by d: change ≈ d*(g[i]-g[j])
            # We want g[i]-g[j] to be negative (decrease objective)
            improved = False
            # Try top pairs by gradient difference
            diffs = np.subtract.outer(g, g)  # diffs[i,j] = g[i] - g[j]
            # We want negative diffs (i up, j down) or positive diffs (i down, j up)
            # Sort by most negative
            for idx in np.argsort(diffs):
                if time.time() > deadline:
                    return h, best_c
                i, j = divmod(int(idx), n)
                if i == j:
                    continue
                # Try moving h[i] up, h[j] down
                d_vals = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
                for d in d_vals:
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                            break
                        else:
                            h[i] -= d; h[j] += d
                    if improved:
                        break
                if not improved:
                    # Try opposite direction
                    for d in d_vals:
                        if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                            h[i] -= d; h[j] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c; improved = True
                                break
                            else:
                                h[i] += d; h[j] -= d
                        if improved:
                            break
                # Only try top 50 pairs per iteration to save time
                if int(idx) // n > 20:  # rough check, not exact
                    pass
            if not improved:
                break
    return h, best_c

def fine_pair_search(h, n, deadline, deltas, radius=None):

            excess = h_t.sum() - target
            if excess <= h_t.sum():
                # Turn the lowest-valued 1s to 0
                ones = np.where(h_t == 1.0)[0]
                # Sort by h value (ascending) and flip the smallest
                order = np.argsort(h[ones])
                for idx in order[:int(excess)]:

            if deficit <= (1.0 - h_t).sum():

        c_t = compute_c5(h_t, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 5.0), (2000, 20.0), (10000, 100.0), (200000, 500.0), (2000000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 1:
            break
        try:
            res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 20,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def subgrad(h, n, deadline, lr=0.005):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(3000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        cmax = c.max()

        active = np.where(c >= cmax - tol)[0]

                g[k:] += (-h[:nv]) * dx_
            else:
                kk = -k
                nv = n - kk
                g[kk:] += (1.0 - h[:nv]) * dx_
                g[:nv] += (-h[kk:]) * dx_
        g /= len(active)
        g -= g.mean()  # project onto sum(d)=0
