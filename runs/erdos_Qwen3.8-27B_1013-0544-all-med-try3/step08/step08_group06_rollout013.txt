```python
    mask_bm = (b_minus_k >= 0) & (b_minus_k < n)
    b_plus_k = b + k
    mask_bp = (b_plus_k >= 0) & (b_plus_k < n)
    
    delta[mask_bm] -= d * (1 - h[b_minus_k[mask_bm]])
    delta[mask_bp] -= d * h[b_plus_k[mask_bp]]
    
    return delta * dx
```

But this creates multiple numpy arrays per pair evaluation, which in Python is slow due to overhead. For 20000 pairs, the Python overhead might dominate.

Alternative: precompute the "influence" matrices. For each index a, precompute which correlation entries it affects and what the coefficients are. Then for a pair (a,b), the delta is just:
delta = d * (influ_a - influ_b)

where influ_a[m] = [(1-h[a-k_m]) + h[a+k_m]] for valid indices.

But influ_a depends on h, so it changes as h changes. However, within a single pass (where h is fixed except for the trial move), influ_a is fixed.

Actually, I can precompute for the current h:
influ[a][m] = (1-h[a-k_m]) * valid + h[a+k_m] * valid

This is an n × (2n-1) matrix. For n=200, that's 200 × 399 = 80000 entries. Computing it once is O(n²) ≈ 40000.

Then for each pair (a,b) with delta d:
Δcorr = d * (influ[a] - influ[b]) * dx

And the new max is max(corr + d * (influ[a] - influ[b]) * dx)

This is a vector addition of length 2n-1 = 399, plus a max. In numpy, this is very fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    if mx > 500:
        v = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        w = np.exp(sh - mx)
    else:
        v = m + np.log(np.sum(np.exp(sh))) / alpha
        w = np.exp(sh)
    w /= w.sum()
    return float(v), w

def smooth_max_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    if mx > 500:
        v = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        w = np.exp(sh - mx)
    else:
        v = m + np.log(np.sum(np.exp(sh))) / alpha
        w = np.exp(sh)
    w_sum = w.sum()
    w = w / w_sum
    
    # Gradient: sum_k w[k] * (d corr[k] / dx)
    # corr[m] = sum_i x[i]*(1-x[i-k])*dx where k = m-(n-1)
    # d corr[m]/dx[a] = dx * [(1-x[a-k]) * valid(a-k) + x[a+k] * valid(a+k)]
    
    k = np.arange(2*n-1) - (n-1)  # shifts
    # Compute Jacobian columns: for each a, the vector d corr/ dx[a]
    # J[a, m] = (1-x[a-k[m]]) * [0<=a-k[m]<n] + x[a+k[m]] * [0<=a+k[m]<n]
    
    # Vectorized:
    a_idx = np.arange(n)[:, None]  # (n, 1)
    k_row = k[None, :]  # (1, 2n-1)
    am = a_idx - k_row  # a - k, shape (n, 2n-1)
    ap = a_idx + k_row  # a + k, shape (n, 2n-1)
    mask_am = (am >= 0) & (am < n)
    mask_ap = (ap >= 0) & (ap < n)
    
    J = np.zeros((n, 2*n-1))
    J[mask_am] = 1.0 - x[am[mask_am]]
    J[mask_ap] += x[ap[mask_ap]]
    
    grad = (w @ J) * dx  # shape (n,)
    return v, w, grad

def slsqp_run(h, n, alpha, maxiter=3000, ftol=1e-15):
    def objective(x):
        v, _ = smooth_max_obj(x, n, alpha)
        return v
    
    def constr_fun(x):
        return np.sum(x) - n / 2.0
    
    def constr_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

        mask_ap = (ap >= 0) & (ap < n)
        influ = np.zeros((n, 2*n-1))
        influ[mask_am] = 1.0 - h[am[mask_am]]
        influ[mask_ap] += h[ap[mask_ap]]
        return influ
    
    for d in deltas:
```

```python
                        nm = (corr + delta).max()
                        if nm < best_c - 1e-13:
                            h[a] = max(0.0, h[a] - d)
                            h[b] = min(1.0, h[b] + d)
                            corr = corr + delta
                            best_c = float(nm)
                            best_h = h.copy()
                            influ = get_influ(h)
                            any_imp = True
                            recompute_counter += 1
                            if recompute_counter % 500 == 0:
                                corr = compute_corr(h, n)
            if not any_imp:
                break
    return best_h, best_c

def multi_start_optimize(h0, n, deadline, seed, n_starts=4):
    rng = np.random.RandomState(seed)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    t0 = time.time()
    alloc = (deadline - t0) / max(1, n_starts)
    for s in range(n_starts):
        if time.time() > deadline - 5:
            break
        if s == 0:
            h_try = best_h.copy()
        else:
            scale = 0.05 + 0.15 * (s / max(1, n_starts - 1))
            h_try = best_h + rng.normal(0, scale, n)
            h_try = project_feasible(h_try, n)
        end_s = t0 + (s + 1) * alloc
        for alpha in [2000, 20000, 100000]:
            if time.time() > end_s - 3:
                break
            h_new = slsqp_optimize(h_try, n, alpha, maxiter=1500)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 200
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: SLSQP with increasing smoothing (35% of time)
    p1_end = t0 + (deadline - t0) * 0.35
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > p1_end - 5:
            break
        h_try = slsqp_optimize(best_h, best_n, alpha, maxiter=2500)
        c_try = compute_c5(h_try, best_n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try

    # Phase 2: Fast pair search - coarse to fine (30%)
    p2_end = t0 + (deadline - t0) * 0.65
    best_h, best_c = fast_pair_search(best_h, best_n, p2_end,
                                       [0.05, 0.02, 0.01, 0.005, 0.002, 0.001])

    # Phase 3: Fine SLSQP + fine pair search (15%)
    p3_end = t0 + (deadline - t0) * 0.80
    for alpha in [500000, 2000000, 5000000]:
        if time.time() > p3_end - 8:
            break
        h_try = slsqp_optimize(best_h, best_n, alpha, maxiter=2000)
        c_try = compute_c5(h_try, best_n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
    if time.time() < p3_end:
        best_h, best_c = fast_pair_search(best_h, best_n, p3_end,
                                           [0.001, 0.0005, 0.0002, 0.0001])

    # Phase 4: Multi-resolution exploration (10%)
    p4_end = t0 + (deadline - t0) * 0.92
    if time.time() < deadline - 25:
        for n_try in [150, 100, 80]:
            if time.time() > deadline - 15:
                break
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_try = slsqp_optimize(h_interp, n_try, 30000, maxiter=1500)
            c_try = compute_c5(h_try, n_try)
            if c_try < best_c - 1e-14:
                h_try, c_try = fast_pair_search(h_try, n_try,
                                                 min(time.time() + 25, deadline - 8),
                                                 [0.01, 0.005, 0.002, 0.001, 0.0005])
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try
                    best_n = n_try

    # Phase 5: Multi-start at current resolution (5%)
    if time.time() < deadline - 20:
        p5_end = min(time.time() + 40, deadline - 5)
        h_ms, c_ms = multi_start_optimize(best_h, best_n, p5_end, seed=7, n_starts=3)
        if c_ms < best_c - 1e-14:
            best_c = c_ms
            best_h = h_ms

    # Final polish
    if time.time() < deadline - 2:
        best_h, best_c = fast_pair_search(best_h, best_n, deadline - 1,
                                           [0.0005, 0.0002, 0.0001, 0.00005])

    # Final SLSQP cleanup
    if time.time() < deadline - 1:
        h_try = slsqp_optimize(best_h, best_n, 5000000, maxiter=1000)
        c_try = compute_c5(h_try, best_n)
        if c_try < best_c:
            best_h = h_try
            best_c = c_try

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
