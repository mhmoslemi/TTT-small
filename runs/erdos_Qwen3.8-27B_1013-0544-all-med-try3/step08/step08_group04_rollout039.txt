from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val_grad(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        shifted = sh - mx
        exps = np.exp(shifted)
        weights = exps / exps.sum()
        val = m + (mx + np.log(exps.sum())) / alpha
    else:
        exps = np.exp(sh)
        weights = exps / exps.sum()
        val = m + np.log(exps.sum()) / alpha
    return val, weights

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    val, weights = smooth_max_val_grad(c, alpha)
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        s = k - n + 1
        # First part: (1-h)[i-s] for valid i
        i_lo = max(0, s)
        i_hi = min(n, s + n)
        if i_lo < i_hi:
            grad_h[i_lo:i_hi] += w * (1 - h[i_lo - s:i_hi - s])
        # Second part: -h[i+s] for valid i
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_lo2 < i_hi2:
            grad_h[i_lo2:i_hi2] -= w * h[i_lo2 + s:i_hi2 + s]
    grad_h *= dx
    return val, grad_h

def gradient_optimize(h, n, deadline, alpha_schedule=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if alpha_schedule is None:
        alpha_schedule = [200, 1000, 5000, 20000, 100000]
    
    for alpha in alpha_schedule:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        lr = 0.05 / alpha * 100
        lr = min(lr, 0.02)
        patience = 0
        for it in range(3000):
            if time.time() > deadline:
                break
            val, grad = grad_smooth_max(h_cur, n, alpha)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-14:
                break
            # Gradient descent (minimize, so go opposite to gradient of max)
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
            h_cur = h_new
            # Adaptive step size
            if patience % 50 == 0 and patience > 0:
                lr *= 0.85
            if patience > 300:
                break
    return best_h, best_c

def fast_pair_eval(h, n, i, j, d, active_ks):
    """Evaluate change in c_k for active shifts when h[i]+=d, h[j]-=d"""
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    best_delta = 0
    for k in active_ks:
        s = k - n + 1
        delta_k = 0.0
        # Contribution from h[i] += d
        m1 = i - s
        if 0 <= m1 < n:
            delta_k += d * (1 - h[m1])
        m2 = i + s
        if 0 <= m2 < n:
            delta_k -= d * h[m2]
        # Contribution from h[j] -= d
        m3 = j - s
        if 0 <= m3 < n:
            delta_k -= d * (1 - h[m3])
        m4 = j + s
        if 0 <= m4 < n:
            delta_k += d * h[m4]
        new_c_k = (c[k] + delta_k) * dx
        if new_c_k > best_delta:
            best_delta = new_c_k
    return best_delta

                        all_pos.add(ii + dd)

                rng_local = np.random.RandomState(rnd)
                sampled = rng_local.choice(all_pos, size=80, replace=False)
                all_pos = sorted(sampled.tolist())

                    break

                            best_h = project_feasible(best_h, n)
                            best_c = compute_c5(best_h, n)
                            improved = True
                            break
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        new_c = fast_pair_eval(best_h, n, j, i, d, active_ks)
                        if new_c < best_c - 1e-12:
                            best_h[j] += d
                            best_h[i] -= d
                            best_h = project_feasible(best_h, n)
                            best_c = compute_c5(best_h, n)
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try smaller deltas in a second pass
            try_better_deltas = [0.001, 0.0005, 0.0002]
            for d in try_better_deltas:
                if time.time() > deadline:
                    break
                for i in range(n):
                    if time.time() > deadline:
                        break
                    for j in range(i + 1, n):
                        if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                            new_c = fast_pair_eval(best_h, n, i, j, d, active_ks)
                            if new_c < best_c - 1e-12:
                                best_h[i] += d
                                best_h[j] -= d
                                best_h = project_feasible(best_h, n)
                                best_c = compute_c5(best_h, n)
                                improved = True
                                break
                        if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                            new_c = fast_pair_eval(best_h, n, j, i, d, active_ks)
                            if new_c < best_c - 1e-12:
                                best_h[j] += d
                                best_h[i] -= d
                                best_h = project_feasible(best_h, n)
                                best_c = compute_c5(best_h, n)
                                improved = True
                                break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def full_pair_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def slsqp_optimize(h, n, alpha, maxiter=1000):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full')
        return smooth_max_val_grad(c * dx, alpha)[0]
    
    def obj_grad(x):
        xc = np.clip(x, 0, 1)
        val, g = grad_smooth_max(xc, n, alpha)
        # Chain rule for clip
        mask = (x > 0) & (x < 1)
        return g * mask
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or compute_c5(res.x, n) < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def slsqp_hargs(x, n):
    return x.sum() - n / 2.0

def slsqp_jargs(x, n):
    return np.ones(n)

def slsqp_opt2(h, n, alpha, maxiter=1000):
    dx = 2.0 / n
    
    def obj(x):
        xc = np.clip(x, 0, 1)
        c = np.correlate(xc, 1 - xc, mode='full') * dx
        return smooth_max_val_grad(c, alpha)[0]
    
    def obj_jac(x):
        xc = np.clip(x, 0, 1)
        val, g = grad_smooth_max(xc, n, alpha)
        mask = (x > 0) & (x < 1)
        return g * mask
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_hargs, 'jac': slsqp_jargs, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def simulated_annealing(h, n, deadline, start_temp=0.05, end_temp=0.001, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    h_cur = h.copy()
    start_time = time.time()
    total_time = max(1, deadline - start_time)
    
    for it in range(50000):
        if time.time() > deadline:
            break
        frac = (time.time() - start_time) / total_time
        temp = start_temp * (end_temp / start_temp) ** frac
        
        # Pick a random pair and delta
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([0.01, 0.02, 0.05, 0.1, 0.005]) * rng.choice([1, -1])
        
        if h_cur[i] + d > 1 or h_cur[i] + d < 0:
            continue
        if h_cur[j] - d > 1 or h_cur[j] - d < 0:
            continue
        
        h_cur[i] += d
        h_cur[j] -= d
        h_cur = project_feasible(h_cur, n)
        new_c = compute_c5(h_cur, n)
        
        delta = new_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_cur.copy()
        else:
            h_cur[i] -= d
            h_cur[j] += d
            h_cur = project_feasible(h_cur, n)
    
    return best_h, best_c
