import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)


def obj_slqp(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    if sm > 50:
        return m
    return m + np.log(np.exp(shifted - sm).sum()) / alpha


def cons_eq(x, n, alpha):
    return x.sum() - n / 2.0


def slsqp_optimize(h0, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: obj_slqp(x, n, alpha), h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

    return [(0.0, 1.0)] * n

def continuous_polish(h0, n, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000, 8000, 30000, 100000]
