import numpy as np
from scipy.optimize import minimize, linprog

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def weighted_subgradient(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2*n - 1) / (2*n - 1)
    else:
        w /= ws
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        if w[idx] < 1e-15:
            continue
        g = shift_grad_single(h, n, idx)
        grad += w[idx] * g
    return grad, float(m)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_step(h, n, active_idx, rho, max_vars=50):
    if len(active_idx) > max_vars:
        c_vec = compute_c5_vec(h, n)
        active_sorted = np.argsort(-c_vec[active_idx])
        active_idx = [active_idx[i] for i in active_sorted[:max_vars]]
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    grads = []
    c_vals = []
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        grads.append(g)
        c_vals.append(c_vec[idx])
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        A_ub[row, 0:n] = grads[j]
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, max(0.01, rho))]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=3000):
    def obj_fn(x, nn, aa):
        c = compute_c5_vec(x, nn)
        m = c.max()
        sh = aa * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / aa)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / aa)
    def sq_fn(x, nn):
        return float(x.sum() - nn / 2.0)
    def sq_jac_fn(x, nn):
        return np.ones(nn)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new

        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            best_n = n_high

    # Phase 9: Random perturbation + reoptimize (80s)
    if time.time() < deadline - 30:
        h9, c9 = perturb_and_reoptimize(best_h, best_n, deadline - 5, n_perturb=3, scale=0.015)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Final SQP + local polish
    if time.time() < deadline - 10:
        h10, c10 = sqp_optimize(best_h, best_n, deadline - 3, max_iter=200, rho_init=0.01, max_active=35)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
