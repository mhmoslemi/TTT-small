import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (50, 1.0), (200, 5.0), (1000, 10.0),
        (5000, 50.0), (20000, 100.0), (100000, 500.0),
        (500000, 2000.0), (2000000, 5000.0), (5000000, 10000.0),
        (20000000, 10000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 1:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def targeted_gradient(h, n, deadline, n_iters=5000):
    """Focus gradient on the worst few shifts."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.01
    dx = 2.0 / n
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        grad = np.zeros(n)
        # Focus on top 5 shifts
        top_idx = np.argsort(c)[-5:]
        for m in top_idx:
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += dx * (1.0 - h[k:n])
                grad[k:n] += dx * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += dx * (1.0 - h[0:nv])
                grad[0:nv] += dx * (-h[kk:n])
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        grad /= g_norm
        # Projected gradient step
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.5
            if lr < 1e-12:
                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = False
        while improved and time.time() < deadline:
            improved = False
            n_improved = 0
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i >= j:
                        continue
                    # Try moving d from j to i
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True; n_improved += 1
                        else:
                            h[i] -= d; h[j] += d
                    # Try moving d from i to j
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True; n_improved += 1
                        else:
                            h[i] += d; h[j] -= d
        if not improved:
            break
    return h, best_c

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    if mx > 500:
        w = np.exp(c_shift - mx)
        w_sum = w.sum()
    else:
        w = np.exp(c_shift)
        w_sum = w.sum()

    dx = 2.0 / n

        (10000000, 50000.0), (50000000, 50000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 1:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def adam_gradient(h, n, deadline, n_shifts=5, lr=0.01, beta1=0.9, beta2=0.999):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stagnant = 0
    
    while time.time() < deadline:
        t += 1
        c = compute_c5_vec(h, n)
        c_max = c.max()
        grad = np.zeros(n)
        top_idx = np.argsort(c)[-n_shifts:]
        for mi in top_idx:
            k = mi - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += dx * (1.0 - h[k:n])
                grad[k:n] -= dx * h[0:nv]
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += dx * (1.0 - h[0:nv])
                grad[0:nv] -= dx * h[kk:n]
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h - step, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.8
            stagnant += 1
            if stagnant > 200 or lr < 1e-12:
                break
    return best_h, best_c

def targeted_gradient(h, n, deadline, n_shifts=5, n_iters=50000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.005
    dx = 2.0 / n
    stagnant = 0
    
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        grad = np.zeros(n)
        top_idx = np.argsort(c)[-n_shifts:]
        for mi in top_idx:
            k = mi - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += dx * (1.0 - h[k:n])
                grad[k:n] -= dx * h[0:nv]
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += dx * (1.0 - h[0:nv])
                grad[0:nv] -= dx * h[kk:n]
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        grad /= g_norm
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
            lr = min(lr * 1.03, 0.02)
        else:
            lr *= 0.7
            stagnant += 1
            if stagnant > 100 or lr < 1e-13:
                break
    return best_h, best_c

def local_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = i + 1
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def full_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS at original resolution
    t1 = min(t0 + 250, deadline - 400)
    if time.time() < t1:
        best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_iters=150)

    # Phase 2: Adam gradient (top 5 shifts)
    if time.time() < deadline - 200:
        t2 = min(time.time() + 80, deadline - 200)
        h_t, c_t = adam_gradient(best_h, best_n, t2, n_shifts=5, lr=0.01)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 3: Targeted gradient descent
    if time.time() < deadline - 120:
        t3 = min(time.time() + 80, deadline - 120)
        h_t, c_t = targeted_gradient(best_h, best_n, t3, n_shifts=3, n_iters=50000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 4: SLSQP
    if time.time() < deadline - 80:
        for a in [200000, 1000000, 5000000]:
            if time.time() > deadline - 75:
                break
            h_s, c_s = slsqp_optimize(best_h, best_n, min(time.time() + 40, deadline - 72),
                                      alpha=a, maxiter=800)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 5: Try higher resolutions
    for n_try in [400, 500, 700, 900]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc = min(70, deadline - time.time() - 40)
        if alloc < 15:
            break
        
        h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.5, max_iters=80)
        if time.time() < deadline - 30:
            h_t, c_t = targeted_gradient(h_t, n_try, min(time.time() + 20, deadline - 27), n_shifts=3, n_iters=20000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 6: Final L-BFGS at best resolution
    if time.time() < deadline - 20:
        h_f, c_f = lbfgs_optimize(best_h, best_n, min(time.time() + 35, deadline - 17), max_iters=200)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 7: Final targeted gradient
    if time.time() < deadline - 10:
        h_t, c_t = targeted_gradient(best_h, best_n, min(time.time() + 25, deadline - 8), n_shifts=3, n_iters=30000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 8: Local pair search (fine)
    if time.time() < deadline - 3:
        best_h, best_c = local_pair_search(best_h, best_n, deadline - 1,
                                           [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                           radius=40)

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
