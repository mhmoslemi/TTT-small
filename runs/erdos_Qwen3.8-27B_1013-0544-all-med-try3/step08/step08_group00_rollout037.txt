def smooth_max_obj(h, n, alpha):
    """Smooth approximation of max using log-sum-exp with constraint penalty."""
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    logsumexp = mx + np.log(np.sum(np.exp(shifted - mx)))
    return m + logsumexp / alpha

def smooth_max_grad(h, n, alpha):
    """Gradient of smooth max."""
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue

        wm = w[mm] * dx
        if k >= 0:

            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])

            nv = n - kk
            if nv > 0:

                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def intensive_local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad_sum = np.zeros(n)
            for idx in active[:30]:
                grad_sum += shift_gradient_single(h, n, idx)
            grad_sum /= len(active)
            n_frac = max(10, int(0.03 * n))
            i_cands = np.argsort(grad_sum)[:n_frac]
            j_cands = np.argsort(-grad_sum)[:n_frac]
            for i in i_cands:
                if time.time() > deadline:
                    return best_h, best_c
                for j in j_cands:
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-10 and h[j] - step >= -1e-10:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
    return best_h, best_c

def targeted_perturb(h, n, deadline, seed=42, n_perturb=8, perturb_scale=0.03):
    """Generate targeted perturbations based on worst shift gradients."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    top_indices = np.argsort(-c_vec)[:10]
    for attempt in range(n_perturb):
        if time.time() > deadline:
            break
        pert = np.zeros(n)
        for idx in top_indices[:5]:
            g = shift_gradient_single(h, n, idx)
            pert += rng.choice([-1, 1]) * g
        pert += rng.randn(n) * perturb_scale * 0.5
        h_pert = project_feasible(best_h + pert, n)
        t_sub = min(time.time() + 20, deadline - 5)
        h_opt, c_opt = run_sqp(h_pert, n, t_sub, max_iter=100, rho_init=0.04)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    return best_h, best_c

def binary_local_search(h, n, deadline, max_flips=5000):
    """Try flipping small regions (blocks) to improve."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    block_sizes = [2, 3, 5, 8]
    flips_done = 0
    while flips_done < max_flips and time.time() < deadline:
        bs = block_sizes[flips_done % len(block_sizes)]
        start = rng.randint(0, n - bs)
        # Try inverting a block (h[i] -> 1-h[i] for i in block)
        # Then project to maintain sum constraint
        h_trial = h.copy()
        h_trial[start:start+bs] = 1.0 - h_trial[start:start+bs]
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-12:
            best_c = c_trial
            best_h = h_trial
        flips_done += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 0: Coarse resolution exploration (n=100, 40s)
    n_coarse = 100
    t_coarse_end = t0 + 40
    if t_coarse_end > time.time() + 5:
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
        h_coarse = np.interp(x_coarse, x_old, h)
        h_coarse = project_feasible(h_coarse, n_coarse)
        # Try L-BFGS at coarse
        h_c, c_c = run_lbfgs(h_coarse, n_coarse, min(time.time() + 20, t_coarse_end - 10),
                            alphas=[500, 5000])
        # Try SQP at coarse
        h_c, c_c = run_sqp(h_c, n_coarse, t_coarse_end, max_iter=200, rho_init=0.04)
        # Interpolate back to fine
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_from_coarse = np.interp(x_new, x_coarse, h_c)
        h_from_coarse = project_feasible(h_from_coarse, n)
        c_from_coarse = compute_c5(h_from_coarse, n)
        if c_from_coarse < best_c:
            best_c = c_from_coarse
            best_h = h_from_coarse.copy()
            h = h_from_coarse

    # Phase 1: L-BFGS warmup (50s)
    t1 = t0 + 90
    if time.time() < t1:
        h, c = run_lbfgs(h, n, t1, alphas=[500, 2000, 10000])
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: SQP main optimization (180s)
    t2 = t0 + 270
    if time.time() < t2:
        h, c = run_sqp(h, n, t2, max_iter=600, rho_init=0.05, active_frac=0.001, max_active=50)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Intensive local search (120s)
    t3 = t0 + 390
    if time.time() < t3:
        h, c = intensive_local_search(h, n, t3)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: SQP with tighter params (80s)
    t4 = t0 + 470
    if time.time() < t4:
        h, c = run_sqp(h, n, t4, max_iter=400, rho_init=0.02, active_frac=0.0005, max_active=30)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Another local search round (60s)
    t5 = t0 + 530
    if time.time() < t5:
        h, c = intensive_local_search(h, n, t5)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Targeted perturbations to escape local optima (150s)
    t6 = t0 + 680
    if time.time() < t6:
        h, c = targeted_perturb(best_h, n, t6, seed=seed, n_perturb=10, perturb_scale=0.04)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 7: SQP final push (80s)
    t7 = t0 + 760
    if time.time() < t7:
        h, c = run_sqp(h, n, t7, max_iter=300, rho_init=0.03, active_frac=0.0005, max_active=40)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 8: Local search final (60s)
    t8 = t0 + 820
    if time.time() < t8:
        h, c = intensive_local_search(h, n, t8)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 9: Upsample to n=1000 (150s)
    n_high = 1000
    t9_end = deadline - 30
    if time.time() < t9_end - 50:
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_high, endpoint=False)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)
        # Optimize at higher resolution
        t9a = min(time.time() + 50, t9_end - 40)
        if time.time() < t9a:
            h_high, c_high = run_lbfgs(h_high, n_high, t9a, alphas=[2000, 10000])
        t9b = min(time.time() + 50, t9_end - 20)
        if time.time() < t9b:
            h_high, c_high = run_sqp(h_high, n_high, t9b, max_iter=200, rho_init=0.03, max_active=40)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n = n_high

    # Phase 10: Final local search on best
    if time.time() < deadline - 5:
        h, c = intensive_local_search(best_h, len(best_h), deadline - 2)
        if c < best_c:
            best_c = c
            best_h = h

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
