def gradient_c5(h, n, K=5):
    """Compute subgradient of smooth top-K objective"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c = c - c.mean()  # center for numerical stability
    # Top K
    topk_idx = np.argsort(c)[-K:]
    # Gradient for each top-k shift
    grad = np.zeros(n)
    for k in topk_idx:
        shift = k - (n - 1)  # actual shift value
        for i in range(n):
            ip = i + shift
            im = i - shift
            contrib = 0.0
            if 0 <= ip < n:
                contrib += (1 - h[ip])
            if 0 <= im < n:
                contrib -= h[im]
            grad[i] += contrib
    grad *= dx / K
    return grad

    c = np.correlate(h, 1 - h, mode='full') * dx
    topk_idx = np.argsort(c)[-K:]
    grad = np.zeros(n)

            # (1-h[i+shift]) for i in [0, n-shift-1]
            grad[:n-shift] += (1 - h[shift:])

            # -h[i+shift] = -h[i-s] for i in [0, n-s-1]
