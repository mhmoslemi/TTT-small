import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_optimize(h, n, i, c_vec):
    """Find optimal h[i] given all others fixed."""
    # Compute slope of each c[m] w.r.t. h[i]
    # For k >= 0 (m = n-1+k, k from 0 to n-1):
    #   slope += (1-h[i+k]) if i+k < n, i.e., k < n-i
    #   slope += (-h[i-k]) if i-k >= 0, i.e., k <= i
    # For k < 0, kk=-k (m = n-1-kk, kk from 1 to n-1):
    #   slope += (1-h[i-kk]) if i-kk >= 0, i.e., kk <= i
    #   slope += (-h[i+kk]) if i+kk < n, i.e., kk < n-i
    
    slopes = np.zeros(2*n-1)
    
    # Positive shifts
    max_k1 = min(n - 1 - i, n - 1)  # k < n-i means k <= n-1-i
    if max_k1 >= 0:
        ks = np.arange(max_k1 + 1)
        ms = n - 1 + ks
        slopes[ms] += (1.0 - h[i + ks])
    
    max_k2 = min(i, n - 1)  # k <= i
    if max_k2 >= 0:
        ks = np.arange(max_k2 + 1)
        ms = n - 1 + ks
        slopes[ms] += (-h[i - ks])
    
    # Negative shifts
    max_kk1 = min(i, n - 1)  # kk <= i
    if max_kk1 >= 1:
        kks = np.arange(1, max_kk1 + 1)
        ms = n - 1 - kks
        slopes[ms] += (1.0 - h[i - kks])
    
    max_kk2 = min(n - 1 - i, n - 1)  # kk < n-i means kk <= n-1-i
    if max_kk2 >= 1:
        kks = np.arange(1, max_kk2 + 1)
        ms = n - 1 - kks
        slopes[ms] += (-h[i + kks])
    
    affected = np.where(np.abs(slopes) > 1e-15)[0]
    if len(affected) == 0:
        return h[i]
    
    # intercept[m] = c[m] - h[i] * slope[m] (value at v=0)
    intercepts = c_vec - h[i] * slopes
    
    # Unaffected floor
    unaffected_mask = np.abs(slopes) < 1e-15
    if unaffected_mask.any():
        floor = c_vec[unaffected_mask].max()
    else:
        floor = -np.inf
    
    # Minimize max(floor, max_{m in affected} (intercept[m] + v * slope[m])) over v in [0,1]
    # Convex piecewise linear -> ternary search
    
    a_af = slopes[affected]
    b_af = intercepts[affected]
    
    def obj(v):
        return max(floor, (b_af + v * a_af).max())
    
    # Check boundaries
    f0 = obj(0.0)
    f1 = obj(1.0)
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1_val = obj(m1)
        f2_val = obj(m2)
        if f1_val < f2_val:
            hi = m2
        else:
            lo = m1
    
    v_mid = (lo + hi) / 2.0
    f_mid = obj(v_mid)
    
    # Pick best
    if f0 <= f1 and f0 <= f_mid:
        return 0.0
    elif f1 <= f0 and f1 <= f_mid:
        return 1.0
    else:
        return v_mid

def coordinate_descent(h, n, deadline, max_sweeps=5000, order='cyclic'):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        improved = False
        
        if order == 'cyclic':
            indices = range(n)
        else:
            indices = np.random.permutation(n)
        
        for i in indices:
            if time.time() > deadline:
                return best_h, best_c
            v_opt = coord_optimize(h, n, i, c_vec)
            if abs(v_opt - h[i]) > 1e-12:
                old = h[i]
                h[i] = v_opt
                c_vec = compute_c5_vec(h, n)
                improved = True
        
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            scale = diff / tot
            h += scale * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            scale = diff / tot
            h += scale * h
    return np.clip(h, 0.0, 1.0)

def coord_optimize(h, n, i, c_vec):
    slopes = np.zeros(2 * n - 1)
    
    # Positive shifts: m = n-1+k, k from 0 to n-1
    # First factor: h[i] appears if i+k < n, i.e., k <= n-1-i
    max_k_f = n - 1 - i
    if max_k_f >= 0:
        ks = np.arange(max_k_f + 1)
        ms = (n - 1) + ks
        slopes[ms] += (1.0 - h[i + ks])
    
    # Second factor: h[i] appears in (1-h[i-k]) if k <= i
    max_k_s = i
    if max_k_s >= 0:
        ks = np.arange(max_k_s + 1)
        ms = (n - 1) + ks
        slopes[ms] -= h[i - ks]
    
    # Negative shifts: kk from 1 to n-1, m = n-1-kk
    # First factor: h[i] appears as h[i+kk]... no wait.
    # c[m] for kk: sum_{j=0}^{n-1-kk} h[j+kk]*(1-h[j])
    # h[i] as h[j+kk]: j+kk=i, j=i-kk. Valid: 0<=i-kk<=n-1-kk, i.e., kk<=i<=n-1.
    max_kk1 = i
    if max_kk1 >= 1:
        kks = np.arange(1, max_kk1 + 1)
        ms = (n - 1) - kks
        slopes[ms] += (1.0 - h[i - kks])
    
    # h[i] as h[j]: j=i. Valid: 0<=i<=n-1-kk, i.e., kk<=n-1-i.
    max_kk2 = n - 1 - i
    if max_kk2 >= 1:
        kks = np.arange(1, max_kk2 + 1)
        ms = (n - 1) - kks
        slopes[ms] -= h[i + kks]
    
    affected = np.where(np.abs(slopes) > 1e-15)[0]
    if len(affected) == 0:
        return h[i]
    
    b_af = c_vec[affected] - h[i] * slopes[affected]
    a_af = slopes[affected]
    
    unaffected_mask = np.abs(slopes) <= 1e-15
    floor = c_vec[unaffected_mask].max() if unaffected_mask.any() else -1e30
    
    # Minimize max(floor, max(b_af + v*a_af)) over v in [0,1]
    # Convex piecewise linear (plus constant floor)
    
    def f(v):
        return max(floor, np.max(b_af + v * a_af))
    
    # Use golden-section or ternary search (40 iterations for precision)
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    
    v_cand = (lo + hi) / 2.0
    f0 = f(0.0)
    f1 = f(1.0)
    fc = f(v_cand)
    
    if f0 <= f1 and f0 <= fc:
        return 0.0
    if f1 <= f0 and f1 <= fc:
        return 1.0
    return v_cand
