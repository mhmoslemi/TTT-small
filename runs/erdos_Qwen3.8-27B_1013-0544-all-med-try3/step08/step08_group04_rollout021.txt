from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_counts(h, n):
    """Return the raw counts (before multiplying by dx) and which index is max."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = int(np.argmax(corr))
    return corr, max_idx, float(corr[max_idx])

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    # Take the top n/2 values as 1, rest as 0
    num_ones = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:num_ones]] = 1.0
    return b

def sa_binary_h(h_start, n, deadline, rng):
    """Simulated annealing for binary optimization."""
    h = to_binary(h_start, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_c = best_c
    
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]

    T_min = 0.0001
    cooling = 0.9995

        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i], h[j] = h[j], h[i]
        
        c = compute_c5(h, n)
        
        if c < curr_c or rng.random() < np.exp(-(c - curr_c) / T):
            curr_c = c
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        
        T *= cooling
    
    # Final greedy improvement
    for _ in range(200):
        improved = False
        for d i in range(n):
            if h[i] == 1:
                for j in range(n):
                    if h[j] == 0:
                        h[i], h[j] = h[j], h[i]
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = h[j], h[i]
        if not improved:
            break
    
    return best_h, best_c

    b = np.zeros(n)
