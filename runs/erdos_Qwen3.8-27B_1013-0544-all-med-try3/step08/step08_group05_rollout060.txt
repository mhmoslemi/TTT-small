    idx = np.arange(n)
    # A[i] = sum_k p[k+n-1] * h[i+k] for 0<=i+k<n
    # j = i+k, k = j-i
    # A[i] = sum_j h[j] * p[j-i+n-1] for 0<=j<n and 0<=j-i+n-1<2n-1
    # The second condition: i-n+1 <= j <= i+n-1, which is always true for j in [0,n-1]
    
    # So A[i] = sum_j h[j] * p[j-i+n-1]
    # Matrix form: A[i] = sum_j P[i,j] * h[j] where P[i,j] = p[j-i+n-1]
    
    # Similarly B[i] = sum_j h[j] * p[i-j+n-1]
    
    # Both P and the B matrix are Toeplitz!
    
    # For A: P[i,j] = p[j-i+n-1]. Let diags be: for j-i = d, P = p[d+n-1]
    # This is a Toeplitz matrix with first column p[n-1-n+1]:p[n-1+0]:... 
    
    # Let me just construct it with numpy
    i_idx = np.arange(n)[:, None]  # n x 1
    j_idx = np.arange(n)[None, :]  # 1 x n

    # j - i + n - 1 ranges from (0-(n-1)+n-1) to ((n-1)-0+n-1) = 0 to 2n-2 ✓
    
    A_indices = j_idx - i_idx + n - 1  # n x n matrix of indices into p
    B_indices = i_idx - j_idx + n - 1  # n x n matrix of indices into p

    # B = P_B @ h where P_B[i,j] = p[B_indices[i,j]]

    B = P_B @ h
    C = P_B.sum(axis=1)  # sum of p over valid j for each i
