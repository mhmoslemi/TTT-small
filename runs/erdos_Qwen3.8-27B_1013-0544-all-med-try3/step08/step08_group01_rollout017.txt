import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha, mu, active):
    c = compute_c5_vec(x, n)
    if active is not None:
        c = c[active]
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu, active):
    c = compute_c5_vec(x, n)
    if active is not None:
        c_sub = c[active]
        m = c_sub.max()
        sh = alpha * (c_sub - m)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        idxs = active
    else:
        m = c.max()
        sh = alpha * (c - m)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        idxs = np.arange(2 * n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    for wi, idx in zip(w, idxs):
        if wi < 1e-16:
            continue
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            g[0:nv] += wm * (1.0 - x[k:n])
            g[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += wm * (1.0 - x[0:nv])
            g[0:nv] += wm * (-x[kk:n])
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def get_active(c_vec, n_top=10, rel_tol=0.002):
    m = c_vec.max()
    thresh = m - max(1e-12, rel_tol * m)
    active = np.where(c_vec >= thresh)[0]
    if len(active) > n_top:
        order = np.argsort(-c_vec[active])
        active = active[order[:n_top]]
    return active

def lbfgs_run(h, n, alpha, mu, active, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu, active),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxiter * 5,
                               'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_polish(h, n, deadline, radius=15):
    best_c = compute_c5(h, n)
    for d in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break

                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d

                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
