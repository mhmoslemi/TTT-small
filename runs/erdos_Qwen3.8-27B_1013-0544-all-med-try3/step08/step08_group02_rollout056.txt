def smooth_max_obj(x, n, active, alpha, mu):

    smx = sh.max()

    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(x, n, active, alpha, mu):

def thue_morse_binary(n):
    """Generate Thue-Morse sequence of length n, with n/2 ones"""
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Ensure exactly n/2 ones
    n_ones = n // 2
    n_zeros = n - n_ones
    if n % 2 != 0:
        n_zeros = n - n_ones
    # Count current ones
    cur_ones = tm.sum()
    if cur_ones > n_ones:
        # Flip some 1s to 0s
        idx_ones = np.where(tm == 1)[0]
        for i in range(int(cur_ones - n_ones)):
            tm[idx_ones[i]] = 0
    elif cur_ones < n_ones:
        idx_zeros = np.where(tm == 0)[0]
        for i in range(int(n_ones - cur_ones)):
            tm[idx_zeros[i]] = 1
    return tm.astype(float)

        c_tm = compute_c5(tm_h, n_pat)
        candidates.append((c_tm, tm_h))

        # Periodic patterns
        for period in [7, 9, 11, 13, 17, 19, 21, 23]:
            for duty in [1, 2, 3]:
                if 2 * duty < period:
                    pat = periodic_binary(n_pat, period, duty)
                    # Adjust to get exactly n/2 ones
                    target = n_pat / 2.0
                    diff = target - pat.sum()
                    if diff > 0:
                        for i in range(n_pat):
                            if pat[i] < 1.0:
                                add = min(1.0 - pat[i], diff)
                                pat[i] += add
                                diff -= add
                                if diff < 1e-12:
                                    break
                    elif diff < 0:
                        for i in range(n_pat - 1, -1, -1):
                            if pat[i] > 0.0:
                                sub = min(pat[i], -diff)
                                pat[i] -= sub
                                diff += sub
                                if diff > -1e-12:
                                    break
                    pat_h = project_feasible(pat, n_pat)
                    c_pat = compute_c5(pat_h, n_pat)
                    candidates.append((c_pat, pat_h))

        # Upsample current best as candidate
        hi_best = upsample(best_h, n_pat)
        c_hib = compute_c5(hi_best, n_pat)
        candidates.append((c_hib, hi_best))

        candidates.sort(key=lambda x: x[0])
        # Optimize the best 2 candidates
        for ci in range(min(2, len(candidates))):
            if time.time() > p4_end - 30:
                break
            cand_c, cand_h = candidates[ci]
            p4i_end = min(time.time() + 50, p4_end - 20)
            hi4, c4 = active_optimize(cand_h, n_pat, p4i_end, top_k=6,
                alphas=[10000, 50000, 200000, 1000000])
            if c4 < best_c:
                best_c = c4
                best_h = hi4
                best_n = n_pat

    # Phase 5: Binary swap search at best resolution (150s)
    if time.time() < deadline - 120:
        p5_end = min(time.time() + 150, deadline - 120)
        # Make a near-binary version
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, best_n)
        h5, c5 = swap_local_search(h_bin, best_n, p5_end)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: SLSQP polish (80s)
    if time.time() < deadline - 60:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 50:
                break
            p6_end = min(time.time() + 35, deadline - 45)
            h6, c6 = slqp_polish(best_h, best_n, p6_end, alpha=alpha, maxiter=1500)
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 7: Subgradient refinement (40s)
    if time.time() < deadline - 30:
        p7_end = min(time.time() + 40, deadline - 25)
        h7, c7 = subgrad_refine(best_h, best_n, p7_end, n_iter=3000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine pair search (remaining time - 10s)
    if time.time() < deadline - 10:
        p8_end = deadline - 3
        h8, c8 = fine_pair_search(best_h, best_n, p8_end, radius=40)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
