```python
import time
import numpy as np
from scipy.optimize import minimize


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def c5_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_box_sum(h, target):
    h = np.asarray(h, dtype=float)
    S = float(target)
    h = np.clip(h, 0.0, 1.0)

    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(h + mid, 0.0, 1.0))
        if val < S:
            lo = mid
        else:
            hi = mid

    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(4):
        diff = S - float(np.sum(h))
        if abs(diff) <= 1e-14:
            break
        free = np.flatnonzero((h > 1e-12) & (h < 1.0 - 1e-12))
        if free.size == 0:
            h = h + diff / max(1, h.size)
        else:
            h[free[0]] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def shift_value(h, n, k):
    dx = 2.0 / float(n)
    if k == 0:
        return dx * float(np.dot(h, 1.0 - h))
    if k > 0:
        nv = n - k
        return dx * float(np.dot(h[:nv], 1.0 - h[k:]))
    p = -k
    nv = n - p
    return dx * float(np.dot(h[p:], 1.0 - h[:nv]))


def shift_grad(h, n, k):
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    if k == 0:
        g = (1.0 - 2.0 * h) * dx
    elif k > 0:
        nv = n - k
        g[:nv] += (1.0 - h[k:]) * dx
        g[k:] += (-h[:nv]) * dx
    else:
        p = -k
        nv = n - p
        g[p:] += (1.0 - h[:nv]) * dx
        g[:nv] += (-h[p:]) * dx
    return g


def nlp_t_objective(z, n):
    return float(z[0])


def nlp_t_objective_jac(z, n):
    g = np.zeros(n + 1, dtype=float)
    g[0] = 1.0
    return g


def nlp_sum_fun(z, n):
    return float(np.sum(z[1:]) - 0.5 * float(n))


def nlp_sum_jac(z, n):
    g = np.zeros(n + 1, dtype=float)
    g[1:] = 1.0
    return g


def nlp_ineq_fun(z, n, active):
    h = z[1:]
    t = float(z[0])
    m = int(np.size(active))
    out = np.empty(m, dtype=float)
    for r in range(m):
        idx = int(active[r])
        k = idx - (n - 1)
        out[r] = t - shift_value(h, n, k)
    return out


def nlp_ineq_jac(z, n, active):
    h = z[1:]
    m = int(np.size(active))
    jac = np.zeros((m, n + 1), dtype=float)
    jac[:, 0] = 1.0
    for r in range(m):
        idx = int(active[r])
        k = idx - (n - 1)
        jac[r, 1:] = -shift_grad(h, n, k)
    return jac


def slqp_active_refine(h0, n, deadline):
    h = project_box_sum(h0, 0.5 * n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0

    active_sizes = (5, 8, 12, 16, 20, 24, 28)
    gap_base = 0.0008

    for outer in range(24):
        if time.time() > deadline - 25.0:
            break

        cvec = c5_vec(h, n)
        cmax = float(np.max(cvec))
        if cmax < best_c:
            best_c = cmax
            best_h = h.copy()

        size = active_sizes[min(outer, len(active_sizes) - 1)]
        gap = gap_base * (1.0 + 0.25 * float(outer))

        top_order = np.argsort(cvec)[::-1]
        active_all = top_order[cvec[top_order] >= cmax - gap]
        if active_all.size < size:
            active = top_order[:int(size)]
        else:
            active = active_all[:int(size)]
        if active.size < 3:
            active = top_order[:3]

        ub = min(1.0, cmax + 1.0e-6)
        t0 = cmax + 1.0e-9
        if t0 >= ub:
            t0 = max(0.0, ub - 1.0e-9)
        if t0 < 0.0:
            t0 = 0.0

        z0 = np.empty(n + 1, dtype=float)
        z0[0] = t0
        z0[1:] = h

        bounds = [(0.0, ub)] + [(0.0, 1.0)] * n
        constraints = (
            {
                "type": "ineq",
                "fun": nlp_ineq_fun,
                "jac": nlp_ineq_jac,
                "args": (n, active),
            },
            {
                "type": "eq",
                "fun": nlp_sum_fun,
                "jac": nlp_sum_jac,
                "args": (n,),
            },
        )

        res = None
        try:
            res = minimize(
                nlp_t_objective,
                z0,
                jac=nlp_t_objective_jac,
                args=(n,),
                bounds=bounds,
                constraints=constraints,
                method="SLSQP",
                options={"maxiter": 80, "ftol": 1.0e-12, "maxls": 30},
            )
        except Exception:
            res = None

        improved = False
        if res is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[1:], 0.5 * n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1.0e-13:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
                stall = 0
                improved = True
            elif c_new < cmax + 1.0e-11:
                h = h_new.copy()
            else:
                h = best_h.copy()

        if not improved:
            stall += 1
            if stall >= 3:
                break

    best_h = project_box_sum(best_h, 0.5 * n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def correlate_binary_int(b, n):
    a = b.astype(np.int32, copy=False)
    c = np.correlate(a, 1 - a, mode="full")
    c = np.asarray(c, dtype=np.int32)
    c[n - 1] = 0
    return c


def flip_bit_update(b, counts, i):
    x = int(b[i])
    n = int(b.size)
    if x == 0:
        counts[n - 1 - i : 2 * n - 1 - i] += 1 - b
        counts[i : i + n] -= b[::-1]
    else:
        counts[n - 1 - i : 2 * n - 1 - i] -= 1 - b
        counts[i : i + n] += b[::-1]
    b[i] = 1 - x
    counts[n - 1] = 0


def plateau_metric(counts, topk=20):
    k = int(topk)
    m = int(counts.size)
    if k <= 0 or m <= 0:
        return 0.0
    if k >= m:
        return float(np.sum(counts))
    part = np.partition(counts, m - k)[-k:]
    return float(np.sum(part))


def ensure_count(b, S, rng):
    b = np.asarray(b, dtype=np.int32).copy()
    S = int(S)
    cur = int(np.sum(b))
    if cur < S:
        need = S - cur
        if need > 0:
            zeros = np.flatnonzero(1 - b)
            if zeros.size < need:
                b[:] = 1
            else:
                perm = rng.choice(zeros, size=need, replace=False)
                b[perm] = 1
    elif cur > S:
        need = cur - S
        if need > 0:
            ones = np.flatnonzero(b)
            if ones.size < need:
                b[:] = 0
            else:
                perm = rng.choice(ones, size=need, replace=False)
                b[perm] = 0
    return b


def get_active_binary(counts, M):
    top = np.argsort(counts)[::-1]
    active = top[counts[top] >= M - 2]
    if active.size == 0:
        active = top[:1]
    if active.size > 40:
        active = active[:40]
    if active.size < 8:
        active = top[:8]
    return active


def active_weights_binary(counts, active, M):
    vals = np.asarray(counts[active], dtype=float)
    return np.exp(0.8 * (vals - float(M)))


def compute_binary_delta(b, active, weights, n):
    D = np.zeros(n, dtype=float)
    bf = b.astype(float, copy=False)
    omf = 1.0 - bf
    nm1 = n - 1
    for r in range(int(size := np.size(active))):
        idx = int(active[r])
        w = float(weights[r])
        if w == 0.0:
            continue
        k = idx - nm1
        if k == 0:
            continue
        if k > 0:
            nv = n - k
            D[:nv] += w * omf[k:]
            D[k:] -= w * bf[:nv]
        else:
            p = -k
            nv = n - p
            D[p:] += w * omf[:nv]
            D[:nv] -= w * bf[p:]
    return D


def binary_candidate_pairs(b, D, K=5, max_pairs=12, rng=None):
    ones = np.flatnonzero(b)
    zeros = np.flatnonzero(1 - b)
    if ones.size == 0 or zeros.size == 0:
        return []

    if ones.size > K:
        top1 = ones[np.argsort(D[ones])[-K:]]
    else:
        top1 = ones

    if zeros.size > K:
        top0 = zeros[np.argsort(D[zeros])[:K]]
    else:
        top0 = zeros

    pairs = []
    seen = []
    for i in top1:
        ii = int(i)
        for j in top0:
            jj = int(j)
            if ii == jj:
                continue
            key = (ii, jj)
            if key in seen:
                continue
            score = float(D[jj] - D[ii])
            if score < 0.0:
                seen.append(key)
                pairs.append((score, ii, jj))

    if rng is not None:
        for _ in range(4):
            ii = int(ones[rng.integers(0, ones.size)])
            jj = int(zeros[rng.integers(0, zeros.size)])
            if ii == jj:
                continue
            key = (ii, jj)
            if key in seen:
                continue
            score = float(D[jj] - D[ii])
            seen.append(key)
            pairs.append((score, ii, jj))

    pairs.sort()
    out = []
    for _, ii, jj in pairs[:int(max_pairs)]:
        out.append((ii, jj))
    return out


def binary_local_search(b_in, n, deadline, seed):
    rng = np.random.default_rng(seed)
    S = int(n) // 2

    b = np.asarray(b_in, dtype=np.int32)
    b = ensure_count(b, S, rng)
    counts = correlate_binary_int(b, n)

    M = int(counts.max())
    best_M = M
    best_b = b.copy()

    stall = 0
    kick_count = 0

    while time.time() < deadline:
        if M < best_M:
            best_M = M
            best_b = b.copy()

        active = get_active_binary(counts, M)
        weights = active_weights_binary(counts, active, M)
        D = compute_binary_delta(b, active, weights, n)

        old_metric = plateau_metric(counts, 20)
        pairs = binary_candidate_pairs(b, D, K=5, max_pairs=12, rng=rng)

        accepted = False
        for i, j in pairs:
            if time.time() > deadline - 5.0:
                break
            if b[i] != 1 or b[j] != 0:
                continue

            flip_bit_update(b, counts, i)
            flip_bit_update(b, counts, j)
            new_M = int(counts.max())

            accept = False
            if new_M < M:
                accept = True
            elif new_M == M:
                if plateau_metric(counts, 20) < old_metric - 1.0e-9:
                    accept = True

            if accept:
                M = new_M
                if M < best_M:
                    best_M = M
                    best_b = b.copy()
                stall = 0
                accepted = True
                break
            else:
                flip_bit_update(b, counts, j)
                flip_bit_update(b, counts, i)

        if not accepted:
            stall += 1
            if stall > 60:
                save_b = b.copy()
                save_counts = counts.copy()
                save_M = M

                b2 = b.copy()
                for _ in range(8):
                    ones = np.flatnonzero(b2)
                    zeros = np.flatnonzero(1 - b2)
                    if ones.size == 0 or zeros.size == 0:
                        break
                    ii = int(ones[rng.integers(0, ones.size)])
                    jj = int(zeros[rng.integers(0, zeros.size)])
                    if ii == jj:
                        continue
                    b2[ii] = 0
                    b2[jj] = 1

                counts2 = correlate_binary_int(b2, n)
                M2 = int(counts2.max())

                if M2 <= best_M + 2:
                    b = b2
                    counts = counts2
                    M = M2
                else:
                    b = save_b
                    counts = save_counts
                    M = save_M

                stall = 0
                kick_count += 1
                if kick_count > 50:
                    b = best_b.copy()
                    counts = correlate_binary_int(b, n)
                    M = best_M
                    kick_count = 0
            elif stall > 2000:
                b = best_b.copy()
                counts = correlate_binary_int(b, n)
                M = best_M
                stall = 0

    best_b = ensure_count(best_b, S, rng)
    best_counts = correlate_binary_int(best_b, n)
    best_M = int(best_counts.max())
    h_out = best_b.astype(float, copy=False)
    return h_out, compute_c5(h_out, n), best_M


def binary_top_from_values(h, n):
    S = int(n) // 2
    h = np.asarray(h, dtype=float)
    b = np.zeros(n, dtype=np.int32)
    if S <= 0:
        return b
    if S >= n:
        b[:] = 1
        return b
    idx = np.argpartition(-h, S - 1)[:S]
    b[idx] = 1
    return b


def make_random_binary(n, rng):
    return (rng.integers(0, 2, size=n)).astype(np.int32)


def make_beatty(n, alpha):
    x = np.mod(np.arange(n, dtype=float) * float(alpha), 1.0)
    return (x < 0.5).astype(np.int32)


def make_thue_morse(n):
    b = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        b[i] = b[i >> 1] ^ 1
    return b


def make_random_blocks(n, rng):
    b = np.empty(n, dtype=np.int32)
    pos = 0
    bit = 0
    while pos < n:
        L = int(rng.integers(3, 40))
        L = min(L, n - pos)
        b[pos : pos + L] = bit
        bit = 1 - bit
        pos += L
    return b


def binary_multi_search(parent, n, seed, deadline, current_h, current_c):
    rng = np.random.default_rng(seed)
    total = max(0.0, deadline - time.time())
    if total <= 0.0:
        return current_h, current_c

    starts = []
    starts.append(binary_top_from_values(parent, n))
    starts.append(binary_top_from_values(0.5 * (parent + parent[::-1]), n))
    starts.append(make_random_binary(n, rng))
    starts.append(make_beatty(n, 0.6180339887498949))
    starts.append(make_thue_morse(n))
    starts.append(make_random_blocks(n, rng))
    starts.append(make_beatty(n, 0.7548776662466927))

    hp = parent + 0.02 * rng.normal(size=n)
    starts.append(binary_top_from_values(project_box_sum(hp, 0.5 * n), n))

    weights = [0.28, 0.10, 0.12, 0.12, 0.10, 0.08, 0.08, 0.12]
    norm_w = sum(weights)
    weights = [w / norm_w for w in weights]

    for idx_w, b in enumerate(starts):
        if time.time() > deadline:
            break
        w = weights[idx_w] if idx_w < len(weights) else 0.1
        lim = min(deadline - time.time(), max(20.0, total * w))
        if time.time() + lim < deadline:
            lim = deadline - time.time()
        if lim <= 5.0:
            break

        h_c, c_c, _ = binary_local_search(b, n, time.time() + lim, seed + 1000 * idx_w)
        if c_c < current_c:
            current_c = c_c
            current_h = h_c

    return current_h, current_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    budget_s = max(1.0, float(budget_s))
    total_deadline = t_start + budget_s - 8.0

    rng = np.random.default_rng(seed)

    try:
        h_raw = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h_raw = np.full(500, 0.5, dtype=float)

    if h_raw.ndim != 1:
        h_raw = h_raw.reshape(-1)
    n0 = int(h_raw.size)
    if n0 == 0:
        n0 = 500
        h_raw = np.full(n0, 0.5, dtype=float)

    if (n0 % 2 != 0) or (n0 < 100) or (n0 > 2000):
        n = 500
        if n0 != n:
            if n0 == 1:
                x_old = np.array([0.0, 1.0], dtype=float)
            else:
                x_old = np.linspace(0.0, 2.0, n0)
            x_new = np.linspace(0.0, 2.0, n)
            h_raw = np.interp(x_new, x_old, h_raw)
    else:
        n = n0

    h0 = project_box_sum(h_raw, 0.5 * n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    reserve = max(60.0, min(500.0, 0.40 * budget_s))
    bin_deadline = min(total_deadline - reserve, t_start + 0.60 * budget_s)
    if bin_deadline > time.time() + 25.0:
        best_h, best_c = binary_multi_search(h0, n, int(seed) + 1, bin_deadline, best_h, best_c)

    slqp_deadline = total_deadline - 5.0
    queue = []
    queue.append(best_h.copy())
    queue.append(h0.copy())
    rev_avg = project_box_sum(0.5 * (h0 + h0[::-1]), 0.5 * n)
    queue.append(rev_avg)
    if time.time() < slqp_deadline - 80.0:
        h_noise1 = project_box_sum(best_h + 0.02 * rng.normal(size=n), 0.5 * n)
        h_noise2 = project_box_sum(h0 + 0.05 * rng.normal(size=n), 0.5 * n)
        queue.append(h_noise1)
        queue.append(h_noise2)

    while queue and time.time() < slqp_deadline - 45.0:
        cand = queue.pop(0)
        cand = project_box_sum(cand, 0.5 * n)
        call_deadline = min(slqp_deadline - 25.0, time.time() + 150.0)
        if call_deadline <= time.time() + 5.0:
            break

        h_c, c_c = slqp_active_refine(cand, n, call_deadline)
        if c_c < best_c - 1.0e-13:
            best_c = c_c
            best_h = h_c.copy()
            queue.append(h_c.copy())
            if len(queue) > 8:
                queue = queue[-8:]

    best_h = project_box_sum(best_h, 0.5 * n)
    best_c = compute_c5(best_h, n)
    return np.asarray(best_h, dtype=float), float(best_c), int(n)
```
