            if tot < 1e-15:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)

        nv = n - kk

            grad[0:nv] += (-h[kk:n]) * dx
    return grad

            c_high = best_c

        # L-BFGS at high resolution (50s)
        t6 = min(time.time() + 50, deadline - 180)
        if t6 > time.time() + 3:
            h_high, c_high = lbfgs_anneal(h_high, n_high, t6,
                alphas=[100, 500, 2000, 8000, 30000], mu=2.0)

        # SQP at high resolution (60s)
        t7 = min(time.time() + 60, deadline - 100)
        if t7 > time.time() + 3:
            h_high, c_high = sqp_optimize(h_high, n_high, t7, max_iter=200, rho_init=0.04)

        # Local search at high resolution (40s)
        t8 = min(time.time() + 40, deadline - 30)
        if t8 > time.time() + 3:
            h_high, c_high = local_search_fine(h_high, n_high, t8, n_top=25, radius=20)

        # Final SQP at high resolution
        if time.time() < deadline - 15:
            h_high, c_high = sqp_optimize(h_high, n_high, deadline - 15, max_iter=100, rho_init=0.015)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high

    # Phase 7: Perturbed restart on best solution
    if time.time() < deadline - 40:
        best_h, best_c = perturbed_restart(best_h, len(best_h), deadline - 15,
                                           n_trials=2, scale=0.003)

    # Phase 8: Final SLSQP polish
    if time.time() < deadline - 8:
        h_final, c_final = slqp_polish(best_h, len(best_h), deadline - 3,
                                       alpha=1000000, maxiter=1500)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
