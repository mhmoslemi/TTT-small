```python
    # Similarly for h[j] (0→1):
    #   term3 = (1-h[j-k]) if 0 <= j-k < n else 0  (h[j] changes 0→1 in h[n+k] slot)
    #   term4 = h[j+k] if 0 <= j+k < n else 0     (h[j] changes 0→1 in (1-h[n]) slot)
    #   delta_c[k] += term3 - term4... 
    
    # Wait, I need to be more careful about signs.
    
    # Original: c[k] = sum_m h[m+k]*(1-h[m])
    # After flipping h[i] from 1 to 0:
    #   The term m=i: was h[i+k]*(1-h[i]) = h[i+k]*0 = 0, becomes h[i+k]*1 = h[i+k]
    #     Change: +h[i+k] (if i is a valid m, i.e., 0<=i<n which is always true,
    #              and i+k is valid, i.e., 0<=i+k<n)
    #   The term m such that m+k=i, i.e., m=i-k: was h[i]*(1-h[i-k]) = 1*(1-h[i-k]),
    #     becomes 0*(1-h[i-k]) = 0
    #     Change: -(1-h[i-k]) (if i-k is a valid m, i.e., 0<=i-k<n,
    #              and i is valid as m+k, i.e., 0<=i<n which is always true)
    
    # After flipping h[j] from 0 to 1:
    #   The term m=j: was h[j+k]*(1-h[j]) = h[j+k]*1 = h[j+k], becomes h[j+k]*0 = 0
    #     Change: -h[j+k] (if 0<=j+k<n)
    #   The term m=j-k: was h[j]*(1-h[j-k]) = 0*(1-h[j-k]) = 0, becomes 1*(1-h[j-k])
    #     Change: +(1-h[j-k]) (if 0<=j-k<n)
    
    # Total delta for shift k:
    # Δc[k] = h[i+k]*[valid] - (1-h[i-k])*[valid] - h[j+k]*[valid] + (1-h[j-k])*[valid]
    
    # In terms of full index p = n-1+k, k = p-(n-1):
    # Δc_full[p] = h[i+p-(n-1)]*[valid] - (1-h[i-(p-(n-1))])*[valid] - h[j+p-(n-1)]*[valid] + (1-h[j-(p-(n-1))])*[valid]
    
    # This is O(n) per swap. Let me vectorize it.
    
    p = np.arange(2*n-1)
    k = p - (n-1)
    
    # Index into h for h[i+k]:
    ik = i + k
    jk = j + k
    ik_prev = i - k  # this is i - (p-(n-1)) = i-k
    jk_prev = j - k
    
    valid_ik = (ik >= 0) & (ik < n)
    valid_jk = (jk >= 0) & (jk < n)
    valid_ikp = (ik_prev >= 0) & (ik_prev < n)
    valid_jkp = (jk_prev >= 0) & (jk_prev < n)
    
    delta = np.zeros(2*n-1)
    # Clip indices for safe access
    ik_safe = np.clip(ik, 0, n-1)
    jk_safe = np.clip(jk, 0, n-1)
    ikp_safe = np.clip(ik_prev, 0, n-1)
    jkp_safe = np.clip(jk_prev, 0, n-1)
    
    delta[valid_ik] += h[ik_safe[valid_ik]]
    delta[valid_ikp] -= (1 - h[ikp_safe[valid_ikp]])
    delta[valid_jk] -= h[jk_safe[valid_jk]]
    delta[valid_jkp] += (1 - h[jkp_safe[valid_jkp]])
    
    return delta
```

Actually, for the simulated annealing, I don't need the full delta array. I just need the new maximum. So I can compute:
- new_r = r + delta
- new_max = new_r.max()

And I only accept if new_max < current_max (or with probability exp(-(new_max - current_max)/T) if it's worse).

Let me think about efficiency. For n=250, the correlation array has 499 elements. Computing the delta involves a few vector operations on arrays of length 499. Each numpy vector op on 499 elements takes maybe 0.5-1μs. So total per swap: maybe 5-10μs. That gives 100,000-200,000 swaps per second.

Over 1100 seconds, that's 10^8 - 2*10^8 swaps. More than enough.

```python
import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr_full(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def get_swap_delta(h, r, i, j, n):
    """Compute change in full correlation when swapping h[i](1→0) and h[j](0→1)."""
    p = np.arange(2 * n - 1)
    k = p - (n - 1)
    
    delta = np.zeros(2 * n - 1)
    
    ik = i + k
    valid_ik = (ik >= 0) & (ik < n)
    if valid_ik.any():
        delta[valid_ik] += h[np.clip(ik, 0, n-1)[valid_ik]]
    
    imk = i - k
    valid_imk = (imk >= 0) & (imk < n)
    if valid_imk.any():
        delta[valid_imk] -= (1 - h[np.clip(imk, 0, n-1)[valid_imk]])
    
    jk = j + k
    valid_jk = (jk >= 0) & (jk < n)
    if valid_jk.any():
        delta[valid_jk] -= h[np.clip(jk, 0, n-1)[valid_jk]]
    
    jmk = j - k
    valid_jmk = (jmk >= 0) & (jmk < n)
    if valid_jmk.any():
        delta[valid_jmk] += (1 - h[np.clip(jmk, 0, n-1)[valid_jmk]])
    
    return delta

def simulated_annealing_binary(h, n, deadline, rng, T0=1.0, T1=0.001, n_iter_max=5000000):
    """Simulated annealing on binary solutions with swap moves."""
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    r = compute_corr_full(h, n)
    dx = 2.0 / n
    r_scaled = r * dx
    current_c = r_scaled.max()
    best_c = current_c
    best_h = h.copy()
    
    T = T0
    cooling = (T1 / T0) ** (1.0 / max(n_iter_max, 1))
    
    # Precompute k array
    p = np.arange(2 * n - 1)
    k_arr = p - (n - 1)
    
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        
        # Choose random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        ik = i + k_arr
        imk = i - k_arr
        jk = j + k_arr
        jmk = j - k_arr
        
        delta = np.zeros(2 * n - 1)
        
        valid = (ik >= 0) & (ik < n)
        if valid.any():
            delta[valid] += h[np.clip(ik, 0, n-1)[valid]]
        
        valid = (imk >= 0) & (imk < n)
        if valid.any():
            delta[valid] -= (1 - h[np.clip(imk, 0, n-1)[valid]])
        
        valid = (jk >= 0) & (jk < n)
        if valid.any():
            delta[valid] -= h[np.clip(jk, 0, n-1)[valid]]
        
        valid = (jmk >= 0) & (jmk < n)
        if valid.any():
            delta[valid] += (1 - h[np.clip(jmk, 0, n-1)[valid]])
        
        new_r = r + delta
        new_c = new_r.max() * dx
        dc = new_c - current_c
        
        if dc < 0 or rng.random() < np.exp(-dc / T):
            # Accept
            h[i], h[j] = 0.0, 1.0
            r = new_r
            current_c = new_c
            # Update ones and zeros
            # i was 1, now 0: move from ones to zeros
            # j was 0, now 1: move from zeros to ones
            ones = np.delete(ones, np.where(ones == i)[0][0])
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        
        T *= cooling
    
    return best_h, best_c
```
