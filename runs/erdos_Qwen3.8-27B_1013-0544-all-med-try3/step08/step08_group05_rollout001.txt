import numpy as np
import time
from scipy.optimize import minimize
from scipy.fftpack import next_fast_len

_G_N = 0
_G_ALPHA = 1.0
_G_MU = 0.0
_G_TARGET = 0.0
_G_STASH = None
_G_CALL_DEADLINE = 0.0
_G_ITER = 0


def _nfft(m):
    m = int(max(2, m))
    try:
        return int(next_fast_len(m))
    except Exception:
        n = 1
        while n < m:
            n <<= 1
        return n


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    if h.size != n:
        raise ValueError("project_feasible: size mismatch")

    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    lo = -1.0 - float(np.min(h))
    hi = 1.0 - float(np.max(h))
    for _ in range(52):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if val > target:
            hi = mid
        else:
            lo = mid

    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        rem = float(diff)
        for idx in order:
            idx = int(idx)
            if rem > 0.0:
                add = min(1.0 - h[idx], rem)
                if add > 1e-18:
                    h[idx] += add
                    rem -= add
                    if abs(rem) < 1e-15:
                        break
            elif rem < 0.0:
                sub = min(h[idx], -rem)
                if sub > 1e-18:
                    h[idx] -= sub
                    rem += sub
                    if abs(rem) < 1e-15:
                        break
    return h


def exact_c(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def consider_update(h, n, best_h, best_c, best_n):
    if h is None:
        return best_h, best_c, best_n, False
    try:
        h = np.asarray(h, dtype=float)
        n = int(n)
        if h.shape[0] != n:
            return best_h, best_c, best_n, False
        h = project_feasible(h, n)
        c = exact_c(h, n)
        if np.isfinite(c) and c < best_c - 1e-15:
            return h.copy(), float(c), n, True
    except Exception:
        pass
    return best_h, best_c, best_n, False


def resample(h, n):
    h = np.asarray(h, dtype=float)
    m = h.size
    n = int(n)
    if m == n:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n),
                     np.linspace(0.0, 2.0, m),
                     h)


def compute_q(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    off = n - 1
    m = 2 * n - 1

    s = np.arange(-off, n, dtype=int)
    abs_s = np.abs(s)
    L = n - abs_s

    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    prefix[1:] = np.cumsum(h, dtype=float)
    total = prefix[n]

    term1 = np.empty(m, dtype=float)
    pos = s >= 0
    term1[pos] = prefix[L[pos]]
    term1[~pos] = total - prefix[abs_s[~pos]]

    N = _nfft(2 * n - 1)
    fh = np.fft.rfft(h, N)
    ac = np.fft.irfft(fh * np.conjugate(fh), N)[:n]
    term2 = ac[abs_s]

    q = term1 - term2
    q = np.asarray(q, dtype=float)
    q[~np.isfinite(q)] = 0.0
    q[np.abs(q) < 1e-14] = 0.0
    return q


def softmax_values(q, alpha):
    q = np.asarray(q, dtype=float)
    m = len(q)
    if not np.any(np.isfinite(q)):
        return 0.0, np.full(m, 1.0 / m)

    qf = np.where(np.isfinite(q), q, -1e300)
    mx = float(np.max(qf))

    if alpha <= 0:
        return mx, np.full(m, 1.0 / m)

    y = float(alpha) * (qf - mx)
    y = np.where(np.isfinite(y), y, -1e300)
    se = np.exp(y)
    ss = float(se.sum())

    if ss <= 0.0 or not np.isfinite(ss):
        active = (qf >= mx - 1e-12).astype(float)
        if active.sum() <= 0:
            active = np.zeros(m)
            active[int(np.argmax(qf))] = 1.0
        active /= active.sum()
        return mx, active

    p = se / ss
    f = mx + float(np.log(ss)) / float(alpha)
    if not np.isfinite(f):
        f = mx
    return float(f), p


def compute_grad(h, n, p):
    h = np.asarray(h, dtype=float)
    p = np.asarray(p, dtype=float)
    n = int(n)

    ps = float(p.sum())
    if ps <= 0.0 or not np.isfinite(ps):
        p = np.full(p.shape, 1.0 / max(1, p.size), dtype=float)
    else:
        p = p / ps

    off = n - 1
    N = _nfft(3 * n - 2)

    c1 = np.fft.irfft(np.fft.rfft(1.0 - h, N) * np.fft.rfft(p[::-1], N), N)
    c2 = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(p, N), N)
    grad = c1[off:off + n] - c2[off:off + n]
    grad = np.asarray(grad, dtype=float)
    grad[~np.isfinite(grad)] = 0.0
    return grad


def smooth_fg(h, n, alpha):
    q = compute_q(h, n)
    f, p = softmax_values(q, alpha)
    grad = compute_grad(h, n, p)
    return f, q, p, grad


def _opt_fun(x):
    global _G_STASH
    try:
        val, grad = smooth_fg(np.asarray(x, dtype=float), _G_N, _G_ALPHA)
        ssum = float(np.sum(x) - _G_TARGET)
        if _G_MU != 0.0:
            val = float(val) + 0.5 * float(_G_MU) * ssum * ssum
            grad = grad + float(_G_MU) * ssum
        if not np.isfinite(val) or not np.all(np.isfinite(grad)):
            _G_STASH = np.zeros_like(np.asarray(x, dtype=float))
            return 1e100
        _G_STASH = grad
        return float(val)
    except Exception:
        _G_STASH = np.zeros(_G_N, dtype=float)
        return 1e100


def _opt_jac(x):
    if _G_STASH is None:
        return _opt_fun(x)
    return _G_STASH


def _sum_fun(x):
    return float(np.sum(x) - _G_TARGET)


def _sum_jac(x):
    return np.ones(_G_N, dtype=float)


def _lbfgs_callback(xk):
    global _G_ITER
    _G_ITER += 1
    return time.time() > _G_CALL_DEADLINE


def run_lbfgs(h, n, alpha, mu, maxiter, deadline):
    global _G_N, _G_ALPHA, _G_MU, _G_TARGET, _G_CALL_DEADLINE, _G_ITER
    try:
        x0 = project_feasible(np.asarray(h, dtype=float), n)
        _G_N = int(n)
        _G_ALPHA = float(alpha)
        _G_MU = float(mu)
        _G_TARGET = 0.5 * _G_N
        _G_CALL_DEADLINE = float(deadline)
        _G_ITER = 0

        bounds = [(0.0, 1.0)] * _G_N
        opts = {
            "maxiter": int(maxiter),
            "ftol": 1e-14,
            "gtol": 1e-6,
            "maxcor": 20,
            "maxls": 20,
            "callback": _lbfgs_callback,
        }
        res = minimize(_opt_fun, x0, method="L-BFGS-B", jac=_opt_jac,
                       bounds=bounds, options=opts)
        xres = getattr(res, "x", x0)
        if not np.all(np.isfinite(xres)):
            xres = x0
        return project_feasible(xres, _G_N)
    except Exception:
        return project_feasible(np.asarray(h, dtype=float), n)


def run_slsqp(h, n, alpha, maxiter):
    global _G_N, _G_ALPHA, _G_MU, _G_TARGET
    try:
        x0 = project_feasible(np.asarray(h, dtype=float), n)
        _G_N = int(n)
        _G_ALPHA = float(alpha)
        _G_MU = 0.0
        _G_TARGET = 0.5 * _G_N

        bounds = [(0.0, 1.0)] * _G_N
        cons = ({'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac},)
        opts = {"maxiter": int(maxiter), "ftol": 1e-13}

        res = minimize(_opt_fun, x0, method="SLSQP", jac=_opt_jac,
                       bounds=bounds, constraints=cons, options=opts)
        xres = getattr(res, "x", x0)
        if not np.all(np.isfinite(xres)):
            xres = x0
        return project_feasible(xres, _G_N)
    except Exception:
        return project_feasible(np.asarray(h, dtype=float), n)


def smooth_descent(h, n, alpha, deadline, maxiter, seed, step0=0.05):
    rng = np.random.RandomState(seed)
    h = project_feasible(np.asarray(h, dtype=float), n)
    n = int(n)
    dx = 2.0 / n

    q = compute_q(h, n)
    f, p = softmax_values(q, alpha)
    cur_m = float(np.max(q))
    best_m = cur_m
    best_h = h.copy()
    grad = compute_grad(h, n, p)

    step = float(step0)
    fail = 0

    for _ in range(int(maxiter)):
        if time.time() > deadline:
            break
        if not np.all(np.isfinite(grad)):
            break

        accepted = False
        for _ls in range(22):
            cand = project_feasible(h - step * grad, n)
            qc = compute_q(cand, n)
            fc, pc = softmax_values(qc, alpha)
            if fc < f - 1e-9:
                accepted = True
                break
            step *= 0.5

        if accepted:
            h = cand
            q = qc
            f = fc
            p = pc
            cur_m = float(np.max(q))
            if cur_m < best_m - 1e-12:
                best_m = cur_m
                best_h = h.copy()
            grad = compute_grad(h, n, p)
            step = min(step * 1.2, 0.5)
            fail = 0
        else:
            h = project_feasible(h + 0.025 * rng.normal(0.0, 1.0, n), n)
            q = compute_q(h, n)
            f, p = softmax_values(q, alpha)
            cur_m = float(np.max(q))
            if cur_m < best_m - 1e-12:
                best_m = cur_m
                best_h = h.copy()
            grad = compute_grad(h, n, p)
            step = float(step0)
            fail += 1
            if fail > 18:
                break

    return best_h, best_m * dx


def binary_threshold(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if h.size != n:
        h = resample(h, n)
    out = np.zeros(n, dtype=float)
    k = int(n // 2)
    if k > 0:
        idx = np.argpartition(-h, k)[:k]
        out[idx] = 1.0
    return out


def binary_local_search(hstart, n, deadline, seed=42):
    try:
        n = int(n)
        if n < 4 or n % 2 != 0:
            return None, None
        target = n // 2
        hstart = np.asarray(hstart, dtype=float)
        if hstart.size != n:
            hstart = resample(hstart, n)

        h = np.zeros(n, dtype=float)
        idx = np.argpartition(-hstart, target)[:target]
        h[idx] = 1.0
        uh = 1.0 - h
        dx = 2.0 / n

        corr = np.correlate(h, uh, mode="full")
        curr = float(np.max(corr) * dx)
        best = curr
        best_h = h.copy()

        ones_list = [int(i) for i in np.where(h == 1.0)[0]]
        zeros_list = [int(i) for i in np.where(h == 0.0)[0]]
        rng = np.random.RandomState(seed)

        T = 0.006
        stag = 0
        evals = 0

        while time.time() < deadline:
            if not ones_list or not zeros_list:
                break

            a = int(ones_list[rng.randint(len(ones_list))])
            b = int(zeros_list[rng.randint(len(zeros_list))])

            h[a] = 0.0
            uh[a] = 1.0
            h[b] = 1.0
            uh[b] = 0.0

            c = float(np.max(np.correlate(h, uh, mode="full")) * dx)

            if c <= curr + 1e-15:
                accept = True
            else:
                if T <= 1e-12:
                    accept = False
                else:
                    u = float(rng.random())
                    if u <= 1e-300:
                        accept = False
                    else:
                        accept = (np.log(1.0 - u) < (curr - c) / T)

            if accept:
                try:
                    ones_list.remove(a)
                except Exception:
                    pass
                try:
                    zeros_list.remove(b)
                except Exception:
                    pass
                ones_list.append(b)
                zeros_list.append(a)
                curr = c
                T *= 0.999
                if c < best - 1e-15:
                    best = c
                    best_h = h.copy()
                    stag = 0
                else:
                    stag += 1
            else:
                h[a] = 1.0
                uh[a] = 0.0
                h[b] = 0.0
                uh[b] = 1.0
                stag += 1

            evals += 1
            if evals % 1000 == 0:
                T *= 0.9
                if T < 1e-5:
                    T = 1e-5
                if stag > 8000:
                    h = best_h.copy()
                    uh = 1.0 - h
                    curr = best
                    ones_list = [int(i) for i in np.where(h == 1.0)[0]]
                    zeros_list = [int(i) for i in np.where(h == 0.0)[0]]
                    T = 0.005
                    stag = 0

        return best_h, best
    except Exception:
        return None, None
