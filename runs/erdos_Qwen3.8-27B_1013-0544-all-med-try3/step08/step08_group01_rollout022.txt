def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    # Project h to {0 ≤ h ≤ 1, sum(h) = n/2}
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            scale = target / h.sum()
            h *= scale
            h = np.clip(h, 0, 1)
    return h

def softmax_obj(h, n, alpha, mu, active_mask=None):
    c = compute_c5_vec(h, n)
    if active_mask is not None:
        c = c[active_mask]
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        logsumexp = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        logsumexp = np.log(np.sum(np.exp(sh)))
    val = m + logsumexp / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(h, n, alpha, mu, active_mask=None):
    c = compute_c5_vec(h, n)
    if active_mask is not None:
        idx = np.where(active_mask)[0]
    else:
        idx = np.arange(len(c))
    c_sel = c[idx]
    m = c_sel.max()
    sh = alpha * (c_sel - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, mm in zip(w, idx):
        if wi < 1e-15:
            continue
        k = mm - (n - 1)
        wm = wi * dx

    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def sm_obj(x, n, alpha, mu, amask):
    c = compute_c5_vec(x, n)
    if amask is not None:
        c = c[amask]
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400.0:
        lse = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        lse = np.log(np.sum(np.exp(sh)))
    val = m + lse / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def sm_grad(x, n, alpha, mu, amask):
    c = compute_c5_vec(x, n)
    if amask is not None:
        idx = np.where(amask)[0]
    else:
        idx = np.arange(2 * n - 1)
    c_s = c[idx]
    m = c_s.max()
    sh = alpha * (c_s - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, mm in zip(w, idx):
        if wi < 1e-16:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def get_active_set(h, n, top_k=5, frac=0.005):
    c = compute_c5_vec(h, n)
    cmax = c.max()
    thresh = cmax * (1.0 - frac)
    amask = c >= thresh
    if amask.sum() > top_k * 3:
        top_idx = np.argsort(-c)[:top_k * 3]
        amask = np.zeros(2 * n - 1, dtype=bool)
        amask[top_idx] = True
    elif amask.sum() < top_k:
        top_idx = np.argsort(-c)[:top_k]
        amask = np.zeros(2 * n - 1, dtype=bool)
        amask[top_idx] = True
    return amask


def run_lbhfgs(h, n, alpha, mu, amask, deadline, maxiter=20000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='L-BFGS-B',
                       bounds=bounds, args=(n, alpha, mu, amask),
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-18, 'maxfun': maxiter * 4,
                                'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if time.time() > deadline:
            return h.copy(), compute_c5(h, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)


def iterative_active_opt(h, n, deadline, alphas, top_k=5, mu=0.5, max_iter_calls=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        for _ in range(max_iter_calls):
            if time.time() > deadline - 5:
                break
            amask = get_active_set(best_h, n, top_k=top_k)
            h_new, c_new = run_lbhfgs(best_h, n, alpha, mu, amask, deadline,
                                       maxiter=15000)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
            else:
                break
    return best_h, best_c


def pair_moves(h, n, deadline, max_trials_per_delta=3000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.default_rng(12345)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline - 2:
            break
        for _ in range(max_trials_per_delta):
            if time.time() > deadline - 1:
                return best_h, best_c
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                h[i] += d
                h[j] -= d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
            if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                h[i] -= d
                h[j] += d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                else:
                    h[i] += d
                    h[j] -= d
    return best_h, best_c


def structured_pair_moves(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    # Focus on pairs where one is in "high" region and other in "low" region
    high_mask = h > 0.6
    low_mask = h < 0.4
    high_idx = np.where(high_mask)[0]
    low_idx = np.where(low_mask)[0]
    if len(high_idx) == 0 or len(low_idx) == 0:
        return best_h, best_c
    rng = np.random.default_rng(67890)
    deltas = [0.01, 0.005, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline - 2:
            break
        for _ in range(2000):
            if time.time() > deadline - 1:
                return best_h, best_c
            i = high_idx[rng.integers(0, len(high_idx))]
            j = low_idx[rng.integers(0, len(low_idx))]
            # Move mass from high to low (makes function more uniform in those spots)
            if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                h[i] -= d
                h[j] += d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                else:
                    h[i] += d
                    h[j] -= d
            # Move mass from low to high
            if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                h[i] += d
                h[j] -= d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
    return best_h, best_c


def coarse_fine_cycle(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x_fine = np.linspace(0, 2, n)
    for n_coarse in [n // 2, n // 4]:
        if n_coarse < 50 or time.time() > deadline - 20:
            continue
        x_coarse = np.linspace(0, 2, n_coarse)
        h_c = np.interp(x_coarse, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)
        # Optimize at coarse level
        amask_c = get_active_set(h_c, n_coarse, top_k=5)
        h_c, c_c = run_lbhfgs(h_c, n_coarse, 1e5, 1.0, amask_c, deadline - 10, maxiter=20000)
        amask_c2 = get_active_set(h_c, n_coarse, top_k=3)
        h_c2, c_c2 = run_lbhfgs(h_c, n_coarse, 1e7, 0.1, amask_c2, deadline - 5, maxiter=20000)
        if c_c2 < c_c:
            h_c, c_c = h_c2, c_c2
        # Upsample
        h_up = np.interp(x_fine, x_coarse, h_c)
        h_up = project_feasible(h_up, n)
        c_up = compute_c5(h_up, n)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up.copy()
    return best_h, best_c


def gradient_polish(h, n, deadline, base_step=0.005):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 1e5
    step = base_step
    stall_count = 0
    while time.time() < deadline - 3 and step > 1e-7:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val * 0.999
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(-c_vec)[:3]
        # Compute gradient for active shifts
        grad = sm_grad(h, n, alpha, 0.0, None)
        # Only use active shift contributions
        grad_active = np.zeros(n)
        for mm in active:
            k = mm - (n - 1)
            if k >= 0:
                nv = n - k
                grad_active[0:nv] += (1.0 - h[k:n])
                grad_active[k:n] += (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad_active[kk:n] += (1.0 - h[0:nv])
                grad_active[0:nv] += (-h[kk:n])
        m = len(active)
        grad_active /= m
        # Find best 2-point move in negative gradient direction
        top_g = np.argsort(-np.abs(grad_active))[:80]
        improved = False
        for ii in range(len(top_g)):
            if time.time() > deadline - 3:
                return best_h, best_c
            i = top_g[ii]
            for jj in range(len(top_g)):
                if ii == jj:
                    continue
                j = top_g[jj]
                gi, gj = grad_active[i], grad_active[j]
                if abs(gi - gj) < 1e-10:
                    continue
                if gi > gj:
                    # decrease h[i], increase h[j]
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                else:
                    # increase h[i], decrease h[j]
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
        if improved:
            stall_count = 0
            step = min(step * 1.2, base_step)
        else:
            stall_count += 1
            step *= 0.5
            if stall_count > 10:
                break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Active-set optimization, moderate to high alpha (top-5)
    if time.time() < deadline - 500:
        t1_end = min(t0 + 120, deadline - 500)
        h, c = iterative_active_opt(h, n, t1_end,
                                     alphas=[1e4, 5e4, 2e5],
                                     top_k=5, mu=1.0, max_iter_calls=2)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: High alpha, top-5 (more iterations)
    if time.time() < deadline - 400:
        t2_end = min(time.time() + 150, deadline - 400)
        h, c = iterative_active_opt(best_h, n, t2_end,
                                     alphas=[5e5, 2e6, 1e7],
                                     top_k=5, mu=0.5, max_iter_calls=2)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            h = h.copy()

    # Phase 3: Very high alpha, top-3 focus
    if time.time() < deadline - 300:
        t3_end = min(time.time() + 150, deadline - 300)
        h, c = iterative_active_opt(best_h, n, t3_end,
                                     alphas=[5e6, 2e7, 1e8, 5e8],
                                     top_k=3, mu=0.1, max_iter_calls=2)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            h = h.copy()

    # Phase 4: Full (no active mask) with high alpha for global view
    if time.time() < deadline - 250:
        t4_end = min(time.time() + 80, deadline - 250)
        for alpha in [2e5, 1e6]:
            if time.time() > t4_end:
                break
            h2, c2 = run_lbhfgs(best_h, n, alpha, 1.0, None, t4_end, maxiter=15000)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()

    # Phase 5: Active-set with recomputed mask, very sharp
    if time.time() < deadline - 200:
        t5_end = min(time.time() + 80, deadline - 200)
        amask = get_active_set(best_h, n, top_k=3, frac=0.001)
        h5, c5 = run_lbhfgs(best_h, n, 1e8, 0.05, amask, t5_end, maxiter=30000)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
        amask2 = get_active_set(best_h, n, top_k=5, frac=0.002)
        h5b, c5b = run_lbhfgs(best_h, n, 5e7, 0.05, amask2, t5_end + 20, maxiter=30000)
        if c5b < best_c:
            best_c = c5b
            best_h = h5b.copy()

    # Phase 6: Coarse-fine cycle
    if time.time() < deadline - 150:
        h6, c6 = coarse_fine_cycle(best_h, n, deadline - 50)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()

    # Phase 7: Multi-start perturbations
    if time.time() < deadline - 100:
        rng = np.random.default_rng(seed)
        for trial in range(4):
            if time.time() > deadline - 90:
                break
            scale = 0.02 * (trial + 1)
            h_pert = best_h + rng.normal(0, scale, n)
            h_pert = project_feasible(h_pert, n)
            amask_p = get_active_set(h_pert, n, top_k=5)
            h_p, c_p = run_lbhfgs(h_pert, n, 1e6, 0.5, amask_p, deadline - 80, maxiter=15000)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p.copy()

    # Phase 8: Gradient polish
    if time.time() < deadline - 50:
        t8_end = min(time.time() + 40, deadline - 50)
        h8, c8 = gradient_polish(best_h, n, t8_end, base_step=0.003)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()

    # Phase 9: Structured pair moves
    if time.time() < deadline - 25:
        t9_end = min(time.time() + 20, deadline - 25)
        h9, c9 = structured_pair_moves(best_h, n, t9_end)
        if c9 < best_c:
            best_c = c9
            best_h = h9.copy()

    # Phase 10: Final very sharp active-set optimization
    if time.time() < deadline - 5:
        amask_f = get_active_set(best_h, n, top_k=3, frac=0.001)
        h10, c10 = run_lbhfgs(best_h, n, 1e9, 0.01, amask_f, deadline - 2, maxiter=50000)
        if c10 < best_c:
            best_c = c10
            best_h = h10.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
