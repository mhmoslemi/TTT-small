```python
            valid = B > A[ai]
            if not valid.any():
                continue
            valid_idx = np.where(valid)[0]
            vals = new_maxs[valid_idx]
            min_idx = valid_idx[np.argmin(vals)]
            min_val = vals[np.argmin(vals)]
            
            if min_val * dx < best_new_c_in_sweep - 1e-15:
                best_new_c_in_sweep = min_val * dx
                best_i_global = ai
                best_j_global = min_idx
                best_corr_global = (base + deltas_add[min_idx]).copy()
        
        if best_i_global >= 0:
            i = ones[best_i_global]
            j = zeros[best_j_global]
            h[i], h[j] = 0.0, 1.0
            corr = best_corr_global
            cur_c = best_new_c_in_sweep
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c
```

**Faster soft optimization: Projected gradient descent on the max**

The objective f(h) = max_k corr_k(h) * dx is not differentiable (max), but we can use subgradient methods:
- At the current h, find the active set A = {k : corr_k(h) = max}
- The subgradient is the average of the gradients of the active functions
- Take a step in the negative subgradient direction
- Project back to {0≤h≤1, sum(h)=n/2}

The gradient of corr_k(h) with respect to h[m] is:
∂corr_k/∂h[m] = (1-h[m+k]) - h[m-k] (with 0 for out-of-range)

This is O(n) to compute for one k, and O(n*|A|) for all active k's.

Let me implement this:

```python
def soft_optimize(h_init, n, deadline, lr=0.01, max_iter=100000):
    h = h_init.copy().astype(float)
    dx = 2.0 / n
    shifts = np.arange(-(n-1), n)
    target = n / 2.0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1-h, mode='full')
        max_val = corr.max()
        # Active set: shifts where corr is within epsilon of max
        active = np.where(corr >= max_val - 0.01)[0]
        
        # Subgradient (average of active gradients)
        grad = np.zeros(n)
        for ai in active:
            s = shifts[ai]
            for m in range(n):
                g = 0.0
                mp = m + s
                if 0 <= mp < n: g += (1 - h[mp])
                mm = m - s
                if 0 <= mm < n: g -= h[mm]
                grad[m] += g
        grad /= len(active)
        
        # Gradient descent step
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        # Fix sum
        diff = target - h_new.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                room = 1.0 - h_new
                s = room.sum()
                if s > 1e-10:
                    h_new = h_new + diff / s * room
            else:
                s = h_new.sum()
                if s > 1e-10:
                    h_new = h_new + diff / s * h_new
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    
    return h
```

But this is O(n * |active| * n) per iteration, which is O(n³) if all shifts are active. Too slow.

```python
# For shift s, grad at position m = (1-h[m+s]) - h[m-s]

mask_p = (mp >= 0) & (mp < n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)


def vectorized_binary_search(h_init, n, deadline, max_sweeps=10000):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n).astype(np.int64)
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    n_shifts = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        no = len(ones)
        nz = len(zeros)
        if nz == 0 or no == 0:
            break

        # Precompute deltas_add for all 0-positions
        deltas_add = np.zeros((nz, n_shifts), dtype=np.float64)
        for k in range(nz):
            j = zeros[k]
            jp = j + shifts
            jm = j - shifts
            mp = (jp >= 0) & (jp < n)
            mm = (jm >= 0) & (jm < n)
            deltas_add[k, mp] += (1.0 - h[jp[mp]])
            deltas_add[k, mm] += h[jm[mm]]

        # Compute A (for 1-positions) and B (for 0-positions) at worst shift
        s_star = shifts[np.argmax(corr)]
        i_sp = ones + s_star
        i_sm = ones - s_star
        A = np.zeros(no)
        mp = (i_sp >= 0) & (i_sp < n)
        mm = (i_sm >= 0) & (i_sm < n)
        A[mp] += h[i_sp[mp]]
        A[mm] += h[i_sm[mm]]

        j_sp = zeros + s_star
        j_sm = zeros - s_star
        B = np.zeros(nz)
        mp = (j_sp >= 0) & (j_sp < n)
        mm = (j_sm >= 0) & (j_sm < n)
        B[mp] += h[j_sp[mp]]
        B[mm] += h[j_sm[mm]]

        A_sorted = np.argsort(A)
        B_max = B.max()

        best_new_c = cur_c
        best_i_g = -1
        best_j_g = -1
        best_corr_g = None

        for ai in A_sorted:
            if A[ai] >= B_max:
                break
            i = ones[ai]
            ip = i + shifts
            im = i - shifts
            delta_rem = np.zeros(n_shifts)
            mp = (ip >= 0) & (ip < n)
            mm = (im >= 0) & (im < n)
            delta_rem[mp] += -(1.0 - h[ip[mp]])
            delta_rem[mm] += h[im[mm]]

            base = corr + delta_rem
            new_maxs = (base[None, :] + deltas_add).max(axis=1)

            valid = B > A[ai]
            if not valid.any():
                continue
            valid_idx = np.where(valid)[0]
            vals = new_maxs[valid_idx]
            min_local = np.argmin(vals)
            best_j_local = valid_idx[min_local]
            min_val = vals[min_local]

            if min_val * dx < best_new_c - 1e-15:
                best_new_c = min_val * dx
                best_i_g = ai
                best_j_g = best_j_local
                best_corr_g = (base + deltas_add[best_j_local]).copy()

        if best_i_g >= 0:
            i = ones[best_i_g]
            j = zeros[best_j_g]
            h[i], h[j] = 0.0, 1.0
            corr = best_corr_g
            cur_c = best_new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break

    return best_h, best_c


def soft_gradient_descent(h_init, n, deadline, max_iter=200000):
    h = h_init.copy().astype(np.float64)
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()

    lr = 0.5 / n
    patience = 5000
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 50 == 0:
            lr = max(0.01 / n, lr * 0.999)

        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        c_val = max_val * dx
        if c_val < best_c - 1e-12:
            best_c = c_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        # Find active shifts
        active = np.where(corr >= max_val - 0.05)[0]

        # Compute subgradient (vectorized over active shifts)
        grad = np.zeros(n)
        for ai in active:
            s = shifts[ai]
            mp = np.arange(n) + s
            mm = np.arange(n) - s
            mask_p = (mp >= 0) & (mp < n)
            mask_m = (mm >= 0) & (mm < n)
            grad[mask_p] += (1.0 - h[mp[mask_p]])
            grad[mask_m] -= h[mm[mask_m]]
        grad /= len(active)

        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        # Project sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - h
                s_sum = room.sum()
                if s_sum > 1e-12:
                    h = h + diff / s_sum * room
            else:
                s_sum = h.sum()
                if s_sum > 1e-12:
                    h = h + diff / s_sum * h
            h = np.clip(h, 0.0, 1.0)

    return best_h, best_c


def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h


def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Load parent

    if cs < best_c:

        best_h = hs.copy()

    # Phase 3: More binary search from multiple starts
    t_phase3_end = t0 + 0.55 * (budget_s - 5)
```
