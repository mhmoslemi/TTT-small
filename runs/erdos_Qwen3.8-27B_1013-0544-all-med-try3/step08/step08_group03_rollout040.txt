    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def overlap_grad(h, n, k):
    """Compute gradient of overlap at shift k with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        # O(k) = sum_{i=0}^{m-1} h[i](1-h[i+k])
        # dO/dh[j] = (1-h[j+k]) if j < m - h[j-k] if j >= k
        grad[:m] += (1.0 - h[k:k+m])  # (1-h[j+k]) for j=0..m-1
        if k > 0:
            grad[k:] += -h[:m]  # -h[j-k] for j=k..n-1
    else:
        kk = -k
        m = n - kk
        # O(k) = sum_{i=kk}^{n-1} h[i](1-h[i-kk])
        # dO/dh[j] = (1-h[j-kk]) if j >= kk - h[j+kk] if j < n-kk
        grad[kk:] += (1.0 - h[:m])  # (1-h[j-kk]) for j=kk..n-1
        grad[:m] += -h[kk:]  # -h[j+kk] for j=0..m-1
    return grad * dx

def subgradient_optimize(h, n, deadline, max_iters=50000):
    """Subgradient method with active-set averaging."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Track recent active constraints
    active_grads = []
    window_size = 50
    
    lr = 0.01
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        current_max = c_vec.max()
        k_star = int(np.argmax(c_vec))
        k_cont = k_star - (n - 1)
        
        g = overlap_grad(h, n, k_cont)
        active_grads.append(g.copy())
        if len(active_grads) > window_size:
            active_grads.pop(0)
        
        # Average active gradients
        g_avg = np.mean(active_grads, axis=0)
        g_norm = np.linalg.norm(g_avg)
        if g_norm < 1e-14:
            break
        g_avg /= g_norm
        
        # Backtracking line search
        step_lr = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step_lr * g_avg, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step_lr *= 0.5
        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break
            # Reset window
            active_grads = [g.copy()]
    
    return best_h, best_c

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    # For numerical stability
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx, wm in enumerate(w):
        if wm < 1e-12:
            continue
        k = idx - (n - 1)
        grad += wm * overlap_grad(h, n, k)
    return grad

def lbfgs_smooth(h, n, deadline, alpha=100000, maxiter=500):
    """L-BFGS-B optimization with smooth max objective."""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    def grad(x):
        return smooth_max_grad(x, n, alpha)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def active_set_optimize(h, n, deadline, eps=0.001):
    """Active-set method: identify tight shifts and equalize them."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        current_max = c_vec.max()
        
        # Find active set (shifts within eps of max)
        active_mask = c_vec > (current_max - eps)
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            break
        if len(active_idx) > 20:
            # Too many active, increase eps
            eps *= 2
            continue
        if len(active_idx) == 1:
            # Only one active, use subgradient
            break
        
        # Optimize to equalize the active shifts
        # Minimize max over active set (using smooth max on just those)
        # with penalty for non-active being above
        
        def obj(x):
            cv = compute_c5_vec(x, n)
            active_vals = cv[active_idx]
            m = active_vals.max()
            sh = 1000.0 * (active_vals - m)
            val = m + np.log(np.sum(np.exp(sh))) / 1000.0
            # Penalty for non-active exceeding current_max
            non_active = ~active_mask
            if non_active.any():
                excess = np.maximum(0, cv[non_active] - current_max)
                val += 10.0 * np.sum(excess)
            return val + 0.001 * (x.sum() - n/2.0)**2
        
        def grad(x):
            cv = compute_c5_vec(x, n)
            active_vals = cv[active_idx]
            m = active_vals.max()
            sh = 1000.0 * (active_vals - m)
            sh_max = sh.max()
            w = np.exp(sh - sh_max)
            w /= w.sum()
            
            g = np.zeros(n)
            for i, idx in enumerate(active_idx):
                if w[i] < 1e-12:
                    continue
                k = idx - (n - 1)
                g += w[i] * overlap_grad(x, n, k)
            
            # Penalty gradient
            non_active = ~active_mask
            if non_active.any():
                excess = cv[non_active] - current_max
                for i, idx in enumerate(np.where(non_active)[0]):
                    if excess[i] > 0:
                        k = idx - (n - 1)
                        g += 10.0 * overlap_grad(x, n, k)
            
            g += 0.002 * (x.sum() - n/2.0)
            return g
        
        bounds = [(0.0, 1.0)] * n
        t_sec = min(30, deadline - time.time())
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 100, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
            else:
                # No improvement, tighten eps
                eps *= 0.5
                if eps < 1e-5:
                    break
        except Exception:
            break
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=80):
    """Very fine pair search for final polishing."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # Find the shifts that are near max (these are the ones we want to reduce)
            c_vec = compute_c5_vec(h, n)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] < 0.01 or h[i] > 0.99:
                        continue
                    if h[j] < 0.01 or h[j] > 0.99:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

                    for idx, j in enumerate(neighbors):
                        frac = room[idx] / total_room
                        delta_j = -d * frac
                        h[j] -= delta_j

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def overlap_grad(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return grad
        grad[:m] += (1.0 - h[k:k+m])
        if k > 0:
            grad[k:] += -h[:m]
    else:
        kk = -k
        m = n - kk
        if m <= 0:
            return grad
        grad[kk:] += (1.0 - h[:m])
        grad[:m] += -h[kk:]
    return grad * dx

def smooth_max_obj_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    if sh_max > 500:
        w = np.exp(sh - sh_max)
        w_sum = w.sum()
        obj = m + (sh_max + np.log(w_sum)) / alpha
    else:
        w = np.exp(sh)
        w_sum = w.sum()
        obj = m + np.log(w_sum) / alpha
    w /= w_sum
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-12:
            continue
        k = idx - (n - 1)
        grad += w[idx] * overlap_grad(h, n, k)
    return obj, grad

        _, g = smooth_max_obj_grad(x, n, alpha)

        t5 = min(time.time() + 80, deadline - 380)
        h5, c5 = focused_search(best_h, best_n, t5, top_k=12)
        if c5 < best_c:
            best_c, best_h = c5, h5
    # Phase 6: Fine pair search (280s)
    if time.time() < deadline - 200:
        t6 = min(time.time() + 280, deadline - 200)
        h6, c6 = fine_pair_search(best_h, best_n, t6, radius=55)
        if c6 < best_c:
            best_c, best_h = c6, h6
    # Phase 7: Focused search again (40s)
    if time.time() < deadline - 150:
        t7 = min(time.time() + 40, deadline - 150)
        h7, c7 = focused_search(best_h, best_n, t7, top_k=10)
        if c7 < best_c:
            best_c, best_h = c7, h7
    # Phase 8: Try higher resolution (100s)
    if time.time() < deadline - 80:
        for n_try in [400, 500]:
            if time.time() > deadline - 50:
                break
            x_old = np.linspace(0, 2, best_n, endpoint=False) + 1.0 / best_n
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(40, deadline - time.time() - 25)
            if alloc < 12:
                break
            h_t, c_t = slsqp_smooth(h_interp, n_try, time.time() + alloc * 0.5, alpha=500000, maxiter=200)
            if time.time() < deadline - 12:
                h_t, c_t = fine_pair_search(h_t, n_try, deadline - 10, radius=20)
            if c_t < best_c:
                best_c, best_h, best_n = c_t, h_t, n_try
    # Phase 9: Final pair search (remaining)
    if time.time() < deadline - 3:
        h9, c9 = fine_pair_search(best_h, best_n, deadline - 1, radius=35)
        if c9 < best_c:
            best_c, best_h = c9, h9
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
