          if tot<=1e-15: break

          if tot<=1e-15: break

def make_random_binary(n, rng):
    h=np.zeros(n, dtype=float)
    target=int(n//2)
    idx=np.arange(n)
    rng.shuffle(idx)

def select_active_lags(c_vec, n, M, tol_frac=0.002):
    L = len(c_vec)
    M = max(1, min(int(M), L))
    maxv = float(c_vec.max())
    tol = max(1e-9, tol_frac * maxv)
    idx = np.where(c_vec >= maxv - tol)[0]
    Meff = min(M + 4, L)
    if len(idx) > Meff:
        order = np.argpartition(-c_vec, Meff - 1)[:Meff]
    else:
        order = idx.astype(int).copy()
    m0 = n - 1
    if m0 not in order:
        if len(order) < Meff:
            order = np.append(order, m0)
        else:
            j = int(np.argmin(c_vec[order]))
            order[j] = m0
    return (n - 1 - order).astype(int)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n

    for _ in range(80):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            slack = 1.0 - h
            tot = float(slack.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * slack
        else:
            tot = float(h.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * h

    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = np.flatnonzero(h < 1.0)
            for i in idx:
                if diff <= 0.0:
                    break
                add = min(1.0 - h[i], diff)
                if add > 0.0:
                    h[i] += add
                    diff -= add
        else:

            for i in idx:
                if diff >= 0.0:
                    break
                sub = min(h[i], -diff)
                if sub > 0.0:
                    h[i] -= sub
                    diff += sub
    return h


def compute_c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def evaluate_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))


def active_values(h, n, lags):
    L = len(lags)
    vals = np.empty(L, dtype=float)
    dx = 2.0 / n
    om = 1.0 - h
    for a, l in enumerate(lags):
        li = int(l)
        if li == 0:
            vals[a] = dx * float(np.dot(h, om))
        elif li > 0:
            nv = n - li
            vals[a] = dx * float(np.dot(h[:nv], om[li:n]))
        else:
            d = -li
            nv = n - d
            vals[a] = dx * float(np.dot(h[d:n], om[:nv]))
    return vals


def active_obj(x, n, lags, alpha, mu):
    vals = active_values(x, n, lags)
    m = float(vals.max())
    z = alpha * (vals - m)
    zmax = float(z.max())
    lse = zmax + float(np.log(np.sum(np.exp(z - zmax))))
    val = m + lse / alpha
    if mu > 0.0:
        val += mu * float((x.sum() - 0.5 * n)) ** 2
    return float(val)


def active_grad(x, n, lags, alpha, mu):
    vals = active_values(x, n, lags)
    m = float(vals.max())
    z = alpha * (vals - m)
    zmax = float(z.max())
    w = np.exp(z - zmax)
    ws = float(w.sum())
    if ws <= 0.0:
        w = np.ones(len(vals)) / len(vals)
    else:
        w = w / ws

    g = np.zeros(n, dtype=float)
    dx = 2.0 / n
    om = 1.0 - x

    for wi, l in zip(w, lags):
        if wi <= 1e-18:
            continue
        li = int(l)
        if li == 0:
            g += (wi * dx) * (om - x)
        elif li > 0:
            nv = n - li
            g[0:nv] += wi * dx * om[li:n]
            g[li:n] -= wi * dx * x[0:nv]
        else:
            d = -li
            nv = n - d
            g[d:n] += wi * dx * om[0:nv]
            g[0:nv] -= wi * dx * x[d:n]

    if mu > 0.0:
        s = float(x.sum() - 0.5 * n)
        if s != 0.0:
            g += 2.0 * mu * s
    return g


def full_smooth_obj(x, n, alpha, mu):

    m = float(c.max())
    z = alpha * (c - m)

def full_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = float(c.max())
    z = alpha * (c - m)
    zmax = float(z.max())
    w = np.exp(z - zmax)
    ws = float(w.sum())
    if ws <= 0.0:
        w = np.ones(c.size) / c.size
    else:
        w = w / ws

    dx = 2.0 / n
    w0 = float(w[n - 1])
    om = 1.0 - x

    if n > 1:
        wpos = w[n - 2::-1]
        wneg = w[n:]
        K = wpos + wneg

        pref_pos = np.zeros(n, dtype=float)
        pref_pos[1:] = np.cumsum(wpos)
        pref_neg = np.zeros(n, dtype=float)
        pref_neg[1:] = np.cumsum(wneg)

        R = pref_pos[::-1] + pref_neg

        Kfull = np.zeros(2 * n - 1, dtype=float)
        shifts = np.arange(1, n)
        Kfull[n - 1 + shifts] = K
        Kfull[n - 1 - shifts] = K

        conv = np.convolve(x, Kfull, mode='full')
        S = conv[n - 1:2 * n - 1]
    else:
        R = np.zeros(n, dtype=float)
        S = np.zeros(n, dtype=float)

    g = dx * (w0 * (om - x) + R - S)
    if mu > 0.0:
        s = float(x.sum() - 0.5 * n)
        if s != 0.0:
            g += 2.0 * mu * s
    return g


def select_active_lags(c_vec, n, M, tol_frac=0.002):
    L = len(c_vec)
    M = max(1, min(int(M), L))
    maxv = float(c_vec.max())
    tol = max(1e-9, tol_frac * maxv)
    idx = np.where(c_vec >= maxv - tol)[0]

    Meff = min(M + 4, L)
    if len(idx) > Meff:
        order = np.argpartition(-c_vec, Meff - 1)[:Meff]
    else:
        order = idx.astype(int).copy()

    m0 = n - 1
    if m0 not in order:
        if len(order) < Meff:
            order = np.append(order, m0)
        else:
            j = int(np.argmin(c_vec[order]))
            order[j] = m0

    return (n - 1 - order).astype(int)

def minimax_obj(z, n, lags):
    return float(z[n])

def minimax_ineq_fun(z, n, lags):
    h = z[:n]
    t = float(z[n])
    vals = active_values(h, n, lags)
    return t - vals


def minimax_ineq_jac(z, n, lags):
    h = z[:n]
    M = len(lags)
    jac = np.empty((M, n + 1), dtype=float)
    jac[:, n] = 1.0

    dx = 2.0 / n
    om = 1.0 - h

    for r, l in enumerate(lags):
        g = jac[r, :n]
        g.fill(0.0)
        li = int(l)
        if li == 0:
            g[:] = dx * (om - h)
        elif li > 0:
            nv = n - li
            g[0:nv] = dx * om[li:n]
            g[li:n] -= dx * h[0:nv]
        else:
            d = -li
            nv = n - d
            g[d:n] = dx * om[0:nv]
            g[0:nv] -= dx * h[d:n]
        g *= -1.0

    return jac

def make_random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    target = int(n // 2)
    idx = np.arange(n)
    rng.shuffle(idx)

    if target > 0:
        h[idx[:target]] = 1.0

    rem = 0.5 * n - float(h.sum())
    if rem > 1e-12:
        if target < n:
            j = idx[target]
            h[j] = min(max(rem, 0.0), 1.0)
    elif rem < -1e-12:
        for j in idx[:target][::-1]:
            if rem >= 0.0:
                break
            need = min(h[j], -rem)
            if need > 0.0:
                h[j] -= need
                rem += need

    return project_feasible(h, n)


def make_periodic_start(n, period, ones):

    if period > 0:
        for p in range(min(int(ones), int(period))):

def linear_upsample_h(h, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)

def upsample_h(h, n_src, n_dst):

        return h.copy()
    if n_dst % n_src == 0:
        return np.repeat(h, n_dst // n_src)
    return linear_upsample_h(h, n_src, n_dst)

                min(D_end, deadline - 60),
                96,
                [50000, 200000, 1000000, 5000000],
                0.1,
                rng,
                0.001
            )
            if cD < best_c:
                best_c = cD
                best_h = hD
                best_n = len(best_h)

    if best_n <= 800 and time.time() < deadline - 35:
        hE1, cE1 = run_slqp_full_smooth(
            best_h,
            best_n,
            deadline - 12,
            200000
        )
        if cE1 < best_c:
            best_c = cE1
            best_h = hE1
            best_n = len(best_h)

    if time.time() < deadline - 18:
        hE2, cE2 = run_minimax_slsqp(
            best_h,
            best_n,
            deadline - 3,
            [16]
        )
        if cE2 < best_c:
            best_c = cE2
            best_h = hE2
            best_n = len(best_h)

    final_n = int(len(best_h))
    best_h = project_feasible(best_h, final_n)
    best_c = evaluate_c5(best_h, final_n)

    return best_h, float(best_c), final_n
