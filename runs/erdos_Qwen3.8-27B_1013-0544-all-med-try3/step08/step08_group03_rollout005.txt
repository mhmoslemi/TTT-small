                best_c = new_c
                best_h = new_h
                h = new_h
    
    # Phase 7: Intensive adaptive pair search
    if time.time() < deadline - 5:
        t7 = deadline - 2
        h, best_c = adaptive_pair_search(h, n, t7,
                                         base_deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                                       0.00005, 0.00002, 0.00001])
        best_h = h
    
    # Phase 8: Final micro polish
    if time.time() < deadline - 2:
        h, best_c = final_micro_polish(h, n, deadline - 1)
        best_h = h
    
    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
