    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - target

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def simulated_annealing(h_init, n, deadline, seed=42, power=20):
    dx = 2.0 / n
    target = n // 2
    rng = np.random.RandomState(seed)

    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0

    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_e = np.sum(corr ** power)
    best_e = cur_e
    best_h = h.copy()
    best_c = corr.max() * dx

    T = max(corr) * 0.05
    T_min = max(corr) * 0.0005
    cooling = 0.9995
    m_range = np.arange(-(n - 1), n)

    while time.time() < deadline and T > T_min:
        T *= cooling
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        i = rng.choice(zi)
        j = rng.choice(oi)

        delta = compute_delta_swap(h, i, j, n, m_range)
        new_corr = corr + delta
        new_e = np.sum(new_corr ** power)
        delta_e = new_e - cur_e

        if delta_e < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_e / max(T, 1e-12))):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_e = new_e
            c_val = corr.max() * dx
            if c_val < best_c:
                best_c = c_val
                best_h = h.copy()

    return best_h, best_c

def greedy_binary_tabu(h_init, n, deadline, seed=42, tabu_len=20, max_sweeps=5000):
    dx = 2.0 / n
    target = n // 2
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)

    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0

    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()

    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    tabu_moves = {}
    sweeper = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweeper += 1
        improved = False

        # Clean tabu
        if sweeper % tabu_len == 0:
            old = [k for k, v in tabu_moves.items() if sweeper - v >= tabu_len]
            for k in old:
                del tabu_moves[k]

        z_list = list(zeros)
        o_list = list(ones)
        rng.shuffle(z_list)
        rng.shuffle(o_list)

        for i in z_list:
            if time.time() > deadline:
                break
            for j in o_list:
                if time.time() > deadline:
                    break
                key = (i, j)
                if key in tabu_moves:
                    continue
                delta = compute_delta_swap(h, i, j, n, m_range)
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    tabu_moves[(j, i)] = sweeper
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try a 2-swap move
            z_list2 = list(zeros)
            o_list2 = list(ones)
            rng.shuffle(z_list2)
            rng.shuffle(o_list2)
            found = False
            for i1 in z_list2[:40]:
                if time.time() > deadline:
                    break
                for j1 in o_list2[:40]:
                    if time.time() > deadline:
                        break
                    d1 = compute_delta_swap(h, i1, j1, n, m_range)
                    tmp_corr = corr + d1
                    h[i1], h[j1] = 1.0, 0.0
                    delta_new_corr = tmp_corr.max()
                    # Now try second swap
                    for i2 in z_list2[:30]:
                        if i2 == j1:
                            continue
                        if time.time() > deadline:
                            break
                        for j2 in o_list2[:30]:
                            if j2 == j1 or j2 == i1:
                                continue
                            d2 = compute_delta_swap(h, i2, j2, n, m_range)
                            new_corr = tmp_corr + d2
                            nc2 = new_corr.max() * dx
                            if nc2 < cur_c - 1e-14:
                                h[i2], h[j2] = 1.0, 0.0
                                corr = new_corr
                                cur_c = nc2
                                ones.discard(j1)
                                ones.discard(j2)
                                ones.add(i1)
                                ones.add(i2)
                                zeros.discard(i1)
                                zeros.discard(i2)
                                zeros.add(j1)
                                zeros.add(j2)
                                tabu_moves[(j1, i1)] = sweeper
                                tabu_moves[(j2, i2)] = sweeper
                                found = True
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if found:
                ones = set(np.where(h == 1)[0].tolist())
                zeros = set(np.where(h == 0)[0].tolist())

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        if not improved and sweep > 50:
            # Restart with perturbation
            n_perturb = rng.randint(3, 10)
            for _ in range(n_perturb):
                oi2 = np.where(h == 1)[0]
                zi2 = np.where(h == 0)[0]
                if len(oi2) > 0 and len(zi2) > 0:
                    a = rng.choice(oi2)
                    b = rng.choice(zi2)
                    h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1 - h, mode='full').copy()
            cur_c = corr.max() * dx
            ones = set(np.where(h == 1)[0].tolist())
            zeros = set(np.where(h == 0)[0].tolist())
            tabu_moves.clear()
            if cur_c >= best_c:
                h = best_h.copy()
                corr = np.correlate(h, 1 - h, mode='full').copy()
                cur_c = corr.max() * dx
                ones = set(np.where(h == 1)[0].tolist())
                zeros = set(np.where(h == 0)[0].tolist())

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    n = n0
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP smooth optimization (30% of time)
    p1 = t0 + 0.25 * (budget_s - 3)
    h = best_h.copy()
    for alpha in [200, 1000, 5000, 20000, 100000, 500000, 2000000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n, alpha, 2000, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Simulated annealing for global exploration (20% of time)
    p2 = t0 + 0.45 * (budget_s - 3)
    if time.time() < p2 - 5:
        ha, ca = simulated_annealing(best_h, n, p2, seed=seed, power=10)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()
        ha, ca = simulated_annealing(best_h, n, p2, seed=seed + 100, power=30)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()

    # Phase 3: Greedy binary search with tabu (25% of time)
    p3 = t0 + 0.70 * (budget_s - 3)
    hg, cg = greedy_binary_tabu(best_h, n, p3, seed=seed, tabu_len=30, max_sweeps=8000)
    if cg < best_c:
        best_c = cg
        best_h = hg.copy()

    # Phase 4: SLSQP polish from best binary (10% of time)
    p4 = t0 + 0.80 * (budget_s - 3)
    if time.time() < p4 - 5:
        h = best_h.copy()
        h = project_feasible(h + 0.01 * rng.randn(n), n)
        for alpha in [100000, 1000000, 5000000]:
            if time.time() > p4 - 5:
                break
            h = slsqp_optimize(h, n, alpha, 3000, p4 - 5)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 5: More SA + greedy (remaining time)
    p5 = dl - 20
    if time.time() < p5:
        ha, ca = simulated_annealing(best_h, n, p5, seed=seed + 200, power=50)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()

    # Phase 6: Final greedy polish
    if time.time() < dl - 10:
        hg, cg = greedy_binary_tabu(best_h, n, dl - 5, seed=seed + 300, tabu_len=20, max_sweeps=5000)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
