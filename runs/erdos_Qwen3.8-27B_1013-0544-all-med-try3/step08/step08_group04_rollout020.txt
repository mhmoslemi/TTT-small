```python
    """Compute subgradient for lag k_star and take a step."""
    # k_star is the index in the full correlation array (0 to 2n-2)
    # Lag d = k_star - (n-1)
    d = k_star - (n - 1)
    
    # g[j] = (1-h[j-d]) * I(0<=j-d<n) - h[j+d] * I(0<=j+d<n)
    g = np.zeros(n)
    
    # First term: (1-h[j-d]) for valid j-d
    if d >= 0:
        # j-d >= 0 means j >= d, j-d < n means j < n+d (always true since j < n)
        j_lo, j_hi = d, n
        g[j_lo:j_hi] += (1 - h[0:n-d])
    else:
        # j-d >= 0 always (since -d > 0 and j >= 0), j-d < n means j < n+d
        j_lo, j_hi = 0, n + d  # n + d = n - |d|
        g[j_lo:j_hi] += (1 - h[-d:n])
    
    # Second term: -h[j+d] for valid j+d
    if d >= 0:
        # j+d >= 0 always, j+d < n means j < n-d
        j_lo, j_hi = 0, n - d
        g[j_lo:j_hi] -= h[d:n]
    else:
        # j+d >= 0 means j >= -d, j+d < n always
        j_lo, j_hi = -d, n
        g[j_lo:j_hi] -= h[0:n+d]
    
    # Project gradient to maintain sum constraint
    g -= g.mean()
    
    return alpha * g

        g = np.zeros(n)

            g[d:] += (1 - h[:n-d])
            g[:n-d] -= h[d:]
        else:
            g[:n+d] += (1 - h[-d:])
            g[-d:] -= h[:n+d]

        g -= g.mean()

        alpha = alpha_init / (1 + t * 0.001)
        v = gamma * v - alpha * g
        h = project_feasible(h + v, n)
    
    c_val = compute_c5(h, n)[0]
    if c_val < best_c:
        best_c = c_val
        best_h = h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def effect_vector(h, j, n):
    """Compute e_j[p] = dc[p]/dh[j] for all p (output indices 0 to 2n-2)."""
    e = np.zeros(2 * n - 1)
    d_range = range(-(n - 1), n)
    for idx, d in enumerate(d_range):
        # e[n-1+d] = (1-h[j-d])*I(0<=j-d<n) - h[j+d]*I(0<=j+d<n)
        val = 0.0
        jd = j - d
        if 0 <= jd < n:
            val += (1.0 - h[jd])
        jd2 = j + d
        if 0 <= jd2 < n:
            val -= h[jd2]
        e[idx] = val
    return e
```
