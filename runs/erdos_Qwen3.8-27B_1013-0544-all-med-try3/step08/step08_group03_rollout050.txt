def compute_gradient_fast(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    # smooth max weights
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    
    grad = np.zeros(n)
    # For each shift m (0 to 2n-2), the weight is w[m] * dx
    # shift k = m - (n-1)
    # d = |k|
    # Contribution: if j >= d, add (1-h[j-d]) * w[m]*dx
    #               if j+d < n, add (-h[j+d]) * w[m]*dx
    
    # Vectorize: for d = 0 to n-1:
    #   there are two shifts with |k|=d: k=d (m = n-1+d) and k=-d (m = n-1-d)
    #   (except d=0 has only one shift)
    
    for d in range(n):
        # shift k = d, m = n-1+d
        wm = w[n-1+d] * dx if wm > 1e-13
        if wm > 1e-13:
            # j >= d: add (1-h[j-d])
            grad[d:] += wm * (1 - h[:n-d])
            # j+d < n i.e. j < n-d: add (-h[j+d])
            # But wait, for k=d: j+d < n means j < n-d
            # h[j+d] for j < n-d
            grad[:n-d] += wm * (-h[d:])
        if d > 0:
            # shift k = -d, m = n-1-d
            wm = w[n-1-d] * dx
            if wm > 1e-13:
                grad[d:] += wm * (1 - h[:n-d])
                grad[:n-d] += wm * (-h[d:])
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        active = (h < 1.0 - 1e-14) & (h > 1e-14)
        if diff > 0:
            with np.errstate(divide='ignore', invalid='ignore'):
                adj = diff / active.sum() if active.sum() > 0 else 0
            h[active] += adj
        else:
            with np.errstate(divide='ignore', invalid='ignore'):
                adj = diff / active.sum() if active.sum() > 0 else 0
            h[active] += adj
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w  # weights for the log-sum-exp

def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    w = smooth_max_grad(c, alpha)
    grad = np.zeros(n)
    for d in range(n):
        # shift k=d (positive)
        if w[n-1+d] > 1e-14:
            wm = w[n-1+d] * dx
            grad[d:] += wm * (1.0 - h[:n-d])
            grad[:n-d] += -wm * h[d:]
        if d > 0 and w[n-1-d] > 1e-14:
            wm = w[n-1-d] * dx
            grad[d:] += wm * (1.0 - h[:n-d])
            grad[:n-d] += -wm * h[d:]
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_value(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = compute_gradient(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return float(smooth_max_value(c, alpha))

def slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def optimize_lbfgsb(h, n, deadline, max_total_iters=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]

def projected_grad_descent(h, n, deadline, lr_init=0.005, alpha_init=1000.0):

def pair_search_fast(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving from j to i
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try moving from i to j
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for d in range(n):
        if d < n and w[n - 1 + d] > 1e-14:
            wm = w[n - 1 + d] * dx
            grad[d:] += wm * (1.0 - h[:n - d])
            grad[:n - d] += -wm * h[d:]
        if d > 0 and w[n - 1 - d] > 1e-14:
            wm = w[n - 1 - d] * dx
            grad[d:] += wm * (1.0 - h[:n - d])
            grad[:n - d] += -wm * h[d:]
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_value(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = compute_gradient(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return float(smooth_max_value(c, alpha))

def slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def optimize_lbfgsb(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (50, 10.0), (200, 50.0), (1000, 100.0), (5000, 500.0),
        (20000, 1000.0), (100000, 5000.0), (500000, 10000.0),
        (2000000, 50000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 150, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': 800,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def optimize_slsqp(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac, 'args': (n,)}]

        return h_new, c_new

def pair_search(h, n, deadline, deltas, radius=60):
    best_c = compute_c5(h, n)
    n_reordered = list(range(n))
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in n_reordered:
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def coordinate_descent(h, n, deadline, n_vals=5):
    """Try small perturbations of each coordinate."""
    best_c = compute_c5(h, n)
    h = h.copy()
    vals = np.linspace(-0.02, 0.02, n_vals * 2 + 1)[1:]
    vals = np.concatenate([vals, -vals])
    for _ in range(200):
        if time.time() > deadline:

        improved = False
        rng = np.random.RandomState(int(time.time() * 1000) % 10000)
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            orig = h[i]
            for delta in vals:
                nv = orig + delta
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                h[i] = orig
        if not improved:
            break
    return h, best_c

    h_new += 0.001 * np.sin(np.pi * x_new) * (1 - h_new) * h_new
    return project_feasible(h_new, n_new)
