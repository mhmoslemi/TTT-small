import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_slsqp(h, n, alpha, deadline, maxiter=1500):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def exact_slsqp(h, n, deadline, maxiter=1000):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        return g.max()

    def cons_fun(x):
        return x.sum() - n / 2.0

    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:

def iterated_perturbation(h, n, deadline, rng, n_iter=50):
    """Perturb and re-optimize to escape local optima"""
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for s in range(n_iter):
        if time.time() > deadline:
            break
        noise_scale = 0.03 + 0.02 * (s % 3)
        pert = best_h + noise_scale * rng.randn(n)
        pert = project_feasible(pert, n)

        # Quick optimize with 2-3 alpha levels
        for alpha in [2000, 20000, 200000]:
            if time.time() > deadline:
                break
            pert = smooth_slsqp(pert, n, alpha, deadline, maxiter=400)
        c = compute_c5(pert, n)
        if c < best_c:
            best_c = c
            best_h = pert.copy()

    return best_h, best_c

    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: SLSQP polish from initial (25% budget)
    p1 = t0 + 0.25 * (budget_s - 3)
    h = best_h.copy()
    for alpha in [1000, 5000, 50000, 500000, 5000000]:
        if time.time() > p1 - 3:
            break
        h = smooth_slsqp(h, n, alpha, p1 - 1, maxiter=1000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Targeted worst-lag descent (15% budget)
    p2 = t0 + 0.40 * (budget_s - 3)
    h, c = targeted_lag_descent(best_h.copy(), n, p2, max_iter=50000)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 3: Binary search on current n (20% budget)
    p3 = t0 + 0.60 * (budget_s - 3)
    if time.time() < p3 - 5:
        hb, cb = binary_block_search(best_h.copy(), n, p3 - 3, max_sweeps=300)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        else:
            # Keep the continuous but do exact SLSQP polish
            h2 = exact_slsqp(best_h.copy(), n, p3 - 3, maxiter=500)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()

    # Phase 4: Try different resolutions (20% budget)
    p4 = t0 + 0.80 * (budget_s - 3)
    for nh in [200, 300, 400]:
        if time.time() > p4 - 60:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = project_feasible(np.interp(xn, xo, best_h, period=2.0), nh)

        for alpha in [5000, 50000, 500000]:
            if time.time() > p4 - 30:
                break
            hh = smooth_slsqp(hh, nh, alpha, p4 - 10, maxiter=500)
        c = compute_c5(hh, nh)
        if c < best_c:
            best_c = c
            best_h = hh.copy()
            n = nh
            # Quick binary on this new resolution
            if time.time() < p4 - 10:
                hb2, cb2 = binary_block_search(best_h.copy(), n, p4 - 5, max_sweeps=100)
                if cb2 < best_c:
                    best_c = cb2
                    best_h = hb2.copy()

    # Phase 5: Iterated perturbation (10% budget)
    p5 = t0 + 0.92 * (budget_s - 3)
    if time.time() < p5 - 5:
        h, c = iterated_perturbation(best_h.copy(), n, p5 - 3, rng, n_iter=30)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Final polish
    if time.time() < dl - 5:
        h = exact_slsqp(best_h.copy(), n, dl - 3, maxiter=800)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
