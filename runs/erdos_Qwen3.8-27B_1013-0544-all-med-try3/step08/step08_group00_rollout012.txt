def project_feasible(h, n):

        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

    """L-BFGS with log-sum-exp smoothing."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    try:

                      args=(n, alpha), method='L-BFGS-B',
                      bounds=bounds,

        c_new = compute_c5(h_new, n)

        h_pert = project_feasible(h_pert, n)
        
        # Local optimize (fewer iterations for speed)
        h_new, c_new = lbfgs_optimize(h_pert, n, deadline - 30, alpha=alpha, maxiter=300)
        
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
            h = h_new
            sigma = sigma_init  # Reset on improvement
        else:
            sigma *= sigma_decay
        
        if sigma < 1e-4:
            break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    # Load parent
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    print(f"Initial: C5 = {best_c:.6f}, n = {n}")
    
    # Phase 1: Quick L-BFGS (30s)
    t1 = min(t0 + 30, deadline - 800)
    if time.time() < t1:
        h, c = lbfgs_optimize(h, n, t1, alpha=3000, maxiter=1000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        print(f"After L-BFGS: C5 = {best_c:.6f}")
    
    # Phase 2: Basin hopping (300s)
    t2 = min(t0 + 330, deadline - 500)
    if time.time() < t2:
        h, c = basin_hopping(best_h, n, t2, n_perturb=15, sigma_init=0.03, sigma_decay=0.97, alpha=5000)
