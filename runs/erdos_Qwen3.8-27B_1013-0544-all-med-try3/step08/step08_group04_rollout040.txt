import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    cs = alpha * (c - m)
    if cs.max() > 500:
        cs2 = cs - cs.max()
        return m + (cs.max() + np.log(np.sum(np.exp(cs2)))) / alpha
    return m + np.log(np.sum(np.exp(cs))) / alpha

def smooth_weights(c, alpha):
    m = c.max()
    cs = alpha * (c - m)
    if cs.max() > 500:
        cs2 = cs - cs.max()
        e = np.exp(cs2)
    else:
        e = np.exp(cs)
    return e / e.sum()

def c5_smooth_fun(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c, alpha)

def c5_smooth_jac(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    w = smooth_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += wk * dx * (1 - x[s:])
            grad[s:] -= wk * dx * x[:n - s]
        else:
            t = -s
            grad[t:] += wk * dx * (1 - x[:n - t])
            grad[:n - t] -= wk * dx * x[t:]
    return grad

def compute_effects(h, n):
    """Compute eff[i][k] = linear change in c[k] (scaled by dx) per unit increase in h[i]."""
    dx = 2.0 / n
    n_c = 2 * n - 1
    eff = np.zeros((n, n_c))
    for i in range(n):
        # (1-h[i+s_k]) for valid k: i+k-(n-1) in [0,n-1] → k in [n-1-i, 2n-2-i]
        lo1 = n - 1 - i
        hi1 = 2 * n - 2 - i
        if lo1 <= hi1:
            # s_k = k - (n-1), i+s_k = i+k-(n-1)
            # indices of h: i+0, i+1, ..., i+(hi1-lo1)
            idx = np.arange(i, i + (hi1 - lo1) + 1)
            eff[i, lo1:hi1+1] += dx * (1 - h[idx])
        
        # -h[i-s_k] for valid k: i-k+(n-1) in [0,n-1] → k in [i, i+n-1]

        hi2 = i + n - 1

        eff[i, lo2:hi2+1] -= dx * h[idx]
    
    return eff

from scipy.optimize import minimize

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def _smooth_max(c, alpha):
    m = c.max()
    cs = alpha * (c - m)
    if cs.max() > 500:
        cs2 = cs - cs.max()
        return m + (cs.max() + np.log(np.sum(np.exp(cs2)))) / alpha
    return m + np.log(np.sum(np.exp(cs))) / alpha


def _smooth_weights(c, alpha):
    m = c.max()
    cs = alpha * (c - m)
    if cs.max() > 500:
        cs2 = cs - cs.max()
        e = np.exp(cs2)
    else:
        e = np.exp(cs)
    return e / e.sum()


def _obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return _smooth_max(c, alpha)


def _grad_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    w = _smooth_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += wk * dx * (1.0 - x[s:])
            grad[s:] -= wk * dx * x[:n - s]
        else:
            t = -s
            grad[t:] += wk * dx * (1.0 - x[:n - t])
            grad[:n - t] -= wk * dx * x[t:]
    return grad


def _sum_con_fun(x, n):
    return x.sum() - n / 2.0


def _sum_con_jac(x, n):
    return np.ones(len(x))


def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_smooth, h, jac=_grad_smooth,
                       args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def compute_effects(h, n):
    dx = 2.0 / n
    n_c = 2 * n - 1
    eff = np.zeros((n, n_c))
    for i in range(n):
        lo1 = n - 1 - i
        hi1 = 2 * n - 2 - i
        if lo1 <= hi1:
            idx = np.arange(i, hi1 - lo1 + i + 1)
            eff[i, lo1:hi1 + 1] += dx * (1.0 - h[idx])
        lo2 = i
        hi2 = i + n - 1
        idx = np.arange(n - 1, -1, -1)
        eff[i, lo2:hi2 + 1] -= dx * h[idx]
    return eff


def pair_search_fast(h, n, deadline, deltas, radius=100):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            eff = compute_effects(h, n)
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        c_new = c + d * (eff[i] - eff[j])
                        new_max = c_new.max()
                        if new_max < best_c - 1e-15:
                            h[i] = min(h[i] + d, 1.0)
                            h[j] = max(h[j] - d, 0.0)
                            c = c_new
                            best_c = new_max
                            improved = True
                            if j < i:
                                continue
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        c_new2 = c - d * (eff[i] - eff[j])
                        new_max2 = c_new2.max()
                        if new_max2 < best_c - 1e-15:
                            h[i] = max(h[i] - d, 0.0)
                            h[j] = min(h[j] + d, 1.0)
                            c = c_new2
                            best_c = new_max2
                            improved = True
            if not improved:
                break
    best_c = compute_c5(h, n)
    return h, best_c


def multi_alpha_slsqp(h, n, alphas, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for a in alphas:
        if time.time() > deadline - 5:

    return best_h, best_c
