from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    # Euclidean projection onto {0<=h<=1, sum(h)=target}
    h = h.copy()
    if h.sum() == target:
        return np.clip(h, 0, 1)
    
    # Binary search for lambda
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    
    # Final adjustment
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h_proj[mask] = np.clip(h_proj[mask] + adj, 0.0, 1.0)
    return h_proj

    m = c.max()
    sh = alpha * (c - m)

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

        bh, bc = opt_lbfgs(bh, nc, deadline - 2, per_phase=50)

    bh = project_feasible(bh, len(bh))
    bc = compute_c5(bh, len(bh))
    return bh, bc, len(bh)
