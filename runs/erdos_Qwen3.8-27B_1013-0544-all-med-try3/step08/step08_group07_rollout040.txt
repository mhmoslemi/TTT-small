```python
def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def compute_all_g(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(g, alpha):
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
```

```python
def compute_delta_g(i, j, h, n):
    """Compute change in all g_k when h[i] goes from 1->0 and h[j] goes from 0->1"""
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For position i (going 1 -> 0):
    delta_i = np.zeros(2*n-1)
    # First factor valid: 0 <= i < n (always) and 0 <= i+l < n
    mask_i_first = (0 <= i + lags) & (i + lags < n)  # wait, i < n is always true
    mask_i_first = (i + lags >= 0) & (i + lags < n)
    # h[i+lags] where valid
    valid_i_first = np.where(mask_i_first, i + lags, 0)
    delta_i[mask_i_first] -= (1 - h[valid_i_first])
    
    # Second factor valid: 0 <= i-l < n and 0 <= i < n (always)
    mask_i_second = (i - lags >= 0) & (i - lags < n)
    valid_i_second = np.where(mask_i_second, i - lags, 0)
    delta_i[mask_i_second] += h[valid_i_second]
    
    # For position j (going 0 -> 1):
    delta_j = np.zeros(2*n-1)
    mask_j_first = (j + lags >= 0) & (j + lags < n)
    valid_j_first = np.where(mask_j_first, j + lags, 0)
    delta_j[mask_j_first] += (1 - h[valid_j_first])
    
    mask_j_second = (j - lags >= 0) & (j - lags < n)
    valid_j_second = np.where(mask_j_second, j - lags, 0)
    delta_j[mask_j_second] -= h[valid_j_second]
    
    return delta_i + delta_j
```

A better approach: for each position i (going from 1 to 0), compute the effect on all g_k at once (O(n)), then for each position j (going from 0 to 1), the total effect is delta_i + delta_j. We want to find the j that minimizes max_k (g_k + delta_i[k] + delta_j[k]).

This is still O(n²) per sweep if we do it naively, but with numpy it should be fast.

For a binary solution h, given current g = compute_all_g(h, n):
1. For each i in ones: compute delta_i = effect of setting h[i]=0 (O(n) using numpy)
2. For each j in zeros: compute delta_j = effect of setting h[j]=1 (O(n) using numpy)
3. For each pair (i, j): new_g = g + delta_i + delta_j, new_max = max(new_g)
4. Choose the pair (i,j) that minimizes new_max

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(g, alpha):
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max_val(g, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):

    idx = np.argsort(h)[::-1]

    g = compute_all_g(h_bin, n)
    current_max = g.max()
    best_c = current_max
    lags = np.arange(-(n-1), n)
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute deltas for all ones (going 1->0) and zeros (going 0->1)
        i_idx = ones[:, None]
        lags_row = lags[None, :]
        
        i_plus_lag = i_idx + lags_row
        mask_fi = (i_plus_lag >= 0) & (i_plus_lag < n)
        h_ip = np.where(mask_fi, h_bin[np.clip(i_plus_lag, 0, n-1)], 0.0)
        d_first_i = -np.where(mask_fi, 1.0 - h_ip, 0.0)
        
        i_minus_lag = i_idx - lags_row
        mask_si = (i_minus_lag >= 0) & (i_minus_lag < n)
        h_im = np.where(mask_si, h_bin[np.clip(i_minus_lag, 0, n-1)], 0.0)
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(g, alpha):
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def slsqp_optimize(h, n, alpha, maxiter, deadline):

        probs = exp_shifted / exp_shifted.sum()
        return probs / dx  # approximate gradient scaling
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

    if time.time() < p2 - 10:
        hb = to_binary(best_h, n)
        hb, cb = binary_local_search_vec(hb, n, p2 - 30, max_sweeps=20000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        
        # Continue with simulated annealing from current best binary
        if time.time() < p2 - 5:
            hb2, cb2 = binary_anneal(hb.copy(), n, p2 - 5,
                                      t_start=0.02, t_end=0.0003, max_iters=50000)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()
            
            # Another round of greedy local search after annealing
            if time.time() < p2 - 3:
                hb3, cb3 = binary_local_search_vec(best_h.copy(), n, p2 - 3,
                                                    max_sweeps=10000)
                if cb3 < best_c:
                    best_c = cb3
                    best_h = hb3.copy()
    
    # Phase 3: Multi-resolution search (25% budget)
    p3 = t0 + 0.85 * (budget_s - 3)
    if time.time() < p3 - 20:
        resolutions = [100, 150, 200, 250, 300, 350, 400]
        # Remove current resolution
        resolutions = [r for r in resolutions if r != n and r < 500]
        # Sort by proximity to current for efficiency
        resolutions.sort(key=lambda r: abs(r - n))
        
        for nh in resolutions[:5]:
            if time.time() > p3 - 15:
                break
            xo = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, nh, endpoint=False)
            hh = np.interp(xn, xo, best_h)
            hh = project_feasible(hh, nh)
            
            # Quick SLSQP
            hh = slsqp_optimize(hh, nh, 5000, 300, p3 - 10)
            
            # Binary local search
            hb = to_binary(hh, nh)
            hb, cb = binary_local_search_vec(hb, nh, p3 - 5, max_sweeps=8000)
            
            c_cont = compute_c5(hh, nh)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                n = nh
            elif c_cont < best_c:
                best_c = c_cont
                best_h = hh.copy()
                n = nh
    
    # Phase 4: Final polish (10% budget)
    if time.time() < dl - 5:
        # SLSQP on the best solution
        h = best_h.copy()
        h = slsqp_optimize(h, n, 1000000, 2000, dl - 10)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Final binary search
        if time.time() < dl - 3:
            hb = to_binary(best_h, n)
            hb, cb = binary_local_search_vec(hb, n, dl - 3, max_sweeps=5000)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
