def solve_1d_minimax(a, b, lo=0.0, hi=1.0):
    """Solve min_{x in [lo,hi]} max_k (a[k]*x + b[k])"""
    lo_t = max(b)  # value at x=lo (if lo=0)
    hi_t = max(a + b)  # value at x=hi (if hi=1)
    
    def feasible(t):
        lower = 0.0
        upper = 1.0
        for ak, bk in zip(a, b):
            if ak > 1e-14:
                upper = min(upper, (t - bk) / ak)
            elif ak < -1e-14:
                lower = max(lower, (t - bk) / ak)
            else:
                if bk > t + 1e-14:
                    return False
        return lower <= upper + 1e-14
    
    # Binary search on t
    for _ in range(60):  # ~1e-18 precision
        mid = (lo_t + hi_t) / 2
        if feasible(mid):
            hi_t = mid
        else:
            lo_t = mid
    
    t_opt = hi_t
    # Find x_opt
    lower = 0.0
    upper = 1.0
    for ak, bk in zip(a, b):
        if ak > 1e-14:
            upper = min(upper, (t_opt - bk) / ak)
        elif ak < -1e-14:
            lower = max(lower, (t_opt - bk) / ak)
    x_opt = (lower + upper) / 2
    return x_opt, t_opt

def compute_c5(h, n):

def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    c = c * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    c = c * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    K = len(c)
    n1 = n - 1
    grad = np.zeros(n)
    for mm in range(K):
        if w[mm] < 1e-15:
            continue
        k = mm - n1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)

3. Clip to [0,1]
4. Project to maintain sum constraint (every few steps)

But this is just gradient descent, which is what L-BFGS already does. The advantage of the 1D solve is that it finds the EXACT optimum in that coordinate (for the linearized problem), taking as large a step as beneficial.

Let me try a different approach: **alternating minimization with exact 1D solves using the linear model**.

For coordinate m:
1. Compute the linear model: for each k, a_k*x + b_k
2. Solve min_x max_k (a_k*x + b_k) s.t. 0≤x≤1 using binary search
3. Update h[m] = x*
4. Every 10 coordinates, re-project to maintain sum

For the binary search:
- lo_t = max_k b_k (value at x=0)
- hi_t = max_k (a_k+b_k) (value at x=1)
- Binary search for 60 iterations (gives ~1e-18 relative precision)

The feasibility check for a given t:
- For each k with a_k > 0: x ≤ (t-b_k)/a_k → upper bound
- For each k with a_k < 0: x ≥ (t-b_k)/a_k → lower bound
- For each k with a_k = 0: need b_k ≤ t
- Feasible iff max(0, lower_bound) ≤ min(1, upper_bound)

This is O(K) per feasibility check, 70 checks per coordinate, n coordinates per sweep.
Total: n * K * 70 = 500 * 999 * 70 ≈ 35M per sweep.
At 100M ops/s: ~0.35s per sweep. 100 sweeps: 35s. Fine.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    c = c * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    c = c * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    K = len(c)
    n1 = n - 1
    grad = np.zeros(n)
    for mm in range(K):
        if w[mm] < 1e-15:
            continue
        k = mm - n1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    c_vals = c_vec[active_idx]
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        g = shift_grad_single(h, n, active_idx[j])
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1

        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1

        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))

            return delta, t_opt
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.05, max_active=30):

def solve_1d_minimax_linear(a, b):
    lo_t = float(np.max(b))
    hi_t = float(np.max(a + b))
    if hi_t - lo_t < 1e-18:
        return 0.0, hi_t
    pos_mask = a > 1e-14
    neg_mask = a < -1e-14
    zero_mask = ~(pos_mask | neg_mask)
    
    def feasible(t):
        lower = 0.0
        upper = 1.0
        if np.any(pos_mask):
            ub = (t - b[pos_mask]) / a[pos_mask]
            upper = min(upper, float(np.min(ub)))
        if np.any(neg_mask):
            lb = (t - b[neg_mask]) / a[neg_mask]
            lower = max(lower, float(np.max(lb)))
        if np.any(zero_mask):
            if float(np.max(b[zero_mask])) > t + 1e-14:
                return False
        return lower <= upper + 1e-14
    
    for _ in range(60):
        mid = (lo_t + hi_t) * 0.5
        if feasible(mid):
            hi_t = mid
        else:
            lo_t = mid

    lower = 0.0
    upper = 1.0
    if np.any(pos_mask):
        ub = (t_opt - b[pos_mask]) / a[pos_mask]

        lower = max(lower, float(np.max(lb)))
    x_opt = float(np.clip((lower + upper) * 0.5, 0.0, 1.0))
