            if s < 1e-14:

                    break
    return h

def lse_eval_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    f = corr * dx  # shape (2n-1,)
    
    # LSE
    f_shift = f - f.max()
    e = np.exp(alpha * f_shift)
    e_sum = e.sum()
    obj = (f.max() + np.log(e_sum)) / alpha
    p = e / e_sum  # softmax
    
    # Gradient: grad[j] = dx * sum_m p[m] * [(1-h[j+lag_m])*V[j,m] - h[j-lag_m])*W[j,m]]
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # Precompute validity and indices
    # For each j (0 to n-1) and each m (0 to 2n-2):
    # V[j,m]: j + lags[m] in [0, n-1]
    # W[j,m]: j - lags[m] in [0, n-1]
    
    j_idx = np.arange(n)[:, None]  # (n, 1)
    m_idx = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    
    pos1 = j_idx + lags[m_idx]  # (n, 2n-1)
    pos2 = j_idx - lags[m_idx]  # (n, 2n-1)
    
    V = (pos1 >= 0) & (pos1 < n)
    W = (pos2 >= 0) & (pos2 < n)
    
    # Gather h values
    h1 = h[np.clip(pos1, 0, n-1)]  # (n, 2n-1)
    h2 = h[np.clip(pos2, 0, n-1)]
    
    # term1[j] = sum_m V[j,m] * p[m] * (1-h1[j,m])
    # term2[j] = sum_m W[j,m] * p[m] * h2[j,m]
    
    term1 = (V * p[None, :] * (1 - h1)).sum(axis=1)
    term2 = (W * p[None, :] * h2).sum(axis=1)
    
    grad = dx * (term1 - term2)
    
    return obj, grad

Actually, the binary search for this problem is inherently O(n²) per sweep (evaluating all n/2 × n/2 = n²/4 swaps, each O(n) to compute the delta and find max). For n=250, that's ~62K evaluations × O(500) = ~31M operations per sweep. In Python, this is probably 1-3 seconds per sweep. So in 110s, we can do maybe 30-100 sweeps.

But the key question is: does binary search actually help? The continuous solution might be better than any binary solution. Let me focus more on the continuous optimization.

Let me reconsider the approach:

**Key insight**: The problem is a finite-dimensional minimax problem with 2n-1 constraints. For n=250, that's 499 constraints and 250 variables. This is a well-posed nonlinear program.

**Best approach**: Use a proper nonlinear solver with:
1. Analytical gradients
2. Good initial point (from parent)
3. Multiple alpha values for the LSE approximation
4. Possibly KKT-based active set identification

Let me also try a different formulation: instead of LSE, use the **max with a smooth penalty**:
g(h, t) = t + (ρ/2) * Σ_k max(0, f_k - t)²

Minimize over (h, t). This is a smooth minimization (the max(0,·)² is C¹). The gradient with respect to t is:
∂g/∂t = 1 - ρ * Σ_k max(0, f_k - t)

And with respect to h:
∂g/∂h_j = -ρ * Σ_k max(0, f_k - t) * ∂f_k/∂h_j

This is the hinge loss formulation. At the optimum, t = max_k f_k and the KKT conditions hold.

The advantage over LSE is that the penalty only activates for active (violated) constraints, so the gradient is sparse in a sense.

But for a smooth solver, LSE is probably better since it's everywhere differentiable.

Let me just go with LSE-SLSQP and make it efficient.

One more thought: I should try **different initial perturbations** to escape local minima. The LSE-SLSQP is a first-order method and will converge to the nearest local minimum. Different perturbations can lead to different minima.

def binary_search(h, n, deadline, n_sweeps=300):
    target = n // 2
    dx = 2.0 / n
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=float)
    hb[idx[:target]] = 1.0
    corr = np.correlate(hb, 1.0 - hb, mode='full').astype(float)
    cur = corr.max() * dx
    best_cur = cur
    best_hb = hb.copy()
    lags = np.arange(-(n - 1), n)

    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(hb == 0)[0]
        ones = np.where(hb == 1)[0]
        for i in zeros:
            best_j = -1
            best_nc = cur
            i1 = i + lags
            i2 = i - lags
            Vi = (i1 >= 0) & (i1 < n)

            for j in ones:

                j2 = j - lags

                    d[Vj] -= 1.0 - hb[np.clip(j1[Vj], 0, n - 1)]

                j1 = j + lags
                j2 = j - lags
                Vj = (j1 >= 0) & (j1 < n)
