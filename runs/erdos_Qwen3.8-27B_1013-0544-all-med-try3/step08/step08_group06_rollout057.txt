        active = np.where(c >= c.max() - 0.005)[0]
        if len(active) > 20:
            active = np.argsort(c)[::-1][:20]
        
        # Compute subgradient
        G = gradient_of_c(h, n)
        subgrad = G[active].mean(axis=0)
        
        # Step with projection
        h_new = h - eta * subgrad
        h_proj = project_feasible(h_new, n)
        
        # Check improvement
        c_new = compute_correlation(h_proj, n)
        if c_new.max() < m_val - 1e-12:
            h = h_proj
            eta = min(eta * 1.01, 0.1)
        else:
            eta *= 0.98
            h = project_feasible(h - eta * 0.5 * subgrad, n)
        
        if eta < 1e-6:
            break
    
    return h, best_c, best_h

def fast_pair_search(h, n, deadline, deltas):
    """Pair search with gradient precomputation for speed."""
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_correlation(h, n)
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                return h, best_c
            # Recompute gradient
            G = gradient_of_c(h, n)
            # For direction (i, j) with step d: change in c_k = d*(G[k,i] - G[k,j])
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Compute effect of increasing h[i] by d and decreasing h[j] by d
                for j in range(n):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    # First-order approximation of new max
                    delta_c = d * (G[:, i] - G[:, j])
                    new_max_approx = (c + delta_c).max()
                    if new_max_approx < best_c - 1e-14:
                        # Accept and verify
                        h[i], h[j] = hi, hj
                        c_new = compute_correlation(h, n)
                        new_c = float(c_new.max())
                        if new_c < best_c - 1e-15:
                            best_c = new_c
                            c = c_new
                            G = gradient_of_c(h, n)
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    
    return h, best_c

def exact_pair_step(h, n, c, i, j, d):
    """Try step: h[i]+=d, h[j]-=d. Return new c if improved."""
    h[i] += d
    h[j] -= d
    c_new = compute_correlation(h, n)
    if c_new.max() < max(c):
        return c_new, True
    h[i] -= d
    h[j] += d
    return c, False

def near_pair_search(h, n, deadline, deltas, radius=50):
    """Restricted pair search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_correlation(h, n)
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(100):
            if time.time() > deadline:
                return h, best_c
            improved = False
            G = gradient_of_c(h, n)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    delta_c = d * (G[:, i] - G[:, j])
                    if (c + delta_c).max() < best_c - 1e-14:
                        h[i], h[j] = hi, hj
                        c_new = compute_correlation(h, n)
                        if c_new.max() < best_c - 1e-15:
                            best_c = c_new
                            c = c_new
                            G = gradient_of_c(h, n)
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: SLSQP with alpha ramp
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=2000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    
    # Phase 2: Subgradient refinement
    rem = deadline - time.time()
    if rem > 20:
        sg_deadline = time.time() + min(rem * 0.3, 80)
        best_h, c_sg, _ = subgradient_refine(best_h, n, sg_deadline)
        if c_sg < best_c:
            best_c = c_sg
    
    # Phase 3: Pair search
    if time.time() < deadline - 8:
        if n <= 250:
            best_h, best_c = fast_pair_search(best_h, n, deadline - 8,
                                              [0.02, 0.01, 0.005, 0.002, 0.001])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 8,
                                              [0.02, 0.01, 0.005, 0.002], radius=40)
    
    # Phase 4: Final SLSQP
    if time.time() < deadline - 5:
        ht = slsqp_optimize(best_h, n, 5000000, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Phase 5: Very fine pair search
    if time.time() < deadline - 2:
        if n <= 250:
            best_h, best_c = fast_pair_search(best_h, n, deadline - 1,
                                              [0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.001, 0.0005], radius=30)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def try_binary_init(n, rng):
    """Try a binary (0/1) initialization with random pattern."""
    h = np.zeros(n)
    indices = rng.permutation(n)
    h[indices[:n//2]] = 1.0
    return project_feasible(h, n)

def structured_init(n, rng):
    """Create a U-shaped or V-shaped initialization."""
    x = np.linspace(0, 2, n)
    # Try different structures
    choices = [
        0.5 + 0.3 * np.sin(np.pi * x),   # bell
        0.5 - 0.3 * np.cos(np.pi * x),   # valley
        0.5 + 0.2 * np.sin(2 * np.pi * x),  # two-hump
        np.where(x < 1, 0.8, 0.2),      # step
        np.where(x < 0.5, 0.9, np.where(x < 1.5, 0.3, 0.9)),  # U-shape
    ]
    idx = rng.randint(len(choices))
    h = choices[idx].copy()
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200
    
    best_h = None
    best_c = float('inf')
    
    # === Phase 1: Optimize at original resolution ===
    if h_init is not None:
        alloc = (deadline - t0) * 0.35
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # === Phase 2: Try other resolutions ===
    resolutions = [100, 150, 200, 250, 300]
    if h_init is not None:
        x_init = np.linspace(0, 2, n_init)
    for n_try in resolutions:
        if h_init is not None and n_try == n_init:
            continue
        if time.time() > deadline - 60:
            break
        
        remaining = deadline - time.time()
        alloc = min(60, remaining * 0.12)
        if alloc < 15:
            break
        
        if best_h is not None:
            # Interpolate from best
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
        else:
            h_interp = structured_init(n_try, rng)
        
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # === Phase 3: Multi-start at best resolution ===
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(3, int(remaining / 30))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            # Different perturbation strategies
            if r % 3 == 0:
                hp = best_h + rng.normal(0, 0.03, n)
            elif r % 3 == 1:
                hp = best_h.copy()
                idx = rng.randint(0, n, n // 4)
                hp[idx] += rng.uniform(-0.05, 0.05, len(idx))
            else:
                # Binary-ish perturbation
                hp = best_h.copy()
                idx = rng.randint(0, n, n // 3)
                hp[idx] = rng.choice([0.0, 0.25, 0.5, 0.75, 1.0], len(idx))
            
            hp = project_feasible(hp, n)
            alloc = min(30, (deadline - time.time() - 20) * 0.3)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # === Phase 4: Try binary/structured initializations at best n ===
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        for init_type in range(4):
            if time.time() > deadline - 15:
                break
            if init_type == 0:
                h_try = try_binary_init(n, rng)
            else:
                h_try = structured_init(n, rng)
            alloc = min(20, (deadline - time.time() - 15) * 0.3)
            if alloc < 8:
                break
            h_opt, c_opt = optimize_n(h_try, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # === Phase 5: Final intensive polishing at best resolution ===
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Final SLSQP with very high alpha
        ht = slsqp_optimize(best_h, n, 10000000, maxiter=2000, ftol=1e-17)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        
        # Very fine pair search
        if time.time() < deadline - 2:
            if n <= 250:
                best_h, best_c = fast_pair_search(best_h, n, deadline - 1,
                                                  [0.001, 0.0005, 0.0003, 0.0001])
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                                  [0.001, 0.0005, 0.0003], radius=30)
        
        # Subgradient polish
        if time.time() < deadline - 2:
            best_h, c_sg, _ = subgradient_refine(best_h, n, deadline - 1, max_iter=5000)
            if c_sg < best_c:
                best_c = c_sg
    
    # Fallback
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
