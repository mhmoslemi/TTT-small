def delta_corr(h, n, a, b, d):
    """Compute the change in correlation vector when h[a]+=d, h[b]-=d."""
    dx = 2.0 / n
    delta = np.zeros(2*n-1)
    for k in range(-(n-1), n):
        # k is the lag
        val = 0.0
        # h[a] as first index: a+k valid?
        if 0 <= a+k < n:
            val += (1 - h[a+k])
        # h[a] as second index: a-k valid? (h[a-k]*(1-h[a]))

            val += -(1 - h[b+k])
        # h[b] as second index: b-k valid?

            val += h[b-k]

    dx = 2.0 / n
    k_range = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # h[a+k] for valid k
    ak = a + k_range  # indices into h
    mask_ak = (ak >= 0) & (ak < n)
    term1 = np.zeros(2*n-1)
    term1[mask_ak] = 1 - h[ak[mask_ak]]
    
    # h[a-k] for valid k  
    ak2 = a - k_range
    mask_ak2 = (ak2 >= 0) & (ak2 < n)
    term2 = np.zeros(2*n-1)
    term2[mask_ak2] = -h[ak2[mask_ak2]]
    
    # h[b+k] for valid k
    bk = b + k_range
    mask_bk = (bk >= 0) & (bk < n)
    term3 = np.zeros(2*n-1)
    term3[mask_bk] = -(1 - h[bk[mask_bk]])
    
    # h[b-k] for valid k
    bk2 = b - k_range
    mask_bk2 = (bk2 >= 0) & (bk2 < n)
    term4 = np.zeros(2*n-1)
    term4[mask_bk2] = h[bk2[mask_bk2]]
    
    return d * dx * (term1 + term2 + term3 + term4)

            rng_local = np.random.RandomState(42 + _pass)
            if len(candidates) > 5000:
                chosen = rng_local.choice(len(candidates), size=5000, replace=False)
                candidates = [candidates[idx] for idx in chosen]
            rng_local.shuffle(candidates)
            for (i, j) in candidates:
                if time.time() > deadline:
                    break
                if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    dc = delta_corr_fast(h, n, i, j, -d, k_range)
                    nm = (c + dc).max()
                    if nm < best_c - 1e-14:
                        h[i] -= d
                        h[j] += d
                        c += dc
                        best_c = nm
                        improved = True
            if not improved:
                break
    return h, best_c


def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)


def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_con_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 200
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha
    for a in [3000, 20000, 100000]:
        if time.time() > deadline - 700:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=2500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = project_feasible(ht, n)

    # Phase 2: Coarse fast pair search
    t2 = t0 + min(350, budget_s * 0.32)
    best_h, best_c = fast_pair_search(best_h, n, t2, [0.02, 0.01, 0.005, 0.002])

    # Phase 3: Guided pair search
    if time.time() < deadline - 500:
        t3 = t0 + min(500, budget_s * 0.45)
        best_h, best_c = guided_pair_search(best_h, n, t3, [0.005, 0.003, 0.002, 0.001])

    # Phase 4: SLSQP polish
    if time.time() < deadline - 350:
        ht = slsqp_optimize(best_h, n, 300000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = project_feasible(ht, n)

    # Phase 5: Fine fast pair search
    t5 = t0 + min(800, budget_s * 0.75)
    best_h, best_c = fast_pair_search(best_h, n, t5, [0.005, 0.002, 0.001, 0.0005])

    # Phase 6: Multi-resolution check
    if time.time() < deadline - 120:
        x_init = np.linspace(0, 2, n)
        for n_try in [160, 240]:
            if time.time() > deadline - 50:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)
            ht = slsqp_optimize(h_interp, n_try, 15000, maxiter=1500)
            h_interp = project_feasible(ht, n_try)
            h_interp, c_try = fast_pair_search(h_interp, n_try,
                                                min(time.time() + 40, deadline - 20),
                                                [0.01, 0.005, 0.002, 0.001])
            c_try = compute_c5(h_interp, n_try)
            if c_try < best_c:
                best_c = c_try
                best_h = h_interp
                n = n_try

    # Phase 7: Final SLSQP
    if time.time() < deadline - 30:
        ht = slsqp_optimize(best_h, n, 500000, maxiter=2500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = project_feasible(ht, n)

    # Phase 8: Final fine pair search
    best_h, best_c = fast_pair_search(best_h, n, deadline - 2,
                                       [0.002, 0.001, 0.0005, 0.0002])

    # Final projected subgradient steps
    c = compute_full_corr(best_h, n)
    for _ in range(500):
        if time.time() > deadline - 1:
            break
        k_star = int(np.argmax(c)) - (n - 1)
        g = grad_active(best_h, n, k_star)
        idx_dec = np.where(g > 0.01)[0]
        idx_inc = np.where(g < -0.01)[0]
        if len(idx_dec) == 0 or len(idx_inc) == 0:
            break
        a = idx_dec[np.argmax(g[idx_dec])]
        b = idx_inc[np.argmin(g[idx_inc])]
        step = min(0.001, best_h[a] * 0.5, (1.0 - best_h[b]) * 0.5)
        if step < 1e-6:
            break
        k_range = make_k_range(n)
        dc = delta_corr_fast(best_h, n, a, b, -step, k_range)
        nm = (c + dc).max()
        if nm < best_c - 1e-15:
            best_h[a] -= step
            best_h[b] += step
            c += dc
            best_c = nm
        else:
            break

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
