                                         max_sweeps=500, n_restarts=3, seed=seed+100)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
            if time.time() < p6_end - 1:
                hs = slsqp_optimize(hb + 0.03 * rng.randn(n), n, 50000000, 2000, p6_end - 1)
                cs = compute_c5(hs, n)
                if cs < best_c:
                    best_c = cs
                    best_h = hs.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
