import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_subproblem(h, i, n):
    """Solve the 1D subproblem for coordinate i."""
    dx = 2.0 / n
    C = compute_c5_vec(h, n)
    
    # Compute A_m for each m
    A = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        a = 0.0
        if k >= 0:
            if i <= n - 1 - k:
                a += (1.0 - h[i + k])
            if k <= i and i <= n - 1:
                a += (-h[i - k])
        else:
            p = -k
            if p <= i and i <= n - 1:
                a += (1.0 - h[i - p])
            if i <= n - 1 - p:
                a += (-h[i + p])
        A[m] = a * dx
    
    # B_m = C[m] - A_m * h[i]
    B = C - A * h[i]
    
    # Minimize f(x) = max_m (A_m * x + B_m) over x in [0, 1]
    # f is convex PL, evaluate at a few points
    x_vals = np.linspace(0, 1, 21)
    f_vals = np.max(A[:, None] * x_vals[None, :] + B[:, None], axis=0)
    
    # Refine around the minimum
    best_idx = np.argmin(f_vals)
    x_lo = x_vals[max(0, best_idx - 1)]
    x_hi = x_vals[min(len(x_vals) - 1, best_idx + 1)]
    
    # Bisection on subgradient
    x_left = x_lo
    x_right = x_hi
    for _ in range(30):
        x_mid = (x_left + x_right) / 2
        f_mid = np.max(A * x_mid + B)
        # Left subgradient: min A among active
        active = np.abs(A * x_mid + B - f_mid) < 1e-12
        if not active.any():
            break
        left_deriv = A[active].min()
        right_deriv = A[active].max()
        if left_deriv > 0:
            x_right = x_mid
        elif right_deriv < 0:
            x_left = x_mid
        else:
            break
    
    x_opt = (x_left + x_right) / 2
    # Evaluate at candidates
    candidates = [0.0, 1.0, x_opt, x_left, x_right, x_lo, x_hi, x_vals[best_idx]]
    best_x = 0
    best_f = np.inf
    for x_c in candidates:
        x_c = max(0.0, min(1.0, x_c))
        f_c = np.max(A * x_c + B)
        if f_c < best_f:
            best_f = f_c
            best_x = x_c
    
    return best_x

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha if mx > 500 else m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas=None, mu=100.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
