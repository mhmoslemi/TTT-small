def compute_delta_c(c, h, i, j, n):
    """Compute change in correlation when flipping h[i]: 1->0, h[j]: 0->1"""
    # c[m] = sum_a h[a] * (1-h[a+m-n+1])
    # delta[m] = -(1-h[i+m-n+1]) + (1-h[j+m-n+1]) + h[i-m+n-1] - h[j-m+n-1]
    # with valid index checks
    
    delta = np.zeros(2*n-1)
    for m in range(2*n-1):
        # a+m-n+1 must be in [0, n-1]
        ai = i + m - n + 1
        aj = j + m - n + 1
        bi = i - m + n - 1
        bj = j - m + n - 1
        
        d = 0
        if 0 <= ai < n:
            d -= (1 - h[ai])
        if 0 <= aj < n:
            d += (1 - h[aj])
        if 0 <= bi < n:
            d += h[bi]
        if 0 <= bj < n:
            d -= h[bj]
        delta[m] = d
    return delta

def project(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break

            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break

    """Compute smooth max of correlation and its gradient."""
    c = np.correlate(h, 1 - h, mode='full')
    x = alpha * c
    xm = x.max()
    s = np.exp(x - xm)
    ss = s.sum()
    s /= ss
