t ≥ dx * Σ_{pairs for lag k} [(1-h̄[j])h[i] - h̄[i]h[j]]

Or equivalently:
dx * Σ [(1-h̄[j])h[i] - h̄[i]h[j]] - t ≤ 0

Now, for the LP, I can use scipy.optimize.linprog or cvxpy. Given the size (n=250, 499 constraints), it should be very fast.

Let me also think about what else to add:

1. **Multiple SLP iterations**: The SLP converges to a local optimum. Multiple iterations with the updated h_bar should improve.

2. **Damping**: Instead of fully updating h to the LP solution, we can use h_new = (1-β)*h_old + β*h_LP for stability.

3. **Binary local search**: After SLP, snap to binary and do local search.

4. **Multi-resolution**: Try different n values.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slp_iteration(h_bar, n):
    """One SLP iteration: linearize bilinear terms and solve LP."""
    dx = 2.0 / n
    num_lags = 2 * n - 1
    # Build constraint matrix
    # Variables: [h_0, ..., h_{n-1}, t], total n+1
    A = np.zeros((num_lags, n + 1))
    b = np.zeros(num_lags)
    
    for k in range(num_lags):
        l = k - (n - 1)
        if l >= 0:
            # c_l = sum_{i=0}^{n-1-l} h[i] * (1-h[i+l])
            for i in range(n - l):
                j = i + l
                A[k, i] += dx * (1 - h_bar[j])
                A[k, j] += -dx * h_bar[i]
        else:
            m = -l
            for i in range(n - m):
                j = i + m
                A[k, j] += dx * (1 - h_bar[i])
                A[k, i] += -dx * h_bar[j]
        A[k, n] = -1.0  # -t
        b[k] = 0.0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: 0 <= h_i <= 1, t >= 0 (or t >= some lower bound)
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        t_new = res.x[n]
        return h_new, t_new
    else:
        return h_bar, None

def slp_optimize(h_init, n, deadline, max_iter=200, damping=1.0):
    """Run SLP iterations."""
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_new, t_val = slp_iteration(h, n)
        if h_new is None or t_val is None:
            break
        # Damped update
        h_new_damped = (1 - damping) * h + damping * h_new
        h_new_damped = project_feasible(h_new_damped, n)
        c = compute_c5(h_new_damped, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_new_damped.copy()
        h = h_new_damped
        # Check convergence
        if abs(c - best_c) < 1e-14 and it > 5:
            # Check if we've truly converged
            if np.max(np.abs(h - best_h)) < 1e-10:
                break
    
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    """SLSQP with log-sum-exp smooth max."""
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, deadline, max_sweeps=500):
    """Local search on binary solutions: try swaps."""
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0])
    zeros = set(np.where(h_bin == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros_list = list(zeros)
        ones_list = list(ones)
        rng_sweep = np.random.RandomState(sweep)
        rng_sweep.shuffle(ones_list)
        for i in zeros_list:
            if time.time() > deadline:
                break

                else:

            break

def simulated_annealing_binary(h, n, deadline, max_iter=100000):
    """Simulated annealing on binary solutions."""
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    
    T = 0.01
    cooldown = 0.9999
    rng = np.random.RandomState(42)
    ones = set(np.where(h_bin == 1)[0])
    zeros = set(np.where(h_bin == 0)[0])
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 1000 == 0:
            T *= cooldown
            if T < 1e-8:
                T = 1e-8
        i = rng.choice(list(zeros))
        j = rng.choice(list(ones))
        h_bin[i], h_bin[j] = 1.0, 0.0
        c = compute_c5(h_bin, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_bin.copy()
            ones.remove(j); ones.add(i)
            zeros.remove(i); zeros.add(j)
        elif rng.random() < np.exp(-(c - best_c) / T):
            ones.remove(j); ones.add(i)
            zeros.remove(i); zeros.add(j)
        else:
            h_bin[i], h_bin[j] = 0.0, 1.0
    
    return best_h, best_c

def pair_refine(h, n, deadline, deltas):
    """Pairwise exchange refinement."""
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in range(0, n, max(1, n // 200)):
                if time.time() > deadline:
                    break
                for j in range(n - 1, i, -max(1, n // 200)):
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s * d, h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                prog = True
                            else:
                                h[i], h[j] = hi - s * d, hj + s * d
    return h, best_c
