import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Add/subtract to non-saturated elements
        if s < target:
            room = 1.0 - h
            rsum = room.sum()
            if rsum < 1e-14:
                break
            h = h + (target - s) / rsum * room
        else:
            hsum = h.copy()
            hsum[h == 0] = 0  # can only reduce where h > 0
            hsum[h == 1] = 0
            if hsum.sum() < 1e-14:
                break
            h = h - (s - target) / hsum.sum() * hsum
    return np.clip(h, 0.0, 1.0)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def adam_optimize(h, n, deadline, lr=0.01, alpha_sm=1000, max_iter=200000):
    """Adam optimizer with softmax-smoothed max objective."""
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    m = np.zeros(n)  # Adam 1st moment
    v = np.zeros(n)  # Adam 2st moment
    step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_c = np.inf
    best_h = h.copy()
    
    while step < max_iter:
        if time.time() > deadline:
            break
        step += 1
        
        corr = compute_corr(h, n)  # shape (2n-1,)
        vals = corr * dx
        cmax = vals.max()
        
        # Softmax over top values for smoother gradient
        # Use top-10 or so
        if alpha_sm > 0:
            shifted = (vals - cmax) * alpha_sm
            # Clip for numerical stability
            shifted = np.clip(shifted, -50, 0)
            weights = np.exp(shifted)
            weights /= weights.sum()
        else:
            idx = np.argmax(vals)
            weights = np.zeros(len(vals))
            weights[idx] = 1.0
        
        # Gradient: sum_k weights[k] * d(corr[k]*dx)/dh[i]
        # d(corr[k])/dh[i] where k is shift (= idx - (n-1))
        grad = np.zeros(n)
        for k_idx in range(len(vals)):
            if weights[k_idx] < 1e-6:
                continue
            shift = k_idx - (n - 1)  # actual shift
            # d/dh[i] of sum_j h[j]*(1-h[j+shift])
            # = (1-h[i+shift]) - h[i-shift] (with OOB=0)
            i_plus = i = np.arange(n)
            # (1-h[i+shift]) valid where 0 <= i+shift < n
            mask_p = (np.arange(n) + shift >= 0) & (np.arange(n) + shift < n)
            # -h[i-shift] valid where 0 <= i-shift < n
            mask_m = (np.arange(n) - shift >= 0) & (np.arange(n) - shift < n)
            
            g = np.zeros(n)
            g[mask_p] += (1 - h[np.arange(n)[mask_p] + shift])
            g[mask_m] -= h[np.arange(n)[mask_m] - shift]
            grad += weights[k_idx] * dx * g
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad**2
        m_hat = m / (1 - b1**step)
        v_hat = v / (1 - b2**step)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h_new = project_feasible(h_new, n)
        h = h_new
        
        c = cmax  # approximate, we computed it before the update
        # Actually recompute
        if step % 100 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

    dx = 2.0 / n
    h = h.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_c = np.inf
    best_h = h.copy()
    
    while step < max_iter:
        if time.time() > deadline:
            break
        step += 1
        
        corr = compute_corr(h, n)
        vals = corr * dx
        cmax = vals.max()
        
        # Find top-k shifts
        idx_sorted = np.argsort(vals)[::-1][:top_k]
        
        grad = np.zeros(n)
        for k_idx in idx_sorted:
            shift = k_idx - (n - 1)
            # g[i] = (1-h[i+shift]) - h[i-shift]
            g = np.zeros(n)
            # (1-h[i+shift]) for valid i
            start_p = max(0, -shift)
            end_p = min(n, n - shift)
            g[start_p:end_p] += (1 - h[start_p + shift:end_p + shift])
            # -h[i-shift] for valid i
            start_m = max(0, shift)
            end_m = min(n, n + shift)
            g[start_m:end_m] -= h[start_m - shift:end_m - shift]
            grad += dx * g
        
        grad /= top_k  # average over top-k
        
        # Adam
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1**step)
        v_hat = v / (1 - b2**step)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        
        if step % 500 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

    """Simple projection to [0,1] with sum = n/2"""

            if fs < 1e-14:

    prev_best = best_c

        m_hat = m_v / (1 - b1**step)

            prev_best = best_c
    
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c
