```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_threshold = 1e-14
    for mm in range(2 * n - 1):
        if w[mm] < idx_threshold:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active_shifts(h, n, fraction=0.002, max_k=20):
    c_vec = compute_c5_vec(h, n)
    m = c_vec.max()
    thresh = m - max(1e-8, fraction * m)
    active = np.where(c_vec >= thresh)[0]
    if len(active) > max_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:max_k]]
    return active

def run_lbfgs_full(h, n, deadline, configs):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(lse_obj, h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-19,
                                   'gtol': 1e-15, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_lbfgs_topk(h, n, deadline, configs, max_active=15):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            active = get_active_shifts(h, n, fraction=0.002, max_k=max_active)
            if len(active) == 0:
                continue
            res = minimize(topk_obj, h, jac=topk_grad,
                          args=(n, active, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-19,
                                   'gtol': 1e-15, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coord_descent_round(h, n, c_cur, max_rounds=50):
    dx = 2.0 / n
    for r in range(max_rounds):
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        thresh = m - max(1e-9, 0.001 * m)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > 25:
            active = active[np.argsort(-c_vec[active])][:25]
        if len(active) == 0:
            break

        n_act = max(1, len(active))
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= n_act

        lr = 0.05
        improved = False
        for _ in range(30):
            h_new = np.clip(h - lr * grad, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_cur - 1e-16:
                h = h_new
                c_cur = c_new
                improved = True
            else:
                lr *= 0.7
                if lr < 1e-12:
                    break
        if not improved:
            break
    return h, c_cur

def pair_local_search(h, n, c_cur, max_iter=500, radius=40):
    best_c = c_cur
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    for _ in range(max_iter):
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        abs_j = min(abs(i - j), n - abs(i - j))
        if abs_j > radius and abs_j < n - radius:
            continue
        for d in [0.001, 0.0005, 0.0001]:
            if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                h[i] += d
                h[j] -= d
                c = compute_c5(h, n)
                if c < best_c - 1e-16:
                    best_c = c
                    best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
            if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                h[i] -= d
                h[j] += d
                c = compute_c5(h, n)
                if c < best_c - 1e-16:
                    best_c = c
                    best_h = h.copy()
                else:
                    h[i] += d
                    h[j] -= d
    return best_h, best_c
```
