```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(h, n, alpha, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    sh = alpha * corr
    sh_max = sh.max()
    exp_sh = np.exp(sh - sh_max)
    total = exp_sh.sum()
    w = exp_sh / total
    f_val = (sh_max + np.log(total)) / alpha + mu * (h.sum() - n / 2.0) ** 2

    n2m1 = 2 * n - 1
    p_idx = np.arange(n)[:, None].astype(np.int64)
    m_idx = (np.arange(n2m1)[None, :] - (n - 1)).astype(np.int64)

    pm = p_idx + m_idx
    valid_fwd = ((pm >= 0) & (pm < n)).astype(np.float64)
    pm_c = np.clip(pm, 0, n - 1)

    pmm = p_idx - m_idx
    valid_bwd = ((pmm >= 0) & (pmm < n)).astype(np.float64)
    pmm_c = np.clip(pmm, 0, n - 1)

    S0 = valid_fwd.sum(axis=1)
    S1 = (valid_fwd * h[pm_c] * w[None, :]).sum(axis=1)
    S2 = (valid_bwd * h[pmm_c] * w[None, :]).sum(axis=1)

    grad = dx * (S0 - S1 - S2)
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return float(f_val), grad

def lbfgs_optimize(h, n, alpha, mu, maxiter, tol, deadline):
    def fun(x):
        return obj_and_grad(x, n, alpha, mu)[0]

    def jac(x):
        return obj_and_grad(x, n, alpha, mu)[1]

    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': tol,
                                'gtol': 1e-12, 'maxls': 50})
        if time.time() > deadline:
            return project_feasible(h, n)
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def pair_search(h, n, deadline, deltas, radius=100):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def grad_descent_polish(h, n, deadline, alpha=50000, mu=1000, n_steps=2000):
    best_c = compute_c5(h, n)
    h = h.copy()
    step_size = 0.01 / n
    for it in range(n_steps):
        if time.time() > deadline:
            break
        _, grad = obj_and_grad(h, n, alpha, mu)
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            step_size *= 0.7
            if step_size < 1e-12:
                break
    return h, best_c

def full_optimize(h_start, n, deadline, alphas, mus, lbfgs_iters=500):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    for a in alphas:
        if time.time() > deadline - 10:
            break
        for mu in mus:
            if time.time() > deadline - 5:
                break
            ht = lbfgs_optimize(h, n, a, mu, lbfgs_iters, 1e-14, deadline - 3)
            c = compute_c5(ht, n)
            if c < best_c - 1e-15:
                best_c = c
                h = ht

    return h, best_c

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B from parent (n=250)
    alloc1 = min(250, (deadline - t0) * 0.25)
    alphas1 = [50, 200, 1000, 5000, 20000, 50000, 100000]
    mus1 = [100, 500, 2000]
    h1, c1 = full_optimize(best_h, n_init, t0 + alloc1, alphas1, mus1, 400)
    if c1 < best_c:
        best_c = c1
        best_h = h1
        best_n = n_init

    # Phase 2: Diverse starting points at n=250
    x_grid = np.linspace(0, 2, n_init)
    diverse_starts = []

    # Sine-based
    h_sine = 0.5 + 0.25 * np.sin(2 * np.pi * x_grid)
    diverse_starts.append(project_feasible(h_sine, n_init))

    # Two-block
    h_block = np.zeros(n_init)
    h_block[:n_init // 2] = 0.7
    h_block[n_init // 2:] = 0.3
    diverse_starts.append(project_feasible(h_block, n_init))

    # Random smooth
    h_rand = 0.5 + 0.2 * rng.normal(0, 1, n_init)
    h_rand = np.convolve(h_rand, np.ones(10) / 10, mode='same')
    diverse_starts.append(project_feasible(h_rand, n_init))

    # Perturbed parent
    h_pert = project_feasible(best_h + 0.1 * rng.normal(0, 1, n_init), n_init)
    diverse_starts.append(h_pert)

    # Anti-symmetric
    h_anti = np.zeros(n_init)
    for i in range(n_init):
        h_anti[i] = 0.5 + 0.3 * np.sin(2 * np.pi * (i + 1) / (n_init / 4.0))
    diverse_starts.append(project_feasible(h_anti, n_init))

    remaining = deadline - time.time()
    per_start = max(20, remaining * 0.2 / len(diverse_starts))
    for h_s in diverse_starts:
        if time.time() > deadline - 60:
            break
        alloc = min(per_start, (deadline - time.time()) * 0.15)
        if alloc < 10:
            break
        alphas2 = [500, 5000, 50000]
        mus2 = [500, 2000]
        h2, c2 = full_optimize(h_s, n_init, time.time() + alloc, alphas2, mus2, 300)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n_init

    # Phase 3: Try different n values
    for n_try in [300, 400, 200]:
        if time.time() > deadline - 100:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, np.linspace(0, 2, best_n), best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(120, (deadline - time.time()) * 0.2)
        if alloc < 30:
            break
        alphas3 = [100, 1000, 10000, 50000, 100000]
        mus3 = [500, 2000, 5000]
        h3, c3 = full_optimize(h_interp, n_try, time.time() + alloc, alphas3, mus3, 350)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_try

    # Phase 4: Gradient descent polish + pair search on best
    if time.time() < deadline - 80:
        n_cur = best_n
        alloc_gd = min(100, (deadline - time.time()) * 0.25)
        best_h, best_c = grad_descent_polish(best_h, n_cur, time.time() + alloc_gd,
                                             alpha=100000, mu=2000, n_steps=5000)

    if time.time() < deadline - 40:
        n_cur = best_n
        if n_cur <= 200:
            radius = 150
        elif n_cur <= 300:
            radius = 80
        else:
            radius = 50
        alloc_ps = deadline - time.time() - 20
        best_h, best_c = pair_search(best_h, n_cur, time.time() + alloc_ps,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                     radius=radius)

    # Phase 5: Final L-BFGS-B polish
    if time.time() < deadline - 10:
        n_cur = best_n
        ht = lbfgs_optimize(best_h, n_cur, 200000, 5000, 500, 1e-14, deadline - 5)
        c = compute_c5(ht, n_cur)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 6: Ultra-fine pair search
    if time.time() < deadline - 3:
        n_cur = best_n
        radius = max(30, n_cur // 5)
        best_h, best_c = pair_search(best_h, n_cur, deadline - 1,
                                     [0.001, 0.0005, 0.0002, 0.0001],
                                     radius=radius)

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
```
