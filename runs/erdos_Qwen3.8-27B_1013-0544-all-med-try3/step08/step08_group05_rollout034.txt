def fft_correlate_full(a, b):
    n = len(a)

            B[valid_jm] += (1.0 - h[j_arr[valid_jm] - s_max])
        if np.any(valid_jp):
            B[valid_jp] -= h[j_arr[valid_jp] + s_max]

        min_A = np.min(A)
        min_B = np.min(B)
        delta_best = min_A + min_B

        if delta_best >= -1e-14:
            # No improvement at s_max; try simulated annealing occasionally
            if temperature > 1e-6 and rng.random() < 0.05:
                # Random swap
                best_i_idx = rng.randint(len(i_arr))
                best_j_idx = rng.randint(len(j_arr))
            else:
                # Cool and break after several failed attempts
                temperature *= 0.9
                if temperature < 1e-8:
                    break
                continue
        else:
            best_i_idx = int(np.argmin(A))
            best_j_idx = int(np.argmin(B))

        i = i_arr[best_i_idx]
        j = j_arr[best_j_idx]

        # Apply swap
        h[i] = 0.0
        h[j] = 1.0
        new_corr = fft_correlate_full(h, 1.0 - h)
        new_max_idx = int(np.argmax(new_corr))
        new_c = new_corr[new_max_idx] * dx

        accept = new_c < cur_c - 1e-15
        if not accept and temperature > 1e-6 and (cur_c - new_c) * dx < 1.0:
            # Accept with SA probability
            delta_c = (new_c - cur_c)
            if delta_c < 0.05 and rng.random() < np.exp(-delta_c / max(temperature, 1e-10)):
                accept = True

        if accept:
            cur_c = new_c
            max_idx = new_max_idx
            corr = new_corr
            one_set.discard(i)
            one_set.add(j)
            zero_set.discard(j)
            zero_set.add(i)
            one_arr = np.array(sorted(one_set))
            zero_arr = np.array(sorted(zero_set))
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            temperature = max(temperature * 0.999, sa_temp * 0.1)
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
            temperature *= 0.95

    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old)
    x_new = np.linspace(0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def refine_binary(h_bin, n, deadline, max_sweeps=3000, seed=99):
    """Local search allowing small continuous perturbations around binary."""
    h = h_bin.astype(float).copy()
    best_c, _ = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    temp = 0.002

    for it in range(max_sweeps):
        if time.time() > deadline:
            break
        # Pick a random position and nudge it
        pos = rng.randint(n)
        delta = rng.randn() * 0.1
        h_new = h.copy()
        h_new[pos] = np.clip(h_new[pos] + delta, 0.0, 1.0)
        # Project to maintain sum
        h_new = project_feasible(h_new, n)
        c_new, _ = compute_c5(h_new, n)
        if c_new < best_c - 1e-15 or (c_new < best_c + temp and rng.random() < 0.3):
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            temp = max(temp * 0.999, 0.0002)
        else:
            temp *= 0.995

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c, _ = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: SLSQP at n=1000 (15% time)
    p1 = t0 + 0.15 * (budget_s - 5)
    n1 = 1000
    h1 = interpolate_h(best_h, best_n, n1)
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > p1 - 5:
            break
        h1 = slsqp_optimize(h1, n1, alpha, 800, p1 - 5)
        c1, _ = compute_c5(h1, n1)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
            best_n = n1

    # Phase 2: Binary local search at n=1000 (25% time)
    p2 = t0 + 0.40 * (budget_s - 5)
    hb2, cb2 = binary_local_search(best_h, n1, p2 - 10, max_sweeps=8000, seed=seed)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()
        best_n = n1

    # Phase 3: SLSQP at n=2000 (15% time)
    p3 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p3 - 30:
        n3 = 2000
        h3 = interpolate_h(best_h, best_n, n3)
        for alpha in [5000, 50000, 200000]:
            if time.time() > p3 - 10:
                break
            h3 = slsqp_optimize(h3, n3, alpha, 500, p3 - 10)
            c3, _ = compute_c5(h3, n3)
            if c3 < best_c:
                best_c = c3
                best_h = h3.copy()
                best_n = n3

    # Phase 4: Binary local search at n=2000 (30% time)
    p4 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p4 - 60:
        n4 = 2000
        if best_n != n4:
            h4 = interpolate_h(best_h, best_n, n4)
        else:
            h4 = best_h.copy()
        idx4 = np.argsort(h4)[::-1]
        h4_bin = np.zeros(n4)
        h4_bin[idx4[:n4//2]] = 1.0
        hb4, cb4 = binary_local_search(h4_bin, n4, p4 - 10, max_sweeps=10000, seed=seed+1)
        if cb4 < best_c:
            best_c = cb4
            best_h = hb4.copy()
            best_n = n4

    # Phase 5: Final refinement (remaining time)
    p5 = dl
    if time.time() < p5 - 30:
        # Continuous refinement
        h5 = best_h.astype(float).copy()
        h5 += rng.randn(best_n) * 0.03
        h5 = project_feasible(h5, best_n)
        for alpha in [50000, 500000]:
            if time.time() > p5 - 20:
                break
            h5 = slsqp_optimize(h5, best_n, alpha, 500, p5 - 20)
            c5v, _ = compute_c5(h5, best_n)
            if c5v < best_c:
                best_c = c5v
                best_h = h5.copy()

        # Final binary polish
        if time.time() < dl - 5:
            h_bin = best_h.copy()
            idx_b = np.argsort(h_bin)[::-1]
            h_bin[idx_b[:best_n//2]] = 1.0
            h_bin[idx_b[best_n//2:]] = 0.0
            hb_f, cb_f = binary_local_search(h_bin, best_n, dl, max_sweeps=2000, seed=seed+10)
            if cb_f < best_c:
                best_c = cb_f
                best_h = hb_f.copy()

    best_h = project_feasible(best_h, best_n)
    best_c, _ = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
