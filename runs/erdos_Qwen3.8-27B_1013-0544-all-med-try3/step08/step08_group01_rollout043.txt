import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vector(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vector(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def v_shift_value(h, n, k_idx):
    """Compute V(k)*dx for a single shift index k_idx in the full correlation array."""
    dx = 2.0 / n
    return float(np.correlate(h, 1 - h, mode='full')[k_idx] * dx)

def grad_single_shift(h, n, k_idx):
    """Gradient of V(k)*dx with respect to h for a single shift."""
    dx = 2.0 / n
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def active_obj(x, n, active, alpha, mu):
    c_vec = compute_c5_vector(x, n)
    vals = c_vec[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c_vec = compute_c5_vector(x, n)
    vals = c_vec[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def full_obj(x, n, alpha, mu):
    c_vec = compute_c5_vector(x, n)
    m = c_vec.max()
    sh = alpha * (c_vec - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_grad(x, n, alpha, mu):
    c_vec = compute_c5_vector(x, n)
    m = c_vec.max()
    sh = alpha * (c_vec - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, range(len(w))):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_active(h, n, active, bounds, alpha, mu, maxiter=3000):
    try:
        res = minimize(active_obj, h, jac=active_grad,
                      args=(n, active, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 5, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

        res = minimize(full_obj, h, jac=full_grad,

                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 5, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

def get_active_shifts(h, n, top_k=10, rel_tol=0.002):
    """Get the active shifts (near-maximum V(k) values)."""
    c_vec = compute_c5_vector(h, n)
    max_val = c_vec.max()
    threshold = max_val - rel_tol * max_val
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.argsort(-c_vec)[:top_k]
    elif len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    return active

def subgradient_steps(h, n, deadline, steps=5000, lr_init=0.01):
    """Subgradient method for the max-V(k) problem."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vector(best_h, n)
    
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.9
    
    for step in range(steps):
        if time.time() > deadline:
            break
        
        # Find worst shift
        k_star = int(np.argmax(c_vec))
        
        # Compute subgradient
        g = grad_single_shift(best_h, n, k_star)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * g
        direction = momentum
        
        # Adaptive step size
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-10:
            break
        lr_step = lr / (1.0 + step * 0.001)
        
        # Take step and project
        h_trial = best_h - lr_step * direction
        h_trial = project_feasible(h_trial, n)
        
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial
            c_vec = compute_c5_vector(best_h, n)
        else:
            # Revert and reduce step
            lr *= 0.95
    
    return best_h, best_c
