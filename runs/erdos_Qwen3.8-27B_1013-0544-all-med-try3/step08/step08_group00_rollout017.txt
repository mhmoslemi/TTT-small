def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, lag_idx):
    """Gradient of overlap at a specific lag index with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:0+nv]) * dx
        grad[k:k+nv] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, lag_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:0+nv]) * dx
        if nv > 0:
            grad[k:k+nv] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:k+nv]) * dx
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:0+nv])
            if nv > 0:

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, alphas=(500, 2000, 8000, 20000, 50000)):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-19,
                                   'gtol': 1e-16, 'maxfun': 60000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:

    return h, best_c

            h_periodic = project_feasible(h_periodic, n)
            c_periodic = compute_c5(h_periodic, n)
            if c_periodic < best_c + 0.005:
                h_opt, c_opt = lbfgs_optimize(h_periodic, n, min(time.time() + 15, deadline - 10),
                                              alphas=(2000, 10000))
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()
        template = best_h[0:p].copy()
        # Re-tile best
    return best_h, best_c

def random_restart_optimize(h, n, deadline, n_restarts=8, amp=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    time_per_restart = max(5.0, (deadline - time.time()) / (n_restarts + 0.5))
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        a = amp * (1.0 - r / float(n_restarts + 1))
        pert = rng.uniform(-a, a, n)
        h_pert = project_feasible(h + pert, n)
        c_pert = compute_c5(h_pert, n)
        t_end = min(time.time() + time_per_restart, deadline - 10)
        if c_pert < best_c + 0.002:
            h_opt, c_opt = lbfgs_optimize(h_pert, n, t_end, alphas=(2000, 8000, 20000))
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick L-BFGS at original resolution (25s)
    t1 = min(t0 + 25, deadline - 500)
    if time.time() < t1:
        best_h, best_c = lbfgs_optimize(best_h, n_orig, t1, alphas=(500, 3000, 10000))

    # Phase 2: SQP at original resolution (40s)
    t2 = min(t0 + 65, deadline - 450)
    if time.time() < t2:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=250, rho_init=0.03, max_active=12)

    # Phase 3: Upsample to n=1000
    n_high = 1000
    x_old = np.linspace(0, 2, n_orig, endpoint=False)
    x_new = np.linspace(0, 2, n_high, endpoint=False)
    h_high = np.interp(x_new, x_old, best_h, period=2)
    h_high = project_feasible(h_high, n_high)
    c_high = compute_c5(h_high, n_high)

    # Phase 4: L-BFGS at n=1000 (100s)
    t4 = min(t0 + 165, deadline - 300)
    if time.time() < t4:
        h_high, c_high = lbfgs_optimize(h_high, n_high, t4,
                                         alphas=(1000, 4000, 15000, 40000))

    # Phase 5: SQP at n=1000 (100s)
    t5 = min(t0 + 265, deadline - 250)
    if time.time() < t5:
        h_high, c_high = sqp_optimize(h_high, n_high, t5, max_iter=300, rho_init=0.03, max_active=15)

    if c_high < best_c:
        best_c = c_high
        best_h = h_high

    # Phase 6: Subgradient with momentum at n=1000 (60s)
    t6 = min(t0 + 325, deadline - 200)
    if time.time() < t6:
        h_sg, c_sg = subgradient_momentum(best_h, n_high, t6, max_iter=300)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    # Phase 7: Periodic exploration (80s)
    t7 = min(t0 + 405, deadline - 170)
    if time.time() < t7:
        best_h, best_c = periodic_explore(best_h, n_high, t7)

    # Phase 8: Random restarts (80s)
    t8 = min(t0 + 485, deadline - 130)
    if time.time() < t8:
        best_h, best_c = random_restart_optimize(best_h, n_high, t8, n_restarts=6, amp=0.008)

    # Phase 9: SQP again after exploration (80s)
    t9 = min(t0 + 565, deadline - 100)
    if time.time() < t9:
        best_h, best_c = sqp_optimize(best_h, n_high, t9, max_iter=250, rho_init=0.02, max_active=20)

    # Phase 10: Fine local search (120s)
    t10 = min(t0 + 685, deadline - 60)
    if time.time() < t10:
        best_h, best_c = local_search_fine(best_h, n_high, t10, radius=20)

    # Phase 11: Final L-BFGS polish
    t11 = min(t0 + 745, deadline - 30)
    if time.time() < t11:
        best_h, best_c = lbfgs_optimize(best_h, n_high, t11, alphas=(20000, 80000, 200000))

    # Phase 12: Final SQP
    t12 = min(t0 + 785, deadline - 15)
    if time.time() < t12:
        best_h, best_c = sqp_optimize(best_h, n_high, t12, max_iter=150, rho_init=0.01, max_active=20)

    # Phase 13: Ultra-fine local search
    t13 = deadline - 5
    if time.time() < t13 - 3:
        best_h, best_c = local_search_fine(best_h, n_high, t13, radius=12)

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
