    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha, penalty):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    max_shifted = shifted.max()
    exp_s = np.exp(shifted - max_shifted)
    p = exp_s / exp_s.sum()
    val = cmax + (max_shifted + np.log(exp_s.sum())) / alpha

    grad_sm = np.zeros(n)
    p_full = p
    for m in range(n):
        idx1 = np.arange(m, m + n)
        first = np.dot(p_full[idx1][::-1], 1.0 - h)
        idx2 = np.arange(n - 1 - m, 2 * n - 1 - m)

    return val + val_pen, grad_sm + grad_pen

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
        if hasattr(res, 'x') and res.x is not None:

    if n <= 200 and time.time() < deadline - 30:
        h, best_c = pair_search(h, n, deadline - 30)
    elif time.time() < deadline - 30:
        h, best_c = near_pair_search(h, n, deadline - 30, radius=50)

    if time.time() < deadline - 15:
        ht = lbfgs_optimize(h, n, 500000, 5000.0, maxiter=200)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    if time.time() < deadline - 10:
        if n <= 200:
            h, best_c = pair_search(h, n, deadline - 5)
        else:
            h, best_c = near_pair_search(h, n, deadline - 5, radius=40)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at initial resolution
    alloc1 = min(250, (deadline - t0) * 0.30)
    h_opt, c_opt = optimize_for_n(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Try different resolutions
    x_init = np.linspace(0, 2, n_init, endpoint=False)
    for n_try in [200, 300, 240, 260]:
        if time.time() > deadline - 200:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.20)
        if alloc < 30:
            break
        h_opt, c_opt = optimize_for_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Gradient-guided + local search on best
    if time.time() < deadline - 80:
        n = len(best_h)
        ht, c = gradient_guided_search(best_h.copy(), n, deadline - 80, alpha=100000)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    # Phase 4: Simulated annealing for exploration
    if time.time() < deadline - 60:
        n = len(best_h)
        remaining_sa = deadline - time.time() - 60
        sa_deadline = time.time() + max(30, min(120, remaining_sa * 0.5))
        h_sa, c_sa = simulated_annealing(best_h.copy(), n, sa_deadline, rng)
        if c_sa < best_c - 1e-14:
            best_c = c_sa
            best_h = h_sa
            # Refine the SA solution
            if time.time() < deadline - 40:
                h_opt, c_opt = optimize_for_n(h_sa, n, deadline - 40, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 5: Multi-restart from best
    if time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 35))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(30, max(15, (deadline - time.time() - 25) * 0.3))
            if alloc < 15:
                break
            h_opt, c_opt = optimize_for_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 6: Final intensive local search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        rem = deadline - time.time() - 3
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, t0 + rem + t0 - t0 + deadline - 3)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 3, radius=80)

    # Phase 7: Final L-BFGS polish
    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        ht = lbfgs_optimize(best_h, n, 1000000, 10000.0, maxiter=100)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 8: Ultra-fine local search
    if best_h is not None and time.time() < deadline - 1:
        n = len(best_h)
        h = best_h.copy()
        best_c_local = compute_c5(h, n)
        for d in [0.0001, 0.00005, 0.00002, 0.00001]:
            improved = True
            while improved and time.time() < deadline - 1:
                improved = False
                for i in range(n):
                    if time.time() > deadline - 1:
                        break
                    lo = max(0, i - 100)
                    hi = min(n, i + 101)
                    for j in range(lo, hi):
                        if i == j:
                            continue
                        if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                            h[i] += d
                            h[j] -= d
                            c = compute_c5(h, n)
                            if c < best_c_local - 1e-15:
                                best_c_local = c
                                improved = True
                            else:
                                h[i] -= d
                                h[j] += d
            if best_c_local < best_c:
                best_c = best_c_local
                best_h = h.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
