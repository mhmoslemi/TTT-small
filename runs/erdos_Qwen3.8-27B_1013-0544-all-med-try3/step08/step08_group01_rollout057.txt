            c_vec = compute_c5_vec(z, n)
            c_max = c_vec.max()
            sh = alpha * (c_vec - c_max)
            shm = sh.max()
            w = np.exp(sh - shm)
            w /= w.sum()
            grad = smooth_max_grad(z, n, w, alpha)
            step = 1.0 / L
            h_next = project_feasible(z - step * grad, n)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            z_new = h_next + ((t - 1.0) / t_new) * (h_next - h_curr)
            z_new = project_feasible(z_new, n)
            h_curr = h_next
            z = z_new
            t = t_new
            c_curr = compute_c5(h_curr, n)
            if c_curr < best_c:
                best_c = c_curr
                best_h = h_curr.copy()
        h = best_h
    return best_h, best_c

def lbfgs_smooth(h, n, alpha, mu, maxiter=5000):
    def obj(x):
        c = compute_c5_vec(x, n)
        v = smooth_max_val(c, alpha)
        v += mu * (x.sum() - n / 2.0) ** 2
        return float(v)
    def grad(x):
        c = compute_c5_vec(x, n)
        c_max = c.max()
        sh = alpha * (c - c_max)
        shm = sh.max()
        w = np.exp(sh - shm)
        w /= w.sum()
        g = smooth_max_grad(x, n, w, alpha)
        g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': 50000})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def top_shifts_lbfgs(h, n, deadline, top_k=5, alphas=None):
    if alphas is None:
        alphas = [5000, 50000, 500000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        for _ in range(5):
            if time.time() > deadline - 3:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-8, 0.002 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = active[np.argsort(-c_vec[active])][:top_k]
            if len(active) < 1:
                break
            def obj_ts(x):
                c = compute_c5_vec(x, n)
                vals = c[active]
                v = smooth_max_val(vals, alpha)
                v += 0.01 * (x.sum() - n / 2.0) ** 2
                return float(v)
            def grad_ts(x):
                c = compute_c5_vec(x, n)
                vals = c[active]
                vmax = vals.max()
                sh = alpha * (vals - vmax)
                shm = sh.max()
                w_act = np.exp(sh - shm)
                w_act /= w_act.sum()
                g = np.zeros(n)
                dx = 2.0 / n
                for wi, m in zip(w_act, active):
                    s = m - (n - 1)
                    wm = wi * dx
                    if s >= 0:
                        ns2 = n - s
                        if ns2 > 0:
                            g[:ns2] += wm * (1.0 - x[s:])
                            g[s:] += wm * (-x[:ns2])
                    else:
                        ss = -s
                        ns2 = n - ss
                        if ns2 > 0:
                            g[ns2:] += wm * (1.0 - x[:ns2])
                            g[:ns2] += wm * (-x[ss:])
                g += 0.02 * (x.sum() - n / 2.0)
                return g
            try:
                res = minimize(obj_ts, best_h, jac=grad_ts, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 30000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def perturbation_restart(h, n, deadline, rng, magnitude=0.02):
    h_pert = h.copy()
    mask = rng.random(n) < 0.3
    h_pert[mask] += rng.uniform(-magnitude, magnitude, mask.sum())
    h_pert = project_feasible(h_pert, n)
    h_pert, c_pert = fista_opt(h_pert, n, min(time.time() + 30, deadline - 10),
                                alphas=[1000, 10000])
    return h_pert, c_pert

def fine_pair_search(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    n = 500
    if n_orig != n:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n)
        h_work = np.interp(x_new, x_old, best_h)
        h_work = project_feasible(h_work, n)
    else:
        h_work = best_h.copy()

    # Phase 1: Bilinear MM at n=500 (~300s)
    t_phase1_end = t0 + 280
    h_mm, c_mm = bilinear_mm(h_work, n, min(t_phase1_end, deadline - 400))
    if c_mm < best_c:
        best_c = c_mm
        best_h = h_mm
    h_work = h_mm if c_mm < compute_c5(h_work, n) else h_work

    # Phase 2: FISTA at n=500 (~150s)
    t_phase2_end = t0 + 430
    h_fista, c_fista = fista_opt(h_work, n, min(t_phase2_end, deadline - 250),
                                   alphas=[500, 5000, 50000, 200000])
    if c_fista < best_c:
        best_c = c_fista
        best_h = h_fista
    h_work = h_fista

    # Phase 3: L-BFGS polish at n=500 (~100s)
    t_phase3_end = t0 + 530
    if time.time() < t_phase3_end:
        for alpha, mu in [(10000, 1.0), (100000, 10.0), (500000, 100.0)]:
            if time.time() > t_phase3_end - 5:
                break
            h_new, c_new = lbfgs_smooth(h_work, n, alpha, mu, maxiter=5000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            h_work = h_new
            if time.time() > t_phase3_end:
                break

    # Phase 4: Bilinear MM again (exponential gap closing) (~100s)
    t_phase4_end = t0 + 630
    if time.time() < t_phase4_end:
        h_mm2, c_mm2 = bilinear_mm(h_work, n, min(t_phase4_end, deadline - 200), max_iter=200)
        if c_mm2 < best_c:
            best_c = c_mm2
            best_h = h_mm2
        h_work = h_mm2

    # Phase 5: Top-shifts focused L-BFGS (~80s)
    t_phase5_end = t0 + 710
    if time.time() < t_phase5_end:
        h_ts, c_ts = top_shifts_lbfgs(h_work, n, min(t_phase5_end, deadline - 150), top_k=5)
        if c_ts < best_c:
            best_c = c_ts
            best_h = h_ts
        h_work = h_ts

    # Phase 6: Try n=1000 (~150s)
    t_phase6_end = t0 + 860
    if time.time() < t_phase6_end:
        n2 = 1000
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n2)
        c_up = compute_c5(h_up, n2)
        h_up, c_up = bilinear_mm(h_up, n2, min(t_phase6_end - 30, deadline - 80), max_iter=100)
        h_up, c_up = fista_opt(h_up, n2, min(t_phase6_end - 10, deadline - 60),
                                alphas=[5000, 50000])
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            n = n2

    # Phase 7: FISTA + restart at current best (~60s)
    if time.time() < deadline - 30:
        h_r, c_r = perturbation_restart(best_h, n, deadline - 15, rng, magnitude=0.03)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r

    # Phase 8: Final polish
    if time.time() < deadline - 10:
        h_fin, c_fin = fista_opt(best_h, n, deadline - 5, alphas=[50000, 200000])
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Phase 9: Fine pair search
    if time.time() < deadline - 3:
        best_h, best_c = fine_pair_search(best_h, n, deadline - 2, radius=20)

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
