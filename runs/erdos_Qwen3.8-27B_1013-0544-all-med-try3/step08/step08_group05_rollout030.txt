    c = np.correlate(h, 1-h, mode='full') * dx
    # Normalize for numerical stability
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()

    # For each t, grad[t] = dx * sum_m w[m] * (d(c[m])/d(h[t]))
    # d(c[m])/d(h[t]) = (1-h[t+m-n+1]) if 0<=t+m-n+1<n else 0

    # dc[m]/dh[t]:
    #   Term 1: t appears as i, contributes (1-h[t+m-n+1]) if 0<=t+m-n+1<n
