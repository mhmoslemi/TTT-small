        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def compute_c5_vec(h, n):

    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])

            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

        best_h, best_c = pair_swap_search(best_h, n, t4_deadline,
            steps=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
            n_cand=12, n_active=12)

    # Phase 4: Second SQP round after local search (100s)
    t5_deadline = min(t0 + 600, deadline - 300)
    if t5_deadline > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n, t5_deadline,
            max_iter=400, rho_init=0.04, max_active=60)

    # Phase 5: Fine pair-swap (80s)
    t6_deadline = min(t0 + 680, deadline - 200)
    if t6_deadline > time.time() + 5:
        best_h, best_c = pair_swap_search(best_h, n, t6_deadline,
            steps=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            n_cand=15, n_active=15)

    # Phase 6: Upsample to n=1000 and optimize (300s)
    n_high = 1000
    t7_start = time.time()
    if t7_start < deadline - 250:
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)

        t7_deadline = min(t7_start + 80, deadline - 150)
        if t7_deadline > time.time() + 5:
            h_high, c_high = run_lbfgs(h_high, n_high, t7_deadline,
                configs=[(500, 100.0), (2000, 1000.0), (5000, 5000.0)])

        t8_deadline = min(time.time() + 120, deadline - 80)
        if t8_deadline > time.time() + 10:
            h_high, c_high = sqp_optimize(h_high, n_high, t8_deadline,
                max_iter=300, rho_init=0.05, max_active=40)

        t9_deadline = min(time.time() + 60, deadline - 25)
        if t9_deadline > time.time() + 5:
            h_high, c_high = pair_swap_search(h_high, n_high, t9_deadline,
                steps=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                n_cand=12, n_active=12)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n = n_high

    # Phase 7: Final SLSQP polish on best solution
    t10_deadline = deadline - 10
    if time.time() < t10_deadline - 20 and len(best_h) <= 1000:
        h_final, c_final = slqp_polish(best_h, len(best_h), t10_deadline,
            alpha=2000000, maxiter=5000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    if time.time() > t0 + budget_s - 2:
        pass

    return best_h, best_c, n_final
