def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

        for _iter in range(50000):
            if time.time() > deadline:
                return h, best_c5
            c_all = np.correlate(h, 1 - h, mode='full') * dx
            c5 = float(c_all.max())
            if c5 < best_c5:
                best_c5 = c5
            
            peak_idx = np.argsort(c_all)[::-1][:top_k]
            
            found_move = False
            for k in peak_idx:
                ki = int(k)
                # For lag ki, delta_c = d*(h[j-ki]-h[i-ki]-h[i+ki]+h[j+ki])
                # We want to maximize h[i-ki]+h[i+ki] - h[j-ki]-h[j+ki]
                g = np.zeros(n)
                for i in range(n):
                    val = 0.0
                    if i - ki >= 0:
                        val += h[i - ki]
                    if i + ki < n:
                        val += h[i + ki]
                    g[i] = val
                
                # i should have high g, j should have low g
                # with constraint h[i]+d <= 1 and h[j]-d >= 0
                valid_i_mask = (h + d <= 1.0 + 1e-12)
                valid_j_mask = (h - d >= -1e-12)
                valid_i_mask &= (g > g.min())
                valid_j_mask &= (g < g.max())
                
                idx_i = np.where(valid_i_mask)[0]
                idx_j = np.where(valid_j_mask)[0]
                
                if len(idx_i) == 0 or len(idx_j) == 0:
                    continue
                
                # Sort by g descending for i, ascending for j
                idx_i = idx_i[np.argsort(g[idx_i])[::-1]][:10]
                idx_j = idx_j[np.argsort(g[idx_j])][:10]
                
                for ii in idx_i:
                    for jj in idx_j:
                        if ii == jj:
                            continue
                        if not (0.0 <= h[ii] + d <= 1.0 and 0.0 <= h[jj] - d <= 1.0):
                            continue
                        h[ii] += d
                        h[jj] -= d
                        c5_new = compute_c5(h, n)
                        if c5_new < best_c5 - 1e-16:
                            best_c5 = c5_new
                            found_move = True
                            break
                        else:
                            h[ii] -= d
                            h[jj] += d
                    if found_move:
                        break
                if found_move:
                    break
            if not found_move:
                break
    return h, best_c5

def optimize_one(h_start, n, deadline, rng):
    """Full optimization pipeline for one resolution."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_run(best_h, n, a, maxiter=4000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht
    
    # Pair search
    if n <= 220:
        if time.time() < deadline - 15:
            best_h, best_c = pair_search_full(best_h, n, deadline - 15,
                                               [0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    else:
        if time.time() < deadline - 15:
            best_h, best_c = pair_search_radius(best_h, n, deadline - 15,
                                                 [0.05, 0.02, 0.01, 0.005, 0.002], radius=50)
    
    # Targeted lag descent
    if time.time() < deadline - 10:
        best_h, best_c = targeted_lag_descent(best_h, n, deadline - 10, top_k=3)
    
    # Another SLSQP
    if time.time() < deadline - 5:
        ht = slsqp_run(best_h, n, 2000000, maxiter=5000, ftol=1e-17)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    # Final pair search
    if n <= 220:
        if time.time() < deadline - 2:
            best_h, best_c = pair_search_full(best_h, n, deadline - 2,
                                               [0.002, 0.001, 0.0005, 0.0002])
    else:
        if time.time() < deadline - 2:
            best_h, best_c = pair_search_radius(best_h, n, deadline - 2,
                                                 [0.002, 0.001, 0.0005], radius=30)
    
    # Another targeted descent
    if time.time() < deadline - 1:
        best_h, best_c = targeted_lag_descent(best_h, n, deadline - 1, top_k=2)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def make_structured_init(n, rng):
    """Create a structured initial guess."""
    x = np.linspace(0, 2, n)
    pattern = rng.choice([0, 1, 2, 3])
    if pattern == 0:
        h = 0.5 + 0.45 * np.sin(np.pi * x)
    elif pattern == 1:
        h = 0.5 + 0.45 * np.cos(2 * np.pi * x)
    elif pattern == 2:
        # Block pattern
        block = max(1, n // 8)
        h = np.zeros(n)
        for i in range(n):
            pos = i % (2 * block)
            h[i] = 1.0 if pos < block else 0.0
    else:
        # Alternating smooth
        h = 0.5 + 0.3 * np.sin(3 * np.pi * x) + 0.2 * np.sin(5 * np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize parent at native resolution (largest allocation)
    if h_init is not None:
        h_good = project_feasible(h_init, n_init)
        alloc1 = min(250, (deadline - t0) * 0.30)
        h_opt, c_opt = optimize_one(h_good, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt
    
    # Phase 2: Try other resolutions
    resolutions = [100, 150, 200, 250, 120]
    tried = set()
    if best_h is not None:
        tried.add(len(best_h))
    
    for n_try in resolutions:
        if n_try in tried:
            continue
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time()
        alloc = min(70, remaining * 0.18)
        if alloc < 30:
            break
        tried.add(n_try)
        
        x_old = np.linspace(0, 2, len(best_h)) if best_h is not None else np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_try)
        if best_h is not None:
            h_interp = np.interp(x_new, x_old, best_h)
        else:
            h_interp = h_init[:len(h_init)]
            if len(h_interp) != n_try:
                x_init = np.linspace(0, 2, len(h_interp))
                h_interp = np.interp(x_new, x_init, h_interp)
        h_interp = project_feasible(h_interp, n_try)
        
        h_opt, c_opt = optimize_one(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Multiple restarts at best resolution
    if best_h is not None:
        n_best = len(best_h)
        restart_count = 0
        while time.time() < deadline - 80:
            restart_count += 1
            # Structured perturbation
            if restart_count % 3 == 1:
                h_try = best_h + rng.normal(0, 0.02, n_best)
            elif restart_count % 3 == 2:
                h_try = best_h + rng.normal(0, 0.05, n_best)
            else:
                h_try = make_structured_init(n_best, rng)
                # Blend with current best
                h_try = 0.5 * best_h + 0.5 * h_try
            h_try = project_feasible(h_try, n_best)
            
            alloc = min(50, (deadline - time.time() - 30) * 0.25)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_one(h_try, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Intensive final refinement at best resolution
    if best_h is not None and time.time() < deadline - 20:
        n_best = len(best_h)
        h_refine = best_h.copy()
        
        # Multiple rounds of fine pair search
        for rnd in range(3):
            if time.time() > deadline - 5:
                break
            h_refine = project_feasible(h_refine, n_best)
            if n_best <= 220:
                h_refine, c_r = pair_search_full(h_refine, n_best, deadline - 5,
                                                  [0.001, 0.0005, 0.0002])
            else:
                h_refine, c_r = pair_search_radius(h_refine, n_best, deadline - 5,
                                                    [0.001, 0.0005], radius=30)
            if c_r < best_c:
                best_c = c_r
                best_h = h_refine
            h_refine = best_h.copy()
        
        # Targeted descent
        if time.time() < deadline - 3:
            h_refine, c_r = targeted_lag_descent(best_h, n_best, deadline - 3, top_k=3)
            if c_r < best_c:
                best_c = c_r
                best_h = h_refine
        
        # Final SLSQP
        if time.time() < deadline - 2:
            ht = slsqp_run(best_h, n_best, 5000000, maxiter=6000, ftol=1e-17)
            c = compute_c5(ht, n_best)
            if c < best_c:
                best_c = c
                best_h = ht
        
        # Very fine pair search
        if n_best <= 220 and time.time() < deadline - 1:
            best_h, best_c = pair_search_full(best_h, n_best, deadline - 1,
                                               [0.0005, 0.0002, 0.0001])
        elif time.time() < deadline - 1:
            best_h, best_c = pair_search_radius(best_h, n_best, deadline - 1,
                                                 [0.0005, 0.0002], radius=25)
    
    # Final: ensure feasibility and compute exact C5
    if best_h is None:
        best_h = project_feasible(np.full(200, 0.5), 200)
        best_c = compute_c5(best_h, 200)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
