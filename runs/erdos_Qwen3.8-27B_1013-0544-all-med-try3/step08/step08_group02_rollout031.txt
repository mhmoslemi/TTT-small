            x = np.concatenate((x, np.zeros(n - x.size, dtype=float)))
        else:
            x = x[:n]

    target = float(n) / 2.0

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)

    for _ in range(5):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            room = 1.0 - h
            idx = np.flatnonzero(room > 0.0)
            if idx.size == 0:
                break
            tot = float(room[idx].sum())
            if tot <= 0.0:
                break
            inc = (diff / tot) * room[idx]
            inc = np.minimum(inc, room[idx])
            h[idx] += inc
        else:
            idx = np.flatnonzero(h > 0.0)
            if idx.size == 0:
                break
            tot = float(h[idx].sum())
            if tot <= 0.0:
                break
            dec = (-diff / tot) * h[idx]
            dec = np.minimum(dec, h[idx])
            h[idx] -= dec

    h = np.clip(h, 0.0, 1.0)
    rem = float(target - h.sum())
    if abs(rem) > 1e-10:
        if rem > 0.0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        for i in order:
            if rem > 0.0:
                step = min(1.0 - h[i], rem)
            else:
                step = min(h[i], -rem)
            if step > 1e-15:
                if rem > 0.0:
                    h[i] += step
                    rem -= step
                else:
                    h[i] -= step
                    rem += step
            if abs(rem) < 1e-12:
                break
    return h


def corr_vec(h, n):
    q = 1.0 - h
    need = 2 * n - 1
    N = 1 << (need - 1).bit_length()
    fH = np.fft.rfft(h, N)
    fQ = np.fft.rfft(q, N)
    raw = np.fft.irfft(fH * np.conj(fQ), N)
    dx = 2.0 / float(n)
    return np.concatenate((raw[N - n + 1:N], raw[0:n])) * dx


def compute_c(h, n):
    return float(corr_vec(h, n).max())


def compute_c_direct(h, n):
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / float(n)))


def select_active(c, m, top_k):
    top_k = max(1, int(top_k))
    tol = max(1e-8, 0.01 * float(m))
    idx = np.flatnonzero(c >= (m - tol))
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))])
    if idx.size > top_k:
        idx = idx[np.argsort(-c[idx])[:top_k]]
    return idx


def active_gradient(h, n, active, weights):
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    for idx, w in zip(active, weights):
        k = int(idx) - (n - 1)
        wm = float(w) * dx
        if wm == 0.0:
            continue
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            q = -k
            nv = n - q
            grad[q:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[q:n])
    return grad


def act_obj(x, n, active, alpha, mu):
    c = corr_vec(x, n)
    vals = c[active]
    m = float(vals.max())
    sh = alpha * (vals - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    s = float(e.sum())
    if s <= 0.0:
        val = m
    else:
        val = m + (mx + np.log(s)) / alpha
    if mu > 0.0:
        diff = float(x.sum()) - 0.5 * float(n)
        val += mu * diff * diff
    return float(val)


def act_grad(x, n, active, alpha, mu):
    c = corr_vec(x, n)
    vals = c[active]
    m = float(vals.max())
    sh = alpha * (vals - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    s = float(e.sum())
    if s > 0.0:
        w = e / s
    else:
        w = np.zeros(vals.shape, dtype=float)
        w[int(np.argmax(vals))] = 1.0
    g = active_gradient(x, n, active, w)
    if mu > 0.0:
        diff = float(x.sum()) - 0.5 * float(n)
        g += (2.0 * mu * diff) * np.ones(n, dtype=float)
    return g


def run_lbfgs_active(h, n, end_time, top_k, alphas, mu=1e-4, gtol=1e-7):
    work_h = project_feasible(h, n)
    best_h = work_h.copy()
    best_c = compute_c(best_h, n)
    work_c = best_c
    bounds = [(0.0, 1.0)] * n

    if n <= 600:
        cap = 600
        factor = 25
    elif n <= 1200:
        cap = 420
        factor = 13
    elif n <= 2500:
        cap = 240
        factor = 7
    else:
        cap = 150
        factor = 4

    for ai, alpha in enumerate(alphas):
        now = time.time()
        if now > end_time - 15.0:
            break
        cvec = corr_vec(work_h, n)
        maxv = float(cvec.max())
        active = select_active(cvec, maxv, top_k)

        nrem = max(1, len(alphas) - ai)
        t_alloc = max(1.0, (end_time - now) / float(nrem))
        maxit = int(t_alloc * factor)
        maxit = max(40, min(cap, maxit))
        if maxit < 10:
            break

        try:
            res = minimize(
                act_obj,
                work_h,
                jac=act_grad,
                args=(n, active, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxit,
                    "ftol": 1e-14,
                    "gtol": gtol,
                    "maxcor": 30,
                    "maxls": 30,
                },
            )
            cand = project_feasible(res.x, n)
            cand_c = compute_c(cand, n)
        except Exception:
            cand = project_feasible(work_h, n)
            cand_c = compute_c(cand, n)

        if not np.isfinite(cand_c):
            continue

        if cand_c < best_c - 1e-14:
            best_c = cand_c
            best_h = cand

        if cand_c <= work_c + 5e-4:
            work_h = cand
            work_c = cand_c
        else:
            work_h = project_feasible(best_h, n)
            work_c = best_c

    return best_h, best_c


def upsample(h, target_n):
    old_n = len(h)
    if target_n == old_n:
        return project_feasible(h, target_n)
    x_old = (np.arange(old_n) + 0.5) * (2.0 / float(old_n))
    x_new = (np.arange(target_n) + 0.5) * (2.0 / float(target_n))
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return project_feasible(h_new, target_n)


def even_linear_start(n, m):
    m = float(m)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = m + (1.0 - 2.0 * m) * np.abs(x - 1.0)
    return project_feasible(h, n)


def random_even_start(n, seed, M=8, amp=0.5):
    rng = np.random.RandomState(int(seed))
    t = (np.arange(n) + 0.5) / float(n) - 1.0
    h = 0.5 * np.ones(n, dtype=float)
    for m in range(1, int(M) + 1):
        a = amp / (float(m) ** 0.8)
        h += rng.uniform(-a, a) * np.cos(m * np.pi * t)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def subgrad_refine(h, n, end_time, max_iter=300, eta0=0.05):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    eta = float(eta0)

    for _ in range(int(max_iter)):
        if time.time() > end_time - 5.0:
            break
        cvec = corr_vec(best_h, n)
        tmax = float(cvec.max())
        active = select_active(cvec, tmax, 16)
        vals = cvec[active]
        m = float(vals.max())
        sh = 50000.0 * (vals - m)
        mx = float(sh.max())
        e = np.exp(sh - mx)
        s = float(e.sum())
        if s > 0.0:
            w = e / s
        else:
            w = np.zeros(vals.shape, dtype=float)
            w[int(np.argmax(vals))] = 1.0

        g = active_gradient(best_h, n, active, w)
        d = -g
        d -= d.mean()
        norm = float(np.linalg.norm(d))
        if norm < 1e-12:
            break
        d /= norm

        improved = False
        for f in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1):
            h2 = project_feasible(best_h + eta * f * d, n)
            c2 = compute_c(h2, n)
            if c2 < best_c - 1e-14:
                best_h = h2
                best_c = c2
                eta = min(eta * 1.05, 0.25)
                improved = True
                break

        if not improved:
            eta *= 0.85
            if eta < 1e-5:
                break

    return best_h, best_c


def mix_line_search(h1, h2, n, end_time, num=60):
    h1 = np.clip(h1, 0.0, 1.0)
    h2 = np.clip(h2, 0.0, 1.0)
    best_h = h1.copy()
    best_c = compute_c(h1, n)
    c2 = compute_c(h2, n)
    if c2 < best_c:
        best_h = h2.copy()
        best_c = c2

    num = max(2, int(num))
    for i in range(1, num + 1):
        if time.time() > end_time:
            break
        a = float(i) / float(num)
        h = a * h1 + (1.0 - a) * h2
        h = np.clip(h, 0.0, 1.0)
        c = compute_c(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if budget_s is None:
        budget_s = 1100.0
    budget_s = float(budget_s)
    deadline = t0 + max(60.0, budget_s - 30.0)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_parent = None

    if h_parent is not None and h_parent.size > 0:
        h_parent = np.ravel(h_parent)
        best_n = int(h_parent.size)
        best_h = project_feasible(h_parent, best_n)
        best_c = compute_c(best_h, best_n)
    else:
        best_n = 512
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c(best_h, best_n)

    if time.time() < deadline - 620.0:
        hs = project_feasible(0.5 * (best_h + best_h[::-1]), best_n)
        cs = compute_c(hs, best_n)
        if cs < best_c:
            best_c = cs
            best_h = hs

        for mval in (0.0, 0.04, 0.08, 0.12, 0.16, 0.24, 0.32):
            hr = even_linear_start(best_n, mval)
            cr = compute_c(hr, best_n)
            if cr < best_c:
                best_c = cr
                best_h = hr

        for r in range(6):
            hr = random_even_start(best_n, int(seed) + 101 * (r + 1), M=10, amp=0.35)
            cr = compute_c(hr, best_n)
            if cr < best_c:
                best_c = cr
                best_h = hr

    stage1_len = min(300.0, max(20.0, budget_s - 800.0))
    stage1_end = time.time() + stage1_len
    stage1_end = min(stage1_end, deadline - 450.0)
    if stage1_end <= time.time() + 10.0:
        stage1_end = min(time.time() + 50.0, deadline - 50.0)

    if stage1_end > time.time() + 20.0:
        top_k1 = 32
        alphas1 = [10, 50, 200, 1000, 5000, 25000, 100000, 500000]
        h1, c1 = run_lbfgs_active(best_h, best_n, stage1_end, top_k1, alphas1,
                                  mu=1e-4, gtol=1e-7)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    targets = [
        (1000, 280, 320),
        (2048, 240, 200),
        (4096, 160, 110),
    ]

    for target_n, stg_len, reserve in targets:
        if target_n <= best_n:
            continue
        if time.time() >= deadline - float(reserve):
            continue

        hup = upsample(best_h, target_n)
        cup = compute_c(hup, target_n)

        if cup < best_c:
            best_h = hup
            best_c = cup
            best_n = target_n
        elif cup > best_c + 0.01:
            continue

        stage_end = time.time() + float(stg_len)
        stage_end = min(stage_end, deadline - max(30.0, 0.5 * float(reserve)))
        if stage_end > time.time() + 20.0:
            if target_n <= 1000:
                top_k = 28
            elif target_n <= 2048:
                top_k = 20
            else:
                top_k = 14
            alphas = [20, 100, 500, 3000, 15000, 80000]
            hx, cx = run_lbfgs_active(hup, target_n, stage_end, top_k, alphas,
                                      mu=1e-4, gtol=1e-7)
            if cx < best_c:
                best_h = hx
                best_c = cx
                best_n = target_n

    if time.time() < deadline - 80.0 and best_n <= 4096:
        hs = project_feasible(0.5 * (best_h + best_h[::-1]), best_n)
        end1 = min(time.time() + 40.0, deadline - 60.0)
        best_h, best_c = mix_line_search(best_h, hs, best_n, end1, num=44)

    if time.time() < deadline - 60.0 and best_n <= 4096:
        hev = even_linear_start(best_n, 0.10)
        end2 = min(time.time() + 30.0, deadline - 45.0)
        best_h, best_c = mix_line_search(best_h, hev, best_n, end2, num=36)

    if time.time() < deadline - 35.0:
        end_sg = deadline - 15.0
        best_h, best_c = subgrad_refine(best_h, best_n, end_sg, max_iter=500, eta0=0.05)

    best_h = project_feasible(best_h, best_n)
    final_c = compute_c(best_h, best_n)

    if time.time() < deadline - 12.0:
        try:
            direct_c = compute_c_direct(best_h, best_n)
            if np.isfinite(direct_c):
                final_c = direct_c
        except Exception:
            pass

    return best_h, float(final_c), int(best_n)
