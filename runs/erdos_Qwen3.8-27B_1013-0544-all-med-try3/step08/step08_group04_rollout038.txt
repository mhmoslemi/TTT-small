    n = len(h)

        pk = p[k]

            grad[j1] += pk * dx * (1.0 - x[j1 - lag])

            grad[j2] -= pk * dx * x[j2 - L]

        if time.time() < deadline - 5:
            best_h, best_c = greedy_worst_lag(best_h, n_f, deadline - 3)

        # Very fine pair search
        if time.time() < deadline - 3:
            if n_f <= 200:
                best_h, best_c = pair_search_fast(best_h, n_f, deadline - 1,
                                                   [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = pair_search_fast(best_h, n_f, deadline - 1,
                                                   [0.001, 0.0005, 0.0002, 0.0001],
                                                   radius=60)

    # Phase 6: Final verification and projection
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
