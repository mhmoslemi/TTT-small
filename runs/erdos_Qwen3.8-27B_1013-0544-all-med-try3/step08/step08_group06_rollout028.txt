def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2} using iterative water-filling."""
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum: add to elements not at 1
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            # Need to decrease sum: subtract proportionally from elements > 0
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                hi_val = h[i]
                for j in range(i + 1, n):
                    # Try h[i]+=d, h[j]-=d
                    if hi_val + d <= 1.0 and h[j] - d >= 0.0:
                        new_c = c + nd * (effect[i] - effect[j])
                        nm = new_c.max()
                        if nm < best_c - 1e-15:
                            update_effect_for_move(effect, h, i, d, n)
                            update_effect_for_move(effect, h, j, -d, n)
                            c = new_c
                            best_c = nm
                            improved = True
                            moves_made += 1
                    # Try h[i]-=d, h[j]+=d
                    if hi_val - d >= 0.0 and h[j] + d <= 1.0:
                        new_c = c - nd * (effect[i] - effect[j])
                        nm = new_c.max()
                        if nm < best_c - 1e-15:
                            update_effect_for_move(effect, h, i, -d, n)
                            update_effect_for_move(effect, h, j, d, n)
                            c = new_c
                            best_c = nm
                            improved = True
                            moves_made += 1
            if not improved:
                break
    return h, best_c

def recompute_and_verify(h, n):
    """Recompute c from scratch to verify (effect updates can drift)."""
    c = compute_c5_array(h, n)
    c5 = c.max()
    h = project_feasible(h, n)
    c = compute_c5_array(h, n)
    c5 = c.max()
    return h, c5

def gradient_subgradient_step(h, n, alpha, step_size):
    """One projected subgradient step."""
    c = compute_c5_array(h, n)
    cur_max = c.max()
    sh = alpha * (c - cur_max)
    sm = sh.max()
    p = np.exp(sh - sm)
    p /= p.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - (n - 1)
        w = pk * dx
        j_lo = max(0, -lag)
        j_hi = min(n, n - lag)
        if j_lo < j_hi:
            g[j_lo:j_hi] += w * (1.0 - h[j_lo + lag:j_hi + lag])
        j_lo2 = max(0, lag)
        j_hi2 = min(n, n + lag)
        if j_lo2 < j_hi2:
            g[j_lo2:j_hi2] -= w * h[j_lo2 - lag:j_hi2 - lag]
    h_new = h - step_size * g
    h_new = project_feasible(h_new, n)
    return h_new

def subgradient_descent(h, n, deadline, alpha_init=200, max_iter=3000,
                        step_init=0.005, decay=0.999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = step_init
    alpha = alpha_init

    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_new = gradient_subgradient_step(h, n, alpha, step)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            h = h_new
            best_c = c_new
            best_h = h_new.copy()
        else:
            step *= 0.85
        if it % 200 == 199:
            alpha = min(100000, alpha * 1.2)
            step = min(step * 1.05, 0.02)
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp_opt(best_h, n, a, maxiter=1500, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    # Fast pair search
    if time.time() < deadline - 10:
        best_h, best_c = fast_pair_search(best_h, n, deadline - 10,
                                           deltas=[0.02, 0.01, 0.005, 0.002, 0.001])

    # Subgradient descent
    if time.time() < deadline - 10:
        hsg, csg = subgradient_descent(best_h, n, deadline - 5,
                                        alpha_init=500, step_init=0.003)
        if csg < best_c:
            best_c = csg
            best_h = hsg

    # Another SLSQP push
    if time.time() < deadline - 5:
        ht = slsqp_opt(best_h, n, 500000, maxiter=2000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    # Final fine pair search
    if time.time() < deadline - 2:
        best_h, best_c = fast_pair_search(best_h, n, deadline - 0.5,
                                           deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        # Phase 1: Optimize at native resolution (45% of budget)
        alloc1 = (deadline - t0) * 0.40
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

        # Phase 2: Try different resolutions (25% of budget)
        x_init = np.linspace(0, 2, n_init)
        n_tries = [100, 150, 250, 300]
        for n_try in n_tries:
            if n_try == n_init:
                continue
            if time.time() > deadline - 40:
                break
            remaining = deadline - time.time() - 30
            alloc = min(60, remaining * 0.20)
            if alloc < 15:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_at_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

        # Phase 3: Random restarts from perturbed best (20% of budget)
        if time.time() < deadline - 30:
            n = len(best_h)
            remaining = deadline - time.time() - 20
            n_restarts = max(2, int(remaining / 40))
            for r in range(n_restarts):
                if time.time() > deadline - 15:
                    break
                scale = 0.01 + 0.08 * rng.rand()
                hp = best_h + rng.normal(0, scale, n)
                hp = project_feasible(hp, n)
                alloc = min(35, (deadline - time.time() - 12) * 0.4)
                if alloc < 10:
                    break
                h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

        # Phase 4: Alternative initializations (10% of budget)
        if time.time() < deadline - 20:
            n = len(best_h)
            # Triplet structure: high-low-high
            a = n // 4
            h_trip = np.zeros(n)
            h_trip[:a] = 1.0
            h_trip[-a:] = 1.0
            # Adjust middle
            mid_start = a
            mid_end = n - a
            h_trip[mid_start:mid_end] = 0.0
            h_trip = project_feasible(h_trip, n)
            alloc = min(25, (deadline - time.time() - 10) * 0.5)
            if alloc > 10:
                h_opt, c_opt = optimize_at_n(h_trip, n, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

            # Smooth sinusoidal init
            x = np.linspace(0, 2, n)
            h_sin = 0.5 + 0.3 * np.sin(np.pi * x)
            h_sin = project_feasible(h_sin, n)
            if time.time() < deadline - 15:
                alloc = min(20, (deadline - time.time() - 8) * 0.5)
                if alloc > 8:
                    h_opt, c_opt = optimize_at_n(h_sin, n, time.time() + alloc, rng)
                    if c_opt < best_c:
                        best_c = c_opt
                        best_h = h_opt

    if best_h is None:
        # Fallback: uniform
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
        if time.time() < deadline - 5:
            best_h, best_c = fast_pair_search(best_h, n, deadline - 1,
                                               deltas=[0.01, 0.005, 0.002])

    # Final ultra-fine refinement
    if time.time() < deadline - 2:
        n = len(best_h)
        remaining_t = deadline - time.time() - 1
        if remaining_t > 1:
            best_h, best_c = fast_pair_search(best_h, n, deadline - 0.5,
                                               deltas=[0.0005, 0.0002, 0.0001])

    best_h = project_feasible(best_h, len(best_h))
    final_n = len(best_h)
    best_c = compute_c5(best_h, final_n)
    return best_h, best_c, final_n
