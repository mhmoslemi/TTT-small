```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {x ∈ [0,1]^n : sum(x) = n/2}."""
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    s = h.sum()
    if abs(s - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    # Bisection on lambda: find lambda s.t. sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h - mid, 0.0, 1.0)
        if h_test.sum() > target:
            lo = mid
        else:
            hi = mid
    lamb = (lo + hi) / 2.0
    h_proj = np.clip(h - lamb, 0.0, 1.0)
    # Fix any remaining tiny error
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        free = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_gradient(h, n, alpha):
    """Compute gradient of the log-sum-exp approximation to max(correlations)."""
    c = compute_correlations(h, n)
    c_max = c.max()
    scaled = alpha * (c - c_max)
    exp_c = np.exp(scaled)
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        # s1: sum_{k=0}^{n-1-j} p[n-1+k]*(1-h[j+k])
        s1 = np.dot(p[n-1:2*n-1-j], 1.0 - h[j:n])
        # s2: -sum_{k=0}^{j} p[n-1+k]*h[j-k]
        s2 = -np.dot(p[n-1:n+j+1], h[j::-1])
        # s3: sum_{l=1}^{j} p[n-1-l]*(1-h[j-l])
        if j > 0:
            s3 = np.dot(p[n-1-j:n-1][::-1], 1.0 - h[j-1::-1])
        else:
            s3 = 0.0
        # s4: -sum_{l=1}^{n-1-j} p[n-1-l]*h[j+l]
        if j < n - 1:
            s4 = -np.dot(p[j:n-1][::-1], h[j+1:n])
        else:
            s4 = 0.0
        grad[j] = alpha * (s1 + s2 + s3 + s4)
    
    obj_val = c_max + np.log(exp_c.sum()) / alpha
    return grad, obj_val

def projected_gradient_step(h, n, alpha, lr, momentum, h_prev):
    """One step of Nesterov projected gradient descent."""
    # Nesterov: evaluate gradient at y = h + momentum * (h - h_prev)
    y = h + momentum * (h - h_prev) if h_prev is not None else h
    y = np.clip(y, 0.0, 1.0)  # keep in bounds for gradient evaluation
    
    grad, obj_val = compute_gradient(y, n, alpha)
    
    # Take gradient step
    h_new = y - lr * grad
    h_new = project_feasible(h_new, n)
    
    return h_new, obj_val

def pgd_optimize(h_start, n, deadline, alphas=None, max_iter=5000):
    """Optimize using projected gradient with Nesterov momentum."""
    if alphas is None:
        alphas = [200, 500, 2000, 10000, 50000, 200000]
    
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_prev = h.copy()
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        momentum = 0.9
        lr = 1.0 / (alpha * 10)  # initial learning rate
        no_improve = 0
        
        for it in range(max_iter):
            if time.time() > deadline - 5:
                break
            
            h_new, obj_val = projected_gradient_step(h, n, alpha, lr, momentum, h_prev)
            c_new = compute_c5(h_new, n)
            
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptive step size
            if c_new > compute_c5(h, n) + 1e-8:
                lr *= 0.5
                # Undo the step
                h_new = h.copy()
            elif c_new < compute_c5(h, n) - 1e-10:
                lr = min(lr * 1.1, 1.0 / alpha)
            
            h_prev = h
            h = h_new
            
            if no_improve > 500:
                break
        
        # Update momentum schedule
        if alpha > 5000:
            momentum = 0.85
    
    return best_h, best_c

def pair_search_fine(h, n, deadline, deltas=None):
    """Fine pair search with small steps."""
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if j - i > 100:  # Limit range for efficiency
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    s = h.sum()
    if abs(s - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h - mid, 0.0, 1.0)
        if h_test.sum() > target:
            lo = mid
        else:
            hi = mid
    lamb = (lo + hi) / 2.0
    h_proj = np.clip(h - lamb, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        free = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_gradient(h, n, alpha):
    c = compute_correlations(h, n)
    c_max = c.max()
    scaled = alpha * (c - c_max)
    exp_c = np.exp(scaled)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for j in range(n):
        s1 = np.dot(p[n-1:2*n-1-j], 1.0 - h[j:n])
        s2 = -np.dot(p[n-1:n+j+1], h[j::-1])
        if j > 0:
            s3 = np.dot(p[n-1-j:n-1][::-1], 1.0 - h[j-1::-1])
        else:
            s3 = 0.0
        if j < n - 1:
            s4 = -np.dot(p[j:n-1][::-1], h[j+1:n])
        else:
            s4 = 0.0
        grad[j] = alpha * (s1 + s2 + s3 + s4)
    obj_val = c_max + np.log(exp_c.sum()) / alpha
    return grad, obj_val

def pgd_optimize(h_start, n, deadline, alphas=None, max_iter_per_alpha=3000):
    if alphas is None:
        alphas = [100, 300, 1000, 5000, 20000, 100000, 300000]
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_prev = h.copy()
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        momentum = 0.85
        lr = 5.0 / alpha
        no_improve = 0
        for it in range(max_iter_per_alpha):
            if time.time() > deadline - 10:
                break
            y = h + momentum * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
            grad, _ = compute_gradient(y, n, alpha)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            h_new = y - lr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            c_cur = compute_c5(h, n)
            if c_new < c_cur - 1e-14:
                h_prev = h
                h = h_new
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                lr = min(lr * 1.05, 10.0 / alpha)
                momentum = min(momentum * 1.01, 0.92)
            else:
                lr *= 0.4
                momentum *= 0.95
                no_improve += 1
            if no_improve > 800:
                break
        h_prev = h.copy()
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter=1000):
    def obj_func(x):
        c = compute_correlations(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sum_constraint(x):
        return x.sum() - n / 2.0
    def sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}]
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    return project_feasible(h_dst, n_dst)
```
