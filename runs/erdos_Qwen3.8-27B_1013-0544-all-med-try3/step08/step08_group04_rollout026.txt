    grad = np.zeros(n)
    if k == 0:

    elif k > 0:

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, k):
    """Gradient of correlate(h, 1-h) at lag k."""
    grad = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    elif k > 0:
        if n - 1 - k >= 0:
            grad[0:n-k] += -h[k:n]

    else:

            grad[0:n-k_neg] += (1.0 - h[k_neg:n])
            grad[k_neg:n] += -h[0:n-k_neg]
    return grad

def projected_gradient_step(h, n, lr, top_k=3, temp=1.0):
    """One step of projected gradient descent on C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    vals = corr * dx
    max_val = vals.max()
    
    # Get top-k lags with softmax weighting
    k = min(top_k, len(vals))
    top_idx = np.argsort(vals)[-k:][::-1]
    
    if k == 1:
        grad = grad_lag(h, n, top_idx[0] - (n-1)) * dx
    else:
        # Softmax weights
        w = np.exp((vals[top_idx] - max_val) / temp)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, pi in zip(w, top_idx):
            k_actual = pi - (n - 1)
            grad += wi * grad_lag(h, n, k_actual)
        grad *= dx
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, max_val

                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def sum_cons(x):
        return float(x.sum() - n / 2.0)
    
    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_cons, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def make_binary_init(n, rng):
    idx = rng.choice(n, size=n//2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_sawtooth_init(n, period, rng):
    h = np.zeros(n)
    for p in range(period, n, period*2):
        end = min(p + period//2, n)
        h[p:end] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global rng
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    
    # Phase 1: Gradient descent on initial resolution
    alloc1 = min(300, (deadline - t0) * 0.40)
    if alloc1 > 30:
        h1, c1 = pd_optimize(h_init, n_init, t0 + alloc1, top_k=3,
                            temp_base=0.001, lr_init=0.004, decay=0.9998)
        if c1 < best_c:
            best_c = c1
            best_h = h1
        # Local search
        if time.time() < deadline - 30:
            h1, c1 = grad_guided_local_search(h1, n_init, time.time() + min(120, (deadline - time.time()) * 0.3))
            if c1 < best_c:
                best_c = c1
                best_h = h1
    
    # Phase 2: SLSQP polish
    if time.time() < deadline - 20:
        h2 = slsqp_polish(best_h, len(best_h), 100000, maxiter=1500)
        c2 = compute_c5(h2, len(best_h))
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Try other resolutions
    n_candidates = [150, 200, n_init, 300, 400]
    n_candidates = [x for x in n_candidates if x != n_init and x < 500]
    
    for n_try in sorted(set(n_candidates)):
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc = min(150, (deadline - time.time()) * 0.25)
        if alloc < 30:
            break
        
        h_opt, c_opt = pd_optimize(h_interp, n_try, time.time() + alloc,
                                   top_k=3, temp_base=0.001, lr_init=0.004, decay=0.9998)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        elif c_opt < best_c + 0.001:
            # Close, try local search
            if time.time() < deadline - 30:
                h_opt, c_opt = grad_guided_local_search(h_opt, n_try, time.time() + 60)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
    
    # Phase 4: Multi-restart with different initializations
    n_best = len(best_h)
    n_restarts = 3
    for r in range(n_restarts):
        if time.time() > deadline - 40:
            break
        # Different perturbation scales
        scale = 0.02 + 0.03 * r
        hp = best_h + rng.normal(0, scale, n_best)
        hp = project_feasible(hp, n_best)
        
        alloc = min(80, (deadline - time.time() - 20) * 0.4)
        if alloc < 20:
            break
        h_opt, c_opt = pd_optimize(hp, n_best, time.time() + alloc,
                                   top_k=3, temp_base=0.002, lr_init=0.003, decay=0.9997)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 5: Binary-aware restart
    if time.time() < deadline - 40:
        h_bin = make_binary_init(n_best, rng)
        alloc = min(60, (deadline - time.time() - 15) * 0.5)
        if alloc > 15:
            h_opt, c_opt = pd_optimize(h_bin, n_best, time.time() + alloc,
                                       top_k=3, temp_base=0.002, lr_init=0.005, decay=0.9996)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 6: Fine local search on best
    if time.time() < deadline - 10:
        n_b = len(best_h)
        if n_b <= 250:
            best_h, best_c = near_neighbor_search(best_h, n_b, deadline - 3,
                                                   radius=30, deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = near_neighbor_search(best_h, n_b, deadline - 3,
                                                   radius=20, deltas=[0.005, 0.002, 0.001, 0.0005])
    
    # Phase 7: Gradient-guided fine search
    if time.time() < deadline - 5:
        n_b = len(best_h)
        best_h, best_c = grad_guided_local_search(best_h, n_b, deadline - 2,
                                                   max_rounds=300)
    
    # Phase 8: Final SLSQP + tiny local moves
    if time.time() < deadline - 3:
        n_b = len(best_h)
        h_final = slsqp_polish(best_h, n_b, 500000, maxiter=3000)
        c_final = compute_c5(h_final, n_b)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        # Tiny local moves
        if time.time() < deadline - 1:
            best_h, best_c = near_neighbor_search(best_h, n_b, deadline - 0.5,
                                                   radius=10, deltas=[0.001, 0.0005, 0.0002])
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
