def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s < target:
            # Need to increase sum
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-14:
                break

            if hs.sum() < 1e-14:

    """Projected subgradient method for minimizing max_k correlation"""
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = step_init

            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        c_val = np.max(corr) * dx

            best_c = c_val
            best_h = h.copy()

        max_c = np.max(corr)

            h = h_new
