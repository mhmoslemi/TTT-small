    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, shift_idx):
    """Gradient of c[shift_idx] with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_feasible_exact(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} via bisection on lambda."""
    target = n / 2.0
    h = np.clip(h.copy(), -10.0, 10.0)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        u = np.clip(h - mid, 0.0, 1.0)
        s = u.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    u = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Tiny correction for floating point
    diff = target - u.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - u
            tot = room.sum()
            if tot > 1e-15:
                u += (diff / tot) * room
        else:
            tot = u.sum()
            if tot > 1e-15:
                u += (diff / tot) * u
    return np.clip(u, 0.0, 1.0)

def shift_gradient(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k == 0:
        grad = (1.0 - 2.0 * h) * dx
    elif k > 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] -= h[0:nv] * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] -= h[kk:n] * dx
    return grad

def smooth_max_obj(x, n, alpha, mu):

    sh = alpha * (c - m)
    mx = sh.max()

    else:

    return float(val)

def smooth_max_grad(x, n, alpha, mu):

def subgradient_polish(h, n, deadline, init_step=0.01, decay=0.9995, max_iters=50000):
    """Exact subgradient method on active set with Armijo line search."""
    h = project_feasible_exact(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = init_step
    stagnant = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        C = c_vec.max()
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 5000:
                step *= 0.5
                stagnant = 0
                if step < 1e-10:
                    break
        
        # Active set: shifts within 0.1% of max
        threshold = C - max(1e-9, 0.001 * max(C, 1e-6))
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            break
        
        # Subgradient: average of active shift gradients
        g = np.zeros(n)
        for idx in active:
            g += shift_gradient(h, n, idx)
        g /= len(active)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Armijo line search
        accepted = False
        s = step
        for _ in range(40):
            h_trial = project_feasible_exact(h - s * g, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < C - 1e-13 * s * gn:
                h = h_trial
                accepted = True
                break
            s *= 0.5
        if not accepted:
            step *= 0.5
            if step < 1e-12:
                break
        else:
            step = min(step * 1.1, init_step)
        step = max(step * decay, 1e-12)
    
    return best_h, best_c

def block_coordinate_descent(h, n, deadline, block_size=10, max_rounds=500):
    """Optimize consecutive blocks of variables."""
    h = project_feasible_exact(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds_block = [(0.0, 1.0)] * block_size
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for start in range(0, n - block_size + 1, block_size // 2):
            if time.time() > deadline:
                return best_h, best_c
            idx = np.arange(start, start + block_size)
            h_block = h[idx].copy()
            
            # Objective: c5 value with only this block varying
            # Use smooth max as proxy
            def blk_obj(xb):
                h_tmp = h.copy()
                h_tmp[idx] = xb
                # Adjust sum constraint approximately
                return compute_c5(h_tmp, n)
            
            def blk_grad(xb):
                h_tmp = h.copy()
                h_tmp[idx] = xb
                c_vec = compute_c5_vec(h_tmp, n)
                C = c_vec.max()
                threshold = C - max(1e-9, 0.002 * max(C, 1e-6))
                active = np.where(c_vec >= threshold)[0]
                g = np.zeros(n)
                for aidx in active:
                    g += shift_gradient(h_tmp, n, aidx)
                g /= len(active)
                return g[idx]
            
            try:
                res = minimize(blk_obj, h_block, jac=blk_grad,
                             method='L-BFGS-B', bounds=bounds_block,
                             options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-10})
                h_candidate = h.copy()
                h_candidate[idx] = np.clip(res.x, 0, 1)
                h_candidate = project_feasible_exact(h_candidate, n)
                c_candidate = compute_c5(h_candidate, n)
                if c_candidate < best_c - 1e-14:
                    h = h_candidate
                    best_c = c_candidate
                    best_h = h.copy()
                    improved = True
            except Exception:
                pass
        if not improved:
            # Try smaller blocks
            if block_size > 4:
                block_size = max(4, block_size // 2)
    
    return best_h, best_c

def two_point_fine_search(h, n, deadline, deltas=None):

                    if jj == ii:
                        continue
                    j = top_idx[jj]
                    # Try move: h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try move: h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def multi_resolution_refine(h, n, deadline):
    """Try at different resolutions."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for n_try in [n * 2, n // 2]:
        if n_try < 100 or n_try > 2000:
            continue
        if time.time() > deadline - 30:
            break
        
        x_orig = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_new = np.interp(x_new, x_orig, h)
        h_new = project_feasible_exact(h_new, n_try)
        c_new = compute_c5(h_new, n_try)
        
        # Quick L-BFGS at new resolution
        t_left = deadline - time.time() - 10
        if t_left > 30:
            h_new, c_new = run_lbfgs(h_new, n_try, time.time() + min(60, t_left),
                                       alpha=200000, mu=100.0, maxiter=3000)
        
        # Try subgradient polish
        if time.time() < deadline - 20:
            h_new, c_new = subgradient_polish(h_new, n_try, time.time() + min(30, deadline - time.time() - 5),
                                              init_step=0.005, max_iters=5000)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def diverse_restarts(h_best, c_best, n, deadline, seed):
    """Try a few diverse starting points."""
    rng = np.random.RandomState(seed)
    
    # Restart 1: Perturb the best
    if time.time() < deadline - 60:
        h1 = h_best + rng.randn(n) * 0.01
        h1 = project_feasible_exact(h1, n)
        c1 = compute_c5(h1, n)
        # Quick optimization
        t1 = min(40, deadline - time.time() - 50)
        if t1 > 15:
            h1, c1 = run_lbfgs(h1, n, time.time() + t1, alpha=50000, mu=50.0, maxiter=3000)
            h1, c1 = subgradient_polish(h1, n, time.time() + 20, init_step=0.005, max_iters=3000)
        if c1 < c_best:
            return h1, c1, n
    
    # Restart 2: Bimodal initialization
    if time.time() < deadline - 40:
        h2 = np.zeros(n)
        blocks = n // 4
        h2[0:blocks] = 1.0
        h2[n//2:n//2 + blocks] = 1.0
        h2 = project_feasible_exact(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < c_best + 0.05:  # Only optimize if reasonably close
            t2 = min(30, deadline - time.time() - 30)
            if t2 > 10:
                h2, c2 = run_lbfgs(h2, n, time.time() + t2, alpha=50000, mu=50.0, maxiter=2000)
                h2, c2 = subgradient_polish(h2, n, time.time() + 15, init_step=0.005, max_iters=2000)
        if c2 < c_best:
            return h2, c2, n
    
    return h_best, c_best, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    # Initialize
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = np.full(n, 0.5)
    
    h = project_feasible_exact(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS with increasing alpha (focus on getting close to true max)
    configs = [
        (50000, 100.0, 4000),
        (200000, 500.0, 3000),
        (1000000, 2000.0, 2000),
    ]
    for alpha, mu, mit in configs:
        if time.time() > deadline - 500:
            break
        t_phase = min(time.time() + 100, deadline - 500)
        h_c, c_c = run_lbfgs(h, n, t_phase, alpha, mu, maxiter=mit)
        if c_c < best_c:
            best_c = c_c
            best_h = h_c
        h = h_c
    
    # Phase 2: Subgradient polish on active set
    if time.time() < deadline - 350:
        t2 = min(time.time() + 200, deadline - 350)
        h_s, c_s = subgradient_polish(h, n, t2, init_step=0.01, decay=0.9998, max_iters=80000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
        h = h_s
    
    # Phase 3: Another L-BFGS at very high alpha
    if time.time() < deadline - 250:
        t3 = min(time.time() + 100, deadline - 250)
        h_l, c_l = run_lbfgs(h, n, t3, 5000000, 5000.0, maxiter=2000)
        if c_l < best_c:
            best_c = c_l
            best_h = h_l
        h = h_l
    
    # Phase 4: Subgradient polish again (finer)
    if time.time() < deadline - 180:
        t4 = min(time.time() + 100, deadline - 180)
        h_s2, c_s2 = subgradient_polish(h, n, t4, init_step=0.005, decay=0.9997, max_iters=60000)
        if c_s2 < best_c:
            best_c = c_s2
            best_h = h_s2
        h = h_s2
    
    # Phase 5: Two-point fine search
    if time.time() < deadline - 100:
        t5 = min(time.time() + 80, deadline - 100)
        h_f, c_f = two_point_fine_search(h, n, t5,
                                         deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
        h = h_f
    
    # Phase 6: Final subgradient with tiny steps
    if time.time() < deadline - 50:
        t6 = min(time.time() + 40, deadline - 50)
        h_p, c_p = subgradient_polish(h, n, t6, init_step=0.002, decay=0.999, max_iters=30000)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    
    # Phase 7: Ultra-fine 2-point search
    if time.time() < deadline - 20:
        t7 = min(time.time() + 15, deadline - 20)
        h_u, c_u = two_point_fine_search(h, n, t7,
                                         deltas=[0.00001, 0.000005, 0.000002, 0.000001, 0.0000005])
        if c_u < best_c:
            best_c = c_u
            best_h = h_u
    
    # Final projection and evaluation
    best_h = project_feasible_exact(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
