Let me also think about the subgradient polishing idea:

After all other methods converge, we can do:
1. Compute c_vec = compute_c5_vec(h, n)
2. Find the argmax index m*
3. Compute grad = shift_grad_single(h, n, m*)
4. Take a step: h_new = h - eta * grad
5. Project: h_new = project_feasible(h_new, n)
6. If it improved, keep it and reduce eta
7. Repeat

This is like a projected subgradient step targeting the max overlap.

For the choice of eta: a good heuristic is to use eta = (current_max - target) / ||grad||² where target is some desired value. Or just use a fixed small eta and decrease it.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    grads = []
    c_vals = []
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        grads.append(g)
        c_vals.append(c_vec[idx])
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        A_ub[row, 0:n] = grads[j]
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=300, rho_init=0.05, max_active=30):

    best_c = compute_c5(h, n)
    rho = rho_init
    consecutive_no_improve = 0
    for it in range(max_iter):

            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()

        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > max_active:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:max_active]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            consecutive_no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.2, 0.08)
        else:
            consecutive_no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if consecutive_no_improve > 30:

                if rho < 1e-10:

    return best_h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:

            h_new = project_feasible(res.x, n)

def coordinate_descent(h, n, deadline, max_sweeps=200, grid_size=31):
    """Coordinate descent with incremental c_vec update."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = np.correlate(h, 1 - h, mode='full')
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                return best_h, best_c
            j = int(j)
            hj = h[j]
            if hj < 1e-12 or hj > 1.0 - 1e-12:
                # At boundary, only try moving inward
                if hj < 1e-12:
                    t_vals = np.array([0.0, 0.01, 0.02, 0.05, 0.1])
                else:
                    t_vals = np.array([1.0, 0.99, 0.98, 0.95, 0.9])
            else:
                # Compute coefficient a[m] for h[j]
                a = np.zeros(2 * n - 1)
                # k ranges from -(n-1) to (n-1)
                # m = k + (n-1)
                # Coefficient: (1-h[j+k]) if valid - h[j-k] if valid
                k_start = max(-(n-1), -j)  # j+k >= 0 → k >= -j
                k_end = min(n-1, n-1-j)    # j+k < n → k < n-j
                for k in range(k_start, k_end + 1):
                    m = k + (n - 1)
                    a[m] += (1.0 - h[j + k])
                
                k_start2 = max(-(n-1), j - (n-1))  # j-k >= 0 → k <= j
                k_end2 = min(n-1, j)                # j-k < n → k > j-n
                for k in range(k_start2, k_end2 + 1):
                    m = k + (n - 1)
                    a[m] += (-h[j - k])
                
                # f(t) = max_m (c_vec[m] + a[m] * (t - hj))
                # Grid search for optimal t
                t_vals = np.linspace(0.0, 1.0, grid_size)
                deltas = t_vals - hj
                vals = c_vec[None, :] + a[:, None] * deltas[None, :]
                f_vals = vals.max(axis=0)
                best_idx = np.argmin(f_vals)
                new_hj = t_vals[best_idx]
                
                if abs(new_hj - hj) > 1e-12:
                    # Update c_vec incrementally
                    c_vec += a * (new_hj - hj)
                    h[j] = new_hj
                    improved = True
        
        # Project and check
        h_proj = project_feasible(h, n)
        if not np.allclose(h_proj, h, atol=1e-10):
            c_vec = np.correlate(h_proj, 1 - h_proj, mode='full')
            h = h_proj
        c_new = float(np.max(c_vec) * 2.0 / n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c

def subgradient_polish(h, n, deadline, max_iter=5000):
    """Projected subgradient method targeting the max overlap."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    stuck_count = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        if m < best_c - 1e-15:
            best_c = m
            best_h = h.copy()
        
        # Find all indices achieving (near) the max
        thresh = m - 1e-6 * m
        active = np.where(c_vec >= thresh)[0]
        
        # Average gradient over active set
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad_single(h, n, idx)
        grad /= len(active)
        
        # Step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < m - 1e-15:
            h = h_new
            stuck_count = 0
            eta = min(eta * 1.1, 0.05)
        else:
            stuck_count += 1
            eta *= 0.9
            if stuck_count > 50:
                eta = max(eta, 0.001)
                stuck_count = 0
            if eta < 1e-8:
                break
    
    return best_h, best_c

def perturbation_restart(h, n, deadline, num_restarts=3, perturbation_scale=0.05):
    """Try small random perturbations and re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]

def coord_coeff(h, n, j):
    """Compute coefficient vector a[m] for h[j] in c[m], fully vectorized."""
    m_vals = np.arange(2 * n - 1)
    k_vals = m_vals - (n - 1)
    a = np.zeros(2 * n - 1)
    idx1 = j + k_vals
    valid1 = (idx1 >= 0) & (idx1 < n)
    a[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = j - k_vals
    valid2 = (idx2 >= 0) & (idx2 < n)
    a[valid2] += (-h[idx2[valid2]])
    return a

def coordinate_descent(h, n, deadline, max_sweeps=300, grid_size=51):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = np.correlate(h, 1 - h, mode='full')
    consec_no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweep_improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                return best_h, best_c
            j = int(j)
            hj = h[j]
            a = coord_coeff(h, n, j)
            if hj <= 1e-12:
                t_vals = np.array([0.0, 0.005, 0.01, 0.02, 0.05, 0.1])
            elif hj >= 1.0 - 1e-12:
                t_vals = np.array([1.0, 0.995, 0.99, 0.98, 0.95, 0.9])
            else:
                t_vals = np.linspace(0.0, 1.0, grid_size)
            deltas = t_vals - hj
            if np.all(np.abs(deltas) < 1e-14):
                continue
            vals = c_vec[None, :] + a[:, None] * deltas[None, :]
            f_vals = vals.max(axis=0)
            best_idx = np.argmin(f_vals)
            new_hj = t_vals[best_idx]
            if abs(new_hj - hj) > 1e-12:
                c_vec += a * (new_hj - hj)
                h[j] = new_hj
                sweep_improved = True
        if sweep_improved:
            h_proj = project_feasible(h, n)
            if not np.allclose(h_proj, h, atol=1e-12):
                c_vec = np.correlate(h_proj, 1 - h_proj, mode='full')
                h = h_proj
            c_new = float(np.max(c_vec) * 2.0 / n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                consec_no_improve = 0
            else:
                consec_no_improve += 1
        else:
            consec_no_improve += 2
        if consec_no_improve > 30:
            break
    return best_h, best_c

def subgradient_polish(h, n, deadline, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.02
    stuck = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        if m < best_c - 1e-15:
            best_c = m
            best_h = h.copy()
        thresh = m - max(1e-8, 0.001 * m)
        active = np.where(c_vec >= thresh)[0]
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad_single(h, n, idx)
        grad /= max(len(active), 1)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < m - 1e-15:
            h = h_new
            stuck = 0
            eta = min(eta * 1.05, 0.04)
        else:
            stuck += 1
            eta *= 0.95
            if stuck > 80:
                eta = max(eta, 0.005)
                stuck = 0
            if eta < 1e-9:
                break
    return best_h, best_c

        improved = False

            if time.time() > deadline:
                return best_h, best_c
            end = min(start + block_size, n)
            js = np.arange(start, end)
            old_vals = h[js].copy()
            
            # Compute a for each j in block
            a_matrix = np.column_stack([coord_coeff(h, n, j) for j in js])
            
            # Grid search over the block (multi-dimensional)
            # For small blocks, use a fine grid
            if block_size == 1:
                deltas_t = np.linspace(-old_vals[0], 1.0 - old_vals[0], grid_size)
                vals = c_vec[None, :] + a_matrix[0, :, None] * deltas_t[None, None, :]
                f_vals = vals.max(axis=(0, 1))
                best_t = old_vals[0] + deltas_t[np.argmin(f_vals)]
                new_vals = np.array([best_t])
            elif block_size == 2:
                t1_vals = np.linspace(0, 1, min(11, grid_size))
                t2_vals = np.linspace(0, 1, min(11, grid_size))
                best_f = np.inf
                new_vals = old_vals.copy()
                for t1 in t1_vals:
                    for t2 in t2_vals:
                        d = np.array([t1 - old_vals[0], t2 - old_vals[1]])
                        val = np.max(c_vec + a_matrix.T @ d)
                        if val < best_f:
                            best_f = val
                            new_vals = np.array([t1, t2])
            elif block_size == 3:
                t_vals = np.linspace(0, 1, 7)
                best_f = np.inf
                new_vals = old_vals.copy()
                for t1 in t_vals:
                    for t2 in t_vals:
                        for t3 in t_vals:
                            d = np.array([t1 - old_vals[0], t2 - old_vals[1], t3 - old_vals[2]])
                            val = np.max(c_vec + a_matrix.T @ d)
                            if val < best_f:
                                best_f = val
                                new_vals = np.array([t1, t2, t3])
            else:
                continue
            
            delta = new_vals - old_vals
            if np.max(np.abs(delta)) > 1e-12:
                c_vec += a_matrix @ delta
                h[js] = new_vals
                improved = True
        
        if improved:
            h_proj = project_feasible(h, n)
            if not np.allclose(h_proj, h, atol=1e-12):
                c_vec = np.correlate(h_proj, 1 - h_proj, mode='full')
                h = h_proj
            c_new = float(np.max(c_vec) * 2.0 / n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
        else:
            break
    return best_h, best_c

def perturbation_restart(h, n, deadline, num_restarts=5, perturbation_scale=0.04):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    remaining = deadline - time.time()
    t_per = remaining / (num_restarts + 0.5)
    
    for r in range(num_restarts):
        if time.time() > deadline - 20:
            break
        t_sub = min(time.time() + t_per, deadline - 15)
        if t_sub < time.time() + 15:
            break
        h_pert = h.copy()
        mask = np.random.random(n) < 0.25
        n_mask = mask.sum()
        if n_mask > 0:
            h_pert[mask] += np.random.normal(0, perturbation_scale, n_mask)
        h_pert = project_feasible(h_pert, n)
        
        h_opt, c_opt = run_lbfgs(h_pert, n, min(t_sub, deadline - 30),
            configs=[(3000, 2000.0), (8000, 10000.0), (20000, 50000.0)])
        if time.time() < deadline - 15:
            h_opt, c_opt = sqp_optimize(h_opt, n, min(time.time() + 20, deadline - 10),
                                         max_iter=80, rho_init=0.03, max_active=30)
        if time.time() < deadline - 5:
            h_opt, c_opt = coordinate_descent(h_opt, n, deadline - 3,
                                               max_sweeps=50, grid_size=51)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: L-BFGS warmup (70s)
    t1 = min(t0 + 70, deadline - 950)
    if t1 > time.time() + 5:
        best_h, best_c = run_lbfgs(best_h, n_orig, t1)
    
    # Phase 2: SQP optimization (110s)
    t2 = min(t0 + 180, deadline - 800)
    if t2 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=400, rho_init=0.06, max_active=40)
    
    # Phase 3: Coordinate descent (180s)
    t3 = min(t0 + 360, deadline - 600)
    if t3 > time.time() + 10:
        best_h, best_c = coordinate_descent(best_h, n_orig, t3, max_sweeps=400, grid_size=51)
    
    # Phase 4: SQP refinement (70s)
    t4 = min(t0 + 430, deadline - 500)
    if t4 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t4, max_iter=250, rho_init=0.03, max_active=35)
    
    # Phase 5: Subgradient polish (70s)
    t5 = min(t0 + 500, deadline - 420)
    if t5 > time.time() + 10:
        best_h, best_c = subgradient_polish(best_h, n_orig, t5, max_iter=20000)
    
    # Phase 6: Coordinate descent with finer grid (180s)
    t6 = min(t0 + 680, deadline - 250)
    if t6 > time.time() + 10:
        best_h, best_c = coordinate_descent(best_h, n_orig, t6, max_sweeps=300, grid_size=101)
    
    # Phase 7: SQP final (50s)
    t7 = min(t0 + 730, deadline - 200)
    if t7 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t7, max_iter=200, rho_init=0.02, max_active=30)
    
    # Phase 8: Perturbation restarts (250s)
    t8 = min(t0 + 980, deadline - 60)
    if t8 > time.time() + 30:
        best_h, best_c = perturbation_restart(best_h, n_orig, t8, num_restarts=4, perturbation_scale=0.03)
    
    # Phase 9: Final coordinate descent + subgradient (remaining)
    if time.time() < deadline - 10:
        best_h, best_c = coordinate_descent(best_h, n_orig, deadline - 5, max_sweeps=200, grid_size=101)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
```
