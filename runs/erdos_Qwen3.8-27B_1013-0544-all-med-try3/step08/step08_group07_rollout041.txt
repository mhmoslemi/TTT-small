def fft_correlate(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    out_len = a.size + b.size - 1
    if out_len <= 1:

    fb = np.fft.rfft(b, n=L)

import numpy as np
import time

try:

except Exception:
    def next_fast_len(x):
        y = 1
        while y < x:
            y <<= 1
        return y


OPT_ALPHA = 1.0
OPT_LAM = 0.0
OPT_TARGET = 0.0

CALLBACK_DEADLINE = 0.0
CALLBACK_LAST_X = None


def minimize_callback(xk):
    global CALLBACK_LAST_X
    if time.time() > CALLBACK_DEADLINE:
        CALLBACK_LAST_X = xk.copy()
        return False
    return None


def fix_sum_exact(y, n):
    target = 0.5 * float(n)
    diff = target - float(np.sum(y))
    if abs(diff) <= 1e-13 * max(1, n):
        return y

    if diff > 0.0:
        mask = y < (1.0 - 1e-15)
    else:
        mask = y > 1e-15

    if np.any(mask):
        y[mask] += diff / float(np.count_nonzero(mask))

    y = np.clip(y, 0.0, 1.0)
    diff2 = target - float(np.sum(y))
    if abs(diff2) > 1e-15:
        if diff2 > 0.0:
            idx = int(np.argmax(1.0 - y))
        else:
            idx = int(np.argmax(y))
        y[idx] += diff2

    return np.clip(y, 0.0, 1.0)


def project_feasible(h, n):
    y = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(np.sum(y))
    diff = target - s

    if diff > 0.0:
        room = 1.0 - y
        sr = float(np.sum(room))
        if sr > 0.0:
            y = y + (diff / sr) * room
    elif diff < 0.0:
        if s > 0.0:
            y = y * (target / s)

    y = np.clip(y, 0.0, 1.0)
    return fix_sum_exact(y, n)


def fft_correlate(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    out_len = int(a.size + b.size - 1)

    if out_len <= 0:
        return np.array([], dtype=float)
    if out_len == 1:
        return np.array([float(a[0] * b[0])], dtype=float)

    L = int(next_fast_len(out_len))
    fa = np.fft.rfft(a[::-1], n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    conv = conv[:out_len]
    return conv[::-1]


def fast_c_array(h):
    n = len(h)
    dx = 2.0 / float(n)
    g = fft_correlate(h, 1.0 - h)
    g = g * dx
    np.maximum(g, 0.0, out=g)
    return g


def fast_c5(h):
    return float(np.max(fast_c_array(h)))


def exact_c5(h):
    n = len(h)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(c)) * dx
    if not np.isfinite(val) or val < 0.0:
        return 0.0
    return val


def direct_gradient(indices, weights, h, n, dx):
    grad = np.zeros(n, dtype=float)
    offset = n - 1

    for idx, w in zip(indices, weights):
        k = int(idx) - offset
        w = float(w)

        if k >= 0:
            ln = n - k
            if ln > 0:
                grad[:ln] += w
                grad[:ln] -= w * h[k:k + ln]
                grad[k:] -= w * h[:ln]
        else:
            d = -k
            ln = n - d
            if ln > 0:
                grad[d:] += w
                grad[d:] -= w * h[:ln]
                grad[:ln] -= w * h[d:]

    grad *= dx
    grad -= grad.mean()
    return grad


def soft_objective_grad(h, alpha):
    n = len(h)
    dx = 2.0 / float(n)
    L = 2 * n - 1

    g = fast_c_array(h)
    m = float(np.max(g))

    if m <= 1e-15 and float(np.sum(g)) <= 1e-12:
        return 0.0, np.zeros(n, dtype=float)

    x = alpha * (g - m)
    w = np.exp(x)
    ws = float(np.sum(w))

    if ws <= 0.0 or not np.isfinite(ws):
        p = np.zeros(L, dtype=float)
        p[int(np.argmax(g))] = 1.0
        soft = m
    else:
        p = w / ws
        soft = m + np.log(ws) / float(alpha)

    nz = np.nonzero(p > 1e-10)[0]
    count = int(nz.size)

    if count == 0:
        nz = np.array([int(np.argmax(g))], dtype=int)
        p_act = np.array([1.0], dtype=float)
        mass = 1.0
    else:
        mass = float(np.sum(p[nz]))
        p_act = p[nz] / mass if mass > 0.0 else p[nz]

    use_direct = (count > 0 and count * n <= 500000 and mass >= 0.98)

    if use_direct:
        return soft, direct_gradient(nz, p_act, h, n, dx)

    cp = np.empty(L + 1, dtype=float)
    cp[0] = 0.0
    np.cumsum(p, out=cp[1:])

    starts = (n - 1) - np.arange(n, dtype=np.intp)
    V = cp[starts + n] - cp[starts]

    idx = (2 * n - 2) - np.arange(n, dtype=np.intp)

    corr1 = fft_correlate(p, h)
    Aplus = corr1[idx]

    p_rev = p[::-1]
    corr2 = fft_correlate(p_rev, h)
    Aminus = corr2[idx]

    grad = dx * (V - Aplus - Aminus)
    grad -= grad.mean()
    return soft, grad


def smooth_objective_penalty(x):
    val, grad = soft_objective_grad(x, OPT_ALPHA)
    s = float(np.sum(x))
    diff = s - OPT_TARGET
    val = val + OPT_LAM * diff * diff
    grad = grad + (2.0 * OPT_LAM * diff)
    return val, grad


def estimate_grad_cost(h, n):
    t0 = time.time()
    for _ in range(2):
        soft_objective_grad(h, 800.0)
    dt = time.time() - t0
    return max(1e-4, dt / 2.0)


def active_polish(h, n, deadline, alpha=3000.0, max_steps=2500):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_exact = exact_c5(h)
    fast_val = fast_c5(h)
    lr = 0.05

    for it in range(max_steps):
        if time.time() > deadline:
            break

        _, grad = soft_objective_grad(h, alpha)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-13:
            break

        d = -grad / gnorm
        improved = False

        for t in (lr, 2.0 * lr, 5.0 * lr, 0.5 * lr, 0.1 * lr):
            if time.time() > deadline:
                break
            ht = project_feasible(h + t * d, n)
            ct = fast_c5(ht)
            if ct < fast_val - 1e-13:
                h = ht
                fast_val = ct
                improved = True
                if it % 10 == 0 or ct < best_exact - 1e-11:
                    ev = exact_c5(h)
                    if np.isfinite(ev) and ev < best_exact:
                        best_exact = ev
                        best_h = h.copy()
                break

        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
        else:
            if lr < 0.15:
                lr *= 1.05

    ev = exact_c5(h)
    if np.isfinite(ev) and ev < best_exact:
        best_h = h.copy()
        best_exact = ev
    else:
        best_h = project_feasible(best_h, n)
        best_exact = exact_c5(best_h)

    return best_h, best_exact


def single_lag_gradient(idx, n, h):
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    k = int(idx) - (n - 1)

    if k >= 0:
        ln = n - k
        if ln > 0:
            grad[:ln] += 1.0
            grad[:ln] -= h[k:k + ln]
            grad[k:] -= h[:ln]
    else:
        d = -k
        ln = n - d
        if ln > 0:
            grad[d:] += 1.0
            grad[d:] -= h[:ln]
            grad[:ln] -= h[d:]

    return dx * grad


def linearize_lp_polish(h, n, deadline, max_rounds=5, max_active=60):
    if linprog is None or n <= 0:
        return h, exact_c5(h)

    best_h = project_feasible(h, n)
    best_c = exact_c5(best_h)

    for _ in range(max_rounds):
        if time.time() > deadline - 5.0:
            break

        cur = project_feasible(best_h, n)
        g = fast_c_array(cur)
        m = float(np.max(g))
        tol = max(1e-10, 1e-4 * m)

        act = np.where(g >= m - tol)[0]
        if act.size == 0:
            act = np.array([int(np.argmax(g))], dtype=int)
        elif act.size > max_active:
            order = np.argsort(g)
            act = order[-max_active:]

        a_rows = np.empty((act.size, n + 1), dtype=float)
        b_ub = np.empty(act.size, dtype=float)

        for r, idx in enumerate(act):
            grad_k = single_lag_gradient(idx, n, cur)
            a_rows[r, :n] = grad_k
            a_rows[r, n] = -1.0
            b_ub[r] = float(np.dot(grad_k, cur)) - float(g[idx])

        c = np.zeros(n + 1, dtype=float)
        c[n] = 1.0

        a_eq = np.zeros((1, n + 1), dtype=float)
        a_eq[0, :n] = 1.0
        b_eq = np.array([0.5 * float(n)], dtype=float)

        bounds = [(0.0, 1.0)] * n + [(None, None)]

        try:
            res = linprog(
                c,
                A_ub=a_rows,
                b_ub=b_ub,
                A_eq=a_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False):
            break

        xh = np.asarray(res.x, dtype=float)[:n]
        xh = project_feasible(xh, n)
        cx = exact_c5(xh)

        if np.isfinite(cx) and cx < best_c - 1e-13:
            best_c = cx
            best_h = xh.copy()

    best_h = project_feasible(best_h, n)
    best_c = exact_c5(best_h)
    return best_h, best_c

def interpolate_h(old, n_old, n_new):
    if n_old == n_new:
        return np.asarray(old, dtype=float).copy()

    if n_old == 1:
        return project_feasible(np.full(int(n_new), float(old[0])), int(n_new))

    x_old = (np.arange(int(n_old)) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(int(n_new)) + 0.5) * (2.0 / float(n_new))
    return project_feasible(np.interp(x_new, x_old, np.asarray(old, dtype=float)), int(n_new))


def candidate_if_good(pairs, h, n, limit):
    h = project_feasible(h, int(n))
    c = exact_c5(h)
    if np.isfinite(c) and c < limit:
        pairs.append((c, h))
    return c


def top_candidates(pairs, k):
    arr = list(pairs)
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        while j >= 0 and arr[j][0] > key[0]:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key

    if len(arr) > k:
        arr = arr[:k]
    return arr


def random_binary_candidate(n, rng):
    m = int(n) // 2
    h = np.zeros(int(n), dtype=float)
    idx = rng.choice(int(n), size=m, replace=False)
    h[idx] = 1.0
    return project_feasible(h, int(n))


def block_candidate(n, rng):
    n = int(n)
    parts = []
    cur = 1 if rng.rand() < 0.5 else 0
    high = int(max(8, int(np.sqrt(n)) + 15))

    while len(parts) < n:
        L = int(rng.randint(5, high))
        parts.extend([cur] * L)
        cur = 1 - cur

    h = np.asarray(parts[:n], dtype=float)
    m = n // 2
    s = int(np.sum(h))

    if s < m:
        zeros = np.flatnonzero(h == 0.0)
        need = m - s
        if zeros.size >= need:
            h[zeros[:need]] = 1.0
    elif s > m:
        ones = np.flatnonzero(h == 1.0)
        need = s - m
        if ones.size >= need:
            h[ones[:need]] = 0.0

    return project_feasible(h, n)


def fourier_candidate(n, rng, low_freq=True):
    n = int(n)
    pos = (np.arange(n) + 0.5) * (2.0 / float(n))
    y = np.full(n, 0.5, dtype=float)
    terms = int(rng.randint(3, 7))

    for _ in range(terms):
        if low_freq:
            f = int(rng.randint(1, 4))
        else:
            f = int(rng.randint(1, 8))

        amp = float(rng.uniform(0.05, 0.45))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        sign = 1.0 if rng.rand() < 0.5 else -1.0
        angle = np.pi * f * pos + phase

        if rng.rand() < 0.5:
            y += sign * amp * np.sin(angle)
        else:
            y += sign * amp * np.cos(angle)

    y = np.clip(y, 0.0, 1.0)
    return project_feasible(y, n)


def thue_morse_candidate(n):
    n = int(n)
    t = np.zeros(n, dtype=np.int8)

    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)

    h = t.astype(float)
    m = n // 2
    s = int(np.sum(h))

    if s < m:
        zeros = np.flatnonzero(h == 0.0)
        need = m - s
        if zeros.size >= need:
            h[zeros[:need]] = 1.0
    elif s > m:
        ones = np.flatnonzero(h == 1.0)
        need = s - m
        if ones.size >= need:
            h[ones[:need]] = 0.0

    return project_feasible(h, n)


def perturb_candidate(h, n, rng, scale=0.1):
    return project_feasible(np.asarray(h, dtype=float) + float(scale) * rng.randn(int(n)), int(n))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0

    if total < 1.0:
        total = 1.0

    deadline = t0 + total - 3.0
    if deadline <= t0:
        deadline = t0 + total - 0.5

    rng = np.random.RandomState(seed)

    init = kwargs.get("initial_h_values", None)
    if init is None:
        try:
            init = initial_h_values
        except Exception:
            init = None

    if init is not None:
        try:
            parent_raw = np.asarray(init, dtype=float).copy()
            if parent_raw.size == 0:
                raise ValueError
        except Exception:
            parent_raw = np.full(256, 0.5, dtype=float)
    else:
        parent_raw = np.full(256, 0.5, dtype=float)

    n_parent = int(parent_raw.size)
    if total >= 800:
        max_n = 2048
    elif total >= 450:
        max_n = 1024
    else:
        max_n = 512

    base_n = max(128, n_parent)
    if base_n % 2:
        base_n += 1
    if base_n > max_n:
        base_n = max_n
    if base_n % 2:
        base_n -= 1

    if base_n == n_parent:
        h0 = parent_raw.copy()
    else:
        h0 = interpolate_h(parent_raw, n_parent, base_n)

    h0 = project_feasible(h0, base_n)
    best_h = h0.copy()
    best_n = int(base_n)
    best_c = exact_c5(best_h)

    # Phase 1: quick optimization at the parent/prescribed resolution.
    phase1_end = t0 + min(0.12 * total, max(0.0, total - 5.0))
    if phase1_end > t0 + 8.0:
        hh, cc = optimize_at_n(best_h, best_n, phase1_end, do_polish=False)
        if cc < best_c:
            best_h = hh
            best_c = cc
            best_n = int(best_n)

    # Phase 2: diverse local starts at the base resolution.
    alt_end = t0 + min(0.24 * total, total - 30.0) if total >= 120 else -1.0
    if total >= 120 and alt_end > time.time() + 30.0:
        pairs = []
        limit = best_c + 0.03

        candidate_if_good(pairs, perturb_candidate(best_h, best_n, rng, 0.12), best_n, limit)
        candidate_if_good(pairs, perturb_candidate(best_h, best_n, rng, 0.30), best_n, limit)
        candidate_if_good(pairs, random_binary_candidate(best_n, rng), best_n, limit)
        candidate_if_good(pairs, block_candidate(best_n, rng), best_n, limit)
        candidate_if_good(pairs, fourier_candidate(best_n, rng, True), best_n, limit)
        candidate_if_good(pairs, fourier_candidate(best_n, rng, False), best_n, limit)
        candidate_if_good(pairs, thue_morse_candidate(best_n), best_n, limit)
        candidate_if_good(pairs, project_feasible(0.5 * (best_h + best_h[::-1]), best_n), best_n, limit)

        top = top_candidates(pairs, 3)
        for i, (_, hc) in enumerate(top):
            remaining = alt_end - time.time()
            if remaining < 15.0:
                break
            alloc = min(remaining / float(len(top) - i), max(20.0, 0.04 * total))
            hh, cc = optimize_at_n(hc, best_n, time.time() + alloc, do_polish=False)
            if cc < best_c:
                best_h = hh
                best_c = cc
                best_n = int(best_n)

    # Higher-resolution phases.
    high_sizes = []
    if total >= 300 and best_n < max_n:
        cur = int(best_n)
        while cur < max_n:
            nxt = cur * 2
            if nxt > max_n:
                nxt = int(max_n)
            if nxt > cur:
                high_sizes.append(nxt)
                cur = nxt
            else:
                break

    if total < 500 and len(high_sizes) > 1:
        high_sizes = high_sizes[:1]
    elif total < 750 and len(high_sizes) > 2:
        high_sizes = high_sizes[:2]

    if high_sizes:
        L = len(high_sizes)
        if L == 1:
            fracs = [0.92]
        elif L == 2:
            fracs = [0.65, 0.92]
        elif L == 3:
            fracs = [0.58, 0.78, 0.92]
        else:
            fracs = []
            for i in range(L):
                fracs.append(0.50 + 0.42 * float(i) / float(L - 1))

        for pi, size in enumerate(high_sizes):
            phase_end = min(deadline - 3.0, t0 + fracs[pi] * total)
            if phase_end <= time.time() + 20.0:
                continue

            starts = []
            h_interp = interpolate_h(best_h, best_n, int(size))
            starts.append((exact_c5(h_interp), h_interp))

            h_pert = perturb_candidate(h_interp, int(size), rng, 0.12)
            starts.append((exact_c5(h_pert), h_pert))

            if total >= 800 and phase_end > time.time() + 80.0:
                h_four = fourier_candidate(int(size), rng, False)
                starts.append((exact_c5(h_four), h_four))

            top = top_candidates(starts, 1)
            if not top:
                continue

            hh, cc = optimize_at_n(top[0][1], int(size), phase_end, do_polish=(pi == L - 1))
            if cc < best_c:
                best_h = hh
                best_c = cc
                best_n = int(size)

    else:
        cont_end = min(deadline - 3.0, t0 + 0.92 * total)
        if cont_end > time.time() + 20.0:
            hh, cc = optimize_at_n(best_h, best_n, cont_end, do_polish=True)
            if cc < best_c:
                best_h = hh
                best_c = cc
                best_n = int(best_n)

    # Symmetry trial.
    h_sym = project_feasible(0.5 * (best_h + best_h[::-1]), best_n)
    c_sym = exact_c5(h_sym)
    if c_sym < best_c:
        best_h = h_sym
        best_c = c_sym

    # Final local polish.
    if time.time() < deadline - 5.0:
        hh, cc = active_polish(best_h.copy(), best_n, deadline - 2.0, alpha=5000.0, max_steps=3000)
        if cc < best_c:
            best_h = hh
            best_c = cc
            best_n = int(best_n)

    # Final convex linearization polish.
    if time.time() < deadline - 12.0:
        hh, cc = linearize_lp_polish(best_h, best_n, deadline - 2.0, max_rounds=5, max_active=60)
        if cc < best_c:
            best_h = hh
            best_c = cc
            best_n = int(best_n)

    best_h = project_feasible(best_h, int(best_n))
    best_c = exact_c5(best_h)
    return best_h, float(best_c), int(best_n)
