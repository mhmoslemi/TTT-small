- i appears as the second factor when the sum index j satisfies j+k=i, i.e., j=i-k: h[i-k]·(1-h[i])·dx (if i-k is valid)

= base_k + [t·(1-h[i+k]) + h[i-k]·(1-t)]·dx  (if both terms valid)
= base_k + [t - t·h[i+k] + h[i-k] - h[i-k]·t]·dx
= base_k + h[i-k]·dx + t·dx·(1 - h[i+k] - h[i-k])

b_k = base_k + h[i-k]·dx (if i-k is valid, else base_k)

def _cons_fun(x, n):
    return float(x.sum() - n / 2.0)

def _cons_jac(x, n):
    return np.ones(len(x))

def slsqp_optimize(h, n, alpha=200000, maxiter=1500, budget_s=60):
    t0 = time.time()
    n_pts = len(h)
    bounds = [(0.0, 1.0)] * n_pts
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n_pts,)}]
    try:
        res = minimize(smooth_obj_func, h, args=(n_pts, alpha),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < smooth_obj_func(h, n_pts, alpha):
            return project_feasible(res.x, n_pts)
    except Exception:
        pass
    return project_feasible(h, n_pts)

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = 0 if radius is None else max(0, i - radius)
                hi = n if radius is None else min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def targeted_local_search(h, n, deadline):
    """Focus on reducing the worst shift specifically."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    t0 = time.time()
    while time.time() < deadline:
        c_vec = np.correlate(h, 1 - h, mode='full') * dx
        k_star = int(np.argmax(c_vec))
        improved = False
        for i in range(n):
            if time.time() > deadline - 5:
                break
            for delta in [0.01, 0.005, 0.002, 0.001]:
                if h[i] + delta <= 1.0:
                    h_old = h[i]
                    h[i] = h_old + delta
                    h = project_feasible(h, n)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] = h_old
                        h = project_feasible(h, n)
                if h[i] - delta >= 0.0:
                    h_old = h[i]
                    h[i] = h_old - delta
                    h = project_feasible(h, n)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] = h_old
                        h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c

def anneal_search(h, n, deadline, T_init=0.005, T_min=0.0001, alpha_cool=0.97):
    """Simulated annealing targeting the max overlap."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = T_init
    rng = np.random.RandomState(12345)
    t0 = time.time()
    
    while time.time() < deadline and T > T_min:
        c_vec = np.correlate(h, 1 - h, mode='full')
        max_idx = int(np.argmax(c_vec))
        c_max = float(c_vec[max_idx])
        
        # Propose a move: pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        direction = rng.choice([1, -1])
        delta = rng.choice([0.005, 0.01, 0.02, 0.03, 0.05])
        
        if direction == 1:
            if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                continue
            h[i] += delta
            h[j] -= delta
        else:
            if h[i] - delta < 0.0 or h[j] + delta > 1.0:
                continue
            h[i] -= delta
            h[j] += delta
        
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        delta_c = c_new - compute_c5(best_h, n) if c_new > best_c else c_new - best_c
        
        if c_new <= best_c or rng.random() < np.exp(-(c_new - best_c) / T):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # Revert
            if direction == 1:
                h[i] -= delta
                h[j] += delta
            else:
                h[i] += delta
                h[j] -= delta
            h = project_feasible(h, n)
        
        T *= alpha_cool
    
    return best_h, best_c

def multi_res_optimize(h_best, n_best, c_best, deadline):
    best_h = h_best.copy()
    best_c = c_best
    best_n = n_best
    x_orig = np.linspace(0, 2, n_best)
    
    for n_try in [180, 220, 280, 320, 350]:
        rem = deadline - time.time()
        if rem < 50:
            break
        if n_try == n_best:
            continue
        
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_orig, h_best)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc_cd = min(60, rem * 0.35)
        h_cd, c_cd = lagrangian_cd(h_interp, n_try, eta=8.0, max_pass=60, budget_s=alloc_cd)
        
        rem = deadline - time.time()
        if rem > 40:
            h_sl = slsqp_optimize(h_cd, n_try, alpha=200000, maxiter=1000,
                                  budget_s=min(40, rem - 20))
            c_sl = compute_c5(h_sl, n_try)
            if c_sl < c_cd:
                h_cd, c_cd = h_sl, c_sl
        
        rem = deadline - time.time()
        if rem > 20:
            h_ps, c_ps = pair_search(h_cd, n_try, deadline - 5,
                                     [0.01, 0.005, 0.001], radius=25)
            if c_ps < c_cd:
                h_cd, c_cd = h_ps, c_ps
        
        if c_cd < best_c:
            best_c = c_cd
            best_h = project_feasible(h_cd, n_try)
            best_n = n_try
    
    return best_h, best_c, best_n

def multi_restart(h_best, n_best, c_best, deadline, rng):
    best_h = h_best.copy()
    best_c = c_best
    n = n_best
    rem_total = deadline - time.time()
    n_restarts = max(2, int(rem_total / 60))
    
    for r in range(n_restarts):
        rem = deadline - time.time()
        if rem < 30:
            break
        scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        
        alloc = min(50, rem - 10)
        if alloc < 15:
            break
        
        h_cd, c_cd = lagrangian_cd(hp, n, eta=10, max_pass=40, budget_s=alloc * 0.6)
        
        rem = deadline - time.time()
        if rem > 25:
            h_sl = slsqp_optimize(h_cd, n, alpha=300000, maxiter=800,
                                  budget_s=min(25, rem - 10))
            c_sl = compute_c5(h_sl, n)
            if c_sl < c_cd:
                h_cd, c_cd = h_sl, c_sl
        
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    deadline = t_start + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    best_n = n_init
    
    # Phase 1: Lagrangian coordinate descent (main optimization)
    alloc1 = min(250, (deadline - t_start) * 0.25)
    h_cd, c_cd = lagrangian_cd(h_init, n_init, eta=8.0, max_pass=150, budget_s=alloc1)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd
        best_n = n_init
    
    # Phase 2: SLSQP polish with high alpha
    rem = deadline - time.time()
    if rem > 60:
        h_sl = slsqp_optimize(best_h, best_n, alpha=300000, maxiter=2000,
                              budget_s=min(80, rem - 30))
        c_sl = compute_c5(h_sl, best_n)
        if c_sl < best_c:
            best_c = c_sl
            best_h = h_sl
    
    # Phase 3: Second coordinate descent pass (from SLSQP result)
    rem = deadline - time.time()
    if rem > 80:
        h_cd2, c_cd2 = lagrangian_cd(best_h, best_n, eta=5.0, max_pass=100,
                                     budget_s=min(100, rem - 30))
        if c_cd2 < best_c:
            best_c = c_cd2
            best_h = h_cd2
    
    # Phase 4: Targeted local search on worst shift
    rem = deadline - time.time()
    if rem > 60:
        h_tl, c_tl = targeted_local_search(best_h, best_n, deadline - 30)
        if c_tl < best_c:
            best_c = c_tl
            best_h = h_tl
    
    # Phase 5: Pair search (fine deltas)
    rem = deadline - time.time()
    if rem > 40:
        if best_n <= 300:
            h_ps, c_ps = pair_search(best_h, best_n, deadline - 15,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005],
                                     radius=40)
        else:
            h_ps, c_ps = pair_search(best_h, best_n, deadline - 15,
                                     [0.01, 0.005, 0.001],
                                     radius=25)
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps
    
    # Phase 6: Multi-resolution exploration
    rem = deadline - time.time()
    if rem > 100:
        h_mr, c_mr, n_mr = multi_res_optimize(best_h, best_n, best_c, deadline - 20)
        if c_mr < best_c:
            best_c = c_mr
            best_h = h_mr
            best_n = n_mr
    
    # Phase 7: Multi-restart
    rem = deadline - time.time()
    if rem > 80:
        h_mr2, c_mr2 = multi_restart(best_h, best_n, best_c, deadline - 20, rng)
        if c_mr2 < best_c:
            best_c = c_mr2
            best_h = h_mr2
    
    # Phase 8: Final SLSQP + pair polish
    rem = deadline - time.time()
    if rem > 20:
        h_fin = slsqp_optimize(best_h, best_n, alpha=500000, maxiter=1500,
                               budget_s=min(30, rem - 10))
        c_fin = compute_c5(h_fin, best_n)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    
    # Phase 9: Ultra-fine pair search
    rem = deadline - time.time()
    if rem > 15 and best_n <= 300:
        h_ultra, c_ultra = pair_search(best_h, best_n, deadline - 3,
                                       [0.0005, 0.0002, 0.0001],
                                       radius=20)
        if c_ultra < best_c:
            best_c = c_ultra
            best_h = h_ultra
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
