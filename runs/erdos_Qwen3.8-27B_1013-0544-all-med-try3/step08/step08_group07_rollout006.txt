    g = compute_all_g(h, n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)

    dx = 2.0 / n

    grad = np.zeros(n)

    for idx in range(len(g)):

            continue
        k = idx - (n - 1)  # lag
        w = weights[idx] * dx
        # dc_k/dh[m] = (1-h[m-k])*[0≤m-k≤n-1] - h[m+k]*[0≤m+k≤n-1]

            grad[m1] += w * (1 - h[m1 - k])
            # m+k in [0,n-1] means m in [0, n-1-k]

            m2 = np.arange(-k, n)
            grad[m2] -= w * h[m2 + k]

                                  alpha=500, max_iter=3000, lr0=0.005)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
