        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 1:
                grad[k:n] += wm * (-x[0:nv-1])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 1:
                grad[0:nv-1] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def fw_optimize(h, n, deadline, max_iter=500):
    """Frank-Wolfe / Mirror Descent with adaptive step"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_target = n // 2
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        alpha = 2.0 / (it + 2)  # FW step size
        # Use higher alpha for smooth max
        a_smooth = max(500, 1000 * (1 + it / 100))
        g = grad_smooth(h, n, a_smooth, 1e-6)
        
        # Linear oracle: binary assignment
        idx = np.argsort(g)
        v = np.zeros(n)
        v[idx[:n_target]] = 1.0
        
        # Frank-Wolfe update
        h_new = (1 - alpha) * h + alpha * v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            alpha2 = alpha * 0.5
            h_new2 = (1 - alpha2) * h + alpha2 * v
            h_new2 = project_feasible(h_new2, n)
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c - 1e-14:
                best_c = c_new2
                best_h = h_new2.copy()
                h = h_new2
    
    return best_h, best_c

def nesterov_descent(h, n, deadline, max_iter=2000):
    """Nesterov accelerated gradient on smooth approximation"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha_smooth = 1000
    mu = 100.0
    L = 0.1  # initial Lipschitz estimate
    x = h.copy()
    y = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_smooth(y, n, alpha_smooth, mu)
        # Gradient step
        x_new = y - (1.0 / L) * g
        x_new = project_feasible(x_new, n)
        # Nesterov momentum
        t = (1 + np.sqrt(1 + 4 * it * it)) / 2
        y_new = x_new + (it / (it + t)) * (x_new - x)
        y_new = project_feasible(y_new, n)
        
        # Check objective
        c_new = compute_c5(y_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = y_new.copy()
        
        # Adaptive L
        if it % 50 == 0 and it > 0:
            # Backtracking
            diff = np.linalg.norm(y_new - x)
            if diff > 1e-10:
                obj_new = obj_smooth(y_new, n, alpha_smooth, mu)
                obj_old = obj_smooth(x, n, alpha_smooth, mu)
                quad = obj_old + g @ (y_new - x) + 0.5 * L * diff * diff
                if obj_new > quad + 1e-12:
                    L *= 2
                elif L > 0.5:
                    L *= 0.7
        x = x_new
        y = y_new
    
    return best_h, best_c

def perturbation_restart(h, n, deadline, n_restarts=10, radius=0.02):
    """Try random perturbations and polish each"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    budget_each = (deadline - time.time()) / n_restarts
    
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        t_end = time.time() + budget_each
        h_pert = h + rng.randn(n) * radius * (rng.rand(n) < 0.1)
        h_pert = project_feasible(h_pert, n)
        
        # Quick L-BFGS polish
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_smooth, h_pert, jac=grad_smooth,
                          args=(n, 5000, 100.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'maxfun': 2000})
            h_pert = project_feasible(res.x, n)
            c_pert = compute_c5(h_pert, n)
            if c_pert < best_c - 1e-14:
                best_c = c_pert
                best_h = h_pert.copy()
        except Exception:
            pass
    
    return best_h, best_c

def refined_local_search(h, n, deadline):
    """Very fine-grained coordinate descent on active constraints"""
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    
    # Focus on positions that affect active constraints
    active_threshold = max_val - 0.005
    active = np.where(c_vec >= active_threshold)[0]
    
    # Compute sensitivity: for each position i, which active shifts does it affect?
    affected = np.zeros(n)
    for mm in active:
        k = mm - (n - 1)
        if k >= 0:
            affected[0:n-k] += 1
            if n-k-1 >= k:
                affected[k:n] += 1
        else:
            kk = -k
            affected[kk:n] += 1
            affected[0:n-kk] += 1
    
    # Sort positions by sensitivity (most affected first)
    order = np.argsort(-affected)
    
    # Try small adjustments at high-sensitivity positions
    steps = [0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in order[:200]:
                if time.time() > deadline:
                    return h, best_c
                i = int(i)
                # Try moving mass from i to j
                for j in order[:200]:
                    j = int(j)
                    if i == j:
                        continue
                    # h[i] += step, h[j] -= step
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # h[i] -= step, h[j] += step
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    
    return h, best_c

def upsample(h_old, n_old, n_new):
    """Conservative upsampling that preserves integral"""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
