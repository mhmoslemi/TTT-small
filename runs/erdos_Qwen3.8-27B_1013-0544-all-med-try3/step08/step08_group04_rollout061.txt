import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, dx, alpha):
    vals = corr * dx
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_max_weights(corr, dx, alpha):
    vals = corr * dx
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def analytic_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    w = smooth_max_weights(corr, dx, alpha)
    grad = np.zeros(n)
    nk = 2 * n - 1
    p_mid = w[n - 1]
    grad += p_mid * (1 - 2 * h)
    for ell in range(1, n):
        w_plus = w[n - 1 + ell]
        w_minus = w[n - 1 - ell]
        wp = w_plus + w_minus
        if wp < 1e-15:
            continue
        idx_lo = ell
        idx_hi = n - 1 - ell
        if idx_lo <= idx_hi:
            seg = slice(idx_lo, idx_hi + 1)
            g1 = w_plus * (1 - h[:n - ell][seg]) if False else None
            # For lag +ell: (1-h[j-ell])*[ell<=j<=n-1] - h[j+ell]*[0<=j<=n-1-ell]
            # For lag -ell: (1-h[j+ell])*[0<=j<=n-1-ell] - h[j-ell]*[ell<=j<=n-1]
            j_range = np.arange(n)
            mask_plus_1 = (j_range >= ell) & (j_range <= n - 1)
            mask_plus_2 = (j_range >= 0) & (j_range <= n - 1 - ell)
            mask_minus_1 = (j_range >= 0) & (j_range <= n - 1 - ell)
            mask_minus_2 = (j_range >= ell) & (j_range <= n - 1)
            term_plus = w_plus * ((1 - np.roll(h, ell)[:n - ell]) * mask_plus_1 - h[:n - ell] * mask_plus_2)
            # Simpler: just compute directly
            dp = np.zeros(n)
            dm = np.zeros(n)
            # lag = +ell
            dp[ell:] = (1 - h[:-ell])
            dp[:n - ell] -= h[ell:]
            # lag = -ell
            dm[:n - ell] = (1 - h[ell:])
            dm[ell:] -= h[:-ell]
            grad += w_plus * dp + w_minus * dm
    return grad * dx

def grad_slsqp(h, n, alpha, maxiter=2000, ftol=1e-15):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return smooth_max_val(corr, dx, alpha)

    def jac(x):
        return analytic_gradient(x, n, alpha)

    def con_fun(x):
        return x.sum() - target

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < smooth_max_val(np.correlate(h, 1-h, mode='full'), dx, alpha):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h, n)

def lbfgs_penalty(h, n, alpha, maxiter=1000, tol=1e-14):
    dx = 2.0 / n
    target = n / 2.0
    mu = 1e4

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        val = smooth_max_val(corr, dx, alpha)
        s = x.sum() - target
        return val + mu * s * s

    def jac(x):
        g = analytic_gradient(x, n, alpha)
        s = x.sum() - target
        g += 2 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': tol,
                                               'gtol': 1e-12, 'maxfun': 10000})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coordinate_descent(h, n, alpha, max_sweeps=100, tol=1e-12):
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old_val = h[j]
            # Compute D_m = d(corr[m])/d(h[j]) for all m
            D = np.zeros(2 * n - 1)
            for ell in range(0, n):
                # lag = +ell
                if ell == 0:
                    D[n - 1] += 1 - 2 * old_val
                else:
                    # lag +ell: d/dh[j] = (1-h[j-ell])*[ell<=j] - h[j+ell]*[j<=n-1-ell]
                    if ell <= j <= n - 1:
                        D[n - 1 + ell] += (1 - h[j - ell])
                    if 0 <= j <= n - 1 - ell:
                        D[n - 1 + ell] -= h[j + ell]
                    # lag -ell: d/dh[j] = (1-h[j+ell])*[j<=n-1-ell] - h[j-ell]*[ell<=j]
                    if 0 <= j <= n - 1 - ell:
                        D[n - 1 - ell] += (1 - h[j + ell])
                    if ell <= j <= n - 1:
                        D[n - 1 - ell] -= h[j - ell]
            # C_m = corr[m] - D_m * old_val
            corr_cur = h[j]  # placeholder
            # Recompute corr with current h for C_m
            corr_full = np.correlate(h, 1 - h, mode='full')
            C = corr_full - D * old_val
            # Minimize F(t) = (1/alpha)*log(sum(exp(alpha*dx*(C_m + D_m*t))))
            # F'(t) = dx * sum(D_m * p_m(t))
            # Find t in [0,1] minimizing F
            def f_prime(t):
                vals = C + D * t
                m = vals.max()
                sh = alpha * dx * (vals - m)
                p = np.exp(sh - sh.max())
                p /= p.sum()
                return dx * np.dot(D, p)
            # Convex, so F' is non-decreasing
            fp0 = f_prime(0.0)
            fp1 = f_prime(1.0)
            if fp0 >= 0:
                t_new = 0.0
            elif fp1 <= 0:
                t_new = 1.0
            else:
                # Bisection
                lo, hi = 0.0, 1.0
                for _ in range(50):
                    mid = (lo + hi) / 2
                    if f_prime(mid) < 0:
                        lo = mid
                    else:
                        hi = mid
                t_new = (lo + hi) / 2
            h[j] = t_new
            if abs(h[j] - old_val) > 1e-10:
                improved = True
        # Project for sum constraint
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
        if not improved:
            break
    return h, best_c

def refine_escalating(h, n, deadline):

def make_structured_init(n, rng):
    h = np.zeros(n)
    pattern = rng.choice([0, 1, 2, 3], p=[0.4, 0.3, 0.2, 0.1])
    if pattern == 0:
        # Block: 1 on first half
        h[:n//2] = 1.0
    elif pattern == 1:
        # Periodic
        for i in range(n):
            h[i] = 1.0 if (i % 4) < 2 else 0.0
    elif pattern == 2:
        # Anti-periodic
        for i in range(n):
            h[i] = 1.0 if (i % 2 == 0) else 0.0
    else:
        # Random binary
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
    return project_feasible(h, n)
