def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def soft_max_obj(params, n, alpha):
    h = params
    g = compute_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        return soft_max_obj(x, n, alpha)
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_sa(h_bin, n, deadline, t_start, cooling, rng):
    h = h_bin.copy()
    c = compute_c5(h, n)
    best_h = h.copy()
    best_c = c
    t = t_start
    
    # Pre-compute for speed
    while time.time() < deadline:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        c_new = compute_c5(h, n)
        delta = c_new - c
        if delta < 0 or (t > 1e-15 and rng.rand() < np.exp(-delta / t)):
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        t *= cooling
    
    # Final greedy polish
    h2 = best_h.copy()
    c2 = best_c
    for _ in range(200):
        if time.time() > deadline:
            break
        ones = list(np.where(h2 == 1.0)[0])
        zeros = list(np.where(h2 == 0.0)[0])
        improved = False
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            found = False
            rng.shuffle(ones)
            for j in ones:
                h2[i], h2[j] = 1.0, 0.0
                c_new = compute_c5(h2, n)
                if c_new < c2 - 1e-15:
                    c2 = c_new
                    improved = True
                    ones[ones.index(j)] = i
                    zeros.remove(i)
                    found = True
                    break
                else:

            if found:
                break
        if not improved:

    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    
    return best_h, best_c

def pair_refine_binary(h_bin, n, deadline, delta_list):
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for d in delta_list:

            break
        for _ in range(20):  # Multiple passes

                break

                    break

                    if time.time() > deadline:
                        break
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj

                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    
    return best_h, best_c

def thue_morse_binary(n, rng):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') & 1
    # Adjust to have exactly n/2 ones

    target = n // 2

        zeros = np.where(tm == 0)[0]
        rng.shuffle(zeros)
        tm[zeros[:target - count]] = 1
    h = tm.astype(float)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def soft_max_obj(x, n, alpha):
    g = compute_g(x, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
