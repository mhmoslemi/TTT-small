def simulated_annealing(initial_state, n, budget, T_initial, T_final):
    """
    Simulated annealing implementation optimized for the step function problem.
    
    Args:
        initial_state (numpy.ndarray): Starting step function representation
        n (int): Number of sample points
        budget (float): Time budget in seconds
        T_initial (float): Initial temperature
        T_final (float): Final temperature
    
    Returns:
        tuple: (best_score, best_state)
    """
    # Validate input
    if len(initial_state) != n:
        raise ValueError(f"Initial state must have {n} elements")
    
    # Setup
    bounds = [(0.0, 1.0)] * n
    best_state = initial_state.copy()
    best_score = compute_c5(best_state, n)
    current_state = best_state.copy()
    cooling_rate = 0.995
    
    # Function to generate neighbors (with mass preservation)
    def generate_neighbor(state):
        i = np.random.randint(0, n)
        j = np.random.choice([idx for idx in range(n) if idx != i])
        d = np.random.uniform(-0.04, 0.04)
        neighbor = state.copy()
        # Update two points to preserve mass
        neighbor[i] = np.clip(neighbor[i] + d, 0, 1)
        neighbor[j] = np.clip(neighbor[j] - d, 0, 1)
        # Ensure valid representation
        return project_feasible(neighbor, n)
    
    start_time = time.time()
    T = T_initial
    
    while time.time() - start_time < budget:
        # Generate neighbor
        new_state = generate_neighbor(current_state)
        
        # Evaluate new state
        new_score = compute_c5(new_state, n)
        
        # Acceptance criterion
        if new_score < best_score:
            best_score = new_score
            best_state = new_state.copy()
            T = max(T_initial, T * cooling_rate)  # Boost for better states
        elif np.random.random() < np.exp(-(new_score - best_score) / T:
            best_state = new_state.copy()
            best_score = new_score
            T = T * cooling_rate
        
        T = max(T_final, T)
    
    return (best_score, best_state)
