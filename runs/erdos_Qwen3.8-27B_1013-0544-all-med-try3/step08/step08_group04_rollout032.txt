            idx2 = np.arange(p, n)
            grad[idx2] -= wd * x[idx2 - p]
        elif p < 0:
            q = -p
            idx1 = np.arange(q, n)
            grad[idx1] += wd * (1.0 - x[idx1 - q])
            idx2 = np.arange(0, n - q)
            grad[idx2] -= wd * x[idx2 + q]
        else:
            grad += wd * (1.0 - 2.0 * x)
    return grad

def penalty_obj(x, n, alpha, penalty):
    val = soft_max_obj(x, n, alpha)
    s = x.sum() - n / 2.0
    out_of_bounds = np.maximum(0, -x).sum() + np.maximum(0, x - 1.0).sum()
    return val + penalty * (s * s) + 1e6 * out_of_bounds

def penalty_grad(x, n, alpha, penalty):
    val = soft_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    val += 2.0 * penalty * s
    val += 1e6 * (-np.sign(-x) * (x < 0) - np.sign(x - 1.0) * (x > 1.0))
    return val

def sum_eq(x, n):
    return float(x.sum() - n / 2.0)

def sum_eq_jac(x, n):
    return np.ones(n)

def optimize_slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(
            lambda x: soft_max_obj(x, n, alpha),
            h,
            jac=lambda x: soft_max_grad(x, n, alpha),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def optimize_lbfgsb(h, n, alpha, penalty=1e6, maxiter=5000):
    x = h.copy()
    for _ in range(3):
        res = fmin_l_bfgs_b(
            lambda v: penalty_obj(v, n, alpha, penalty),
            x,
            fprime=lambda v: penalty_grad(v, n, alpha, penalty),
            pgtol=1e-18,
            gtol=1e-12,
            maxiter=maxiter
        )
        x = project_feasible(res[0], n)
    return x

def pair_search_full(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def pair_search_near(h, n, deadline, deltas, radius):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                if time.time() > deadline:
                    return h, best_c
    return h, best_c

def gradient_step(h, n, alpha, lr, num_steps):
    for _ in range(num_steps):
        g = soft_max_grad(h, n, alpha)
        s = h.sum() - n / 2.0
        g -= s
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        c_old = compute_c5(h, n)
        if c_new < c_old:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-12:
                break
    return h

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    alphas = [500, 2000, 10000, 50000, 200000, 500000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        if n <= 300:
            ht = optimize_slsqp(h, n, a, maxiter=2000)
        else:
            ht = optimize_lbfgsb(h, n, a, penalty=1e7, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            h = ht

    if n <= 200:
        if time.time() < deadline - 10:
            h, best_c = pair_search_full(h, n, deadline - 5,
                                         [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    elif n <= 300:
        if time.time() < deadline - 10:
            h, best_c = pair_search_near(h, n, deadline - 5,
                                         [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005], radius=60)
    else:
        if time.time() < deadline - 10:
            h, best_c = pair_search_near(h, n, deadline - 5,
                                         [0.01, 0.005, 0.002, 0.001], radius=40)

    if time.time() < deadline - 5:
        if n <= 300:
            ht = optimize_slsqp(h, n, 1000000, maxiter=3000)
        else:
            ht = optimize_lbfgsb(h, n, 1000000, penalty=1e8, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    if n <= 200 and time.time() < deadline - 2:
        h, best_c = pair_search_full(h, n, deadline - 1,
                                     [0.0005, 0.0002, 0.0001, 0.00005])
    elif time.time() < deadline - 2:
        h, best_c = pair_search_near(h, n, deadline - 1,
                                     [0.0005, 0.0002, 0.0001], radius=30)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=True)
    x_new = np.linspace(0, 2, n_new, endpoint=True)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def structured_perturbation(h, n, rng, pattern):
    hp = h.copy()
    if pattern == 0:
        hp += rng.normal(0, 0.05, n)
    elif pattern == 1:
        hp[:n//4] += 0.03
        hp[-n//4:] -= 0.03
    elif pattern == 2:
        period = max(2, n // 8)
        for i in range(0, n, period):
            end = min(i + period // 2, n)
            hp[i:end] += 0.04
            end2 = min(i + period, n)
            hp[end:

end2] += -0.04
    elif pattern == 3:
        mid = n // 2
        hp[mid//2:mid] += 0.05
        hp[mid:mid + mid//2] -= 0.05
    return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at initial resolution
    alloc1 = min(180, (deadline - t0) * 0.20)
    h_opt, c_opt = optimize_at_resolution(best_h, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Interpolate to higher resolution and optimize
    for n_try in [400, 500, 600]:
        if time.time() > deadline - 120:
            break
        h_interp = interpolate_h(best_h, len(best_h), n_try)
        alloc = min(150, (deadline - time.time()) * 0.20)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Also try from coarser resolution for global exploration
    if time.time() < deadline - 100:
        for n_coarse in [120, 150]:
            if time.time() > deadline - 80:
                break
            if n_coarse == n_init:
                continue
            h_coarse = interpolate_h(best_h, len(best_h), n_coarse)
            h_coarse += rng.normal(0, 0.02, n_coarse)
            h_coarse = project_feasible(h_coarse, n_coarse)
            alloc = min(80, (deadline - time.time()) * 0.15)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_resolution(h_coarse, n_coarse, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Multi-restart with structured perturbations at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 60))
        n_restarts = min(n_restarts, 8)
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            pattern = r % 4
            hp = structured_perturbation(best_h, n_best, rng, pattern)
            alloc = min(60, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Interpolate best to even higher resolution for final push
    n_final = max(400, n_best)
    if n_final != n_best and time.time() < deadline - 60:
        h_interp = interpolate_h(best_h, n_best, n_final)
        alloc = min(80, deadline - time.time() - 20)
        if alloc > 30:
            h_opt, c_opt = optimize_at_resolution(h_interp, n_final, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n_best = n_final

    # Phase 6: Final intensive pair search on best solution
    if time.time() < deadline - 8:
        n = len(best_h)
        if n <= 200:
            best_h, best_c = pair_search_full(best_h, n, deadline - 3,
                                             [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        elif n <= 300:
            best_h, best_c = pair_search_near(best_h, n, deadline - 3,
                                             [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=50)
        else:
            best_h, best_c = pair_search_near(best_h, n, deadline - 3,
                                             [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=35)

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 3:
        n = len(best_h)
        if n <= 300:
            ht = optimize_slsqp(best_h, n, 2000000, maxiter=2000)
        else:
            ht = optimize_lbfgsb(best_h, n, 2000000, penalty=1e8, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    if best_c > 0.381:
        best_h = project_feasible(h_init.copy(), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
