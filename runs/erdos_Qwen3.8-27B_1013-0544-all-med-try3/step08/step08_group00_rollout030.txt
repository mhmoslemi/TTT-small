```
def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(1000):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
```

```python
def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(1000):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def compute_c5(h, n):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def b_counts_vec(h, n):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr[::-1] * dx


def binary_counts(h, n):
    h32 = np.asarray(h, dtype=np.int32)
    o32 = (1 - h32).astype(np.int32)
    corr = np.correlate(h32, o32, mode="full")
    return corr[::-1].astype(np.int32)


def apply_flip(h, c, i, delta, off):
    n = len(h)
    rhs = n - i - 1
    lhs = i
    pos0 = off + 1
    neg0 = off - 1

    if delta == 1:
        if rhs > 0:
            rs = h[i + 1:i + 1 + rhs]
            c[pos0:pos0 + rhs] += 1
            c[pos0:pos0 + rhs] -= rs
            c[neg0:off - rhs - 1:-1] -= rs
        if lhs > 0:
            ls = h[i - 1::-1]
            c[pos0:pos0 + lhs] -= ls
            c[neg0:off - lhs - 1:-1] += 1
            c[neg0:off - lhs - 1:-1] -= ls
    else:
        if rhs > 0:
            rs = h[i + 1:i + 1 + rhs]
            c[pos0:pos0 + rhs] -= 1
            c[pos0:pos0 + rhs] += rs
            c[neg0:off - rhs - 1:-1] += rs

            c[pos0:pos0 + lhs] += ls
            c[neg0:off - lhs - 1:-1] -= 1
            c[neg0:off - lhs - 1:-1] += ls


def apply_swap(h, c, i, j, off):
    apply_flip(h, c, i, -1, off)
    h[i] = 0
    apply_flip(h, c, j, 1, off)
    h[j] = 1


def select_active(c, max_lags=24, band=1):
    M = int(np.max(c))
    active = np.flatnonzero(c >= M - band)
    if active.size > max_lags:
        order = np.argsort(c[active])[::-1]
        active = active[order[:max_lags]]
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=np.int64)
    return active


def add_lag_gradient(D, h, n, idx, weight=1.0):
    if weight == 0.0:
        return
    k = idx - (n - 1)

    if k >= 0:
        nv = n - k
        if weight == 1.0:
            D[0:nv] += 1.0
            D[0:nv] -= h[k:n]
            D[k:n] -= h[0:nv]
        else:
            D[0:nv] += weight
            D[0:nv] -= h[k:n] * weight
            D[k:n] -= h[0:nv] * weight
    else:
        kk = -k
        nv = n - kk
        if weight == 1.0:
            D[kk:n] += 1.0
            D[kk:n] -= h[0:nv]
            D[0:nv] -= h[kk:n]
        else:
            D[kk:n] += weight
            D[kk:n] -= h[0:nv] * weight
            D[0:nv] -= h[kk:n] * weight


def compute_D(h, c, n, max_lags=24):
    active = select_active(c, max_lags, 1)
    D = np.zeros(n, dtype=float)
    for p in range(active.size):
        add_lag_gradient(D, h, n, int(active[p]), 1.0)
    if active.size > 1:
        D /= float(active.size)
    return D


def choose_candidates(D, h, ones, zeros, khigh, klow):
    if ones.size == 0:
        high = ones
    else:
        d = D[ones]
        if ones.size > khigh:
            part = np.argpartition(d, ones.size - khigh)[-khigh:]
        else:
            part = np.arange(ones.size)
        high = ones[part]
        high = high[np.argsort(-D[high])]

    if zeros.size == 0:
        low = zeros
    else:
        dz = D[zeros]
        if zeros.size > klow:
            part = np.argpartition(dz, klow - 1)[:klow]
        else:
            part = np.arange(zeros.size)
        low = zeros[part]
        low = low[np.argsort(D[low])]

    return high, low


def try_best_pairs(h, c, high_i, low_j, cur_M, off, t_end):
    if high_i is None or low_j is None or high_i.size == 0 or low_j.size == 0:
        return False, cur_M

    best_m = cur_M
    best_i = -1
    best_j = -1
    best_snap = None
    timeout = False

    for ii in high_i:
        if timeout:
            break
        ii = int(ii)
        if h[ii] != 1:
            continue
        for jj in low_j:
            jj = int(jj)
            if jj == ii or h[jj] != 0:
                continue

            snap = c.copy()
            apply_swap(h, c, ii, jj, off)
            m = int(c.max())

            if m < best_m:
                best_m = m
                best_i = ii
                best_j = jj
                if best_snap is None:
                    best_snap = c.copy()
                else:
                    best_snap[:] = c

            c[:] = snap
            h[ii] = 1
            h[jj] = 0

            if time.time() > t_end:
                timeout = True
                break

    if best_i >= 0 and best_m < cur_M and best_snap is not None:
        c[:] = best_snap
        h[best_i] = 0
        h[best_j] = 1
        return True, best_m

    return False, cur_M


def hill_climb(h, n, t_end, seed=1, khigh=10, klow=10,
               recompute_every=10, random_batch=200, max_lags=24):
    off = n - 1
    h = np.asarray(h, dtype=np.uint8).copy()
    c = binary_counts(h, n)

    cur_M = int(c.max())
    best_M = cur_M
    best_h = h.copy()

    rng = np.random.RandomState(seed)
    count = recompute_every
    no_improve = 0

    D = None
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    high_i = None
    low_j = None

    while time.time() < t_end:
        if D is None or count >= recompute_every or high_i is None:
            D = compute_D(h, c, n, max_lags)
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
            high_i, low_j = choose_candidates(D, h, ones, zeros, khigh, klow)
            count = 0

        ok, newM = try_best_pairs(h, c, high_i, low_j, cur_M, off, t_end)
        if ok:
            cur_M = newM
            count += 1
            no_improve = 0
            if cur_M < best_M:
                best_M = cur_M
                best_h = h.copy()
            continue

        accepted = False
        allow = 0
        if no_improve > 0:
            allow = min(3 + no_improve, 8)

        for st in range(random_batch):
            if time.time() > t_end:
                break

            use_biased = (high_i is not None and low_j is not None and
                          high_i.size > 0 and low_j.size > 0 and st % 2 == 0)

            if use_biased:
                ii = int(high_i[int(rng.randint(high_i.size))])
                jj = int(low_j[int(rng.randint(low_j.size))])
            else:
                if ones.size == 0 or zeros.size == 0:
                    break
                ii = int(ones[int(rng.randint(ones.size))])
                jj = int(zeros[int(rng.randint(zeros.size))])

            if ii == jj or h[ii] != 1 or h[jj] != 0:
                if st % 32 == 31:
                    D = compute_D(h, c, n, max_lags)
                    ones = np.flatnonzero(h)
                    zeros = np.flatnonzero(1 - h)
                    high_i, low_j = choose_candidates(D, h, ones, zeros, khigh, klow)
                continue

            snap = c.copy()
            apply_swap(h, c, ii, jj, off)
            m = int(c.max())

            if m < cur_M:
                cur_M = m
                accepted = True
                count += 1
                if cur_M < best_M:
                    best_M = cur_M
                    best_h = h.copy()
                break
            elif allow > 0 and m <= cur_M + allow and rng.random() < 0.15:
                cur_M = m
                count += 1
            else:
                c[:] = snap
                h[ii] = 1
                h[jj] = 0

        if accepted:
            no_improve = 0
            continue

        no_improve += 1

        if cur_M > best_M + 12:
            h = best_h.copy()
            c = binary_counts(h, n)
            cur_M = best_M
            count = recompute_every
            no_improve = 0
            continue

        if no_improve > 2 and time.time() < t_end - 2.0:
            h = best_h.copy()
            c = binary_counts(h, n)
            cur_M = best_M

            allow = min(4 + no_improve, 18)
            p_moves = min(40 + no_improve * 15, 160)

            for _ in range(p_moves):
                if time.time() > t_end - 2.0:
                    break
                ones2 = np.flatnonzero(h)
                zeros2 = np.flatnonzero(1 - h)
                if ones2.size == 0 or zeros2.size == 0:
                    break

                ii = int(ones2[int(rng.randint(ones2.size))])
                jj = int(zeros2[int(rng.randint(zeros2.size))])
                if ii == jj:
                    continue

                snap = c.copy()
                apply_swap(h, c, ii, jj, off)
                m = int(c.max())

                if m <= best_M + allow and (m < cur_M or rng.random() < 0.35):
                    cur_M = m
                    if cur_M < best_M:
                        best_M = cur_M
                        best_h = h.copy()
                else:
                    c[:] = snap
                    h[ii] = 1
                    h[jj] = 0

            no_improve = 0
            count = recompute_every

    return best_h, int(best_M)


def random_binary(n, rng):
    h = np.zeros(n, dtype=np.uint8)
    perm = rng.permutation(n)
    h[perm[:n // 2]] = 1
    return h


def parent_to_binary(parent_vals, n, rng=None, sigma=0.0):
    m = parent_vals.size
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    vals = np.interp(x_new, x_old, parent_vals.astype(float))
    if sigma > 0.0 and rng is not None:
        vals = vals + rng.normal(0.0, sigma, n)

    k = n // 2
    if k <= 0:
        return np.zeros(n, dtype=np.uint8)
    if k >= n:
        return np.ones(n, dtype=np.uint8)

    idx = np.argpartition(-vals, k)[:k]
    h = np.zeros(n, dtype=np.uint8)
    h[idx] = 1
    return h


def optimize_binary_n(n, t_end, seed=123, parent_vals=None):
    n = int(n)
    if n % 2 != 0:
        n += 1
    if n < 2:
        return None, None

    if n <= 700:
        khigh = 12
        klow = 12
        recompute_every = 8
        random_batch = 250
        max_lags = 20
        base_trial = max(15.0, min(90.0, float(n) / 25.0))
    elif n <= 1200:
        khigh = 10
        klow = 10
        recompute_every = 12
        random_batch = 170
        max_lags = 24
        base_trial = max(20.0, min(90.0, float(n) / 25.0))
    else:
        khigh = 8
        klow = 8
        recompute_every = 16
        random_batch = 120
        max_lags = 28
        base_trial = max(25.0, min(100.0, float(n) / 22.0))

    rng = np.random.RandomState(seed)
    best_h = None
    best_M = None
    trial = 0

    while time.time() < t_end - 5.0:
        if parent_vals is not None and parent_vals.size > 0:
            if trial % 4 == 3:
                h0 = random_binary(n, rng)
            else:
                sigmas = [0.0, 0.02, 0.05, 0.10, 0.18, 0.30, 0.45, 0.60, 0.80]
                sigma = sigmas[trial % len(sigmas)]
                h0 = parent_to_binary(parent_vals, n, rng, sigma)
        else:
            h0 = random_binary(n, rng)

        t_start = time.time()
        mult = 2.0 if trial == 0 else 1.0
        t_trial = min(t_end - 2.0, t_start + base_trial * mult)
        if t_trial <= time.time():
            break

        h1, M1 = hill_climb(
            h0, n, t_trial,
            seed=seed + 1000 * trial,
            khigh=khigh, klow=klow,
            recompute_every=recompute_every,
            random_batch=random_batch,
            max_lags=max_lags
        )

        if M1 is not None and (best_M is None or M1 < best_M):
            best_M = int(M1)
            best_h = h1

        trial += 1

    return best_h, best_M


def select_active_float(c, max_lags=80, band_frac=0.015):
    M = float(c.max())
    thresh = M - max(1e-12, band_frac * max(1.0, M))
    active = np.flatnonzero(c >= thresh)
    if active.size > max_lags:
        active = active[np.argsort(c[active])[::-1][:max_lags]]
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=np.int64)
    return active


def smooth_max_obj(x, n, alpha, mu):
    if _RUN_DEADLINE > 0.0 and time.time() > _RUN_DEADLINE:
        raise RuntimeError("continuous_time_out")

    c = b_counts_vec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    lse = mx + float(np.log(np.sum(np.exp(sh - mx))))
    val = m + lse / alpha + mu * (float(np.sum(x)) - 0.5 * float(n)) ** 2
    return float(val)


def smooth_max_grad(x, n, alpha, mu, n_lag=80):
    if _RUN_DEADLINE > 0.0 and time.time() > _RUN_DEADLINE:
        raise RuntimeError("continuous_time_out")

    c = b_counts_vec(x, n)
    m = float(c.max())
    active = select_active_float(c, n_lag, 0.02)

    sh = alpha * (c[active] - m)
    mx = float(sh.max())
    w = np.exp(sh - mx)
    s = float(w.sum())
    if s <= 0.0:
        w = np.ones_like(w, dtype=float)
    else:
        w = w / s

    grad = np.zeros(n, dtype=float)
    for p in range(active.size):
        add_lag_gradient(grad, x, n, int(active[p]), float(w[p]))

    grad += 2.0 * mu * (float(np.sum(x)) - 0.5 * float(n))
    return grad


def continuous_polish(h0, n, t_end):
    if minimize is None or n <= 0:
        return h0, compute_c5(h0, n)

    global _RUN_DEADLINE
    _RUN_DEADLINE = max(0.0, float(t_end) - 1.0)

    hh = np.asarray(h0, dtype=float).copy()
    best_c = float(compute_c5(hh, n))

    if time.time() > t_end:
        return hh, best_c

    bounds = [(0.0, 1.0)] * int(n)
    configs = [
        (20.0, 50.0, 300),
        (80.0, 200.0, 250),
        (300.0, 500.0, 200),
    ]

    for alpha, mu, maxiter in configs:
        if time.time() > t_end - 3.0:
            break
        try:
            res = minimize(
                smooth_max_obj,
                hh,
                jac=smooth_max_grad,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-15,
                    "gtol": 1e-12,
                    "maxls": 50,
                }
            )
            h1 = project_feasible(res.x, n)
            c1 = compute_c5(h1, n)
            if c1 < best_c:
                best_c = float(c1)
                hh = h1
        except Exception:
            pass

    _RUN_DEADLINE = 0.0
    return hh, best_c


def run(seed=42, budget_s=1100, **kwargs):
    global _RUN_DEADLINE

    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(30.0, budget_s - 12.0)
    _RUN_DEADLINE = 0.0

    np.random.seed(seed)

    parent_vals = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if pv.size >= 4:
            parent_vals = project_feasible(pv, pv.size)
    except Exception:
        parent_vals = None

    if parent_vals is not None:
        best_h = parent_vals.copy()
        best_n = int(parent_vals.size)
        best_c = compute_c5(best_h, best_n)
    else:
        best_n = 500
        best_h = np.full(best_n, 0.5)
        best_c = compute_c5(best_h, best_n)

    if (parent_vals is not None and minimize is not None and
            best_n <= 900 and time.time() < deadline - 250.0):
        t_pol = min(time.time() + 60.0, deadline - 250.0)
        try:
            h_pol, c_pol = continuous_polish(parent_vals, best_n, t_pol)
            if c_pol < best_c:
                best_h = h_pol
                best_c = float(c_pol)
                best_n = int(len(h_pol))
        except Exception:
            pass

    if parent_vals is not None and parent_vals.size > 0:
        n0 = int(parent_vals.size)
        if n0 % 2 == 1:
            n0 += 1
        if n0 < 256:
            n0 = 256
        if n0 > 2048:
            n0 = 2048
    else:
        n0 = 500

    if n0 <= 700:
        raw_phases = [
            (n0, 0.35),
            (1024, 0.40),
            (1536, 0.18),
            (2048, 0.07),
        ]
    elif n0 <= 1100:
        raw_phases = [
            (n0, 0.45),
            (1536, 0.30),
            (2048, 0.15),
            (512, 0.10),
        ]
    else:
        other = 2048 if n0 != 2048 else 1536
        raw_phases = [
            (n0, 0.55),
            (other, 0.30),
            (1024, 0.15),
        ]

    seen = set()
    phases = []
    for nv, frac in raw_phases:
        m = int(nv)
        if m % 2 != 0:
            m += 1
        if m < 256:
            continue
        if m in seen:
            continue
        seen.add(m)
        phases.append((m, float(frac)))

    base_seconds = max(30.0, budget_s - 30.0)

    for nphase, frac in phases:
        if time.time() > deadline - 25.0:
            break
        seconds = frac * base_seconds
        t_end_phase = min(deadline - 15.0, time.time() + seconds)
        if t_end_phase <= time.time() + 10.0:
            continue

        try:
            h_bin, M_bin = optimize_binary_n(
                nphase,
                t_end_phase,
                seed=7919 + 13 * int(nphase) + int(seed),
                parent_vals=parent_vals
            )
        except Exception:
            h_bin = None
            M_bin = None

        if h_bin is not None and M_bin is not None:
            h_float = np.asarray(h_bin, dtype=float)
            c_act = compute_c5(h_float, nphase)
            if c_act < best_c:
                best_h = h_float
                best_c = float(c_act)
                best_n = int(nphase)

    if (minimize is not None and best_n <= 700 and
            time.time() < deadline - 12.0):
        t_final = min(time.time() + 45.0, deadline - 5.0)
        try:
            h_f, c_f = continuous_polish(best_h, best_n, t_final)
            if c_f < best_c:
                best_h = h_f
                best_c = float(c_f)
                best_n = int(len(h_f))
        except Exception:
            pass

    _RUN_DEADLINE = 0.0

    best_h = project_feasible(best_h, int(len(best_h)))
    best_n = int(len(best_h))
    best_c = compute_c5(best_h, best_n)

    return best_h, float(best_c), best_n
```
