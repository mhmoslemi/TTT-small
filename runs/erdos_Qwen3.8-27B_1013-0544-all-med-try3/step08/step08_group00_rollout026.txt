    """Smooth approximation of max using log-sum-exp"""
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

    """Gradient of smoothed max w.r.t. c"""
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

    """Gradient of ∫h(x)(1-h(x+k))dx w.r.t. h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    idx = k + (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] = (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] = (-h[kk:n]) * dx
    return grad

    val = smoothed_max(c, alpha)
    w = smoothed_max_grad(c, alpha)
    grad = np.zeros(n)

        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)

        if k >= 0:

    best_c = compute_c5(h, n)

    v = np.zeros_like(h)
    beta = momentum
    lr = lr_init

        y = h + beta * v
        
        # Compute gradient at y
        val, grad = obj_and_grad_smoothed(y, n, alpha)

            no_improve = 0

                if lr < 1e-8:
                    break
                v = np.zeros_like(h)
                no_improve = 0
            elif no_improve % 100 == 0:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def lse_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def obj_smoothed(h, n, alpha):
    c = compute_c5_vec(h, n)
    return lse_max(c, alpha)

def grad_smoothed(h, n, alpha):
    c = compute_c5_vec(h, n)
    w = lse_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_smoothed, h, jac=lambda x: grad_smoothed(x, n, alpha),
                  args=(n, alpha), method='L-BFGS-B', bounds=bounds,
                  options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                           'maxfun': 50000, 'maxls': 40, 'iprint': -1})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new, n)

def cyclic_cd(h, n, deadline, alpha=5000, step_decay=0.999):
    """Cyclic coordinate descent on smoothed objective"""
    best_c = compute_c5(h, n)
    step = 0.05
    bounds = [(0.0, 1.0)] * n
    no_improve = 0
    while time.time() < deadline and step > 1e-6:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            # Try increasing and decreasing h[i]
            for direction in [step, -step]:
                old_val = h[i]
                new_val = old_val + direction
                if new_val < -1e-12 or new_val > 1.0 + 1e-12:
                    continue
                h[i] = new_val
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                else:
                    h[i] = old_val
        if improved:
            no_improve = 0
            step *= step_decay
        else:
            no_improve += 1
            step *= 0.8
            if no_improve > 10:
                break
    return h, best_c

def pair_swap_search(h, n, deadline, steps=None):

def perturbation_restart(h, n, deadline, n_restarts=5):
    """Try random perturbations and re-optimize"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    per_restart = (deadline - time.time()) / max(n_restarts, 1)
    
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        t_restart = min(time.time() + per_restart, deadline)
        
        # Create perturbation
        scale = 0.02 * (1 + r)  # Increase perturbation with restart number
        noise = np.random.randn(n) * scale
        h_pert = project_feasible(h + noise, n)
        
        # Optimize the perturbed solution
        h_opt, c_opt = lbfgs_optimize(h_pert, n, t_restart, alpha=1000, maxiter=500)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def lse_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def obj_smoothed(x, n, alpha):
    c = compute_c5_vec(x, n)
    return lse_max(c, alpha)

def grad_smoothed(x, n, alpha):
    c = compute_c5_vec(x, n)
    w = lse_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

        return grad_smoothed(x, n, alpha)

        return np.ones(n)

def sqp_lp_step(h, n, active_idx, rho):

def upsample(h, n_old, n_new):

    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
