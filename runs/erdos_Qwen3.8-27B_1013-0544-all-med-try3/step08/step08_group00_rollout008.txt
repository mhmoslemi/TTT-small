import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    k = shift_idx - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    if k >= 0:
        mask1 = j >= k
        mask2 = j < (n - k)
        g[mask1] = (1.0 - h[j[mask1] - k])
        g[mask2] -= h[j[mask2] + k]
    else:
        kk = -k
        mask1 = j >= kk
        mask2 = j < (n - kk)
        g[mask1] = (1.0 - h[j[mask1] - kk])
        g[mask2] -= h[j[mask2] + kk]
    return g * dx

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_fast(x, n, alpha, mu, topk=50):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    # Only use top shifts with significant weight
    if topk is not None and topk < len(w):
        top_order = np.argsort(-w)[:topk]
        w_sub = w[top_order]
        w_sub /= w_sub.sum()
    else:
        top_order = np.arange(len(w))
        w_sub = w
    grad = np.zeros(n)
    dx = 2.0 / n
    j = np.arange(n)
    for ii in range(len(top_order)):
        idx = top_order[ii]
        wm = w_sub[ii] * dx
        k = idx - (n - 1)
        if k >= 0:
            mask1 = j >= k
            mask2 = j < (n - k)
            g = np.zeros(n)
            g[mask1] = (1.0 - x[j[mask1] - k])
            g[mask2] -= x[j[mask2] + k]
        else:
            kk = -k
            mask1 = j >= kk
            mask2 = j < (n - kk)
            g = np.zeros(n)
            g[mask1] = (1.0 - x[j[mask1] - kk])
            g[mask2] -= x[j[mask2] + kk]
        grad += wm * g * (len(w) / len(top_order)) if topk is not None and topk < len(w) else wm * g
    # Rescale
    if topk is not None and topk < len(w):
        grad *= (w_sub.sum() / w.sum())
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def smooth_grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    j = np.arange(n)
    for idx in range(len(w)):
        if w[idx] < 1e-12:
            continue
        wm = w[idx] * dx
        k = idx - (n - 1)
        if k >= 0:
            mask1 = j >= k
            mask2 = j < (n - k)
            g = np.zeros(n)
            g[mask1] = (1.0 - x[j[mask1] - k])
            g[mask2] -= x[j[mask2] + k]
        else:
            kk = -k
            mask1 = j >= kk
            mask2 = j < (n - kk)
            g = np.zeros(n)
            g[mask1] = (1.0 - x[j[mask1] - kk])
            g[mask2] -= x[j[mask2] + kk]
        grad += wm * g
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None, topk=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 20.0), (1000, 100.0), (3000, 500.0), (8000, 2000.0), (20000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            if topk is not None and topk < 2 * n - 1:
                res = minimize(smooth_obj, h, jac=smooth_grad_fast,
                              args=(n, alpha, mu, topk), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-18,
                                       'gtol': 1e-16, 'maxfun': 20000, 'maxls': 40})
            else:
                res = minimize(smooth_obj, h, jac=smooth_grad_full,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-18,
                                       'gtol': 1e-16, 'maxfun': 20000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def subgradient_polish(h, n, deadline, tol=0.002, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.001
    n_active_max = 30
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - tol
        active = np.where(c_vec >= thresh)[0]
        if len(active) > n_active_max:
            top_order = np.argsort(-c_vec[active])[:n_active_max]
            active = active[top_order]
        grad = np.zeros(n)
        n_act = len(active)
        if n_act == 0:
            break
        for idx in active:
            grad += shift_grad_single(h, n, idx)
        grad /= n_act
        # Project gradient onto sum=0 subspace
        grad -= np.mean(grad)
        # Check if we're at a minimum of the max (gradient points outward)
        obj_val = c_vec.max()
        # Adaptive step
        h_trial = project_feasible(h - lr * grad, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-15:
            best_c = c_trial
            best_h = h_trial.copy()
            h = h_trial
            if lr < 0.005:
                lr = min(lr * 1.1, 0.01)
        else:
            lr *= 0.9
            if lr < 1e-8:
                # Try with smaller tol
                tol *= 0.7
                lr = 0.0005
                if tol < 1e-6:
                    break
    return best_h, best_c

def pair_local_search(h, n, deadline, steps=None, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for step in steps:

        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)

                g = np.abs(shift_grad_single(h, n, idx))

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    c_vals = []
    A_ub_list = []
    b_ub_list = []
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub_list.append(np.append(g, -1.0))
        b_ub_list.append(-c_vec[idx])
    n_act = len(active_idx)
    # Bound constraints
    for i in range(n):
        A_ub_list.append(np.append(np.r_[-1.0 + np.zeros(i), -1.0, np.zeros(n - i - 1)], 0.0))
        b_ub_list.append(h[i])
    for i in range(n):
        A_ub_list.append(np.append(np.r_[np.zeros(i), 1.0, np.zeros(n - i - 1)], 0.0))
        b_ub_list.append(1.0 - h[i])
    # Trust region
    for i in range(n):
        A_ub_list.append(np.append(np.r_[np.zeros(i), -1.0, np.zeros(n - i - 1)], 0.0))
        b_ub_list.append(rho)
    for i in range(n):
        A_ub_list.append(np.append(np.r_[np.zeros(i), 1.0, np.zeros(n - i - 1)], 0.0))
        b_ub_list.append(rho)
    total_ineq = len(A_ub_list)
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    for r in range(total_ineq):
        A_ub[r, :] = A_ub_list[r]
        b_ub[r] = b_ub_list[r]
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=2000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h, n_old, n_new):

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
