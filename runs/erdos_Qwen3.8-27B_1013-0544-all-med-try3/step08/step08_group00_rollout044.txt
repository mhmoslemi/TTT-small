    # Compute gradient of each correlation
    # c[k] for k = 0, ..., 2n-2
    # k corresponds to shift (k - (n-1))
    for kk in range(2 * n - 1):
        if w[kk] < 1e-14:
            continue
        k = kk - (n - 1)
        wm = w[kk] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk2 = -k
            nv = n - kk2
            if nv > 0:
                grad[kk2:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk2:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slp_optimize(h, n, deadline, max_iter=200, alpha=5000):
    """Successive Linear Programming: solve LP at each iteration to minimize smoothed max."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rho = 0.02  # trust region radius
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        # Get active constraints (top correlations)
        thresh = m - max(1e-8, 0.005 * m)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > 30:
            top_order = np.argsort(-c_vec[active])
            active = active[top_order[:30]]
        
        # Solve LP: minimize t s.t. c_k + grad_k @ delta <= t, ||delta||_inf <= rho
        # Variables: delta (n), t (1)
        # Constraints:
        #   grad_k @ delta + c_k - t <= 0 for active k
        #   -rho <= delta_i <= rho
        #   sum(delta) = 0 (maintain constraint)
        #   0 <= h_i + delta_i <= 1
        
        from scipy.optimize import linprog
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0  # minimize t
        
        A_ub = []
        b_ub = []
        
        for idx in active:
            k = idx - (n - 1)
            g = np.zeros(n)
            wm = 1.0  # coefficient
            dx_v = 2.0 / n
            if k >= 0:
                nv = n - k
                if nv > 0:
                    g[0:nv] = (1.0 - h[k:n]) * dx_v
                    g[k:n] = (-h[0:nv]) * dx_v
            else:
                kk2 = -k
                nv = n - kk2
                if nv > 0:
                    g[kk2:n] = (1.0 - h[0:nv]) * dx_v
                    g[0:nv] = (-h[kk2:n]) * dx_v
            A_ub.append(np.concatenate([g, [-1.0]]))
            b_ub.append(-c_vec[idx])
        
        # Bound constraints: 0 <= h_i + delta_i <= 1 → -h_i <= delta_i <= 1-h_i
        # And -rho <= delta_i <= rho
        lower = np.maximum(-h, -rho)
        upper = np.minimum(1.0 - h, rho)
        
        A_ub_full = np.array(A_ub) if A_ub else None
        b_ub_full = np.array(b_ub) if b_ub else None
        
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(lower[i], upper[i]) for i in range(n)] + [(float('-inf'), float('inf'))]
        
        try:
            res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full,
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.success:
                delta = res.x[0:n]
                h_new = project_feasible(h + delta, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    if rho < 0.05:
                        rho = min(rho * 1.1, 0.05)
                else:
                    rho *= 0.7
                    if rho < 1e-8:
                        break
        except Exception:
            rho *= 0.5
    
    return best_h, best_c

def binary_local_search(h_bin, n, deadline, rng, max_per_iter=200):
    """Directed binary local search."""
    best_c = compute_c5(h_bin, n)
    h_best = h_bin.copy()
    no_improve = 0
    max_no_improve = 50000
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h_bin, n)
        m = c_vec.max()
        # Find worst lags
        worst_lags = np.argsort(-c_vec)[:5]
        
        improved = False
        for wl in worst_lags:
            if time.time() > deadline:
                break
            k = wl - (n - 1)
            ones_idx = np.where(h_bin == 1)[0]
            zeros_idx = np.where(h_bin == 0)[0]
            
            # Find contributing positions
            if k > 0:
                contrib = [i for i in ones_idx if i + k < n and h_bin[i + k] == 0]
            elif k < 0:
                kk = -k
                contrib = [i for i in ones_idx if i - kk >= 0 and h_bin[i - kk] == 0]
            else:
                contrib = []
            
            # For each contributing position, try a few swaps
            rng.shuffle(contrib)
            for i in contrib[:max_per_iter // 5]:
                if time.time() > deadline:
                    break
                # Try swapping with a few random zeros
                # Prefer zeros that are "far" from other 1s
                for _ in range(3):
                    j = zeros_idx[rng.integers(len(zeros_idx))]
                    if j == i:
                        continue
                    h_bin[i] = 0
                    h_bin[j] = 1
                    new_c = compute_c5(h_bin, n)
                    if new_c < best_c - 1e-15:
                        best_c = new_c
                        h_best = h_bin.copy()
                        no_improve = 0
                        improved = True
                    else:
                        h_bin[i] = 1
                        h_bin[j] = 0
                    if improved:
                        break
            
            if not improved:
                no_improve += 1
        
        if no_improve > max_no_improve:
            # Perturbation: random 3-5 swaps
            ones_idx = np.where(h_bin == 1)[0]
            zeros_idx = np.where(h_bin == 0)[0]
            n_perturb = rng.integers(3, 6)
            for _ in range(n_perturb):
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                i = ones_idx[rng.integers(len(ones_idx))]
                j = zeros_idx[rng.integers(len(zeros_idx))]
                h_bin[i] = 0
                h_bin[j] = 1
                ones_idx = np.where(h_bin == 1)[0]
                zeros_idx = np.where(h_bin == 0)[0]
            no_improve = 0
    
    return h_best, best_c
