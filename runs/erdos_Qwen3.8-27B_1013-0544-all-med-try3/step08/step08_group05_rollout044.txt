    mask = (p4 >= 0) & (p4 < 2*n-1)
    delta[p4[mask]] += h[np.arange(n)[mask]]
    return delta

Wait, p3 = n-1-i+t (where t plays the role of s). Yes, that's the same as n-1+s-i.

This is O(n) per swap evaluation, which is very fast!

Now, the key question: when we swap, the new correlation is corr + delta. We need max(corr + delta) < cur_max.

The max of corr + delta can be computed in O(n) using numpy. So total per iteration: O(n).

With n=250, that's very fast. I can do millions of iterations.

Let me now design the full algorithm:

1. Start from initial_h_values, make it binary (round to 0/1 with correct count)
2. Simulated annealing on binary solutions:
   - Temperature schedule: start high, geometrically cool
   - At each step: pick random (i,j), compute delta, evaluate new max
   - Accept if improvement, or accept worse with prob exp(-Δ/T)
3. SLSQP on the continuous relaxation
4. Try multiple resolutions
5. Inter-residue refinement

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    """Round h to binary with exactly n//2 ones"""
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def compute_swap_delta(i, j, h, n):
    delta = np.zeros(2 * n - 1)
    t = np.arange(n)
    delta[n - 1 + i - t] += (1.0 - h)
    delta[n - 1 + j - t] -= (1.0 - h)
    delta[n - 1 - i + t] -= h
    delta[n - 1 - j + t] += h
    return delta

def sim_anneal_binary(h_init, n, t_start, t_end, deadline, max_iters, rng, verbose=False):
    """Simulated annealing on binary solutions"""
    h = make_binary(h_init, n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    n_iter = 0
    t_current = t_start
    cooling = (t_end / t_start) ** (1.0 / max(max_iters, 1))
    
    # Track best few max positions for faster evaluation
    while n_iter < max_iters and time.time() < deadline:
        # Pick random swap
        i = rng.integers(len(zeros))
        j = rng.integers(len(ones))
        i = zeros[i]
        j = ones[j]
        
        delta = compute_swap_delta(i, j, h, n)
        new_corr = corr + delta
        new_max = new_corr.max()
        
        delta_obj = new_max - cur_max
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(t_current, 1e-10)):
            # Accept
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_max = new_max
            # Update ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        
        n_iter += 1
        t_current *= cooling
        if t_current < t_end:
            t_current = t_end
    
    return best_h, best_max

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
