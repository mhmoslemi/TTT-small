def logsumexp_alpha(all_c, alpha):

    c_max = all_c.max()
    if alpha * (all_c - c_max).max() > 500:

        sh = np.where(all_c - c_max > -1000/alpha, alpha * (all_c - c_max), -500)

        return c_max + np.log(np.sum(np.exp(sh))) / alpha

    val = logsumexp_alpha(c, alpha)
    if mu > 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-15:
                break
            h += (diff / room.sum()) * room
        else:
            avail = h
            if avail.sum() < 1e-15:
                break
            h -= ((-diff) / avail.sum()) * avail
    return np.clip(h, 0.0, 1.0)

def compute_weights_all(c, alpha):
    c_max = c.max()
    sh = alpha * (c - c_max)
    sh_max = sh.max()
    if sh_max > 500:
        w = np.exp(np.maximum(sh - sh_max, -500))
    else:
        w = np.exp(sh - sh_max)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c_max == c] = 1.0 / (c_max == c).sum()
    else:
        w /= s
    return w

def compute_grad_from_w(h, n, w):

    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-16:

        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        cw[:n - k] += wk * h[k:n]
        bw[k:] += wk * h[:n - k]
    grad[0:n-1] += dx * ((w[1:].sum() - w[n - 1] * (n - 1)) - cw[0:n-1])
    # Actually let me redo this more carefully
    # grad[j] = dx * (w0*(1-2h[j]) + A[j] - B[j])
    # A[j] = sum_{k=1}^{n-1-j} w[n-1+k] * (1-h[j+k])
    # B[j] = sum_{k=1}^{j} w[n-1+k] * h[j-k]
    # = sum_{k=1}^{n-1-j} w[n-1+k] - sum_{k=1}^{n-1-j} w[n-1+k]*h[j+k] - sum_{k=1}^{j} w[n-1+k]*h[j-k]

    for j in range(n):

        if idx > 0:

    c_max = c.max()
    sh = alpha * (c - c_max)
    sh_max = sh.max()
    if sh_max > 500:
        val = c_max + (500.0 + np.log(np.sum(np.exp(np.maximum(sh - sh_max, -500) - 500)))) / alpha
    else:
        val = c_max + np.log(np.sum(np.exp(sh - sh_max))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

    return float(val)

def topshift_grad(h, n, shift_idx, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[shift_idx]
    v_max = vals.max()
    sh = alpha * (vals - v_max)
    sh_max = sh.max()
    w_local = np.exp(np.maximum(sh - sh_max, -500))
    s = w_local.sum()
    if s < 1e-300:
        w_local = np.zeros_like(vals)
        w_local[vals == v_max] = 1.0 / (vals == v_max).sum()
    else:
        w_local /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(len(shift_idx)):
        if w_local[ii] < 1e-15:
            continue
        si = shift_idx[ii]
        k = si - (n - 1)
        wm = w_local[ii] * dx
        if k == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif k > 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def slqp_obj_fn(h, n, alpha):

    c_max = c.max()
    sh = alpha * (c - c_max)
    sh_max = sh.max()

    else:
        return float(c_max + np.log(np.sum(np.exp(sh - sh_max))) / alpha)

def slqp_sum_fn(h, n):
    return float(h.sum() - n / 2.0)

def slqp_sum_jac(h, n):
    return np.ones(n)

def run_lbfgs_obj(h, n, deadline, obj_func, grad_func, obj_args, grad_args, maxiter, ftol):

def fine_pair_search(h, n, deadline, center_radius=40):
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    active = np.where(c_vec >= max_val - 0.01 * max_val)[0]
    
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Find active shifts affected by h[i]
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def gradient_descent_steps(h, n, deadline, n_steps=2000):
    best_c = compute_c5(h, n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    si = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        w = compute_weights_all(c_vec, 5000)
        grad = compute_grad_from_w(h, n, w)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        # Try moving in -grad direction with current step size
        step_size = step_sizes[min(si, len(step_sizes) - 1)]
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            h = h_new
            si = max(0, si - 1)
        else:
            si = min(si + 1, len(step_sizes) - 1)
    return h, best_c
