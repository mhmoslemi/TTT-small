import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = np.max(sh)
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = np.max(sh)
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w

def grad_smooth_max(h, n, alpha):

    dx = 2.0 / n
    grad = np.zeros(n)

        if k >= 0:

    return float(smooth_max_val(c, alpha))

def interpolate_to_n(h_old, n_old, n_new):
