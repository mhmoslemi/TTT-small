def frank_wolfe(h_init, n, deadline, alpha=1000, max_iter=10000):
    h = h_init.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    step_t = 0
    for t in range(max_iter):
        if time.time() > deadline:
            break
        step_t = 2.0 / (t + 2)
        
        # Compute subgradient of max(c)
        g = compute_corr(h, n)
        k_star = np.argmax(g)
        
        # Subgradient: ∂c[k*]/∂h[m] = (1-h[m+k*-(n-1)]) - h[m-k*+(n-1)]
        grad = np.zeros(n)
        for m in range(n):
            # First term: (1-h[m+k*-(n-1)]) if 0 ≤ m+k*-(n-1) < n
            j1 = m + k_star - (n-1)
            if 0 <= j1 < n:
                grad[m] += (1 - h[j1])
            # Second term: -h[m-k*+(n-1)] if 0 ≤ m-k*+(n-1) < n
            j2 = m - k_star + (n-1)
            if 0 <= j2 < n:
                grad[m] -= h[j2]
        
        # Linear minimization: min grad·h s.t. sum(h)=n/2, 0≤h≤1
        # Solution: sort grad, set top n/2 to 0, bottom n/2 to 1 (minimize dot product)
        # Wait: we want to MINIMIZE grad·h. So set h=1 where grad is most negative, h=0 where grad is most positive.
        # With sum(h) = n/2: set the n/2 smallest grad values to 1, rest to 0.
        order = np.argsort(grad)
        v = np.zeros(n)
        v[order[:n//2]] = 1.0
        
        # Update
        h = (1 - step_t) * h + step_t * v
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def sa_binary(h_init, n, deadline, seed=42, t_init=0.05, t_min=0.001):
    rng = np.random.RandomState(seed)
    h = binarize(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    it = 0
    while time.time() < deadline:
        # Adaptive temperature
        frac = min(it / 1000000.0, 1.0)
        T = t_init * (t_min / t_init) ** frac
        if T < 1e-6:
            T = 1e-6
        
        it += 1
        # Random swap
        ii = rng.randint(len(ones_list))
        jj = rng.randint(len(zeros_list))
        i = ones_list[ii]
        j = zeros_list[jj]
        
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h, n)
        
        delta = c_new - best_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # Accept
            ones_list[ii] = j
            zeros_list[jj] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # Reject
            h[j] = 0.0
            h[i] = 1.0
    
    return best_h, best_c

def local_search_binary(h, n, deadline, max_sweeps=1000):
    """Steepest descent: try all swaps, take best improvement per sweep."""
    best_c = compute_c5(h, n)
    ones_pos = list(np.where(h == 1)[0])
    zeros_pos = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_delta = 0
        best_i = -1
        best_j = -1
        
        for i in ones_pos:
            for j in zeros_pos:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                delta = best_c - c
                if delta > best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
                h[j] = 0.0
                h[i] = 1.0
        
        if best_delta > 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            best_c -= best_delta
            ones_pos.remove(best_i)
            ones_pos.append(best_j)
            zeros_pos.remove(best_j)
            zeros_pos.append(best_i)
        else:
            break
    
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha_list, maxiter_each=1000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            sm = shifted.max()
            return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        
        def cons_fun(x):
            return x.sum() - n / 2.0
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]
        
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def subgrad_polish(h, n, deadline, max_iter=10000):
    """Subgradient method for max correlation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    n_corr = 2 * n - 1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full')
        m = g.max()
        k_star = int(np.argmax(g))
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j1 = i + k_star - (n - 1)
            if 0 <= j1 < n:
                grad[i] += (1.0 - h[j1])
            j2 = i - k_star + (n - 1)
            if 0 <= j2 < n:
                grad[i] -= h[j2]
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        elif c_new > m + lr * 0.1:
            lr *= 0.9
        else:
            lr = min(lr * 1.05, 0.1)
        
        h = h_new  # always accept for exploration? or only if improves?
        # Actually for subgradient, we should always take the step
    
    return best_h, best_c
