def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad_h(c, h, n, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n1 = n - 1
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - n1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] -= wm * h[0:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] -= wm * h[kk:n]
    return grad

def lag_grad_single(h, n, lag_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n])
        grad[k:n] -= h[0:nv]
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv])
        grad[0:nv] -= h[kk:n]
    return grad * dx

def solve_sqp_lp(h, n, active_lags, c_vals, rho):
    nvar = n + 1
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0
    n_act = len(active_lags)
    if n_act == 0:
        return None
    A_ub_rows = []
    b_ub_vals = []
    for idx, ci in zip(active_lags, c_vals):
        g = lag_grad_single(h, n, idx)
        row = np.zeros(nvar)
        row[0:n] = g
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(-ci)
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    A_eq = np.zeros((1, nvar))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        if lo > hi:
            lo = 0.0
            hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n]
    except Exception:
        pass
    return None

def lbfgs_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smooth_max_grad_h(c, x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def sqp_loop(h, n, deadline, rho_init=0.05, max_active=30, max_no_improve=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    it = 0
    while time.time() < deadline and it < 5000:
        it += 1
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 3e-5 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            break
        if len(active) > max_active:
            top_order = np.argsort(-c_vec[active])
            active = active[top_order[:max_active]]
        c_act = c_vec[active]
        delta = solve_sqp_lp(h, n, active.tolist(), c_act, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-9:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.3, 0.12)
        else:
            no_improve += 1
            rho *= 0.6
            if rho < 1e-10 or no_improve > max_no_improve:
                break
    return best_h, best_c

def pairwise_search(h, n, deadline, base_radius=50, max_pairs=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 2e-4 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            w_act = c_vec[active]
            exponents = 200.0 * (w_act - w_act.max())
            w_act = np.exp(exponents)
            w_act /= w_act.sum()
            grad = np.zeros(n)
            for idx, wi in zip(active, w_act):
                grad += wi * lag_grad_single(h, n, int(idx))
            top_pos = np.argsort(-grad)[:base_rad
                ius]
            top_neg = np.argsort(grad)[:base_radius]
            candidates = []
            for i_val in top_pos[:40]:
                for j_val in top_neg[:40]:
                    i = int(i_val)
                    j = int(j_val)
                    if i != j:
                        candidates.append((i, j))
            np.random.shuffle(candidates)
            for i, j in candidates[:max_pairs]:
                if time.time() > deadline:
                    return best_h, best_c
                if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                    h[i] -= step
                    h[j] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += step
                        h[j] -= step
                if h[j] - step >= -1e-12 and h[i] + step <= 1.0 + 1e-12:
                    h[j] -= step
                    h[i] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[j] += step
                        h[i] -= step
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    h_parent = project_feasible(h_parent, n_parent)
    c_parent = compute_c5(h_parent, n_parent)

    best_h = h_parent.copy()
    best_c = c_parent
    best_n = n_parent

    # Phase 1: Upsample to n=2000
    n_work = 2000
    h = upsample_func(h_parent, n_parent, n_work)
    c_h = compute_c5(h, n_work)
    if c_h < best_c:
        best_c = c_h
        best_h = h.copy()
        best_n = n_work

    # Phase 2: L-BFGS with moderate smoothness (50s)
    t2 = min(t0 + 50, deadline - 500)
    if t2 > time.time() + 5:
        bounds_w = [(0.0, 1.0)] * n_work
        try:
            res = minimize(lbfgs_smooth, h, jac=lbfgs_smooth_grad,
                          args=(n_work, 3000.0, 100.0),
                          method='L-BFGS-B', bounds=bounds_w,
                          options={'maxiter': 5000, 'ftol': 1e-19,
                                   'gtol': 1e-16, 'maxfun': 30000, 'maxls': 30})
            h_new = project_feasible(res.x, n_work)
            c_new = compute_c5(h_new, n_work)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    # Phase 3: L-BFGS with higher smoothness (40s)
    t3 = min(t0 + 90, deadline - 450)
    if t3 > time.time() + 5:
        try:
            res = minimize(lbfgs_smooth, h, jac=lbfgs_smooth_grad,
                          args=(n_work, 20000.0, 500.0),
                          method='L-BFGS-B', bounds=bounds_w,
                          options={'maxiter': 5000, 'ftol': 1e-19,
                                   'gtol': 1e-17, 'maxfun': 20000, 'maxls': 30})
            h_new = project_feasible(res.x, n_work)
            c_new = compute_c5(h_new, n_work)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    # Phase 4: SQP main phase (350s)
    t4 = min(t0 + 440, deadline - 200)
    if t4 > time.time() + 10:
        h = best_h.copy()
        h, c = sqp_loop(h, n_work, t4, rho_init=0.06, max_active=35, max_no_improve=40)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = n_work

    # Phase 5: Another L-BFGS round (60s)
    t5 = min(time.time() + 60, deadline - 150)
    if t5 > time.time() + 5:
        try:
            res = minimize(lbfgs_smooth, h, jac=lbfgs_smooth_grad,
                          args=(n_work, 100000.0, 2000.0),
                          method='L-BFGS-B', bounds=bounds_w,
                          options={'maxiter': 5000, 'ftol': 1e-19,
                                   'gtol': 1e-18, 'maxfun': 15000, 'maxls': 30})
            h_new = project_feasible(res.x, n_work)
            c_new = compute_c5(h_new, n_work)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                best_n = n_work
        except Exception:
            pass

    # Phase 6: Pairwise coordinate search (120s)
    t6 = min(time.time() + 120, deadline - 80)
    if t6 > time.time() + 5:
        h = best_h.copy()
        h, c = pairwise_search(h, n_work, t6, base_radius=40, max_pairs=300)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = n_work

    # Phase 7: Final SQP polish (60s)
    t7 = min(time.time() + 60, deadline - 20)
    if t7 > time.time() + 5:
        h = best_h.copy()
        h, c = sqp_loop(h, n_work, t7, rho_init=0.02, max_active=20, max_no_improve=20)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = n_work

    # Phase 8: Final tiny pairwise (20s)
    t8 = min(time.time() + 20, deadline - 5)
    if t8 > time.time() + 3:
        h = best_h.copy()
        h, c = pairwise_search(h, n_work, t8, base_radius=25, max_pairs=150)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = n_work

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
