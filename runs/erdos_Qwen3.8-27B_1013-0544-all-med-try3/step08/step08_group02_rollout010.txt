def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_wts(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[m == c.max()] = 1.0 / np.sum(c == c.max())
        return w
    return w / s

def grad_corr(x, n):

            g[0:nv] += (1.0 - x[k:n]) * dx
            g[k:n] += (-x[0:nv]) * dx
        else:

            nv = n - kk
            g[kk:n] += (1.0 - x[0:nv]) * dx

def pair_descent(h, n, deadline, radius=50, deltas=None):

        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)

                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:

                        h[j] += d
                        c = compute_c5(h, n)

                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d

def periodic_init(n, period, frac_on):
    h = np.zeros(n)
    on = int(round(period * frac_on))

        on = 1
    if on >= period:

        h[i::period] = 1.0
    target = n / 2.0

    if diff > 0:

            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
    else:

            sub = min(h[i], -diff)
            h[i] -= sub
            diff += sub
    return project_feasible(h, n)

    if time.time() < p3_end - 30:
        h3, c3 = run_lbfgs(best_h, best_n, p3_end,
                           alphas=[20, 100, 500, 2000, 10000, 50000,
                                   200000, 1000000],
                           mu=0.005,
                           maxit_per=[800, 1000, 1500, 2000, 2500,
                                       2500, 2000, 1500])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_main

    # Phase 4: SLP refinement
    p4_end = min(time.time() + 120, deadline - 200)
    if time.time() < p4_end - 10:
        h4, c4 = slp_refine(best_h, best_n, p4_end, max_iters=3000, top_k=5)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_main

    # Phase 5: Active set L-BFGS
    p5_end = min(time.time() + 120, deadline - 150)
    if time.time() < p5_end - 20:
        h5, c5_val = run_active_lbfgs(best_h, best_n, p5_end, top_k=4,
                                      alphas=[50000, 200000, 1000000, 5000000])
        if c5_val < best_c:
            best_c = c5_val
            best_h = h5
            best_n = n_main

    # Phase 6: Subgradient with momentum
    p6_end = min(time.time() + 80, deadline - 120)
    if time.time() < p6_end - 5:
        h6, c6 = subgrad_momentum(best_h, best_n, p6_end, n_iter=10000, eta0=0.002)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = n_main

    # Phase 7: SLP again after subgradient
    p7_end = min(time.time() + 80, deadline - 80)
    if time.time() < p7_end - 10:
        h7, c7 = slp_refine(best_h, best_n, p7_end, max_iters=2000, top_k=4)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = n_main

    # Phase 8: Try periodic restart at high res
    p8_end = min(time.time() + 100, deadline - 60)
    if time.time() < p8_end - 30:
        n_per = best_n
        candidates = [
            periodic_init(n_per, 10, 0.5),
            periodic_init(n_per, 20, 0.5),
            periodic_init(n_per, 50, 0.5),
            periodic_init(n_per, 100, 0.5),
        ]
        for cand in candidates:
            if time.time() > p8_end - 25:
                break
            cc = compute_c5(cand, n_per)
            if cc < best_c:
                best_c = cc
                best_h = cand.copy()
                best_n = n_per
            elif cc < best_c + 0.01:
                h8, c8 = run_lbfgs(cand, n_per, min(time.time() + 30, p8_end - 10),
                                   alphas=[100, 1000, 10000, 100000],
                                   mu=0.01, maxit_per=[500, 1000, 1000, 1000])
                if c8 < best_c:
                    best_c = c8
                    best_h = h8
                    best_n = n_per

    # Phase 9: Final SLP + active set
    p9_end = min(time.time() + 60, deadline - 30)
    if time.time() < p9_end - 10:
        h9, c9 = slp_refine(best_h, best_n, min(p9_end, time.time() + 30),
                            max_iters=1500, top_k=3)
        if c9 < best_c:
            best_c = c9
            best_h = h9
            best_n = n_main

    # Phase 10: Pair descent fine-tuning
    p10_end = deadline - 5
    if time.time() < p10_end - 10:
        radius = max(10, best_n // 50)
        h10, c10 = pair_descent(best_h, best_n, p10_end, radius=radius,
                                deltas=[0.0005, 0.0002, 0.0001, 0.00005])
        if c10 < best_c:
            best_c = c10
            best_h = h10
            best_n = n_main

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
