    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()

    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:

    return float(val)

def active_grad(x, n, active, alpha, mu):

    vals = c[active]
    mx = vals.max()

    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue

        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def pair_search(h, n, deadline, radius=50, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    break
                # Try moving mass from i to nearby j
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # i gains d, j loses d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            # Don't save copy every time, just continue
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # i loses d, j gains d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def binary_local_search(h, n, deadline, max_rounds=50):
    """Simulated annealing on binary swaps."""
    h_bin = np.zeros(n)
    # Round to binary
    threshold = np.sort(h)[n//2]  # median
    h_bin[h >= threshold] = 1.0
    # Fix the count
    n_ones = h_bin.sum()
    target = n // 2
    if n_ones > target:
        # Remove excess ones
        ones_idx = np.where(h_bin == 1.0)[0]
        rng = np.random.RandomState(42)
        remove = rng.choice(ones_idx, int(n_ones - target), replace=False)
        h_bin[remove] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0.0)[0]
        rng = np.random.RandomState(42)
        add = rng.choice(zeros_idx, int(target - n_ones), replace=False)
        h_bin[add] = 1.0
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    rng = np.random.RandomState(42)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones_idx = np.where(h_bin == 1.0)[0]
        zeros_idx = np.where(h_bin == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        n_trials = max(1000, int((deadline - time.time()) * 2))
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            i = rng.integers(len(ones_idx))
            j = rng.integers(len(zeros_idx))
            a = ones_idx[i]
            b = zeros_idx[j]
            h_bin[a] = 0.0
            h_bin[b] = 1.0
            c = compute_c5(h_bin, n)
            if c < best_c:
                best_c = c
                best_h = h_bin.copy()
            else:
                h_bin[a] = 1.0
                h_bin[b] = 0.0
    
    return best_h, best_c

def golay_like_start(n):
    """Create a Golay-like initial pattern."""
    # Start with a m-sequence or Gold code pattern
    h = np.zeros(n)
    # Use a simple periodic pattern with good autocorrelation properties
    # Period of 5: 1,1,0,1,0 (has low peak sidelobe)
    pattern = np.array([1, 1, 0, 1, 0, 0, 1, 0, 1, 1])  # length 10
    h[:] = np.tile(pattern, n // len(pattern) + 1)[:n]
    # Adjust to get exactly n/2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        excess = n_ones - target
        ones_pos = np.where(h == 1.0)[0]
        rng = np.random.RandomState(0)
        to_remove = rng.choice(ones_pos, excess, replace=False)
        h[to_remove] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        zero_pos = np.where(h == 0.0)[0]
        rng = np.random.RandomState(0)
        to_add = rng.choice(zero_pos, deficit, replace=False)
        h[to_add] = 1.0
    return project_feasible(h, n)

def subgrad_step(h, n, active, eta, dx):
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    grad /= len(active)
    return h - eta * grad

def subgrad_refine(h, n, deadline, n_iter=10000):

            pass
    return best_h, best_c

def active_refine(h, n, deadline, top_k, alphas):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(50):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0008 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(400, int((deadline - time.time()) * 3))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.003),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def pair_search(h, n, deadline, radius, deltas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline - 2:
            break
        improved = True
        passes = 0
        while improved and time.time() < deadline - 2 and passes < 20:
            improved = False
            passes += 1
            for i in range(n):
                if time.time() > deadline - 2:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 + 1e-15 and best_h[j] - d >= -1e-15:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= -1e-15 and best_h[j] + d <= 1.0 + 1e-15:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline - 1:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-11, 0.0004 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.008, 0.015)
        else:
            eta *= 0.985
        h = h_new
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha, maxiter):
    def sq_fun(x, nn):
        return x.sum() - nn / 2.0
    def sq_jac(x, nn):
        return np.ones(nn)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(sm_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick L-BFGS at original resolution to get a good starting shape
    p1_end = min(t0 + 100, deadline - 500)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 50000],
        mu=0.01, maxiters=[1000, 1500, 2000, 2000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=1000, thorough L-BFGS
    n2 = 1000
    p2_end = min(time.time() + 380, deadline - 280)
    h_fine = upsample(best_h, best_n, n2)
    h2, c2 = run_lbfgs(h_fine, n2, p2_end,
        alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.005, maxiters=[800, 1200, 1500, 1500, 2000, 2000, 2500, 2500, 3000])
    if c2 < best_c:
        best_c = c2
        best_h = h2
        best_n = n2

    # Phase 3: Active-set refinement at n=1000
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 120, deadline - 160)
        h3, c3 = active_refine(best_h, best_n, p3_end, top_k=4,
            alphas=[20000, 100000, 500000, 2000000, 10000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Subgradient polish
    if time.time() < deadline - 100:
        p4_end = min(time.time() + 50, deadline - 60)
        h4, c4 = subgrad_refine(best_h, best_n, p4_end, n_iter=30000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: SLSQP polish
    if time.time() < deadline - 60:
        for alpha in [100000, 500000]:
            if time.time() > deadline - 40:
                break
            p5_end = min(time.time() + 25, deadline - 30)
            h5, c5 = slqp_polish(best_h, best_n, p5_end, alpha, 800)
            if c5 < best_c:
                best_c = c5
                best_h = h5

    # Phase 6: Pair search at current resolution
    if time.time() < deadline - 25:
        p6_end = min(time.time() + 35, deadline - 10)
        h6, c6 = pair_search(best_h, best_n, p6_end, radius=35,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try n=2000 if time remains
    n3 = 2000
    remaining = deadline - time.time()
    if remaining > 80:
        h_fine3 = upsample(best_h, best_n, n3)
        p7_end = min(time.time() + min(55, int(remaining * 0.5)), deadline - 20)
        h7, c7 = run_lbfgs(h_fine3, n3, p7_end,
            alphas=[30, 300, 3000, 30000],
            mu=0.003, maxiters=[1000, 1500, 2000, 2000])
        if time.time() < deadline - 5:
            h7b, c7b = pair_search(h7, n3, deadline - 3, radius=25,
                deltas=[0.001, 0.0005, 0.0002])
            if c7b < c7:
                c7 = c7b
                h7 = h7b
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = n3
    elif remaining > 40:
        # Shorter n=2000 attempt
        h_fine3 = upsample(best_h, best_n, n3)
        p7_end = min(time.time() + 25, deadline - 5)
        h7, c7 = run_lbfgs(h_fine3, n3, p7_end,
            alphas=[100, 2000],
            mu=0.003, maxiters=[1000, 1500])
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = n3

    # Phase 8: Final micro pair search
    if time.time() < deadline - 2:
        h8, c8 = pair_search(best_h, best_n, deadline - 1, radius=20,
            deltas=[0.0001, 0.00005, 0.00002])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection and verification
    fn = len(best_h)
    best_h = project_feasible(best_h, fn)
    best_c = compute_c5(best_h, fn)

    return best_h, best_c, fn
