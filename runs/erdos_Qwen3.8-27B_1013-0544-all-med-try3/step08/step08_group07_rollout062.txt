        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_f(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

    except Exception:
        return h

def grad_c_at_k(h, k, n):
    """Gradient of c[k] w.r.t. h."""
    dx = 2.0 / n
    s = k - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        js = j - s
        if 0 <= js < n:
            grad[j] += (1.0 - h[js])
        jp = j + s
        if 0 <= jp < n:
            grad[j] += (-h[jp])
    grad *= dx
    return grad

def targeted_opt(h, n, deadline, max_rounds=20000):
    """Gradient descent targeting the worst shift(s)."""
    h = h.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    g = compute_g(h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        m = g.max()
        k_max = int(np.argmax(g))
        
        # Use weighted combination of gradients for top-3 shifts
        top_ks = np.argsort(g)[::-1][:3]
        weights = g[top_ks] - m + 0.1  # small bias to include second/third
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_ks):
            grad += w * grad_c_at_k(h, k, n)
        
        # Project to maintain sum
        grad -= grad.mean()
        
        # Line search
        found = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                g = compute_g(h)
                found = True
                break
        if not found:
            # Try a smaller step or break
            break
    
    return best_h, best_c

    p1_end = t0 + 0.35 * (total - 5)
    h_work = h_best.copy()
    h_work, c_w = multi_alpha_slsqp(h_work, n_best, p1_end,
                                     [500, 2000, 10000, 50000, 200000, 1000000, 5000000], 3000)
    if c_w < best_c:
        best_c = c_w
        h_best = h_work.copy()

    # Phase 2: Targeted gradient descent (20% budget)
    p2_end = t0 + 0.55 * (total - 5)
    if time.time() < p2_end - 5:
        h_t, c_t = targeted_descent(h_best.copy(), n_best, p2_end, max_rounds=50000)
        if c_t < best_c:
            best_c = c_t
            h_best = h_t.copy()

    # Phase 3: Pair moves (15% budget)
    p3_end = t0 + 0.70 * (total - 5)
    if time.time() < p3_end - 5:
        h_p, c_p = pair_search(h_best.copy(), n_best, p3_end, max_rounds=200000)
        if c_p < best_c:
            best_c = c_p
            h_best = h_p.copy()

    # Phase 4: Try binary + relaxation (10% budget)
    p4_end = t0 + 0.80 * (total - 5)
    if time.time() < p4_end - 10:
        # Make a binary version
        h_bin = np.zeros(n_best)
        idx_sorted = np.argsort(h_best)[::-1]
        target = int(n_best / 2)
        h_bin[idx_sorted[:target]] = 1.0
        # Relax
        h_rel, c_rel = binary_relax(h_bin, n_best, p4_end)
        if c_rel < best_c:
            best_c = c_rel
            h_best = h_rel.copy()

    # Phase 5: Different resolutions (15% budget)
    p5_end = dl - 15
    if time.time() < p5_end - 20:
        for n_try in [249, 251, 252, 254, 200, 300]:
            if time.time() > p5_end - 15:
                break
            h_try = project_feasible(interpolate_h(h_best, n_best, n_try), n_try)
            h_try, c_try = multi_alpha_slsqp(h_try, n_try, min(p5_end - 10, time.time() + 120),
                                              [5000, 50000, 500000], 2000)
            if time.time() < p5_end - 10:
                h_try2, c_try2 = targeted_descent(h_try, n_try, p5_end - 5, max_rounds=5000)
                if c_try2 < c_try:
                    h_try, c_try = h_try2, c_try2
            if c_try < best_c:
                best_c = c_try
                h_best = h_try.copy()
                n_best = n_try

    # Phase 6: Final SLSQP + targeted polish
    if time.time() < dl - 8:
        h_final = slsqp_opt(h_best, n_best, 20000000, 1500)
        c_final = compute_c5(h_final)
        if c_final < best_c:
            best_c = c_final
            h_best = h_final.copy()

    if time.time() < dl - 3:
        h_fin2, c_fin2 = targeted_descent(h_best, n_best, dl - 2, max_rounds=10000)
        if c_fin2 < best_c:
            best_c = c_fin2
            h_best = h_fin2.copy()

    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best)
    return h_best, best_c, n_best
