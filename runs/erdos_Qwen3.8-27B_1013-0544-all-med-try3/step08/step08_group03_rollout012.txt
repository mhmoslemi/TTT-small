import numpy as np
import time
import math
from scipy.optimize import minimize


def _sum_clipped(x, lam):
    return float(np.sum(np.clip(x + lam, 0.0, 1.0)))


def project_feasible(x, n):
    x = np.asarray(x, dtype=float).copy()
    target = 0.5 * float(n)
    x = np.clip(x, 0.0, 1.0)

    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x

    lo = -1.0
    hi = 1.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        if _sum_clipped(x, mid) < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(120):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-13:
            break
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        x[mask] += diff / float(cnt)
        x = np.clip(x, 0.0, 1.0)

    diff = target - float(np.sum(x))
    if abs(diff) > 1e-15:
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        cnt = int(np.count_nonzero(mask))
        if cnt:
            x[mask] += diff / float(cnt)
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5_val(h, n, hm1=None):
    if hm1 is None:
        hm1 = 1.0 - h
    return float(np.max(np.correlate(h, hm1, mode="full")) * (2.0 / float(n)))


def _c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def _smooth_max_val_and_weights(cvec, alpha):
    cvec = np.asarray(cvec, dtype=float)
    m = float(np.max(cvec))
    z = alpha * (cvec - m)
    zmax = float(np.max(z))
    z = z - zmax
    ez = np.exp(z)
    s = float(np.sum(ez))
    if s <= 1e-300:
        w = np.zeros_like(cvec)
        idx = int(np.argmax(cvec))
        w[idx] = 1.0
        return m, w
    val = m + float(np.log(s)) / alpha
    return val, ez / s


def _grad_raw_from_weights(h, n, w):
    grad = np.zeros(n, dtype=float)
    L = n - 1
    hlocal = h
    for p in range(2 * n - 1):
        wm = float(w[p])
        if wm <= 1e-15:
            continue
        k = p - L
        if k >= 0:
            m = n - k
            if m <= 0:
                continue
            grad[k:n] += wm * (1.0 - hlocal[:m])
            grad[:m] -= wm * hlocal[k:]
        else:
            kk = -k
            m = n - kk
            if m <= 0:
                continue
            grad[:m] += wm * (1.0 - hlocal[kk:n])
            grad[kk:n] -= wm * hlocal[:m]
    return grad


def _penalty_obj(x, n, alpha, mu):
    cvec = np.correlate(x, 1.0 - x, mode="full") * (2.0 / float(n))
    cvec = np.nan_to_num(cvec, nan=0.0, posinf=0.0, neginf=0.0)
    val, _ = _smooth_max_val_and_weights(cvec, alpha)
    sdiff = float(np.sum(x)) - 0.5 * float(n)
    return float(val + mu * sdiff * sdiff)


def _penalty_grad(x, n, alpha, mu):
    cvec = np.correlate(x, 1.0 - x, mode="full") * (2.0 / float(n))
    cvec = np.nan_to_num(cvec, nan=0.0, posinf=0.0, neginf=0.0)
    _, w = _smooth_max_val_and_weights(cvec, alpha)
    grad = _grad_raw_from_weights(x, n, w) * (2.0 / float(n))
    sdiff = float(np.sum(x)) - 0.5 * float(n)
    grad += 2.0 * mu * sdiff
    return grad


def _lbfgs_polish(h, n, deadline, maxiter=80):
    h = project_feasible(h, n)
    best_c = _c5_val(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0) for _ in range(n)]

    configs = [
        (10.0, 1.0),
        (200.0, 5.0),
        (2000.0, 50.0),
        (20000.0, 300.0),
        (100000.0, 1200.0),
    ]

    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(
                _penalty_obj,
                h,
                jac=_penalty_grad,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-16,
                    "gtol": 1e-12,
                    "maxfun": int(maxiter) * 8,
                    "maxls": 30,
                },
            )
            xnew = getattr(res, "x", None)
            if xnew is None:
                continue
            hnew = project_feasible(xnew, n)
            cnew = _c5_val(hnew, n)
            if cnew < best_c - 1e-15:
                best_c = cnew
                best_h = hnew.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c = _c5_val(best_h, n)
    return best_h, best_c


def _slsqp_obj(x, n, alpha):
    cvec = np.correlate(x, 1.0 - x, mode="full") * (2.0 / float(n))
    cvec = np.nan_to_num(cvec, nan=0.0, posinf=0.0, neginf=0.0)
    val, _ = _smooth_max_val_and_weights(cvec, alpha)
    return float(val)


def _slsqp_grad(x, n, alpha):
    cvec = np.correlate(x, 1.0 - x, mode="full") * (2.0 / float(n))
    cvec = np.nan_to_num(cvec, nan=0.0, posinf=0.0, neginf=0.0)
    _, w = _smooth_max_val_and_weights(cvec, alpha)
    return _grad_raw_from_weights(x, n, w) * (2.0 / float(n))


def _slsqp_constraint(x, n):
    return float(np.sum(x)) - 0.5 * float(n)


def _slsqp_jac(x, n):
    return np.ones(n, dtype=float)


def _slsqp_polish(h, n, deadline, alpha=30000.0, maxiter=150):
    if n > 400 or time.time() > deadline - 25:
        return project_feasible(h, n), _c5_val(project_feasible(h, n), n)

    h = project_feasible(h, n)
    best_c = _c5_val(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [{
        "type": "eq",
        "fun": _slsqp_constraint,
        "jac": _slsqp_jac,
        "args": (n,),
    }]

    try:
        res = minimize(
            _slsqp_obj,
            h,
            jac=_slsqp_grad,
            args=(n, alpha),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-15, "disp": 0},
        )
        xnew = getattr(res, "x", None)
        if xnew is not None:
            hnew = project_feasible(xnew, n)
            cnew = _c5_val(hnew, n)
            if cnew < best_c - 1e-15:
                best_c = cnew
                best_h = hnew.copy()
    except Exception:
        pass

    best_h = project_feasible(best_h, n)
    best_c = _c5_val(best_h, n)
    return best_h, best_c


def _pair_scan(h, n, deadline, initial_step=0.01, min_step=1e-6, alpha=200.0, max_passes=35):
    h = project_feasible(h, n)
    hm1 = 1.0 - h
    dx = 2.0 / float(n)

    cmax = float(np.max(np.correlate(h, hm1, mode="full")) * dx)
    best_c = cmax
    best_h = h.copy()

    step = float(initial_step)
    a_cur = float(alpha)

    for _pass in range(int(max_passes)):
        if time.time() > deadline or step < min_step:
            break

        cvec = np.correlate(h, hm1, mode="full") * dx
        cvec = np.nan_to_num(cvec, nan=0.0, posinf=0.0, neginf=0.0)
        cmax = float(np.max(cvec))
        if cmax < best_c - 1e-15:
            best_c = cmax
            best_h = h.copy()

        w = _smooth_max_val_and_weights(cvec, a_cur)[1]
        grad = _grad_raw_from_weights(h, n, w)
        grad -= float(np.mean(grad))

        improved = False
        evals = 0

        for i in range(n - 1):
            if time.time() > deadline:
                break
            gi = grad[i]
            for j in range(i + 1, n):
                diff = gi - grad[j]
                if diff > 1e-12:
                    t = j
                    s = i
                elif diff < -1e-12:
                    t = i
                    s = j
                else:
                    continue

                maxd = min(1.0 - h[t], h[s])
                if maxd <= 1e-8:
                    continue

                d = step
                if d > maxd:
                    d = maxd
                if d <= 1e-12:
                    continue

                old_t = h[t]
                old_s = h[s]
                old_1t = hm1[t]
                old_1s = hm1[s]

                h[t] = old_t + d
                h[s] = old_s - d
                hm1[t] = old_1t - d
                hm1[s] = old_1s + d

                cnew = float(np.max(np.correlate(h, hm1, mode="full")) * dx)

                if cnew < cmax - 1e-15:
                    cmax = cnew
                    improved = True
                else:
                    h[t] = old_t
                    h[s] = old_s
                    hm1[t] = old_1t
                    hm1[s] = old_1s

                evals += 1
                if (evals & 2047) == 0 and time.time() > deadline:
                    break

            if time.time() > deadline:
                break

        if time.time() > deadline:
            break

        if improved:
            if cmax < best_c - 1e-15:
                best_c = cmax
                best_h = h.copy()
            step = max(min_step, step * 0.85)
            a_cur = min(60000.0, a_cur * 1.5)
        else:
            step = max(min_step, step * 0.5)
            a_cur = min(60000.0, a_cur * 2.0)

    best_h = project_feasible(best_h, n)
    best_c = _c5_val(best_h, n)
    return best_h, best_c


def _top_pair_line_search(h, n, deadline, alphas=(100.0, 1000.0, 5000.0), top=24):
    h = project_feasible(h, n)
    best_c = _c5_val(h, n)
    best_h = h.copy()
    dx = 2.0 / float(n)
    fracs = (1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02, 0.005)

    for alpha in alphas:
        if time.time() > deadline:
            break

        hm1 = 1.0 - h
        cvec = np.correlate(h, hm1, mode="full") * dx
        cvec = np.nan_to_num(cvec, nan=0.0, posinf=0.0, neginf=0.0)
        cmax = float(np.max(cvec))
        if cmax < best_c - 1e-15:
            best_c = cmax
            best_h = h.copy()

        w = _smooth_max_val_and_weights(cvec, alpha)[1]
        grad = _grad_raw_from_weights(h, n, w)
        grad -= float(np.mean(grad))

        m = min(int(top), max(1, n // 2))
        if m <= 0:
            continue

        inc = np.argpartition(grad, m - 1)[:m]
        dec = np.argpartition(grad, n - m)[n - m:]

        tested = 0
        for ia in inc:
            a = int(ia)
            for ib in dec:
                b = int(ib)
                if a == b:
                    continue
                if (tested & 7) == 0 and time.time() > deadline:
                    break

                maxd = min(1.0 - h[a], h[b])
                if maxd <= 1e-8:
                    tested += 1
                    continue

                base_c = cmax
                best_d = 0.0
                best_v = base_c

                for frac in fracs:
                    d = float(frac) * maxd
                    if d <= 1e-9 or d > maxd:
                        continue

                    old_a = h[a]
                    old_b = h[b]
                    old_1a = hm1[a]
                    old_1b = hm1[b]

                    h[a] = old_a + d
                    h[b] = old_b - d
                    hm1[a] = old_1a - d
                    hm1[b] = old_1b + d

                    cnew = float(np.max(np.correlate(h, hm1, mode="full")) * dx)
                    if cnew < best_v - 1e-15:
                        best_v = cnew
                        best_d = d

                    h[a] = old_a
                    h[b] = old_b
                    hm1[a] = old_1a
                    hm1[b] = old_1b

                if best_d > 1e-9 and best_v < base_c - 1e-15:
                    h[a] += best_d
                    h[b] -= best_d
                    hm1[a] -= best_d
                    hm1[b] += best_d
                    cmax = best_v
                    if cmax < best_c - 1e-15:
                        best_c = cmax
                        best_h = h.copy()

                tested += 1

            if time.time() > deadline:
                break

        h = project_feasible(h, n)

    best_h = project_feasible(best_h, n)
    best_c = _c5_val(best_h, n)
    return best_h, best_c


def _random_pair_search(h, n, deadline, rng, anneal=False, max_evals=200000):
    h = project_feasible(h, n)
    hm1 = 1.0 - h
    dx = 2.0 / float(n)
    current_c = _c5_val(h, n, hm1)
    best_c = current_c
    best_h = h.copy()
    T = 1.0e-4
    count = 0

    for _ in range(int(max_evals)):
        if time.time() > deadline:
            break

        a = int(rng.randint(n))
        b = int(rng.randint(n))
        if a == b:
            continue

        if rng.rand() < 0.8:
            if h[a] < h[b]:
                t = a
                s = b
            else:
                t = b
                s = a
        else:
            if rng.rand() < 0.5:
                t = a
                s = b
            else:
                t = b
                s = a

        maxd = min(1.0 - h[t], h[s])
        if maxd <= 1e-7:
            continue

        u = float(rng.rand())
        if u < 0.15:
            d = maxd
        elif u < 0.6:
            d = min(maxd, float(rng.uniform(0.001, 0.05)))
        else:
            d = min(maxd, float(rng.uniform(0.05, 0.5)))
        if d <= 1e-9:
            continue

        old_t = h[t]
        old_s = h[s]
        old_1t = hm1[t]
        old_1s = hm1[s]

        h[t] = old_t + d
        h[s] = old_s - d
        hm1[t] = old_1t - d
        hm1[s] = old_1s + d

        cnew = float(np.max(np.correlate(h, hm1, mode="full")) * dx)

        accept = cnew < current_c - 1e-15
        if (not accept) and anneal and cnew > current_c:
            delta = cnew - current_c
            if delta < 5.0e-3:
                if float(rng.rand()) < math.exp(-delta / max(1e-8, T)):
                    accept = True

        if accept:
            current_c = cnew
        else:
            h[t] = old_t
            h[s] = old_s
            hm1[t] = old_1t
            hm1[s] = old_1s

        if cnew < best_c - 1e-15:
            best_c = cnew
            best_h = h.copy()

        if anneal:
            T = max(1e-7, T * 0.9998)

        count += 1
        if (count & 4095) == 0 and time.time() > deadline:
            break

    best_h = project_feasible(best_h, n)
    best_c = _c5_val(best_h, n)
    return best_h, best_c


def _binarize(h, n):
    h = np.asarray(h, dtype=float)
    b = np.zeros(n, dtype=float)
    m = n // 2
    order = np.argsort(-h)
    if m > 0:
        b[order[:m]] = 1.0
    if n - 2 * m == 1:
        b[order[m]] = 0.5
    return b


def _binary_search(h0, n, deadline, rng, exhaustive=True):
    h0 = project_feasible(h0, n)
    hb = _binarize(h0, n)
    hb = project_feasible(hb, n)
    hm1 = 1.0 - hb
    dx = 2.0 / float(n)

    initial_raw = float(np.max(np.correlate(hb, hm1, mode="full")))
    cur_raw = initial_raw
    best_raw = initial_raw
    best_c = initial_raw * dx
    best_b = hb.copy()

    if exhaustive and n <= 420:
        passes = 0
        while time.time() < deadline and passes < 80:
            ones = np.nonzero(hb > 0.5)[0]
            zeros = np.nonzero(hb < 0.5)[0]
            improved = False
            evals = 0

            for aa in ones:
                if time.time() > deadline:
                    break
                if hb[aa] <= 0.5:
                    continue
                for bb in zeros:
                    if hb[bb] >= 0.5:
                        continue

                    hb[aa] = 0.0
                    hb[bb] = 1.0
                    hm1[aa] = 1.0
                    hm1[bb] = 0.0

                    newraw = float(np.max(np.correlate(hb, hm1, mode="full")))

                    if newraw < cur_raw - 0.5:
                        cur_raw = newraw
                        if newraw < best_raw - 0.5:
                            best_raw = newraw
                            ctry = newraw * dx
                            if ctry < best_c - 1e-15:
                                best_c = ctry
                                best_b = hb.copy()
                        improved = True
                    else:
                        hb[aa] = 1.0
                        hb[bb] = 0.0
                        hm1[aa] = 0.0
                        hm1[bb] = 1.0

                    evals += 1
                    if (evals & 1023) == 0 and time.time() > deadline:
                        break

                if time.time() > deadline:
                    break

            passes += 1
            if not improved:
                break

    T = 0.5
    evals = 0
    while time.time() < deadline:
        aa = int(rng.randint(n))
        bb = int(rng.randint(n))
        if aa == bb:
            continue
        if hb[aa] == hb[bb]:
            continue

        olda = hb[aa]
        oldb = hb[bb]
        old1a = hm1[aa]
        old1b = hm1[bb]

        hb[aa] = oldb
        hb[bb] = olda
        hm1[aa] = old1b
        hm1[bb] = old1a

        newraw = float(np.max(np.correlate(hb, hm1, mode="full")))

        accept = False
        if newraw < cur_raw - 0.5:
            accept = True
        elif abs(newraw - cur_raw) < 1e-9:
            if rng.rand() < 0.02:
                accept = True
        elif newraw > cur_raw + 0.5:
            if rng.rand() < math.exp(-0.5 * (newraw - cur_raw) / max(0.05, T)):
                accept = True

        if accept:
            cur_raw = newraw
            if cur_raw < best_raw - 0.5:
                best_raw = cur_raw
                ctry = cur_raw * dx
                if ctry < best_c - 1e-15:
                    best_c = ctry
                    best_b = hb.copy()
        else:
            hb[aa] = olda
            hb[bb] = oldb
            hm1[aa] = old1a
            hm1[bb] = old1b

        T = max(0.05, T * 0.99995)
        evals += 1
        if (evals & 4095) == 0 and time.time() > deadline:
            break

    best_b = project_feasible(best_b, n)
    best_c = _c5_val(best_b, n)
    return best_b, best_c


def _resample(h, n, n2):
    h = np.asarray(h, dtype=float)
    if int(n) == int(n2):
        return h.copy()

    x = (np.arange(int(n)) + 0.5) * (2.0 / float(n))
    x2 = (np.arange(int(n2)) + 0.5) * (2.0 / float(n2))
    y = np.interp(x2, x, h, left=float(h[0]), right=float(h[-1]))
    return project_feasible(y, int(n2))


def _fallback_h(n, seed):
    rng = np.random.RandomState(seed)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))

    h1 = 0.5 + 0.5 * np.cos(np.pi * x)
    h2 = 0.5 + 0.35 * np.cos(np.pi * x) + 0.20 * np.cos(2.0 * np.pi * x)
    h3 = 0.5 + 0.20 * rng.normal(size=n)

    h4 = np.zeros(n, dtype=float)
    pos = 0
    val = 1.0
    while pos < n:
        run = 1 + int(rng.randint(0, 4))
        end = min(n, pos + run)
        h4[pos:end] = val
        pos = end
        val = 2.0 - val

    cands = [h1, h2, h3, h4]
    best = None
    bestc = float("inf")
    for hc in cands:
        hp = project_feasible(hc, n)
        c = _c5_val(hp, n)
        if c < bestc:
            bestc = c
            best = hp
    return project_feasible(best, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_budget = float(max(1.0, float(budget_s)))
    deadline = t0 + total_budget - 2.0
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c = float("inf")
    best_n = 0
    h_binary = None

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = None

    candidates = []
    if parent is not None and parent.size >= 20:
        n_parent = int(parent.size)
        h_parent = project_feasible(parent, n_parent)
        if 80 <= n_parent <= 800:
            cp = _c5_val(h_parent, n_parent)
            candidates.append((h_parent.copy(), n_parent, cp))

        if n_parent != 300:
            h300 = project_feasible(_resample(h_parent, n_parent, 300), 300)
            c300 = _c5_val(h300, 300)
            candidates.append((h300, 300, c300))
        else:
            candidates.append((h_parent.copy(), 300, _c5_val(h_parent, 300)))
    else:
        h300 = _fallback_h(300, seed)
        c300 = _c5_val(h300, 300)
        candidates.append((h300, 300, c300))

    for ch, cn, cc in candidates:
        if cc < best_c - 1e-15:
            best_c = cc
            best_h = ch.copy()
            best_n = int(cn)

    if best_h is None:
        best_h = _fallback_h(300, seed)
        best_n = 300
        best_c = _c5_val(best_h, best_n)

    n_main = int(best_n)
    h_main = best_h.copy()
    if (n_main % 2 != 0) or n_main > 600:
        n_main = 300
        h_main = project_feasible(_resample(best_h, best_n, 300), 300)
        cm = _c5_val(h_main, n_main)
        if cm < best_c - 1e-15:
            best_c = cm
            best_h = h_main.copy()
            best_n = n_main

    if (n_main % 2 == 0) and n_main <= 500 and time.time() < deadline - 40:
        phase_deadline = min(time.time() + 90.0, deadline - 40.0)
        if phase_deadline > time.time() + 8.0:
            hb, cb = _binary_search(h_main, n_main, phase_deadline, rng, exhaustive=True)
            h_binary = hb.copy()
            if cb < best_c - 1e-15:
                best_c = cb
                best_h = hb.copy()
                best_n = n_main
                h_main = h_binary.copy()

    starts = [best_h.copy()]
    if h_binary is not None:
        cb_start = _c5_val(h_binary, int(len(h_binary)))
        if cb_start < best_c + 0.004:
            starts.append(h_binary)

    for st in starts:
        ns = int(len(st))
        if time.time() < deadline - 180:
            phase_deadline = min(time.time() + 70.0, deadline - 180.0)
            if phase_deadline > time.time() + 8.0:
                hl, cl = _lbfgs_polish(st, ns, phase_deadline, maxiter=70)
                if cl < best_c - 1e-15:
                    best_c = cl
                    best_h = hl.copy()
                    best_n = ns
                    h_main = hl.copy()

    if best_n <= 380 and time.time() < deadline - 130:
        phase_deadline = min(time.time() + 330.0, deadline - 130.0)
        if phase_deadline > time.time() + 20.0:
            hp, cp = _pair_scan(
                best_h,
                best_n,
                phase_deadline,
                initial_step=0.008,
                min_step=1e-7,
                alpha=300.0,
                max_passes=45,
            )
            if cp < best_c - 1e-15:
                best_c = cp
                best_h = hp.copy()
                best_n = int(len(hp))

    if time.time() < deadline - 90:
        phase_deadline = min(time.time() + 120.0, deadline - 90.0)
        if phase_deadline > time.time() + 10.0:
            ht, ct = _top_pair_line_search(
                best_h,
                best_n,
                phase_deadline,
                alphas=(100.0, 1000.0, 5000.0),
                top=24,
            )
            if ct < best_c - 1e-15:
                best_c = ct
                best_h = ht.copy()
                best_n = int(len(ht))

    if time.time() < deadline - 45:
        phase_deadline = min(time.time() + 70.0, deadline - 45.0)
        if phase_deadline > time.time() + 8.0:
            hr, cr = _random_pair_search(
                best_h,
                best_n,
                phase_deadline,
                rng,
                anneal=False,
                max_evals=180000,
            )
            if cr < best_c - 1e-15:
                best_c = cr
                best_h = hr.copy()
                best_n = int(len(hr))

    if time.time() < deadline - 25:
        phase_deadline = min(time.time() + 25.0, deadline - 25.0)
        if phase_deadline > time.time() + 5.0:
            hr, cr = _random_pair_search(
                best_h,
                best_n,
                phase_deadline,
                rng,
                anneal=True,
                max_evals=60000,
            )
            if cr < best_c - 1e-15:
                best_c = cr
                best_h = hr.copy()
                best_n = int(len(hr))

    high_ns = []
    if best_n <= 350:
        high_ns = [600, 450]
    elif best_n == 450:
        high_ns = [600]
    elif best_n <= 650:
        high_ns = [900]

    for n2 in high_ns:
        if time.time() < deadline - 70:
            h2 = project_feasible(_resample(best_h, best_n, n2), n2)
            c2 = _c5_val(h2, n2)

            if c2 < best_c + 0.003:
                if time.time() < deadline - 45:
                    phase_deadline = min(time.time() + 45.0, deadline - 45.0)
                    if phase_deadline > time.time() + 8.0:
                        h2, c2 = _lbfgs_polish(h2, n2, phase_deadline, maxiter=45)

                if (n2 % 2 == 0) and time.time() < deadline - 35:
                    phase_deadline = min(time.time() + 30.0, deadline - 35.0)
                    if phase_deadline > time.time() + 5.0:
                        hb2, cb2 = _binary_search(h2, n2, phase_deadline, rng, exhaustive=(n2 <= 380))
                        if cb2 < c2:
                            h2, c2 = hb2, cb2

                if time.time() < deadline - 20:
                    phase_deadline = min(time.time() + 30.0, deadline - 20.0)
                    if phase_deadline > time.time() + 5.0:
                        h2, c2 = _top_pair_line_search(
                            h2,
                            n2,
                            phase_deadline,
                            alphas=(200.0, 2000.0),
                            top=18,
                        )

                if c2 < best_c - 1e-15:
                    best_c = c2
                    best_h = h2.copy()
                    best_n = int(n2)

    if best_n <= 380 and time.time() < deadline - 18:
        phase_deadline = min(time.time() + 18.0, deadline - 12.0)
        if phase_deadline > time.time() + 3.0:
            hs, cs = _slsqp_polish(
                best_h,
                best_n,
                phase_deadline,
                alpha=40000.0,
                maxiter=120,
            )
            if cs < best_c - 1e-15:
                best_c = cs
                best_h = hs.copy()
                best_n = int(len(hs))

    if time.time() < deadline - 3:
        hp, cp = _pair_scan(
            best_h,
            best_n,
            deadline - 2.0,
            initial_step=5e-4,
            min_step=1e-8,
            alpha=5000.0,
            max_passes=3,
        )
        if cp < best_c - 1e-15:
            best_c = cp
            best_h = hp.copy()
            best_n = int(len(hp))

    if time.time() < deadline - 1:
        hr, cr = _random_pair_search(
            best_h,
            best_n,
            deadline - 0.5,
            rng,
            anneal=False,
            max_evals=20000,
        )
        if cr < best_c - 1e-15:
            best_c = cr
            best_h = hr.copy()
            best_n = int(len(hr))

    best_h = project_feasible(best_h, best_n)
    best_n = int(len(best_h))
    final_hm1 = 1.0 - best_h
    final_c = float(np.max(np.correlate(best_h, final_hm1, mode="full")) * (2.0 / float(best_n)))
    if final_c < best_c:
        best_c = final_c

    return best_h, best_c, best_n
