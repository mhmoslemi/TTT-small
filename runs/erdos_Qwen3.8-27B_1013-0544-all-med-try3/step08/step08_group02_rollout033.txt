def project_feasible(y, n):

def compute_c_dx(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(compute_c_dx(h, n).max())


def get_weights(c, alpha):
    m = c.size
    if m == 0:
        return np.zeros(0, dtype=float)
    if alpha <= 0.0:
        return np.full(m, 1.0 / float(m), dtype=float)
    z = alpha * (c - c.max())
    zm = float(z.max())
    if np.isfinite(zm):
        z = z - zm
    else:
        z = np.zeros_like(z)
    w = np.exp(z)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(m, 1.0 / float(m), dtype=float)
    return w / s


def weighted_grad_from_weights(h, w, n):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)

    if n == 1:
        return np.array([dx * float(w[0]) * (1.0 - 2.0 * h[0])], dtype=float)

    w0 = float(w[n - 1])
    a = np.zeros(n, dtype=float)
    b = np.zeros(n, dtype=float)

    a[1:] = w[n:]
    b[1:] = w[n - 2::-1]

    grad = w0 * (1.0 - 2.0 * h)

    pa = np.cumsum(a)
    pb = np.cumsum(b)
    grad = grad + pa + pb[np.arange(n - 1, -1, -1)]

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    ih = np.fft.rfft(h, N)

    ia = np.fft.rfft(a, N)
    conv_a = np.fft.irfft(ia * ih, N)[:n]
    grad -= conv_a

    a_rev = a[::-1]
    ira = np.fft.rfft(a_rev, N)
    conv_ar = np.fft.irfft(ira * ih, N)
    grad -= conv_ar[n - 1:n - 1 + n]

    ib = np.fft.rfft(b, N)
    conv_b = np.fft.irfft(ib * ih, N)[:n]
    grad -= conv_b

    b_rev = b[::-1]
    irb = np.fft.rfft(b_rev, N)
    conv_br = np.fft.irfft(irb * ih, N)
    grad -= conv_br[n - 1:n - 1 + n]

    return dx * grad


def energy_obj(x, n, alpha, mu):
    c = compute_c_dx(x, n)
    if alpha <= 0.0:
        val = float(c.mean())
    else:
        mx = float(c.max())
        z = alpha * (c - mx)
        zmax = float(z.max())
        if not np.isfinite(zmax):
            z = np.zeros_like(z)
            zmax = 0.0
        else:
            z = z - zmax
        val = (mx + float(np.log(np.sum(np.exp(z))))) / alpha
    if mu:
        s = float(x.sum() - 0.5 * n)
        val += mu * s * s
    return float(val)


def energy_grad(x, n, alpha, mu):
    c = compute_c_dx(x, n)
    w = get_weights(c, alpha)
    g = weighted_grad_from_weights(x, w, n)
    if mu:
        s = float(x.sum() - 0.5 * n)
        g = g + (2.0 * mu * s) * np.ones_like(g)
    return g


def smooth_refine(h, n, deadline, alphas=None, mu=0.2, max_iter_cap=2500, time_budget=None):
    if time_budget is not None:
        deadline = min(deadline, time.time() + float(time_budget))

    h = project_feasible(h, n)
    cur_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = cur_c

    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 60000]
    alphas = list(alphas)

    if n <= 800:
        rate = 35.0
        cap_iter = 3000
    elif n <= 1500:
        rate = 18.0
        cap_iter = 2600
    else:
        rate = 9.0
        cap_iter = 2400

    if max_iter_cap is not None and int(max_iter_cap) > 0:
        cap_iter = min(cap_iter, int(max_iter_cap))

    bounds = [(0.0, 1.0)] * n
    allow_increase = 0.004

    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 15.0:
            break

        alloc = max(10.0, min(rem * 0.45, 240.0))
        maxit = int(alloc * rate)
        maxit = max(80, min(cap_iter, maxit))

        try:
            res = minimize(
                energy_obj,
                h,
                jac=energy_grad,
                args=(n, float(alpha), float(mu)),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxit,
                    "ftol": 1e-12,
                    "gtol": 1e-9,
                    "maxfun": maxit * 8,
                    "maxls": 50,
                    "maxcor": 40,
                    "pgtol": 1e-9,
                    "disp": False,
                },
            )
        except Exception:
            h = best_h.copy()
            cur_c = best_c
            break

        if not np.all(np.isfinite(res.x)):
            h = best_h.copy()
            cur_c = best_c
            continue

        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)

        if not np.isfinite(c_new):
            h = best_h.copy()
            cur_c = best_c
            continue

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

            cur_c = c_new
        elif c_new <= cur_c + allow_increase:
            h = h_new
            cur_c = c_new
        else:

                cur_c = best_c
            else:
                h = h_new
                cur_c = c_new

    best_c = compute_c5(best_h, n)

def upsample_h(h, n_new):
    n_old = int(h.size)
    n_new = int(n_new)
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, np.asarray(h, dtype=float))
    return project_feasible(h_new, n_new)


def lowpass_h(h, n, keep_frac=0.05):
    h = np.asarray(h, dtype=float)
    m = 0.5 * n
    y = h - m / float(n)
    spec = np.fft.rfft(y)
    cut = max(2, int(n * keep_frac))
    if cut < len(spec):
        spec[cut:] = 0.0
    y2 = np.fft.irfft(spec, n) + m / float(n)
    return project_feasible(y2, n)


def make_binary_start(h, n):
    h = project_feasible(np.asarray(h, dtype=float), n)
    target = 0.5 * n
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    diff = target - float(h_bin.sum())

    if diff > 0.0:
        zeros = np.where(h_bin < 1.0)[0]
        order = np.argsort(-h[zeros])
        remain = diff
        for idx in order:
            if remain <= 1e-13:
                break
            i = zeros[idx]
            add = min(1.0 - h_bin[i], remain)
            if add <= 0.0:
                continue
            h_bin[i] += add
            remain -= add
    elif diff < 0.0:
        ones = np.where(h_bin > 0.0)[0]
        order = np.argsort(h[ones])
        remain = -diff
        for idx in order:
            if remain <= 1e-13:
                break
            i = ones[idx]
            sub = min(h_bin[i], remain)
            if sub <= 0.0:
                continue
            h_bin[i] -= sub
            remain -= sub

    return project_feasible(h_bin, n)


def noise_perturb(h, n, seed, sigma=0.01, keep_frac=0.125):
    h = np.asarray(h, dtype=float)
    sd = int(seed) % 4294967296
    rng = np.random.RandomState(sd)
    spec = np.fft.rfft(rng.normal(size=n))
    cut = max(2, int(n * keep_frac))
    if cut < len(spec):
        spec[cut:] = 0.0
    noise = np.fft.irfft(spec, n)
    sdv = float(np.std(noise))
    if sdv > 1e-12:
        noise = noise / sdv * float(sigma)
    else:
        noise = np.zeros(n, dtype=float)
    return project_feasible(h + noise, n)


def shift_search(h, n, end_time, max_trials=80, seed=123):
    h = project_feasible(np.asarray(h, dtype=float), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if time.time() >= end_time or n <= 1:
        return best_h, best_c

    sd = int(seed) % 4294967296
    rng = np.random.RandomState(sd)

    offsets = [
        0,
        1, n - 1,
        2, n - 2,
        3, n - 3,
        5, n - 5,
        n // 8, n - n // 8,
        n // 4, n - n // 4,
        n // 6, n - n // 6,
        n // 12, n - n // 12,
    ]

    while len(offsets) < max_trials and time.time() < end_time:
        offsets.append(int(rng.randint(1, max(2, n))))

    tried = set()
    for off in offsets:
        if time.time() >= end_time:
            break
        off = int(off) % int(n)
        if off in tried:
            continue
        tried.add(off)
        h2 = np.roll(h, off)
        c2 = compute_c5(h2, n)
        if c2 < best_c - 1e-15:
            best_c = c2
            best_h = h2.copy()

    return best_h, float(best_c)


def shift_and_polish(h, n, best_c, deadline, seed, time_shift=8.0, time_refine=20.0):

def subgrad_refine(h, n, deadline, alpha=30000.0, step_init=0.03):
    h = project_feasible(np.asarray(h, dtype=float), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    base = float(step_init)

    if time.time() >= deadline:
        return best_h, float(best_c)

    factors = (1.0, 1.6, 0.55)

    while time.time() < deadline - 1.0:
        cdx = compute_c_dx(h, n)
        if not np.all(np.isfinite(cdx)):
            break

        w = get_weights(cdx, float(alpha))
        g = weighted_grad_from_weights(h, w, n)
        gn = float(np.linalg.norm(g))
        if not np.isfinite(gn) or gn < 1e-16:
            break

        best_cand = None
        best_cand_c = best_c

        for fac in factors:
            if time.time() >= deadline - 1.0:
                break
            step = (base * float(fac)) / gn
            cand = project_feasible(h - step * g, n)
            cc = compute_c5(cand, n)
            if np.isfinite(cc) and cc < best_cand_c:
                best_cand_c = cc
                best_cand = cand

        if best_cand is not None and best_cand_c < best_c - 1e-15:
            best_c = best_cand_c
            best_h = best_cand.copy()
            h = best_cand.copy()
            base = min(base * 1.07, 0.25)
        else:
            base *= 0.85
            if base < 1e-5:
                break

    return best_h, float(best_c)


def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)


def sum_jac(x, n):
    return np.ones_like(x, dtype=float)


def slqp_refine(h, n, deadline, alpha=20000.0, maxiter=30):
    h = project_feasible(np.asarray(h, dtype=float), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    rem = deadline - time.time()
    if rem < 25.0:
        return best_h, float(best_c)

    maxit = int(min(maxiter, max(5, rem * 0.4)))
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_fun, "jac": sum_jac, "args": (int(n),)}]

    try:
        res = minimize(
            energy_obj,
            h,
            jac=energy_grad,
            args=(int(n), float(alpha), 0.0),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxit, "ftol": 1e-12, "disp": False},
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if np.isfinite(c_new) and c_new < best_c:
                best_h = h_new
                best_c = c_new
    except Exception:
        pass

    return best_h, float(best_c)


def fallback_start(n):
    x = np.linspace(0.0, 2.0, n)
    h = (
        0.5
