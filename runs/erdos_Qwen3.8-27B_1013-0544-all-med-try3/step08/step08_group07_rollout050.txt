def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

            break

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def grad_obj(x):

        m = g.max()

        exps = np.exp(shifted - sm)
        smax = exps.sum()
        weights = exps / smax
        # grad of log-sum-exp + m wrt x
        # d/dx [m + (sm + log_sum_exp(shifted))/alpha]
        # = d/dx [m] + (1/alpha)*sum_k weights[k]*alpha*d(g[k])/dx

                grad += weights[k] * gk
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,

    try:

                       bounds=bounds, constraints=cons,

    a, b = 0, 1
    fib = [0, 1]
    while fib[-1] < n:
        fib.append(fib[-1] + fib[-2])
    # Use golden ratio rotation
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = sorted(int((i * phi) % 1.0 * n) for i in range(n))
    h[positions[:target_ones]] = 1.0
    return h

    t0 = time.time()

    # Phase 2: SLSQP with increasing alpha at parent resolution (20% budget)
    p2 = t0 + 0.30 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > p2 - 10:
            break
        h = slsqp_step_fast(h, best_n, alpha, 800, p2 - 10)
        c = compute_c5(h, best_n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 3: Projected gradient longer (20% budget)
    p3 = t0 + 0.50 * (budget_s - 5)
    for top_k in [5, 10, 20, 30]:
        if time.time() > p3 - 20:
            break
        h = best_h.copy()
        h, c = projected_gradient_active(h, best_n, p3 - 20, top_k=top_k,
                                          max_iter=30000, init_step=0.01)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 4: Try higher resolutions (25% budget)
    p4 = t0 + 0.75 * (budget_s - 5)
    for nh in [300, 400, 500, 600, 800]:
        if time.time() > p4 - 30:
            break
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        # Quick projected gradient
        deadline_curr = min(p4 - 30, time.time() + (p4 - t0 - budget_s * 0.2) / 2)
        hh, ch = projected_gradient_active(hh, nh, deadline_curr, top_k=10,
                                            max_iter=20000, init_step=0.015)
        # SLSQP polish
        if time.time() < deadline_curr - 15:
            hh = slsqp_step_fast(hh, nh, 50000, 500, deadline_curr - 10)
            ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            best_n = nh
    
    # Phase 5: Binary search (10% budget)
    p5 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p5 - 5:
        # SA from current
        hb, cb = binary_sa(best_h, best_n, p5 - 5, max_sweeps=30, T_init=0.005)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        # SA from Thue-Morse
        if time.time() < p5 - 5:
            ht = thue_morse_binary(best_n, best_n // 2)
            hb2, cb2 = binary_sa(ht, best_n, p5 - 5, max_sweeps=20, T_init=0.008)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()
        # Binary local search from best
        if time.time() < p5 - 3:
            hbl, cbl = binary_local_search(best_h, best_n, p5 - 3, max_sweeps=100)
            if cbl < best_c:
                best_c = cbl
                best_h = hbl.copy()
    
    # Phase 6: Final refinement (5% budget)
    if time.time() < dl - 10:
        # SLSQP with very high alpha
        h_fin = slsqp_step_fast(best_h, best_n, 500000, 1000, dl - 10)
        c_fin = compute_c5(h_fin, best_n)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()
    if time.time() < dl - 5:
        h_fin = slsqp_step_fast(best_h, best_n, 2000000, 800, dl - 3)
        c_fin = compute_c5(h_fin, best_n)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()
    if time.time() < dl - 2:
        h, c = projected_gradient_active(best_h, best_n, dl - 1, top_k=5,
                                          max_iter=10000, init_step=0.005)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
