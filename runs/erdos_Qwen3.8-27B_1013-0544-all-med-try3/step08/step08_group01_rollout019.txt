```python
    # alpha: is i in [max(0,-l), min(n-1, n-1-l)]?
    m_lo = np.maximum(0, -l)
    m_hi = np.minimum(n-1, n-1-l)
    alpha = ((m_lo <= i) & (i <= m_hi)).astype(float)
    
    # beta: h[i+l] if valid
    idx_beta = i + l
    beta = np.zeros(2*n-1)
    valid_beta = (idx_beta >= 0) & (idx_beta < n) & alpha
    beta[valid_beta] = h[idx_beta[valid_beta]]
    
    # gamma: h[i-l] if valid
    idx_gamma = i - l
    gamma = np.zeros(2*n-1)
    valid_gamma = (idx_gamma >= 0) & (idx_gamma < n) & (m_lo <= idx_gamma) & (idx_gamma <= m_hi)
    gamma[valid_gamma] = h[idx_gamma[valid_gamma]]
    
    return alpha - beta - gamma
```
