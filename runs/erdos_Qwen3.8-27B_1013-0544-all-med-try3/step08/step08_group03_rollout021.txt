def project_feasible(h, n):
    target = n / 2.0

    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

def _eq_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    # Alpha schedule: start moderate, increase
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        remaining = deadline - time.time() - 5
        # Allocate time based on alpha
        t_sec = min(remaining * 0.35, 80.0)
        if t_sec < 5:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, 1.0),
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 5,
                                   'maxls': 40, 'maxcor': 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=1000000, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
    except Exception:
        pass
    return h, best_c

def interp_h(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new points"""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old/2  # cell centers
    # Actually, for step functions, the natural interpolation is:
    x_old = np.linspace(0, 2, n_old)  # or use midpoints
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old, period=2)  # No, not periodic
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def interp_h_mid(h_old, n_old, n_new):
    """Interpolate using cell midpoints"""
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def pair_search(h, n, deadline, deltas=None, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i]+d, h[j]-d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved_any = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i]-d, h[j]+d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved_any = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def gradient_local_search(h, n, deadline, alpha=10000, step=0.001):
    """Gradient-guided local search: move mass from high-grad to low-grad positions"""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        g = grad_smooth_max(h, n, alpha)
        # Find the pair (i,j) that gives the most improvement
        # Moving mass from j to i: h[i]+=δ, h[j]-=δ
        # Change in objective ≈ δ*(g[i] - g[j])
        # We want g[i] - g[j] < 0, i.e., g[i] < g[j]
        # Most negative: maximize g[j] - g[i]
        g_diff = g[np.newaxis, :] - g[:, np.newaxis]  # [i,j] = g[j] - g[i]
        # Wait: g_diff[i,j] = g[j] - g[i]. We want this to be positive for improvement.
        # Actually: if we move δ from j to i, Δf ≈ δ*(g[i] - g[j])
        # For improvement, we need g[i] - g[j] < 0, i.e., g[j] > g[i]
        # g_diff[i,j] = g[j] - g[i] > 0 means moving from j to i helps
        
        best_improvement = 0
        best_i, best_j = -1, -1
        for i in range(n):
            for j in range(n):
                if i == j: continue
                imp = g[j] - g[i]
                if imp > best_improvement:
                    best_improvement = imp
                    best_i, best_j = i, j
        
        if best_i < 0 or best_improvement < 1e-10:
            break
        
        # Try the move
        for scale in [1.0, 0.5, 0.25, 0.1]:
            δ = step * scale
            if h[best_i] + δ <= 1.0 and h[best_j] - δ >= 0.0:
                h[best_i] += δ; h[best_j] -= δ
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    break
                h[best_i] -= δ; h[best_j] += δ
        
        # Also try a few random pairs
        for _ in range(10):
            i = np.random.randint(n); j = np.random.randint(n)
            if i == j: continue
            δ = step * 0.5
            if g[j] > g[i]:
                if h[i] + δ <= 1.0 and h[j] - δ >= 0.0:
                    h[i] += δ; h[j] -= δ
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                    else:
                        h[i] -= δ; h[j] += δ
            elif g[i] > g[j]:
                if h[j] + δ <= 1.0 and h[i] - δ >= 0.0:
                    h[j] += δ; h[i] -= δ
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                    else:
                        h[j] -= δ; h[i] += δ
        
        alpha = min(alpha * 1.2, 500000)
    
    return h, best_c
