    """Returns the Jacobian matrix of shape (n, 2n-1) where J[m, i] = dg_i/dh[m]."""
    dx = 2.0 / n
    J = np.zeros((n, 2*n-1))
    k_vals = np.arange(2*n-1) - (n-1)

                val += (1 - h[m-k])
            if 0 <= m+k <= n-1:
                val -= h[m+k]
            J[m, i] = dx * val
    return J

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    g = compute_all_g(h, n)
    g_max = g.max()
    if alpha < 1e6:
        exp_g = np.exp(alpha * (g - g_max))
        return g_max + np.log(exp_g.sum()) / alpha
    else:
        return g_max

def smooth_grad(h, n, alpha):
    g = compute_all_g(h, n)
    g_max = g.max()
    exp_g = np.exp(alpha * (g - g_max))
    sum_exp = exp_g.sum()
    p = exp_g / sum_exp
    
    dx = 2.0 / n
    k_vals = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    
    for m in range(n):
        k_lo1 = max(-(n-1), m-(n-1))
        k_hi1 = min(n-1, m)
        if k_lo1 <= k_hi1:
            ks = np.arange(k_lo1, k_hi1+1)
            j = m - ks
            p_idx = ks + (n-1)
            grad[m] += np.dot(p[p_idx], (1.0 - h[j]))
        
        k_lo2 = max(-(n-1), -m)
        k_hi2 = min(n-1, n-1-m)
        if k_lo2 <= k_hi2:
            ks = np.arange(k_lo2, k_hi2+1)
            j = m + ks
            p_idx = ks + (n-1)
            grad[m] -= np.dot(p[p_idx], h[j])
        
        grad[m] *= dx
    
    return grad

def slsqp_analytical(h0, n, alpha, maxiter, deadline):
    """SLSQP with analytical gradient."""
    def obj(x):
        return smooth_obj(x, n, alpha)
    
    def grad_obj(x):
        return smooth_grad(x, n, alpha)
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h0
