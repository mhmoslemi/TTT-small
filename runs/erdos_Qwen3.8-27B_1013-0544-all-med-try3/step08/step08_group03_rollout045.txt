def gradient_descent(h, n, deadline, lr_init=0.02, alpha_init=500.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    for _ in range(10000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        step_lr = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step_lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step_lr *= 0.5
        if not improved:
            lr *= 0.4
            alpha = min(alpha * 1.3, 1e7)
        if lr < 1e-13:
            break
    return best_h, best_c


def pair_search(h, n, deadline, deltas, radius=80):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline - 1:
            improved = False
            for i in range(n):
                if time.time() > deadline - 1:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c


def gradient_pair_search(h, n, deadline, step_size=0.003):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000, 20000]
    for alpha in alphas:
        while time.time() < deadline - 1:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline - 1:
                    return h, best_c
                gi = g[i]
                for j in range(n):
                    if i == j:
                        continue
                    move = -step_size * (gi - g[j])
                    if abs(move) < 1e-10:
                        continue
                    if (h[i] + move >= -1e-12 and h[i] + move <= 1.0 + 1e-12 and
                            h[j] - move >= -1e-12 and h[j] - move <= 1.0 + 1e-12):
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return h, best_c


def perturb_h(h, n, rng, scale=0.02):
    h2 = h.copy() + rng.randn(n) * scale
    return project_feasible(h2, n)


def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, n_new)


def multi_resolution_opt(h, n, deadline, target_ns, allocations):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for n_try, alloc in target_ns:
        if time.time() > deadline - 30:
            break
        if n_try == n:
            continue
        t_start = time.time()
        t_end = min(t_start + alloc, deadline - 20)
        h_interp = interpolate_h(best_h, n, n_try)
        h_new, c_new = lbfgs_phase(h_interp, n_try, t_end * 0.55, max_iters=100)
        if time.time() < t_end * 0.7:
            h_new, c_new = slsqp_phase(h_new, n_try, time.time() + min(30, t_end - time.time() - 5),
                                       alpha=500000, maxiter=1000)
        if time.time() < t_end - 5:
            h_new, c_new = pair_search(h_new, n_try, t_end - 2,
                                       [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=50)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B with increasing alpha (30% of budget)
    t1 = t0 + budget_s * 0.28
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=300)

    # Phase 2: SLSQP refinement (12%)
    if time.time() < deadline - budget_s * 0.4:
        t2 = min(time.time() + budget_s * 0.10, deadline - budget_s * 0.25)
        for a in [200000, 1000000, 5000000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_phase(best_h, n_init, min(time.time() + 40, t2), alpha=a, maxiter=2000)
            if c2 < best_c - 1e-15:
                best_c = c2
                best_h = h2

    # Phase 3: Gradient descent (10%)
    if time.time() < deadline - budget_s * 0.25:
        t3 = min(time.time() + budget_s * 0.08, deadline - budget_s * 0.20)
        h3, c3 = gradient_descent(best_h, n_init, t3, lr_init=0.02, alpha_init=500.0)
        if c3 < best_c - 1e-15:
            best_c = c3
            best_h = h3

    # Phase 4: Gradient-guided pair search (10%)
    if time.time() < deadline - budget_s * 0.18:
        t4 = min(time.time() + budget_s * 0.08, deadline - budget_s * 0.15)
        best_h, best_c = gradient_pair_search(best_h, n_init, t4, step_size=0.003)

    # Phase 5: Standard pair search (12%)
    if time.time() < deadline - budget_s * 0.10:
        t5 = min(time.time() + budget_s * 0.10, deadline - budget_s * 0.08)
        best_h, best_c = pair_search(best_h, n_init, t5,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=100)

    # Phase 6: Multi-resolution attempt (15%)
    if time.time() < deadline - budget_s * 0.08:
        remaining = deadline - time.time() - 10
        t6_start = time.time()
        h6, c6 = lbfgs_phase(best_h, n_init, t6_start + remaining * 0.3, max_iters=150)
        if c6 < best_c - 1e-15:
            best_c = c6
            best_h = h6
        # Try higher resolution
        if time.time() < deadline - 60:
            for n_try, alloc in [(400, remaining * 0.35), (600, remaining * 0.35)]:
                if time.time() > deadline - 50:
                    break
                t_end = min(time.time() + alloc, deadline - 20)
                h_interp = interpolate_h(best_h, len(best_h), n_try)
                t_mid = t_end - (t_end - time.time()) * 0.5
                h_new, c_new = lbfgs_phase(h_interp, n_try, t_mid, max_iters=100)
                if time.time() < t_end - 10:
                    h_new, c_new = slsqp_phase(h_new, n_try, min(time.time() + 25, t_end - 5),
                                               alpha=1000000, maxiter=800)
                if time.time() < t_end - 3:
                    h_new, c_new = pair_search(h_new, n_try, t_end - 2,
                                               [0.005, 0.002, 0.001, 0.0005], radius=40)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new

    # Phase 7: Perturbation + re-optimize (5%)
    if time.time() < deadline - 20:
        for pert_scale in [0.01, 0.005]:
            if time.time() > deadline - 15:
                break
            h_pert = perturb_h(best_h, len(best_h), rng, scale=pert_scale)
            t_pert_end = min(time.time() + 15, deadline - 5)
            h_p, c_p = lbfgs_phase(h_pert, len(best_h), t_pert_end, max_iters=50)
            if c_p < best_c - 1e-15:
                best_c = c_p
                best_h = h_p

    # Phase 8: Final refinement
    if time.time() < deadline - 3:
        nf = len(best_h)
        h_f, c_f = lbfgs_phase(best_h, nf, deadline - 2, max_iters=50)
        if c_f < best_c - 1e-15:
            best_c = c_f
            best_h = h_f
        h_f, c_f = pair_search(best_h, nf, deadline - 1,
                               [0.001, 0.0005, 0.0002, 0.0001], radius=30)
        if c_f < best_c - 1e-15:
            best_c = c_f
            best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
