        if d_norm < 1e-12:

        best_c_ls = best_c

            c_try = compute_c5(h_try, n)
            if c_try < best_c_ls - 1e-14:
                best_c_ls = c_try
                best_alpha = alpha

            h = project_feasible(h + best_alpha * d, n)
            best_c = best_c_ls
        else:

                    h = h_try
                    best_c = c_try
                    break
            else:
                break

def top_k_gradient_descent(h, n, deadline, k_top=5):
    """Gradient descent focusing on top-k lags."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    lr = 0.05
    
    for _ in range(5000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for m in top_idx:
            g += grad_single_lag(h, n, m) / k_top
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        improved = False
        for try_alpha in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.01]:
            h_try = project_feasible(h - try_alpha * g, n)
            c_try = compute_c5(h_try, n)
            if c_try <= best_c - 1e-14:
                h = h_try
                best_c = c_try
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return h, best_c

def fine_pair_search(h, n, deadline, deltas=None):

    best_c = compute_c5(h, n)

            g = np.zeros(n)
            for m in top_idx:
                g += grad_single_lag(h, n, m) / 10
            
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Try moving from i to j where j has lower gradient
                for j in range(n):
                    if i == j:
                        continue

                        continue

                        h[j] += d
                        c_try = compute_c5(h, n)
                        if c_try < best_c - 1e-14:
                            best_c = c_try
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_lag(h, n, m):
    dx = 2.0 / n
    k = m - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n])
        grad[k:n] += (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv])
        grad[0:nv] += (-h[kk:n])
    return grad * dx

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    active = w > 1e-10
    for m in np.where(active)[0]:
        grad += w[m] * grad_single_lag(h, n, m)
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(20, 1e5), (100, 1e6), (500, 1e7), (2000, 1e8),
               (10000, 1e9), (50000, 1e10), (200000, 1e10)]
    
    for alpha, mu in configs:
        remaining = deadline - time.time()
        if remaining < 3:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 3,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def lp_minimax(h, n, deadline, top_k=10):
    """Linearized minimax using LP for direction, line search for step."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds_lp = [(0, 1)] * n + [(0, None)]
    
    no_improve_count = 0
    
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        current_max = c.max()
        
        if current_max <= best_c:
            best_c = current_max
        
        # Get top k lags
        top_idx = np.argsort(c)[-top_k:]
        
        # Build LP
        A_ub = np.zeros((top_k, n + 1))
        b_ub = np.zeros(top_k)
        for j in range(top_k):
            g = grad_single_lag(h, n, top_idx[j])
            A_ub[j, :n] = g
            A_ub[j, n] = -1.0
            b_ub[j] = g @ h - c[top_idx[j]]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            break
        
        h_lp = res.x[:n]
        d = h_lp - h
        d_norm = np.linalg.norm(d)
        
        if d_norm < 1e-12:
            # Try gradient direction
            g = np.zeros(n)
            for m in top_idx:
                g += grad_single_lag(h, n, m) / top_k
            d = -g
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-12:
                break
        
        # Line search
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            h_try = project_feasible(h + alpha * d, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:

                best_c = c_try
                improved = True
                break
        
        if improved:
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 5:
                break
    
    return h, best_c

def topk_grad_descent(h, n, deadline, k_top=5):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    lr = 0.1
    no_improve = 0
    
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for m in top_idx:
            g += grad_single_lag(h, n, m) / k_top
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        improved = False
        for factor in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.01]:
            h_try = project_feasible(h - factor * g, n)
            c_try = compute_c5(h_try, n)
            if c_try <= best_c - 1e-14:
                h = h_try
                best_c = c_try
                improved = True
                break

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_lag(h, n, m):
    dx = 2.0 / n
    k = m - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n])
        grad[k:n] += (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv])
        grad[0:nv] += (-h[kk:n])
    return grad * dx

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    active = w > 1e-10
    for m in np.where(active)[0]:
        grad += w[m] * grad_single_lag(h, n, m)
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(20, 1e5), (100, 1e6), (500, 1e7), (2000, 1e8),
               (10000, 1e9), (50000, 1e10), (200000, 1e11)]
    for alpha, mu in configs:
        remaining = deadline - time.time()
        if remaining < 3:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 3,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def lp_minimax(h, n, deadline, top_k=10):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds_lp = [(0, 1)] * n + [(0, None)]
    no_improve_count = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        current_max = c.max()
        if current_max < best_c:
            best_c = current_max
        top_idx = np.argsort(c)[-top_k:]
        A_ub = np.zeros((top_k, n + 1))
        b_ub = np.zeros(top_k)
        for j in range(top_k):
            g = grad_single_lag(h, n, top_idx[j])
            A_ub[j, :n] = g
            A_ub[j, n] = -1.0
            b_ub[j] = g @ h - c[top_idx[j]]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        h_lp = res.x[:n]
        d = h_lp - h
        if np.linalg.norm(d) < 1e-12:
            g = np.zeros(n)
            for m in top_idx:
                g += grad_single_lag(h, n, m) / top_k
            d = -g
            if np.linalg.norm(d) < 1e-12:

                h = h_try
                best_c = c_try
                improved = True
                break
        if improved:
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 8:
                break
    return h, best_c

        h5, c5 = subgradient_descent(best_h, n, t5)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        if time.time() < deadline - 100:
            h5b, c5b = topk_grad_descent(best_h, n, deadline - 100, k_top=3)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b
    # Phase 6: Targeted pair search
    if time.time() < deadline - 80:
        t6 = min(time.time() + 80, deadline - 80)
        best_h, best_c = targeted_pair_search(best_h, n, t6)
    # Phase 7: LP minimax round 3
    if time.time() < deadline - 60:
        t7 = min(time.time() + 50, deadline - 60)
        h7, c7 = lp_minimax(best_h, n, t7, top_k=12)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    # Phase 8: Random pair search
    if time.time() < deadline - 35:
        t8 = min(time.time() + 40, deadline - 35)
        best_h, best_c = random_pair_search(best_h, n, t8, num_deltas=5)
    # Phase 9: Random restarts
    if time.time() < deadline - 30:
        t9 = min(time.time() + 30, deadline - 30)
        h9, c9 = random_restart_refine(best_h, n, t9, seed + 7, n_restarts=3)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    # Phase 10: Try different n
    if time.time() < deadline - 20:
        for n_try in [200, 400]:
            if time.time() > deadline - 10:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(25, deadline - time.time() - 5)
            if alloc < 10:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.6, max_iters=80)
            if time.time() < deadline - 3:
                h_t, c_t = lp_minimax(h_t, n_try, min(time.time() + 8, deadline - 2), top_k=8)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
    # Phase 11: Final polish
    if time.time() < deadline - 2:
        best_h, best_c = random_pair_search(best_h, len(best_h), deadline - 1, num_deltas=3)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
