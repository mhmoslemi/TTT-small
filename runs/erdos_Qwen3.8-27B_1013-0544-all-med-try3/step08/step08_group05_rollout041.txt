            mask = h > 0.0
            s = h[mask].sum()
            if s < 1e-14:
                break
            h[mask] += diff / s * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max(corr, alpha):
    m = corr.max()
    scaled = alpha * (corr - m)
    sm = scaled.max()
    return m + (sm + np.log(np.exp(scaled - sm).sum())) / alpha

def softmax_weights(corr, alpha):
    m = corr.max()
    scaled = alpha * (corr - m)
    sm = scaled.max()
    w = np.exp(scaled - sm)
    w /= w.sum()
    return w

def compute_grad(corr, h, n, weights):
    """Compute gradient of smooth max correlation w.r.t. h"""
    dx = 2.0 / n
    g = np.zeros(n)
    active = weights > 1e-12
    active_idx = np.where(active)[0]
    for l in active_idx:
        k = l - (n - 1)
        w = weights[l]
        if k >= 0:
            if k == 0:
                g += w * (1.0 - 2.0 * h)
            else:
                g[:n-k] += w * (1.0 - h[k:])
                g[k:] -= w * h[:n-k]
        else:
            kk = -k
            g[kk:] += w * (1.0 - h[:n-kk])
            g[:n-kk] -= w * h[kk:]
    g *= dx
    return g

def slsqp_optimize(h0, n, alpha, maxiter, deadline):

        # And d(corr[k])/dx[i] = (1-h[i+k]) - h[i-k] (with bounds)

        w = softmax_weights(corr, alpha)
        return compute_grad(corr, x, n, w)
    
    def constraint(x):

def binary_local_search(h0, n, deadline, max_sweeps=100):

    # Ensure exactly n/2 ones

        idx = np.argsort(h0)[::-1]
        h = np.zeros(n)
        h[idx[:n//2]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').copy()

    # influence[i][l] = contribution of position i to correlation at lag l
