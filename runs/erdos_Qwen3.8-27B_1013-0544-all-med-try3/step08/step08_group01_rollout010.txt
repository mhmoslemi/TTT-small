import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, n, idx):
    """Gradient of correlation at shift index idx w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def subgrad_descent_max(h, n, deadline, tol=1e-14, max_iter=50000):
    """Subgradient descent on the max of correlation values."""
    h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    n_iter = 0
    
    while time.time() < deadline - 5 and n_iter < max_iter:
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        if c_max < best_c:
            best_c = c_max
            h[0] = h[0]  # keep copy
        
        # Active set: shifts within 0.5% of max
        threshold = c_max - max(1e-8, 0.005 * c_max)
        active = np.where(c_vec >= threshold)[0]

            avg_grad += grad_shift(h, n, idx)

        n_iter += 1

    vals = np.zeros(len(active))
    for i, idx in enumerate(active):

        if k >= 0:
            nv = n - k
            vals[i] = np.sum(x[0:nv] * (1.0 - x[k:n])) * dx
        else:
            kk = -k
            nv = n - kk
            vals[i] = np.sum(x[kk:n] * (1.0 - x[0:nv])) * dx
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 400:

def lse_grad_active(x, n, active, alpha):

    for i, idx in enumerate(active):
        k = idx - (n - 1)

    smx = sh.max()
