Alternatively, I can precompute the "shift kernel" and use convolution. Let me think...

For shift k ≥ 0:
- grad[0:n-k] += w[m] * dx * (1-h[k:n])
- grad[k:n] += w[m] * dx * (-h[0:n-k])

For shift k < 0 (let kk = -k):
- grad[kk:n] += w[m] * dx * (1-h[0:n-kk])
- grad[0:n-kk] += w[m] * dx * (-h[kk:n])

This can be seen as:
- The contribution to grad[j] from shift k is: wx[k] * (1-h[j+k]) for j+k < n, minus wx[k] * h[j-k] for j >= k (when k ≥ 0)

Where wx[k] = w[k+n-1] * dx.

So: grad[j] = Σ_k wx[k] * (1-h[j+k]) * 1(j+k<n) - Σ_k wx[k] * h[j-k] * 1(j-k≥0)

The first sum is a cross-correlation of wx with (1-h), evaluated at position j (with appropriate shifting).
The second sum is a cross-correlation of wx with h, evaluated at position j (with appropriate shifting).

Let me define:
- a = (1-h) (length n)
- b = h (length n)  
- w_shift = wx reordered so that w_shift[k] = wx[k] for k = -(n-1), ..., 0, ..., (n-1)

- First sum at position j = Σ_k w_shift[k] * a[j+k] = (a ⊛ w_shift)[j] where ⊛ is cross-correlation
- Second sum at position j = Σ_k w_shift[k] * b[j-k] = (b * w_shift_flipped)[j] where * is convolution

                        h[i] = max(0.0, h[i] - d)

                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick polish at original resolution
    p1_end = t0 + 120
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[100, 500, 2000, 10000, 50000],
        mu=0.01, maxiter_list=[2000, 2000, 3000, 3000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set refinement at original resolution
    if time.time() < deadline - 500:
        p2_end = min(time.time() + 100, deadline - 500)
        h2, c2 = run_active_set(best_h, best_n, p2_end, top_k=4,
            alphas=[20000, 100000, 500000, 2000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Hinge penalty approach
    if time.time() < deadline - 400:
        p3_end = min(time.time() + 80, deadline - 400)
        C_target = best_c * 0.9999
        h3, c3 = run_hinge(best_h, best_n, p3_end, C_target, mu=0.005)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to higher resolution and optimize
    n_high = 1000
    if time.time() < deadline - 350:
        x_fine = np.linspace(0, 2, n_high)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_high)
        c_fine = compute_c5(h_fine, n_high)

        p4_end = min(time.time() + 250, deadline - 200)
        h4, c4 = run_lbfgs(h_fine, n_high, p4_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000],
            mu=0.005, maxiter_list=[2000, 2500, 2500, 3000, 3000, 3000])
        if time.time() < deadline - 180:
            p4b_end = min(time.time() + 80, deadline - 180)
            h4b, c4b = run_active_set(h4, n_high, p4b_end, top_k=4,
                alphas=[50000, 200000, 1000000])
            if c4b < c4:
                h4, c4 = h4b, c4b
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_high

    # Phase 5: Try binary start at best resolution
    if time.time() < deadline - 150:
        h_bin = make_binary_start(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        p5_end = min(time.time() + 80, deadline - 130)
        h5, c5 = run_lbfgs(h_bin, best_n, p5_end,
            alphas=[100, 1000, 10000, 100000, 500000],
            mu=0.01, maxiter_list=[2000, 2000, 2500, 2500, 3000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Periodic starts (exploration)
    if time.time() < deadline - 100:
        for (per, ones) in [(4,2), (6,3), (5,2), (7,3), (8,4)]:
            if time.time() > deadline - 90:
                break
            h_per = periodic_start(best_n, per, ones)
            c_per = compute_c5(h_per, best_n)
            if c_per < best_c + 0.01:
                p6_end = min(time.time() + 30, deadline - 80)
                h6, c6 = run_lbfgs(h_per, best_n, p6_end,
                    alphas=[100, 1000, 10000, 100000],
                    mu=0.01, maxiter_list=[1500, 1500, 2000, 2000])
                if c6 < best_c:
                    best_c = c6
                    best_h = h6

    # Phase 7: Active set at best solution
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 50, deadline - 50)
        h7, c7 = run_active_set(best_h, best_n, p7_end, top_k=3,
            alphas=[100000, 500000, 2000000, 5000000])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Subgradient refinement
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 30, deadline - 30)
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=50000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Fine pair search
    if time.time() < deadline - 5:
        p9_end = deadline - 2
        h9, c9 = pair_search(best_h, best_n, p9_end, radius=40)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
