import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max_objective(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    g_max = g.max()
    g_shifted = alpha * (g - g_max)
    # Numerically stable log-sum-exp
    g_max2 = g_shifted.max()
    return g_max + (g_max2 + np.log(np.exp(g_shifted - g_max2).sum())) / alpha

def soft_max_gradient(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    g_max = g.max()
    g_shifted = alpha * (g - g_max)
    g_max2 = g_shifted.max()
    weights = np.exp(g_shifted - g_max2)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - (n - 1)
        # grad[i] += w * dx * [(1-x[i+shift]) - x[i-shift]] (for valid indices)
        if shift >= 0:
            # i+shift valid for i=0..n-1-shift
            # i-shift valid for i=shift..n-1
            for i in range(n - shift):
                grad[i] += w * (1 - x[i + shift])
            for i in range(shift, n):
                grad[i] -= w * x[i - shift]
        else:
            s = -shift
            # i+shift = i-s valid for i=s..n-1
            # i-shift = i+s valid for i=0..n-1-s
            for i in range(s, n):
                grad[i] += w * (1 - x[i - s])
            for i in range(n - s):
                grad[i] -= w * x[i + s]
    grad *= dx
    return grad

- In f_k: h[i-k] * (1-h[i]) → linear in h[i] (since h[i-k] is fixed)

So f_k as a function of h[i] is:
f_k(h_i) = base_k + h_i * [(1-h[i+k]) - h[i-k]] (for the shifts where these terms are valid)

Wait, but h[i] also appears in f_0 (shift 0): h[i]*(1-h[i]) = h[i] - h[i]². That's quadratic!

Let me be more careful. f_k = dx * Σ_j h[j]*(1-h[j+k-(n-1)])

For the 'full' correlation with indices:
f_k (for k = 0..2n-2) = dx * Σ_j h[j] * (1-h[j - (k - n + 1)])  [for valid j and j-(k-n+1)]

Let shift = k - (n-1). Then:
f_{shift} = dx * Σ_j h[j] * (1-h[j+shift]) for valid j, j+shift

Terms involving h[i]:
1. j = i: h[i] * (1-h[i+shift]) → linear in h[i]
2. j+shift = i, i.e., j = i-shift: h[i-shift] * (1-h[i]) → linear in h[i]
3. If i+shift = i (shift=0): h[i]*(1-h[i]) → quadratic!

So for shift ≠ 0, f_shift is linear in h[i].
For shift = 0, f_0 is quadratic in h[i]: h[i] - h[i]².

The soft-max of all these is still a max of (mostly) linear and one quadratic function.

For coordinate descent, we minimize:
min_{h_i ∈ [0,1]} max_shift [linear_or_quadratic_in_h_i(shift)]

This can be solved via bisection or by finding the intersection points.

But this is getting very complex. Let me just go with a practical approach:

1. SLSQP with smooth max (like before but tuned better)
2. Coordinate descent with a few iterations
3. Binary local search for the final polish

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    g_max = g.max()
    gs = alpha * (g - g_max)
    gs_max = gs.max()
    return g_max + (gs_max + np.log(np.exp(gs - gs_max).sum())) / alpha

def soft_max_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    g_max = g.max()
    gs = alpha * (g - g_max)
    gs_max = gs.max()
    w = np.exp(gs - gs_max)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        shift = k - (n - 1)
        if shift > 0:
            grad[:n - shift] += wk * (1.0 - x[shift:])
            grad[shift:] -= wk * x[:n - shift]
        elif shift < 0:
            s = -shift
            grad[s:] += wk * (1.0 - x[:n - s])
            grad[:n - s] -= wk * x[s:]
        else:
            grad += wk * (1.0 - 2.0 * x)
    grad *= dx
    return grad

def slsqp_opt(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(soft_max_obj, h, jac=lambda x: soft_max_grad(x, n, alpha),
                       args=(n, alpha), method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_search(h, n, deadline, max_sweeps=2000, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        worst_lag = np.argmax(corr)
        shift = worst_lag - (n - 1)
        # Pairs at worst lag first
        cands = []
        if shift > 0:
            for i in range(n - shift):
                if h[i] == 1 and h[i + shift] == 0:
                    cands.append((i, i + shift))
        elif shift < 0:
            s = -shift
            for i in range(s, n):
                if h[i] == 1 and h[i - s] == 0:
                    cands.append((i, i - s))
        # Add random candidates
        zl = list(zeros)
        ol = list(ones)
        if zl and ol:
            for _ in range(min(30, len(zl) * len(ol))):
                cands.append((rng.choice(zl), rng.choice(ol)))
        rng.shuffle(cands)

        for (i, j) in cands:
            if time.time() > deadline:
                break
            if h[i] != 0 or h[j] != 1:
                continue
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            if 0 <= k1 < 2 * n - 1:
                delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            if 0 <= k2 < 2 * n - 1:
                delta[k2] += 1.0
            new_corr = corr + delta
            nc = new_corr.max() * dx
            if nc < cur_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = nc
                ones.discard(j)
                ones.add(i)
                zeros.discard(i)
                zeros.add(j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def iterated_binary(h, n, deadline, seed=42, n_iter=30):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    cur = best_h.copy()
    for it in range(n_iter):
        if time.time() > deadline - 10:
            break
        cur, c = binary_search(cur, n, deadline - 20, max_sweeps=500, seed=seed + it * 7)
        if c < best_c:
            best_c = c
            best_h = cur.copy()
        # Perturb
        h_p = best_h.copy()
        ones = np.where(h_p == 1)[0]
        zeros = np.where(h_p == 0)[0]
        n_p = rng.randint(3, 12)
        for _ in range(n_p):
            if len(ones) == 0 or len(zeros) == 0:
                break
            a = rng.choice(ones)
            b = rng.choice(zeros)
            h_p[a], h_p[b] = 0.0, 1.0
            ones = np.where(h_p == 1)[0]
            zeros = np.where(h_p == 0)[0]
        cur = h_p
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)

    # Phase 2: Binary local search (20% time)

    hb = np.zeros(n)
    idx = np.argsort(-best_h)
    hb[idx[:n // 2]] = 1.0
    hb2, cb2 = iterated_binary(hb, n, p2, seed=seed, n_iter=15)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()

    # Phase 3: SLSQP from binary (15% time)
    p3 = t0 + 0.53 * (budget_s - 5)
    hp = project_feasible(best_h + 0.03 * rng.randn(n), n)

        if time.time() > p3:
            break
        hp = slsqp_opt(hp, n, alpha, 4000, p3)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()

    # Phase 4: Higher resolution exploration (20% time)
    p4 = t0 + 0.73 * (budget_s - 5)
    for nh in [500, 750]:
        if time.time() > p4 - 40:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)

        ch = compute_c5(hh, nh)
        # Binary round

        ib = np.argsort(-hh)
        hbin[ib[:nh // 2]] = 1.0
        hbin2, cbin2 = binary_search(hbin, nh, p4 - 15, max_sweeps=300, seed=seed + 100)
        if cbin2 < best_c:
            best_c = cbin2
            best_h = hbin2.copy()
            n = nh
        elif ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh

    # Phase 5: Final SLSQP + binary (20% time)
    p5 = dl
    h = slsqp_opt(project_feasible(best_h + 0.01 * rng.randn(n), n), n, 1000000, 5000, dl - 20)
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Final binary polish
    hb = np.zeros(n)
    idx = np.argsort(-best_h)
    hb[idx[:n // 2]] = 1.0
    hb2, cb2 = binary_search(hb, n, dl, max_sweeps=500, seed=seed + 200)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()
    # One more SLSQP
    if time.time() < dl - 5:
        h = slsqp_opt(project_feasible(best_h + 0.02 * rng.randn(n), n), n, 2000000, 2000, dl - 2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
