def continuous_refine(h_bin, n, deadline):

    h = h_bin.astype(float).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()

        if time.time() > deadline:
            break
        dx = 2.0 / n
        
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        def cons_fun(x):
            return x.sum() - n / 2.0
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]
        try:

                           constraints=cons,

                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

                i = zeros[rng.randint(len(zeros))]
            if targets_j:
                j = targets_j[rng.randint(len(targets_j))]
            else:
                j = ones[rng.randint(len(ones))]
        else:
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
        if i == j:
            continue
        h_bin[i], h_bin[j] = 1.0, 0.0
        new_c = compute_c5(h_bin, n)
        d = new_c - cur_c
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-14)):
            cur_c = new_c
            cur_g = compute_all_g(h_bin, n)
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = 0.0, 1.0
        T *= 0.9998
    return best_h, best_c

def slsqp_refine(h, n, deadline, alphas=(5000, 20000, 100000, 500000)):
    best_h = h.astype(float).copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
            return val
        def cons_fun(x):
            return x.sum() - n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def multi_res_binary(h_best, n_best, deadline, resolutions):
    best_h = h_best.copy()
    best_c = compute_c5(h_best, n_best)
    for nh in resolutions:
        if time.time() > deadline - 30:
            break
        if nh % 2 != 0:
            nh += 1
        xo = np.linspace(0, 2, n_best)
        xn = np.linspace(0, 2, nh)
        h_interp = np.interp(xn, xo, h_best)
        h_bin = project_binary(h_interp, nh)
        c_bin = compute_c5(h_bin, nh)
        h_bin, c_bin = greedy_binary(h_bin, nh, deadline - 60, max_sweeps=30)
        h_bin, c_bin = anneal_binary(h_bin, nh, 0.008, 0.0001,
                                     deadline - 30, 80000, two_opt_frac=0.3, seed=7)
        h_bin, c_bin = targeted_anneal(h_bin, nh, deadline - 20, 60000, seed=8)
        if c_bin < best_c - 1e-9:
            best_c = c_bin
            best_h = h_bin.copy()
            n_best = nh
    return best_h, best_c, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 200
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    
    # Phase 1: Binary optimization at parent resolution (30% time)
    p1 = t0 + 0.30 * (budget_s - 5)
    if n0 % 2 == 0:
        h_bin = project_binary(h0, n0)
        c_bin = compute_c5(h_bin, n0)
        h_bin, c_bin = greedy_binary(h_bin, n0, min(p1 - 60, t0 + 150), max_sweeps=40)
        h_bin, c_bin = anneal_binary(h_bin, n0, 0.012, 0.00005,
                                     p1 - 30, 150000, two_opt_frac=0.35, seed=seed)
        h_bin, c_bin = targeted_anneal(h_bin, n0, p1 - 10, 100000, seed=seed + 1)
        h_bin, c_bin = greedy_binary(h_bin, n0, p1, max_sweeps=20)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
            best_n = n0
    
    # Phase 2: Multi-resolution binary search (35% time)
    p2 = t0 + 0.65 * (budget_s - 5)
    res_list = [r for r in [200, 300, 400, 500, 600] if r != best_n]
    best_h, best_c, best_n = multi_res_binary(best_h, best_n, p2, res_list)
    
    # Phase 3: Another round of annealing at best resolution (15% time)
    p3 = t0 + 0.80 * (budget_s - 5)
    if best_n % 2 == 0:
        h_bin = project_binary(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        h_bin, c_bin = anneal_binary(h_bin, best_n, 0.006, 0.00001,
                                     p3, 120000, two_opt_frac=0.4, seed=seed + 100)
        h_bin, c_bin = targeted_anneal(h_bin, best_n, p3, 80000, seed=seed + 200)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
    
    # Phase 4: Continuous refinement (15% time)
    if time.time() < dl - 5:
        h_cont, c_cont = slsqp_refine(best_h, best_n, dl,
                                      alphas=(10000, 50000, 200000, 1000000))
        if c_cont < best_c:
            best_c = c_cont
            best_h = h_cont.copy()
        # Try rounding the continuous solution
        h_round = project_binary(best_h, best_n)
        c_round = compute_c5(h_round, best_n)
        if c_round < best_c:
            best_c = c_round
            best_h = h_round.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
