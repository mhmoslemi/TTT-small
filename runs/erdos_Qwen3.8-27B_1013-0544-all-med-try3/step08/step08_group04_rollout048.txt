    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()

    exp_c = np.exp(shifted - shifted.max())  # for numerical stability
    exp_sum = exp_c.sum()
    smax = c_max + (shifted.max() + np.log(exp_sum)) / alpha
    w = exp_c / exp_sum

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
