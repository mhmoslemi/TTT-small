def softmax_weights(vals, alpha):
    vals=np.asarray(vals,dtype=float)
    if len(vals)==1: return np.ones(1)
    if alpha <=0: return np.full(len(vals),1.0/len(vals))
    mx=vals.max()
    s=alpha*(vals-mx)
    # max s 0, no overflow
    e=np.exp(s)
    e/=e.sum()
    return e

def lag_gradient(h,n,idx):
    center=n-1
    l=idx-center
    g=np.zeros(n,dtype=float)
    if l>=0:
       r=l; m=n-r
       if m>0:
          g[r:n] += (1.0 - h[0:m])
          g[0:m] -= h[r:n]
    else:
       r=-l; m=n-r
       if m>0:
          g[0:m] += (1.0 - h[r:n])
          g[r:n] -= h[0:m]
    dx=2.0/float(n)
    g *= dx
    return g

_GRAD_MODE = 0


def compute_c5_vec(h, n):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    if n <= 800:
        return np.correlate(h, 1.0 - h, mode="full") * dx

    v = 1.0 - h
    L = 1
    need = 2 * n - 1
    while L < need:
        L <<= 1
    fa = np.fft.rfft(h, L)
    fv = np.fft.rfft(v[::-1], L)
    c = np.fft.irfft(fa * fv, L)[:need]
    c = np.maximum(c, 0.0)
    return c * dx


def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))


def compute_c5_direct(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

        if diff > 0.0:
            idx = np.where(y < 1.0 - 1e-15)[0]
            if idx.size == 0:
                break
            add = diff / idx.size
            maxadd = float((1.0 - y[idx]).max())
            if add > maxadd:
                add = maxadd
            if add <= 0.0:
                break
            y[idx] += add
        else:
            idx = np.where(y > 1e-15)[0]
            if idx.size == 0:
                break
            sub = -diff / idx.size
            maxsub = float(y[idx].min())
            if sub > maxsub:
                sub = maxsub
            if sub <= 0.0:
                break
            y[idx] -= sub

    return y


def reflect_h(h):
    return h[::-1].copy()


def complement_h(h):
    return 1.0 - h


def choose_best_candidates(h, n):
    cands = [
        h,
        reflect_h(h),
        complement_h(h),
        complement_h(reflect_h(h))
    ]
    best_h = None
    best_c = np.inf
    for hh in cands:
        hh = project_feasible(hh, n)
        c = compute_c5(hh, n)
        if c < best_c:
            best_c = c
            best_h = hh
    return best_h, float(best_c)


def active_indices(cvec, max_count=24, abs_tol=1e-6, rel_tol=0.002):
    cvec = np.asarray(cvec, dtype=float)
    m = float(np.max(cvec))
    thr = m - max(float(abs_tol), float(rel_tol) * m)
    idx = np.where(cvec >= thr - 1e-15)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(cvec))], dtype=int)
    elif idx.size > max_count:
        order = np.argsort(-cvec[idx])
        idx = idx[order[:max_count]]
    amax = int(np.argmax(cvec))
    if not bool(np.any(idx == amax)):
        idx = np.concatenate((idx, np.array([amax], dtype=int)))
        if idx.size > max_count:
            cvals = cvec[idx]
            order = np.argsort(-cvals)
            idx = idx[order[:max_count]]
    return idx


def softmax_weights(vals, alpha):
    vals = np.asarray(vals, dtype=float)
    m = vals.size
    if m == 1:
        return np.ones(1, dtype=float)
    if alpha <= 0.0:
        return np.full(m, 1.0 / m, dtype=float)
    mx = float(np.max(vals))
    s = alpha * (vals - mx)
    e = np.exp(s)
    e = e / float(e.sum())
    return e


def lag_gradient(h, n, idx):
    center = n - 1
    lag = int(idx) - center
    g = np.zeros(n, dtype=float)
    if lag >= 0:
        r = lag
        m = n - r
        if m > 0:
            g[r:n] += 1.0 - h[0:m]
            g[0:m] -= h[r:n]
    else:
        r = -lag
        m = n - r
        if m > 0:
            g[0:m] += 1.0 - h[r:n]
            g[r:n] -= h[0:m]
    g *= 2.0 / float(n)
    return g


def lag_gradient_parent(h, n, idx):
    center = n - 1
    lag = int(idx) - center
    g = np.zeros(n, dtype=float)
    if lag >= 0:
        r = lag
        m = n - r
        if m > 0:
            g[0:m] += 1.0 - h[r:n]
            g[r:n] -= h[0:m]
    else:
        r = -lag
        m = n - r
        if m > 0:
            g[r:n] += 1.0 - h[0:m]
            g[0:m] -= h[r:n]
    g *= 2.0 / float(n)
    return g


def lag_gradient_chosen(h, n, idx):
    if _GRAD_MODE == 0:
        return lag_gradient(h, n, idx)
    return lag_gradient_parent(h, n, idx)


def lag_value(h, n, idx):
    dx = 2.0 / float(n)
    return float(np.correlate(h, 1.0 - h, mode="full")[idx] * dx)


def finite_diff_lag(h, n, idx, eps=1e-5):
    h = np.asarray(h, dtype=float)
    c0 = lag_value(h, n, idx)
    g = np.zeros(n, dtype=float)
    for i in range(n):
        hp = h.copy()
        hm = h.copy()
        hp[i] = min(1.0, h[i] + eps)
        hm[i] = max(0.0, h[i] - eps)
        if abs(hp[i] - h[i]) < 1e-18 and abs(hm[i] - h[i]) < 1e-18:
            g[i] = 0.0
        elif abs(hp[i] - h[i]) < 1e-18:
            g[i] = (c0 - lag_value(hm, n, idx)) / eps
        elif abs(hm[i] - h[i]) < 1e-18:
            g[i] = (lag_value(hp, n, idx) - c0) / eps
        else:
            scale = 2.0 * eps
            g[i] = (lag_value(hp, n, idx) - lag_value(hm, n, idx)) / scale
    return g


def calibrate_gradient():
    n = 16
    rng = np.random.RandomState(1234567)
    h = np.clip(rng.rand(n), 0.0, 1.0)
    h = project_feasible(h, n)
    idxs = [0, 2, 5, 8, 12, 15, 22, 27, 30, 31]
    err0 = 0.0
    err1 = 0.0
    for idx in idxs:
        if 0 <= idx <= 2 * n - 2:
            fd = finite_diff_lag(h, n, idx, 1e-5)
            g0 = lag_gradient(h, n, idx)
            g1 = lag_gradient_parent(h, n, idx)
            err0 += float(np.sum((fd - g0) * (fd - g0)))
            err1 += float(np.sum((fd - g1) * (fd - g1)))
    return 0 if err0 <= err1 else 1


def soft_obj_active(h, n, act, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[act]
    mx = float(np.max(vals))
    if alpha <= 0.0:
        val = mx
    else:
        s = alpha * (vals - mx)
        e = np.exp(s)
        val = mx + float(np.log(float(e.sum()))) / alpha
    if mu != 0.0:
        diff = float(h.sum()) - 0.5 * float(n)
        val += float(mu) * diff * diff
    return float(val)


def soft_grad_active(h, n, act, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[act]
    w = softmax_weights(vals, alpha)
    g = np.zeros(n, dtype=float)
    for wi, ai in zip(w, act):
        if wi > 1e-16:
            g += wi * lag_gradient_chosen(h, n, ai)
    if mu != 0.0:
        diff = float(h.sum()) - 0.5 * float(n)
        g += 2.0 * float(mu) * diff
    return g


def slqp_obj_active(h, n, act, alpha):
    return soft_obj_active(h, n, act, alpha, 0.0)


def slqp_grad_active(h, n, act, alpha):
    return soft_grad_active(h, n, act, alpha, 0.0)


def sum_constraint(x, n):
    return float(x.sum()) - 0.5 * float(n)


def sum_jac(x, n):
    return np.ones_like(x, dtype=float)


def run_lbfgs_active(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if configs is None:
        configs = [
            (8, 500.0, 0.05, 320, 900),
            (8, 2500.0, 0.02, 320, 900),
            (16, 1000.0, 0.02, 420, 1200),
            (16, 6000.0, 0.01, 420, 1200),
            (32, 2500.0, 0.01, 420, 1200),
            (32, 12000.0, 0.005, 320, 900),
            (64, 6000.0, 0.005, 260, 800)
        ]

    bounds = [(0.0, 1.0)] * n

    for count, alpha, mu, maxiter, maxfun in configs:
        if time.time() > deadline - 8.0:
            break
        cvec = compute_c5_vec(h, n)
        act = active_indices(cvec, max_count=count, abs_tol=1e-6, rel_tol=0.002)
        try:
            res = minimize(
                soft_obj_active,
                best_h,
                jac=soft_grad_active,
                args=(n, act, float(alpha), float(mu)),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": int(maxiter),
                    "maxfun": int(maxfun),
                    "ftol": 1e-18,
                    "gtol": 1e-12,
                    "maxls": 20,
                    "maxcor": 20,
                    "disp": False
                }
            )
        except Exception:
            continue

        h_new = project_feasible(np.asarray(res.x, dtype=float), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new

    return best_h, float(best_c)


def run_slqp_polish(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if configs is None:
        configs = [
            (6, 1000.0, 700),
            (8, 4000.0, 800),
            (12, 1500.0, 900),
            (12, 8000.0, 900)
        ]

    bounds = [(0.0, 1.0)] * n
    cons = [{
        "type": "eq",
        "fun": sum_constraint,
        "jac": sum_jac,
        "args": (n,)
    }]

    for count, alpha, maxiter in configs:
        if time.time() > deadline - 10.0:
            break
        cvec = compute_c5_vec(h, n)
        act = active_indices(cvec, max_count=count, abs_tol=1e-6, rel_tol=0.001)
        try:
            res = minimize(
                slqp_obj_active,
                best_h,
                jac=slqp_grad_active,
                args=(n, act, float(alpha)),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-18,
                    "disp": False
                }
            )
        except Exception:
            continue

        h_new = project_feasible(np.asarray(res.x, dtype=float), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new

    return best_h, float(best_c)


def projected_subgrad_polish(h, n, deadline, seed=1):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)

    for count in (5, 10, 20):
        if time.time() > deadline - 5.0:
            break

        cvec = compute_c5_vec(h, n)
        act = active_indices(cvec, max_count=count, abs_tol=1e-7, rel_tol=0.001)
        vals = cvec[act]

        dirs = []

        for ai in act[:min(count, 12)]:
            d = lag_gradient_chosen(h, n, ai)
            nm = float(np.linalg.norm(d))
            if nm > 1e-14:
                dirs.append(-d / nm)

        for alpha in (200.0, 2000.0):
            w = softmax_weights(vals, alpha)
            d = np.zeros(n, dtype=float)
            for wi, ai in zip(w, act):
                if wi > 1e-15:
                    d += wi * lag_gradient_chosen(h, n, ai)
            nm = float(np.linalg.norm(d))
            if nm > 1e-14:
                dirs.append(-d / nm)

        d = np.zeros(n, dtype=float)
        for ai in act:
            d += lag_gradient_chosen(h, n, ai)
        d /= max(1, len(act))
        nm = float(np.linalg.norm(d))
        if nm > 1e-14:
            dirs.append(-d / nm)

        for d in dirs:
            if time.time() > deadline - 3.0:
                break
            for step in (0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002):
                y = project_feasible(h - step * d, n)
                c = compute_c5(y, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = y
                    h = y
                    break

    return best_h, float(best_c)


def perturb_h(h, n, rng, sigma):
    y = h + float(sigma) * rng.randn(n)
    return project_feasible(y, n)


def run_short_optimus(h, n, deadline):
    cfgs = [
        (8, 1200.0, 0.04, 220, 650),
        (16, 5000.0, 0.015, 220, 650)
    ]
    h, c = run_lbfgs_active(h, n, deadline, configs=cfgs)
    h, c = choose_best_candidates(h, n)
    if time.time() < deadline - 8.0:
        h, c = projected_subgrad_polish(h, n, deadline, seed=99)
    return h, c


def resample_h(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, int(n_old))
    x_new = np.linspace(0.0, 2.0, int(n_new))
    y = np.interp(x_new, x_old, np.asarray(h, dtype=float))
    return project_feasible(y, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    global _GRAD_MODE

    t0 = time.time()
    total_budget = float(budget_s)
    deadline = t0 + total_budget - 5.0

    _GRAD_MODE = calibrate_gradient()

    h_init = None
    n0 = 0
    try:
        h_init = np.asarray(initial_h_values, dtype=float).ravel()
        n0 = int(h_init.size)
        if n0 < 10:
            h_init = None
            n0 = 0
    except Exception:
        h_init = None
        n0 = 0

    if h_init is None:
        n0 = 500
        x = np.linspace(0.0, 2.0, n0)
        h_init = 0.5 + 0.25 * np.cos(np.pi * x) + 0.10 * np.cos(2.0 * np.pi * x)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_feasible(h_init, n0)

    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)
    best_h, best_c = choose_best_candidates(best_h, n0)

    t1 = t0 + 0.34 * total_budget
    if time.time() < min(t1, deadline - 180.0) - 10.0:
        best_h, best_c = run_lbfgs_active(best_h, n0, min(t1, deadline - 180.0))
        best_h, best_c = choose_best_candidates(best_h, n0)

    t2 = t0 + 0.48 * total_budget
    if time.time() < min(t2, deadline - 160.0) - 10.0:
        best_h, best_c = run_slqp_polish(best_h, n0, min(t2, deadline - 160.0))
        best_h, best_c = choose_best_candidates(best_h, n0)

    t3 = t0 + 0.58 * total_budget
    if time.time() < min(t3, deadline - 140.0) - 10.0:
        best_h, best_c = projected_subgrad_polish(best_h, n0, min(t3, deadline - 140.0), seed=int(seed))
        best_h, best_c = choose_best_candidates(best_h, n0)

    if time.time() < deadline - 260.0:
        rng = np.random.RandomState(int(seed) + 17)
        sigmas = (0.05, 0.02, 0.01, 0.005)
        for sig in sigmas:
            if time.time() > deadline - 120.0:
                break
            hp = perturb_h(best_h, n0, rng, sig)
            hp, cp = choose_best_candidates(hp, n0)
            short_dl = min(time.time() + 65.0, deadline - 95.0)
            if short_dl > time.time() + 15.0:
                hp, cp = run_short_optimus(hp, n0, short_dl)
                hp, cp = choose_best_candidates(hp, n0)
            if cp < best_c:
                best_c = cp
                best_h = hp

    n_cur = int(best_h.size)
    if n_cur < 1500 and time.time() < deadline - 180.0:
        n_up = max(1000, 2 * n_cur)
        if n_up % 2 == 1:
            n_up += 1
        n_up = min(1500, n_up)
        h_up = resample_h(best_h, n_cur, n_up)
        h_up, c_up = choose_best_candidates(h_up, n_up)
        up_dl = min(time.time() + 150.0, deadline - 70.0)
        if up_dl > time.time() + 20.0:
            h_up, c_up = run_short_optimus(h_up, n_up, up_dl)
            h_up, c_up = choose_best_candidates(h_up, n_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            n_cur = int(h_up.size)

    if time.time() < deadline - 25.0:
        fin_cfgs = [
            (16, 4000.0, 0.015, 180, 550),
            (32, 10000.0, 0.008, 180, 550)
        ]
        final_dl = deadline - 12.0
        best_h, best_c = run_lbfgs_active(best_h, n_cur, final_dl, configs=fin_cfgs)
        best_h, best_c = choose_best_candidates(best_h, n_cur)

    if time.time() < deadline - 8.0:
        best_h, best_c = projected_subgrad_polish(best_h, n_cur, deadline - 2.0, seed=int(seed) + 777)
        best_h, best_c = choose_best_candidates(best_h, n_cur)

    n_final = int(best_h.size)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5_direct(best_h, n_final)

    return best_h, float(best_c), int(n_final)
