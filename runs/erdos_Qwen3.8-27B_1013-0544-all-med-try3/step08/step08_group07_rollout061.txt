def soft_max_topk(g, k, alpha):

def sturmian_binary(n, slope):
    """Generate a Sturmian (mechanical) word of length n with given slope"""
    target = n // 2
    # Use Beatty sequence
    alpha = slope
    positions = []
    for k in range(1, n+1):
        pos = int((k+1) * alpha) - 1
        if 0 <= pos < n and pos not in positions:
            positions.append(pos)
    # If we don't have exactly target positions, add/remove
    while len(positions) < target:
        # Add the position that maximizes minimum spacing
        best_pos = -1
        best_min_dist = -1
        for p in range(n):
            if p in positions:
                continue
            min_dist = min(abs(p - q) for q in positions) if positions else 0
            if min_dist > best_min_dist:
                best_min_dist = min_dist
                best_pos = p
        if best_pos >= 0:
            positions.append(best_pos)
        else:
            break
    
    h = np.zeros(n)
    for p in positions[:target]:
        h[p] = 1.0
    return h

def compute_all_g(h, n):

def project_feasible(h, n):

            if s < 1e-14:

def grad_shift(h, n, s):

    if s >= 0:

        if s > 0:

def delta_c(h, n, i, j):
    """Compute delta in all c[k] when h[i]: 0->1 and h[j]: 1->0 (i was 0, j was 1)"""
    dx = 2.0 / n
    dc = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - n + 1
        val = 0.0
        if 0 <= i + s < n:
            val += (1.0 - h[i + s])
        if 0 <= i - s < n:
            val -= h[i - s]
        if 0 <= j + s < n:
            val -= (1.0 - h[j + s])
        if 0 <= j - s < n:
            val += h[j - s]
        dc[k] = val * dx
    return dc

def relax_binary(h_bin, n, deadline):
    """Take binary solution, identify positions to make fractional, optimize"""
    h = h_bin.astype(float).copy()
    g = compute_all_g(h, n)
    gm = g.max()
    k_worst = int(np.argmax(g))
    s_worst = k_worst - n + 1
    grad = grad_shift(h, n, s_worst)
    # Find positions where flipping slightly would help
    # For positions with h=1 where grad < 0, decreasing helps
    # For positions with h=0 where grad > 0, increasing helps
    candidates = []
    for i in range(n):
        if h[i] == 1.0 and grad[i] < -0.001:
            candidates.append((i, 'dec', -grad[i]))
        elif h[i] == 0.0 and grad[i] > 0.001:
            candidates.append((i, 'inc', grad[i]))
    candidates.sort(key=lambda x: -x[2])
    # Take top ~10 candidates and make them fractional
    top = candidates[:min(15, len(candidates))]
    h_relax = h.copy()
    for (i, direction, score) in top:
        frac = 0.3 * min(score / max(gm, 1e-10) * 10, 1.0)
        if direction == 'dec':
            h_relax[i] = 1.0 - frac
        else:
            h_relax[i] = frac
    h_relax = project_feasible(h_relax, n)
    # Optimize the fractional positions
    free_idx = np.array([i for (i, _, _) in top])
    if len(free_idx) == 0:
        return h, compute_c5(h, n)

    def obj_fixed(x):
        hh = h_relax.copy()
        hh[free_idx] = np.clip(x, 0, 1)
        hh = project_feasible(hh, n)
        return compute_c5(hh, n)

    x0 = h_relax[free_idx].copy()
    res = minimize(obj_fixed, x0, method='Nelder-Mead',
                   options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
    h_final = h_relax.copy()
    h_final[free_idx] = np.clip(res.x, 0, 1)
    h_final = project_feasible(h_final, n)
    return h_final, compute_c5(h_final, n)

def quasi_random_binary(n, seed_val):
    target = n // 2
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    positions = []
    for k in range(target * 2):
        pos = int(((k + 0.5) / target) * n * 0.61803) % n
        if pos not in positions:
            positions.append(pos)
    remaining = [p for p in range(n) if p not in positions]
    rng = np.random.RandomState(seed_val)
    while len(positions) < target and remaining:
        p = remaining[rng.randint(len(remaining))]
        if positions:
            score = min((p - q) % n for q in positions)
        else:
            score = 0
        positions.append(p)
        remaining.remove(p)
    h = np.zeros(n)
    for p in positions[:target]:
        h[p] = 1.0
    return h

def sturmian_binary(n, slope, seed_val):
    target = n // 2
    rng = np.random.RandomState(seed_val)
    positions = []
    alpha = slope
    for k in range(1, target * 3):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.append(pos)
    while len(positions) < target and len(positions) < n:
        # Add position maximizing min distance
        best_p, best_d = -1, -1
        for p in range(n):
            if p in positions:
                continue
            if positions:
                d = min(min(abs(p - q), n - abs(p - q)) for q in positions)
            else:
                d = 0
            if d > best_d:
                best_d = d
                best_p = p
        if best_p >= 0:
            positions.append(best_p)
        else:
            break
    h = np.zeros(n)
    for p in positions[:target]:
        h[p] = 1.0
    return h
