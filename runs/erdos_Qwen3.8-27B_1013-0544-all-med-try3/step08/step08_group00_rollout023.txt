import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += 0.5 * mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def c5_grad(x, n, alpha, mu):

    m = c.max()

            for idx in active:
                kk = idx - (n - 1)
                if kk >= 0:
                    nv = n - kk
                    grad[0:nv] += (1.0 - h[kk:n])
                    grad[kk:n] += (-h[0:nv])
                else:
                    kk2 = -kk
                    nv = n - kk2
                    grad[kk2:n] += (1.0 - h[0:nv])
                    grad[0:nv] += (-h[kk2:n])
            grad /= max(1, len(active))
            
            # Find best pair to swap
            top_idx = np.argsort(-np.abs(grad))[:min(radius * 2, n)]
            for i_idx in top_idx:
                i = int(i_idx)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from i to j
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                    # Try moving mass from j to i
                    if h[j] - step >= -1e-12 and h[i] + step <= 1.0 + 1e-12:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[j] += step
                            h[i] -= step
    return h, best_c

def sqp_step(h, n, active_idx, rho):
    """SQP step: minimize t s.t. g_k(h+δ) ≤ c_k + t for active k, |δ| ≤ rho"""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    dx = 2.0 / n
    
    for idx in active_idx:
        kk = idx - (n - 1)
        g = np.zeros(n)
        if kk >= 0:
            nv = n - kk
            g[0:nv] = (1.0 - h[kk:n]) * dx
            g[kk:n] += (-h[0:nv]) * dx
        else:
            kk2 = -kk
            nv = n - kk2
            g[kk2:n] = (1.0 - h[0:nv]) * dx
            g[0:nv] += (-h[kk2:n]) * dx
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

            rho *= 0.5
            if rho < 1e-10:

            continue

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)

def downsample(h_old, n_old, n_new):
    # Average groups of adjacent points
    if n_old % n_new == 0:
        ratio = n_old // n_new
        h_new = h_old.reshape(n_new, ratio).mean(axis=1)
    else:
        h_new = np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h_old)
    return project_feasible(h_new, n_new)

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        kk = idx - (n - 1)
        g = np.zeros(n)
        if kk >= 0:
            nv = n - kk
            g[0:nv] = (1.0 - h[kk:n]) * dx
            g[kk:n] += (-h[0:nv]) * dx
        else:
            kk2 = -kk
            nv = n - kk2
            g[kk2:n] = (1.0 - h[0:nv]) * dx
            g[0:nv] += (-h[kk2:n]) * dx
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=200, rho_init=0.03):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.002 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 12:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:12]]
        delta, t_opt = sqp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            if rho < 0.05:
                rho = min(rho * 1.4, 0.07)
        else:
            rho *= 0.5
            if rho < 1e-11:
                break
    return best_h, best_c

def local_search(h, n, deadline, radius=25):
    best_c = compute_c5(h, n)
    steps = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00003, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad = np.zeros(n)
            for idx in active:
                kk = idx - (n - 1)
                if kk >= 0:
                    nv = n - kk
                    grad[0:nv] += (1.0 - h[kk:n])
                    grad[kk:n] += (-h[0:nv])
                else:
                    kk2 = -kk
                    nv = n - kk2
                    grad[kk2:n] += (1.0 - h[0:nv])
                    grad[0:nv] += (-h[kk2:n])
            grad /= max(1, len(active))
            top_idx = np.argsort(-np.abs(grad))[:min(radius * 2, n)]
            for i_idx in top_idx:
                i = int(i_idx)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                    if h[j] - step >= -1e-12 and h[i] + step <= 1.0 + 1e-12:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[j] += step
                            h[i] -= step
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def downsample(h_old, n_old, n_new):
    if n_old % n_new == 0:
        ratio = n_old // n_new
        h_new = h_old.reshape(n_new, ratio).mean(axis=1)
    else:
        h_new = np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)

    # Phase 1: Quick L-BFGS polish (35s)
    t1 = min(t0 + 35, deadline - 900)
    if t1 > time.time() + 3:
        h0, c = lbfgs_optimize(h0, n0, t1,
            alphas=[500, 2000, 8000], mus=[100, 500, 2000], maxiter=2500)
        if c < best_c:
            best_c = c
            best_h = h0.copy()

    # Phase 2: Multi-start perturbations (55s)
    t2 = min(t0 + 90, deadline - 800)
    if t2 > time.time() + 5:
        for trial in range(5):
            if time.time() > t2 - 8:
                break
            if trial == 0:
                h_try = fourier_perturb(best_h, n0, rng, 4, 0.012)
            elif trial == 1:
                h_try = noise_perturb(best_h, n0, rng, 0.006)
            elif trial == 2:
                h_try = fourier_perturb(best_h, n0, rng, 2, 0.025)
            elif trial == 3:
                h_try = noise_perturb(best_h, n0, rng, 0.012)
            else:
                h_try = fourier_perturb(best_h, n0, rng, 6, 0.008)
            c_try = compute_c5(h_try, n0)
            if c_try > best_c + 0.004:
                continue
            h_try, c_try = lbfgs_optimize(h_try, n0, t2 - 8,
                alphas=[3000, 10000], mus=[2000, 8000], maxiter=1500)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()

    # Phase 3: Subgradient (30s)
    t3 = min(t0 + 120, deadline - 750)
    if t3 > time.time() + 3:
        h_sg, c_sg = subgradient_optimize(best_h, n0, t3, max_iter=4000)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg.copy()

    # Phase 4: SQP (45s)
    t4 = min(t0 + 165, deadline - 700)
    if t4 > time.time() + 3:
        best_h, best_c = sqp_optimize(best_h, n0, t4, max_iter=250, rho_init=0.025)

    # Phase 5: High-alpha L-BFGS (45s)
    t5 = min(t0 + 210, deadline - 650)
    if t5 > time.time() + 3:
        best_h, best_c = lbfgs_optimize(best_h, n0, t5,
            alphas=[20000, 80000, 200000], mus=[5000, 20000, 80000], maxiter=2000)

    # Phase 6: Higher resolution n=1000 (90s)
    n_hi = 1000
    t6 = min(t0 + 300, deadline - 550)
    if t6 > time.time() + 10:
        h_hi = upsample(best_h, len(best_h), n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        t6a = min(time.time() + 40, deadline - 300)
        if t6a > time.time() + 5:
            h_hi, c_hi = lbfgs_optimize(h_hi, n_hi, t6a,
                alphas=[1000, 5000, 20000, 60000], mus=[500, 2000, 8000, 30000], maxiter=2000)
        t6b = min(time.time() + 30, deadline - 180)
        if t6b > time.time() + 3:
            h_hi, c_hi = sqp_optimize(h_hi, n_hi, t6b, max_iter=120, rho_init=0.02)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()

    # Phase 7: n=200 for cleaner structure (30s)
    n_lo = 200
    t7 = min(t0 + 330, deadline - 500)
    if t7 > time.time() + 5:
        h_lo = downsample(best_h, len(best_h), n_lo)
        c_lo = compute_c5(h_lo, n_lo)
        t7a = min(time.time() + 22, deadline - 350)
        if t7a > time.time() + 3:
            h_lo, c_lo = lbfgs_optimize(h_lo, n_lo, t7a,
                alphas=[500, 2000, 8000, 30000], mus=[200, 1000, 4000, 15000], maxiter=2000)
        if c_lo < best_c:
            best_c = c_lo
            best_h = h_lo.copy()

    # Phase 8: Best resolution, very high alpha (50s)
    n_best = len(best_h)
    t8 = min(t0 + 380, deadline - 400)
    if t8 > time.time() + 3:
        best_h, best_c = lbfgs_optimize(best_h, n_best, t8,
            alphas=[50000, 150000, 400000], mus=[10000, 40000, 120000], maxiter=3000)

    # Phase 9: SQP refinement (30s)
    t9 = min(t0 + 410, deadline - 350)
    if t9 > time.time() + 3:
        best_h, best_c = sqp_optimize(best_h, n_best, t9, max_iter=150, rho_init=0.015)

    # Phase 10: Local search (45s)
    t10 = min(t0 + 455, deadline - 300)
    if t10 > time.time() + 3:
        best_h, best_c = local_search(best_h, n_best, t10, radius=20)

    # Phase 11: SLSQP polish (40s)
    t11 = min(t0 + 495, deadline - 200)
    if t11 > time.time() + 5:
        h_sp, c_sp = slqp_polish(best_h, n_best, t11, alpha=3000000)
        if c_sp < best_c:
            best_c = c_sp
            best_h = h_sp.copy()

    # Phase 12: Second multi-start (40s)
    t12 = min(t0 + 535, deadline - 150)
    if t12 > time.time() + 5:
        for trial in range(3):
            if time.time() > t12 - 8:
                break
            if trial == 0:
                h_try = fourier_perturb(best_h, n_best, rng, 3, 0.008)
            elif trial == 1:
                h_try = noise_perturb(best_h, n_best, rng, 0.004)
            else:
                h_try = fourier_perturb(best_h, n_best, rng, 5, 0.005)
            h_try, c_try = lbfgs_optimize(h_try, n_best, t12 - 8,
                alphas=[20000, 80000], mus=[5000, 20000], maxiter=1500)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()

    # Phase 13: Final subgradient + micro (remaining)
    if time.time() < deadline - 15:
        h_final, c_final = subgradient_optimize(best_h, n_best, deadline - 10, max_iter=3000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()

    # Final micro local search
    if time.time() < deadline - 5:
        best_h, best_c = local_search(best_h, n_best, deadline - 1, radius=10)

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
