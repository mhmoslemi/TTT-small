def project_feasible(h, n):

import numpy as np

import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_obj(x, n, top_k, mu):
    c = compute_c5_vec(x, n)
    # Sum of top k values
    idx = np.argpartition(-c, top_k)[:top_k]
    val = c[idx].sum()
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, top_k, mu):
    c = compute_c5_vec(x, n)
    idx = np.argpartition(-c, top_k)[:top_k]
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in idx:
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += dx * (1.0 - x[k:n])
            grad[k:n] += dx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += dx * (1.0 - x[0:nv])
            grad[0:nv] += dx * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def smoothmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smoothmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(len(c)):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_obj(x, n, top_k, mu):
    c = compute_c5_vec(x, n)
    idx = np.argpartition(-c, top_k)[:top_k]
    val = float(c[idx].sum())
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def topk_grad(x, n, top_k, mu):
    c = compute_c5_vec(x, n)
    idx = np.argpartition(-c, top_k)[:top_k]
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in idx:
        k = mi - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += dx * (1.0 - x[k:n])
            grad[k:n] += dx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += dx * (1.0 - x[0:nv])
            grad[0:nv] += dx * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(len(c)):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:

        k = idx - (n - 1)

        if k >= 0:

            grad[0:nv] += wm * (1.0 - x[k:n])

            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:

    return grad

def target_obj(x, n, T, mu):
    c = compute_c5_vec(x, n)
    viol = np.maximum(0.0, c - T)
    val = 0.5 * float(viol @ viol)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def target_grad(x, n, T, mu):
    c = compute_c5_vec(x, n)
    viol = np.maximum(0.0, c - T)
    active = viol > 1e-12
    dx = 2.0 / n
    grad = np.zeros(n)
    am = np.where(active)[0]
    for mi in am:
        k = mi - (n - 1)
        w = viol[mi] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += w * (1.0 - x[k:n])
            grad[k:n] += w * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += w * (1.0 - x[0:nv])
            grad[0:nv] += w * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, fn, gfn, args, bounds, maxiter, ftol=1e-18, gtol=1e-15):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    maxit = max(200, int((deadline - time.time()) * 8))
    try:
        res = minimize(fn, best_h, jac=gfn, args=args, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxit, 'ftol': ftol,
                                'gtol': gtol, 'maxfun': maxit * 10,
                                'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    except Exception:
        pass
    return best_h, best_c

def run_multi_lbfgs(h, n, deadline, fn, gfn, arg_list, bounds, maxiters):

def coordinate_refine(h, n, deadline, radius=50, shifts=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if shifts is None:
        shifts = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
    for d in shifts:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def run_slqp(h, n, deadline, alpha=500000, maxiter=1000):

def upsample(h_old, n_old, n_new):

            [(n0, 10000, 0.005), (n0, 50000, 0.005)], bounds0, [800, 800])
        if c1 < best_c:
            best_c, best_h = c1, h1

    # Phase 2: Optimize at n=1000 with top-k
    n2 = 1000
    if time.time() < deadline - 350:
        h2 = upsample(best_h, best_n, n2)
        c2 = compute_c5(h2, n2)
        if c2 > best_c:
            h2 = project_feasible(h2, n2)
            c2 = compute_c5(h2, n2)
        bounds2 = [(0.0, 1.0)] * n2

        # Top-k optimization (k=5)
        p2a_end = min(time.time() + 200, deadline - 300)
        h2a, c2a = run_multi_lbfgs(h2, n2, p2a_end, topk_obj, topk_grad,
            [(n2, 5, 0.01), (n2, 3, 0.005), (n2, 5, 0.005), (n2, 2, 0.005)],
            bounds2, [2000, 2000, 2000, 2000])
        if c2a < best_c:
            best_c, best_h, best_n = c2a, h2a, n2

        # Smooth max at n=1000
        if time.time() < deadline - 250:
            p2b_end = min(time.time() + 150, deadline - 250)
            h2b, c2b = run_multi_lbfgs(best_h, n2, p2b_end, sm_obj, sm_grad,
                [(n2, 20000, 0.003), (n2, 100000, 0.003), (n2, 500000, 0.001)],
                bounds2, [2000, 2000, 2000])
            if c2b < best_c:
                best_c, best_h, best_n = c2b, h2b, n2

    # Phase 3: Target-level optimization at n=1000
    if time.time() < deadline - 200:
        T_init = compute_c5_vec(best_h, n2).max()
        p3_end = min(time.time() + 120, deadline - 200)
        bounds2 = [(0.0, 1.0)] * n2
        targets = [T_init * 0.999, T_init * 0.998, T_init * 0.997, T_init * 0.995]
        h3 = best_h.copy()
        c3 = best_c
        for T in targets:
            if time.time() > p3_end - 10:
                break
            mi = max(500, int((p3_end - time.time()) * 5))
            try:
                res = minimize(target_obj, h3, jac=target_grad,
                             args=(n2, T, 0.01), method='L-BFGS-B',
                             bounds=bounds2,
                             options={'maxiter': mi, 'ftol': 1e-16,
                                      'gtol': 1e-14, 'maxfun': mi * 8,
                                      'maxls': 40, 'maxcor': 40})
                h3_new = project_feasible(res.x, n2)
                c3_new = compute_c5(h3_new, n2)
                if c3_new < c3:
                    c3 = c3_new
                    h3 = h3_new
            except Exception:
                pass
        if c3 < best_c:
            best_c, best_h, best_n = c3, h3, n2

    # Phase 4: Higher resolution n=2000
    n4 = 2000
    if time.time() < deadline - 150:
        h4 = upsample(best_h, best_n, n4)
        c4 = compute_c5(h4, n4)
        bounds4 = [(0.0, 1.0)] * n4
        p4_end = min(time.time() + 180, deadline - 150)

        h4a, c4a = run_multi_lbfgs(h4, n4, p4_end, topk_obj, topk_grad,
            [(n4, 5, 0.005), (n4, 3, 0.005), (n4, 5, 0.001)],
            bounds4, [1500, 1500, 1500])
        if c4a < best_c:
            best_c, best_h, best_n = c4a, h4a, n4

        if time.time() < deadline - 100:
            p4b_end = min(time.time() + 60, deadline - 100)
            h4b, c4b = run_multi_lbfgs(best_h, n4, p4b_end, sm_obj, sm_grad,
                [(n4, 50000, 0.002)], bounds4, [1000])
            if c4b < best_c:
                best_c, best_h, best_n = c4b, h4b, n4

    # Phase 5: Active set refinement at best resolution
    if time.time() < deadline - 80:
        p5_end = min(time.time() + 60, deadline - 80)
        bounds_b = [(0.0, 1.0)] * best_n
        h5, c5 = active_set_refine(best_h, best_n, p5_end, top_k=4,
            alphas=[50000, 200000, 1000000])
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: SLSQP polish
    if time.time() < deadline - 40:
        for alpha in [100000, 500000, 2000000]:
            if time.time() > deadline - 35:
                break
            p6_end = min(time.time() + 15, deadline - 30)
            h6, c6 = run_slqp(best_h, best_n, p6_end, alpha=alpha, maxiter=800)
            if c6 < best_c:
                best_c, best_h = c6, h6

    # Phase 7: Coordinate descent
    if time.time() < deadline - 10:
        p7_end = deadline - 3
        h7, c7 = coordinate_refine(best_h, best_n, p7_end, radius=30,
            shifts=[0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05])
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
