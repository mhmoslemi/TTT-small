def project_box_sum(y,target):
    y=np.asarray(y,float)
    n=y.size
    target=float(target)
    if target <=0: return np.zeros(n)
    if target >=n: return np.ones(n)
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(80):
        mid=(lo+hi)/2
        z=np.clip(y-mid,0,1)
        s=z.sum()
        if s > target: lo=mid
        else: hi=mid
    z=np.clip(y-(lo+hi)/2,0,1)
    # correct residual
    diff=target-z.sum()
    if abs(diff)>1e-14:
       idx=np.where((z>1e-12)&(z<1-1e-12))[0]
       if idx.size>0:
          add=diff/idx.size
          z2=z.copy(); z2[idx]+=add; z2=np.clip(z2,0,1)
          # if fail, adjust one
          if abs(z2.sum()-target)>1e-12:
             z=z2
             # repeat few
       return z
    return z

def active_gradient(h,g,n,tol):
    M=float(g.max())
    idx=np.where(g >= M - tol)[0]
    if idx.size == 0:
        idx=np.array([int(g.argmax())])
    w = np.maximum(g[idx] - M + 1.0, 1e-15)
    ws = float(w.sum())
    if ws <= 0: w = np.ones(idx.size); ws=idx.size
    else: w = w/ws
    grad=np.zeros(n,float)
    off=n-1
    for k,wk in zip(idx,w):
        if wk <= 1e-15: continue
        p = int(k) - off
        if p >= 0:
            m = n - p
            grad[:m] += wk * (1.0 - h[p:p+m])
            grad[p:p+m] -= wk * h[:m]
        else:
            q = -p
            m = n - q
            grad[q:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[q:]
    return grad

import numpy as np

def compute_counts(h):
    return np.correlate(h, 1.0 - h, mode="full").astype(float)


def c5_from_counts(g, n):
    return float(np.max(g)) * (2.0 / n)


def evaluate_h(h):
    n = len(h)
    g = compute_counts(h)
    return c5_from_counts(g, n)


def project_box_sum(y, target):
    y = np.asarray(y, dtype=float)
    n = y.size
    target = float(target)
    if n == 0:
        return y
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(25):
        diff = target - float(np.sum(z))
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = np.where(z < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            add = diff / float(idx.size)
            z = np.clip(z + (add if add > 0 else -1e-15), 0.0, 1.0)
            z[idx] += add
            z = np.clip(z, 0.0, 1.0)
        else:
            idx = np.where(z > 1e-12)[0]
            if idx.size == 0:
                break
            add = diff / float(idx.size)
            z[idx] += add
            z = np.clip(z, 0.0, 1.0)
    return z


def build_pre(n):
    off = n - 1
    L = 2 * n - 1
    F = np.empty((n, n), dtype=np.int64)
    S = np.empty((n, n), dtype=np.int64)
    for i in range(n):
        F[i, :] = np.arange(off - i, off - i + n)
        S[i, :] = np.arange(i, i + n)
    return F, S, L, off


def active_gradient(h, g, n, tol):
    M = float(np.max(g))
    idx = np.where(g >= M - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(g))])

    w = np.maximum(g[idx] - M + 1.0, 1e-15)
    ws = float(np.sum(w))
    if ws <= 1e-15:
        w = np.ones(idx.size)
        ws = float(idx.size)
    else:
        w = w / ws

    grad = np.zeros(n, dtype=float)
    off = n - 1

    for k, wk in zip(idx, w):
        if wk <= 1e-15:

        p = int(k) - off
        if p >= 0:
            m = n - p
            grad[:m] += wk * (1.0 - h[p:p + m])

        else:
            q = -p
            m = n - q
            grad[q:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[q:]
    return grad

            g = compute_counts(h)
            cur_c = c5_from_counts(g, n)
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best_h = h.copy()
            moves_grad = 0

        if cur_c > best_c + 0.05:
            h = best_h.copy()
            g = compute_counts(h)
            cur_c = c5_from_counts(g, n)
            step = float(step0)
            moves_grad = 0
            continue

        if moves_grad <= 0:
            tol = tols[tol_idx % len(tols)]
            grad = active_gradient(h, g, n, tol)
            grad -= grad.mean()
            moves_grad = 12
        else:
            moves_grad -= 1

        psize = min(max_p, max(8, n // 4))
        order = np.argsort(grad)
        targets = order[:psize]
        sources = order[-psize:]

        omh = 1.0 - h
        hr = h[::-1]

        improved = False
        inner_deadline = deadline - 1.0

        for i in targets:
            if time.time() > inner_deadline:
                break
            ai = 1.0 - h[i]
            if ai <= 1e-14:
                continue
            Fi = F[i]
            Si = S[i]
            base_d = min(step, ai)
            if base_d <= 1e-14:
                continue

            for j in sources:
                if i == j:
                    continue
                if time.time() > inner_deadline:
                    break

                d = base_d
                hj = h[j]
                if d > hj:
                    d = hj
                if d <= 1e-14:
                    continue

                Fj = F[j]
                Sj = S[j]
                d2 = d * d

                cand = g.copy()
                cand[Fi] += d * omh
                cand[Si] -= d * hr
                cand[Fj] -= d * omh
                cand[Sj] += d * hr

                idx1 = j - i + off
                if 0 <= idx1 < L:
                    cand[idx1] -= d2
                idx2 = i - j + off
                if 0 <= idx2 < L:
                    cand[idx2] += d2

                cnew = c5_from_counts(cand, n)
                if cnew < cur_c - 1e-13:
                    g = cand
                    h[i] += d
                    h[j] -= d
                    cur_c = cnew
                    if cur_c < best_c - 1e-15:
                        best_c = cur_c
                        best_h = h.copy()
                    accepted += 1
                    moves_grad = 0
                    improved = True
                    break

                if d > 1e-7:
                    d05 = 0.5 * d
                    d2_05 = d05 * d05
                    cand = g.copy()
                    cand[Fi] += d05 * omh
                    cand[Si] -= d05 * hr
                    cand[Fj] -= d05 * omh
                    cand[Sj] += d05 * hr

                    if 0 <= idx1 < L:
                        cand[idx1] -= d2_05
                    if 0 <= idx2 < L:
                        cand[idx2] += d2_05

                    cnew = c5_from_counts(cand, n)
                    if cnew < cur_c - 1e-13:
                        g = cand
                        h[i] += d05
                        h[j] -= d05
                        cur_c = cnew
                        if cur_c < best_c - 1e-15:
                            best_c = cur_c
                            best_h = h.copy()
                        accepted += 1
                        moves_grad = 0
                        improved = True
                        break

            if improved:
                break

        if improved:
            continue

        step *= 0.65
        if step < 1e-5:
            step = float(step0)
            tol_idx += 1
            moves_grad = 0
            perturbs += 1

            if perturbs <= 250:
                if perturbs % 5 == 0:
                    base = best_h.copy()
                else:
                    base = h.copy()
                noise = 0.025 + 0.125 * float(rng.rand())
                h = project_box_sum(base + rng.normal(0.0, noise, n), target)
                g = compute_counts(h)
                cur_c = c5_from_counts(g, n)
                if cur_c < best_c - 1e-15:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                break

    h = project_box_sum(best_h, target)
    g = compute_counts(h)
    best_c = c5_from_counts(g, n)
    return h, best_c


def resample_h(h_old, n_old, n_new):
    if n_new == n_old:
        return np.array(h_old, dtype=float, copy=True)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    out = np.interp(x_new, x_old, h_old)
    return out


def make_binary(h, n):
    b = (h > 0.5).astype(float)
    target = float(n) / 2.0
    if abs(target - round(target)) > 1e-12:
        return project_box_sum(b, target)

    int_target = int(round(target))
    cnt = int(np.sum(b))

    if cnt > int_target:
        idx = np.where(b == 1.0)[0]
        if idx.size > 0:
            idx = idx[np.argsort(h[idx])]
            need = cnt - int_target
            b[idx[:need]] = 0.0
    elif cnt < int_target:
        idx = np.where(b == 0.0)[0]
        if idx.size > 0:
            idx = idx[np.argsort(-h[idx])]
            need = int_target - cnt
            b[idx[:need]] = 1.0

    return b


def random_binary(n, rng):
    b = np.zeros(n, dtype=float)
    target = n // 2
    if target > 0:
        idx = rng.choice(n, target, replace=False)
        b[idx] = 1.0
    if n % 2 != 0:
        return project_box_sum(b, float(n) / 2.0)
    return b


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float("inf")
    base_n = 200

    try:
        h0 = np.array(initial_h_values, dtype=float, copy=True)
        n0 = len(h0)
        base_n = n0
        h0 = project_box_sum(h0, n0 / 2.0)
        c0 = evaluate_h(h0)
        best_h = h0.copy()
        best_c = c0
    except Exception:
        n0 = 200
        x = np.arange(n0) * (2.0 / n0)
        h0 = 0.5 + 0.4 * np.cos(np.pi * x)
        h0 = project_box_sum(h0, n0 / 2.0)
        best_h = h0.copy()
        best_c = evaluate_h(best_h)

    def keep_if_better(h, c):
        nonlocal best_h, best_c
        if h is not None and c < best_c:
            best_h = np.array(h, dtype=float, copy=True)
            best_c = float(c)

    if best_h is not None:
        d = min(t0 + 0.45 * (deadline - t0), deadline - 35.0)
        if d > time.time() + 5.0:
            h, c = local_search(best_h, len(best_h), d, rng, step0=0.05, max_p=50)
            keep_if_better(h, c)

        if time.time() < deadline - 45.0:
            hb = make_binary(best_h, len(best_h))
            keep_if_better(hb, evaluate_h(hb))
            d2 = min(time.time() + 55.0, deadline - 30.0)
            if d2 > time.time() + 5.0:
                h, c = local_search(hb, len(hb), d2, rng, step0=0.025, max_p=45)
                keep_if_better(h, c)

    seen = set()
    cand_ns = [base_n, 160, 240, 300, 120, 360, 180, 220]
    for n_try in cand_ns:
        if n_try <= 0:
            continue
        if n_try % 2 != 0:
            n_try += 1
        if n_try < 80 or n_try > 420:
            continue
        if n_try in seen:
            continue
        seen.add(n_try)

        if n_try == base_n:
            continue
        if best_h is None:
            continue
        if time.time() > deadline - 120.0:
            break

        start = resample_h(best_h, len(best_h), n_try)
        d = min(time.time() + 65.0, deadline - 90.0)
        if d <= time.time() + 8.0:
            break

        h, c = local_search(start, n_try, d, rng, step0=0.04, max_p=40)
        keep_if_better(h, c)

        if time.time() < deadline - 55.0:
            hb = make_binary(start, n_try)
            keep_if_better(hb, evaluate_h(hb))
            d2 = min(time.time() + 22.0, deadline - 40.0)
            if d2 > time.time() + 4.0:
                h, c = local_search(hb, n_try, d2, rng, step0=0.02, max_p=35)
                keep_if_better(h, c)

    if best_h is not None and time.time() < deadline - 40.0:
        hb = random_binary(base_n, rng)
        keep_if_better(hb, evaluate_h(hb))
        d = min(time.time() + 28.0, deadline - 20.0)
        if d > time.time() + 4.0:
            h, c = local_search(hb, base_n, d, rng, step0=0.02, max_p=40)
            keep_if_better(h, c)

    if best_h is not None and time.time() < deadline - 5.0:
        h, c = local_search(best_h, len(best_h), deadline - 1.5, rng,
                            step0=0.004, max_p=60)
        keep_if_better(h, c)

    h = project_box_sum(best_h, len(best_h) / 2.0)
    g = compute_counts(h)
    c = c5_from_counts(g, len(h))
    return h, c, len(h)
