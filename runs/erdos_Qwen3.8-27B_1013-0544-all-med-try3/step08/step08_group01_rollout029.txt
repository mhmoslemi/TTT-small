import numpy as np
import time


def compute_c5_direct(h):
    """Exact evaluator-compatible objective."""
    n = len(h)
    if n <= 0:
        return float("inf")
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def make_edge_v(n):
    """
    Symmetric V-shaped construction:
        h((i + 0.5) * 2/n) = |(i + 0.5) * 2/n - 1|.

    Equivalently, for even n:
        h[i] = |2*i - (n - 1)| / n.

    For powers of two n, this gives values with denominator n, and the
    discrete sum is exactly n/2 up to binary-float representation.
    """
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    return np.abs(2.0 * i - float(n - 1)) / float(n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    try:
        b = float(budget_s)
    except Exception:
        b = 1100.0
    if b != b or b <= 0.0:
        b = 1100.0

    # Leave a small safety margin to ensure we return before the deadline.
    deadline = t0 + max(0.2, b - 0.5)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # If a parent construction is available, consider it, but do not depend on it.
    try:
        h_parent = np.asarray(initial_h_values, dtype=np.float64)
        if h_parent.size > 0:
            n_parent = int(h_parent.size)
            feasible = (
                np.isfinite(h_parent).all()
                and float(h_parent.min()) >= -1e-12
                and float(h_parent.max()) <= 1.0 + 1e-12
                and abs(float(h_parent.sum()) - 0.5 * n_parent) <= 1e-8
            )
            if feasible:
                h_test = np.clip(h_parent, 0.0, 1.0)
                c_test = compute_c5_direct(h_test)
                if c_test < best_c:
                    best_c = float(c_test)
                    best_h = h_test
                    best_n = n_parent
    except Exception:
        pass

    # Evaluate the simple V-shaped construction at several resolutions.
    # All sizes are powers of two, so h.sum() is exact for this construction.
    resolutions = (8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096)

    for n in resolutions:
        rem = deadline - time.time()
        if rem <= 0.05:
            break

        # Very conservative time guards for larger exact correlate calls.
        if n >= 4096 and rem < 1.0:
            break
        if n >= 2048 and rem < 0.3:
            break
        if n >= 1024 and rem < 0.1:
            break

        h = make_edge_v(n)
        h = np.clip(h, 0.0, 1.0)

        # Safety: if floating-point summation ever drifts, correct the tiny error.
        target = 0.5 * float(n)
        s = float(h.sum())
        if abs(s - target) > 1e-12:
            diff = target - s
            interior = (h > 1e-14) & (h < 1.0 - 1e-14)
            k = int(np.count_nonzero(interior))
            if k > 0:
                h[interior] += diff / float(k)
                h = np.clip(h, 0.0, 1.0)

        c = compute_c5_direct(h)
        if c < best_c - 1e-15:
            best_c = float(c)
            best_h = h
            best_n = int(n)

    # Fallback should never be needed, but guarantees a valid feasible return.
    if best_h is None:
        n_fb = 16
        best_h = make_edge_v(n_fb)
        best_h = np.clip(best_h, 0.0, 1.0)
        target = 0.5 * float(n_fb)
        diff = target - float(best_h.sum())
        if abs(diff) > 1e-12:
            interior = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
            k = int(np.count_nonzero(interior))
            if k > 0:
                best_h[interior] += diff / float(k)
                best_h = np.clip(best_h, 0.0, 1.0)
        best_c = compute_c5_direct(best_h)
        best_n = n_fb

    # Final exact value for the returned candidate.
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5_direct(best_h)

    return best_h, float(best_c), int(best_n)
