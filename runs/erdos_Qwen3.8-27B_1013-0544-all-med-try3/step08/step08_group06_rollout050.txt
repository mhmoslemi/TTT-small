        alpha_t = min(alpha_t, 0.5)

        h_new = h - alpha_t * subgrad
        h_new = project_feasible(h_new, n)

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()

        h = h_new
        t += 1

        if it > 200 and it % 200 == 0:
            # Check for stagnation
            alpha *= 0.95
            if alpha < 0.01:
                break

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    n_best = n_init

    # Phase 1: Coordinate descent (fast, exact 1D convex steps)
    alloc1 = min(80, (deadline - t0) * 0.12)
    best_h, best_c = coord_descent(best_h, n_init, max_passes=400,
                                   deadline=t0 + alloc1, random_order=True, rng=rng)
    n_best = n_init

    # Phase 2: SLSQP with increasing sharpness
    if time.time() < deadline - 40:
        for alpha in [3000, 20000, 100000]:
            if time.time() > deadline - 40:
                break
            ht = slsqp_optimize(best_h, n_best, alpha, maxiter=1500)
            c = compute_c5(ht, n_best)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht.copy()

    # Phase 3: Coordinate descent again (refined landscape)
    if time.time() < deadline - 80:
        alloc3 = min(100, (deadline - time.time()) * 0.2)
        best_h, best_c = coord_descent(best_h, n_best, max_passes=600,
                                       deadline=time.time() + alloc3,
                                       random_order=True, rng=rng)

    # Phase 4: Subgradient descent (explores different region)
    if time.time() < deadline - 80:
        alloc4 = min(60, (deadline - time.time()) * 0.2)
        ht, ct = subgradient_descent(best_h, n_best, max_iter=3000,
                                     deadline=time.time() + alloc4)
        if ct < best_c - 1e-14:
            best_c = ct
            best_h = ht.copy()

    # Phase 5: Pair search (fine-tuning)
    if time.time() < deadline - 80:
        alloc5 = min(150, (deadline - time.time()) * 0.3)
        if n_best <= 200:
            best_h, best_c = pair_search(best_h, n_best, time.time() + alloc5,
                                         [0.005, 0.002, 0.001, 0.0005])
        else:
            best_h, best_c = near_pair_search(best_h, n_best, time.time() + alloc5,
                                              [0.005, 0.002, 0.001], radius=40)

    # Phase 6: SLSQP polish
    if time.time() < deadline - 30:
        ht = slsqp_optimize(best_h, n_best, 500000, maxiter=3000)
        c = compute_c5(ht, n_best)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    # Phase 7: Try other resolutions
    if time.time() < deadline - 60:
        x_init = np.linspace(0, 2, n_best)
        for n_try in [100, 150, 250, 300]:
            if n_try == n_best or time.time() > deadline - 35:
                continue
            remaining = deadline - time.time() - 35
            if remaining < 30:
                break
            alloc = min(50, remaining * 0.35)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)

            ht, ct = coord_descent(h_interp, n_try, max_passes=200,
                                   deadline=time.time() + alloc * 0.5, rng=rng, random_order=True)
            if time.time() < deadline - 35:
                ht = slsqp_optimize(ht, n_try, 50000, maxiter=1000)
                ct = compute_c5(ht, n_try)
            if ct < best_c - 1e-14:
                best_c = ct
                best_h = ht.copy()
                n_best = n_try

    # Phase 8: Final coordinate descent + pair search
    if time.time() < deadline - 20:
        alloc8 = min(60, (deadline - time.time()) - 15)
        best_h, best_c = coord_descent(best_h, n_best, max_passes=300,
                                       deadline=time.time() + alloc8 * 0.5, rng=rng, random_order=True)

    if time.time() < deadline - 8:
        if n_best <= 200:
            best_h, best_c = pair_search(best_h, n_best, deadline - 3,
                                         [0.002, 0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = near_pair_search(best_h, n_best, deadline - 3,
                                              [0.002, 0.001, 0.0005], radius=30)

    # Phase 9: Final SLSQP polish
    if time.time() < deadline - 5:
        ht = slsqp_optimize(best_h, n_best, 2000000, maxiter=3000)
        c = compute_c5(ht, n_best)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 10: Final micro pair search
    if time.time() < deadline - 2:
        if n_best <= 150:
            best_h, best_c = pair_search(best_h, n_best, deadline - 1,
                                         [0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = near_pair_search(best_h, n_best, deadline - 1,
                                              [0.0005, 0.0002], radius=25)

    # Final projection and evaluation
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)

    # Multi-restart if time permits
    if time.time() < deadline - 30:
        n = n_best
        for r in range(5):
            if time.time() > deadline - 15:
                break
            scale = 0.03 + 0.05 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            ht, ct = coord_descent(hp, n, max_passes=100,
                                   deadline=deadline - 15, rng=rng, random_order=True)
            if time.time() < deadline - 15:
                ht = slsqp_optimize(ht, n, 100000, maxiter=1000)
                ct = compute_c5(ht, n)
            if ct < best_c - 1e-14:
                best_c = ct
                best_h = ht.copy()
                n_best = n

    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)

    return best_h, best_c, n_best
