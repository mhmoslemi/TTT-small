    # Phase 2: Spectral-inspired restarts at current resolution
    for restart in range(4):
        if time_left() < 40:
            break
        h_spec = project_h(best_h + np.random.randn(n) * 0.15, target)
        h_spec = spectral_flatness_step(h_spec, n_iter=80, alpha=0.3)
        h_spec = project_h(h_spec, target)
        c5_spec = compute_c5(h_spec)
        if c5_spec < best_c5:
            best_h = h_spec
            best_c5 = c5_spec
        for beta in [200, 1000, 3000, 8000]:
            if time_left() < 15:
                break
            pen_w = 80.0 * (1.0 + beta / 20.0)
            obj_fn = lgbfs_obj_factory(dx, beta, pen_w, target)
            try:
                res = minimize(obj_fn, best_h, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 1200, 'ftol': 1e-19})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 3: Resolution refinement
    for n_new in [300, 400, 500]:
        if time_left() < 80:
            break
        target_new = n_new / 2.0
        dx_new = 2.0 / n_new
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_interp = project_h(np.interp(x_new, x_old, best_h), target_new)
        c5_interp = compute_c5(h_interp)
        bounds_new = Bounds(np.zeros(n_new), np.ones(n_new))

        improved_here = False
        for attempt in range(3):
            if time_left() < 30:
                break
            start = h_interp.copy()
            if attempt == 1:
                start = project_h(h_interp + np.random.randn(n_new) * 0.02, target_new)
            elif attempt == 2:
                start = project_h(spectral_flatness_step(h_interp, 40, 0.3) +
                                  np.random.randn(n_new) * 0.01, target_new)

            for beta in [50, 200, 800, 3000, 8000, 20000]:
                if time_left() < 12:
                    break
                pen_w = 40.0 * (1.0 + beta / 15.0)
                obj_fn = lgbfs_obj_factory(dx_new, beta, pen_w, target_new)
                try:
                    res = minimize(obj_fn, start, method='L-BFGS-B', bounds=bounds_new,
                                   options={'maxiter': 1000, 'ftol': 1e-19})
                    h_try = project_h(res.x, target_new)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_interp:
                        h_interp = h_try
                        c5_interp = c5_try
                        improved_here = True
                except Exception:
                    pass

        if improved_here and c5_interp < best_c5:
            best_h = h_interp.copy()
            best_c5 = c5_interp

    # Phase 4: Subgradient fine-tuning
    n_f = len(best_h)
    target_f = n_f / 2.0
    no_improve = 0

    for it in range(80000):
        if time_left() < 3:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_k = min(5, max(3, n_f // 40))
        top_idx = np.argsort(corr)[::-1][:top_k]

        grad = np.zeros(n_f)
        for idx in top_idx:
            lag = int(idx) - (n_f - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * best_h)
            elif lag > 0:
                grad[lag:] += (1.0 - best_h[:-lag])
                grad[:n_f - lag] -= best_h[lag:]
            else:
                m = -lag
                grad[:n_f - m] += (1.0 - best_h[m:])
                grad[m:] -= best_h[:n_f - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm

        improved = False
        steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        for step in steps:
            h_try = project_h(best_h - step * grad, target_f)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 300:
                if no_improve % 200 == 0:
                    pert = np.random.randn(n_f) * 0.004
                    h_try = project_h(best_h + pert, target_f)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        best_h = h_try
                        best_c5 = c5_try
                        no_improve = 0
                if no_improve > 3000:
                    break

    # Phase 5: Final L-BFGS-B polish
    if time_left() > 10:
        n_p = len(best_h)
        target_p = n_p / 2.0
        dx_p = 2.0 / n_p
        bounds_p = Bounds(np.zeros(n_p), np.ones(n_p))
        for beta in [5000, 10000, 20000]:
            if time_left() < 8:
                break
            pen_w = 200.0
            obj_fn = lgbfs_obj_factory(dx_p, beta, pen_w, target_p)
            try:
                res = minimize(obj_fn, best_h, method='L-BFGS-B', bounds=bounds_p,
                               options={'maxiter': 3000, 'ftol': 1e-20})
                h_new = project_h(res.x, target_p)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 6: Final subgradient sweep
    for it in range(30000):
        if time_left() < 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_k = min(4, max(2, n_f // 50))
        top_idx = np.argsort(corr)[::-1][:top_k]

        grad = np.zeros(n_f)
        for idx in top_idx:
            lag = int(idx) - (n_f - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * best_h)
            elif lag > 0:
                grad[lag:] += (1.0 - best_h[:-lag])
                grad[:n_f - lag] -= best_h[lag:]
            else:
                m = -lag
                grad[:n_f - m] += (1.0 - best_h[m:])
                grad[m:] -= best_h[:n_f - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = project_h(best_h - step * grad, target_f)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
