    dx = 2.0 / n

    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(sh))) / beta

def sml_objective(h, beta, target, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    val = m + np.log(np.sum(np.exp(sh))) / beta
    penalty = lam * (np.sum(h) - target) ** 2
    return val + penalty

def sml_gradient(h, beta, target, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    ex = np.exp(sh)
    w = ex / ex.sum()

        if wl < 1e-16:
            continue
        grad[la:] += wl * (1.0 - h[:-la])
        grad[:-la] -= wl * h[la:]
    grad += 2.0 * lam * (np.sum(h) - target)
    return grad

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

    )

def coord_descent(h, target, n_sweeps, t0, budget):
    best_c5 = compute_c5(h)
    n = len(h)
    step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for sw in range(n_sweeps):

            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1.5:
                break

                    c = compute_c5(trial)

                        h = trial
                        best_c5 = c
                        improved = True
                        break
                if improved:
                    break

            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(sh))) / beta

def sml_objective(h, beta, target, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    val = m + np.log(np.sum(np.exp(sh))) / beta
    penalty = lam * (np.sum(h) - target) ** 2
    return val + penalty

def sml_gradient(h, beta, target, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    ex = np.exp(sh)
    w = ex / ex.sum()
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for la in range(1, n):
        wl = 2.0 * w[n - 1 + la]
        if wl < 1e-16:
            continue
        grad[la:] += wl * (1.0 - h[:-la])
        grad[:-la] -= wl * h[la:]
    grad += 2.0 * lam * (np.sum(h) - target)
    return grad

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def lbfgsb_optimize(h0, n, target, beta, lam, maxiter, t0, budget):
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        result = minimize(
            sml_objective, h0, method='L-BFGS-B',
            jac=sml_gradient, bounds=bounds,
            args=(beta, target, lam),
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50}
        )
        return project_h(result.x, target)
    except Exception:
        return h0

def lag_focus_descent(h, target, n_iter, top_k, step, t0, budget):
    n = len(h)
    best_c5 = compute_c5(h)
    mom = np.zeros(n)
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            la = abs(int(idx) - (n - 1))
            if la == 0:
                grad += (1.0 - 2.0 * h)
            else:
                grad[la:] += (1.0 - h[:-la])
                grad[:-la] -= h[la:]
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad /= gn
        mom = 0.82 * mom - step * grad
        trial = project_h(h + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-14:
            h = trial
            best_c5 = c_try
            stall = 0
        else:
            stall += 1
            mom *= 0.5
            trial = project_h(h + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-14:
                h = trial
                best_c5 = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 300:
            break
    return h, best_c5

def coord_descent(h, target, n_sweeps, t0, budget, rng):
    best_c5 = compute_c5(h)
    n = len(h)
    order = np.arange(n)
    for sw in range(n_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        rng.shuffle(order)
        improved = False
        for j in order:
            if budget - (time.time() - t0) < 1.0:
                break
            for step in [0.06, 0.03, 0.015, 0.008, 0.004]:
                for sign in [1, -1]:
                    trial = h.copy()
                    trial[j] = np.clip(trial[j] + sign * step, 0.0, 1.0)
                    trial = project_h(trial, target)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-13:
                        h = trial
                        best_c5 = c
                        improved = True
                        break
                if improved:
                    break
        if not improved and sw > 4:
            break
    return h, best_c5
