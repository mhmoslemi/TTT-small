def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = exps[k] / sw
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
        elif s < 0:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:n - t]
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    return val, grad * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-9) & (h < 1.0 - 1e-9)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)

    return project_feasible(best_h, n / 2.0), compute_c5(project_feasible(best_h, n / 2.0))

This **augmented Lagrangian / quadratic penalty** approach is often better than pure Lagrangian because the penalty term becomes stronger as we iterate, forcing the constraint to be satisfied.

But there's a problem: the penalty parameter ρ grows exponentially, and the objective becomes dominated by the penalty, making it hard to optimize the original objective. The standard approach is to alternate between increasing ρ and re-optimizing.

Actually, I think the cleanest approach for this problem is:

1. Use the **KKT conditions** directly. At the optimum:
   - ∂f/∂h[i] + λ + μ_i - ν_i = 0 for all i
   - where f is the smooth max, λ is the multiplier for sum(h) = n/2, μ_i ≥ 0 for h[i] ≥ 0, ν_i ≥ 0 for h[i] ≤ 1
   - μ_i * h[i] = 0, ν_i * (1 - h[i]) = 0

2. This means: for "free" variables (0 < h[i] < 1), we have ∂f/∂h[i] = -λ
   So all free variables should have the same partial derivative.

This suggests an **iterative water-filling** algorithm:
- Fix λ, then for each h[i], the optimal value satisfies ∂f/∂h[i] + λ = 0 (if free) or is at a boundary
- This is a fixed-point iteration

But computing ∂f/∂h[i] individually is expensive. Let me stick with the numerical optimization approach.

Let me finalize my code. I'll use:
1. L-BFGS-B with augmented penalty (quadratic penalty on constraint)
2. Progressive β
3. SLSQP for final polish
4. Multi-restart
5. Higher resolution

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_ctx = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    thresh = 1e-13
    for k in range(len(corr)):
        wk = exps[k] / sw
        if wk < thresh:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
        elif s < 0:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:n - t]
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    return val, grad * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-9) & (h < 1.0 - 1e-9)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def _obj_pen(x):
    beta = _ctx['beta']
    rho = _ctx['rho']
    target = _ctx['target']
    v, _ = smooth_max_grad(x, beta)
    cv = x.sum() - target
    return v + rho * cv * cv

def _grad_pen(x):
    beta = _ctx['beta']
    rho = _ctx['rho']
    target = _ctx['target']
    _, g = smooth_max_grad(x, beta)
    cv = x.sum() - target
    g = g + 2.0 * rho * cv
    return g

def lbfgs_opt(h0, beta, max_outer, max_inner, t_limit):
    n = len(h0)
    target = n / 2.0
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    _ctx['beta'] = beta
    _ctx['target'] = target
    bounds = [(0.0, 1.0)] * n
    rho = 1.0
    for outer in range(max_outer):
        if time.time() > t_limit:
            break
        _ctx['rho'] = rho
        try:
            res = minimize(_obj_pen, h, jac=_grad_pen, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-18, 'gtol': 1e-9})
            h = res.x
        except Exception:
            break
        h = np.clip(h, 0.0, 1.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        viol = abs(h.sum() - target)
        if viol < 1e-7:
            break
        rho *= 5.0
        if rho > 1e6:
            break
    return project_feasible(best_h, target), compute_c5(project_feasible(best_h, target))

def slsqp_obj(x):
    beta = _ctx['beta']
    v, _ = smooth_max_grad(x, beta)
    return v

def slsqp_grad(x):
    beta = _ctx['beta']
    _, g = smooth_max_grad(x, beta)
    return g

def slsqp_con(x):
    return x.sum() - _ctx['target']

def slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_refine(h0, beta, maxiter):
    n = len(h0)
    _ctx['beta'] = beta
    _ctx['target'] = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    try:
        res = minimize(slsqp_obj, h0, jac=slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h0.copy(), n / 2.0)

def nesterov_topk(h0, n_iters, lr0, decay, top_k, t_limit):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    x = h0.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_c5 = compute_c5(x)
    for t in range(n_iters):
        if time.time() > t_limit:
            break
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            if s >= 0:
                grad[s:] += (1.0 - y[:n - s])
                grad[:n - s] += (-y[s:])
            elif s < 0:
                tt = -s
                grad[:n - tt] += (1.0 - y[tt:])
                grad[tt:] += (-y[:n - tt])
            else:
                grad[:] += (1.0 - 2.0 * y)
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr_t * grad, target)
        c5 = compute_c5(x_new)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: L-BFGS with progressive beta (fast, many iterations)
    for beta in [30, 80, 200, 500, 1500]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new, c_new = lbfgs_opt(h_best, beta, max_outer=15, max_inner=300,
                                  t_limit=t0 + rem - 20)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 2: Nesterov top-k (explores landscape)
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_ng, c_ng = nesterov_topk(h_best, 4000, 0.03, 0.9999, top_k=6,
                                    t_limit=t0 + rem - 10)
        if c_ng < best_c5:
            h_best, best_c5 = h_ng, c_ng

    # Phase 3: SLSQP refinement
    for beta in [500, 2000, 8000]:
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        h_new = slsqp_refine(h_best, beta=beta, maxiter=400)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 4: Multi-restart with L-BFGS
    n = len(h_best)
    for i in range(12):
        rem = budget_s - (time.time() - t0)
        if rem < 30:
            break
        scale = 0.004 + 0.004 * (i % 6)
        h_pert = project_feasible(h_best + np.random.randn(n) * scale, n / 2.0)
        h_new, c_new = lbfgs_opt(h_pert, beta=800, max_outer=10, max_inner=250,
                                  t_limit=t0 + rem - 10)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new
            n = len(h_best)

    # Phase 5: Higher resolution
    if budget_s - (time.time() - t0) > 180:
        for n_try in [600, 800]:
            rem = budget_s - (time.time() - t0)
            if rem < 150:
                break
            h_interp = interp_to_n(h_best, n_try)
            for beta in [200, 800, 3000]:
                rem = budget_s - (time.time() - t0)
                if rem < 60:
                    break
                h_new, c_new = lbfgs_opt(h_interp, beta, max_outer=12, max_inner=250,
                                          t_limit=t0 + rem - 20)
                if c_new < best_c5:
                    h_best, best_c5 = h_new, c_new
                    h_interp = h_best.copy()
            n = len(h_best)
            # SLSQP at new resolution
            rem = budget_s - (time.time() - t0)
            if rem > 50:
                h_sls = slsqp_refine(h_best, beta=2000, maxiter=300)
                c_sls = compute_c5(h_sls)
                if c_sls < best_c5:
                    h_best, best_c5 = h_sls, c_sls

    # Phase 6: Final Nesterov + SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_ng2, c_ng2 = nesterov_topk(h_best, 3000, 0.02, 0.99995, top_k=4,
                                      t_limit=t0 + rem - 15)
        if c_ng2 < best_c5:
            h_best, best_c5 = h_ng2, c_ng2
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_fin = slsqp_refine(h_best, beta=10000, maxiter=500)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            h_best, best_c5 = h_fin, c_fin

    # Final feasibility
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
