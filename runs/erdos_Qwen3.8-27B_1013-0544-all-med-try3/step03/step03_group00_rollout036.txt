def project_feasible(h, target_sum):

            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=40):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-12:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def lbfgsb_opt(h, beta, top_k, maxiter, t0, budget_s):
    n = len(h)
    target = n / 2.0
    mu = 500.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        v, g = smoothmax_val_grad(x, beta, top_k)
        return v + mu * (x.sum() - target) ** 2
    
    def grad(x):
        v, g = smoothmax_val_grad(x, beta, top_k)
        return g + 2.0 * mu * (x.sum() - target)
    
    try:
        res = minimize(h, obj, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9, 'maxls': 50})
        h_out = project_feasible(res.x, target)
    except Exception:
        h_out = project_feasible(h, target)
    return h_out

def nesterov_subgrad(h, beta, top_k, n_iters, lr0, t0, budget_s):
    n = len(h)
    target = n / 2.0
    mu = 500.0
    y = h.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for t in range(n_iters):
        if budget_s - (time.time() - t0) < 5:
            break

        grad = grad + 2.0 * mu * (y.sum() - target)
        lr = lr0 / (1.0 + 0.05 * t)
        mom = min(0.9, 0.5 + 0.005 * t)

        y = y + v
        y = project_feasible(y, target)
        c5 = compute_c5(y)
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()
    return best_h, best_c5

    h_new = np.clip(h_new, 0.0, 1.0)
