    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def make_structured_pert(n, seed, trial):
    rng = np.random.RandomState(seed + trial * 1000)
    if trial % 4 == 0:
        return rng.randn(n) * 0.015
    elif trial % 4 == 1:
        freqs = rng.choice([2, 3, 5, 7, 11], 3, replace=False)
        p = np.zeros(n)
        xx = np.arange(n)
        for fq in freqs:
            p += 0.008 * np.sin(2.0 * np.pi * fq * xx / n + rng.uniform(0, 2*np.pi))
        return p
    elif trial % 4 == 2:
        nz = max(1, n // 16)
        p = np.zeros(n)
        idx = rng.choice(n, nz, replace=False)
        p[idx] = rng.uniform(-0.05, 0.05, nz)
        return p
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.01
        p[half:] = -0.01
        return p

def run(seed=42, budget_s=1100, **kwargs):
    global _g_t0, _g_budget
    np.random.seed(seed)
    _g_t0 = time.time()
    _g_budget = budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: L-BFGS-B with penalty, increasing beta
    for beta, top_k, lam in [(50, 40, 1e5), (100, 35, 1e6), (200, 30, 1e7), (500, 25, 1e7)]:
        if time_left() < 80:
            break
        h_new = lbfgsb_opt(h_best, beta, top_k, lam, maxiter=300)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: SLSQP refinement
    for beta, top_k in [(200, 30), (500, 25), (1000, 20)]:
        if time_left() < 70:
            break
        h_new = slsqp_opt(h_best, beta, top_k, maxiter=300)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 3: Accelerated projected gradient
    if time_left() > 150:
        h_sg, c_sg = accelerated_proj(h_best, max_iters=6000, top_k=16, lr0=0.03, decay=0.99992)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg
        if time_left() > 50:
            h_new = slsqp_opt(h_best, 800, 20, maxiter=200)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 4: Structured multi-start
    for trial in range(10):
        if time_left() < 70:
            break
        pert = make_structured_pert(n, seed, trial)
        h_pert = project_feasible(h_best + pert, n / 2.0)
        h_new = lbfgsb_opt(h_pert, 300, 30, 1e6, maxiter=200)
        c = compute_c5(h_new)
        if c < best_c5:
            h_new = slsqp_opt(h_new, 800, 25, maxiter=200)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 5: Resolution scaling
    for n_try in [600, 800]:
        if time_left() < 250:
            break
        h_interp = interp_to_n(h_best, n_try)
        for beta, top_k in [(100, 30), (400, 25)]:
            if time_left() < 80:
                break
            h_new = lbfgsb_opt(h_interp, beta, top_k, 1e6, maxiter=250)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()
        n = len(h_best)

    # Phase 6: More SLSQP at current resolution
    for beta, top_k in [(500, 20), (1000, 15), (2000, 15)]:
        if time_left() < 50:
            break
        h_new = slsqp_opt(h_best, beta, top_k, maxiter=300)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 7: Final L-BFGS-B polish
    if time_left() > 40:
        h_new = lbfgsb_opt(h_best, 1500, 20, 1e7, maxiter=400)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 8: Another round of perturbations at best resolution
    for trial in range(6):
        if time_left() < 50:
            break
        scale = 0.008 * (1.0 + 0.3 * trial)
        pert = np.random.randn(len(h_best)) * scale
        h_pert = project_feasible(h_best + pert, len(h_best) / 2.0)
        h_new = slsqp_opt(h_pert, 1000, 20, maxiter=250)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Final feasibility and evaluation
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
