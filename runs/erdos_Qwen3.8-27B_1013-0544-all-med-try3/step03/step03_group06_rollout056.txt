            _g_target = target_t
            cons_t = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
            try:
                res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds_t,
                              constraints=cons_t, options={'maxiter': 4000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_t)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5:
                    h_cur = h_new
                    cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = cur_c5
                    n_best = n_try
            except Exception:
                pass

    # Phase 3: Weighted subgradient descent
    n_final = len(best_h)
    target_final = n_final / 2.0
    no_improve = 0

    for it in range(40000):
        if time_left() < 15:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        n_act = min(15, max(5, n_final // 12))
        top_idx = np.argsort(corr)[::-1][:n_act]
        lags = [int(i) - (n_final - 1) for i in top_idx]
        # Weight by correlation value (normalized)
        top_corr = corr[top_idx].copy()
        top_corr = np.maximum(top_corr, 1e-10)
        wts = top_corr / top_corr.sum()
        wts = wts / max(wts.max(), 1e-15)

        grad = subgrad(best_h, lags, wts)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad /= gn

        found = False
        for step in [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0007, 0.0003]:
            ht = project_h(best_h - step * grad, target_final)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_h = ht
                best_c5 = ct
                found = True
                no_improve = 0
                break
        if not found:
            no_improve += 1
            if no_improve % 150 == 0:
                for sc in [0.012, 0.006, 0.003]:
                    ht = project_h(best_h + np.random.randn(n_final) * sc, target_final)
                    ct = compute_c5(ht)
                    if ct < best_c5:
                        best_h = ht
                        best_c5 = ct
                        no_improve = 0
                        break
            if no_improve > 2500:
                break

    # Phase 4: Pairwise moves targeting worst lags
    for it in range(120000):
        if time_left() < 8:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:5]
        improved = False
        for idx in top_idx:
            if time_left() < 2:
                break
            lag = int(idx) - (n_final - 1)
            if lag == 0:
                for _ in range(5):
                    j = np.random.randint(n_final)
                    for d in [0.012, -0.012, 0.006, -0.006, 0.003, -0.003]:
                        ht = best_h.copy()
                        ht[j] += d
                        if 0 <= ht[j] <= 1:
                            ht = project_h(ht, target_final)
                            ct = compute_c5(ht)
                            if ct < best_c5:
                                best_h = ht
                                best_c5 = ct
                                improved = True
                                break
                        if improved:
                            break
            else:
                al = abs(lag)
                if al >= n_final:
                    continue
                nv = n_final - al
                for _ in range(8):
                    i = np.random.randint(nv)
                    j = i + al if lag > 0 else i - lag
                    for d in [0.012, -0.012, 0.006, -0.006, 0.003, -0.003]:
                        ht = best_h.copy()
                        ht[i] += d
                        ht[j] -= d
                        if 0 <= ht[i] <= 1 and 0 <= ht[j] <= 1:
                            ht = project_h(ht, target_final)
                            ct = compute_c5(ht)
                            if ct < best_c5:
                                best_h = ht
                                best_c5 = ct
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
            if improved:
                break

    # Phase 5: Final high-precision SLSQP
    if time_left() > 15:
        n_pol = len(best_h)
        target_pol = n_pol / 2.0
        dx_pol = 2.0 / n_pol
        bounds_pol = Bounds(np.zeros(n_pol), np.ones(n_pol))
        cons_pol = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
        for beta in [10000, 20000, 50000]:
            if time_left() < 8:
                break
            _g_dx = dx_pol
            _g_beta = beta
            _g_target = target_pol
            try:
                res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds_pol,
                              constraints=cons_pol, options={'maxiter': 10000, 'ftol': 1e-18})
                h_new = project_h(res.x, target_pol)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 6: Ultra-fine subgradient steps
    for it in range(8000):
        if time_left() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        n_act = min(10, max(4, n_final // 15))
        top_idx = np.argsort(corr)[::-1][:n_act]
        lags = [int(i) - (n_final - 1) for i in top_idx]
        top_corr = corr[top_idx].copy()
        top_corr = np.maximum(top_corr, 1e-10)
        wts = top_corr / top_corr.sum()
        wts = wts / max(wts.max(), 1e-15)

        grad = subgrad(best_h, lags, wts)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        grad /= gn
        for step in [0.003, 0.0015, 0.0007, 0.0004, 0.0002, 0.0001, 5e-05]:
            ht = project_h(best_h - step * grad, target_final)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_h = ht
                best_c5 = ct
                break

    return best_h, best_c5, len(best_h)
