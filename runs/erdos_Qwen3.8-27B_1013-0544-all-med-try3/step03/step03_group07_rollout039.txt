```python
def augmented_lagrangian_optimize(h_init, beta, n_inner, lam, mu, max_outer):
    """
    Minimize smoothmax(h) subject to 0<=h<=1, sum(h)=n/2
    using augmented Lagrangian + L-BFGS-B
    """
    n = len(h_init)
    target = n / 2.0
    x = h_init.copy()
    
    for outer in range(max_outer):
        def obj(x):
            v, g = smoothmax_and_grad(x, beta)
            penalty = lam * (x.sum() - target) + 0.5 * mu * (x.sum() - target)**2
            return v + penalty
        
        def grad_obj(x):
            _, g = smoothmax_and_grad(x, beta)
            g = g + lam * np.ones(n) + mu * (x.sum() - target) * np.ones(n)
            return g
        
        res = minimize(obj, x, jac=grad_obj, method='L-BFGS-B',
                      bounds=[(0,1)]*n, options={'maxiter': n_inner, 'ftol': 1e-15})
        x = res.x
        constraint_viol = x.sum() - target
        lam = lam + mu * constraint_viol
        if abs(constraint_viol) > 0.1:
            mu = min(mu * 2, 1e6)
    
    return project_feasible(x, target)
```

```python
def project_onto_hypersimplex(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}"""
    n = len(h)
    h = h.copy()
    
    def project_simplex(v, s):
        """Project v onto {x: x >= 0, sum(x) = s}"""
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
        theta = (cssv[rho] - s) / (rho + 1.0)
        return np.maximum(v - theta, 0)
    
    # Bisection on the upper bound multiplier
    # We want: sum(max(min(h + t, 1), 0)) = target_sum
    # This is monotone in t, so bisection works
    
    lo, hi = -max(h) - 1, max(1 - h) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid

    if abs(diff) > 1e-14:

            h_proj = np.clip(h_proj, 0, 1)
    return h_proj
```

```python
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h + mid, 0.0, 1.0).sum() > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + mid, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            free = (h > 1e-12) & (h < 1.0 - 1e-12)
            if free.sum() > 0:
                h[free] += d2 / free.sum()
                h = np.clip(h, 0.0, 1.0)

    val = m + np.log(sw) / beta
    w = e / sw
    return val, w

    g = np.zeros(n)

        if wk < 1e-14:

            g[s:] += wk * (1.0 - h[:n - s])
            g[:n - s] -= wk * h[s:]
        else:
            t = -s
            g[:n - t] += wk * (1.0 - h[t:])
            g[t:] -= wk * h[:n - t]
    g *= dx
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h + mid, 0.0, 1.0).sum() > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + mid, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            free = (h > 1e-12) & (h < 1.0 - 1e-12)
            if free.sum() > 0:
                h[free] += d2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smoothmax_cv(h, corr, beta):
    m = np.max(corr)
    s = corr - m
    e = np.exp(beta * s)
    sw = e.sum()
    val = m + np.log(sw) / beta
    w = e / sw
    return val, w

def compute_grad(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[s:] += wk * (1.0 - h[:n - s])
            g[:n - s] -= wk * h[s:]
        else:
            t = -s
            g[:n - t] += wk * (1.0 - h[t:])
            g[t:] -= wk * h[:n - t]
    g *= dx
    return g

def nesterov_pg(h0, beta, n_iter, step):
    n = len(h0)
    target = n * 0.5
    x = h0.copy()
    x_prev = x.copy()
    best_x = h0.copy()
    best_v = compute_c5(h0)
    for it in range(n_iter):
        if it > 0:
            a = float(it) / (it + 2.0)
            y = x + a * (x - x_prev)
        else:
            y = x.copy()
        y = project_feasible(y, target)
        corr = get_corr(y)
        _, w = smoothmax_cv(y, corr, beta)
        g = compute_grad(y, w)
        x_new = project_feasible(y - step * g, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def lbfgs_refine(h, beta, maxiter):
    n = len(h)
    target = n * 0.5
    pen = 200.0
    def obj(x):
        corr = get_corr(x)
        v, _ = smoothmax_cv(x, corr, beta)
        c = x.sum() - target
        return v + pen * c * c
    def gfun(x):
        corr = get_corr(x)
        _, w = smoothmax_cv(x, corr, beta)
        g = compute_grad(x, w)
        c = x.sum() - target
        return g + 2.0 * pen * c
    res = minimize(obj, h, jac=gfun, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, target)

def interp_to_n(h, n_new):
    n = len(h)
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(xo, h, kind='cubic', fill_value=(0.0, 0.0), assume_sorted=True)
    h_new = np.clip(f(xn), 0.0, 1.0)
    return project_feasible(h_new, n_new * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: Accelerated projected gradient with increasing smoothness
    for beta in [150, 400, 800, 1500, 3000]:
        if tl() < 55:
            break
        h_new, c = nesterov_pg(h_best, beta, 7000, 35.0)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: L-BFGS polish
    for beta in [1500, 4000]:
        if tl() < 35:
            break
        h_new = lbfgs_refine(h_best, beta, 400)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 3: Diverse restarts with structured perturbations
    for i in range(16):
        if tl() < 50:
            break
        r = i % 4
        if r == 0:
            pert = np.random.randn(n) * 0.015
        elif r == 1:
            freq = np.random.randint(2, 10)
            phase = np.random.uniform(0, 2 * np.pi)
            pert = 0.025 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        elif r == 2:
            pert = np.random.randn(n) * 0.025
        else:
            blk = np.random.randint(5, 20)
            pert = np.random.randn(n // blk + 1) * 0.03
            pert = np.repeat(pert, blk)[:n]
        h_pert = project_feasible(h_best + pert, n * 0.5)
        h_new, c = nesterov_pg(h_pert, 1000, 5000, 30.0)
        if c < best_c5:
            h_best, best_c5 = h_new, c
        if tl() > 30 and c < best_c5 + 0.005:
            h_new = lbfgs_refine(h_new, 3000, 300)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 4: Higher resolution
    if tl() > 200:
        n_hi = 800
        h_hi = interp_to_n(h_best, n_hi)

            if c < c_hi:

            if tl() > 35:
                h_new = lbfgs_refine(h_hi, beta, 300)
                c = compute_c5(h_new)
                if c < c_hi:
                    h_hi, c_hi = h_new, c
        if c_hi < best_c5:
            h_best, best_c5 = h_hi, c_hi

    # Phase 5: Additional restarts at final resolution
    for i in range(6):
        if tl() < 45:

        h_new, c = nesterov_pg(h_pert, 2000, 4000, 20.0)
        if c < best_c5:

        if tl() > 25 and c < best_c5 + 0.003:

        c = compute_c5(h_new)
        if c < best_c5:

    h_best = project_feasible(h_best, len(h_best) * 0.5)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
```
