        # Backtracking line search
        step = 0.05 / (1.0 + 0.001 * it)
        improved = False
        for _ in range(30):

            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5

            T = 0.003 * np.exp(-it / 500.0)

            h_trial = project_h(cur + noise, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 + T * 2:

                cur_c5 = c5_trial

def coordinate_polish(h, n, target, dx, max_iters, time_fn, min_time, best_state):
    """Coordinate descent focusing on the worst lag."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    center = n - 1
    
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        worst_idx = int(np.argmax(corr_val))
        lag = worst_idx - center
        
        # For each coordinate, compute the optimal value given the current lag
        # A(lag) = sum_i h[i](1-h[i+lag]) (for valid i)
        # dA/dh[j] = (1-h[j+lag]) - h[j-lag] (for valid indices)
        # To minimize A, if dA/dh[j] > 0, decrease h[j]; if < 0, increase h[j]
        
        # Batch update: try to move each coordinate in the direction that reduces A
        # But we need to maintain sum(h) = target
        # Strategy: compute desired updates, then project to maintain sum
        
        dA = grad_for_lag(cur, lag, n, dx)  # gradient of A(lag) w.r.t. h
        # We want to decrease A, so move h in direction -dA
        # But maintain sum constraint: subtract mean
        dA_centered = dA - np.mean(dA)
        
        # Step size
        step = 0.02
        for _ in range(20):
            h_trial = project_h(cur - step * dA_centered, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
    
    return cur, cur_c5, n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad


def smooth_max_obj(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(x_shift)) / len(c)) / beta) * dx
