```python
def build_lp_matrices(h_t, n):
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    # A is (n_shifts) x (n+1), b is (n_shifts)
    # Variable: [h[0],...,h[n-1], t]
    
    A = np.zeros((n_shifts, n + 1))
    b = np.zeros(n_shifts)
    
    for m in range(n_shifts):
        k = m - (n - 1)
        # Coefficients for h[j]
        # j+k valid: j in [max(0,-k), min(n-1, n-1-k)]
        # j-k valid: j in [max(0,k), min(n-1, n-1+k)]
        
        coeff = np.zeros(n)
        if k >= 0:
            # j+k valid for j in [0, n-1-k]
            coeff[:n-k] += (1.0 - h_t[k:n])
            # j-k valid for j in [k, n-1]
            coeff[k:] -= h_t[:n-k]
        else:
            # k < 0, let kk = -k
            kk = -k
            # j+k = j-kk valid for j in [kk, n-1]
            coeff[kk:] += (1.0 - h_t[:n-kk])
            # j-k = j+kk valid for j in [0, n-1-kk]
            coeff[:n-kk] -= h_t[kk:]
        
        A[m, :n] = coeff * dx
        A[m, n] = -1.0
        
        # constant
        if k >= 0:
            const = np.sum(h_t[:n-k] * h_t[k:n]) * dx
        else:
            kk = -k
            const = np.sum(h_t[kk:] * h_t[:n-kk]) * dx
        b[m] = -const
    
    return A, b
```
