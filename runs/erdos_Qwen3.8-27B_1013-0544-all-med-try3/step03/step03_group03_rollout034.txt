import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def grad_smooth(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    weights = exp_s / total
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        k_minus_j = k - j_range
        grad_h[j_range] += w * (1.0 - 2.0 * h[k_minus_j])
    return grad_h * dx

def sum_constraint_fun(h, target_sum):
    return h.sum() - target_sum

def sum_constraint_jac(h, target_sum):
    return np.ones(len(h))

def slsqp_optimize(h0, dx, beta, target_sum, maxiter=3000):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint_fun, 'jac': sum_constraint_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                       jac=grad_smooth, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h0

def pair_swap_search(h, dx, n_swaps=5000, time_limit=5.0, t0=0.0):
    """Try random pair adjustments to reduce C5."""
    n = len(h)
    best_c5 = compute_c5(h)
    for _ in range(n_swaps):
        if time.time() - t0 > time_limit:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        for eps in [0.05, 0.02, 0.01, -0.05, -0.02, -0.01]:
            h_try = h.copy()
            h_try[i] += eps
            h_try[j] -= eps
            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            c = compute_c5(h_try)
            if c < best_c5 - 1e-8:
                h = h_try
                best_c5 = c
                break
    return h, best_c5
