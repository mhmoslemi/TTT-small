def subgradient_step(h, target_sum, n_top=5, steps=None):
    """Subgradient descent step focusing on top lags."""
    if steps is None:
        steps = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005]
    n = len(h)
    dx = 2.0 / n
    c = compute_corr(h) * dx
    m = np.max(c)
    # Softmax weights for top lags
    top_idx = np.argsort(c)[::-1][:n_top]
    top_vals = c[top_idx]
    w_top = np.exp((top_vals - np.max(top_vals)) / 0.001)
    w_top /= w_top.sum()
    
    grad = np.zeros(n)
    for weight, idx in zip(w_top, top_idx):
        k = int(idx) - (n - 1)
        if k == 0:
            grad += weight * (1.0 - 2.0 * h)
        elif k > 0:
            valid = n - k
            grad[:valid] += weight * (1.0 - h[k:])
            grad[k:] -= weight * h[:valid]
        else:
            mk = -k
            valid = n - mk
            grad[mk:] += weight * (1.0 - h[:valid])
            grad[:valid] -= weight * h[mk:]
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-12:
        return h, compute_c5(h), False
    grad /= gnorm
    
    best_c5 = compute_c5(h)
    for step in steps:
        h_trial = project_h(h - step * grad, target_sum)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5 - 1e-14:
            return h_trial, c5_t, True
    return h, best_c5, False

def pairwise_refine(h, target_sum, n_rand=8):
    """Random pairwise moves to reduce C5."""
    n = len(h)
    best_c5 = compute_c5(h)
    corr = compute_corr(h) * (2.0 / n)
    top_idx = np.argsort(corr)[::-1][:3]
    
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        abs_lag = abs(lag)
        if abs_lag >= n:
            continue
        n_valid = n - abs_lag
        for _ in range(n_rand):
            i = np.random.randint(n_valid)
            j = i + abs_lag
            for d in [0.02, -0.02, 0.01, -0.01, 0.03, -0.03, 0.015, -0.015]:
                h_trial = h.copy()
                h_trial[i] += d
                h_trial[j] -= d
                if 0 <= h_trial[i] <= 1 and 0 <= h_trial[j] <= 1:
                    h_trial = project_h(h_trial, target_sum)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        h = h_trial.copy()
                        best_c5 = c5_t
                        break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_and_weights(c, beta):
    m = np.max(c)
    s = np.clip(beta * (c - m), -500, 500)
    e = np.exp(s)
    se = np.sum(e)
    weights = e / se
    val = m + np.log(se) / beta
    return val, weights


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)


def obj_fn(x, dx, beta, target, pen):

    val, _ = smooth_max_and_weights(c, beta)
    v = np.sum(x) - target
    return val * dx + pen * v * v


def grad_fn(x, dx, beta, target, pen):
