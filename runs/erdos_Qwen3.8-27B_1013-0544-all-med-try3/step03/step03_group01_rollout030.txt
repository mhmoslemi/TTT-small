import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    a = s.max()
    return m + (a + np.log(np.sum(np.exp(s - a)))) / alpha

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    a = s.max()
    ev = np.exp(s - a)
    w = ev / ev.sum()
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = lags[k]
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * x[lo2 + lag:hi2 + lag]
    grad *= dx
    return grad

def active_set_obj(x, n, alpha, active_idx):
    dx = 2.0 / n
    c_full = np.correlate(x, 1.0 - x, mode='full')
    c_a = c_full[active_idx] * dx
    m = c_a.max()
    s = alpha * (c_a - m)
    a = s.max()
    return m + (a + np.log(np.sum(np.exp(s - a)))) / alpha

def active_set_grad(x, n, alpha, active_idx):
    dx = 2.0 / n
    c_a = np.correlate(x, 1.0 - x, mode='full')[active_idx] * dx
    m = c_a.max()
    s = alpha * (c_a - m)
    a = s.max()
    ev = np.exp(s - a)
    w = ev / ev.sum()
    lags = active_idx - (n - 1)
    grad = np.zeros(n)
    for idx in range(len(active_idx)):
        wk = w[idx]
        if wk < 1e-14:
            continue
        lag = lags[idx]
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * x[lo2 + lag:hi2 + lag]
    grad *= dx
    return grad

def lbfbs_optimize(h, n, alpha, maxiter, deadline):
    target = n / 2.0
    lam = 500.0
    def obj(x):
        return smooth_max_val(x, n, alpha) + lam * (x.sum() - target) ** 2
    def obj_grad(x):
        return smooth_max_grad(x, n, alpha) + 2.0 * lam * (x.sum() - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def active_optimize(h, n, alpha, top_k, maxiter, deadline):
    target = n / 2.0
    lam = 500.0
    active_idx = top_k
    def obj(x):
        return active_set_obj(x, n, alpha, active_idx) + lam * (x.sum() - target) ** 2
    def obj_grad(x):
        return active_set_grad(x, n, alpha, active_idx) + 2.0 * lam * (x.sum() - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def fourier_optimize(n, K, deadline, rng):
    i = np.arange(n)
    basis = []
    for k in range(1, K + 1):
        basis.append(np.cos(2.0 * np.pi * k * i / n))
        basis.append(np.sin(2.0 * np.pi * k * i / n))
    basis = np.array(basis)  # shape (n, 2K)
    
    def params_to_h(params):
        return 0.5 + basis @ params
    
    best_params = None
    best_c = float('inf')
    
    for trial in range(30):
        if time.time() > deadline - 10:
            break
        scale = 0.1 + 0.4 * (trial / 30.0)
        params = rng.normal(0, scale, 2 * K)
        h = params_to_h(params)
        if h.min() < -0.01 or h.max() > 1.01:
            params *= 0.5
            h = params_to_h(params)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_params = h.copy()
    
    if best_params is None:
        return None, float('inf')
    
    # Refine with L-BFGS-B on h directly
    for alpha in [1000, 5000, 20000]:
        if time.time() > deadline - 5:
            break
        h_new = lbfbs_optimize(best_params, n, alpha, 2000, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_params = h_new
    
    return best_params, best_c

def local_refine(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    target = n / 2.0
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(2000):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for s in (1.0, -1.0):
                hi = h[i] + s * d
                hj = h[j] - s * d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        break
                    else:
                        h[i], h[j] = h[i] - s * d, h[j] + s * d
    return h, best_c
