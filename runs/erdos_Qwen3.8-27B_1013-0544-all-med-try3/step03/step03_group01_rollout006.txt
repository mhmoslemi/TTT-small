def project_feasible(v, n):
    """Project v onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on threshold
    lo, hi = -2.0, 2.0
    for _ in range(100):

        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid

    dx = 2.0 / n

    alpha = 200.0
    step = 0.05
    obj_curr, _ = smooth_obj_and_grad(h, n, alpha, ctx)
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj_curr, grad = smooth_obj_and_grad(h, n, alpha, ctx)
        s = step
        for bt in range(15):
            h_trial = project_feasible(h - s * grad, n)
            obj_trial, _ = smooth_obj_and_grad(h_trial, n, alpha, ctx)
            if obj_trial <= obj_curr - 1e-5 * s * s * (grad * grad).sum():
                break
            s *= 0.5
        h_new = project_feasible(h - s * grad, n)
        c_exact = compute_c5(h_new, n)
        if c_exact < best_c - 1e-15:
            best_c = c_exact
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        h = h_new
        obj_curr, _ = smooth_obj_and_grad(h, n, alpha, ctx)
        step = min(s * 1.3, 0.5)
        if it % 400 == 399:
            alpha = min(alpha * 2.5, max_alpha)
            step = min(step * 0.4, 0.02)
            obj_curr, _ = smooth_obj_and_grad(h, n, alpha, ctx)
        if stagnant > 3000 and alpha >= max_alpha * 0.5:
            break
    return best_h, best_c

def pair_local_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001]
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        for _pass in range(2):
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if time.time() > deadline:
                        return h, best_c
                    for sgn in (1.0, -1.0):
                        hi = h[i] + sgn * d
                        hj = h[j] - sgn * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c:
                                best_c = c
                            else:
                                h[i], h[j] = h[i] - sgn * d, h[j] + sgn * d
    return h, best_c

def diverse_starts(n, rng, h_parent):
    starts = []
    if h_parent is not None and len(h_parent) == n:
        starts.append(h_parent.copy())
    x = np.linspace(0, 2, n)
    starts.append(np.full(n, 0.5))
    for k in range(1, 12):
        amp = 0.25 + 0.1 * np.cos(0.3 * k)
        fn = 0.5 + amp * np.sin(k * np.pi * x / 2.0) + 0.08 * np.sin(3 * k * np.pi * x / 2.0)
        starts.append(project_feasible(fn, n))
    for _ in range(12):
        starts.append(project_feasible(rng.uniform(0, 1, n), n))
    for period in range(2, min(n, 20)):
        pat = np.array([1.0 if (i % period) < period // 2 else 0.0 for i in range(n)])
        starts.append(project_feasible(pat, n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 100
    best_h_overall = None
    best_c_overall = float('inf')
    # Phase 1: Main optimization from parent
    if h_parent is not None:
        dl1 = t0 + (deadline - t0) * 0.50
        h1, c1 = pgd_optimize(h_parent, n_parent, dl1)
        if c1 < best_c_overall:
            best_c_overall = c1
            best_h_overall = (h1, n_parent)
    # Phase 2: Try alternative n and starts
    dl2 = t0 + (deadline - t0) * 0.78
    n_alts = [60, 80, n_parent, 150, 200]
    n_alts = sorted(set(nv for nv in n_alts if nv <= 250))
    remaining_starts = len(n_alts)
    for n_try in n_alts:
        if time.time() > dl2:
            break
        sub_dl = min(dl2, time.time() + (dl2 - time.time()) / max(1, remaining_starts) * 1.2)
        remaining_starts -= 1
        x = np.linspace(0, 2, n_try)
        if h_parent is not None and n_try == n_parent:
            h_start = h_parent.copy()
        else:
            h_start = project_feasible(0.5 + 0.3 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x), n_try)
        h_r, c_r = pgd_optimize(h_start, n_try, sub_dl)
        if c_r < best_c_overall:
            best_c_overall = c_r
            best_h_overall = (h_r, n_try)
    # Phase 3: Multi-restart refinement on best n
    if best_h_overall is not None:
        h_best, n_best = best_h_overall
        dl3 = t0 + (deadline - t0) * 0.90
        starts = diverse_starts(n_best, rng, h_best)
        for s in starts[:8]:
            if time.time() > dl3:
                break
            sub_dl = min(dl3, time.time() + 25)
            h_r, c_r = pgd_optimize(s, n_best, sub_dl)
            if c_r < best_c_overall:
                best_c_overall = c_r
                best_h_overall = (h_r, n_best)
    # Phase 4: Pair local search + final PGD
    if best_h_overall is not None:
        h_best, n_best = best_h_overall
        dl4 = deadline - 15
        h_best, c4 = pair_local_search(h_best, n_best, dl4, [0.005, 0.002, 0.001, 0.0005])
        if c4 < best_c_overall:
            best_c_overall = c4
            best_h_overall = (h_best, n_best)
        if time.time() < deadline - 10:
            h_f, c_f = pgd_optimize(best_h_overall[0], best_h_overall[1], deadline,
                                    max_alpha=5000000, max_iter=3000)
            if c_f < best_c_overall:
                best_c_overall = c_f
                best_h_overall = (h_f, best_h_overall[1])
    if best_h_overall is not None:
        h_out = project_feasible(best_h_overall[0], best_h_overall[1])
        c_out = compute_c5(h_out, best_h_overall[1])
        return h_out, c_out, best_h_overall[1]
    n = n_parent
    h_out = project_feasible(np.full(n, 0.5), n)
    return h_out, compute_c5(h_out, n), n
