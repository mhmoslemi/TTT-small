    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    else:
        la = -k
        grad[la:] += (1.0 - h[:n-la])
        grad[:n-la] -= h[la:]
    return grad

def active_subgrad(h0, target, n, max_iter=2000, t0=0, budget=100):
    """Subgradient method focusing on active (top) lags with adaptive step"""
    dx = 2.0 / n
    h = h0.copy()
    best_c = compute_c5(h)
    step = 0.05
    
    for it in range(max_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Top 3 lags
        top = np.argsort(corr)[::-1][:3]
        
        # Gradient = average of gradients at top lags (subgradient of max)
        grad = np.zeros(n)
        for idx in top:
            grad += corr_grad(h, idx, n)
        grad /= len(top)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        
        # Adaptive step: try to decrease C by ~step * gnorm
        alpha = step / gnorm if gnorm > 0 else step
        best_trial = None
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            trial = project_h(h - alpha * scale * grad, target)
            c_try = compute_c5(trial)
            if c_try < best_c:
                best_trial = trial
                best_c = c_try
                break
        if best_trial is None:
            step *= 0.7
            if step < 1e-8:
                break
        else:
            h = best_trial
            step = min(step * 1.1, 0.1)
    
    return h, best_c

def penalty_refine(h0, target, n, C_target, max_iter=500, t0=0, budget=100):
    """Penalty method to try to achieve C5 <= C_target"""
    dx = 2.0 / n
    h = h0.copy()
    
    for it in range(max_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        violations = np.maximum(corr - C_target, 0)
        penalty = np.sum(violations**2)
        if penalty < 1e-18:
            return h, compute_c5(h), True
        
        # Gradient of penalty
        grad = np.zeros(n)
        active_mask = violations > 1e-12
        active_idx = np.where(active_mask)[0]
        for idx in active_idx:
            grad += 2.0 * violations[idx] * corr_grad(h, idx, n) * dx
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        alpha = 1.0 / (gnorm + 1e-12)
        success = False
        for _ in range(30):
            trial = project_h(h - alpha * grad, target)
            new_corr = np.correlate(trial, 1.0 - trial, mode="full") * dx
            new_viol = np.sum(np.maximum(new_corr - C_target, 0)**2)
            if new_viol < penalty * (1 - 1e-4):
                h = trial
                penalty = new_viol
                success = True
                break
            alpha *= 0.5
        
        if not success:
            # Try smaller step
            trial = project_h(h - 0.01 * grad / gnorm, target)
            new_corr = np.correlate(trial, 1.0 - trial, mode="full") * dx
            new_viol = np.sum(np.maximum(new_corr - C_target, 0)**2)
            if new_viol < penalty:
                h = trial
                penalty = new_viol
            else:
                break
    
    return h, compute_c5(h), False

def fine_coord_descent(h0, target, n, t0=0, budget=100):
    """Very fine coordinate descent"""
    h = h0.copy()
    best_c = compute_c5(h)
    
    # Multi-scale steps
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

                    if c_try < best_c - 1e-15:
                        h = trial
                        best_c = c_try
                        improved = True
                        break
    return h, best_c

def smax_obj(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(beta * (c - m))) / len(c)) / beta

def smax_grad(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    exps = np.exp(beta * (c - m))
    weights = exps / np.sum(exps)
    grad = np.zeros(len(h))
    # Only compute for non-negligible weights
    active = weights > 0.01
    for idx in np.where(active)[0]:
        grad += weights[idx] * corr_grad(h, idx, len(h))
    grad *= dx
    return grad

def slsqp_optimize(h0, n, target, beta, maxiter=2000):
    def obj(x):
        return smax_obj(x, beta, 2.0/n)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except:
        return h0
