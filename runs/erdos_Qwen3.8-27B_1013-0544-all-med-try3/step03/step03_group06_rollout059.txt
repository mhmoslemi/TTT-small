def compute_c5(h):

    dx = 2.0 / n

def project_h(h, target):

        c = np.correlate(x, 1.0 - x, mode="full") * dx
        cm = c.max()
        e = np.exp(np.clip(beta * (c - cm), -500, 500))
        sm = cm + np.log(e.sum() / (2 * n - 1)) / beta
        return sm + pen * (x.sum() - target) ** 2

        c = np.correlate(x, 1.0 - x, mode="full") * dx
        cm = c.max()
