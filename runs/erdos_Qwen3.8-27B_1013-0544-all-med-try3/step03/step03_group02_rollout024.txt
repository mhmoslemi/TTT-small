    x = np.clip(beta * (corr - m), -500, 500)
    exp_x = np.exp(x)
    s = exp_x.sum()
    w = exp_x / s
    val = m + np.log(s / len(corr)) / beta
    return val, w

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def c5_obj_grad(h, beta, target, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, w = smooth_max_w(corr, beta)
    fval = val * dx + pen * (h.sum() - target) ** 2
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wi = w[m]
        if wi < 1e-10:
            continue
        k = m - (n - 1)
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[k:] += wi * (1.0 - h[:nk])
                grad[:nk] -= wi * h[k:]
        else:
            la = -k
            nk = n - la
            if nk > 0:
                grad[la:] += wi * (1.0 - h[:nk])
                grad[:nk] -= wi * h[la:]
    grad *= dx
    grad += 2.0 * pen * (h.sum() - target)
    return fval, grad

def make_pattern(n, seed):

        if on:

        pos += bl
        on = 1 - on
    return project_h(h, n / 2.0)
