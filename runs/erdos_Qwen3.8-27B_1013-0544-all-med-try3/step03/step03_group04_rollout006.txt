        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

def corr_grad_lag(h, lag):
    """Gradient of correlation at given lag with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n - lag] = (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
    else:
        l = -lag
        grad[l:] = (1.0 - h[:n - l])
        grad[:n - l] -= h[l:]
    return grad

def projected_grad_step(h, target, dx, lr, top_k, beta):
    """One step of projected gradient descent on smooth max of top-k correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Get top-k lags
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_vals = corr[top_idx]
    
    # Softmax weights over top-k
    top_shift = top_vals - np.max(top_vals)
    weights = np.exp(beta * top_shift)
    weights /= weights.sum()
    
    # Accumulate gradient
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        lag = int(idx) - (n - 1)
        grad += w * corr_grad_lag(h, lag)
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_h(h_new, target)
    return h_new, corr

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_single(h, lag):
    n = len(h)
    if lag == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] = 1.0 - h[lag:]
        grad[lag:] -= h[:n - lag]
    else:
        l = -lag
        grad[l:] = 1.0 - h[:n - l]
        grad[:n - l] -= h[l:]
    return grad

def gradient_topk(h, top_k, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_vals = corr[top_idx]
    shift = top_vals - np.max(top_vals)
    weights = np.exp(beta * shift)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        lag = int(idx) - (n - 1)
        grad += w * corr_grad_single(h, lag)
    return grad, corr

def smooth_max_obj(x, dx, top_k, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    if top_k >= len(corr):
        tv = corr
    else:
        tv = np.partition(corr, -top_k)[-top_k:]
    m = np.max(tv)
    s = np.clip(beta * (tv - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s)) / len(tv)) / beta) * dx

def try_binary(h, target):
    best_h, best_c5 = None, compute_c5(h)[0]
    for t in np.linspace(0.15, 0.85, 29):
        hb = (h >= t).astype(float)
        deficit = target - hb.sum()
        if abs(deficit) < 1e-6:
            c5_b, _ = compute_c5(hb)
            if c5_b < best_c5:
                best_h, best_c5 = hb.copy(), c5_b
            continue
        if deficit > 0:
            zeros = np.where(hb == 0)[0]
            order = zeros[np.argsort(h[zeros])[::-1]]
            k = int(np.ceil(deficit))
            if k <= len(order):
                hb[order[:k]] = 1.0
        else:
            ones = np.where(hb == 1)[0]
            order = ones[np.argsort(h[ones])]
            k = int(np.ceil(-deficit))
            if k <= len(order):
                hb[order[:k]] = 0.0
        if abs(hb.sum() - target) < 1e-6:
            c5_b, _ = compute_c5(hb)
            if c5_b < best_c5:
                best_h, best_c5 = hb.copy(), c5_b
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def _slsqp_con(x, target):
    return x.sum() - target

def _slsqp_jac(x):
    return np.ones(x.shape)

    resolutions = [r for r in resolutions if 100 <= r <= 1200]

        cur_c5 = compute_c5(h_cur)[0]
