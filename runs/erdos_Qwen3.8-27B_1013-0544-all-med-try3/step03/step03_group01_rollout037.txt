    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

    return np.correlate(h.astype(float), (1 - h).astype(float), mode='full')

def make_binary_from_continuous(h_cont, n):

    threshold = np.sort(h_cont)[n//2]
