from scipy.interpolate import interp1d
import time

# Global state for SLSQP
_g_n = 0
_g_beta = 100.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    """Project h onto [0,1]^n ∩ {sum = target_sum}"""
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
        else:
            free = h > 1e-12
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    """Compute smooth max approximation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx

    for k in range(len(corr)):

            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += wk * (-h[s:])
        elif s < 0:
            t = -s
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n-t])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

    return g

    _g_beta = beta
    target = _g_n / 2.0

                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:

def topk_subgrad_nesterov(h, max_iters, lr0, decay, top_k):
    """Nesterov-accelerated subgradient using only top-k lags."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
            y = project_feasible(y, target)

            elif s < 0:

        if v < best_v:

        x_prev = x
        x = x_new

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_g_n = 0
_g_beta = 100.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
        else:
            free = h > 1e-12
        if free.sum() == 0:
            break
        step = diff / free.sum()
        h[free] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] += wk * (-h[s:])
        elif s < 0:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n - t])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def _obj(x):
    v, _ = smoothmax_val_grad(x, _g_beta)
    return v

def _grad(x):
    _, g = smoothmax_val_grad(x, _g_beta)
    return g

def _con_fun(x):
    return x.sum() - _g_n / 2.0

def _con_jac(x):
    return np.ones(len(x))

def slsqp_refine(h, maxiter=400, beta=500):
    global _g_n, _g_beta
    _g_n = len(h)
    _g_beta = float(beta)
    target = _g_n / 2.0
    bounds = Bounds(np.zeros(_g_n), np.ones(_g_n))
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def topk_subgrad(h, max_iters, lr0, top_k, nesterov=True):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy() if nesterov else None
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (1.0 / (1.0 + 0.001 * t))
        if nesterov and t > 0:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
            y = project_feasible(y, target)
        else:
            y = x
        corr = compute_corr(y)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            if s >= 0:
                grad[s:] += (1.0 - y[:n - s])
                grad[:n - s] += (-y[s:])
            elif s < 0:
                t2 = -s
                grad[:n - t2] += (1.0 - y[t2:])
                grad[t2:] += (-y[:n - t2])
            else:
                grad[:] += (1.0 - 2.0 * y)
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        if nesterov:
            x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    def tl():
