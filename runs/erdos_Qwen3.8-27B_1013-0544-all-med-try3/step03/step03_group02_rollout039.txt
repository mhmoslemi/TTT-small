                idx2 = np.arange(lag, n)

            # Negative part: -h[j+la] for j in [0, n-1-la]
            idx2 = np.arange(0, n - la)
            grad[idx2] -= w * h[idx2 + la]
    
    # Add penalty gradient
    diff = np.sum(h) - target
    grad += 2.0 * penalty * diff
    
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sm_obj(h, beta, target, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(sh)) / len(corr)) / beta
    diff = h.sum() - target
    return sm * dx + pen * diff * diff

def sm_grad(h, beta, target, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    ex = np.exp(sh)
    w = ex / ex.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - 1 - lag >= 0:
                i1 = np.arange(0, n - lag)
                grad[i1] += wk * (1.0 - h[i1 + lag])
            if lag <= n - 1:
                i2 = np.arange(lag, n)
                grad[i2] -= wk * h[i2 - lag]
        else:
            la = -lag
            i1 = np.arange(la, n)
            grad[i1] += wk * (1.0 - h[i1 - la])
            i2 = np.arange(0, n - la)
            grad[i2] -= wk * h[i2 + la]
    diff = h.sum() - target
    grad += 2.0 * pen * diff
    return grad

def lbfgs_opt(h0, target, beta, pen, maxiter, t0, budget):
    n = len(h0)
    bnd = Bounds(np.zeros(n), np.ones(n))
    try:
        r = minimize(sm_obj, h0, jac=lambda x: sm_grad(x, beta, target, pen),
                     args=(beta, target, pen),
                     method='L-BFGS-B', bounds=bnd,
                     options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e12})
        return project(r.x, target)
    except Exception:
        return h0

def coord_descent(h, target, t0, budget, max_sweeps=50):
    n = len(h)
    best = compute_c5(h)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            # Try several step sizes
            for step in (0.05, -0.05, 0.02, -0.02, 0.01, -0.01):
                trial = h.copy()
                trial[j] = np.clip(trial[j] + step, 0.0, 1.0)
                trial = project(trial, target)
                c = compute_c5(trial)
                if c < best - 1e-13:
                    h = trial
                    best = c
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return h, best

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)
    
    # Resolutions to try
    res_list = [best_n]
    for r in [240, 280, 320, 360, 400, 440]:
        if r not in res_list:
            res_list.append(r)
    res_list = sorted(set(res_list))
    
    for n_pts in res_list:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        target = n_pts / 2.0
        
        cur = project(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
        
        # L-BFGS-B with progressive beta
        for beta in [50, 100, 200, 400, 800]:
            rem = budget_s - (time.time() - t0)
            if rem < 20:
                break
            pen = 1000.0
            cur = lbfgs_opt(cur, target, beta, pen, 500, t0, budget_s)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
                cur_c5 = best_c5
        
        # Coordinate descent
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur, cur_c5 = coord_descent(cur, target, t0, budget_s, max_sweeps=30)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
        
        # Random restarts
        rem = budget_s - (time.time() - t0)
        if rem > 40:
            for att in range(3):
                rem = budget_s - (time.time() - t0)
                if rem < 15:
                    break
                scale = 0.03 / (att + 1)
                hp = project(cur + np.random.randn(n_pts) * scale, target)
                hp = lbfgs_opt(hp, target, 300, 1000.0, 300, t0, budget_s)
                hp_c5 = compute_c5(hp)
                if hp_c5 < best_c5:
                    best_h, best_c5, best_n = hp.copy(), hp_c5, n_pts
                    cur = hp
    
    # Final polish at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        nf = len(best_h)
        tf = nf / 2.0
        for beta in [500, 1000, 2000]:
            rem = budget_s - (time.time() - t0)
            if rem < 5:
                break
            best_h = lbfgs_opt(best_h, tf, beta, 5000.0, 1000, t0, budget_s)
            best_c5 = compute_c5(best_h)
        rem = budget_s - (time.time() - t0)
        if rem > 5:
            best_h, best_c5 = coord_descent(best_h, tf, t0, budget_s, max_sweeps=40)
    
    return best_h, best_c5, best_n

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(h, beta, target, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(sh)) / len(corr)) / beta
    diff = h.sum() - target
    return sm * dx + pen * diff * diff

def _grad(h, beta, target, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    ex = np.exp(sh)
    w = ex / ex.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            i1 = np.arange(0, n - lag)
            grad[i1] += wk * (1.0 - h[i1 + lag])
            i2 = np.arange(lag, n)
            grad[i2] -= wk * h[i2 - lag]
        else:
            la = -lag
            i1 = np.arange(la, n)
            grad[i1] += wk * (1.0 - h[i1 - la])
            i2 = np.arange(0, n - la)
            grad[i2] -= wk * h[i2 + la]
    diff = h.sum() - target
    grad += 2.0 * pen * diff
    return grad

def lbfgs_opt(h0, beta, target, pen, maxiter):
    n = len(h0)
    bnd = Bounds(np.zeros(n), np.ones(n))
    try:
        r = minimize(_obj, h0, jac=_grad, args=(beta, target, pen),
                     method='L-BFGS-B', bounds=bnd,
                     options={'maxiter': maxiter, 'gtol': 1e-12, 'factr': 1e14})
        return project(r.x, target)
    except Exception:
        return h0

def subgrad_step(h, target, steps, lr, top_k, t0, budget):
    n = len(h)
    mom = np.zeros(n)
    best = compute_c5(h)
    for it in range(steps):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            if lag >= 0:
                i1 = np.arange(0, n - lag)
                g[i1] += (1.0 - h[i1 + lag])
                i2 = np.arange(lag, n)
                g[i2] -= h[i2 - lag]
            else:
                la = -lag
                i1 = np.arange(la, n)
                g[i1] += (1.0 - h[i1 - la])
                i2 = np.arange(0, n - la)
                g[i2] -= h[i2 + la]
        g /= top_k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        mom = 0.8 * mom - lr * g
        trial = project(h + mom, target)
        c = compute_c5(trial)
        if c < best - 1e-14:
            h = trial
            best = c
        else:
            mom *= 0.3
            trial = project(h + mom, target)
            c = compute_c5(trial)
            if c < best - 1e-14:
                h = trial
                best = c
            else:
                mom = np.zeros(n)
    return h, best

def coord_descent(h, target, t0, budget, max_sweeps=40):
    n = len(h)
    best = compute_c5(h)
    steps_arr = (0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for step in steps_arr:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + step, 0.0, 1.0)
                trial = project(trial, target)
                c = compute_c5(trial)
                if c < best - 1e-13:
                    h = trial
                    best = c
                    improved = True
                    break
        if not improved and sw > 8:
            break
    return h, best

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def boundary_round(h, target, eps):
    h = h.copy()
    h[h < eps] = 0.0
    h[h > 1.0 - eps] = 1.0
    return project(h, target)
