def compute_grad_vectorized(h, weights):
    """Compute gradient of smooth max using vectorized shifts."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s, ws in enumerate(weights):
        if ws < 1e-12:
            continue
        lag = s - (n - 1)
        if lag >= 0:
            # shift(1-h, -lag): (1-h)[j-lag] for j = lag..n-1
            grad[lag:] += ws * one_minus_h[:n-lag]
            # -h[j+lag] for j = 0..n-1-lag
            grad[:n-lag] -= ws * h[lag:]
        elif lag < 0:
            t = -lag
            # shift(1-h, t): (1-h)[j+t] for j = 0..n-1-t
            grad[:n-t] += ws * one_minus_h[t:]
            # -h[j-t] for j = t..n-1
            grad[t:] -= ws * h[:n-t]
        else:
            grad[:] += ws * one_minus_h
            grad[:] -= ws * h
    grad *= dx
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum} using bisection."""
    h = np.clip(h.copy(), 0.0, 1.0)
    # Bisection on the offset
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_test = np.clip(h + mid, 0.0, 1.0)
        s = h_test.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2, 0.0, 1.0)

from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_and_weights(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    return corr, m, w, sw

def smoothmax_value(h, beta):
    corr, m, w, sw = corr_and_weights(h, beta)
    dx = 2.0 / len(h)
    return (m + np.log(sw) / beta) * dx

def smoothmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr, m, w, sw = corr_and_weights(h, beta)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(2 * n - 1):
        ws = w[s]
        if ws < 1e-14:
            continue
        lag = s - (n - 1)
        if lag == 0:
            grad[:] += ws * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += ws * one_minus_h[:n - lag]
            grad[:n - lag] -= ws * h[lag:]
        else:
            t = -lag
            grad[:n - t] += ws * one_minus_h[t:]
            grad[t:] -= ws * h[:n - t]
    grad *= dx
    return grad

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    # Bisection for exact sum
    lo, hi = -1.0, 1.0
    base = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(base + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h2 = np.clip(base + (lo + hi) / 2, 0.0, 1.0)
    diff = target - h2.sum()
    if abs(diff) > 1e-10:
        free = (h2 > 1e-9) & (h2 < 1.0 - 1e-9)
        if free.any():
            h2[free] += diff / free.sum()
            h2 = np.clip(h2, 0.0, 1.0)
    return h2

    target = n / 2.0
    def f(x):
        return smoothmax_value(x, beta)
    def g(x):
        return smoothmax_grad(x, beta)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(res.x, target)
    except:
        return project(h.copy(), target)

def nesterov_pg(h, beta, max_iters=3000, lr=0.05, decay=0.9999):
    n = len(h)
    target = n / 2.0
    x = h.copy()
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr * (decay ** t)
        if t == 0:
            y = x
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project(y, target)
        grad = smoothmax_grad(y, beta)
        x_new = project(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(base + mid, 0, 1).sum() < target:
            lo = mid
        else:
            hi = mid
    h2 = np.clip(base + (lo + hi) / 2, 0.0, 1.0)
    diff = target - h2.sum()
    if abs(diff) > 1e-10:
        free = (h2 > 1e-9) & (h2 < 1.0 - 1e-9)
        if free.any():
            h2[free] += diff / free.sum()
            h2 = np.clip(h2, 0, 1)
    return h2

def slsqp_opt(h, beta, maxiter=300):
    n = len(h)
    target = n / 2.0
    def f(x):
        return smoothmax_val(x, beta)
    def g(x):
        return smoothmax_grad(x, beta)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(res.x, target)
    except Exception:
        return project(h.copy(), target)

def nesterov_pg(h, beta, max_iters=4000, lr=0.04, decay=0.99995):
    n = len(h)
    target = n / 2.0
    x = h.copy()
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr * (decay ** t)
        if t == 0:
            y = x
        else:
            mu = (t - 1.0) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = project(y, target)
        grad = smoothmax_grad(y, beta)
        x_new = project(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to(h, n_new):
    n_old = len(h)
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(xo, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    return project(np.clip(f(xn), 0, 1), n_new / 2.0)

def alt_init(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project(0.5 + 0.4 * np.sin(np.pi * x), n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(200, 0.5)
    h = project(h, len(h) / 2.0)
    best = h.copy()
    best_c5 = compute_c5(h)
    n = len(best)

    # Phase 1: Progressive SLSQP beta ramp
    for beta in [100, 300, 700, 1500, 3000]:
        if tl() < 35:
            break
        h = slsqp_opt(best, beta, maxiter=350)
        c = compute_c5(h)
        if c < best_c5:
            best, best_c5 = h.copy(), c

    # Phase 2: Nesterov momentum projected gradient
    if tl() > 80:
        h2, c2 = nesterov_pg(best, 2000, max_iters=5000, lr=0.03, decay=0.99996)
        if c2 < best_c5:
            best, best_c5 = h2, c2

    # Phase 3: High-beta SLSQP polish
    for beta in [3000, 6000]:
        if tl() < 35:
            break
        h = slsqp_opt(best, beta, maxiter=400)
        c = compute_c5(h)
        if c < best_c5:
            best, best_c5 = h.copy(), c

    # Phase 4: Strategic perturbations + re-optimize
    n = len(best)
    for i in range(14):
        if tl() < 40:
            break
        scale = 0.006 + 0.004 * (i % 4)
        h_p = project(best + np.random.randn(n) * scale, n / 2.0)
        h_new = slsqp_opt(h_p, 2500, maxiter=350)
        c = compute_c5(h_new)
        if c < best_c5:
            best, best_c5 = h_new, c
        if tl() > 25 and c <= best_c5 + 1e-5:
            h3, c3 = nesterov_pg(h_new, 4000, max_iters=2000, lr=0.015, decay=0.99997)
            if c3 < best_c5:
                best, best_c5 = h3, c3

    # Phase 5: Alternative structure exploration
    if tl() > 120:
        h_alt = alt_init(n)
        h_alt = slsqp_opt(h_alt, 500, maxiter=300)
        c_alt = compute_c5(h_alt)
        if c_alt < best_c5:
            for beta in [1000, 3000]:
                h_alt = slsqp_opt(h_alt, beta, maxiter=300)
                c_alt = compute_c5(h_alt)
            if c_alt < best_c5:
                best, best_c5 = h_alt, c_alt

    # Phase 6: Multi-resolution
    for n_try in [300, 500]:
        if tl() < 150:
            break
        h_int = interp_to(best, n_try)
        for beta in [500, 2000, 5000]:
            if tl() < 50:
                break
            h_new = slsqp_opt(h_int, beta, maxiter=300)
            c = compute_c5(h_new)
            if c < best_c5:
                best, best_c5 = h_new, c
                h_int = h_new.copy()
        if len(best) != n_try and tl() > 40:
            h_new = slsqp_opt(best, 3000, maxiter=300)
            c = compute_c5(h_new)
            if c < best_c5:
                best, best_c5 = h_new, c

    # Phase 7: Final perturbations at best resolution
    n = len(best)
    for i in range(8):
        if tl() < 35:
            break
        h_p = project(best + np.random.randn(n) * 0.004, n / 2.0)
        h_new = slsqp_opt(h_p, 5000, maxiter=350)
        c = compute_c5(h_new)
        if c < best_c5:
            best, best_c5 = h_new, c

    # Phase 8: Ultimate polish
    if tl() > 40:
        h = slsqp_opt(best, 8000, maxiter=500)
        c = compute_c5(h)
        if c < best_c5:
            best, best_c5 = h, c

    best = project(best, len(best) / 2.0)
    best_c5 = compute_c5(best)
    return best, best_c5, len(best)
