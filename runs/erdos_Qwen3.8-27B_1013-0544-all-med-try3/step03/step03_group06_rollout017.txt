```python
    n = len(h)
    dx = 2.0 / n
    a = 1.0 - h
    corr = np.correlate(h, a, mode="full")
    # corr has length 2n-1, index n-1 corresponds to lag 0
    
    # Smooth max weights
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(x_shift) / np.sum(np.exp(x_shift))
    
    # Gradient: for each lag k (index = n-1+k in corr),
    # d(corr[n-1+k])/dh = shift_{-k}(a) - shift_k(h)
    # We need to sum w_k * (shift_{-k}(a) - shift_k(h))
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if w[k_idx] < 1e-12:
            continue
        k = k_idx - (n - 1)
        # shift_{-k}(a): element j of this is a[j+k] if 0 <= j+k < n
        # shift_k(h): element j of this is h[j-k] if 0 <= j-k < n
        if k >= 0:
            # shift_{-k}(a): a[j+k] for j = 0 to n-1-k
            grad[:n-k] += w[k_idx] * a[k:]
            # shift_k(h): h[j-k] for j = k to n-1
            grad[k:] -= w[k_idx] * h[:n-k]
        else:

            # shift_{-k}(a) = shift_{k_abs}(a): a[j-k_abs] for j = k_abs to n-1
            grad[k_abs:] += w[k_idx] * a[:n-k_abs]

            grad[:n-k_abs] -= w[k_idx] * h[k_abs:]
    
    return dx * grad, dx * (m + np.log(np.sum(np.exp(x_shift))/len(corr))/beta)
```

```python
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            return h
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)

    sum_exp = np.exp(np.log(np.sum(np.exp(x_shift))))  # avoid overflow
    # Actually, since we shifted by max, x_shift <= 0, so exp is safe
    w = exp_x / np.sum(exp_x)
    val = m + np.log(np.sum(exp_x) / len(x)) / beta
    return val, w

    """Compute gradient of smooth_max(corr) * dx with respect to h."""

    dx = 2.0 / n
    a = 1.0 - h
    corr = np.correlate(h, a, mode="full")

        corr_masked[idx] = corr[idx]
        corr = corr_masked
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            return h
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(corr, beta):
    """Return smooth max value and gradient weights."""
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)
    exp_x = np.exp(xs)
    w = exp_x / np.sum(exp_x)
    val = m + np.log(np.sum(exp_x) / len(corr)) / beta
    return val, w

def compute_analytic_grad(h, beta, dx):
    """Compute gradient of smooth_max(corr)*dx wrt h."""
    n = len(h)
    a = 1.0 - h
    corr = np.correlate(h, a, mode="full")
    val, w = smooth_max_and_grad(corr, beta)
    grad = np.zeros(n)
    center = n - 1
    for k_idx in range(len(corr)):
        wk = w[k_idx]
        if wk < 1e-13:
            continue
        k = k_idx - center
        if k >= 0:
            grad[:n-k] += wk * a[k:]
            grad[k:] -= wk * h[:n-k]
        else:
            ka = -k
            grad[ka:] += wk * a[:n-ka]
            grad[:n-ka] -= wk * h[ka:]
    return dx * val, dx * grad

class LPProblem:
    def __init__(self, h0, n, target_sum):
        self.h0 = h0
        self.n = n
        self.target = target_sum
        self.dx = 2.0 / n
        self._build()
    
    def _build(self):
        n = self.n
        h0 = self.h0
        a = 1.0 - h0
        h = cp.Variable(n)
        t = cp.Variable()
        constraints = [h >= 0, h <= 1, cp.sum(h) == self.target, t >= 0]
        center = n - 1
        for k_idx in range(2 * n - 1):
            k = k_idx - center
            coeff = np.zeros(n)
            if k >= 0:
                const = np.sum(h0[:n-k] * h0[:n-k])
                coeff[:n-k] += (1 - h0[:n-k])
                coeff[k:] -= h0[:n-k]
            else:
                ka = -k
                const = np.sum(h0[ka:] * h0[:n-ka])
                coeff[ka:] += (1 - h0[:n-ka])
                coeff[:n-ka] -= h0[ka:]
            constraints.append(coeff @ h + const * self.dx <= t)
        self.prob = cp.Problem(cp.Minimize(t), constraints)
        self.h_var = h
        self.t_var = t
    
    def solve(self, solver=None):
        if solver is None:
            solvers = [cp.GLOP, cp.CBC, cp.GLPK]
        else:
            solvers = [solver]
        for s in solvers:
            try:
                self.prob.solve(solver=s)
                if self.h_var.value is not None:
                    return self.h_var.value.copy(), self.t_var.value
            except Exception:
                continue
        return None, None

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(100, 0.5)
    
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)
    
    # Phase 1: LP relaxation iterations
    for n_lp in [n_best, 100, 150, 200, 250, 300]:
        if time_left() < 60:
            break
            
        target = n_lp / 2.0
        if n_lp == n_best:
            h_lp = best_h.copy()
        else:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_lp)
            h_lp = project_h(np.interp(x_new, x_old, best_h), target)
        
        # Iterate LP relaxation
        for it in range(50):
            if time_left() < 30:
                break
            lp = LPProblem(h_lp, n_lp, target)
            h_new, t_new = lp.solve()
            if h_new is None:
                break
            h_new = project_h(h_new, target)
            c5_new = compute_c5(h_new)
            
            if c5_new < compute_c5(h_lp) - 1e-10:
                h_lp = h_new
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    n_best = n_lp
            else:
                # Converged (LP solution doesn't improve the true objective)
                # Try one more with the new h_lp
                if it > 3:
                    break
    
    # Phase 2: SLSQP with analytical gradient
    n_slsqp = len(best_h)
    target_s = n_slsqp / 2.0
    dx_s = 2.0 / n_slsqp
    bounds_s = Bounds(np.zeros(n_slsqp), np.ones(n_slsqp))
    
    for beta in [50, 100, 200, 500, 1000, 2000, 5000, 10000]:
        if time_left() < 20:
            break
        
        # Use a class to avoid closures
        class ObjFunc:
            def __init__(self, dx, beta):
                self.dx = dx
                self.beta = beta
            def __call__(self, x):
                corr = np.correlate(x, 1.0 - x, mode="full")
                m = np.max(corr)
                xs = np.clip(self.beta * (corr - m), -500, 500)
                return (m + np.log(np.sum(np.exp(xs)) / len(corr)) / self.beta) * self.dx
        
        class GradFunc:
            def __init__(self, dx, beta):
                self.dx = dx
                self.beta = beta
            def __call__(self, x):
                _, g = compute_analytic_grad(x, self.beta, self.dx)
                return g
        
        class ConFun:
            def __init__(self, target):
                self.target = target
            def __call__(self, x):
                return np.sum(x) - self.target
        
        class ConJac:
            def __init__(self, target):
                self.target = target
            def __call__(self, x):
                return np.ones(x.shape)
        
        obj = ObjFunc(dx_s, beta)
        grad = GradFunc(dx_s, beta)
        cons = [{'type': 'eq', 'fun': ConFun(target_s), 'jac': ConJac(target_s)}]
        
        try:
            res = minimize(obj, best_h, jac=grad, method='SLSQP',
                          bounds=bounds_s, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-18})
            h_new = project_h(res.x, target_s)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass
    
    # Phase 3: Subgradient refinement
    n_sub = len(best_h)
    target_sub = n_sub / 2.0
    no_improve = 0
    
    for iteration in range(8000):
        if time_left() < 3:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        n_active = min(8, max(4, n_sub // 30))
        top_idx = np.argsort(corr)[::-1][:n_active]
        
        grad = np.zeros(n_sub)
        for idx in top_idx:
            lag = int(idx) - (n_sub - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * best_h)
            elif lag > 0:
                grad[:n_sub - lag] += (1.0 - best_h[lag:])
                grad[lag:] -= best_h[:n_sub - lag]
            else:
                m = -lag
                grad[m:] += (1.0 - best_h[:n_sub - m])
                grad[:n_sub - m] -= best_h[m:]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        
        found = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = project_h(best_h - step * grad, target_sub)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_h = h_trial
                best_c5 = c5_trial
                found = True
                no_improve = 0
                break
        
        if not found:
            no_improve += 1
            if no_improve > 80 and iteration % 40 == 39:
                perturb = np.random.randn(n_sub) * 0.004
                h_trial = project_h(best_h + perturb, target_sub)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial
                    no_improve = 0
            if no_improve > 1200:
                break
    
    return best_h, best_c5, len(best_h)
```

```python
class ObjFunc:
    def __init__(self, dx, beta):
        self.dx = dx
        self.beta = beta
    def __call__(self, x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        xs = np.clip(self.beta * (corr - m), -500, 500)
        return (m + np.log(np.sum(np.exp(xs)) / len(corr)) / self.beta) * self.dx

class GradFunc:
    def __init__(self, dx, beta):
        self.dx = dx
        self.beta = beta
    def __call__(self, x):
        _, g = compute_analytic_grad(x, self.beta, self.dx)
        return g

class ConFun:
    def __init__(self, target):
        self.target = target
    def __call__(self, x):
        return np.sum(x) - self.target

class ConJac:
    def __init__(self, target):
        self.target = target
    def __call__(self, x):
        return np.ones(x.shape)
```
