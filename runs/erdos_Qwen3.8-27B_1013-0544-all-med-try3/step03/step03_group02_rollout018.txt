    deltas = [0.03, -0.03, 0.015, -0.015, 0.007, -0.007, 0.003, -0.003]
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                nv = cur[j] + d
                if nv < 0 or nv > 1:
                    continue
                trial = cur.copy()
                trial[j] = nv
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def pair_moves(cur, target, n, n_tries, t0, budget):
    best_c5 = compute_c5(cur)
    h = cur.copy()
    for _ in range(n_tries):
        if budget - (time.time() - t0) < 1:
            break
        i, j = np.random.choice(n, 2, replace=False)
        d = np.random.choice([0.02, 0.04, 0.01, 0.06])
        trial = h.copy()
        trial[i] += d
        trial[j] -= d
        if trial[i] < 0 or trial[i] > 1 or trial[j] < 0 or trial[j] > 1:
            continue
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            h = trial
            best_c5 = c_try
    return h, best_c5

def anneal(cur, target, n, t0, budget, n_iter=4000, T0=0.004, T1=0.00005):
    h = cur.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = T0
    decay = (T1 / T0) ** (1.0 / max(n_iter, 1))
    for it in range(n_iter):
        if budget - (time.time() - t0) < 1:
            break
        T *= decay
        n_ch = 1 if np.random.random() < 0.6 else 2
        idxs = np.random.choice(n, n_ch, replace=False)
        for idx in idxs:
            d = np.random.uniform(-0.06, 0.06)
            h[idx] = np.clip(h[idx] + d, 0, 1)
        h = project_h(h, target)
        c_try = compute_c5(h)
        delta = c_try - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)

    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base
    target = n_base / 2.0

    # Phase 1: SLSQP polish
    for beta in [100, 300, 800, 1500]:
        rem = budget_s - (time.time() - t0)
        if rem < 12:
            break
        h_new = do_slsqp(best_h, n_base, target, beta, 3000)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new

    # Phase 2: Active set descent (focus on max lag)
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_new, c_new = active_set_descent(best_h, target, n_base, 4000, t0, budget_s, top_k=2, lr_init=0.035)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new

    # Phase 3: Coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        best_h, best_c5 = coord_descent(best_h, target, n_base, 35, t0, budget_s)

    # Phase 4: Pair moves
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        best_h, best_c5 = pair_moves(best_h, target, n_base, 3000, t0, budget_s)

    # Phase 5: Annealing from current best
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_new, c_new = anneal(best_h, target, n_base, t0, budget_s, n_iter=3000, T0=0.003, T1=0.00004)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new
        else:
            # Perturb and anneal
            hp = project_h(best_h + np.random.randn(n_base) * 0.04, target)
            h_new, c_new = anneal(hp, target, n_base, t0, budget_s, n_iter=2000, T0=0.005, T1=0.0001)
            if c_new < best_c5:
                best_h, best_c5 = h_new, c_new

    # Phase 6: Try different resolution
    rem = budget_s - (time.time() - t0)
    if rem > 70:
        for n_try in [240, 200, 280]:
            rem = budget_s - (time.time() - t0)
            if rem < 40:
                break
            target_t = n_try / 2.0
            h_t = np.interp(np.linspace(0, 2, n_try), np.linspace(0, 2, n_base), best_h)
            h_t = project_h(h_t, target_t)
            for beta in [200, 600, 1200]:
                rem = budget_s - (time.time() - t0)
                if rem < 10:
                    break
                h_t = do_slsqp(h_t, n_try, target_t, beta, 2000)
                c_t = compute_c5(h_t)
                if c_t < best_c5:
                    best_h, best_c5, best_n = h_t.copy(), c_t, n_try
            rem = budget_s - (time.time() - t0)
            if rem > 40:
                h_t, c_t = active_set_descent(h_t, target_t, n_try, 2000, t0, budget_s, top_k=2, lr_init=0.03)
                if c_t < best_c5:
                    best_h, best_c5, best_n = h_t.copy(), c_t, n_try
            rem = budget_s - (time.time() - t0)
            if rem > 20:
                h_t, c_t = coord_descent(h_t, target_t, n_try, 25, t0, budget_s)
                if c_t < best_c5:
                    best_h, best_c5, best_n = h_t.copy(), c_t, n_try

    # Phase 7: Final polish
    rem = budget_s - (time.time() - t0)
    nf = len(best_h)
    tf = nf / 2.0
    if rem > 15:
        for beta in [2000, 4000]:
            rem = budget_s - (time.time() - t0)
            if rem < 6:
                break
            h_new = do_slsqp(best_h, nf, tf, beta, 5000)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_h, best_c5 = h_new, c_new
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h, best_c5 = active_set_descent(best_h, tf, nf, 3000, t0, budget_s, top_k=1, lr_init=0.02)
    rem = budget_s - (time.time() - t0)
    if rem > 8:
        best_h, best_c5 = coord_descent(best_h, tf, nf, 40, t0, budget_s)
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        best_h, best_c5 = pair_moves(best_h, tf, nf, 2000, t0, budget_s)

    return best_h, best_c5, len(best_h)
