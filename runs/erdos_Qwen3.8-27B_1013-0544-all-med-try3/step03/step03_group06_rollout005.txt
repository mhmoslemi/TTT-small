def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:

def coord_descent(h, target_sum, num_lags=25, max_sweeps=500, time_limit=None, start_time=None):

    c5 = compute_c5(h)

def slsqp_refine(h, target_sum, beta=2000, maxiter=3000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        shifted = np.clip(beta * (corr - m), -500, 500)
        return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        shifted = np.clip(beta * (corr - m), -500, 500)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        grad = np.zeros(2 * n - 1)
        for idx in range(len(corr)):
            if weights[idx] < 1e-15:
                continue
            lag = idx - (n - 1)
            w = weights[idx] * beta * dx
            if lag == 0:
                grad[idx] += w * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[idx + lag:] += w * (1.0 - x[:-lag])
                grad[idx:idx + n - lag] -= w * x[lag:]
            else:
                ml = -lag
                grad[idx:idx + n - ml] += w * (1.0 - x[ml:])
                grad[idx + ml:] -= w * x[:n - ml]
        return grad

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(len(x))}]
    try:

                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)
