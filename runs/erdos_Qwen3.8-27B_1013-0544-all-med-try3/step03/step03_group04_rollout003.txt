def project_h(h, target_sum):

    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    w = np.exp(xs)
    s = w.sum()
    if s < 1e-300:

        w[np.argmax(x)] = 1.0
    else:

        h_fin = project_h(h_fin, tgt_fin)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin.copy(), c5_fin

    return best_h, best_c5, int(len(best_h))
