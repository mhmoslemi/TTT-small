def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def objective_topk(h, top_idx, k):
    """Mean of top-k correlations (smooth, differentiable surrogate for max)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.mean(corr[top_idx[:k]])

def objective_scmax(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s)) / len(corr)) / beta

def subgrad_corr(h, idx):
    n = len(h)
    grad = np.zeros(n)
    lag = idx - (n - 1)
    if lag >= 0:
        grad[:n-lag] += (1.0 - h[lag:])
        grad[lag:] += -h[:n-lag]
    else:
        la = -lag
        grad[la:] += (1.0 - h[:n-la])
        grad[:n-la] += -h[la:]
    return grad

def subgrad_topk(h, k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:k]
    grad = np.zeros(n)
    for idx in top_idx:
        grad += subgrad_corr(h, int(idx))
    return grad / k, corr

    return np.sum(x) - target

    return np.ones(x.shape)

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def antisym_project(h, target):
    """Project to anti-symmetric: h[i] + h[n-1-i] = 1, sum = target"""
    n = len(h)
    half = n // 2
    g = h[:half].copy()
    # Anti-sym: h[n-1-i] = 1 - h[i], so sum = sum_{i=0}^{half-1} (g[i] + 1-g[i]) = half
    # For n even: sum = n/2 = target (since target = n/2). So it's automatically satisfied!
    # Just clip g to [0,1] and reconstruct
    g = np.clip(g, 0.0, 1.0)
    h_out = np.zeros(n)
    h_out[:half] = g
    h_out[half:] = 1.0 - g[::-1]
    return h_out

def subgrad_at_idx(h, idx, n):
    grad = np.zeros(n)
    if idx == n-1:
        grad[:] = 1.0 - 2.0*h
    elif idx > n-1:
        k = idx - (n-1)  # positive lag
        # h[j] as h[i+k]: valid for j in [k, n-1]
        grad[k:] += (1.0 - h[:n-k])
        # h[j] as h[i]: valid for j in [0, n-1-k]
        grad[:n-k] += -h[k:]
    else:
        k = (n-1) - idx  # negative lag magnitude
        # h[j] as h[i]: valid for j in [0, n-1-k]
        grad[:n-k] += (1.0 - h[k:])
        # h[j] as h[i+k]: valid for j in [k, n-1]
        grad[k:] += -h[:n-k]
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_scmax(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s)) / len(corr)) / beta

def subgrad_at_idx(h, idx, n):
    grad = np.zeros(n)
    if idx == n - 1:
        grad[:] = 1.0 - 2.0 * h
    elif idx > n - 1:
        k = idx - (n - 1)
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] += -h[k:]
    else:
        k = (n - 1) - idx
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] += -h[:n-k]
    return grad
