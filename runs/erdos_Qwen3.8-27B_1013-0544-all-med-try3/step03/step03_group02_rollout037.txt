import numpy as np
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_binary(h, target):
    """Project h to binary {0,1} with sum = target."""
    scores = h.copy()
    # Sort by score descending, take top target
    idx = np.argsort(scores)[::-1]
    h_bin = np.zeros(len(h))
    h_bin[idx[:int(target)]] = 1.0
    return h_bin

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_values(h):
    """Return the correlation array (before dx scaling)."""
    return np.correlate(h, 1.0 - h, mode="full")

    return m + np.log(np.sum(np.exp(np.clip(beta*(x-m),-500,0)))/len(x))/beta

def obj_sml(x, beta, dx):
    return sml_max(np.correlate(x, 1.0-x, mode="full"), beta) * dx

def _sum_fun(x, t): return x.sum() - t
def _sum_jac(x, t): return np.ones(x.shape)

def slsqp(h0, n, target, beta, maxiter=5000):
    b = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type':'eq','fun':_sum_fun,'jac':_sum_jac,'args':(target,)}]
    try:
        r = minimize(obj_sml, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=b, constraints=cons, options={'maxiter':maxiter,'ftol':1e-15})
        return proj_h(r.x, target)
    except: return h0

def subgrad_step(h, target, n_steps, step_size, t0, budget, top_k=5):
    n = len(h)
    for _ in range(n_steps):
        if budget - (time.time()-t0) < 2: break
        c = np.correlate(h, 1.0-h, mode="full")
        top = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for t in top:
            k = int(t) - (n-1)
            if k >= 0:
                grad[:n-k] += (1.0 - h[k:])
                grad[k:] -= h[:n-k]
            else:
                la = -k
                grad[la:] += (1.0 - h[:n-la])
                grad[:n-la] -= h[la:]
        grad /= max(np.linalg.norm(grad), 1e-12)
        h = proj_h(h - step_size * grad, target)
    return h
