```python
    """Compute C5 for a periodic pattern with given values repeated."""
    n = 280
    h = np.tile(values, n // period)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")

def optimize_periodic(period, seed=42):
    n = 280
    n_reps = n // period
    target_sum = period * 0.5  # sum of one period must be period/2
    
    def obj(x):
        # Ensure non-negativity
        x = np.maximum(x, 0)
        # Normalize to have correct sum
        s = x.sum()
        if s < 1e-10:
            return 1.0
        x = x * target_sum / s
        x = np.minimum(x, 1.0)
        h = np.tile(x, n_reps)
        dx = 2.0 / n
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    # Initial guess
    x0 = np.full(period, 0.5)
    res = minimize(obj, x0, method='Nelder-Mead', options={'maxiter': 10000})
    best_x = np.maximum(res.x, 0)
    best_x = np.minimum(best_x * target_sum / best_x.sum(), 1.0)
    h = np.tile(best_x, n_reps)
    return h, obj(best_x)
```

```python
def compute_all_sensitivities(h, n):
    """Compute sensitivity for all coordinates. Returns array of shape (n, 2n-1)."""
    # sensitivity[i, k] = (1 - h[i+offset_k]) * 1_{valid} - h[i-offset_k] * 1_{valid}
    # offset_k = k - (n-1)
    offsets = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # For each i and k: offset = offsets[k]
    # i + offset valid: 0 <= i+offset < n
    # i - offset valid: 0 <= i-offset < n
    
    # Create index arrays
    # i ranges 0..n-1, offset ranges -(n-1)..(n-1)
    i_idx = np.arange(n)[:, None]  # (n, 1)
    off_idx = offsets[None, :]  # (1, 2n-1)
    
    i_plus = i_idx + off_idx  # (n, 2n-1)
    i_minus = i_idx - off_idx  # (n, 2n-1)
    
    valid_plus = (i_plus >= 0) & (i_plus < n)
    valid_minus = (i_minus >= 0) & (i_minus < n)
    
    # Clip for safe indexing
    i_plus_clipped = np.clip(i_plus, 0, n-1)
    i_minus_clipped = np.clip(i_minus, 0, n-1)
    
    # h[i+offset] and h[i-offset]
    h_plus = h[i_plus_clipped] * valid_plus  # where invalid, 0
    h_minus = h[i_minus_clipped] * valid_minus
    
    sens = (1.0 - h_plus) - h_minus  # shape (n, 2n-1)
    
    # Handle offset=0 (k=n-1) case: sensitivity = 1 - 2*h[i]
    # Actually the formula above gives: (1-h[i]) - h[i] = 1-2*h[i] which is correct!
    # Because when offset=0: i_plus = i, i_minus = i, both valid
    # sens = (1-h[i]) - h[i] = 1-2*h[i] ✓
    
    return sens
```

```python
import numpy as np
from scipy.optimize import minimize, differential_evolution, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def compute_all_sensitivities(h, n):
    i_idx = np.arange(n)[:, None]
    offsets = (np.arange(2 * n - 1) - (n - 1))[None, :]
    i_plus = i_idx + offsets
    i_minus = i_idx - offsets
    vp = (i_plus >= 0) & (i_plus < n)
    vm = (i_minus >= 0) & (i_minus < n)
    ipc = np.clip(i_plus, 0, n - 1)
    imc = np.clip(i_minus, 0, n - 1)
    hp = h[ipc] * vp
    hm = h[imc] * vm
    return (1.0 - hp) - hm

def perc_obj(x, period, n):
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-12:
        return 10.0
    x = x * (period * 0.5) / s
    x = np.minimum(x, 1.0)
    nr = n // period
    h = np.tile(x[:period], nr)
    return compute_c5(h)

def try_periodic(period, n, seed=42):
    np.random.seed(seed)
    x0 = np.full(period, 0.5) + 0.2 * np.random.randn(period)
    x0 = np.clip(x0, 0.01, 0.99)
    res = minimize(perc_obj, x0, args=(period, n), method='Nelder-Mead',
                   options={'maxiter': 50000, 'xatol': 1e-12, 'fatol': 1e-14})
    x = np.maximum(res.x, 0.0)
    s = x.sum()
    if s < 1e-12:
        return None, 1e9
    x = x * (period * 0.5) / s
    x = np.minimum(x, 1.0)
    h = np.tile(x[:period], n // period)
    return h, compute_c5(h)

def coord_descent(h, max_passes, step, t_start, time_limit):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)
    cur_max = float(np.max(corr))
    rng = np.random.RandomState(12345)
    for p in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        sens = compute_all_sensitivities(h, n)
        order = rng.permutation(n)
        improved = False
        for i in order:
            best_d = 0.0
            best_m = cur_max
            best_c = None
            for d in (-step, -step * 0.4, step * 0.4, step):
                nv = h[i] + d
                if nv < -1e-6 or nv > 1.0 + 1e-6:
                    continue
                d = nv - h[i]
                if abs(d) < 1e-9:
                    continue
                nc = corr + d * sens[i]
                nm = float(np.max(nc))
                if nm < best_m - 1e-14:
                    best_m = nm
                    best_d = d
                    best_c = nc
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                if best_c is not None:
                    corr = best_c
                cur_max = best_m
                improved = True
        if not improved:
            break
        if p % 50 == 49:
            corr = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)
            cur_max = float(np.max(corr))
    h = project_h(h, target)
    return h, compute_c5(h)

def _slsqp_obj(x, n_pts):
    dx = 2.0 / n_pts
    corr = np.correlate(x, 1.0 - x, mode="full")
    beta = max(200.0, 50.0 * (n_pts / 50.0))
    return smooth_max(corr, beta) * dx

def _slsqp_con(x, n_pts):
    return np.sum(x) - n_pts / 2.0

def _slsqp_con_jac(x, n_pts):
    return np.ones(x.shape)

def slsqp_refine(h, n_pts, t_start, time_limit):
    target = n_pts / 2.0
    h = project_h(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    constr = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac, 'args': (n_pts,)}]
    for beta_scale in [1, 2, 4, 8, 16, 32]:
        if time.time() - t_start > time_limit:
            break
        for attempt in range(3):
            if time.time() - t_start > time_limit - 10:
                break
            try:
                res = minimize(_slsqp_obj, best_h, args=(n_pts,), method='SLSQP',
                               bounds=bounds, constraints=constr,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass
    return best_h, best_c5

def _de_obj(x, target, dx):
    h = project_h(x, target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, 60.0) * dx
```
