def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_and_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:

        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def lbfgs_step(h, beta, penalty, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    
    def obj(x):

        return v + penalty * (x.sum() - target) ** 2

            return np.ones(n)
        v, g = smoothmax_and_grad(x, beta, top_k)
        return g + 2 * penalty * (x.sum() - target)

                   bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_and_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(-corr)[:top_k]
    else:
        idx = np.arange(len(corr))
    c_sel = corr[idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad


def _lbfgs_obj(x, beta, penalty, target, top_k):
    v, _ = smoothmax_and_grad(x, beta, top_k)
    return v + penalty * (x.sum() - target) ** 2


def _lbfgs_objg(x, beta, penalty, target, top_k):
    v, g = smoothmax_and_grad(x, beta, top_k)
    return g + 2.0 * penalty * (x.sum() - target)


def lbfgs_solve(h, beta, penalty, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    obj = partial(_lbfgs_obj, beta=beta, penalty=penalty, target=target, top_k=top_k)
    objg = partial(_lbfgs_objg, beta=beta, penalty=penalty, target=target, top_k=top_k)
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def basin_hop(h_best, best_c5, n_hops, beta_list, maxiter, noise_base,
              noise_decay, seed_hop, t0, budget_s):
    rng = np.random.RandomState(seed_hop)
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 35:
            break
        noise = noise_base * (noise_decay ** hop)
        h_pert = project_feasible(h_best + rng.randn(n) * noise, target)
        h_local = h_pert
        for beta in beta_list:
            if budget_s - (time.time() - t0) < 20:
                break
            h_local = lbfgs_solve(h_local, beta, 50.0, maxiter, top_k=25)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5


def symmetric_init(n):
    h = np.zeros(n)
    for i in range(n // 2):
        h[i] = 1.0
        h[i + n // 2] = 0.0
    return project_feasible(h, n / 2.0)


def sinusoidal_init(n, freq=3):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(np.pi * freq * x)
    return project_feasible(h, n / 2.0)


def block_init(n, nblocks=4):
    h = np.zeros(n)
    block_size = n // nblocks
    for b in range(nblocks):
        if b % 2 == 0:
            h[b * block_size:(b + 1) * block_size] = 1.0
    return project_feasible(h, n / 2.0)


def optimize_from_start(h_start, t0, budget_s, max_total_iter=3000):
    if budget_s - (time.time() - t0) < 30:
        return h_start, compute_c5(h_start)
    n = len(h_start)
    h = h_start.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    for beta in [10, 30, 80, 200, 500, 800]:
        if budget_s - (time.time() - t0) < 30:

        h = lbfgs_solve(h, beta, 50.0, 400, top_k=None)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
