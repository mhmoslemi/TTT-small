def compute_gradient(h, w):
    """Compute gradient of smoothmax given weights w on lags."""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    # For lag s = k - (n-1), k ranges from 0 to 2n-2
    # s ranges from -(n-1) to (n-1)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            # corr[k] = sum_{i=0}^{n-1-s} h[i+s] * (1-h[i])
            # d(corr[k])/dh[j] = (1-h[j-s]) for j in [s, n-1]
            #                     -h[j+s]     for j in [0, n-1-s]
            grad[s:] += wk * one_minus_h[:n-s]
            grad[:n-s] -= wk * h[s:]
        else:
            t = -s
            # corr[k] = sum_{i=t}^{n-1} h[i] * (1-h[i-t])
            # d(corr[k])/dh[j] = (1-h[j+t]) for j in [0, n-1-t]
            #                     -h[j-t]    for j in [t, n-1]
            grad[:n-t] += wk * one_minus_h[t:]
            grad[t:] -= wk * h[:n-t]
    
    return grad

from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-9) & (h < 1.0 - 1e-9)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def get_smoothmax_weights(corr, beta):
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta)
    return val, w

def compute_grad_from_weights(h, w):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * one_minus_h[:n-s]
            grad[:n-s] -= wk * h[s:]
        else:
            t = -s
            grad[:n-t] += wk * one_minus_h[t:]
            grad[t:] -= wk * h[:n-t]
    return grad

def smoothmax_step(h, beta, lr, top_k=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, w = get_smoothmax_weights(corr, beta)
    
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
        w2 = w.copy()
        w[:] = 0.0
        w[top_idx] = w2[top_idx]
        w[top_idx] /= w[top_idx].sum()
    
    grad = compute_grad_from_weights(h, w) * dx
    h_new = project_feasible(h - lr * grad, target)
    return h_new

def slsqp_optimize(h, beta, maxiter, top_k=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        val, w = get_smoothmax_weights(corr, beta)
        if top_k is not None and top_k < len(corr):
            top_idx = np.argsort(corr)[-top_k:]
            w2 = w.copy()
            w[:] = 0.0
            w[top_idx] = w2[top_idx]
            w[top_idx] /= w[top_idx].sum()
            val = np.max(corr)  # use true max for objective
        return val * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        _, w = get_smoothmax_weights(corr, beta)
        if top_k is not None and top_k < len(corr):
            top_idx = np.argsort(corr)[-top_k:]
            w2 = w.copy()
            w[:] = 0.0
            w[top_idx] = w2[top_idx]
            w[top_idx] /= w[top_idx].sum()
        g = compute_grad_from_weights(x, w) * dx
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))

def subgrad_descent(h, max_iters, lr0, decay, beta, top_k=None):

    target = n / 2.0
    x = h.copy()

        v = compute_c5(x)
        if v < best_v:

3. Take a step in direction δ (with line search to ensure the true objective decreases)

def active_set_lp_step(h, top_k=20):

    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Get top k lags
    top_idx = np.argsort(corr)[-top_k:]
    t_max = corr[top_idx].max()
    
    # Compute gradients of f_k at current h
    # f_k(h) = corr[k] * dx
    # grad of corr[k] w.r.t. h:
    # We need A such that corr[k] = sum_j A_kj * h[j] (approximately, linearized)
    # Actually, corr[k] = sum_i h[i](1-h[i+s]) = sum_i h[i] - sum_i h[i]h[i+s]
    # Linearized around h: f_k(h+δ) ≈ f_k(h) + df_k/dh * δ
    # df_k/dh[j] = (1-h[j+s]) - h[j-s] ... (from our gradient formula)

    # minimize t
    # subject to: f_k(h) + grad_k · δ ≤ t  for k in top_k
    #             -h[i] ≤ δ[i] ≤ 1-h[i]  for all i
    #             sum(δ) = 0

    # Objective: minimize t → c = [0,...,0, 1]

    # For each active lag k: grad_k · δ - t ≤ -f_k(h)
    #   → A_eq row: [grad_k, -1], b = -f_k(h)
    # sum(δ) = 0 → A_eq row: [ones, 0], b = 0
    # -h[i] ≤ δ[i] ≤ 1-h[i] → bounds

            g[s:] += (-h[:n-s])

            g[:n-t_lag] += (-h[t_lag:])

import numpy as np
from scipy.optimize import minimize, Bounds, linprog
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-9) & (h < 1.0 - 1e-9)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k):
    """Gradient of corr[k] w.r.t. h"""
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] += (-h[:n-s])
    else:
        t = -s
        g[t:] += (1.0 - h[:n-t])
        g[:n-t] += (-h[t:])
    return g

def smoothmax_val(h, corr, beta):
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = exps.sum()
    return (m + np.log(sw) / beta)

def smoothmax_weights(corr, beta):
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = exps.sum()
    return exps / sw

def slsqp_smoothmax(h, beta, maxiter, top_k=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smoothmax_val(x, corr, beta) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        w = smoothmax_weights(corr, beta)
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-12: continue
            g += wk * lag_gradient(x, k)
        return g * dx
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except:
        return h.copy()

def lp_active_step(h, top_k=20):
    """LP-based active set optimization step"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    
    # Build LP: minimize t s.t. f_k(h) + grad_k·δ ≤ t, -h≤δ≤1-h, sum(δ)=0
    c = np.zeros(n + 1)
    c[-1] = 1.0

    b_ub = np.zeros(top_k)

        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[k] * dx
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds_lp = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(None, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
        if result.success:
            delta = result.x[:n]
            # Line search
            best_h = h.copy()
            best_val = compute_c5(h)
            for alpha in [1.0, 0.8, 0.5, 0.3, 0.1, 0.05, 0.01]:
                h_trial = project_feasible(h + alpha * delta, target)
                val = compute_c5(h_trial)
                if val < best_val:
                    best_val = val
                    best_h = h_trial
            return best_h
    except:
        pass
    return h

def lp_active_loop(h, max_iters, top_k=20):

            best_x = x_new.copy()
        if np.max(np.abs(x_new - x)) < 1e-10:
            break
        x = x_new
    
    return best_x, best_v

def subgrad_descent(h, max_iters, lr0, decay, beta):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        corr = np.correlate(x, 1.0 - x, mode="full")
        w = smoothmax_weights(corr, beta)
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-12: continue
            g += wk * lag_gradient(x, k)
        g *= dx
        x = project_feasible(x - lr_t * g, target)
        v = compute_c5(x)
        if v < best_v:
            best_v = v
            best_x = x.copy()
    
    return best_x, best_v

def project_feasible(h, target_sum):

    diff = target_sum - h.sum()

        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] += (-h[:n-s])
    else:
        t = -s
        g[t:] += (1.0 - h[:n-t])
        g[:n-t] += (-h[t:])
    return g

def sum_lag_gradients(h, w):
