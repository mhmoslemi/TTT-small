def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:

        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:

            h = np.clip(h, 0.0, 1.0)

def smoothmax_val_grad(h, beta, top_k=None):

    dx = 2.0 / n
