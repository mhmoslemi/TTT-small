import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def binary_push(h, target_sum, level=0.5):
    h_bin = (h >= level).astype(float)
    ones = int(h_bin.sum())
    target = int(target_sum)
    if ones > target:
        idx = np.where(h_bin == 1)[0]
        remove = np.random.choice(idx, ones - target, replace=False)
        h_bin[remove] = 0
    elif ones < target:
        idx = np.where(h_bin == 0)[0]
        add = np.random.choice(idx, target - ones, replace=False)
        h_bin[add] = 1
    return h_bin

def targeted_hill_climb(h, target_sum, max_iters=5000, tol=1e-10):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        corr = compute_corr(best_h)
        top_k = np.argsort(corr)[-4:]
        improved = False
        for k_idx in top_k:
            d = int(k_idx - (n - 1))
            if d == 0:
                continue
            ad = abs(d)
            if d > 0:
                pairs = [(i, i + d) for i in range(n - d)]
            else:
                pairs = [(i - d, i) for i in range(ad)]
            np.random.shuffle(pairs)
            for (i, j) in pairs[:15]:
                for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                    hi_new = best_h[i] + delta
                    hj_new = best_h[j] - delta
                    if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
                        continue
                    h_try = best_h.copy()
                    h_try[i], h_try[j] = hi_new, hj_new
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - tol:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5

    start = time.time()

    def tleft():

        return x.sum() - target_sum

        return np.ones(n)
