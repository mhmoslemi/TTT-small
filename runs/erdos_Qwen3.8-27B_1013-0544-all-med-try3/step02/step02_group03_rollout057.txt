    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_cost = np.max(corr) * dx
    best_cost = cur_cost
    best_h = h.copy()

    zeros = [i for i in range(n) if i not in set(ones)]
    
    T = 0.1
    T_min = 0.001
    cool_rate = (T_min / T) ** (1.0 / max(1, max_iters))

        ones[ones.index(i)] = j
        zeros[zeros.index(j)] = i

        new_cost = np.max(new_corr) * dx

            if new_cost < best_cost:

                best_h = h.copy()

            h[i], h[j] = h[j], h[i]
            ones[ones.index(j)] = i
            zeros[zeros.index(i)] = j
        
        T *= cool_rate

This is NOT linear because of the h[i]*h[i+k] term! So it's a quadratic constraint.

Hmm, but we can linearize it. Note that:
h[i]*(1-h[i+k]) = h[i] - h[i]*h[i+k]

The h[i]*h[i+k] term is the product of two variables, which makes this a quadratic (actually bilinear) constraint. This can be handled by SDP relaxation (S-lemma / semidefinite programming) but that's expensive.

Alternatively, for a fixed h[i+k], the constraint becomes linear in h[i]. We could use an alternating approach.

Actually, let me think about this differently. The bilinear term h[i]*h[i+k] can be upper bounded by:
h[i]*h[i+k] ≤ (h[i] + h[i+k])²/4

And lower bounded by:
h[i]*h[i+k] ≥ 0

So h[i]*(1-h[i+k]) = h[i] - h[i]*h[i+k] ≤ h[i]

This gives a trivial bound. Not helpful.

Another approach: use the identity h[i]*(1-h[i+k]) ≤ max(h[i], 1-h[i+k])/2... no that's not right either.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def overlap_grad(h, s, n):
    grad = np.zeros(n)
    if s > 0:
        idx1 = np.arange(0, n - s)
        grad[idx1] += (1.0 - h[idx1 + s])
        idx2 = np.arange(s, n)
        grad[idx2] -= h[idx2 - s]
    elif s < 0:
        t = -s
        idx1 = np.arange(t, n)
        grad[idx1] += (1.0 - h[idx1 - t])
        idx2 = np.arange(0, n - t)
        grad[idx2] -= h[idx2 + t]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def tl_ok(budget_s, t0, min_remaining):
    return (time.time() - t0) < budget_s - min_remaining

def binary_sa(n, seed, max_iters, t0, budget_s):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_cost = float(np.max(corr) * dx)
    best_cost = cur_cost
    best_h = h.copy()
    ones_set = set(ones_idx.tolist())
    ones_list = ones_idx.tolist()
    zeros_list = [i for i in range(n) if i not in ones_set]
    T = 0.05
    T_min = 0.0005
    cool_rate = (T_min / T) ** (1.0 / max(1, max_iters))
    for it in range(max_iters):
        if not tl_ok(budget_s, t0, 10):
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        ones_list[ones_list.index(i)] = j
        zeros_list[zeros_list.index(j)] = i
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_cost = float(np.max(new_corr) * dx)
        delta = new_cost - cur_cost
        if delta < 0 or rng.rand() < np.exp(-max(delta, -50) / T):
            cur_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            ones_list[ones_list.index(j)] = i
            zeros_list[zeros_list.index(i)] = j
        T *= cool_rate
    return best_h, best_cost

def subgrad_adam(h, n, t0, budget_s, max_iters=80000):
    dx = 2.0 / n
    target_sum = n / 2.0
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0
    for it in range(max_iters):
        if not tl_ok(budget_s, t0, 15):
            break
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = 3
        top_idx = np.argsort(corr)[-top_k:][::-1]
        shifts = (top_idx - (n - 1)).astype(int)
        vals = corr[top_idx]
        w = np.exp(150.0 * (vals - vals[0]))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, shifts):
            grad += wi * overlap_grad(h, int(si), n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_feasible(h, target_sum)
        lr = 0.01 / (1.0 + 0.001 * np.sqrt(t))
        if it % 3000 == 2999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(40, 0.5)

    n_parent = len(h_parent)
    n = 70
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = project_feasible(h_parent, target_sum) if n_parent == n else \
        project_feasible(np.interp(np.linspace(0, 2, n, endpoint=False),
                                    np.linspace(0, 2, n_parent, endpoint=False), h_parent), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase A: Binary SA for diverse global exploration
    if tl_ok(budget_s, t0, 60):
        for sa_seed in range(8):
            if not tl_ok(budget_s, t0, 15):
                break
            h_bin, c5_bin = binary_sa(n, seed * 100 + sa_seed, 40000, t0, budget_s)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.astype(float).copy()

    # Phase B: SLSQP continuation from diverse starts
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]

    all_starts = [best_h.copy()]
    x = np.linspace(0, 2, n, endpoint=False)
    for freq, amp in [(1, 0.15), (1, 0.3), (2, 0.15), (2, 0.25), (3, 0.1)]:
        all_starts.append(project_feasible(0.5 + amp * np.sin(np.pi * freq * x), target_sum))
    np.random.seed(seed + 999)
    for _ in range(3):
        all_starts.append(project_feasible(0.3 + 0.4 * np.random.rand(n), target_sum))
    # From parent
    if n_parent != n:
        all_starts.append(project_feasible(
            np.interp(np.linspace(0, 2, n, endpoint=False),
                      np.linspace(0, 2, n_parent, endpoint=False), h_parent), target_sum))

    for h0 in all_starts:
        if not tl_ok(budget_s, t0, 25):
            break
        h_cur = h0.copy()
        for beta in [10, 50, 300, 2000, 15000]:
            if not tl_ok(budget_s, t0, 8):
                break
            try:
                res = minimize(obj_func, h_cur, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-13})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                h_cur = h_new
            except Exception:
                pass
            # Small perturbation between betas
            if tl_ok(budget_s, t0, 5):
                h_cur += np.random.randn(n) * 0.002
                h_cur = project_feasible(h_cur, target_sum)

    # Phase C: Adam subgradient
    if tl_ok(budget_s, t0, 30):
        h_adam, c5_adam = subgrad_adam(best_h, n, t0, budget_s, max_iters=60000)
        if c5_adam < best_c5:
            best_c5 = c5_adam
            best_h = h_adam.copy()

    # Phase D: High-beta SLSQP polish
    if tl_ok(budget_s, t0, 25):
        for beta in [50000, 200000]:
            if not tl_ok(budget_s, t0, 8):
                break
            try:
                res = minimize(obj_func, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase E: Trust-constr attempt
    if tl_ok(budget_s, t0, 30):
        try:
            def obj_tc(h, dx, beta):
                corr = np.correlate(h, 1.0 - h, mode='full')
                return smooth_max(corr, beta) * dx
            res = minimize(obj_tc, best_h, args=(dx, 5000), method='trust-constr',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase F: Adjacent resolutions
    if tl_ok(budget_s, t0, 40):
        n_curr = len(best_h)
        for n_try in [n_curr - 5, n_curr + 5, n_curr - 10, n_curr + 10, n_curr + 20]:
            if n_try < 30 or not tl_ok(budget_s, t0, 15):
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                            np.linspace(0, 2, n_curr, endpoint=False), best_h)
            h_t = project_feasible(h_t, ts_t)
            bounds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
            for beta in [500, 5000, 50000]:
                if not tl_ok(budget_s, t0, 6):
                    break
                try:
                    res = minimize(obj_func, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bounds_t, constraints=cons_t,
                                   options={'maxiter': 4000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        n_curr = n_try
                except Exception:
                    pass

    # Phase G: Final quick subgradient + SLSQP
    if tl_ok(budget_s, t0, 15):
        h_fin, c5_fin = subgrad_adam(best_h, len(best_h), t0, budget_s, max_iters=20000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        if tl_ok(budget_s, t0, 8):
            n_f = len(best_h)
            dx_f = 2.0 / n_f
            ts_f = n_f / 2.0
            bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
            cons_f = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_f, n_f)}]
            try:
                res = minimize(obj_func, best_h, args=(dx_f, 100000), method='SLSQP',
                               bounds=bounds_f, constraints=cons_f,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    return best_h, best_c5, len(best_h)
