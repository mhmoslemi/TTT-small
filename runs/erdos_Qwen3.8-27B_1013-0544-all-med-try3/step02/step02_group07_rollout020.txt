                    tmp = h_try[i:i+blen].copy()
                    h_try[i:i+blen] = h_try[j:j+blen]
                    h_try[j:j+blen] = tmp
            elif r < 0.78:
                # Gaussian on subset
                frac = 0.10 + 0.30 * (T / 0.012)
                mask = rng.random(n) < frac
                nm = int(mask.sum())
                if nm > 0:
                    h_try[mask] += rng.randn(nm) * (0.03 + 0.09 * T / 0.012)
            elif r < 0.90:
                # Multi-point push toward binary
                np_pts = rng.randint(4, min(14, n // 3 + 1))
                idx = rng.choice(n, np_pts, replace=False)
                for i in idx:
                    h_try[i] += 0.04 if h_try[i] < 0.5 else -0.04
            else:
                # Kick: larger random perturbation
                np_pts = rng.randint(6, min(20, n // 2 + 1))
                idx = rng.choice(n, np_pts, replace=False)
                h_try[idx] += rng.randn(np_pts) * 0.08

            h_try = project_h(h_try, target_sum)
            c5_try = compute_c5(h_try)
            delta = c5_try - c5_cur
            if delta < 0 or rng.random() < np.exp(-min(200.0, delta / max(T, 1e-10))):
                h_cur, c5_cur = h_try, c5_try
                no_imp = 0 if c5_cur < c5_cur - 1e-12 else no_imp + 1
            else:
                no_imp += 1

            T = max(T_min, T * cooling)
            if no_imp > 500:
                T = max(T, 0.009)
                no_imp = 0

    # Final SLSQP polish
    if budget_left() > 15:
        def obj_fin(h, b=800, dx_local=dx):
            c = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(c, b) * dx_local
        try:
            res = minimize(obj_fin, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < c5_cur:
                h_cur, c5_cur = h_new.copy(), c5
        except Exception:
            pass

    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def budget_left():
        return budget_s - (time.time() - start_time)

    best_h = None
    best_c5 = 1.0

    # Build candidate list
    candidates = []

    # Primary: initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        candidates.append(np.array(initial_h_values, dtype=float).copy())

    # Alternative n=60: interpolated or fresh
    if len(candidates) > 0:
        h0 = candidates[0]
        n0 = len(h0)
        # Interpolate to n=60
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x60 = np.linspace(0, 2, 60, endpoint=False)
        h60 = np.interp(x60, x0, h0, period=2.0)
        candidates.append(h60)
    else:
        candidates.append(np.full(70, 0.5))

    # n=80 from interpolation of best so far
    if len(candidates) > 0:
        h0 = candidates[0]
        n0 = len(h0)
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x80 = np.linspace(0, 2, 80, endpoint=False)
        h80 = np.interp(x80, x0, h0, period=2.0)
        candidates.append(h80)

    # Random binary n=64
    h_bin = (rng.random(64) > 0.5).astype(float)
    candidates.append(h_bin)

    # Process candidates, allocating time budget
    n_cands = len(candidates)
    # Give primary candidate more time
    time_alloc = [0.35, 0.20, 0.20, 0.15]
    for i in range(n_cands):
        if budget_left() < 50:
            break
        h_cand = candidates[i]
        n = len(h_cand)
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': lambda x, ts=target_sum: con_fun(x, ts),
                 'jac': lambda x: con_jac(x)}]

        h_res, c5_res = optimize_one(h_cand, target_sum, bounds, cons, budget_left, rng)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    if best_h is None:
        best_h = project_h(np.array(candidates[0]), len(candidates[0]) / 2.0)
        best_c5 = compute_c5(best_h)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
