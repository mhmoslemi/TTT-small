        if time.time() - start_time > budget_s - 120:
            break
        def obj_sl(hh):
            corr = np.correlate(hh, 1.0 - hh, mode="full")
            return smooth_max(corr, float(beta)) * dx
        try:
            res = minimize(obj_sl, h, method='SLSQP', bounds=bounds,
                          constraints=con, options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                update(h, c5, n)
        except Exception:
            pass

    # Phase 2: Active-lag focused optimization
    if time.time() - start_time < budget_s - 100:
        for top_k in [3, 5, 7]:
            if time.time() - start_time > budget_s - 40:
                break
            corr_arr = np.correlate(h, 1 - h, mode="full") * dx
            active = np.argsort(corr_arr)[::-1][:top_k]
            def obj_act(hh):
                c = np.correlate(hh, 1 - hh, mode="full")
                return sum(c[k] for k in active) * dx
            try:
                res = minimize(obj_act, h, method='SLSQP', bounds=bounds,
                              constraints=con, options={'maxiter': 2000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    update(h_new, c5_new, n)
                    h = h_new
            except Exception:
                pass

    # Phase 3: Multi-resolution search
    if time.time() - start_time < budget_s - 30:
        resolutions = [60, 65, 70, 75, 80, 90]
        for n2 in resolutions:
            if time.time() - start_time > budget_s - 20:
                break
            if n2 == n:
                continue
            ts2 = n2 / 2.0
            dx2 = 2.0 / n2
            bounds2 = Bounds(np.zeros(n2), np.ones(n2))
            con2 = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts2,
                    'jac': lambda x: np.ones(n2)}]
            h2 = project_feasible(resample_h(h, n2), ts2)
            for beta2 in [300, 1000, 3000]:
                if time.time() - start_time > budget_s - 15:
                    break
                def obj_r(hh):
                    corr = np.correlate(hh, 1.0 - hh, mode="full")
                    return smooth_max(corr, float(beta2)) * dx2
                try:
                    res = minimize(obj_r, h2, method='SLSQP', bounds=bounds2,
                                  constraints=con2, options={'maxiter': 2000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts2)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        update(h_new, c5_new, n2)
                        h2 = h_new
                except Exception:
                    pass

    # Phase 4: Adam-based local search on best
    if time.time() - start_time < budget_s - 20:
        nf = best_n
        tsf = nf / 2.0
        dxf = 2.0 / nf
        def obj_adam(hh):
            corr = np.correlate(hh, 1.0 - hh, mode="full")
            return smooth_max(corr, 2000.0) * dxf
        try:
            h_adam, val_adam = adam_projected(obj_adam, best_h, np.zeros(nf), np.ones(nf), tsf,
                                              lr=0.005, iters=3000)
            c5_adam = compute_c5(h_adam)
            if c5_adam < best_c5:
                update(h_adam, c5_adam, nf)
        except Exception:
            pass

    # Phase 5: Final high-precision polish
    if best_h is not None and time.time() - start_time < budget_s - 3:
        nf = best_n
        tsf = nf / 2.0
        dxf = 2.0 / nf
        boundsf = Bounds(np.zeros(nf), np.ones(nf))
        conf = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tsf,
                'jac': lambda x: np.ones(nf)}]
        for beta_f in [5000, 10000]:
            if time.time() - start_time > budget_s - 1:
                break
            def obj_f(hh):
                corr = np.correlate(hh, 1.0 - hh, mode="full")
                return smooth_max(corr, float(beta_f)) * dxf
            try:
                res = minimize(obj_f, best_h, method='SLSQP', bounds=boundsf,
                              constraints=conf, options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, tsf)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass

    if best_h is None:
        best_h = h_parent
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
