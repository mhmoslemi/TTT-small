import numpy as np
from scipy.optimize import minimize, Bounds, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def get_grad_at_lag(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        idx = np.arange(m)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones_like(h)

        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        h = project_feasible(h_start.copy(), target_sum)
        local_best = compute_c5(h)

        # Continuation SLSQP
        for beta in [10, 50, 200, 1000, 5000, 20000, 100000]:
            if tl() < alloc:
                break
            for trial in range(2):
                if tl() < alloc - 1:
                    break
                x0 = h.copy()
                if trial == 1:
                    x0 += np.random.randn(n) * 0.002
                    x0 = project_feasible(x0, target_sum)
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 10000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < local_best:
                        h = h_new
                        local_best = c5_new
                except Exception:
                    pass

        # Momentum subgradient descent
        if tl() > alloc + 15:
            h2 = h.copy()
            v = np.zeros(n)
            mom = 0.9
            for it in range(150000):
                if tl() < alloc + 3:
                    break
                corr = np.correlate(h2, 1.0 - h2, mode='full')
                k_worst = np.argmax(corr)
                grad = get_grad_at_lag(h2, k_worst, n)
                lr = 0.006 / (1.0 + 0.002 * it)
                v = mom * v - lr * grad
                h2 = project_feasible(h2 + v, target_sum)
                if it % 4000 == 3999:
                    c5_ck = compute_c5(h2)
                    if c5_ck < local_best:
                        h = h2.copy()
                        local_best = c5_ck

        # High-beta polish
        if tl() > alloc + 5:
            for beta in [50000, 200000, 1000000]:
                if tl() < alloc + 2:
                    break
                try:
                    res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 15000, 'ftol': 1e-16})
                    h_new = project_feasible(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < local_best:
                        h = h_new
                        local_best = c5_new
                except Exception:
                    pass

        if local_best < best_c5:
            best_c5 = local_best
            best_h = h.copy()
            best_n = n

    # Phase 2: Main optimization at parent resolution
    optimize_full(best_h, best_n, 280)

    # Phase 3: Resolution sweep
    n_cur = len(best_h)
    tried_n = {n_cur}
    res_list = []
    for dn in [-30, -20, -10, -5, 5, 10, 20, 30, 40, 50, 60]:
        nc = n_cur + dn
        if 30 <= nc <= 200 and nc not in tried_n:
            res_list.append(nc)
    for nc in [40, 50, 60, 80, 90, 100, 110, 120, 140]:
        if nc not in tried_n:
            res_list.append(nc)

    for nc in sorted(res_list, key=lambda x: abs(x - n_cur)):
        if tl() < 70:
            break
        h_try = resample_h(best_h, nc)
        alloc = max(30, min(55, tl() - 70))
        optimize_full(h_try, nc, tl() - alloc)
        tried_n.add(nc)

    # Phase 4: Multi-start diversification at best resolution
    if tl() > 60:
        n_b = len(best_h)
        for rnd in range(6):
            if tl() < 25:
                break
            scale = 0.008 * (1.0 - rnd * 0.12)
            h_r = best_h + np.random.randn(n_b) * scale
            # Also try complement
            if rnd == 3:
                h_r = 1.0 - best_h
            h_r = project_feasible(h_r, n_b / 2.0)
            alloc = max(20, min(35, tl() - 25))
            optimize_full(h_r, n_b, tl() - alloc)

    # Phase 5: Ultra-high-beta final polish
    if tl() > 8:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        bnds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        cons_f = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_f,)}]
        for beta in [500000, 2000000, 10000000]:
            if tl() < 4:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx_f, beta), method='SLSQP',
                               bounds=bnds_f, constraints=cons_f,
                               options={'maxiter': 25000, 'ftol': 1e-17})
                h_new = project_feasible(res.x, ts_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new
            except Exception:
                pass

    return best_h, best_c5, len(best_h)
