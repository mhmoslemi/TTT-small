    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)

                    j = mid[np.argmin(h_cur[mid])]
                    h_cur[i] = min(1.0, h_cur[i] + 0.04)
                    h_cur[j] = max(0.0, h_cur[j] - 0.04)
                    improved = True
            else:
                da = abs(d)
                if d > 0:
                    prods = h_cur[:n-da] * h_cur[da:]
                else:
                    prods = h_cur[da:] * h_cur[:n-da]
                n_worst = min(3, len(prods))
                worst_idx = np.argsort(prods)[-n_worst:]
                for w in worst_idx:
                    if prods[w] > 0.10:
                        if d > 0:
                            i, j = w, w + d
                        else:
                            i, j = w + da, w
                        if h_cur[i] >= h_cur[j]:
                            delta = min(h_cur[i] - 0.05, 0.05)
                            h_cur[i] -= delta
                            h_cur[j] += delta
                        else:
                            delta = min(h_cur[j] - 0.05, 0.05)
                            h_cur[j] -= delta
                            h_cur[i] += delta
                        improved = True
        if not improved:
            break
        h_cur = project_h(h_cur, target)
        c5_new = compute_c5(h_cur)
        if c5_new < c5_cur:
            c5_cur = c5_new
            stuck = 0
            if c5_cur < best_c5:
                best = h_cur.copy()
                best_c5 = c5_cur
        else:
            stuck += 1
    return best, best_c5

def targeted_anneal(h, target, n, max_time, total_budget, t0):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best = h_cur.copy()
    best_c5 = c5_cur
    T = 0.005
    T_min = 0.00003
    cooling = 0.9993
    it0 = time.time()
    while True:
        elapsed_total = time.time() - t0
        if time.time() - it0 > max_time:
            break
        if total_budget - elapsed_total < 45:
            break
        if T < T_min:
            break
        h_new = h_cur.copy()
        r = np.random.random()
        if r < 0.40:
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            top_k = np.argsort(corr)[-2:]
            for ki in top_k:
                d = ki - (n - 1)
                if d == 0:
                    idx = np.random.choice(n, 2, replace=False)
                    for i in idx:
                        if h_cur[i] > 0.5:
                            h_new[i] += 0.04
                        else:
                            h_new[i] -= 0.04
                else:
                    da = abs(d)
                    n_adj = np.random.randint(1, 4)
                    if d > 0:
                        idx = np.random.choice(n - da, min(n_adj, n - da), replace=False)
                    else:
                        idx = np.random.choice(n - da, min(n_adj, n - da), replace=False)
                    for i in idx:
                        j = i + d
                        if 0 <= j < n:
                            if h_cur[i] > h_cur[j]:
                                h_new[i] -= 0.02 + 0.03 * np.random.random()
                                h_new[j] += 0.02 + 0.03 * np.random.random()
                            else:
                                h_new[j] -= 0.02 + 0.03 * np.random.random()
                                h_new[i] += 0.02 + 0.03 * np.random.random()
        elif r < 0.60:
            blen = np.random.randint(2, 7)
            i = np.random.randint(0, max(1, n - blen))
            shift = np.random.randint(-3, 4)
            if shift != 0:
                block = h_new[i:i+blen].copy()
                h_new[i:i+blen] = np.roll(block, shift)
        elif r < 0.80:
            mask = np.random.random(n) < 0.12
            cnt = int(mask.sum())
            if cnt > 0:
                h_new[mask] += np.random.randn(cnt) * 0.04
        else:
            i, j = np.random.choice(n, 2, replace=False)
            delta = 0.04 * (1.0 if h_cur[i] > h_cur[j] else -1.0)
            h_new[i] += delta
            h_new[j] -= delta
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        delta = c5_new - c5_cur
        if delta < 0 or np.random.random() < np.exp(-min(400.0, delta / max(T, 1e-12))):
            h_cur = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best = h_cur.copy()
                best_c5 = c5_cur
        T *= cooling
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    n_init = len(h_init)

    n_candidates = [70, 90, 110]

    for n in n_candidates:
        if budget_s - (time.time() - t0) < 90:
            break

        if n == n_init:
            h_cur = h_init.copy()
        else:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n)
            h_cur = np.interp(x_new, x_old, h_init)

        target = n / 2.0
        h_cur = project_h(h_cur, target)
        c5_cur = compute_c5(h_cur)

        if c5_cur < best_c5:
            best_h, best_c5 = h_cur.copy(), c5_cur

        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

        # Phase A: SLSQP with increasing beta
        for beta in [60, 150, 350]:
            if budget_s - (time.time() - t0) < 55:
                break
            try:
                res = minimize(obj_smooth, h_cur, args=(beta, dx),
                             method='SLSQP', bounds=bounds,
                             constraints=constraints,
                             options={'maxiter': 2500, 'ftol': 1e-14})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < c5_cur:
                    h_cur, c5_cur = h_new.copy(), c5
            except Exception:
                pass

        # Phase B: Targeted pair adjustment
        if budget_s - (time.time() - t0) > 120:
            adj_time = min(55, (budget_s - (time.time() - t0)) // 3)
            h_cur, c5_cur = targeted_adjust(h_cur, target, n, adj_time)

        # Phase C: Targeted annealing
        if budget_s - (time.time() - t0) > 120:
            sa_time = min(200, (budget_s - (time.time() - t0)) // 2)
            h_cur, c5_cur = targeted_anneal(h_cur, target, n, sa_time, budget_s, t0)

        # Phase D: SLSQP after annealing
        for beta in [300, 600]:
            if budget_s - (time.time() - t0) < 25:
                break
            try:
                res = minimize(obj_smooth, h_cur, args=(beta, dx),
                             method='SLSQP', bounds=bounds,
                             constraints=constraints,
                             options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < c5_cur:
                    h_cur, c5_cur = h_new.copy(), c5
            except Exception:
                pass

        # Phase E: Final high-beta polish
        for beta in [800, 1500, 3000]:
            if budget_s - (time.time() - t0) < 18:
                break
            try:
                res = minimize(obj_smooth, h_cur, args=(beta, dx),
                             method='SLSQP', bounds=bounds,
                             constraints=constraints,
                             options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < c5_cur:
                    h_cur, c5_cur = h_new.copy(), c5
            except Exception:
                pass

        if c5_cur < best_c5:
            best_h, best_c5 = h_cur.copy(), c5_cur

    # Final attempt: perturb best and SLSQP
    if budget_s - (time.time() - t0) > 30:
        n_best = len(best_h)
        target_best = n_best / 2.0
        dx_best = 2.0 / n_best
        bounds_best = Bounds(np.zeros(n_best), np.ones(n_best))
        constraints_best = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        for _ in range(3):
            if budget_s - (time.time() - t0) < 15:
                break
            h_pert = best_h.copy()
            n_pert = np.random.randint(3, 8)
            idx = np.random.choice(n_best, n_pert, replace=False)
            h_pert[idx] += np.random.randn(n_pert) * 0.03
            h_pert = project_h(h_pert, target_best)
            for beta in [500, 1200]:
                if budget_s - (time.time() - t0) < 12:
                    break
                try:
                    res = minimize(obj_smooth, h_pert, args=(beta, dx_best),
                                 method='SLSQP', bounds=bounds_best,
                                 constraints=constraints_best,
                                 options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_h(res.x, target_best)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
