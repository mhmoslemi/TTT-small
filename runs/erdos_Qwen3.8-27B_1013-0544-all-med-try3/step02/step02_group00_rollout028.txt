import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    else:
        kp = -k
        grad[kp:] += (1.0 - h[:n-kp])
        grad[:n-kp] -= h[kp:]
    return grad

def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = free.sum()
        if nfree == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / nfree
            h = np.clip(h, 0.0, 1.0)
    return h

def optimize_subgrad(h, target_sum, start_time, budget_s, max_iters=20000, top_m=5, beta=5.0, name=""):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    cur = h.copy()
    stall_count = 0
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 5:
            break
        
        corr = compute_corr(cur)
        top_idx = np.argsort(corr)[::-1][:top_m]
        top_vals = corr[top_idx]

        w = np.exp(beta * (top_vals - top_vals[0]))
        w /= w.sum()
        
        # Weighted gradient
        grad = np.zeros(n)

            if c5t < best_c5 - 1e-14:
                cur = trial
                best_c5 = c5t
                stall_count = 0
                improved = True
                break
            step *= 0.5

            if stall_count > 50:

                stall_count = 0

**Alternative: Use CVXPY for the bilinear subproblem?**

At each subgradient iteration, we take a linearized step. But what if instead, we solve a sequence of QP subproblems?

The objective is max_k corr_k, where corr_k = sum_i h[i] - sum_i h[i]*h[i+k].

If we fix all h[i+k] for i ≠ j, then corr_k as a function of h[j] is linear (as we showed). So we can't form a QP directly.

But we CAN form a linearization: at the current point h*, corr_k ≈ corr_k(h*) + grad_k(h*)^T (h - h*).

So the linearized problem is:
minimize t
subject to: corr_k(h*) + grad_k(h*)^T (h - h*) ≤ t for all k
            sum(h) = n/2
            0 ≤ h ≤ 1

This IS a linear program! We can solve it with CVXPY.

This is the **cutting-plane** or **linearization** approach. At each iteration:
1. Evaluate all corr_k at current h
2. Compute all gradients
3. Solve the LP
4. Update h
5. Repeat

This should converge faster than subgradient because it uses all the linearization information simultaneously.

The LP has:
- n+1 variables (h_0, ..., h_{n-1}, t)
- 2n-1 + 1 + 2n = 5n-1 constraints (approximately)

For n=140, that's about 700 constraints and 141 variables. This is very solvable with any LP solver.

import cvxpy as cp

def lp_step(h, target_sum):
    n = len(h)
    h_vec = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_vec >= 0, h_vec <= 1, cp.sum(h_vec) == target_sum]
    
    for k in range(-(n-1), n):
        # corr_k = sum_i h[i]*(1-h[i+k]) for valid i
        # Linearization at h: corr_k ≈ corr_k(h) + grad_k^T (h_vec - h)
        grad = grad_lag(h, k)
        corr_val = compute_corr(h)[k + n - 1]
        constraints.append(corr_val + grad @ (h_vec - h) <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if h_vec.value is not None:
        h_new = project_h(h_vec.value, target_sum)
        return h_new
    
    return h

    """Solve min-max via linearization (cutting plane)."""
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    best_c5 = compute_c5(cur)
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 5:
            break
        
        corr = compute_corr(cur)
        # Only include lags that are "active" (within epsilon of max)
        max_corr = corr.max()
        active = np.where(corr >= max_corr - 0.5)[0]
        
        # Build LP
        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        for idx in active:
            k = int(idx) - (n - 1)
            g = grad_lag(cur, k)
            c_val = corr[idx]
            constraints.append(c_val + g @ (h_var - cur) <= t)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except:
            try:
                prob.solve(solver=cp.SCS)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = project_h(h_var.value, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            cur = h_new
            best_c5 = c5_new
        elif abs(c5_new - best_c5) < 1e-10:
            # Converged
            break
        else:

            alpha = 0.5
            h_mixed = project_h(alpha * h_var.value + (1-alpha) * cur, target_sum)
            c5_mixed = compute_c5(h_mixed)
            if c5_mixed < best_c5:
                cur = h_mixed
                best_c5 = c5_mixed
    
    return cur, best_c5

import numpy as np
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] -= h[:n - k]
    else:
        kp = -k
        grad[kp:] += (1.0 - h[:n - kp])
        grad[:n - kp] -= h[kp:]
    return grad

def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_step(h, target_sum, top_m=15):
    n = len(h)
    corr = compute_corr(h)
    max_c = corr.max()
    active = np.where(corr >= max_c - 3.0)[0]
    top_idx = np.argsort(corr)[::-1][:top_m]
    active = np.union1d(active, top_idx)
    h_var = cp.Variable(n)
    t = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    for idx in active:
        k = int(idx) - (n - 1)
        g = grad_lag(h, k)
        cons.append(corr[idx] + g @ (h_var - h) <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, rel_tol=1e-9, abs_tol=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=10000)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return project_h(h_var.value, target_sum)

def lp_optimize(h, target_sum, start_time, budget_s, max_iters=300):
    n = len(h)
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 10:
            break
        h_new = lp_step(cur, target_sum)
        if h_new is None:
            break
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            cur = h_new
            best_c5 = c5_new
        else:
            alpha = 0.5
            h_mix = project_h(alpha * h_new + (1 - alpha) * cur, target_sum)
            c5_mix = compute_c5(h_mix)
            if c5_mix < best_c5 - 1e-14:
                cur = h_mix
                best_c5 = c5_mix
            elif it > 5:
                break
    return cur, best_c5

def subgrad_optimize(h, target_sum, start_time, budget_s, max_iters=10000, top_m=3, beta=10.0):
    n = len(h)
    cur = h.copy()
    best_c5 = compute_c5(cur)
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 5:
            break
        corr = compute_corr(cur)
        top_idx = np.argsort(corr)[::-1][:top_m]
        top_vals = corr[top_idx]
        w = np.exp(beta * (top_vals - top_vals[0]))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            lag = int(idx) - (n - 1)
            grad += wi * grad_lag(cur, lag)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        step = 0.04
        improved = False
        for _ in range(12):
            trial = project_h(cur - step * grad, target_sum)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-14:
                cur = trial
                best_c5 = c5t
                stall = 0
                improved = True
                break
            step *= 0.5
        if not improved:
            stall += 1
            if stall > 80:
                cur = project_h(cur + 0.015 * np.random.randn(n), target_sum)
                best_c5 = compute_c5(cur)
                stall = 0
            beta = min(40.0, beta * 1.005)
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

        n_base = len(h_parent)
    else:
        n_base = 140
        h_parent = np.full(n_base, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base

    tl = budget_s - (time.time() - start_time)
    if tl > 60:
        target = n_base / 2.0
        h1, c5_1 = lp_optimize(h_parent.copy(), target, start_time, budget_s, max_iters=250)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_h, best_c5, best_n = h1, c5_1, n_base

    tl = budget_s - (time.time() - start_time)
    if tl > 30:
        target = len(best_h) / 2.0
        h2, c5_2 = subgrad_optimize(best_h.copy(), target, start_time, budget_s,
                                     max_iters=15000, top_m=3, beta=10.0)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    tl = budget_s - (time.time() - start_time)
    if tl > 80:
        n2 = n_base * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n2)
        h_init = project_h(np.interp(x_new, x_old, best_h), target2)
        h3, c5_3 = lp_optimize(h_init, target2, start_time, budget_s, max_iters=150)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n2
        tl = budget_s - (time.time() - start_time)
        if tl > 20:
            target = len(best_h) / 2.0
            h4, c5_4 = subgrad_optimize(best_h.copy(), target, start_time, budget_s,

            if c5_4 < best_c5:
                best_h, best_c5 = h4, c5_4

    tl = budget_s - (time.time() - start_time)
    if tl > 60:
        n_use = best_n
        target = n_use / 2.0
        inits = []
        for freq in [1, 2, 3]:
            s = 0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n_use) / n_use)
            inits.append(project_h(s, target))

        for bs in [7, 14]:
            block = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(bs) / bs)
            pat = np.tile(block, n_use // bs + 1)[:n_use]
            inits.append(project_h(pat, target))
        for h_init in inits:
            if budget_s - (time.time() - start_time) < 20:
                break
            hi, c5i = lp_optimize(h_init, target, start_time, budget_s, max_iters=80)
            c5i = compute_c5(hi)
            if c5i < best_c5:
                best_h, best_c5, best_n = hi, c5i, n_use

    tl = budget_s - (time.time() - start_time)
    if tl > 20:
        target = len(best_h) / 2.0
        h5, c5_5 = subgrad_optimize(best_h.copy(), target, start_time, budget_s,
                                     max_iters=30000, top_m=2, beta=20.0)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    tl = budget_s - (time.time() - start_time)
    if tl > 15:
        target = len(best_h) / 2.0
        h6, c5_6 = lp_optimize(best_h.copy(), target, start_time, budget_s, max_iters=100)
        c5_6 = compute_c5(h6)
        if c5_6 < best_c5:
            best_h, best_c5 = h6, c5_6

    return best_h, compute_c5(best_h), len(best_h)
