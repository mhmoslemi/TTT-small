import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_lags(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            return h
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() > 0:
            step = diff / free.sum()
            h[free] += step
        else:
            h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    return h


def weighted_lag_gradient(h, weights):
    n = len(h)
    offset = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        d = k - offset
        g = np.ones(n)
        if d > 0:
            g[:n - d] -= h[d:]
            g[d:] -= h[:n - d]
        elif d < 0:
            da = -d
            g[da:] -= h[:n - da]
            g[:n - da] -= h[da:]
        else:
            g -= 2.0 * h
        grad += w * g
    return grad


def obj_func(h, beta, dx):
    corr = all_lags(h)
    return smooth_max(corr, beta) * dx


def slsqp_step(h, beta, dx, n, bounds, constraints, maxiter=5000):
    try:
        res = minimize(obj_func, h, args=(beta, dx),
                       method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(res.x, n / 2.0)
    except Exception:
        return h


def subgradient_phase(h, dx, n, target, budget_fn, max_iters=15000):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    v = np.zeros(n)
    mu = 0.85
    lr = 0.005
    beta_sg = 150.0
    no_improve = 0

    for it in range(max_iters):
        if budget_fn() < 30:
            break
        corr = all_lags(h_cur)
        weights = np.exp(beta_sg * (corr - np.max(corr)))
        weights /= weights.sum()
        grad = weighted_lag_gradient(h_cur, weights)
        v = mu * v - lr * grad
        h_try = project_h(h_cur + v, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
            h_cur = h_try
            no_improve = 0
        else:
            h_cur = h_try
            no_improve += 1
            if no_improve > 300:
                v *= 0.5
                lr *= 0.97
                no_improve = 0
            if no_improve > 800:
                beta_sg = min(beta_sg * 1.4, 3000.0)
                no_improve = 400
        if it % 1000 == 0 and it > 0:
            beta_sg = min(beta_sg * 1.3, 3000.0)
    return best_h, best_c5


def coord_descent_phase(h, dx, n, target, budget_fn, max_iters=8000):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()

    for it in range(max_iters):
        if budget_fn() < 20:
            break
        corr = all_lags(h_cur)
        top_k = np.argsort(corr)[::-1][:4]
        c5_cur = compute_c5(h_cur)

        improved = False

            elif d < 0:
                valid = -d
            else:

            if valid <= 0:

            idx = np.random.choice(valid, min(n_adj, valid), replace=False)

                elif d < 0:
                    a, b = i + (-d), i
                else:

                step = 0.015 * diff * (1.0 + 0.5 * np.random.random())
                step = np.clip(step, -0.06, 0.06)
                h_cur[a] -= step
                h_cur[b] += step
            h_cur = project_h(h_cur, target)
            c5_new = compute_c5(h_cur)
