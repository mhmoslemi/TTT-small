import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(x, beta, N):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, beta) * (2.0 / N)

def slsqp_grad(x, beta, N):
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    se = np.sum(e)
    w = e / se
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - (N - 1)
        if d >= 0:
            idx = np.arange(d, N)
            grad[idx] += wk * (1.0 - x[idx - d])
            idx2 = np.arange(0, N - d)
            grad[idx2] += wk * (-x[idx2 + d])
        else:
            m_ = -d
            idx = np.arange(0, N - m_)
            grad[idx] += wk * (1.0 - x[idx + m_])
            idx2 = np.arange(m_, N)
            grad[idx2] += wk * (-x[idx2 - m_])
    return grad * dx

def con_fun(x, N):
    return x.sum() - N / 2.0

def con_jac(x, N):
    return np.ones(N)

def time_remaining(start_time, budget_s):
    return budget_s - (time.time() - start_time)

def make_bimodal(N, rng):
    h = np.zeros(N)
    perm = rng.permutation(N)
    h[perm[:N // 2]] = 1.0
    return h

def make_block(N, rng):
    h = np.zeros(N)
    half = N // 2
    for _ in range(200):
        a = rng.randint(0, N)
        length = rng.randint(2, N // 2 + 1)
        for j in range(length):
            h[(a + j) % N] = 1.0
        if h.sum() == half:
            return h
    return make_bimodal(N, rng)

def make_sinusoidal(N, rng):
    x = np.linspace(0, 2, N, endpoint=False)
    freq = rng.uniform(1.0, 6.0)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x + phase)
    return project_h(h, N / 2.0)

def targeted_adjust(h, target_sum, n_steps=100):

    c5 = compute_c5(h)
    rng = np.random.RandomState(42)
    for _ in range(n_steps):

        if d > 0:

            for i in idx:

    best_c5 = compute_c5(h)
    delta = delta_init

            n_p = rng.randint(4, 14)
            idx = rng.choice(N, n_p, replace=False)
            h0[idx] += rng.randn(n_p) * 0.07
            h0 = project_h(h0, target_sum)
        else:
            h0 = 1.0 - best_h
            h0 = project_h(h0, target_sum)

        for beta in (60, 150, 400):
            if time_remaining(start_time, budget_s) < 30:
                break
            try:
                res = minimize(slsqp_obj, h0, args=(beta, N),
                              jac=slsqp_grad, method='SLSQP',
                              bounds=bnds, constraints=constraints,
                              options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h0 = best_h.copy()
            except Exception:
                pass

    # Phase 4: Coordinate descent
    if time_remaining(start_time, budget_s) > 40:
        best_h, best_c5 = coord_descent(best_h, target_sum, max_sweeps=400, delta_init=0.025)

    # Phase 5: Another targeted adjustment
    if time_remaining(start_time, budget_s) > 30:
        h_t, c5_t = targeted_adjust(best_h, target_sum, n_steps=150)
        if c5_t < best_c5:
            best_h = h_t.copy()
            best_c5 = c5_t

    # Phase 6: Final SLSQP polish with high beta
    for beta in (800, 1200, 2000):
        if time_remaining(start_time, budget_s) < 15:
            break
        try:
            res = minimize(slsqp_obj, best_h, args=(beta, N),
                          jac=slsqp_grad, method='SLSQP',
                          bounds=bnds, constraints=constraints,
                          options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 7: Final fine coordinate descent
    if time_remaining(start_time, budget_s) > 10:
        best_h, best_c5 = coord_descent(best_h, target_sum, max_sweeps=300, delta_init=0.008)

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, N
