    h = np.clip(h, 0.0, 1.0)

        s = h.sum()
        diff = target_sum - s

            top_lags = np.argsort(corr)[-5:]
            for j in top_lags:
                lag = j - (n_points - 1)
                if lag >= 0:
                    nn = n_points - lag
                    h0[lag:nn] -= 0.002 * (1.0 - h0[:nn])
                    h0[:nn] += 0.002 * h0[lag:nn]
                else:
                    kk = -lag
                    nn = n_points - kk
                    h0[:nn] -= 0.002 * (1.0 - h0[kk:])
                    h0[kk:] += 0.002 * h0[:nn]
        elif strat == 5:
            # Partial mirror
            alpha = 0.1 + 0.15 * (attempt % 3)
            h0 = (1 - alpha) * h0 + alpha * h0[::-1]
            h0 += rng.randn(n_points) * 0.003
        elif strat == 6:
            # Simulated annealing style: larger jump
            scale = 0.03 * np.exp(-attempt / 80.0)
            idx = rng.choice(n_points, size=rng.randint(3, 10), replace=False)
            h0[idx] += rng.randn(len(idx)) * scale
        else:
            # Flatten towards mean in high-value regions
            thr = np.mean(h0) + 0.1 * np.std(h0)
            mask = h0 > thr
            h0[mask] = 0.95 * h0[mask] + 0.05 * thr

        h0 = project_h(h0, _TARGET_SUM)

        # Refine
        beta_r = 200 + 200 * (attempt % 5)
        try:
            res = minimize(obj_smooth, h0, jac=grad_smooth,
                          args=(beta_r, dx), method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 4000, 'ftol': 1e-18})
            h_new = project_h(res.x, _TARGET_SUM)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        # If stuck for too long, try a bigger jump
        if no_improve >= 15:
            h_jump = best_h + rng.randn(n_points) * 0.05
            h_jump = project_h(h_jump, _TARGET_SUM)
            try:
                res = minimize(obj_smooth, h_jump, jac=grad_smooth,
                              args=(100, dx), method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_h(res.x, _TARGET_SUM)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
            no_improve = 0

    # Phase 4: Final ultra-precise polish
    for beta in [10000, 20000, 50000]:
        if time_remaining(start_t, budget_s) < 8:
            break
        try:
            res = minimize(obj_smooth, best_h, jac=grad_smooth,
                          args=(beta, dx), method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 20000, 'ftol': 1e-20})
            h_new = project_h(res.x, _TARGET_SUM)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # One more active-set pass at the end
    if time_remaining(start_t, budget_s) > 5:
        corr = get_corr(best_h)
        corr_shifted = corr - np.max(corr)
        for bw in [500, 1000]:
            ev = np.exp(bw * corr_shifted)
            w = ev / np.sum(ev)
            try:
                res = minimize(obj_weighted, best_h, jac=grad_weighted,
                              args=(w, dx), method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 10000, 'ftol': 1e-20})
                h_new = project_h(res.x, _TARGET_SUM)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    best_h = project_h(best_h, _TARGET_SUM)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
