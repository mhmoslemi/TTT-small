def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    """Compute subgradient of max_k corr[k] w.r.t. h, vectorized."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top-k indices
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        d = int(idx) - (n - 1)
        if d >= 0:
            # corr[idx] = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
            # d/dh[j] = (1-h[j-d]) for j>=d, minus h[j+d] for j<=n-1-d
            grad[d:] += (1.0 - h[:n-d])
            grad[:n-d] -= h[d:]
        else:

            dp = -d
            # corr[idx] = sum_{i=0}^{n-1-dp} h[i]*(1-h[i+dp])
            # d/dh[j] = (1-h[j+dp]) for j<=n-1-dp, minus h[j-dp] for j>=dp
            grad[:n-dp] += (1.0 - h[dp:])
            grad[dp:] -= h[:n-dp]
    # Weight by corr values (for a smoother gradient)
    # Actually for subgradient of max, any subgradient works, 
    # but averaging over top-k gives a better direction
    return grad, corr

    """Backtracking line search for projected gradient step."""
    n = len(h)
    dx = 2.0 / n
    step = 0.1
    h_trial = project_h(h - step * grad, target_sum)
    c5_trial = compute_c5(h_trial)
    for _ in range(max_iter):
        if c5_trial <= max_corr - c * step * np.dot(grad, h - h_trial):
            return h_trial, c5_trial
        step *= alpha
        h_trial = project_h(h - step * grad, target_sum)
        c5_trial = compute_c5(h_trial)
    return h_trial, c5_trial

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def vec_subgradient(h, top_k=4):
    """Vectorized subgradient of max_k corr[k] w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        d = int(idx) - (n - 1)
        if d >= 0:
            grad[d:] += (1.0 - h[:n - d])
            if n - d > 0:
                grad[:n - d] -= h[d:]
        else:
            dp = -d
            if n - dp > 0:
                grad[:n - dp] += (1.0 - h[dp:])
                grad[dp:] -= h[:n - dp]
    return grad, corr


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 140
        h_parent = np.full(n_base, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base

    # Resolutions to try
    n_options = sorted(set([n_base, n_base - 10, n_base + 10, n_base + 30, n_base - 30]))
    n_options = [n for n in n_options if 60 <= n <= 300]

    for n_pts in n_options:
        if time_left() < 90:
            break

        target_sum = n_pts / 2.0
        dx_cur = 2.0 / n_pts
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))

        if n_pts == n_base:
            h_init = project_h(h_parent.copy(), target_sum)
        else:
            x_old = np.linspace(0, 2, n_base, endpoint=False)
            x_new = np.linspace(0, 2, n_pts, endpoint=False)
            h_init = project_h(np.interp(x_new, x_old, h_parent), target_sum)

        cur_c5 = compute_c5(h_init)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = h_init.copy(), cur_c5, n_pts

        h_cur = h_init.copy()

        # Phase 1: Vectorized subgradient with adaptive step
        max_iters = 4000
        step = 0.08
        stalled = 0
        for it in range(max_iters):
            if time_left() < 30:
                break
            if stalled > 200:
                break

            grad, corr = vec_subgradient(h_cur, top_k=4)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            grad = grad / gnorm

            improved = False
            for _ in range(15):
                h_trial = project_h(h_cur - step * grad, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-15:
                    h_cur = h_trial
                    cur_c5 = c5_trial
                    improved = True
                    stalled = 0
                    if cur_c5 < best_c5:
                        best_h, best_c5, best_n = h_cur.copy(), cur_c5, n_pts
                    break
                step *= 0.7

            if not improved:
                stalled += 1
                step = max(0.01, step * 1.2)
                # Occasional random kick
                if stalled % 50 == 0:
                    kick = 0.03 * np.random.randn(n_pts)
                    h_trial = project_h(h_cur + kick, target_sum)
                    if compute_c5(h_trial) < cur_c5:
                        h_cur = h_trial
                        cur_c5 = compute_c5(h_cur)
                        stalled = 0
                        if cur_c5 < best_c5:
                            best_h, best_c5, best_n = h_cur.copy(), cur_c5, n_pts

        # Phase 2: SLSQP with smooth max at increasing beta
        for beta in [50, 150, 500, 1500]:
            if time_left() < 40:
                break

            def obj_sb(h_in):
                c = np.correlate(h_in, 1.0 - h_in, mode="full")
                return smooth_max(c, beta) * dx_cur

            def con_sb(x_in):
                return np.sum(x_in) - target_sum

            def jac_sb(x_in):
                return np.ones(x_in.shape)

            constraints = [{'type': 'eq', 'fun': con_sb, 'jac': jac_sb}]
            try:
                res = minimize(obj_sb, h_cur, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 4000, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5:
                    h_cur = h_new.copy()
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_h, best_c5, best_n = h_cur.copy(), cur_c5, n_pts
            except Exception:
                pass

        # Phase 3: Targeted pair optimization on worst lag
        if time_left() < 20:
            continue
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        worst_lag = int(np.argmax(corr)) - (n_pts - 1)
        ad = abs(worst_lag)

        for it in range(3000):
            if time_left() < 5:
                break
            i = np.random.randint(0, n_pts - ad)
            j = i + ad
            # Try to make h[i] and h[j] more anti-correlated
            if h_cur[i] > h_cur[j]:
                delta = 0.05 * (h_cur[i] - h_cur[j])
            else:
                delta = -0.05 * (h_cur[j] - h_cur[i])
            h_trial = h_cur.copy()
            h_trial[i] = np.clip(h_cur[i] + delta, 0, 1)
            h_trial[j] = np.clip(h_cur[j] - delta, 0, 1)
            h_trial = project_h(h_trial, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                h_cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = h_cur.copy(), cur_c5, n_pts

        # Phase 4: Random pair swaps
        if time_left() < 10:
            continue
        for it in range(4000):
            if time_left() < 5:
                break
            i, j = np.random.choice(n_pts, 2, replace=False)
            delta = np.random.uniform(-0.04, 0.04)
            h_trial = h_cur.copy()
            h_trial[i] = np.clip(h_cur[i] + delta, 0, 1)
            h_trial[j] = np.clip(h_cur[j] - delta, 0, 1)
            h_trial = project_h(h_trial, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                h_cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = h_cur.copy(), cur_c5, n_pts

    # Final polish on best solution
    if time_left() > 15 and best_h is not None:
        n_fin = len(best_h)
        target_fin = n_fin / 2.0
        dx_fin = 2.0 / n_fin
        bounds_fin = Bounds(np.zeros(n_fin), np.ones(n_fin))

        for beta in [500, 2000, 5000]:
            if time_left() < 10:
                break

            def obj_fin(h_in):
                c = np.correlate(h_in, 1.0 - h_in, mode="full")
                return smooth_max(c, beta) * dx_fin

            def con_fin(x_in):
                return np.sum(x_in) - target_fin

            def jac_fin(x_in):
                return np.ones(x_in.shape)

            constraints = [{'type': 'eq', 'fun': con_fin, 'jac': jac_fin}]
            try:
                res = minimize(obj_fin, best_h, method='SLSQP', bounds=bounds_fin,
                               constraints=constraints,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_fin)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass

        # Final subgradient pass
        step = 0.03
        for it in range(5000):
            if time_left() < 3:
                break
            grad, _ = vec_subgradient(best_h, top_k=3)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            grad /= gnorm
            improved = False
            for _ in range(10):
                h_trial = project_h(best_h - step * grad, target_fin)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    best_h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
                step *= 0.7
            if not improved:
                step = max(0.005, step * 1.1)

    return best_h, best_c5, best_n
