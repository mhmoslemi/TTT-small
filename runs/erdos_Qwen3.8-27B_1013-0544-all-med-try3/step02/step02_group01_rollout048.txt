    rng = np.random.RandomState(seed)
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    
    # Compute all lags
    corr = np.correlate(h_init, 1.0 - h_init, mode="full")
    n_lags = len(corr)
    
    # Initialize weights
    w = np.ones(n_lags) / n_lags
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    for outer in range(n_outer):
        # Inner: minimize sum_k w[k] * f_k(h) * dx
        # This is minimize sum_k w[k] * sum_i h[i]*(1-h[i+k]) * dx
        # = dx * sum_i h[i] * (1 - sum_k w[k]*h[i+k])  ... not quite, the ranges differ

        h = h_init.copy()
        for inner in range(n_inner):
            corr = np.correlate(h, 1.0 - h, mode="full")
            # Gradient of weighted sum
            grad = np.zeros(n)
            j = np.arange(n)
            for k in range(n_lags):
                wk = w[k]
                if wk < 1e-15:
                    continue
                s = k - (n-1)
                js = j - s
                jp = j + s
                ms = (js >= 0) & (js < n)
                mp = (jp >= 0) & (jp < n)
                if ms.any():
                    grad[ms] += wk * (1.0 - h[js[ms]])
                if mp.any():
                    grad[mp] += (-wk * h[jp[mp]])
            grad *= dx
            # Projected gradient step
            lr = 0.1 / (outer + 1)
            h_new = project_feasible(h - lr * grad, target)
            h = h_new
        
        # Evaluate
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update weights
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        w = w * np.exp(eta * corr)
        w = w / w.sum()
    
    return best_h, best_c5

from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-6) & (h < 1.0 - 1e-6)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, beta, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    val = (m + np.log(np.sum(exps)) / beta) * dx
    return val

def smooth_max_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - x[js[ms]])
        if mp.any():
            grad[mp] += (-wk * x[jp[mp]])
    grad *= dx
    return grad

def nesterov_fast(h, max_iters, lr0, decay, top_k, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new

    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smooth_max_obj(x, beta, target)
    
    def objg(x):
        return smooth_max_grad(x, beta)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def lbfgs_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smooth_max_obj(x, beta, target)
    
    def objg(x):
        return smooth_max_grad(x, beta)
    
    bounds = list(zip(np.zeros(n), np.ones(n)))
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

            best_h = h_new.copy()
