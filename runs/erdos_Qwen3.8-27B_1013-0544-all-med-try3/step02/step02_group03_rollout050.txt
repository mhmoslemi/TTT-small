import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_subgradient(h, k_idx, n):
    """Get subgradient of corr[k_idx] w.r.t. h"""
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # corr[k_idx] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
        # d/dh[j] = (1-h[j+lag]) for j < n-lag
        #           -h[j-lag] for j >= lag
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            if i + lag < n:
                grad[i + lag] += (-h[i])
    else:
        la = -lag
        for i in range(n - la):
            grad[i + la] += (1.0 - h[i])
            grad[i] += (-h[i + la])
    return grad

def get_subgradient_fast(h, k_idx, n):
    """Vectorized subgradient computation"""
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])

    return grad

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    return smooth_max(corr, beta) * dx

    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

        return budget_s - (time.time() - t0)

                    best_c5 = c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def subgrad_fast(h, k_idx, n):
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation
    for beta in [20, 100, 500, 2000]:
        if tl() < 50:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for trial in range(3):
            if tl() < 20:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * (0.015 / trial)
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Subgradient with top-k averaging and momentum
    if tl() > 200:
        h = best_h.copy()
        momentum = np.zeros(n)
        decay = 0.9
        stale_count = 0
        for it in range(300000):
            if tl() < 50:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_cur = float(np.max(corr) * dx)
            top_k = min(5, len(corr))
            top_idx = np.argsort(corr)[-top_k:]
            top_vals = corr[top_idx]
            w = np.exp(40.0 * (top_vals - top_vals[-1]))
            w /= w.sum()
            grad = np.zeros(n)
            for wi, ki in zip(w, top_idx):
                grad += wi * subgrad_fast(h, int(ki), n)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            gap = max(c5_cur - best_c5, 1e-10)
            lr = min(0.03, gap / (gn * gn + 1e-12))
            lr = min(lr, 0.4 / np.sqrt(it + 1))
            if stale_count > 20000:
                lr *= 2.0
                decay = min(decay, 0.7)
            momentum = decay * momentum - lr * grad
            h_new = h + momentum
            h_new = project_feasible(h_new, target_sum)
            h = h_new
            if it % 2000 == 1999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    stale_count = 0
                else:
                    stale_count += 2000
                if stale_count > 50000:
                    decay = max(0.5, decay - 0.1)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 3: Coordinate descent
    if tl() > 80:
        h = best_h.copy()
        base_c5 = best_c5
        for sweep in range(40):
            if tl() < 25:
                break
            order = np.random.permutation(n)
            sweep_gain = 0.0
            for j in order:
                if tl() < 15:
                    break
                old_val = h[j]
                lo = max(0.0, old_val - 0.15)
                hi = min(1.0, old_val + 0.15)
                cands = np.unique(np.concatenate([
                    np.linspace(lo, hi, 25),
                    np.array([0.0, 1.0, 0.5, old_val])
                ]))
                cv_best = old_val
                c5_best = compute_c5(h)
                for cv in cands:
                    h[j] = cv
                    c5_j = compute_c5(h)
                    if c5_j < c5_best:
                        c5_best = c5_j
                        cv_best = cv
                h[j] = cv_best
                h = project_feasible(h, target_sum)
                sweep_gain += (base_c5 - compute_c5(h)) if compute_c5(h) < c5_best else 0
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                base_c5 = c5
            elif sweep > 8:
                h = best_h.copy()
                break

    # Phase 4: High-beta SLSQP
    if tl() > 40:
        for beta in [5000, 20000, 80000]:
            if tl() < 15:
                break
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
            for trial in range(3):
                if tl() < 10:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n) * 0.003
                    x0 = project_feasible(x0, target_sum)
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 10000, 'ftol': 1e-16})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 5: Try resolution changes
    if tl() > 50:
        for n_try in [n + 10, n + 20, n - 5, n + 15]:
            if n_try < 30 or tl() < 25:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
            for beta in [500, 5000, 20000]:
                if tl() < 15:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 5000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 6: Final coordinate descent on best
    if tl() > 30:
        h = best_h.copy()
        for sweep in range(30):
            if tl() < 10:
                break
            for j in range(len(h)):
                if tl() < 8:
                    break
                old_val = h[j]
                cands = np.linspace(max(0, old_val - 0.08), min(1, old_val + 0.08), 17)
                c5_best = compute_c5(h)
                cv_best = old_val
                for cv in cands:
                    h[j] = cv
                    c5_j = compute_c5(h)
                    if c5_j < c5_best:
                        c5_best = c5_j
                        cv_best = cv
                h[j] = cv_best
                h = project_feasible(h, n / 2.0)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            else:
                h = best_h.copy()
                if sweep > 3:
                    break

    return best_h, best_c5, len(best_h)
