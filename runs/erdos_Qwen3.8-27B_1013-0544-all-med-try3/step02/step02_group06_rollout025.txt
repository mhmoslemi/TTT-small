def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_value(corr, dx, beta):
    vals = beta * dx * corr
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(vals - m)))) / beta

def smooth_max_gradient(h, n, beta):
    """Compute gradient of the smooth max objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = beta * dx * corr
    vals -= np.max(vals)
    exp_vals = np.exp(vals)
    p = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        pi = p[i]
        if pi < 1e-18:
            continue
        s = i - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[s:] += pi * (1.0 - h[:n-s])
                grad[:n-s] -= pi * h[s:]
        else:
            ns = -s
            if n - ns > 0:
                grad[:n-ns] += pi * (1.0 - h[ns:])
                grad[ns:] -= pi * h[:n-ns]
    return dx * grad

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def make_objective(n, beta, lam):
    target = n / 2.0
    def obj(h):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        vals = beta * dx * corr
        m = np.max(vals)
        sm = (m + np.log(np.sum(np.exp(vals - m)))) / beta
        penalty = lam * (h.sum() - target) ** 2
        return sm + penalty
    def grad(h):
        g = smooth_max_gradient(h, n, beta)
        g += 2 * lam * (h.sum() - target)
        return g
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def s_max_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = beta * dx * corr
    vals -= vals.max()
    ev = np.exp(vals)
    p = ev / ev.sum()
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        pi = p[i]
        if pi < 1e-16:
            continue
        s = i - (n - 1)
        if s >= 0:
            r = n - s
            if r > 0:
                grad[s:] += pi * (1.0 - h[:r])
                grad[:r] -= pi * h[s:]
        else:
            ns = -s
            r = n - ns
            if r > 0:
                grad[:r] += pi * (1.0 - h[ns:])
                grad[ns:] -= pi * h[:r]
    return dx * grad

def s_obj(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = beta * dx * corr
    m = vals.max()
    return (m + np.log(np.exp(vals - m).sum())) / beta + lam * (h.sum() - n / 2.0) ** 2

def s_grad(h, n, beta, lam):
    return s_max_grad(h, n, beta) + 2.0 * lam * (h.sum() - n / 2.0)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_up(h, n_dst):
    n_src = len(h)
    xs = np.arange(n_src) * 2.0 / n_src
    xd = np.arange(n_dst) * 2.0 / n_dst
    return project(np.interp(xd, xs, h), n_dst / 2.0)

def interp_dn(h, n_dst):
    n_src = len(h)
    out = np.zeros(n_dst)
    for i in range(n_dst):
        lo = int(i * n_src / n_dst)
        hi = max(lo + 1, int((i + 1) * n_src / n_dst))
        out[i] = h[lo:hi].mean()
    return project(out, n_dst / 2.0)

def subgrad(h, n, s, step):
    g = np.zeros(n)
    if s >= 0:
        r = n - s
        g[s:] += 1.0 - h[:r]
        g[:r] -= h[s:]
    else:
        ns = -s
        r = n - ns
        g[:r] += 1.0 - h[ns:]
        g[ns:] -= h[:r]
    return h - step * g

def do_minimize(h, n, beta, lam, maxiter, maxfun):
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(s_obj, h, args=(n, beta, lam), jac=s_grad,
                     method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-17, 'maxfun': maxfun})
        return project(r.x, n / 2.0)
    except Exception:
        return h
