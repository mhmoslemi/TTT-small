import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-13:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-13:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_obj(x, n, alpha):
    return obj_smooth(x, n, alpha)

def slsqp_con(x, n):
    return x.sum() - n / 2.0

def slsqp_optimize(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: slsqp_con(x, n)}]
    try:
        res = minimize(slsqp_obj, h.copy(), args=(n, alpha), bounds=bounds,
                      constraints=cons, method='SLSQP',
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def eval_pair_delta(h, n, p, q, hp, hq, d):
    h[p] = hp + d
    h[q] = hq - d
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))

def pair_optimize(h, n, p, q):
    dx = 2.0 / n
    hp, hq = h[p], h[q]
    d_lo = max(-hp, hq - 1.0)
    d_hi = min(1.0 - hp, hq)
    if d_hi - d_lo < 1e-10:
        return compute_c5(h, n)
    best_d, best_c = 0.0, eval_pair_delta(h, n, p, q, hp, hq, 0.0)
    coarse = np.linspace(d_lo, d_hi, 16)
    for d in coarse:
        c = eval_pair_delta(h, n, p, q, hp, hq, d)
        if c < best_c:
            best_c, best_d = c, d
    w = (d_hi - d_lo) / 32.0
    lo = max(d_lo, best_d - w)
    hi = min(d_hi, best_d + w)
    fine = np.linspace(lo, hi, 32)
    for d in fine:
        c = eval_pair_delta(h, n, p, q, hp, hq, d)
        if c < best_c:
            best_c, best_d = c, d
    h[p] = hp + best_d
    h[q] = hq - best_d
    return best_c

def coord_sweep(h, n, rng):
    best_c5 = compute_c5(h, n)
    order = np.arange(n)
    rng.shuffle(order)
    for i in range(n):
        p = int(order[i])
        for j in range(i + 1, n):
            q = int(order[j])
            c = pair_optimize(h, n, p, q)
            if c < best_c5:
                best_c5 = c
    return h, best_c5
