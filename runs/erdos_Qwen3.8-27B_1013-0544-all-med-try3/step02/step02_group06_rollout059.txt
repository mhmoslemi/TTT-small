import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_1d_pair(h, i, j, n, n_grid=50):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi_val = min(1.0, s)

        return h[i]

def coord_descend(h, target, max_seconds, n_grid=50, seed=0):
    n = len(h)
    h = h.copy()
    t_end = time.time() + max_seconds
    rng = np.random.RandomState(seed)
    prev_c5 = compute_c5(h)
    while time.time() < t_end:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(pairs)
        improved = False
        for i, j in pairs:
            s = h[i] + h[j]
            new_i = solve_1d_pair(h, i, j, n, n_grid)
            if abs(new_i - h[i]) > 1e-13:
                h[i] = new_i
                h[j] = s - new_i
                improved = True
        if not improved:
            break
        c5_now = compute_c5(h)
        if c5_now >= prev_c5 - 1e-10 and improved:
            prev_c5 = c5_now
            break
        prev_c5 = c5_now
    return h


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def _lbfgs_objective(h, n, target, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target) ** 2
