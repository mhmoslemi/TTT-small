```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    diff2 = target_sum - s2
    if abs(diff2) > 1e-12:
        mask = (h > 0.01) & (h < 0.99)
        if mask.sum() > 0:
            h[mask] += diff2 / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_step(h0, target_sum, dx, beta, maxiter=500):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    con = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
            'jac': lambda x: np.ones(n)}]
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, beta) * dx
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0.copy(), compute_c5(h0)

def active_descent(h, target_sum, n_steps=50, scale=0.02):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    dx = 2.0 / n
    s = scale
    for _ in range(n_steps):
        if s < 1e-7:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx = np.argsort(corr)[::-1][:5]
        av = np.mean(corr[idx]) * dx
        grad = np.zeros(n)
        eps = 1e-5
        for i in range(n):
            if best_h[i] <= 0.001 or best_h[i] >= 0.999:
                continue
            hu = best_h.copy()
            hu[i] = min(1.0, best_h[i] + eps)
            hu = project_h(hu, target_sum)
            cu = np.correlate(hu, 1.0 - hu, mode="full")
            grad[i] = (np.mean(cu[idx]) * dx - av) / eps
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        h_new = project_h(best_h - (s / gn) * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            s = min(s * 1.05, 0.05)
        else:
            s *= 0.6
    return best_h, best_c5

def coord_descent(h, target_sum, n_passes=4, delta=0.02):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    d = delta
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            if best_h[i] <= 0.001 or best_h[i] >= 0.999:
                continue
            for dv in [d, -d, d * 2, -d * 2]:
                ht = best_h.copy()
                ht[i] = best_h[i] + dv
                ht = project_h(ht, target_sum)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-13:
                    best_h, best_c5 = ht, c5t
                    improved = True
                    break
        if not improved:
            break
        d *= 0.7
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(70, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    n_list = [70, 80, 90]
    total_budget = budget_s - 40

    for n_pts in n_list:
        if time.time() - start > total_budget - 50:
            break
        ts = n_pts / 2.0
        dx = 2.0 / n_pts
        t0 = time.time()
        t_alloc = total_budget / len(n_list)

        if len(h_parent) == n_pts:
            h0 = h_parent.copy()
        else:
            xp = np.linspace(0, 2, len(h_parent), endpoint=False)
            xn = np.linspace(0, 2, n_pts, endpoint=False)
            h0 = np.interp(xn, xp, h_parent, period=2.0)
            h0 = project_h(h0, ts)

        cur_h, cur_c5 = h0.copy(), compute_c5(h0)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_h.copy(), cur_c5, n_pts

        # Phase 1: Progressive SLSQP
        t_phase1 = t0 + t_alloc * 0.55
        for beta in [40, 80, 150, 300, 500, 800]:
            if time.time() > t_phase1:
                break
            h_new, c5_new = slsqp_step(cur_h, ts, dx, beta, maxiter=600)
            if c5_new < cur_c5 - 1e-14:
                cur_h, cur_c5 = h_new, c5_new

        # Phase 2: Active set descent
        if time.time() < t0 + t_alloc * 0.72:
            cur_h, cur_c5 = active_descent(cur_h, ts, n_steps=60)

        # Phase 3: Coordinate descent
        if time.time() < t0 + t_alloc * 0.85:
            cur_h, cur_c5 = coord_descent(cur_h, ts, n_passes=6, delta=0.02)

        # Phase 4: Final SLSQP polish
        if time.time() < t0 + t_alloc * 0.95:
            h_new, c5_new = slsqp_step(cur_h, ts, dx, 1000, maxiter=1500)
            if c5_new < cur_c5:
                cur_h, cur_c5 = h_new, c5_new

        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_h.copy(), cur_c5, n_pts
        h_parent = best_h if best_n == n_pts else h_parent

    # Random restarts
    if time.time() - start < budget_s - 25:
        nf = best_n
        ts = nf / 2.0
        dx_f = 2.0 / nf
        rng = np.random.RandomState(seed + 7)
        for _ in range(6):
            if time.time() - start > budget_s - 20:
                break
            h_try = best_h + rng.randn(nf) * 0.06
            h_try = project_h(h_try, ts)
            h_new, c5_new = slsqp_step(h_try, ts, dx_f, 200, maxiter=400)
            c5_cd, _ = (lambda hh: (coord_descent(hh, ts, 3, 0.01)))(h_new)
            c5_new = compute_c5(c5_cd)
            if c5_new < best_c5:
                best_h, best_c5 = c5_cd, c5_new

    # Ultimate polish
    if time.time() - start < budget_s - 10 and best_h is not None:
        nf = best_n
        ts = nf / 2.0
        dx_f = 2.0 / nf
        h_new, c5_new = slsqp_step(best_h, ts, dx_f, 1200, maxiter=3000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        best_h, best_c5 = coord_descent(best_h, ts, 3, 0.005)

    return best_h, best_c5, best_n
```
