def project_to_simplex_with_bounds(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    h = h.copy()
    # Bisection on threshold t
    # We want sum(clip(h_i - t, 0, 1)) = target_sum
    lo, hi = h.min() - target_sum, h.max()
    for _ in range(100):
        t = (lo + hi) / 2
        s = np.clip(h - t, 0, 1).sum()
        if s > target_sum:

        else:
            hi = t

    """Project h onto {x: 0<=x_i<=1, sum(x_i)=target_sum} using bisection"""
    n = len(h)
    lo = h.min() - target_sum / n - 1
    hi = h.max() + 1
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target_sum:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def smooth_max_value(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def softmax_weights(corr, beta):
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    return e / e.sum()

def compute_grad(h, beta):
    """Analytical gradient of smooth_max(corr(h), beta) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = softmax_weights(corr, beta)
    
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - n1
        i_lo = max(0, -d)
        i_hi = min(n, n - d)  # exclusive
        
        # Term 1: (1-h[j-d]) for j where j-d in [i_lo, i_hi)
        j_lo = max(0, i_lo + d)
        j_hi = min(n, i_hi + d)
        if j_lo < j_hi:
            js = np.arange(j_lo, j_hi)
            grad[js] += wk * (1.0 - h[js - d])
        
        # Term 2: -h[j+d] for j in [i_lo, i_hi)
        if i_lo < i_hi:
            js = np.arange(i_lo, i_hi)
            grad[js] -= wk * h[js + d]
    
    return dx * grad

def obj_fn(h, beta):
    return smooth_max_value(np.correlate(h, 1.0 - h, mode="full"), beta) * (2.0 / len(h))

def grad_fn(h, beta):
    return compute_grad(h, beta)

def constraint_eq(x):
    return np.sum(x) - x.size / 2.0

def constraint_eq_jac(x):
    return np.ones(x.size)

def pgd_step(h, target_sum, beta, lr, momentum):
    """One round of projected gradient with momentum, returns improved h"""
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(800):
        g = compute_grad(h, beta)
        v = momentum * v - lr * g
        h_new = project_onto_set(h + v, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        # Adaptive lr
        if it > 0 and it % 50 == 0:
            lr *= 0.8
    
    return best_h, best_c5

from scipy.optimize import minimize, Bounds
import time

_gbeta = 100.0
_global_t0 = 0.0
_global_budget = 1100.0

def _time_left():
    return _global_budget - (time.time() - _global_t0)

def project_onto_set(h, target_sum):
    n = len(h)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target_sum:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def softmax_w(corr, beta):
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    return e / e.sum()

def compute_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = softmax_w(corr, beta)
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = n1 - k
        i_lo = max(0, -L)
        i_hi = min(n, n - L)
        if i_lo < i_hi:
            js = np.arange(i_lo, i_hi)
            grad[js] += wk * (1.0 - h[js + L])
        j_lo = max(0, i_lo + L)
        j_hi = min(n, i_hi + L)
        if j_lo < j_hi:
            js = np.arange(j_lo, j_hi)
            grad[js] -= wk * h[js - L]
    return dx * grad

def _obj(x):
    n = len(x)
    return smooth_max_val(np.correlate(x, 1.0 - x, mode="full"), _gbeta) * (2.0 / n)

def _grad(x):
    return compute_grad(x, _gbeta)

def _cfun(x):
    return np.sum(x) - x.size / 2.0

def _cjac(x):
    return np.ones(x.size)

def pgd_run(h, target_sum, beta, lr, momentum, n_iter):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        g = compute_grad(h, beta)
        v = momentum * v - lr * g
        h_new = project_onto_set(h + v, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        if it % 80 == 79:
            lr *= 0.7
    return best_h, best_c5
