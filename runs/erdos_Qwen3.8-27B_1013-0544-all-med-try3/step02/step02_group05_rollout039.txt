- By KKT: h'_i = clip(x_i - λ, 0, 1) for some λ
- Find λ by bisection: sum(clip(x_i - λ, 0, 1)) = n/2

This is O(n log(precision)) which is very efficient.

- Use polyak step size: α_k = (f_best - f_lower_bound) / ||g_k||² (needs a lower bound)
- Use Nesterov acceleration: maintain a momentum vector

- Find the pair (j, j') that most reduces corr[k*] while maintaining feasibility

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum}."""
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def corr_gradient(h, k_star):
    """Compute gradient of corr[k_star] w.r.t. h.
    corr[k] for k = n-1+m: sum_{i=m}^{n-1} h[i]*(1-h[i-m])
    corr[k] for k = n-1-m: sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
    """
    n = len(h)
    grad = np.zeros(n)
    mid = n - 1  # zero-shift lag
    
    if k_star == mid:
        # corr[mid] = sum h[i]*(1-h[i])
        # d/dh[j] = 1 - 2*h[j]
        grad = 1.0 - 2.0 * h
    elif k_star > mid:
        m = k_star - mid
        # corr = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        # d/dh[j] = (1-h[j-m]) for j in [m, n-1] minus h[j+m] for j in [0, n-1-m]
        for j in range(m, n):
            grad[j] += (1.0 - h[j - m])
        for j in range(0, n - m):
            grad[j] -= h[j + m]
    else:
        m = mid - k_star
        # corr = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        # d/dh[j] = (1-h[j+m]) for j in [0, n-1-m] minus h[j-m] for j in [m, n-1]
        for j in range(0, n - m):
            grad[j] += (1.0 - h[j + m])
        for j in range(m, n):
            grad[j] -= h[j - m]
    
    return grad

def subgradient_optimize(h, n, target_sum, max_iter=5000, verbose=False):
    """Subgradient method with momentum for minimizing max corr."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    # Momentum
    v = np.zeros(n)
    beta_mom = 0.9
    alpha = 0.5
    alpha_min = 0.001
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        k_star = np.argmax(corr)
        grad = corr_gradient(h, k_star)
        
        # Momentum
        v = beta_mom * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-12:
            break
        
        # Step
        h_new = h - alpha * v / v_norm
        h_new = project_h(h_new, target_sum)
        
        # Check improvement
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        c5_new = np.max(corr_new) * dx
        
        if c5_new > c5 * (1.0 + 1e-6):
            # Step too large, reduce alpha
            alpha *= 0.7
            if alpha < alpha_min:
                break
        else:
            h = h_new
            # Occasionally increase alpha
            if c5_new < c5 * (1.0 - 1e-4):
                alpha = min(alpha * 1.05, 0.5)
        
        if verbose and it % 100 == 0:
            print(f"  iter {it}, c5={c5:.6f}, alpha={alpha:.4f}")
    
    return best_h, best_c5

def slsqp_polish(h, n, target_sum, beta=200, maxiter=300):
    """SLSQP polish with smooth max."""
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, beta) * dx
    
    def con_fun(x):
        return np.sum(x) - target_sum
    
    def con_jac(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(res.x, target_sum)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h, compute_c5(h)

def coordinate_exchange(h, n, target_sum, max_iter=2000):
    """2-coordinate exchange: increase h[j], decrease h[j'] to reduce max corr."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_val = np.max(corr)
        
        # Find active lag
        k_star = np.argmax(corr)
        grad = corr_gradient(h, k_star)
        
        # Find best pair (j for increase, j' for decrease)
        # We want to maximize grad[j'] - grad[j] (most positive grad gets decreased, most negative gets increased)
        # j should have h[j] < 1 (can increase)
        # j' should have h[j'] > 0 (can decrease)
        
        # Find the coordinate with most positive gradient (decrease it)
        # and most negative gradient (increase it)
        j_decrease = np.argmax(np.where(h > 1e-8, grad, -np.inf))
        j_increase = np.argmin(np.where(h < 1.0 - 1e-8, grad, np.inf))
        
        if j_decrease == -1 or j_increase == -1 or j_decrease == j_increase:
            break
        
        delta_max = min(h[j_decrease], 1.0 - h[j_increase])
        
        # Try a fraction of delta_max
        for frac in [1.0, 0.5, 0.25, 0.1]:
            delta = frac * delta_max
            if delta < 1e-10:
                continue
            h_new = h.copy()
            h_new[j_increase] += delta
            h_new[j_decrease] -= delta
            
            if h_new[j_increase] > 1.0 + 1e-10 or h_new[j_decrease] < -1e-10:
                continue
            
            c5_new = np.max(np.correlate(h_new, 1.0 - h_new, mode="full"))
            if c5_new < c5_val - 1e-10:
                h = h_new
                c5_final = compute_c5(h)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h.copy()
                break
        
        else:
            break
    
    return best_h, best_c5

def interpolate_h(h_from, n_from, n_to):
    """Interpolate h from n_from points to n_to points."""
    x_old = np.linspace(0, 2.0, n_from, endpoint=False)
    x_new = np.linspace(0, 2.0, n_to, endpoint=False)
    h_new = np.interp(x_new, x_old, h_from, period=2.0)
    return project_h(h_new, n_to / 2.0)

def smooth_grad_descent(h, n, target_sum, max_iter=10000, beta0=50, beta_max=500):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    v = np.zeros(n)  # momentum
    beta_mom = 0.85
    alpha = 0.5
    
    for it in range(max_iter):
        beta = min(beta0 * (1.001 ** it), beta_max)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute weights (softmax over corr values)
        shifted = corr - np.max(corr)
        weights = np.exp(beta * shifted)
        wsum = weights.sum()
        weights /= wsum
        
        # Only use lags with significant weight
        threshold = 1.0 / len(corr)
        active = np.where(weights > threshold)[0]
        
        if len(active) == 0:
            active = [np.argmax(corr)]
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k in active:
            grad += weights[k] * corr_gradient(h, k)
        
        # Momentum
        v = beta_mom * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-10:
            break
        
        # Step with backtracking
        for _ in range(10):
            h_new = h - alpha * v / v_norm
            h_new = project_h(h_new, target_sum)
            c5_new = np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) * (2.0 / n)
            if c5_new < c5 * (1.0 + 1e-8):
                h = h_new
                break
            alpha *= 0.7
        else:
            break
        
        # Occasionally increase alpha
        if c5_new < c5 * (1.0 - 0.01):
            alpha = min(alpha * 1.1, 1.0)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(x, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    if k_star == mid:
        return 1.0 - 2.0 * h
    elif k_star > mid:
        m = k_star - mid
        grad = np.zeros(n)
        for j in range(m, n):
            grad[j] += (1.0 - h[j - m])
        for j in range(0, n - m):
            grad[j] -= h[j + m]
        return grad
    else:
        m = mid - k_star
        grad = np.zeros(n)
        for j in range(0, n - m):
            grad[j] += (1.0 - h[j + m])
        for j in range(m, n):
            grad[j] -= h[j - m]
        return grad

def slsqp_polish(h, n, target_sum, beta, maxiter):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, beta) * dx
    
    def cfun(x):
        return np.sum(x) - target_sum
    
    def cjac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

    best_c5 = compute_c5(h)

            break

    h_new = np.interp(x_new, x_old, h_from, period=2.0)
