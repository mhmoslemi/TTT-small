                    grad[j] += (1.0 - h[j + d])
                if j >= d:
                    grad[j] -= h[j - d]

                if j <= n - 1 - d:
                    grad[j] -= h[j + d]
                if j >= d:
                    grad[j] += (1.0 - h[j - d])

def solve_lp_cuts(h, n, top_k=5):
    """Solve LP with cutting plane constraints from top_k worst lags."""
    overlaps = all_overlaps(h)
    # Get top_k worst lags
    idx = np.argsort(overlaps)[::-1][:top_k]
    
    # Build LP: min t
    # s.t. grad_k · x <= t - g_k(h) + grad_k · h for each active k
    #      0 <= x <= 1
    #      sum(x) = n/2
    # Variables: x[0..n-1], t
    # 
    # c = [0, 0, ..., 0, 1] (minimize t)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    A_ub = []
    b_ub = []
    for k in idx:
        g = grad_overlap_at(h, k, n)
        gk = overlaps[k]
        # g · x <= t - gk + g · h
        # g · x - t <= -gk + g · h
        row = np.append(g, -1.0)
        rhs = -gk + g.dot(h)
        A_ub.append(row)
        b_ub.append(rhs)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.fun
    return None, None

import numpy as np
from scipy.optimize import minimize, linprog, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def all_overlaps(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_overlap_at(h, k_out, n):
    lag = k_out - (n - 1)
    grad = np.zeros(n)
    d = abs(lag)
    if d == 0:
        grad = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] -= h[:n - d]
    else:
        grad[:n - d] -= h[d:]
        grad[d:] += (1.0 - h[:n - d])
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def run_lp_step(h, n, top_k=10, use_all=False):
    """One LP cutting-plane step. Returns new h or None."""
    overlaps = all_overlaps(h)
    if use_all:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argsort(overlaps)[::-1][:top_k]
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_rows = []
    b_vals = []
    
    for k in idx:
        g = grad_overlap_at(h, k, n)
        gk = overlaps[k]
        row = np.concatenate([g, [-1.0]])
        rhs = g.dot(h) - gk
        A_rows.append(row)
        b_vals.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

import numpy as np
from scipy.optimize import minimize, linprog, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_overlaps(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_overlap_at(h, k_out, n):
    lag = k_out - (n - 1)
    grad = np.zeros(n)
    d = abs(lag)
    if d == 0:
        grad = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] -= h[:n - d]
    else:
        grad[:n - d] -= h[d:]
        grad[d:] += (1.0 - h[:n - d])
    return grad


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def run_lp_step(h, n, top_k=None):
    overlaps = all_overlaps(h)
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(overlaps)[::-1][:top_k]
    else:
        idx = np.arange(2 * n - 1)

    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_rows = []
    b_vals = []
    for k in idx:
        g = grad_overlap_at(h, k, n)
        gk = overlaps[k]
        row = np.empty(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(g.dot(h) - gk)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def run_lp_optimize(h_init, n, time_limit):
    target = n / 2.0
    t0 = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_size = 1.0
    stagnant = 0
    iter_count = 0

    while time.time() - t0 < time_limit:
        x_new = run_lp_step(h, n, top_k=None)
        if x_new is None:
            break
        iter_count += 1
        improved = False
        alphas = [step_size, 0.5 * step_size, 0.25 * step_size, 0.1 * step_size]
        for alpha in alphas:
            h_try = h + alpha * (x_new - h)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_h = h_try.copy()
                best_c5 = c5_try
                h = h_try.copy()
                improved = True
                stagnant = 0
                step_size = min(1.0, step_size * 1.08)
                break
            elif c5_try < compute_c5(h) + 1e-12:
                h = h_try.copy()
                improved = True
                stagnant = 0
                if c5_try < best_c5:
                    best_h = h_try.copy()
                    best_c5 = c5_try
                step_size = min(1.0, step_size * 1.02)
                break
        if not improved:
            stagnant += 1
            step_size *= 0.75
            if stagnant > 80:
                h = best_h + np.random.randn(n) * 0.008
                h = project_feasible(h, target)
                step_size = 0.6
                stagnant = 0
                if time.time() - t0 > time_limit * 0.85:
                    break
            elif step_size < 0.003:
                break
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_h = h.copy()
            best_c5 = c5_cur

    final_c5 = compute_c5(best_h)
    return best_h, final_c5


def smooth_max_obj(h, dx, beta, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    pen = 100.0 * (h.sum() - target) ** 2
    return sm * dx + pen


def slsqp_con_fun(h, target):
    return h.sum() - target


def slsqp_con_jac(h, target):
    return np.ones(len(h))

    t0 = time.time()

        h_init = np.full(70, 0.5)

    n = len(h_init)

    h_init = project_feasible(h_init, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: LP cutting-plane from initial point
    b1 = remaining() * 0.4
    if b1 > 20:
        h, c5 = run_lp_optimize(h_init, n, b1)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 2: LP with top_k variants and different starting points
    starts = []
    starts.append(best_h + np.random.randn(n) * 0.015)
    # Binary pattern: alternating blocks
    h_bin1 = np.zeros(n)
    block = n // 10
    for b in range(0, n, 2 * block):
        h_bin1[b:b + block] = 1.0
    starts.append(h_bin1)
    # Random binary
    h_bin2 = np.zeros(n)
    perm = np.random.permutation(n)
    h_bin2[perm[:n // 2]] = 1.0
    starts.append(h_bin2)
    # Smoothed version of best with noise
    starts.append(best_h + np.random.randn(n) * 0.04)
    # Structured: two-scale pattern
    h_struct = np.zeros(n)
    for i in range(n):
        h_struct[i] = 1.0 if (i % 4 < 2) else 0.0
    starts.append(h_struct)

    n_starts = len(starts)
    b2 = remaining() * 0.3
    per_start = b2 / n_starts
    for i in range(n_starts):
        if remaining() < 80:
            break
        h_s = project_feasible(starts[i], target)
        h, c5 = run_lp_optimize(h_s, n, per_start)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 3: SLSQP polishing
    if remaining() > 60:
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac,
                 'args': (target,)}]
        for beta in [300, 3000, 30000]:
            if remaining() < 25:
                break
            try:
                res = minimize(smooth_max_obj, best_h, args=(dx, beta, target),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 4: Second LP pass on the polished result
    if remaining() > 40:
        b4 = min(remaining() * 0.5, 120)
        h, c5 = run_lp_optimize(best_h, n, b4)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 5: Try nearby resolutions
    if remaining() > 50:
        for n_try in [n + 10, n - 10, n + 20, n - 20]:
            if n_try < 30 or remaining() < 25:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new_pts = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new_pts, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            bgt = min(25, remaining() * 0.4)
            h_r, c5_r = run_lp_optimize(h_t, n_try, bgt)
            if c5_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c5_r

    # Phase 6: Final SLSQP polish
    if remaining() > 15:
        dx = 2.0 / len(best_h)
        target_f = len(best_h) / 2.0
        bounds_f = Bounds(np.zeros(len(best_h)), np.ones(len(best_h)))
        cons_f = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac,

            h_new = project_feasible(res.x, target_f)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
