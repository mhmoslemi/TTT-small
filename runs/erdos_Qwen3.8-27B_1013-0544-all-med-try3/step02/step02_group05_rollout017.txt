    idx_minus = i - offsets  # indices for h in the second term

    c[valid_minus] -= h[idx_minus[valid_minus]]

    a = corr - h[i] * c
    # We want to minimize max_k (a_k + c_k * t) for t in [0, 1]
    
    # Find the minimum of max of linear functions
    # The optimum is either at t=0, t=1, or at an intersection of two lines
    # For efficiency, find the two largest lines at t=0 and t=1
    
    vals_at_0 = a
    vals_at_1 = a + c

                    best_val = val_cross
