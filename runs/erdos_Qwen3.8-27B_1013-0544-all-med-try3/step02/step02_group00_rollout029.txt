```python
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")

            # For j in [n-absL, n): only first term applies (j >= absL)
            #   grad[j] += (1-h[j-absL])
            
            # Vectorized:
            # Start with -h[j+absL] for j in [0, n-absL)
            grad[:n-absL] += -h[absL:n]  # -h[j+absL] for j=0..n-absL-1
            # Add (1-h[j-absL]) for j in [absL, n)
            grad[absL:] += (1.0 - h[:n-absL])  # (1-h[j-absL]) for j=absL..n-1
    
    return grad * dx
```

Let me also think about whether I should try different n values. The parent is n=140. Maybe n=100 (coarser, but the discrete problem is easier to optimize) could give a better C5? Or n=200 (finer, but harder to optimize)?

Actually, the C5 value depends on the discretization. A coarser grid gives a larger dx, which means each correlation value is larger (since corr * dx is the integral approximation). Wait no:

C5 = max_k corr[k] * dx where corr[k] = sum of products.

For a fixed continuous function h, as n increases:
- dx = 2/n decreases
- corr[k] = sum of about n terms, each of size about h(1-h) ≈ 0.25
- So corr[k] ≈ n * 0.25
- And C5 ≈ n * 0.25 * (2/n) = 0.5

So C5 converges to the continuous value regardless of n. But for small n, the discretization error means we might get a slightly different value. In general, for this problem, finer discretization should give a value closer to the true continuous optimum, which is what we want to minimize.

But the discrete optimization problem is harder for larger n (more dimensions). So there's a tradeoff.

Given that the parent is n=140 and we're trying to shave off 0.0001, I think sticking with n=140 is the right call. The improvement needs to come from better optimization, not finer discretization.

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def compute_grad_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    tol = max(val * 0.01 for val in [max_val]) + 1e-12
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k_idx in active:
        L = int(k_idx) - (n - 1)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        else:
            absL = abs(L)
            if absL >= n:
                continue
            grad[:n - absL] += -h[absL:n]
            grad[absL:] += (1.0 - h[:n - absL])
    return grad * dx


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(140, 0.5)

    n = len(best_h)
    best_c5 = compute_c5(best_h)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    # Phase 1: SLSQP with increasing beta
    for beta in [20, 50, 100, 200, 400, 800, 1500]:
        if time_left() < 130:
            break

        def obj(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx

        def con(x_in):
            return np.sum(x_in) - target_sum

        def con_jac(x_in):
            return np.ones(n)

        constraints = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 3000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 2: Gradient descent with backtracking line search
    for iteration in range(200000):
        if time_left() < 380:
            break
        grad = compute_grad_c5(best_h)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        step = 0.08 / (1.0 + 0.001 * iteration)
        improved = False
        for _ in range(40):
            h_trial = project_h(best_h - step * grad, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_h = h_trial
                best_c5 = c5_trial
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try perturbation to escape
            if iteration % 500 == 0:
                pert = np.random.normal(0, 0.02, n)
                pert -= pert.sum() / n
                h_trial = project_h(best_h + pert, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial

    # Phase 3: Random pair search with simulated annealing
    T = 0.0005
    T_min = 1e-8
    no_improve = 0
    for iteration in range(5000000):
        if time_left() < 100:
            break
        i, j = np.random.choice(n, 2, replace=False)
        # Gradient-informed: move h in direction that decreases C5
        grad = compute_grad_c5(best_h)
        if grad[i] > grad[j]:
            delta = np.random.uniform(0.002, 0.15)
        elif grad[j] > grad[i]:
            delta = -np.random.uniform(0.002, 0.15)
        else:
            delta = np.random.uniform(-0.1, 0.1)

        h_trial = best_h.copy()
        h_trial[i] += delta
        h_trial[j] -= delta
        if h_trial[i] < 0 or h_trial[i] > 1 or h_trial[j] < 0 or h_trial[j] > 1:
            # Clip and reproject
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = project_h(h_trial, target_sum)
        else:
            h_trial = project_h(h_trial, target_sum)

        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-15:
            best_h = h_trial
            best_c5 = c5_trial
            no_improve = 0
        else:
            no_improve += 1
            # Accept with probability based on temperature
            if no_improve % 100 == 0 and c5_trial < best_c5 + T:
                if np.random.random() < np.exp(-(c5_trial - best_c5) / max(T, 1e-12)):
                    best_h = h_trial
        # Cool down every 50000 iterations
        if iteration % 50000 == 49999:
            T = max(T * 0.85, T_min)
        if no_improve > 200000 and T <= T_min:
            break

    # Phase 4: Final SLSQP polish with very high beta
    for beta in [1000, 2000, 5000, 10000]:
        if time_left() < 5:
            break

        def obj_final(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx

        def con_final(x_in):
            return np.sum(x_in) - target_sum

        def con_jac_final(x_in):
            return np.ones(n)

        constraints_f = [{'type': 'eq', 'fun': con_final, 'jac': con_jac_final}]
        try:
            res = minimize(obj_final, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints_f,
                           options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 5: Fine-tune with very small gradient steps
    for iteration in range(50000):
        if time_left() < 2:
            break
        grad = compute_grad_c5(best_h)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        step = 0.01 / (1.0 + 0.0005 * iteration)
        h_trial = project_h(best_h - step * grad, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-16:
            best_h = h_trial
            best_c5 = c5_trial

    return best_h, best_c5, n
```
