def coordinate_pair_minimize(h, i, j, n, target_sum):
    """Given h with h[i] + h[j] = target_sum - sum(others), optimize u = h[i]."""
    # v = target_sum - u where target_sum = sum(others) + u + v... 
    # Actually, let me redefine. The constraint is sum(h) = n/2.
    # If we fix all h[k] for k ≠ i,j, then h[i] + h[j] = n/2 - sum(h[k] for k≠i,j)
    # Let S = n/2 - sum(h[k] for k≠i,j)
    # Then h[j] = S - h[i] = S - u
    # Constraint: 0 ≤ u ≤ 1 and 0 ≤ S-u ≤ 1
    # So max(0, S-1) ≤ u ≤ min(1, S)
    
    S = target_sum - (h.sum() - h[i] - h[j])
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if lo > hi:

    # For each lag s, compute c[s] as function of u
    # c[s] = sum over valid k of h[k](1-h[k+s])

    def obj_1d(u):
        h[i] = u
        h[j] = S - u
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr)
    
    # Use dense sampling + refinement
    us = np.linspace(lo, hi, 200)
    vals = np.array([obj_1d(u) for u in us])
    best_idx = np.argmin(vals)
    # Refine around best
    if 0 < best_idx < len(us)-1:

        hi_r = us[best_idx+1]
    else:
        lo_r, hi_r = lo, hi
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(obj_1d, bounds=(lo_r, hi_r), method='bounded', 
                          options={'xatol': 1e-10})
    h[i] = res.x
    h[j] = S - res.x
    return h

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def coord_descent_sweep(h, n, n_samples=60, rng=None):
    """One sweep of coordinate descent."""
    half = n - 1
    order = np.arange(n)
    if rng is not None:
        rng.shuffle(order)
    for i in order:
        h[i] = 0.0
        base = np.correlate(h, 1.0 - h, mode='full')
        # Compute delta for all 2n-1 lags
        delta = np.zeros(2 * n - 1)
        for lag in range(-(n-1), n):
            d = 0.0
            if 0 <= i - lag <= n - 1:
                d += 1.0 - h[i - lag]
            if 0 <= i + lag <= n - 1:
                d -= h[i + lag]
            delta[lag + half] = d
        # Dense sampling
        us = np.linspace(0.0, 1.0, n_samples)
        vals = np.max(base[None, :] + delta[None, :] * us[:, None], axis=1)
        best_idx = np.argmin(vals)
        if best_idx == 0:
            u_opt = 0.0
        elif best_idx == n_samples - 1:
            u_opt = 1.0
        else:
            lo = us[best_idx - 1]
            hi = us[best_idx + 1]
            us2 = np.linspace(lo, hi, 50)
            vals2 = np.max(base[None, :] + delta[None, :] * us2[:, None], axis=1)
            u_opt = us2[np.argmin(vals2)]
        h[i] = u_opt
    return h

def slsqp_polish(h, n, target_sum, dx, budget_left, beta_seq=(500, 2000, 10000)):
    """SLSQP polishing with continuation on beta."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    
    for beta in beta_seq:
        if budget_left() < 20:
            break
        x0 = best_h.copy()
        try:
            res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass
    return best_h, best_c5
