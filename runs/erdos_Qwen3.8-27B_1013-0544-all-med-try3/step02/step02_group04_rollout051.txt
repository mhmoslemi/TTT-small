    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    n = len(h)

                grad[j] += 1.0 - h[j - L]
            if 0 <= j <= n - 1 - L:

            if M <= j <= n - 1:

    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_all(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:-L]
        g[:n-L] -= h[L:]
    else:
        M = -L
        g[:n-M] += 1.0 - h[M:]
        g[M:] -= h[:n-M]
    return g

    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

        if abs(diff) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)

    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

        h_new = project_feasible(res.x, tsum)

from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_all(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        M = -L
        g[:n-M] += 1.0 - h[M:]
        g[M:] -= h[:n-M]
    return g

def project_feasible(h, tsum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = tsum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if not free.any():
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def slsqp_run(h0, n, tsum, bounds, constr, beta, maxiter):
    def obj(x):
        return smooth_max(corr_all(x), beta) * (2.0 / n)

                      constraints=constr, options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, tsum)
        return h_new
    except:
        return h0.copy()
