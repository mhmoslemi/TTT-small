from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def build_B(w, n):
    """Build the shift matrix B = sum_k w[k] * S_{lag_k}"""
    B = np.zeros((n, n))
    for k in range(len(w)):
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            B[idx, idx + lag] += w[k]
        else:
            la = -lag
            idx = np.arange(n - la)
            B[idx + la, idx] += w[k]
    return B

def weighted_obj_and_grad(h, B, target_sum):
    """Objective: -h^T B h (to minimize the weighted overlap)
       Plus penalty for constraint violation"""
    val = -h @ (B @ h)
    grad = -(B + B.T) @ h
    # Gradient of constraint penalty
    diff = h.sum() - target_sum
    grad += 2.0 * 100.0 * diff  # penalty gradient
    val += 100.0 * diff ** 2
    return val, grad

def project_feasible(h, target_sum):

        if abs(diff) < 1e-14:

    """Compute weighted sum of correlations and its gradient."""
    n = len(h)
    # Compute correlations
    corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h


def wcorr_grad(h, w):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = float(np.dot(w, corr))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for j in range(n - lag):
                grad[j] += wk * (1.0 - h[j + lag])
                grad[j + lag] -= wk * h[j]
        else:
            la = -lag
            for j in range(n - la):
                grad[j + la] += wk * (1.0 - h[j])
                grad[j] -= wk * h[j + la]
    return obj, grad


def slb_step(h, w, target_sum, pen, bounds):
    n = len(h)
    obj0, g0 = wcorr_grad(h, w)
    d0 = h.sum() - target_sum
    obj0 += pen * d0 * d0
    g0 += 2.0 * pen * d0
    res = minimize(
        lambda x: wcorr_grad(x, w)[0] + pen * (x.sum() - target_sum) ** 2,
        h,
        jac=lambda x: wcorr_grad(x, w)[1] + 2.0 * pen * (x.sum() - target_sum),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-11}
    )
    return project_feasible(res.x, target_sum)


def subgrad_polish(h, target_sum, steps, tl, best_c5_ref):
    n = len(h)
    h = h.copy()
    for it in range(steps):
        rem = tl()
        if rem < 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        for k in top_k:
            lag = k - (n - 1)
            if lag >= 0:
                for j in range(n - lag):
                    grad[j] += (1.0 - h[j + lag])
                    grad[j + lag] -= h[j]
            else:
                la = -lag
                for j in range(n - la):
                    grad[j + la] += (1.0 - h[j])
                    grad[j] -= h[j + la]
        lr = 0.0004 / (1.0 + it * 0.0004)
        h = h - lr * grad
        h = project_feasible(h, target_sum)
        if it % 3000 == 2999:
            c5 = compute_c5(h)
            if c5 < best_c5_ref[0]:
                best_c5_ref[0] = c5
                best_c5_ref[1] = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5_ref[0]:
        best_c5_ref[0] = c5
        best_c5_ref[1] = h.copy()
    return h, best_c5_ref


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    # Structured binary start

    for (h_start, n) in starts:
        if tl() < 100:

        dx = 2.0 / n
        target_sum = n / 2.0

            if ws < 1e-30:

                w[np.argmax(corr)] = 1.0

            elif outer % 30 == 29:

        # Phase B: Subgradient polish
        if tl() > 25:

        n = len(h)

        def obj_sm(x, dx_v, b):
            c = np.correlate(x, 1.0 - x, mode='full')
            return smax(c, b) * dx_v

            return x.sum() - ts

            return np.ones(len(x))

            if tl() < 25:
                break
            cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j, 'args': (target_sum,)}]
            try:

                               bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5_ref[0]:
                    best_c5_ref[0] = c5
                    best_c5_ref[1] = h_new.copy()
            except Exception:
                pass

    # Final subgradient polish on overall best
    if tl() > 20 and best_c5_ref[1] is not None:
        h = best_c5_ref[1].copy()
        n = len(h)
        target_sum = n / 2.0
        h, best_c5_ref = subgrad_polish(h, target_sum, 20000, tl, best_c5_ref)

    return best_c5_ref[1], best_c5_ref[0], len(best_c5_ref[1])
