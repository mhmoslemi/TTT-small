import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(obj, beta):
    m = np.max(obj)
    shifted = beta * (obj - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(c)
    w = np.exp(beta * (c - c_max))
    w /= w.sum()
    g = np.zeros(n)
    for j in range(n):
        k_min1 = max(0, j - n + 1)
        k_max1 = min(2 * n - 2, j + n - 1)
        ks1 = np.arange(k_min1, k_max1 + 1)
        idx1 = j - ks1 + n - 1
        s1 = np.dot(w[ks1], 1.0 - h[idx1])
        k_min2 = max(0, n - 1 - j)
        k_max2 = min(2 * n - 2, n - 1 + j)
        ks2 = np.arange(k_min2, k_max2 + 1)
        idx2 = j + ks2
        idx2 -= n - 1
        s2 = np.dot(w[ks2], h[idx2])
        g[j] = dx * (s1 - s2)
    return g

def obj_smax(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(c, beta) * dx

def obj_grad_smax(h, beta, dx):
    return grad_smooth_max(h, beta)

def obj_penalty(h, beta, target, lam, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(c, beta) * dx + lam * (h.sum() - target) ** 2

def obj_penalty_grad(h, beta, target, lam, dx):
    g = grad_smooth_max(h, beta)
    g += 2.0 * lam * (h.sum() - target)
    return g

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    best_h = project_h(h_init, target)
    best_c5 = compute_c5(best_h)

    def tleft():
        return budget_s - (time.time() - t0)

    bounds = Bounds(np.zeros(n), np.ones(n))

    # Phase 1: L-BFGS-B with penalty for integral constraint

                               bounds=bounds, options={'maxiter': 500, 'ftol': 1e-18})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)

                    best_c5 = c5

        c_cur = compute_c5(h_cur)
        lr = 0.03

                break

                c_try = compute_c5(h_try)
                if c_try < c_cur:
                    h_cur = h_try
                    c_cur = c_try
                    improved = True
                    break

            if improved:
