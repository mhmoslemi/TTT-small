Hmm, let me reconsider. The key issues with my approach:
1. The subgradient computation in a loop over j is O(n) per lag, and we do this for each iteration. For n=140, that's fine, but for n=420 it could be slow.
2. The backtracking line search calls compute_c5 multiple times per iteration, which is O(n²) due to np.correlate.
3. The targeted coordinate descent is clever but might not be fast enough.

Let me optimize the subgradient computation. For a given lag k:
grad[j] = (1-h[j+k]) if j+k < n, else 0) + (-h[j-k] if j-k >= 0, else 0)

This can be computed in vectorized form:
- For the first term: grad[j] += (1-h[j+k]) for j = 0 to n-1-k
- For the second term: grad[j] -= h[j-k] for j = k to n-1

So:
grad[:n-k] += (1 - h[k:])
grad[k:] -= h[:n-k]

This is O(n) and vectorizable! Let me use this.

Also, for the compute_c5, I should cache the correlation when possible.

Let me also think about whether there's a smarter approach to the whole problem.

**Alternative: Direct optimization of the max constraint**

The problem is:
minimize t
subject to: sum_i h[i]*(1-h[i+k]) * dx <= t for all k
           sum_i h[i] = n/2
           0 <= h[i] <= 1

This is a non-convex problem (because h[i]*(1-h[i+k]) is bilinear). But if we fix t, the feasibility problem is also non-convex.

However, we can try a **penalty method**: minimize t + lambda * max(0, max_k corr[k]*dx - t)² or something similar.

Actually, the cleanest approach is probably:
1. Use the exact subgradient (fast, vectorized)
2. Use momentum (Nesterov) to accelerate
3. Use a smart step size (Polyak step size if we know a lower bound, or backtracking)
4. Do targeted local repair

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def fast_subgrad(h):
    """Compute subgradient of max_k corr[k]*dx using the argmax lag. Vectorized."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr)) - (n - 1)
    if k_star >= 0:
        grad = np.zeros(n)
        if k_star < n:
            grad[:n - k_star] += (1.0 - h[k_star:])
        grad[k_star:] -= h[:n - k_star]
        return grad * dx, k_star
    else:
        k = -k_star
        grad = np.zeros(n)
        if k < n:
            grad[:-k] += (1.0 - h[:n - k])
            grad[k:] -= (0.0)  # negative lag handled differently
        # For negative lag -k: corr = sum_j h[j+k]*(1-h[j])
        # d/dh[j]: -h[j+k] if j-k valid, (1-h[j+k]) if j+k valid
        grad = np.zeros(n)
        if k < n:
            grad[:n - k] -= h[k:]
            grad[k:] += (1.0 - h[:n - k])
        return grad * dx, k_star


def smooth_max_fn(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta


def subgrad_descent(h, target_sum, n_points, time_budget, seed_rng):
    """Accelerated subgradient with momentum and backtracking."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    velocity = np.zeros(n_points)
    mu = 0.7  # momentum

    for it in range(8000):
        if time_budget() < 10:
            break
        grad, k_star = fast_subgrad(cur)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        # Nesterov momentum
        velocity = mu * velocity - grad
        step = 0.04 / (1.0 + 0.001 * it)

        # Backtracking
        improved = False
        for _bt in range(12):
            h_trial = project_h(cur + step * velocity, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-16:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                break
            step *= 0.4

        if not improved:
            # Plain gradient step
            step = 0.03
            for _bt in range(8):
                h_trial = project_h(cur - step * grad, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-16:
                    cur = h_trial
                    cur_c5 = c5_trial
                    improved = True
                    break
                step *= 0.5

        if not improved and it % 40 == 0 and it > 0:
            # Perturbation
            sigma = 0.04 * (1.0 + 0.5 * np.sin(0.1 * it))
            pert = seed_rng.normal(0, sigma, n_points)
            h_trial = project_h(cur + pert, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5:
                cur = h_trial
                cur_c5 = c5_trial
            velocity *= 0.5

    return cur, cur_c5


def targeted_local_search(h, target_sum, n_points, time_budget, seed_rng):
    """Targeted coordinate descent focusing on the active lag."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    max_rounds = 500

    for rnd in range(max_rounds):
        if time_budget() < 8:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        k_star = int(np.argmax(corr)) - (n_points - 1)
        k_abs = abs(k_star)
        if k_abs >= n_points or k_abs == 0:
            # Try at lag 1
            k_abs = 1
            k_star = 1 if k_star >= 0 else -1

        # Compute overlap contributions at this lag
        if k_star > 0:
            overlap = cur[:n_points - k_star] * (1.0 - cur[k_star:])
            valid_idx = np.arange(n_points - k_star)
        else:
            k_abs = -k_star
            overlap = cur[k_abs:] * (1.0 - cur[:n_points - k_abs])
            valid_idx = np.arange(k_abs, n_points)

        # Find worst position
        j_worst = valid_idx[np.argmax(overlap)]
        # Find best position to receive mass (where cur is small but neighbor is large)
        if k_star > 0:
            help_metric = np.zeros(n_points)
            help_metric[:n_points - k_star] = (1.0 - cur[:n_points - k_star]) * cur[:n_points - k_star]
        else:
            help_metric = np.zeros(n_points)
            help_metric[k_abs:] = (1.0 - cur[k_abs:]) * cur[k_abs:]

        j_help = np.argmax(help_metric)

        if j_worst != j_help:

                h_trial[i] += sign * delta
                h_trial[j] -= sign * delta

                    c5_trial = compute_c5(h_trial)
                    if c5_trial < cur_c5 - 1e-15:
                        cur = h_trial.copy()
                        cur_c5 = c5_trial
                        break
