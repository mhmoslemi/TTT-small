def max_corr_val(h):

def project_h(h, target_sum):

        if abs(s - target_sum) < 1e-14:

        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def binary_hill_climb(bits, n, max_iters=20000, seed=0):
    """Hill climbing on binary sequence."""
    rng = np.random.RandomState(seed)
    best = bits.copy()
    best_mc = max_corr_val(best.astype(float))
    stuck = 0
    
    for it in range(max_iters):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        test = best.copy()
        test[i] = 0
        test[j] = 1
        mc = max_corr_val(test.astype(float))
        if mc < best_mc:
            best = test
            best_mc = mc
            stuck = 0
        else:
            stuck += 1
            if stuck > 500:
                # Simulated annealing: sometimes accept worse
                if rng.random() < np.exp(-(mc - best_mc) / max(0.1, 1.0 - it/max_iters)):
                    best = test
                    best_mc = mc
                    stuck = 0
                else:
                    stuck += 1
        if stuck > 2000:
            break
    
    return best, best_mc

def con_jac(x):
