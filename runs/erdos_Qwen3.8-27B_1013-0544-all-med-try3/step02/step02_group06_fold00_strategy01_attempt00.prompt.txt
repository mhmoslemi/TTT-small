$$C_5 = \max_k \int h(x)(1 - h(x+k)) dx$$

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _objective(h, n, target_sum, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target_sum) ** 2


def _subgradient_step(h, n, k_shift, step_size):
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = j + k_shift
        i2 = j - k_shift
        if 0 <= i1 < n:
            val += (1.0 - h[i1])
        if 0 <= i2 < n:
            val -= h[i2]
        g[j] = val
    return h - step_size * g
