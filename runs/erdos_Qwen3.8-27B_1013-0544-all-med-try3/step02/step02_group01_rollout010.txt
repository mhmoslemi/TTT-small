    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.size > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def nesterov_accel(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def swap_descent(h, n_rounds=5, n_attempts=2000, seed=42):
    n = len(h)
    best = h.copy()
    best_v = compute_c5(best)
    rng = np.random.default_rng(seed)
    for rnd in range(n_rounds):
        improved = False
        for _ in range(n_attempts):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:
                continue
            vi, vj = best[i], best[j]
            best[i], best[j] = vj, vi
            v = compute_c5(best)
            if v < best_v - 1e-12:
                best_v = v
                improved = True
            else:
                best[i], best[j] = vi, vj
        if not improved:
            break
    return best, best_v

def interp_to_n(h, n_new):

    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)

    return project_feasible(h_new, n_new / 2.0)

from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.size > 0:
                h[free] += diff2 / free.size
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def nesterov_accel(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def swap_descent(h, n_rounds, n_attempts, seed):
    n = len(h)
    best = h.copy()
    best_v = compute_c5(best)
    rng = np.random.default_rng(seed)
    for rnd in range(n_rounds):
        improved = False
        for _ in range(n_attempts):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            vi, vj = best[i], best[j]
            best[i], best[j] = vj, vi
            v = compute_c5(best)
            if v < best_v - 1e-13:
                best_v = v
                improved = True
            else:
                best[i], best[j] = vi, vj
        if not improved:
            break
    return best, best_v
