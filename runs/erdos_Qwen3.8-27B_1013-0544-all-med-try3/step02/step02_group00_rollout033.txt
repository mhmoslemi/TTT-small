def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs))) / beta

def grad_weighted_corr(h, p):
    """Gradient of dot(p, corr) w.r.t. h, where corr = correlate(h, 1-h, 'full')"""
    n = len(h)
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k_idx in range(n_lags):
        pk = p[k_idx]
        if abs(pk) < 1e-14:
            continue
        lag = k_idx - (n - 1)
        # First part: (1-h[j+lag]) for valid j+lag
        jp = np.arange(n) + lag
        vp = (jp >= 0) & (jp < n)
        if np.any(vp):
            grad[vp] += pk * (1.0 - h[jp[vp]])
        # Second part: -h[j-lag] for valid j-lag
        jm = np.arange(n) - lag
        vm = (jm >= 0) & (jm < n)
        if np.any(vm):
            grad[vm] -= pk * h[jm[vm]]
    return grad
