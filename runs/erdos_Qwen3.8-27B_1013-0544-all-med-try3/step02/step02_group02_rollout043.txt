def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        s = k_idx - (n - 1)
        # Term 1: (1-h[j+s]) for j where 0<=j+s<n
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wk * (1.0 - h[j_lo+s:j_hi+s])
        # Term 2: -h[j-s] for j where 0<=j-s<n
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wk * h[j_lo2-s:j_hi2-s]
    
    grad *= dx
    return grad

        if abs(diff) < 1e-14:

            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:

            h += (diff / total) * room

            total = h.sum()
            if total < 1e-14:

            h += (diff / total) * h

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()

        w = np.exp(shifted - mx)

    for k_idx in range(len(w)):

        if wk < 1e-10:

        s = k_idx - (n - 1)
        j_lo = max(0, -s)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        s = k_idx - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wk * (1.0 - h[j_lo + s:j_hi + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wk * h[j_lo2 - s:j_hi2 - s]
    grad *= dx
    return grad

def _sobj(hv, n, alpha):
    return smooth_max_val(hv, n, alpha)

def _sjac(hv, n, alpha):
    return smooth_max_grad(hv, n, alpha)

def _scon_fun(hv, n):
    return np.sum(hv) - n / 2.0

def _scon_jac(hv, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scon_fun, 'jac': _scon_jac, 'args': (n,)}]
    try:
        res = minimize(_sobj, h, jac=_sjac, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def momentum_descent(h, n, end_time, alpha, n_iter_max):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    v = np.zeros(n)
    target_sum = n / 2.0
    lr = 0.01
    decay = (1e-8 / lr) ** (1.0 / max(n_iter_max, 1))
    for it in range(n_iter_max):
        if time.time() >= end_time:
            break
        grad = smooth_max_grad(h, n, alpha)
        v = 0.9 * v - lr * grad
        h_new = np.clip(h + v, 0.0, 1.0)
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-14:
            if diff > 0:
                room = 1.0 - h_new
                tot = room.sum()
                if tot > 1e-14:
                    h_new += (diff / tot) * room
            else:
                tot = h_new.sum()
                if tot > 1e-14:
                    h_new += (diff / tot) * h_new
        h = np.clip(h_new, 0.0, 1.0)
        lr *= decay
        if it % 100 == 0:
            c5 = compute_c5(h, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h, n)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def binary_search(h, n, end_time, rng):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    ones_arr = np.where(h_bin > 0.5)[0]
    zeros_arr = np.where(h_bin <= 0.5)[0]
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin, n)
    no_improve = 0
    max_no_improve = 20000
    while time.time() < end_time and no_improve < max_no_improve:
        i = ones_arr[rng.randint(len(ones_arr))]
        j = zeros_arr[rng.randint(len(zeros_arr))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5 = compute_c5(h_bin, n)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_bin.copy()
            ones_arr = np.where(h_bin > 0.5)[0]
            zeros_arr = np.where(h_bin <= 0.5)[0]
            no_improve = 0
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1
    return best_h, best_c5

def generate_initial(rng, n):
    choice = rng.randint(6)
    if choice == 0:
        h = np.full(n, 0.5)
    elif choice == 1:
        h = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
    elif choice == 2:
        p = max(4, n // 10)
        h = np.array([1.0 if (i % p) < p // 2 else 0.0 for i in range(n)])
    elif choice == 3:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.4 * np.sin(np.pi * x)
    elif choice == 4:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.35 * np.sin(2.0 * np.pi * x)
    else:
        h = np.zeros(n)
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
    return project_feasible(h, n)

def optimize_for_n(h, n, end_time, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

    for alpha in [100, 500, 2000, 10000, 50000]:
        if time.time() >= end_time - 15:
            break
        ht = slsqp_opt(best_h, n, alpha, 3000, 1e-15)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    if time.time() < end_time - 40:
        md_end = min(end_time - 15, time.time() + 80)
        ht, c = momentum_descent(best_h, n, md_end, 5000, 50000)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    if time.time() < end_time - 20:
        bs_end = min(end_time - 10, time.time() + 180)
        ht, c = binary_search(best_h, n, bs_end, rng)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c
        if time.time() < end_time - 8:
            ht2 = slsqp_opt(ht, n, 50000, 5000, 1e-15)
            c2 = compute_c5(ht2, n)
            if c2 < best_c5 - 1e-14:
                best_h, best_c5 = ht2, c2

    for alpha in [100000, 500000]:
        if time.time() >= end_time - 5:
            break
        ht = slsqp_opt(best_h, n, alpha, 5000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    return best_h, best_c5
