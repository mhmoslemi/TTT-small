```python
    dx = 2.0 / n

        grad[:nv] += (1.0 - h[k:]) * dx
        grad[k:] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv]) * dx
        grad[:nv] += (-h[kk:]) * dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            h = np.where(h > 0.0, h, 0.0)
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    return float(m + np.logaddexp.reduce(sh) / alpha + mu * (h.sum() - n/2.0)**2)

def jac_smooth(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx = np.where(w > 1e-12 * w.max())[0]
    for mm in idx:
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    grad += 2.0 * mu * (h.sum() - n/2.0)
    return grad

def obj_topk(h, n, shifts, alpha, mu):
    c = compute_c5_vec(h, n)
    v = c[shifts]
    m = v.max()
    sh = alpha * (v - m)
    return float(m + np.logaddexp.reduce(sh) / alpha + mu * (h.sum() - n/2.0)**2)

def jac_topk(h, n, shifts, alpha, mu):
    c = compute_c5_vec(h, n)
    v = c[shifts]
    m = v.max()
    sh = alpha * (v - m)
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, mm in zip(w, shifts):
        if wi < 1e-14: continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    grad += 2.0 * mu * (h.sum() - n/2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas_mus):
    best = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in alphas_mus:
        if time.time() > deadline - 2: break
        try:
            res = minimize(obj_smooth, h, jac=jac_smooth, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-22, 'gtol': 1e-16,
                                   'maxfun': 100000, 'maxls': 100})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c = cn; best = hn; h = hn
        except: pass
    return best, best_c

def run_topk(h, n, deadline, top_k=10):
    best = h.copy()

    bounds = [(0.0, 1.0)] * n
    alphas = [500, 5000, 50000, 500000, 5000000, 50000000]

        mx = cv.max()

        active = np.where(cv >= thresh)[0]

    return best, best_c
```
