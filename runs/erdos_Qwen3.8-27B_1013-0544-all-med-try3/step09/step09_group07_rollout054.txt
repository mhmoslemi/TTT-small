import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    nc = 2 * n - 1
    for k in range(nc):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e10):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full')
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        w = smooth_max_weights(c, alpha)
        grad = np.zeros(n)
        dx = 2.0 / n
        nc = 2 * n - 1
        for k in range(nc):
            if w[k] < 1e-12:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[lag:n] += w[k] * (1.0 - x[0:n - lag])
                grad[0:n - lag] += w[k] * (-x[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += w[k] * (1.0 - x[idx:n])
                grad[idx:n] += w[k] * (-x[0:n - idx])
        grad *= dx
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def interpolate_h(h_src, n_src, n_dst):

    x_dst = np.linspace(0, 2, n_dst)

            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 6: High-alpha L-BFGS polish
    if time.time() < deadline - 50:
        n_final = len(best_h)
        for alpha in [5000000, 50000000, 200000000]:
            if time.time() > deadline - 20:
                break
            h_opt = lbfgsb_optimize(best_h, n_final, alpha, time.time() + 20, maxiter=3000)
            c_opt = compute_c5(h_opt, n_final)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 7: Top-lag local search (intensive)
    if time.time() < deadline - 15:
        n_final = len(best_h)
        ls_deadline = deadline - 3
        h_opt, c_opt = top_lag_local_search(best_h, n_final, ls_deadline, rng, top_k=10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 8: Final micro-polish
    if time.time() < deadline - 2:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 500000000, time.time() + 5, maxiter=2000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
