    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_shifts_obj(x, n, active_indices, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_indices]
    vals_max = vals.max()
    sh = alpha * (vals - vals_max)
    smx = sh.max()
    if smx > 500:
        val = vals_max + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = vals_max + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_shifts_grad(x, n, active_indices, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_indices]
    vals_max = vals.max()
    sh = alpha * (vals - vals_max)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_indices):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_run(h, n, alpha, mu, maxiter, maxfun, deadline):
    if time.time() > deadline - 5:
        return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(c5_smooth_obj, h, jac=c5_smooth_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxfun, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def lbfgs_multi_alpha(h, n, deadline, alpha_list, mu, maxiter=5000, maxfun=10000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alpha_list:
        if time.time() > deadline - 5:
            break
        h_new, c_new = lbfgs_run(best_h, n, alpha, mu, maxiter, maxfun, deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def active_shifts_optimize(h, n, deadline, top_k, alpha_list):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for _ in range(20):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])[:top_k]]
        if len(active) == 0:
            break
        for alpha in alpha_list:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(active_shifts_obj, best_h, jac=active_shifts_grad,
                              args=(n, active, alpha, 1.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxfun': 5000, 'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def sinusoidal_perturb(h, n, amplitude, seed_rng):
    m = seed_rng.randint(1, 20)
    phase = seed_rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n, endpoint=False)
    pert = amplitude * np.sin(2 * np.pi * m * x / 2.0 + phase)
    h_new = h + pert
    return project_feasible(h_new, n)

def multi_freq_perturb(h, n, amplitude, seed_rng):
    x = np.linspace(0, 2, n, endpoint=False)
    pert = np.zeros(n)
    n_freqs = seed_rng.randint(2, 6)
    for _ in range(n_freqs):
        m = seed_rng.randint(1, 15)
        phase = seed_rng.uniform(0, 2 * np.pi)
        amp = amplitude * seed_rng.uniform(0.5, 1.0)
        pert += amp * np.sin(2 * np.pi * m * x / 2.0 + phase)
    h_new = h + pert
    return project_feasible(h_new, n)

def random_coarse_perturb(h, n, amplitude, seed_rng):
    n_blocks = seed_rng.randint(4, 12)
    block_size = n // n_blocks
    pert = np.zeros(n)
    for b in range(n_blocks):
        val = seed_rng.uniform(-amplitude, amplitude)
        lo = b * block_size
        hi = min(n, (b + 1) * block_size)
        pert[lo:hi] = val
    h_new = h + pert
    return project_feasible(h_new, n)

def nesterov_optimize(h, n, deadline, alpha, mu, lr_init, momentum, patience):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    x = h.copy()
    v = np.zeros(n)
    lr = lr_init
    stuck = 0
    while time.time() < deadline:
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-14:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
        grad += 2.0 * mu * (x.sum() - n / 2.0)
        v = momentum * v - lr * grad
        x_new = np.clip(x + v, 0.0, 1.0)
        x = project_feasible(x_new, n)
        c_new = compute_c5(x, n)
        if c_new < best_c:
            best_c = c_new
            best_h = x.copy()
            stuck = 0
        else:
            stuck += 1
        lr *= 0.9995
        if stuck > patience:
            lr *= 0.5
            v *= 0.3
            stuck = 0
            if lr < 1e-9:
                break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha, maxiter):
    def sl_obj(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        return float(val)
    
    def sl_jac(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-14:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
        return grad
    
    def sq_fun(x, n):
        return float(x.sum() - n / 2.0)
    
    def sq_jac(x, n):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(sl_obj, h, jac=sl_jac, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def multires_cycle(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for n_coarse in [max(100, n // 4), max(100, n // 2)]:
        if time.time() > deadline - 30:
            break
        h_c = downsample(best_h, n, n_coarse)
        t_c = min(60, (deadline - time.time()) * 0.2)
        if t_c > 20:
            h_c, c_c = lbfgs_multi_alpha(h_c, n_coarse, time.time() + t_c,
                                          [200, 1000, 5000, 20000], 10.0, 3000, 5000)
        h_f = upsample(h_c, n_coarse, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: L-BFGS at original resolution (n=500) - aggressive alpha sweep
    if time.time() < deadline - 600:
        t1 = min(t0 + 120, deadline - 600)
        best_h, best_c = lbfgs_multi_alpha(best_h, n_orig, t1,
            [100, 500, 2000, 10000, 50000, 200000, 500000], 50.0, 5000, 8000)
    
    # Phase 2: Multi-start with diverse perturbations at n=500
    if time.time() < deadline - 500:
        t2_deadline = min(t0 + 300, deadline - 500)
        perturbations = [
            (sinusoidal_perturb, 0.1),
            (sinusoidal_perturb, 0.3),
            (multi_freq_perturb, 0.15),
            (multi_freq_perturb, 0.3),
            (random_coarse_perturb, 0.1),
            (random_coarse_perturb, 0.25),
            (sinusoidal_perturb, 0.05),
            (multi_freq_perturb, 0.05),
        ]
        for pert_fn, amp in perturbations:
            if time.time() > t2_deadline - 20:
                break
            h_pert = pert_fn(best_h, n_orig, amp, rng)
            # Quick L-BFGS on perturbed solution
            h_try, c_try = lbfgs_multi_alpha(h_pert, n_orig, t2_deadline,
                [1000, 10000, 100000], 50.0, 2000, 3000)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
    
    # Phase 3: Upsample to n=1000 and optimize
    n_high = 1000
    if time.time() < deadline - 350:
        h_high = upsample(best_h, n_orig, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c:
            best_h = h_high
            best_c = c_high
        else:
            best_h = h_high
            best_c = c_high
        
        t3 = min(time.time() + 200, deadline - 350)
        best_h, best_c = lbfgs_multi_alpha(best_h, n_high, t3,
            [200, 1000, 5000, 20000, 100000, 300000], 20.0, 4000, 8000)
        n_orig = n_high
    
    # Phase 4: More restarts at higher resolution
    if time.time() < deadline - 250:
        t4_deadline = min(time.time() + 120, deadline - 250)
        perturbations2 = [
            (sinusoidal_perturb, 0.05),
            (multi_freq_perturb, 0.08),
            (random_coarse_perturb, 0.05),
            (multi_freq_perturb, 0.15),
        ]
        for pert_fn, amp in perturbations2:
            if time.time() > t4_deadline - 20:
                break
            h_pert = pert_fn(best_h, n_orig, amp, rng)
            h_try, c_try = lbfgs_multi_alpha(h_pert, n_orig, t4_deadline,
                [2000, 20000, 200000], 20.0, 2000, 4000)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
    
    # Phase 5: Active shifts optimization
    if time.time() < deadline - 150:
        t5 = min(time.time() + 100, deadline - 150)
        best_h, best_c = active_shifts_optimize(best_h, n_orig, t5, top_k=8,
            alpha_list=[10000, 50000, 200000])
    
    # Phase 6: Nesterov acceleration
    if time.time() < deadline - 80:
        t6 = min(time.time() + 60, deadline - 80)
        best_h, best_c = nesterov_optimize(best_h, n_orig, t6,
            alpha=50000, mu=1.0, lr_init=0.005, momentum=0.9, patience=150)
    
    # Phase 7: Multi-resolution cycle
    if time.time() < deadline - 60:
        t7 = min(time.time() + 40, deadline - 60)
        best_h, best_c = multires_cycle(best_h, n_orig, t7)
    
    # Phase 8: SLSQP polish
    if time.time() < deadline - 15:
        for alpha_sl in [100000, 500000]:
            if time.time() > deadline - 10:
                break
            t8 = min(time.time() + 8, deadline - 8)
            h8, c8 = slqp_polish(best_h, n_orig, t8, alpha=alpha_sl, maxiter=2000)
            if c8 < best_c:
                best_c = c8
                best_h = h8
    
    # Final check and return
    best_h = project_feasible(best_h, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    return best_h, best_c, n_orig
