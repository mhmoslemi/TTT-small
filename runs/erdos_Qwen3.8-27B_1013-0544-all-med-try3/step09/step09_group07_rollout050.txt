def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    h_bin = (h >= 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    return h_bin

def targeted_binary_search(h, n, deadline, rng, max_no_improve=300000):

                break

                contributing_i = []
                for i in range(n - m):

            else:
                contributing_i = []
                for i in range(-lag, n):
                    if h[i] == 1 and h[i - m] == 0:
                        contributing_i.append(i)
            
            if not contributing_i:
                continue
            
            # Try flipping a contributing 1 to 0, and a "safe" 0 to 1
            zeros = np.where(h == 0)[0]
            if len(zeros) == 0:
                continue
            
            # Score zeros: prefer those far from contributing positions of top lags
            n_try = min(15, len(contributing_i))
            for _ in range(n_try):
                if time.time() > deadline:
                    break
                i = contributing_i[rng.randint(len(contributing_i))]

def projected_gradient_descent(h, n, alpha, deadline, step_init=0.02, max_iter=100000):
    """Projected gradient descent on smooth max objective."""
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target_sum = n / 2.0
    dx = 2.0 / n
    step = step_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_all = np.correlate(h, 1 - h, mode='full') * dx
        max_c = c_all.max()
        
        # Soft-max weights
        shifted = alpha * (c_all - max_c)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(c_all)
            w[int(np.argmax(c_all))] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * h) * dx
            elif lag > 0:
                grad[lag:n] += w[k] * (1.0 - h[0:n-lag]) * dx
                grad[0:n-lag] += w[k] * (-h[lag:n]) * dx
            else:
                idx = -lag
                grad[0:n-idx] += w[k] * (1.0 - h[idx:n]) * dx
                grad[idx:n] += w[k] * (-h[0:n-idx]) * dx
        
        # Constraint gradient (maintain sum)
        diff = h.sum() - target_sum
        grad += 1e4 * diff * np.ones(n)
        
        # Projected gradient step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.1)
        else:
            h = h_new  # still move for exploration
            step *= 0.95
        
        if it % 5000 == 0 and step < 0.001:
            break
    
    return best_h, best_c

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    dx = 2.0 / n
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w_sum = w.sum()
        if w_sum < 1e-30:
            obj = m
            grad = np.zeros(n)
        else:
            obj = m + np.log(w_sum) / alpha
            w2 = w / w_sum
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                if w2[k] < 1e-12:
                    continue
                lag = k - (n - 1)
                if lag == 0:
                    grad += w2[k] * (1.0 - 2.0 * x)
                elif lag > 0:
                    grad[lag:n] += w2[k] * (1.0 - x[0:n-lag])
                    grad[0:n-lag] += w2[k] * (-x[lag:n])
                else:
                    idx = -lag
                    grad[0:n-idx] += w2[k] * (1.0 - x[idx:n])
                    grad[idx:n] += w2[k] * (-x[0:n-idx])
            grad *= dx
        
        diff = x.sum() - target_sum
        obj += 1e6 * diff * diff
        grad = grad + 2.0 * 1e6 * diff * np.ones(n)
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h.copy(), method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'maxls': 40, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def thue_morse_start(n):
    """Generate a Thue-Morse-like binary sequence as starting point."""
    h = np.zeros(n)
    for i in range(n):
        # Count bits
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 0 else 0.0
    return make_binary(h, n)

def periodic_start(n, period=7):
    """Generate a periodic starting pattern."""
    h = np.zeros(n)
    pattern = np.zeros(period)
    # Optimal periodic pattern for small periods
    if period == 7:
        pattern = [1, 1, 0, 1, 0, 1, 0]
    elif period == 5:
        pattern = [1, 1, 0, 1, 0]
    else:
        pattern = np.tile([1, 1, 0, 1, 0, 1, 0], period // 7 + 1)[:period]
    h = np.tile(pattern, n // period + 1)[:n].astype(float)
    return make_binary(h, n)
