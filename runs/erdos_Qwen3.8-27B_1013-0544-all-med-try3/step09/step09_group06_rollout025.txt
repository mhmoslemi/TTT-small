    x = np.linspace(0, 2, n, endpoint=False) + 2.0/n/2  # center of each cell
    t = (x + offset) % period
    # Use a triangular wave: high in first half, low in second
    h = np.where(t < period/2, 0.8, 0.2)
    return project_exact(h, n)

def multi_period_init(n, periods):
    """Create initial solution based on multiple periods (beat pattern)."""
    x = np.linspace(0, 2, n, endpoint=False) + 2.0/n/2
    h = np.zeros(n)
    for p in periods:
        t = x % p
        h += np.where(t < p/2, 0.5, -0.5)
    h = 0.5 + 0.3 * h / len(periods)
    return project_exact(h, n)

def project_exact(y, n):

    target = n / 2.0

        else:

            if tr > 1e-15:

            best_h = h_new.copy()
        h_c = h_new
        t += 1
    return best_h, best_c

def optimize_pipeline(h_start, n, deadline, seed):
    h = project_exact(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    remaining = deadline - time.time()
    if remaining < 10:
        return best_h, best_c
    
    # Phase 1: Nesterov with increasing alpha (50% of pipeline time)
    alphas = [30, 80, 200, 500, 1500]
    p1_time = remaining * 0.45
    per_a = p1_time / len(alphas)
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        h_new, c_new = nesterov_opt(h, n, alpha, time.time() + per_a, lr=0.25, beta=0.85)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    # Phase 2: Subgradient (15%)
    if time.time() < deadline - 20:
        p2 = min(25, (deadline - time.time()) * 0.15)
        if p2 > 5:
            h_new, c_new = subgradient_opt(h, n, time.time() + p2, lr=0.03, rng_seed=seed)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    
    # Phase 3: SLSQP (15%)
    if time.time() < deadline - 15:
        for tm in [3, 5]:
            if time.time() > deadline - 8:
                break
            h_new = slsqp_polish(h, n, alpha=50000, maxiter=3000, top_m=tm)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    
    # Phase 4: Local search (15%)
    if time.time() < deadline - 5:
        ls_dead = min(deadline - 2, time.time() + (deadline - time.time()) * 0.5)
        h_new, c_new = local_search(h, n, ls_dead, max_rounds=100)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    best_h = project_exact(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # === Phase A: Optimize from initial (35% budget) ===
    if h_init is not None:
        alloc = min(350, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_pipeline(h_init, n_init, t0 + alloc, seed)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(n_init, 0.5)
        alloc = min(250, (deadline - t0) * 0.3)
        h_opt, c_opt = optimize_pipeline(h_start, n_init, t0 + alloc, seed)
        best_h, best_c = h_opt, c_opt
    
    # === Phase B: Try other resolutions (25% budget) ===
    n_best = len(best_h)
    n_tries = [500, 600, 400, 700]
    if n_best not in n_tries:
        n_tries.insert(0, n_best)
    
    for n_try in n_tries:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 40
        if remaining < 50:
            break
        alloc = min(120, remaining * 0.3)
        if alloc < 40:
            break
        
        if n_try == len(best_h):
            h_start = best_h.copy()
        else:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_start = np.interp(x_new, x_old, best_h)
            h_start = project_exact(h_start, n_try)
        
        h_opt, c_opt = optimize_pipeline(h_start, n_try, time.time() + alloc, seed + n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # === Phase C: Multistart at best resolution (25% budget) ===
    n_best = len(best_h)
    if time.time() < deadline - 60:
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 45))
        
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r == 0:
                hp = best_h + rng.normal(0, 0.015, n_best)
            elif r == 1:
                hp = best_h + rng.normal(0, 0.04, n_best)
            elif r == 2:
                # Perturbed periodic
                x = np.linspace(0, 2, n_best, endpoint=False) + 1.0 / n_best
                period = 2.0 / 3.0
                t = x % period
                hp = np.where(t < period * 0.5, 0.7, 0.3)
                hp = project_exact(hp, n_best)
            elif r == 3:
                # Different periodic
                x = np.linspace(0, 2, n_best, endpoint=False) + 1.0 / n_best
                period = 2.0 / 7.0
                t = x % period
                hp = np.where(t < period * 0.45, 0.75, 0.25)
                hp = project_exact(hp, n_best)
            else:
                # 0-1 random with structure
                hp = np.zeros(n_best)
                chunk = n_best // 5
                for ci in range(5):
                    start = ci * chunk
                    end = start + chunk if ci < 4 else n_best
                    frac = 0.4 + 0.1 * rng.random()
                    n_ones = int(frac * (end - start))
                    idx = rng.choice(end - start, size=n_ones, replace=False)
                    hp[start + idx] = 1.0
                hp = project_exact(hp, n_best)
            
            alloc = min(40, (deadline - time.time() - 20) * 0.3)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_pipeline(hp, n_best, time.time() + alloc, seed + 100 + r)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # === Phase D: Final intensive local search (15% budget) ===
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        
        # SLSQP one more time
        if time.time() < deadline - 15:
            h_new = slsqp_polish(best_h, n, alpha=100000, maxiter=4000, top_m=8)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Extended local search
        if time.time() < deadline - 3:
            ls_dead = deadline - 1
            best_h, best_c = local_search(best_h, n, ls_dead, max_rounds=300)
            best_h = project_exact(best_h, n)
            best_c = compute_c5(best_h, n)
        
        # Binary search (0-1 moves)
        if time.time() < deadline - 2:
            h_new, c_new = binary_search(best_h, n, deadline - 1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            else:
                # Also try: take the binary solution and do a quick continuous optimization
                h_cont = project_exact(h_new + 0.3 * (best_h - h_new), n)
                h_new2, c_new2 = local_search(h_cont, n, deadline - 1, max_rounds=20)
                if c_new2 < best_c:
                    best_c = c_new2
                    best_h = h_new2
    
    if best_h is None:
        best_h = project_exact(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
