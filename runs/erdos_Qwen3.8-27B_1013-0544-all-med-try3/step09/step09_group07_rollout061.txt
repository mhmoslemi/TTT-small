```python
def project_sum(h, target):
    """Project h onto {0 <= h <= 1, sum(h) = target} using bisection on Lagrange multiplier."""
    # We want sum(clip(h - lambda, 0, 1)) = target
    # f(lambda) = sum(clip(h - lambda, 0, 1)) is monotonically decreasing in lambda
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)
```

```python
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    k_star = np.argmax(c)
    lag = k_star - (n - 1)
    
    grad = np.zeros(n)
    if lag == 0:
        grad = (1 - 2*h)
    elif lag > 0 and lag < n:
        grad[lag:n] = (1 - h[0:n-lag])
        grad[0:n-lag] = -h[lag:n]
    elif lag < 0 and -lag < n:
        idx = -lag
        grad[0:n-idx] = (1 - h[idx:n])
        grad[idx:n] = -h[0:n-idx]

    h_new = project_sum(h_new, n/2)
    return h_new, c * dx
```

```python
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - result
            tot = room.sum()
            if tot > 1e-15:
                result = result + (diff / tot) * room
        else:
            tot = result.sum()
            if tot > 1e-15:
                result = result + (diff / tot) * result
    return np.clip(result, 0.0, 1.0)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < lag < n:
        g[0:n - lag] = -h[lag:n]
        g[lag:n] += 1.0 - h[0:n - lag]
    elif -n < lag < 0:
        m = -lag
        g[0:n - m] = -h[m:n]
        g[m:n] += 1.0 - h[0:n - m]
    return g

def lbfgs_objective(x, args_tuple):
    n, alpha, target_sum = args_tuple
    dx = 2.0 / n
    x = np.clip(x, 0.0, 1.0)
    c = compute_all_lags(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    obj_val = m + (mx + np.log(exps.sum())) / alpha
    diff = x.sum() - target_sum
    obj_val += 1e8 * diff * diff
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if abs(lag) >= n:
            continue
        grad += w[k] * lag_grad(x, n, lag)
    grad *= dx
    grad += 2e8 * diff
    return obj_val, grad
```
