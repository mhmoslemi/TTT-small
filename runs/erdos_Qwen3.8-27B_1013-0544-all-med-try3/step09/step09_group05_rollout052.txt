import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def vec_gradient_full(h, n, w):
    """Compute grad of sum_k w[k]*c[k] where c = correlate(h,1-h)*dx."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-16:
            continue
        d = k - (n - 1)
        wkdx = wk * dx
        if d >= 0:
            nv = n - d
            if nv <= 0:
                continue
            grad[0:nv] += wkdx * (1.0 - h[d:n])
            grad[d:n] += -wkdx * h[0:nv]
        else:
            dd = -d
            nv = n - dd
            if nv <= 0:
                continue
            grad[dd:n] += wkdx * (1.0 - h[0:nv])
            grad[0:nv] += -wkdx * h[dd:n]
    return grad

def vec_gradient_top(h, n, w, active_idx):
    """Gradient for subset of shifts only."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_idx:
        wk = w[k]
        if abs(wk) < 1e-16:
            continue
        d = k - (n - 1)
        wkdx = wk * dx
        if d >= 0:
            nv = n - d
            if nv <= 0:
                continue
            grad[0:nv] += wkdx * (1.0 - h[d:n])
            grad[d:n] += -wkdx * h[0:nv]
        else:
            dd = -d
            nv = n - dd
            if nv <= 0:
                continue
            grad[dd:n] += wkdx * (1.0 - h[0:nv])
            grad[0:nv] += -wkdx * h[dd:n]
    return grad

def softmax_smooth(c, alpha):
    """Return (smoothed_value, weights) for log-sum-exp smoothing."""
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    exp_sh = np.exp(sh - mx)
    w = exp_sh / exp_sh.sum()
    val = m + (mx + np.log(exp_sh.sum())) / alpha
    return val, w

def lbfgs_smoothed(h, n, alpha, mu, maxiter=5000, ftol=1e-16, gtol=1e-12):
    """L-BFGS on smoothed full correlation objective."""
    def obj(x):
        c = compute_c5_vec(x, n)
        v, _ = softmax_smooth(c, alpha)
        v += mu * (x.sum() - n / 2.0) ** 2
        return v
    def grad(x):
        c = compute_c5_vec(x, n)
        _, w = softmax_smooth(c, alpha)
        g = vec_gradient_full(x, n, w)
        g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': gtol, 'maxfun': maxiter * 5, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgs_top_shifts(h, n, active_idx, alpha, mu, maxiter=3000):
    """L-BFGS focusing on top shifts."""
    def obj(x):

        vals = c[active_idx]
        m = vals.max()
        sh = alpha * (vals - m)
        mx = sh.max()
        exp_sh = np.exp(sh - mx)

        vals = c[active_idx]
        m = vals.max()
        sh = alpha * (vals - m)
        mx = sh.max()
        exp_sh = np.exp(sh - mx)
        w_full = np.zeros(2 * n - 1)

        g = vec_gradient_top(x, n, w_full, active_idx)

    bounds = [(0.0, 1.0)] * n

        return project_feasible(res.x, n)
    except Exception:
        return h

def custom_projected_grad(h, n, alpha, deadline, lr_init=0.005, decay=0.9995):
    """Custom projected gradient with momentum and adaptive learning rate."""
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    momentum = np.zeros(n)
    lr = lr_init
    stall_count = 0
    best_stall_c = best_c
    
    while time.time() < deadline:
        c = compute_c5_vec(h_curr, n)
        _, w = softmax_smooth(c, alpha)
        grad = vec_gradient_full(h_curr, n, w)
        
        # Add constraint gradient
        constraint_viol = h_curr.sum() - n / 2.0
        grad += 0.01 * constraint_viol * np.ones(n)
        
        # Momentum (heavy ball)
        momentum = 0.8 * momentum - lr * grad
        
        # Projected update
        h_new = h_curr + momentum
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h_curr = h_new.copy()
            stall_count = 0
        else:
            h_curr = h_new.copy()
            stall_count += 1
            if c_new < best_stall_c:
                best_stall_c = c_new
            if stall_count % 100 == 0:
                lr *= 0.7
                momentum *= 0.5
                best_stall_c = best_c
        lr *= decay
    
    return h_curr, best_c

def custom_top_shifts_grad(h, n, alpha, deadline, top_k=10, lr_init=0.003):
    """Adaptively track top shifts and do projected gradient on them."""
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    momentum = np.zeros(n)
    lr = lr_init
    bound = [(0.0, 1.0)]
    
    while time.time() < deadline:
        c = compute_c5_vec(h_curr, n)
        max_val = c.max()
        threshold = max_val - max(1e-8, 0.01 * max_val)
        active = np.where(c >= threshold)[0]
        if len(active) > top_k * 2:
            active = active[np.argsort(-c[active])][:top_k]
        
        vals = c[active]
        m = vals.max()
        sh = alpha * (vals - m)
        mx = sh.max()
        exp_sh = np.exp(sh - mx)
        w_sub = exp_sh / exp_sh.sum()
        
        w_full = np.zeros(len(c))
        w_full[active] = w_sub
        grad = vec_gradient_top(h_curr, n, w_full, active)
        
        constraint_viol = h_curr.sum() - n / 2.0
        grad += 0.01 * constraint_viol * np.ones(n)
        
        momentum = 0.7 * momentum - lr * grad
        h_new = h_curr + momentum
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
        h_curr = h_new
        lr *= 0.9997
    
    return h_curr, best_c

def smart_pair_search(h, n, deadline, max_radius=40):
    """Efficient pairwise search focusing on critical shifts."""
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:

            break
        improved = True

            improved = False
            # Focus on indices where h is in (0.1, 0.9) - most impactful to change

            for i in candidate_idx:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:

                        if c < best_c - 1e-14:

def slqp_optimize(h, n, alpha, maxiter=2000):
    """SLSQP on smoothed objective with equality constraint."""
    def obj(x):
        c = compute_c5_vec(x, n)
        v, _ = softmax_smooth(c, alpha)
        return v
    def jac(x):
        c = compute_c5_vec(x, n)
        _, w = softmax_smooth(c, alpha)
        return vec_gradient_full(x, n, w)
    bounds = [(0.0, 1.0)] * n

    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:
        return h
