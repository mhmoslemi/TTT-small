import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha):
    xc = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(xc, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(w.sum()) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - xc[k:n])
            grad[k:n] += wm * (-xc[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - xc[0:nv])
            grad[0:nv] += wm * (-xc[kk:n])
    return float(val), grad

def nesterov_optimize(h, n, deadline, alpha, n_iter=5000, lr_init=0.01):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    y = h.copy()
    momentum = 0.9
    lr = lr_init
    lr_decay = 0.999
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it > 0:
            t_new = (1 + np.sqrt(1 + 4 * momentum**2)) / 2
            momentum_new = (1 - t_new) / (t_new + 1)
            y = h + momentum_new * (h - y)
            momentum = momentum_new
        
        val, grad = smooth_obj_and_grad(np.clip(y, 0, 1), n, alpha)
        y_new = y - lr * grad
        h_new = project_feasible(y_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        lr *= lr_decay
    
    return best_h, best_c

        except Exception:
            pass
    return h, best_c


def multi_start_refine(h, n, deadline, n_starts=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    for s in range(n_starts):
        if time.time() > deadline - 30:
            break
        perturbation = 0.05 * (rng.rand(n) - 0.5)
        h_pert = project_feasible(h + perturbation, n)
        t_sub = min(time.time() + 30, deadline - 20)
        if t_sub < time.time() + 5:
            break
        h_t, c_t = optimize_lbfgs(h_pert, n, t_sub, alpha=5000, maxiter=1000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n = 400

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS warmup with low alpha
    t1 = min(t0 + 40, deadline - 800)
    if t1 > time.time() + 5:
        best_h, best_c = optimize_lbfgs(best_h, n, t1, alpha=200, maxiter=2000)

    # Phase 2: Nesterov with medium alpha
    t2 = min(t0 + 110, deadline - 600)
    if t2 > time.time() + 5:
        h2, c2 = optimize_nesterov(best_h, n, t2, alpha=5000, n_iter=8000, lr_init=0.004)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: L-BFGS with higher alpha
    t3 = min(t0 + 220, deadline - 450)
    if t3 > time.time() + 10:
        best_h, best_c = optimize_lbfgs(best_h, n, t3, alpha=20000, maxiter=8000)

    # Phase 4: Focused gradient descent
    t4 = min(t0 + 320, deadline - 350)
    if t4 > time.time() + 10:
        best_h, best_c = optimize_focused(best_h, n, t4, lr_init=0.003, top_frac_init=0.015)

    # Phase 5: L-BFGS very high alpha
    t5 = min(t0 + 420, deadline - 280)
    if t5 > time.time() + 10:
        best_h, best_c = optimize_lbfgs(best_h, n, t5, alpha=100000, maxiter=6000)

    # Phase 6: SLSQP with constraint
    t6 = min(t0 + 520, deadline - 200)
    if t6 > time.time() + 10:
        h6, c6 = optimize_slqp(best_h, n, t6, alpha=200000, maxiter=5000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Penalized L-BFGS for tighter constraint handling
    t7 = min(t0 + 620, deadline - 120)
    if t7 > time.time() + 10:
        h7, c7 = optimize_slqp_nocons(best_h, n, t7, alpha=500000, maxiter=4000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Try alternative resolutions
    for n_try in [400, 600, 800]:
        if time.time() > deadline - 50:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / (2.0 * n)
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / (2.0 * n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, n_try)
        c_try = compute_c5(h_try, n_try)

        t_a = min(time.time() + 25, deadline - 30)
        if t_a > time.time() + 5:
            h_try, c_try = optimize_lbfgs(h_try, n_try, t_a, alpha=20000, maxiter=2000)

        t_b = min(time.time() + 15, deadline - 15)
        if t_b > time.time() + 5:
            h_try, c_try = optimize_slqp(h_try, n_try, t_b, alpha=100000, maxiter=2000)

        if c_try < best_c:
            best_c = c_try
            best_h = h_try
            n = n_try

    # Phase 9: Final polish with SLSQP
    if time.time() < deadline - 10:
        best_h, c_final = optimize_slqp(best_h, n, deadline - 5, alpha=500000, maxiter=5000)
        if c_final < best_c:
            best_c = c_final

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
