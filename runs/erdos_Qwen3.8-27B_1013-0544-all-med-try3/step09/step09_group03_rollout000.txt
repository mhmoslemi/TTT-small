import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def constrained_obj(x, n, alpha, mu):
    base = smooth_max_obj(x, n, alpha)
    base += mu * (x.sum() - n / 2.0) ** 2
    return base

def constrained_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, configs=None):

    best_c = compute_c5(h, n)

        configs = [(100, 500.0), (500, 2000.0), (2000, 10000.0),

    for alpha, mu in configs:

            break

            h_new = project_feasible(res.x, n)

def threshold_continuous(h, n):
    target = n / 2
    thresh = np.sort(h)[::-1][int(target) - 1] if target > 0 else 1.0
    h_b = (h >= thresh).astype(float)
    if h_b.sum() > target:
        excess = int(h_b.sum() - target)
        above = np.where(h_b == 1)[0]
        order = np.argsort(h[above])
        for i in range(excess):
            h_b[above[order[i]]] = 0.0
    elif h_b.sum() < target:
        deficit = int(target - h_b.sum())
        below = np.where(h_b == 0)[0]
        order = np.argsort(-h[below])
        for i in range(deficit):
            h_b[below[order[i]]] = 1.0
    return h_b

    return project_feasible(h_new, n_new)

            kk = -k_worst
            region1 = list(range(kk, n))
            region2 = list(range(0, n - kk))
        for i in region1:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                nv = h[i] + delta
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    break
                else:
                    h[i] = best_h[i]
            else:
                continue
            break
        for i in region2:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                nv = h[i] + delta
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    break
                else:
                    h[i] = best_h[i]
            else:
                continue
            break
    return best_h, best_c

def perturbation_refine(h, n, deadline, rng, n_trials=6):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        mags = [0.01, 0.02, 0.03, 0.015, 0.025, 0.04]
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(constrained_obj, h_p, jac=constrained_grad,
                          args=(n, 5000, 20000.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-20, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
        h_b = threshold_continuous(h, n)
        h_sf = soften_binary(h_b, n, sigma=0.06)
        try:
            res2 = minimize(constrained_obj, h_sf, jac=constrained_grad,
                           args=(n, 3000, 10000.0), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-20, 'maxls': 50})
            h_new2 = project_feasible(res2.x, n)
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c:
                best_c = c_new2
                best_h = h_new2.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        n_base = 500
        h_init = np.full(n_base, 0.5)
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Generate and evaluate diverse binary candidates at n_base (30s)
    t1 = min(t0 + 30, deadline - 800)
    if time.time() < t1 - 5:
        patterns = diverse_binary_patterns(n_base, rng)
        for pat in patterns[:40]:
            if time.time() > t1 - 5:
                break
            pat_c = compute_c5(pat, n_base)
            if pat_c < best_c:
                best_c = pat_c
                best_h = pat.copy()
        # Soften best binary and quick optimize
        h_soft = soften_binary(best_h, n_base, sigma=0.08)
        try:
            res = minimize(constrained_obj, h_soft, jac=constrained_grad,
                          args=(n_base, 500, 5000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n_base,
                          options={'maxiter': 5000, 'ftol': 1e-18, 'maxls': 50})
            h_s = project_feasible(res.x, n_base)
            c_s = compute_c5(h_s, n_base)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
        except Exception:
            pass
    
    # Phase 2: Binary annealing at reduced resolution (80s)
    n_anneal = min(200, n_base)
    t2 = min(t0 + 110, deadline - 700)
    if time.time() < t2 - 10:
        # Downsample best_h to n_anneal
        h_coarse = interpolate_to(best_h, n_base, n_anneal)
        h_binary_coarse = threshold_continuous(h_coarse, n_anneal)
        h_anneal, c_anneal = binary_annealing(n_anneal, t2, rng, t_start=0.1, t_end=0.002)
        h_up = interpolate_to(h_anneal, n_anneal, n_base)
        c_up = compute_c5(h_up, n_base)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
    
    # Phase 3: L-BFGS optimization at n_base (200s)
    t3 = min(t0 + 310, deadline - 500)
    if time.time() < t3 - 10:
        best_h, best_c = run_lbfgs(best_h, n_base, t3,
            configs=[(200, 1000.0), (1000, 5000.0), (4000, 20000.0),
                     (10000, 50000.0), (50000, 200000.0)])
    
    # Phase 4: SLSQP polish at n_base (80s)
    t4 = min(t0 + 390, deadline - 400)
    if time.time() < t4 - 10:
        h4, c4 = slqp_polish(best_h, n_base, t4, alpha=2000000, maxiter=4000)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Iterative binary-continuous cycling (150s)
    t5 = min(t0 + 540, deadline - 300)
    if time.time() < t5 - 10:
        for cycle in range(4):
            if time.time() > t5 - 20:
                break
            h_b = threshold_continuous(best_h, n_base)
            h_sf = soften_binary(h_b, n_base, sigma=0.05 + 0.03 * cycle)
            try:
                res = minimize(constrained_obj, h_sf, jac=constrained_grad,
                              args=(n_base, 8000, 50000.0), method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n_base,
                              options={'maxiter': 4000, 'ftol': 1e-20, 'maxls': 50})
                h_c = project_feasible(res.x, n_base)
                c_c = compute_c5(h_c, n_base)
                if c_c < best_c:
                    best_c = c_c
                    best_h = h_c
            except Exception:
                pass
    
    # Phase 6: Upsample to n=800 and optimize (200s)
    n_high = 800
    t6 = min(t0 + 740, deadline - 150)
    if time.time() < t6 - 50:
        h800 = interpolate_to(best_h, n_base, n_high)
        h800, c800 = run_lbfgs(h800, n_high, t6,
            configs=[(500, 2000.0), (2000, 10000.0), (8000, 40000.0),
                     (30000, 150000.0)])
        if c800 < best_c:
            best_c = c800
            best_h = h800
    
    # Phase 7: SLSQP at n=800 (60s)
    t7 = min(t0 + 800, deadline - 100)
    if time.time() < t7 - 10:
        h7, c7 = slqp_polish(best_h, len(best_h), t7, alpha=5000000, maxiter=3000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Perturbation + refine (80s)
    t8 = min(t0 + 880, deadline - 60)
    if time.time() < t8 - 10:
        best_h, best_c = perturbation_refine(best_h, len(best_h), t8, rng, n_trials=5)
    
    # Phase 9: Targeted coordinate descent (60s)
    t9 = deadline - 5
    if time.time() < t9 - 5:
        best_h, best_c = targeted_descent(best_h, len(best_h), t9, n_rounds=200)
    
    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
