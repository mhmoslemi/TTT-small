import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def lse_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:

                grad[k:n] += wm * (-x[0:nv])

def run_lbfgs_stages(h, n, deadline, alpha_schedule=None, mu_schedule=None):
    """Run L-BFGS with increasing alpha and mu schedules."""

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    if alpha_schedule is None:

def steepest_max_descent(h, n, deadline, n_steps=10000, lr_init=0.001):
    """Steepest descent on the max shift value."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline - 5:
            break
        c = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c))
        k = worst_idx - (n - 1)
        dx_val = 2.0 / n
        
        # Gradient of c at the worst shift
        if k >= 0:
            nv = n - k
            grad = np.zeros(n)
            grad[0:nv] += (1.0 - h[k:n]) * dx_val
            grad[k:n] += (-h[0:nv]) * dx_val
        else:
            kk = -k
            nv = n - kk
            grad = np.zeros(n)
            grad[kk:n] += (1.0 - h[0:nv]) * dx_val
            grad[0:nv] += (-h[kk:n]) * dx_val
        
        # Project gradient onto feasible direction (maintain sum)
        mean_grad = grad.mean()
        grad_feas = grad - mean_grad
        
        # Line search
        for ls_try in range(10):
            t = lr * (0.5 ** ls_try)
            h_trial = h + t * grad_feas
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c:
                best_c = c_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                lr = min(lr * 1.1, 0.01)
                break
            elif ls_try == 9:
                lr *= 0.5
        
        if step % 500 == 0 and lr < 1e-7:
            break
    
    return best_h, best_c
