import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def threshold_to_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    target = n // 2
    # Sort by value descending, pick top n/2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def compute_f(h, n, i, j):
    """Compute f[l] for all 2n-1 lags, where Δc[l] = d * f[l] for move d from i to j."""
    lags = np.arange(2 * n - 1) - (n - 1)
    f = np.zeros(2 * n - 1)
    
    idx1 = i + lags
    valid1 = (idx1 >= 0) & (idx1 < n)
    f[valid1] += (1.0 - h[idx1[valid1]])
    
    idx2 = i - lags
    valid2 = (idx2 >= 0) & (idx2 < n)
    f[valid2] -= h[idx2[valid2]]
    
    idx3 = j + lags
    valid3 = (idx3 >= 0) & (idx3 < n)
    f[valid3] -= (1.0 - h[idx3[valid3]])
    
    idx4 = j - lags
    valid4 = (idx4 >= 0) & (idx4 < n)
    f[valid4] += h[idx4[valid4]]
    
    return f

def find_optimal_d(c_vals, f, d_lo, d_hi, n_bisect=50):
    """
    Find d in [d_lo, d_hi] minimizing max(c_vals + d*f).
    Returns (best_d, min_max_value).
    """
    if d_lo >= d_hi:
        return 0.0, float(c_vals.max())
    
    t_lo = float(c_vals.min()) - 1.0
    t_hi = float(c_vals.max())
    
    best_d = 0.0
    for _ in range(n_bisect):
        t_mid = (t_lo + t_hi) / 2.0
        
        pos = f > 1e-15
        neg = f < -1e-15
        zer = ~(pos | neg)
        
        if zer.any():
            if c_vals[zer].max() > t_mid + 1e-14:
                t_lo = t_mid
                continue
        
        lo = d_lo
        hi = d_hi
        if pos.any():
            val = np.min((t_mid - c_vals[pos]) / f[pos])
            if val < hi:
                hi = float(val)
        if neg.any():
            val = np.max((t_mid - c_vals[neg]) / f[neg])
            if val > lo:
                lo = float(val)
        
        if lo <= hi:
            t_hi = t_mid
            best_d = (lo + hi) / 2.0
        else:
            t_lo = t_mid
    
    return best_d, t_hi

def pair_exchange_local_search(h, n, deadline, rng, K=15, S=300, max_sweeps=200):
    """
    Local search using optimal pair exchanges.
    For top K lags, try S random pairs at each lag, find optimal d.
    """

    best_h = h.copy()

                    h[i] += d

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e10):
    target_sum = n / 2.0
    lags = np.arange(2 * n - 1) - (n - 1)
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c_r = np.correlate(x, 1 - x, mode='full')
        dx = 2.0 / n
        c = c_r * dx
        
        # Smooth max
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            obj_val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        else:
            obj_val = m + np.log(np.sum(np.exp(shifted))) / alpha
        
        # Penalty
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        
        # Gradient
        w = np.exp(shifted - mx) if mx > 500 else np.exp(shifted)
        w = w / w.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            lag = lags[k]
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[lag:n] += w[k] * (1.0 - x[0:n-lag])
                grad[0:n-lag] += w[k] * (-x[lag:n])
            else:
                idx = -lag
                grad[0:n-idx] += w[k] * (1.0 - x[idx:n])
                grad[idx:n] += w[k] * (-x[0:n-idx])
        
        grad *= dx
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 80,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def project_feasible(h, n):
