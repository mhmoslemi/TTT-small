    """Standard DE 'DE/rand/1' mutation"""
    n = len(pop[0])
    pop_size = len(pop)
    idx = rng.choice(pop_size, 3, replace=False)
    a, b, c = pop[idx[0]], pop[idx[1]], pop[idx[2]]
    v = a + F * (b - c)
    return v

def de_crossover(v, h_target, rng, CR=0.8):
    """Binary crossover"""
    n = len(h_target)
    mask = rng.random(n) < CR
    if not mask.any():
        mask[rng.integers(n)] = True
    child = np.where(mask, v, h_target)
    return child

def run_de(h_init, n, deadline, pop_size=40, F=0.8, CR=0.8):

    pop = de_population(h_init, n, pop_size, rng)
    fitness = [compute_c5(h, n) for h in pop]
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]

    while time.time() < deadline:
        generation += 1
        for i in range(pop_size):

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def downsample(h_old, n_old, n_new):
    if n_new >= n_old:
        return upsample(h_old, n_old, n_new)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(100, int(remaining * 1.5))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_opt(h, n, deadline, top_k=6, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    for rnd in range(100):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        improved = False
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            maxit = max(50, int(rem2 * 1.0))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(n, active, alpha, 0.002), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': maxit * 10,
                                        'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    improved = True
            except Exception:
                pass
        if not improved and rnd > 20:
            break
    return best_h, best_c

def de_obj_with_penalty(h, n, penalty):
    c = compute_c5(h, n)
    viol = abs(h.sum() - n / 2.0)
    return c + penalty * viol * viol

def run_de(h_init, n, deadline, pop_size=40, F=None, CR=None, seed=42):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    
    if F is None:
        F = 0.7
    if CR is None:
        CR = 0.8
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    pop = [best_h.copy()]
    for i in range(pop_size - 1):
        h = best_h + rng.normal(0, 0.03 + 0.02 * i / pop_size, n)
        h = project_feasible(h, n)
        pop.append(h)
    
    fitness = [compute_c5(h, n) for h in pop]
    
    for gen in range(200000):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            others = list(range(pop_size))
            others.remove(i)
            pick = rng.choice(others, 3, replace=False)
            v = pop[i] + F * (pop[pick[0]] - pop[pick[1]])
            v = np.clip(v, 0.0, 1.0)
            mask = rng.random(n) < CR
            if not mask.any():
                mask[rng.integers(n)] = True
            child = np.where(mask, v, pop[i])
            child = project_feasible(child, n)
            c_child = compute_c5(child, n)
            if c_child < fitness[i]:
                pop[i] = child
                fitness[i] = c_child
                if c_child < best_c:
                    best_c = c_child
                    best_h = child.copy()
            if gen % 50 == 0:
                F = max(0.3, F * 0.999)
    
    top_indices = np.argsort(fitness)[:min(5, pop_size)]
    return best_h, best_c, [pop[ti].copy() for ti in top_indices]

def run_de_multi(h_init, n, deadline, n_configs=3, seed=42):
    best_overall_h = h_init.copy()
    best_overall_c = compute_c5(h_init, n)
    rng = np.random.default_rng(seed)
    
    configs = [
        {'pop_size': 30, 'F': 0.6, 'CR': 0.8},
        {'pop_size': 40, 'F': 0.8, 'CR': 0.7},
        {'pop_size': 50, 'F': 0.5, 'CR': 0.9},
    ]
    
    per_deadline = (deadline - time.time()) / n_configs
    for ci in range(n_configs):
        sub_deadline = time.time() + per_deadline
        cfg = configs[ci % len(configs)]
        s = seed + ci * 1000
        h_res, c_res, _ = run_de(h_init, n, sub_deadline,
                                  pop_size=cfg['pop_size'], F=cfg['F'],
                                  CR=cfg['CR'], seed=s)
        if c_res < best_overall_c:
            best_overall_c = c_res
            best_overall_h = h_res.copy()
    
    return best_overall_h, best_overall_c

def perturb_and_optimize(h, n, deadline, n_perturbs=10, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    per_time = max(10, (deadline - time.time()) / n_perturbs)
    for p in range(n_perturbs):
        if time.time() > deadline:
            break
        pert_end = time.time() + per_time * 0.7
        opt_end = time.time() + per_time * 0.95
        
        scale = 0.02 * (1 + p / n_perturbs)
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)
        
        c_pert = compute_c5(h_pert, n)
        
        if c_pert < best_c or True:
            h_opt, c_opt = run_lbfgs(h_pert, n, opt_end,
                alphas=[10000, 100000, 500000], mu=0.01,
                maxiters=[300, 400, 500])
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
    
    return best_h, best_c

def block_perturb(h, n, rng):
    h_new = h.copy()
    n_blocks = rng.integers(2, 6)
    block_size = n // n_blocks
    for block_idx in range(n_blocks):
        if rng.random() < 0.3:
            start = block_idx * block_size
            end = min(n, (block_idx + 1) * block_size)
            delta = rng.uniform(-0.1, 0.1)
            h_new[start:end] += delta
    return project_feasible(h_new, n)

def sinusoidal_perturb(h, n, rng):
    h_new = h.copy()
    freq = rng.uniform(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.01, 0.05)
    x = np.linspace(0, 2 * np.pi, n)
    h_new += amp * np.sin(freq * x + phase)
    return project_feasible(h_new, n)

def subgrad_refine(h, n, deadline, n_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.005
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.003 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.02, 0.02)
        else:
            eta *= 0.96
        h = h_new
        if eta < 1e-6:
            break
    return best_h, best_c

def pair_search_fast(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # Phase 1: Quick polish at original resolution
    if time.time() < deadline - 100:
        p1_end = t0 + 40
        h1, c1 = run_lbfgs(best_h, best_n, p1_end,
            alphas=[500, 5000, 50000, 500000],
            mu=0.01, maxiters=[800, 1200, 1500, 2000])
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Upsample to n=1000 and optimize
    n2 = 1000
    if time.time() < deadline - 500:
        h2_start = upsample(best_h, best_n, n2)
        c2_start = compute_c5(h2_start, n2)
        p2_end = min(time.time() + 120, deadline - 400)
        h2, c2 = run_lbfgs(h2_start, n2, p2_end,
            alphas=[50, 500, 5000, 50000, 500000, 2000000],
            mu=0.005, maxiters=[500, 800, 1000, 1200, 1500, 2000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n2
    
    # Phase 3: Upsample to n=2000 and optimize
    n3 = 2000
    if time.time() < deadline - 350:
        h3_start = upsample(best_h, best_n, n3)
        c3_start = compute_c5(h3_start, n3)
        p3_end = min(time.time() + 100, deadline - 250)
        h3, c3 = run_lbfgs(h3_start, n3, p3_end,
            alphas=[20, 200, 2000, 20000, 100000, 500000],
            mu=0.003, maxiters=[400, 600, 800, 1000, 1200, 1500])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n3
    
    # Phase 4: DE at moderate resolution for global exploration
    n_de = 300
    if time.time() < deadline - 300:
        h_de_init = downsample(best_h, best_n, n_de)
        p4_end = min(time.time() + 180, deadline - 200)
        h_de, c_de, top_de = run_de_multi(h_de_init, n_de, p4_end,
                                            n_configs=2, seed=seed)
        # Upsample the best DE result and try to improve
        if c_de < best_c + 0.01:
            h_de_up = upsample(h_de, n_de, best_n)
            c_de_up = compute_c5(h_de_up, best_n)
            p4b_end = min(time.time() + 40, deadline - 150)
            h_de_opt, c_de_opt = run_lbfgs(h_de_up, best_n, p4b_end,
                alphas=[10000, 100000, 1000000],
                mu=0.005, maxiters=[400, 600, 800])
            if c_de_opt < best_c:
                best_c = c_de_opt
                best_h = h_de_opt
            # Also try upsampled to higher res
            if best_n < 2000:
                h_de_up2 = upsample(h_de, n_de, 1500)
                c_de_up2 = compute_c5(h_de_up2, 1500)
                p4c_end = min(time.time() + 40, deadline - 100)
                h_de_opt2, c_de_opt2 = run_lbfgs(h_de_up2, 1500, p4c_end,
                    alphas=[5000, 50000, 500000],
                    mu=0.005, maxiters=[400, 600, 800])
                if c_de_opt2 < best_c:
                    best_c = c_de_opt2
                    best_h = h_de_opt2
                    best_n = 1500
    
    # Phase 5: Active set optimization
    if time.time() < deadline - 150:
        p5_end = min(time.time() + 80, deadline - 100)
        h5, c5 = run_active_opt(best_h, best_n, p5_end, top_k=6,
            alphas=[50000, 200000, 1000000, 5000000, 20000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Structured perturbations + local optimization
    if time.time() < deadline - 120:
        p6_end = min(time.time() + 60, deadline - 80)
        for ptype in ['block', 'sinusoidal', 'noise']:
            if time.time() > deadline - 80:
                break
            if ptype == 'block':
                h_pert = block_perturb(best_h, best_n, rng)
            elif ptype == 'sinusoidal':
                h_pert = sinusoidal_perturb(best_h, best_n, rng)
            else:
                h_pert = best_h + rng.normal(0, 0.02, best_n)
                h_pert = project_feasible(h_pert, best_n)
            sub_end = min(time.time() + (p6_end - time.time()) / 3, p6_end)
            h_p_opt, c_p_opt = run_lbfgs(h_pert, best_n, sub_end,
                alphas=[10000, 100000, 1000000],
                mu=0.01, maxiters=[300, 400, 500])
            if c_p_opt < best_c:
                best_c = c_p_opt
                best_h = h_p_opt
                best_n = best_n
    
    # Phase 7: Subgradient refinement
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 30, deadline - 30)
        h7, c7 = subgrad_refine(best_h, best_n, p7_end, n_iter=10000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Active set final polish
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 25, deadline - 15)
        h8, c8 = run_active_opt(best_h, best_n, p8_end, top_k=4,
            alphas=[100000, 500000, 5000000, 20000000])
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Pair search
    if time.time() < deadline - 15:
        p9_end = deadline - 5
        h9, c9 = pair_search_fast(best_h, best_n, p9_end, radius=25)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: Final single-constraint push
    if time.time() < deadline - 2:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(active_obj, best_h, jac=active_grad,
                           args=(best_n, active, 10000000, 0.001),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 200, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxfun': 500})
            h_final = project_feasible(res.x, best_n)
            c_final = compute_c5(h_final, best_n)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        except Exception:
            pass
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
