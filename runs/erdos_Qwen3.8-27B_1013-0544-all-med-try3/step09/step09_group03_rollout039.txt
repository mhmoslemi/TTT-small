    x=np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    target=n/2.0
    x=np.clip(x,0.0,1.0)
    s=x.sum()

        return x
    lo=x.min()-1.0; hi=x.max()+1.0
    # If target==0? not, target>0
    for _ in range(60):
       mid=(lo+hi)*0.5
       y=np.clip(x-mid,0.0,1.0)
       ys=y.sum()
       if ys > target:
           lo=mid
       else:
           hi=mid
    lam=(lo+hi)*0.5
    h=np.clip(x-lam,0.0,1.0)
    for _ in range(n+5):
       res=target-h.sum()
       if abs(res)<=1e-14: break

          idx=int(np.argmax(room))
          add=min(res, room[idx])

          if sub<=0: break

    rng=np.random.RandomState(seed)
    m = int(rng.randint(1, max_modes+1))
    amps=rng.uniform(0.1,1.0,m)
    phases=rng.uniform(-np.pi,np.pi,m)
    s=np.sum(np.abs(amps))

    for k in range(m):
        h += amps[k]*np.cos((k+1)*theta + phases[k])
    h -= (h.mean()-0.5)
    return project_feasible(h,n)

def smooth_grad(h,n,alpha,mu):
    vals,m=compute_c5_values(h,n)
    z=alpha*(vals-m)
    zmax=float(z.max())
    w=np.exp(z-zmax)
    w /= float(w.sum())
    W0 = float(w[0])
    l1 = 2*n-1
    # Allocate
    WA = w[1:n]
    WB = w[n:l1]
    P = np.zeros(n, dtype=float)
    P[1:] = WA + WB
    cross = even_kernel_convolution(h, P, n)
    left = np.zeros(n, dtype=float)
    right = np.zeros(n, dtype=float)
    if n > 1:

    grad = dx * (W0 * (1.0 - 2.0 * h) + left + right - cross)
    diff = float(h.sum()) - n / 2.0

import numpy as np
from scipy.optimize import minimize
import time


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def project_feasible(x, n):
    x = np.asarray(x, dtype=float)
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    s = float(x.sum())
    if abs(s - target) <= 1e-13:
        return x

    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        ys = float(y.sum())
        if ys > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)

    for _ in range(n + 10):
        res = target - float(h.sum())
        if abs(res) <= 1e-14:
            break
        if res > 0.0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            add = min(res, float(room[idx]))
            if add <= 0.0:
                break
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-res, float(h[idx]))
            if sub <= 0.0:
                break
            h[idx] -= sub

    return np.clip(h, 0.0, 1.0)


def compute_c5_values(h, n):
    dx = 2.0 / n
    total = float(h.sum())

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])

    N = next_pow2(2 * n)
    H = np.fft.rfft(h, N)
    S = np.fft.irfft(H * np.conj(H), N)
    S = S[:n]

    A = pref[n:n - 1:-1] - S
    B = total - pref[:-1] - S

    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = A[0]
    vals[1:n] = A[1:]
    vals[n:2 * n - 1] = B[1:]

    vals *= dx
    neg = vals < 0.0
    if np.any(neg):
        vals[neg] = 0.0

    m = float(vals.max())
    return vals, m


def c5_fft(h, n):
    if n <= 0:
        return float("inf")
    _, m = compute_c5_values(h, n)
    return float(m)


def c5_direct(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def evaluated_c5(h, n):
    if n <= 5000:
        return c5_direct(h, n)
    return c5_fft(h, n)


def even_kernel_convolution(h, weights, n):
    N = next_pow2(2 * n)
    kernel = np.zeros(N, dtype=float)
    if n > 1:
        idx = np.arange(1, n, dtype=int)
        wv = weights[1:]
        kernel[idx] = wv
        kernel[N - idx] = wv

    H = np.fft.rfft(h, N)
    K = np.fft.rfft(kernel, N)
    conv = np.fft.irfft(H * K, N)
    return conv[:n]


def smooth_obj(h, n, alpha, mu):
    vals, m = compute_c5_values(h, n)
    z = alpha * (vals - m)
    zmax = float(z.max())
    expz = np.exp(z - zmax)
    logsum = zmax + float(np.log(np.sum(expz)))
    val = m + logsum / alpha
    diff = float(h.sum()) - n / 2.0
    val += mu * diff * diff
    return float(val)


def smooth_grad(h, n, alpha, mu):
    vals, m = compute_c5_values(h, n)
    z = alpha * (vals - m)
    zmax = float(z.max())
    w = np.exp(z - zmax)
    w /= float(w.sum())

    W0 = float(w[0])
    total_terms = 2 * n - 1

    WA = w[1:n]
    WB = w[n:total_terms]

    P = np.zeros(n, dtype=float)
    P[1:] = WA + WB

    cross = even_kernel_convolution(h, P, n)

    left = np.zeros(n, dtype=float)
    right = np.zeros(n, dtype=float)
    if n > 1:
        wa_pref = np.cumsum(WA)
        left[:n - 1] = wa_pref[::-1]
        wb_pref = np.cumsum(WB)
        right[1:] = wb_pref

    dx = 2.0 / n
    grad = dx * (W0 * (1.0 - 2.0 * h) + left + right - cross)
    diff = float(h.sum()) - n / 2.0
    grad += 2.0 * mu * diff
    return grad

def symmetrize(h, n):
    h = 0.5 * (h + h[::-1])
    return project_feasible(h, n)


def run_lbfgs(h, n, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_fft(h, n)

    bounds = [(0.0, 1.0)] * n
    configs = get_configs(n)

    for alpha, mu in configs:

        maxiter = 700 if n > 3000 else 900
        try:
            res = minimize(
                smooth_obj,
                h,
                jac=smooth_grad,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxiter,
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxfun": maxiter * 20,
                    "maxls": 8,
                    "maxcor": 20,
                },
            )
            h_new = project_feasible(res.x, n)
            c_new = c5_fft(h_new, n)

            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
            elif c_new < best_c + 1e-10:
                h = h_new.copy()

            h_sym = symmetrize(h_new, n)
            c_sym = c5_fft(h_sym, n)
            if c_sym < best_c:
                best_c = c_sym
                best_h = h_sym.copy()
                h = h_sym.copy()
            elif c_sym < best_c + 1e-10 and c_sym < c_new:
                h = h_sym.copy()
        except Exception:
            pass

def cosine_single_start(n, amp, phase, offset):
    amp = float(amp)
    phase = float(phase)
    offset = float(offset)
    if amp == 0.0:
        h = np.full(n, 0.5, dtype=float)
    else:
        theta = 2.0 * np.pi * ((np.arange(n) + offset) / float(n)) + phase
        h = 0.5 + amp * np.cos(theta)

    h = h - (h.mean() - 0.5)
    return project_feasible(h, n)


def harmonic_start(n, amps, phases, offset=0.5):
    offset = float(offset)
    theta = 2.0 * np.pi * ((np.arange(n) + offset) / float(n))
    h = np.full(n, 0.5, dtype=float)
    for k in range(len(amps)):
        h += float(amps[k]) * np.cos((k + 1) * theta + float(phases[k]))
    h = h - (h.mean() - 0.5)
    return project_feasible(h, n)


def random_fourier_start(n, seed, total_amp=0.45, max_modes=5):
    rng = np.random.RandomState(seed)
    m = int(rng.randint(1, max_modes + 1))
    amps = rng.uniform(0.05, 1.0, m)
    phases = rng.uniform(-np.pi, np.pi, m)
    s = float(np.sum(np.abs(amps)))
    if s <= 0.0:
        amps = np.zeros(m, dtype=float)
        amps[0] = total_amp
    else:
        amps = amps * (total_amp / s)

    theta = 2.0 * np.pi * (np.arange(n) / float(n))
    h = np.full(n, 0.5, dtype=float)
    for k in range(m):
        h += amps[k] * np.cos((k + 1) * theta + phases[k])

    h = h - (h.mean() - 0.5)
    return project_feasible(h, n)


def interpolate_to(h_old, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

        top_reserve = 320.0 if len(ns) == 4 else 120.0
        per = max(30.0, (remain - top_reserve) / float(top_count))

        top_candidates = candidates[:top_count]
        for idx in range(top_count):
            if time.time() > deadline - 60.0:
                break
            h0, n0, _ = top_candidates[idx]
            alloc = min(per, deadline - time.time() - 40.0)
            if alloc < 20.0:
                break

            hopt, _ = run_lbfgs(h0, n0, time.time() + alloc)

            if n0 <= 5000 and time.time() < deadline - 10.0:
                cd = c5_direct(hopt, n0)
            else:
                cd = c5_fft(hopt, n0)
            if cd < best_c:
                best_h = hopt
                best_n = n0
                best_c = cd

            hs = symmetrize(hopt, n0)
            if n0 <= 5000 and time.time() < deadline - 10.0:
                cds = c5_direct(hs, n0)
            else:
                cds = c5_fft(hs, n0)
            if cds < best_c:
                best_h = hs
                best_n = n0
                best_c = cds

    if best_n < 4096 and len(ns) == 4 and time.time() < deadline - 200.0:
        hup = interpolate_to(best_h, best_n, 4096)
        hopt, _ = run_lbfgs(hup, 4096, deadline - 80.0)
        cd = c5_direct(hopt, 4096)
        if cd < best_c:
            best_h = hopt
            best_n = 4096
            best_c = cd

        hs = symmetrize(hopt, 4096)
        cds = c5_direct(hs, 4096)
        if cds < best_c:
            best_h = hs
            best_n = 4096
            best_c = cds

    if time.time() < deadline - 90.0:
        hpol, _ = run_lbfgs(best_h, best_n, deadline - 30.0)
        if best_n <= 5000 and time.time() < deadline - 10.0:
            cp = c5_direct(hpol, best_n)
        else:
            cp = c5_fft(hpol, best_n)
        if cp < best_c:
            best_h = hpol
            best_c = cp

        hs = symmetrize(hpol, best_n)
        if best_n <= 5000 and time.time() < deadline - 10.0:
            csp = c5_direct(hs, best_n)
        else:
            csp = c5_fft(hs, best_n)
        if csp < best_c:
            best_h = hs
            best_c = csp

    best_h = project_feasible(best_h, best_n)
    if best_n <= 5000 and time.time() < deadline - 1.0:
        best_c = c5_direct(best_h, best_n)
    else:
        best_c = c5_fft(best_h, best_n)

    return best_h, float(best_c), int(best_n)
