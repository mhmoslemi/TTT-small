            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_set_opt(h, n, deadline, top_k_list=None, alphas=None, rounds_per_config=3):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if top_k_list is None:
        top_k_list = [3, 5, 8, 12, 20, 30]
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    for top_k in top_k_list:
        if time.time() > deadline - 10:
            break
        for rnd in range(rounds_per_config):
            if time.time() > deadline - 8:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                break
            for alpha in alphas[:2]:
                if time.time() > deadline - 5:
                    break
                maxit = max(100, int((deadline - time.time()) * 1.5))
                try:
                    def _obj(x):
                        return obj_active(x, n, list(active), alpha, 0.001)
                    def _jac(x):
                        c = compute_c5_vec(x, n)
                        val = obj_active(x, n, list(active), alpha, 0.001)
                        g = grad_active(c, alpha, x, n, list(active))
                        g += 0.002 * (x.sum() - n / 2.0)
                        return g
                    res = minimize(_obj, best_h, jac=_jac, method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': maxit, 'ftol': 1e-20,
                                           'gtol': 1e-14, 'maxfun': maxit * 6,
                                           'maxls': 30, 'maxcor': 30})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_new
                except Exception:
                    pass
    return best_h, best_c

def projected_gradient(h, n, deadline, base_lr=0.001, decay=0.9995, n_steps=50000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    curr_h = best_h.copy()
    lr = base_lr
    dx_val = 2.0 / n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if step % 500 == 0:
            lr = base_lr * (decay ** step)
            if lr < 1e-8:
                lr = 1e-8
        c_vec = compute_c5_vec(curr_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        grad = np.zeros(n)
        count = 0
        for idx in active[:15]:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - curr_h[k:n])
                grad[k:n] += (-curr_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - curr_h[0:nv])
                grad[0:nv] += (-curr_h[kk:n])
            count += 1
        if count > 0:
            grad *= (dx_val / count)
        mean_h = curr_h.sum() / n
        if mean_h < 0.4:
            grad -= 5.0 * (curr_h.sum() / n - 0.5)
        elif mean_h > 0.6:
            grad -= 5.0 * (curr_h.sum() / n - 0.5)
        new_h = curr_h - lr * grad
        new_h = project_feasible(new_h, n)
        c_new = compute_c5(new_h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h.copy()
            curr_h = new_h
        else:
            lr *= 0.9
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=60, deltas=None, top_pairs=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            candidates = []
            for i in range(n):
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        candidates.append((i, j, d))
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        candidates.append((i, j, -d))
            rng = np.random.RandomState(int(time.time() * 1000) % 100000)
            if len(candidates) > top_pairs * 20:
                idx = rng.choice(len(candidates), size=top_pairs * 20, replace=False)
                candidates = [candidates[i] for i in idx]
            for (i, j, dd) in candidates:
                if time.time() > deadline:
                    break
                best_h[i] += dd
                best_h[j] -= dd
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-16:
                    best_c = c
                    improved = True
                else:
                    best_h[i] -= dd
                    best_h[j] += dd
                    best_h = project_feasible(best_h, n)
    return best_h, best_c

def simulated_annealing(h, n, deadline, n_iterations=50000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    curr_h = best_h.copy()
    curr_c = best_c
    T_start = 0.01
    T_end = 0.0001
    rng = np.random.RandomState(12345)
    n_iter = min(n_iterations, int((deadline - time.time()) * 80))
    for it in range(n_iter):
        frac = it / max(1, n_iter)
        T = T_start * (T_end / T_start) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        amt = min(0.02, min(curr_h[i], 1.0 - curr_h[j])) * rng.uniform(0.1, 1.0)
        if amt < 1e-8:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        if direction > 0:
            curr_h[i] -= amt
            curr_h[j] += amt
        else:
            curr_h[i] += amt
            curr_h[j] -= amt
        curr_h = project_feasible(curr_h, n)
        new_c = compute_c5(curr_h, n)
        delta = new_c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = curr_h.copy()
        else:
            if direction > 0:
                curr_h[i] += amt
                curr_h[j] -= amt
            else:
                curr_h[i] -= amt
                curr_h[j] += amt
            curr_h = project_feasible(curr_h, n)
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 2 * n + 10):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0, 1)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [('beatty_golden', make_beatty_start(n_orig, golden / 2.0)),
              ('beatty_sqrt2', make_beatty_start(n_orig, np.sqrt(2.0) / 2.0)),
              ('rand1', make_random_binary(n_orig, rng)),
              ('rand2', make_random_binary(n_orig, rng))]
    for name, h_alt in starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS full smooth max (increasing alpha)
    p1_end = t0 + 180
    h1, c1 = lbfgs_phase(best_h, best_n, p1_end,
        alphas=[20, 100, 500, 2000, 10000, 50000, 200000],
        mu=0.01, maxit_list=[800, 1000, 1200, 1500, 2000, 2500, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set optimization
    if time.time() < deadline - 300:
        p2_end = min(time.time() + 150, deadline - 250)
        h2, c2 = active_set_opt(best_h, best_n, p2_end,
            top_k_list=[3, 5, 8, 12, 20],
            alphas=[100000, 500000, 2000000],
            rounds_per_config=3)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Projected gradient with adaptive step
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 120, deadline - 200)
        h3, c3 = projected_gradient(best_h, best_n, p3_end,
            base_lr=0.002, decay=0.9997, n_steps=40000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: More active set rounds
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 100, deadline - 160)
        h4, c4 = active_set_opt(best_h, best_n, p4_end,
            top_k_list=[3, 5, 10],
            alphas=[1000000, 5000000, 20000000],
            rounds_per_config=4)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Simulated annealing escape
    if time.time() < deadline - 160:
        p5_end = min(time.time() + 80, deadline - 130)
        h5, c5 = simulated_annealing(best_h, best_n, p5_end, n_iterations=30000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        # Polish after SA
        if c5 < best_c + 0.005 and time.time() < deadline - 100:
            p5b_end = min(time.time() + 40, deadline - 80)
            h5b, c5b = lbfgs_phase(best_h, best_n, p5b_end,
                alphas=[20000, 100000, 500000], mu=0.01, maxit_list=[500, 800, 1000])
            if c5b < best_c:
                best_c = c5b
                best_h = h5b

    # Phase 6: Fine pair search
    if time.time() < deadline - 100:
        p6_end = min(time.time() + 60, deadline - 70)
        h6, c6 = fine_pair_search(best_h, best_n, p6_end, radius=50,
            deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
            top_pairs=80)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Very fine adjustments
    if time.time() < deadline - 50:
        p7_end = min(time.time() + 35, deadline - 35)
        h7, c7 = fine_pair_search(best_h, best_n, p7_end, radius=30,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001],
            top_pairs=100)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final projected gradient micro-tuning
    if time.time() < deadline - 15:
        p8_end = deadline - 5
        h8, c8 = projected_gradient(best_h, best_n, p8_end,
            base_lr=0.0005, decay=0.9999, n_steps=20000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final very fine pair search with tiny steps
    if time.time() < deadline - 3:
        h9, c9 = fine_pair_search(best_h, best_n, deadline - 1, radius=20,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005],
            top_pairs=120)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
