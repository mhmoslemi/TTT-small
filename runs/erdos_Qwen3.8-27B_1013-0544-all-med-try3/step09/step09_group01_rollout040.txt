def c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(np.ascontiguousarray(h), np.ascontiguousarray(1.0 - h), mode='full') * dx


def c5_val(h):
    return float(c5_vec(h).max())


def proj(h):
    h = np.ascontiguousarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    y = np.clip(h, 0.0, 1.0)
    for _ in range(6000):
        diff = target - y.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - y
            s = room.sum()
            if s < 1e-15:
                break
            y = y + (diff / s) * room
        else:
            s = y.sum()
            if s < 1e-15:
                break
            y = y + (diff / s) * y
        y = np.clip(y, 0.0, 1.0)
    return y


def upsample(h0, n1):
    n0 = len(h0)
    if n1 == n0:
        return proj(h0.copy())
    ratio = n1 / n0
    if abs(ratio - round(ratio)) < 1e-6:
        r = int(round(ratio))
        out = np.repeat(h0, r)
        return proj(out)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    out = np.interp(x1, x0, h0)
    return proj(out)

        if nv > 0:
            out[0:nv] += w * (-x[t:n])
            out[t:n] += w * (1.0 - x[0:nv])
    else:
        u = -t
        nv = n - u
        if nv > 0:
            out[u:n] += w * (-x[0:nv])
            out[0:nv] += w * (1.0 - x[u:n])


def obj_grad(x, n, alpha, which):
    c = c5_vec(x)
    if which is None:
        idx = np.arange(c.size)
    else:
        idx = np.ascontiguousarray(which, dtype=int)
    cs = c[idx]
    m = float(cs.max())
    s = alpha * (cs - m)
    A = float(s.max())
    e = np.exp(s - A)
    se = float(e.sum())
    val = m + (A + np.log(se)) / alpha
    out = np.zeros(n)
    nb = n - 1
    inv = 1.0 / se
    for j in range(idx.size):
        wj = e[j] * inv
        if wj < 1e-12:
            continue
        t = int(idx[j]) - nb
        add_weighted(out, x, t, wj)
    return val, out


def pen_fun(x, n, alpha, which, mu):
    v, g = obj_grad(x, n, alpha, which)
    d = x.sum() - 0.5 * n
    return v + mu * d * d


def pen_jac(x, n, alpha, which, mu):
    v, g = obj_grad(x, n, alpha, which)
    d = x.sum() - 0.5 * n
    g = SIGN * g + (2.0 * mu * d) * np.ones_like(g)
    return g


def run_lbfgs(x0, n, alpha, which, deadline):
    global CUR_END
    if time.time() > deadline - 4:
        return proj(x0.copy())
    CUR_END = deadline - 3
    bounds = [(0.0, 1.0)] * n
    maxit = 9000
    res = None
    try:
        res = minimize(pen_fun, x0, jac=pen_jac, args=(n, alpha, which, 5e-3),
                       method='L-BFGS-B', bounds=bounds,
                       options=dict(maxiter=maxit, maxfun=3 * maxit,
                                    ftol=1e-20, gtol=1e-16, maxls=40, maxcor=30),
                       callback=stop_cb)
    except TimeoutHit:
        pass
    except Exception:
        return proj(x0.copy())
    try:
        if res is not None and hasattr(res, 'x'):
            return proj(res.x)
    except Exception:
        pass
    return proj(x0.copy())


def allshifts(x0, n, deadline, alist):
    best = proj(x0.copy())
    bc = c5_val(best)
    x = best
    for a in alist:
        if time.time() > deadline - 8:
            break
        h2 = run_lbfgs(x, n, a, None, deadline)
        c2 = c5_val(h2)
        if c2 < bc:
            best = h2
            bc = c2
        x = h2
    return best, bc


def chase(x0, n, deadline, alist, klist):
    best = proj(x0.copy())
    bc = c5_val(best)
    x = best
    for i, a in enumerate(alist):
        if time.time() > deadline - 8:
            break
        cvec = c5_vec(x)
        K = klist[min(i, len(klist) - 1)]
        top = np.argsort(-cvec)[:K]
        which = np.sort(top)
        h2 = run_lbfgs(x, n, a, which, deadline)
        c2 = c5_val(h2)
        if c2 < bc:
            best = h2
            bc = c2
        x = h2
    return best, bc


def swap_polish(h, n, deadline):
    h = proj(h.copy())
    h = np.clip(h, 0.0, 1.0)
    m = c5_val(h)
    best_h = h.copy()
    best_m = m
    if n < 12 or time.time() > deadline - 4:
        return best_h, best_m
    cvec = c5_vec(h)
    rng = np.random.RandomState(99)
    it = 0
    while time.time() < deadline - 3 and it < 400000:
        it += 1
        sstar = int(np.argmax(cvec))
        tstar = sstar - (n - 1)
        cands = []
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if tstar >= 0:
            nv = n - tstar
            if nv > 0:
                src = np.where((h[0:nv] < 0.5) & (h[tstar:n] > 0.5))[0]
                tgt = src + tstar
                if len(src) > 0:
                    for _ in range(6):
                        q = int(src[rng.randint(len(src))])
                        p = int(tgt[rng.randint(len(tgt))])
                        cands.append((q, p))
        else:
            u = -tstar
            nv = n - u
            if nv > 0:
                src = np.where((h[u:n] < 0.5) & (h[0:nv] > 0.5))[0] + u
                tgt = src - u
                if len(src) > 0:
                    for _ in range(6):
                        q = int(src[rng.randint(len(src))])
                        p = int(tgt[rng.randint(len(tgt))])
                        cands.append((q, p))
        for _ in range(4):
            if len(ones) and len(zeros):
                q = int(zeros[rng.randint(len(zeros))])
                p = int(ones[rng.randint(len(ones))])
                cands.append((q, p))
        bestcand = None
        bestval = m
        for (q, p) in cands:
            if p == q:
                continue
            if not (h[p] > 0.5 and h[q] < 0.5):
                continue
            op = h[p]
            oq = h[q]
            h[p] = 0.0
            h[q] = 1.0
            cm = c5_val(h)
            h[p] = op
            h[q] = oq
            if cm < bestval - 1e-12:
                bestval = cm
                bestcand = (p, q)
        if bestcand is not None:
            p, q = bestcand
            h[p] = 0.0
            h[q] = 1.0
            m = bestval
            cvec = c5_vec(h)
            if m < best_m:
                best_m = m
                best_h = h.copy()
        else:
            if len(ones) and len(zeros) and rng.rand() < 0.15:
                q = int(zeros[rng.randint(len(zeros))])
                p = int(ones[rng.randint(len(ones))])
                if p != q:
                    h[p] = 0.0
                    h[q] = 1.0
                    m = c5_val(h)
                    cvec = c5_vec(h)
                    if m < best_m:
                        best_m = m
                        best_h = h.copy()
    return best_h, best_m


def process_res(h0, n, t_end):
    x = proj(h0.copy())
    x = np.clip(x, 0.0, 1.0)
    c = c5_val(x)
    now = time.time()
    seg = max(0.1, t_end - now)
    c1 = now + 0.30 * seg
    c2 = now + 0.74 * seg
    if n <= 600:
        all_a = [25.0, 100.0, 400.0, 1500.0, 6000.0]
    else:
        all_a = [18.0, 70.0, 260.0, 1000.0, 4500.0]
    chase_a = [800.0, 3500.0, 14000.0, 55000.0]
    chase_k = [5, 9, 16]
    if time.time() < c1 - 3:
        try:
            b, c2n = allshifts(x, n, c1, all_a)
            if c2n < c:
                x = b
                c = c2n
        except Exception:
            pass
    if time.time() < c2 - 3:
        try:
            b, c2n = chase(x, n, c2, chase_a, chase_k)
            if c2n < c:
                x = b
                c = c2n
        except Exception:
            pass
    if time.time() < t_end - 3:
        try:
            b, c2n = swap_polish(x, n, t_end)
            if c2n < c:
                x = b
                c = c2n
        except Exception:
            pass
    return x, c


def calib_sign():
    n = 60
    rs = np.random.RandomState(7)
    x = proj(rs.rand(n))
    _, g = obj_grad(x, n, 300.0, None)
    idx = rs.choice(n, 8, replace=False)
    fd = np.zeros(len(idx))
    eps = 1e-5
    for jj in range(len(idx)):
        j = int(idx[jj])
        xp = x.copy()
        xm = x.copy()
        xp[j] = min(1.0, x[j] + eps)
        xm[j] = max(0.0, x[j] - eps)
        vp, _ = obj_grad(xp, n, 300.0, None)
        vm2, _ = obj_grad(xm, n, 300.0, None)
        den = xp[j] - xm[j]
        fd[jj] = (vp - vm2) / den if abs(den) > 1e-30 else 0.0
    num = float(g[idx].dot(fd))
    denc = float(np.linalg.norm(g[idx]) * np.linalg.norm(fd))
    corr = num / (denc + 1e-300)
    return 1.0 if corr > 0.0 else -1.0


def run(seed=42, budget_s=1100, **kwargs):
    global CUR_END, SIGN, SEED
    t0 = time.time()
    SEED = int(seed)
    deadline = t0 + float(budget_s) - 10.0
    try:
        SIGN = calib_sign()
    except Exception:
        SIGN = 1.0

    try:
        h0 = np.ascontiguousarray(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        rng = np.random.RandomState(SEED)
        h0 = np.clip(0.5 + 0.2 * rng.randn(500), 0.0, 1.0)
        n0 = 500

    h0 = proj(h0)
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_c = c5_val(h0)
    best_n = n0

    span = max(60.0, deadline - t0)
    capA = t0 + 0.25 * span
    capB = t0 + 0.74 * span
    capC = t0 + 0.87 * span

    # Stage A: parent resolution with corrected smoothed-max L-BFGS + chase + polish
    try:
        hA, cA = process_res(h0, n0, capA)
        if cA < best_c:
            best_h, best_c, best_n = hA, cA, n0
    except Exception:
        pass

    nB = int(min(2 * n0, 2500))
    # Stage B: doubled resolution
    try:
        if time.time() < capB - 20 and nB > n0:
            hB0 = upsample(h0, nB)
            hB, cB = process_res(hB0, nB, capB)
            if cB < best_c:
                best_h, best_c, best_n = hB, cB, nB
    except Exception:
        pass

    # Diverse basin from alternating template (cheap hot-lag polish only)
    try:
        if time.time() < capC - 20:
            nD = int(min(2 * n0, 2500))
            hD0 = alt_len(nD)
            hD, cD = swap_polish(hD0, nD, capC)
            if cD < best_c:
                best_h, best_c, best_n = hD, cD, nD
    except Exception:
        pass

    # Final polish on the best
    try:
        if time.time() < deadline - 25:
            hP, cP = swap_polish(best_h, best_n, deadline - 2)
            if cP < best_c:
                best_h, best_c = hP, cP
    except Exception:
        pass

    # Verification projection
    try:
        best_h = proj(best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c = c5_val(best_h)
        best_n = len(best_h)
    except Exception:
        best_h = np.clip(proj(h0), 0.0, 1.0)
        best_c = c5_val(best_h)
        best_n = len(best_h)

    return best_h, float(best_c), int(best_n)
