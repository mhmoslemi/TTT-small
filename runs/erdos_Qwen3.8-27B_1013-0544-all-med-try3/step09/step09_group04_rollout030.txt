    target = n / 2.0
    # Bisection on lambda
    lo, hi = h.min() - target, h.max() - target + target  # rough bounds

    # lam_max: all h[i] - lam <= 0 => lam >= max(h), sum = 0 < target
    lo = h.min() - 1.0 - 1.0  # sum will be n > target

        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-16:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def single_obj(x, n, idx, mu):
    c = compute_c5_vec(x, n)
    val = c[idx]
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def single_grad(x, n, idx, mu):
    c = compute_c5_vec(x, n)
    k = idx - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - x[k:n]) * dx
        grad[k:n] += (-x[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - x[0:nv]) * dx
        grad[0:nv] += (-x[kk:n]) * dx
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, obj, jac, obj_args, maxit, mu_name=None):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, args=obj_args,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-30,
                                'gtol': 1e-20, 'maxfun': maxit * 8,
                                'maxls': 30, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def cycle_constraints(h, n, deadline, max_rounds=200, maxit_per=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_full = 2 * n - 1
    bounds = [(0.0, 1.0)] * n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-7, 0.005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for idx in active:
            if time.time() > deadline:
                break
            rem = deadline - time.time()
            if rem < 3:
                break
            mi = max(50, int(rem * 0.3))
            try:
                res = minimize(single_obj, best_h, jac=single_grad,
                               args=(n, int(idx), 0.005),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': mi, 'ftol': 1e-30,
                                        'gtol': 1e-20, 'maxfun': mi * 6,
                                        'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def active_set_refine(h, n, deadline, top_k=8, alphas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000, 100000000]
    for rnd in range(100):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem = deadline - time.time()
            if rem < 5:
                break
            mi = max(100, int(rem * 0.8))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, alpha, 0.001),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': mi, 'ftol': 1e-30,
                                        'gtol': 1e-20, 'maxfun': mi * 8,
                                        'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def perturb_restart(h, n, deadline, rng, n_restarts=5, perturb_scale=0.02):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        if rem < 30:
            break
        h_pert = h + rng.normal(0, perturb_scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert > best_c + 0.05:
            perturb_scale *= 0.5
            continue
        p_end = time.time() + min(rem * 0.7, 60)
        h_opt, c_opt = active_set_refine(h_pert, n, p_end, top_k=5,
                                          alphas=[100000, 1000000, 10000000])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            perturb_scale = max(0.005, perturb_scale * 0.7)
        else:
            perturb_scale *= 0.7
    return best_h, best_c

def pair_local_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def subgrad_step(h, n, deadline, n_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.003 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.02, 0.01)
        else:
            eta *= 0.98
        h = h_new
    return best_h, best_c

def multi_alpha_sweep(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        mi = max(200, int(rem * 0.4))
        try:
            res = minimize(csm_obj, best_h, jac=csm_grad,
                           args=(n, alpha, 0.003),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': mi, 'ftol': 1e-30,
                                    'gtol': 1e-20, 'maxfun': mi * 8,
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        n_orig = 500
        h0 = np.full(n_orig, 0.5)

    n = n_orig
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Quick smooth-max L-BFGS polish
    p1_end = t0 + 40
    bounds = [(0.0, 1.0)] * n
    for alpha in [1000, 10000, 100000]:
        if time.time() > p1_end:
            break
        mi = max(500, int((p1_end - time.time()) * 2))
        try:
            res = minimize(csm_obj, best_h, jac=csm_grad,
                           args=(n, alpha, 0.01),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': mi, 'ftol': 1e-25,
                                    'gtol': 1e-18, 'maxfun': mi * 8,
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Multi-alpha sweep
    p2_end = min(t0 + 180, deadline - 400)
    if time.time() < p2_end:
        h2, c2 = multi_alpha_sweep(best_h, n, p2_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Active set refinement
    p3_end = min(t0 + 350, deadline - 250)
    if time.time() < p3_end:
        h3, c3 = active_set_refine(best_h, n, p3_end, top_k=8,
                                   alphas=[50000, 200000, 1000000, 5000000, 20000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Individual constraint cycling
    p4_end = min(t0 + 550, deadline - 200)
    if time.time() < p4_end:
        h4, c4 = cycle_constraints(best_h, n, p4_end, max_rounds=100, maxit_per=150)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Multi-start perturbations
    p5_end = min(t0 + 750, deadline - 150)
    if time.time() < p5_end:
        h5, c5 = perturb_restart(best_h, n, p5_end, rng, n_restarts=8, perturb_scale=0.03)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Active set refinement again
    p6_end = min(t0 + 900, deadline - 100)
    if time.time() < p6_end:
        h6, c6 = active_set_refine(best_h, n, p6_end, top_k=6,
                                   alphas=[100000, 1000000, 10000000, 50000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Individual constraint cycling again
    p7_end = min(t0 + 1020, deadline - 50)
    if time.time() < p7_end:
        h7, c7 = cycle_constraints(best_h, n, p7_end, max_rounds=150, maxit_per=100)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Subgradient refinement
    p8_end = deadline - 15
    if time.time() < p8_end:
        h8, c8 = subgrad_step(best_h, n, p8_end, n_iter=50000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final active set push
    if time.time() < deadline - 2:
        c_vec = compute_c5_vec(best_h, n)
        active = np.array([np.argmax(c_vec)])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(n, active, 100000000, 0.001),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-30,
                                    'gtol': 1e-20, 'maxfun': 2000,
                                    'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
