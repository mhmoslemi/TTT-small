    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    # Phase 1: Quick L-BFGS-B polish at parent resolution
    n = len(best_h)
    for alpha in [100000, 1000000, 10000000]:
        if time.time() > deadline - 400:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 30, maxiter=500)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Expand to n=750
    if time.time() < deadline - 300:
        n750 = 750
        h750 = interpolate_to(best_h, n, n750)
        c750 = compute_c5(h750, n750)
        for alpha in [100000, 1000000, 10000000, 50000000]:
            if time.time() > deadline - 250:
                break
            h_opt = lbfgsb_optimize(h750, n750, alpha, time.time() + 25, maxiter=400)
            c_opt = compute_c5(h_opt, n750)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n750_final = h_opt
            h750 = h_opt
        # Targeted local search at 750
        if time.time() < deadline - 200:
            ls_deadline = min(deadline - 180, time.time() + 80)
            h_ls, c_ls = targeted_local_search(h750, n750, ls_deadline, rng, k_top=5, max_rounds=30000)
            if c_ls < best_c:
                best_c = c_ls
                best_h = h_ls

    # Phase 3: Expand to n=1000
    if time.time() < deadline - 250:
        n1000 = 1000
        h1000 = interpolate_to(best_h, len(best_h), n1000)
        c1000 = compute_c5(h1000, n1000)
        for alpha in [500000, 5000000, 50000000]:
            if time.time() > deadline - 200:
                break
            h_opt = lbfgsb_optimize(h1000, n1000, alpha, time.time() + 20, maxiter=300)
            c_opt = compute_c5(h_opt, n1000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            h1000 = h_opt
        # Local search at 1000
        if time.time() < deadline - 150:
            ls_deadline = min(deadline - 130, time.time() + 60)
            h_ls, c_ls = targeted_local_search(h1000, n1000, ls_deadline, rng, k_top=5, max_rounds=20000)
            if c_ls < best_c:
                best_c = c_ls
                best_h = h_ls

    # Phase 4: Intensive local search at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 100:
        ls_deadline = min(deadline - 80, time.time() + 100)
        h_ls, c_ls = targeted_local_search(best_h, n_best, ls_deadline, rng, k_top=7, max_rounds=50000)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Phase 5: Subgradient balance refinement
    if time.time() < deadline - 80:
        sub_deadline = min(deadline - 60, time.time() + 50)
        h_sub, c_sub = subgradient_balance_refine(best_h, n_best, sub_deadline, num_iters=3000)
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub

    # Phase 6: Multi-perturbation refinement
    if time.time() < deadline - 80:
        mp_deadline = min(deadline - 50, time.time() + 60)
        h_mp, c_mp = multi_perturb_refine(best_h, n_best, mp_deadline, rng,
                                          num_perturbs=8, alpha=10000000)
        if c_mp < best_c:
            best_c = c_mp
            best_h = h_mp

    # Phase 7: Gradient descent polish
    if time.time() < deadline - 40:
        gd_deadline = min(deadline - 20, time.time() + 30)
        h_gd, c_gd = gradient_descent_polish(best_h, n_best, gd_deadline, alpha=50000000, max_iter=10000)
        if c_gd < best_c:
            best_c = c_gd
            best_h = h_gd

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 10:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 100000000, deadline - 3, maxiter=2000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final micro local search
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_ls, c_ls = targeted_local_search(best_h, n_final, deadline - 1, rng, k_top=5, max_rounds=20000)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
