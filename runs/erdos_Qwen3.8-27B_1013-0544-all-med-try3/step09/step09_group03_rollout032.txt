import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def min_1d_convex(A, B, d, mu, n_iter=60):
    """Minimize f(t) = max(A + B*t) + mu*(d+t)^2 over t in [0,1].
    A, B are arrays of length m. d is scalar.
    """
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        f1 = np.max(A + B * t1) + mu * (d + t1) ** 2
        f2 = np.max(A + B * t2) + mu * (d + t2) ** 2
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0

def coordinate_descent(h, n, deadline, mu=50.0, seed=42):
    """Coordinate descent with exact 1D minimization."""
    dx = 2.0 / n
    h = h.copy()
    h = project_feasible(h, n)
    S_target = n / 2.0
    S = h.sum()
    c_vec = compute_c5_vec(h, n)
    best_c = c_vec.max()
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    n_suite = 2 * n - 1
    
    pass_num = 0
    while time.time() < deadline - 10:
        pass_num += 1
        # Random order
        order = rng.permutation(n)
        
        # Precompute ks for efficiency
        # (we'll compute B on the fly for each i)
        
        for i in order:
            if time.time() > deadline - 10:
                break
            old = h[i]
            
            # Compute B[j] for all j = 0, ..., 2n-2
            # k = j - (n-1)
            # B[j] = dx * [(1-h[i+k]) * I(0<=i+k<n) + (-h[i-k]) * I(0<=i-k<n)]
            
            # Vectorized computation
            ks = np.arange(-(n-1), n)  # length 2n-1
            idx1 = i + ks  # for first factor
            idx2 = i - ks  # for second factor
            
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            
            B = np.zeros(n_suite)
            B[v1] = (1.0 - h[idx1[v1]]) * dx
            B[v2] += (-h[idx2[v2]]) * dx
            
            # A = c_vec - B * old
            A = c_vec - B * old
            
            # d = S - S_target - old (the constant in mu*(d+t)^2)
            # f(t) = max(A + B*t) + mu*(S - S_target - old + t)^2
            d = S - S_target - old
            
            t_opt = min_1d_convex(A, B, d, mu)
            
            delta = t_opt - old
            if abs(delta) > 1e-14:
                c_vec += delta * B
                h[i] = t_opt
                S += delta

        h = project_feasible(h, n)
        S = h.sum()
        c_vec = compute_c5_vec(h, n)
        c = c_vec.max()
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def min_1d_convex(A, B, d, mu, n_iter=50):
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        f1 = np.max(A + B * t1) + mu * (d + t1) ** 2
        f2 = np.max(A + B * t2) + mu * (d + t2) ** 2
        if f1 <= f2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) * 0.5

def coordinate_descent(h, n, deadline, mu=50.0, seed=42, verbose=False):
    h = project_feasible(h, n).copy()
    n_suite = 2 * n - 1
    c_vec = compute_c5_vec(h, n)
    best_c = c_vec.max()
    best_h = h.copy()
    S_target = n / 2.0
    S = h.sum()
    rng = np.random.RandomState(seed)
    
    # Precompute ks
    ks = np.arange(-(n-1), n)
    
    while time.time() < deadline - 5:
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline - 5:
                return best_h, best_c
            old = h[i]
            
            idx1 = i + ks
            idx2 = i - ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            
            B = np.zeros(n_suite)
            B[v1] = (1.0 - h[idx1[v1]]) * (2.0/n)
            B[v2] += (-h[idx2[v2]]) * (2.0/n)
            
            A = c_vec - B * old
            d = S - S_target - old
            
            t_opt = min_1d_convex(A, B, d, mu)
            delta = t_opt - old
            
            if abs(delta) > 1e-12:
                c_vec += delta * B
                h[i] = t_opt
                S += delta
        
        h = project_feasible(h, n)
        S = h.sum()
        c_vec = compute_c5_vec(h, n)
        c = c_vec.max()
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 100.0), (2000, 500.0), (5000, 2000.0), (10000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_optimize(h, n, deadline, seed=123):
    """Try random perturbations and optimize each."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    
    for trial in range(20):
        if time.time() > deadline - 30:
            break
        mag = 0.03 * (1.0 + rng.rand())
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        
        # Quick coordinate descent
        h_new, c_new = coordinate_descent(h_p, n, min(time.time() + 30, deadline - 10), mu=50.0, seed=trial)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    return best_h, best_c

Now, if I change h[a] from old to new (delta = new - old), how does out_k change?

h[a] appears in out_k in two ways:
1. As h[n+k] where n+k = a, i.e., n = a-k. Term: h[a] * (1-h[a-k]). Change: delta * (1-h[a-k]) * dx. Valid when 0≤a-k<n (and 0≤a<n which is true).
2. As (1-h[n]) where n = a. Term: h[a+k] * (1-h[a]). Change: -delta * h[a+k] * dx. Valid when 0≤a+k<n.

So: Δout_k = dx * [(1-h[a-k])*I(0≤a-k<n) + (-h[a+k])*I(0≤a+k<n)]

Wait, this is slightly different from what I had before! Before I had:
Δ_k = dx * [(1-h[i+k])*I(0≤i+k<n) + (-h[i-k])*I(0≤i-k<n)]

But now I get:
Δ_k = dx * [(1-h[i-k])*I(0≤i-k<n) + (-h[i+k])*I(0≤i+k<n)]

The signs and indices are swapped! Let me re-derive carefully.

out_k = Σ_n h[n+k] * (1-h[n]) * dx

Change h[a] from old to old+delta:

Case 1: h[a] appears as h[n+k] (first factor). This happens when n+k = a, i.e., n = a-k.
The term is h[a] * (1-h[a-k]) * dx.
Change: delta * (1-h[a-k]) * dx.
Valid when 0 ≤ a-k < n (so that n = a-k is a valid index for the sum).

Case 2: h[a] appears in (1-h[n]) (second factor). This happens when n = a.
The term is h[a+k] * (1-h[a]) * dx.
Change: -delta * h[a+k] * dx.
Valid when 0 ≤ a+k < n (so that a+k is a valid index for h in the first factor).

So: B_k = dx * [(1-h[a-k])*I(0≤a-k<n) + (-h[a+k])*I(0≤a+k<n)]

These are different! The correct one is:
