CONV_DIRECT_THRESHOLD = 1200


def convolve_fast(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    if a.size == 0 or b.size == 0:
        return np.empty(0, dtype=float)

    if fftconvolve is not None and max(a.size, b.size) > CONV_DIRECT_THRESHOLD:
        try:
            out = fftconvolve(a, b, mode="full")
            if np.iscomplexobj(out):
                out = out.real
            return np.asarray(out, dtype=float)
        except Exception:
            pass

    return np.convolve(a, b, mode="full").astype(float, copy=False)


def compute_lags_fast(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    p = n - 1

    q = convolve_fast(h, h[::-1])
    need = 2 * n - 1
    if q.size < need:
        tmp = np.zeros(need, dtype=float)
        tmp[:q.size] = q
        q = tmp

    sumh = float(np.sum(h))
    c = np.empty(need, dtype=float)
    c[p] = sumh - float(q[p])

    if p > 0:
        prefix = np.empty(n + 1, dtype=float)
        prefix[0] = 0.0
        np.cumsum(h, out=prefix[1:])

        idx = np.arange(1, n, dtype=int)
        head = prefix[n - idx]
        tail = sumh - prefix[idx]
        qm = q[p + idx]

        c[p - idx] = head - qm
        c[p + idx] = tail - qm

    return c


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def project_simplex_box(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)

    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0

    for _ in range(70):
        tau = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - tau, 0.0, 1.0)))
        if s > target:
            lo = tau
        else:
            hi = tau

    tau = 0.5 * (lo + hi)
    z = np.clip(h - tau, 0.0, 1.0)

    for _ in range(6):
        diff = target - float(np.sum(z))
        if abs(diff) <= 1e-14 * max(1.0, target):
            break
        free = (z > 1e-12) & (z < 1.0 - 1e-12)
        if free.any():
            z[free] += diff / float(np.count_nonzero(free))
        else:
            if diff > 0.0:
                room = 1.0 - z
                tot = float(np.sum(room))
                if tot <= 1e-15:
                    break
                z += (diff / tot) * room
            else:
                tot = float(np.sum(z))
                if tot <= 1e-15:
                    break
                z += (diff / tot) * z
        z = np.clip(z, 0.0, 1.0)

    diff = target - float(np.sum(z))
    if abs(diff) > 1e-15:
        free = (z > 1e-10) & (z < 1.0 - 1e-10)

            rem = float(diff)
            for pos in idx:
                if abs(rem) <= 1e-15:
                    break
                if rem > 0.0:
                    add = min(rem, 1.0 - z[pos])
                else:
                    add = max(rem, -z[pos])
                z[pos] += add
                rem -= add

    return z


def resize_h(h_src, n_src, n_dst):

def smooth_value_weights(c, tau):
    c = np.asarray(c, dtype=float)
    if tau <= 1e-10:
        tau = 1e-10

    vmax = float(np.max(c))
    shifted = (c - vmax) / tau
    shifted = np.minimum(shifted, 0.0)
    w = np.exp(shifted)
    s = float(np.sum(w))

    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0

    val = vmax + tau * float(np.log(s))
    w = w / s
    return float(val), w


def smooth_max_grad(h, n, w):
    h = np.asarray(h, dtype=float)
    n = int(n)
    dx = 2.0 / n
    p = n - 1

    if p <= 0:
        return (1.0 - 2.0 * h) * dx

    A = w[p + 1: 2 * p + 1].astype(float, copy=False)
    B = w[0: p][::-1].astype(float, copy=False)
    C = A + B

    prefA = np.concatenate(([0.0], np.cumsum(A)))
    prefB = np.concatenate(([0.0], np.cumsum(B)))
    const = prefA + prefB[::-1]

    left = np.zeros(n, dtype=float)
    conv_l = convolve_fast(C, h)
    left[1:] = conv_l[:p]

    right = np.zeros(n, dtype=float)
    conv_r = convolve_fast(C[::-1], h)
    right[0:p] = conv_r[p:2 * p]

    grad = w[p] * (1.0 - 2.0 * h) + const - (left + right)
    return grad * dx


def lbfgsb_obj_grad(x, n, tau, penalty_coeff, deadline):
    global _LAST_X, _LAST_VAL

    if deadline is not None and time.time() > deadline:
        raise TimeoutError

    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)

    dx = 2.0 / float(n)
    c = compute_lags_fast(x, int(n))
    val_counts, w = smooth_value_weights(c, tau)
    val = val_counts * dx

    target = 0.5 * float(n)
    diff = float(np.sum(x)) - target
    val += float(penalty_coeff) * diff * diff

    grad = smooth_max_grad(x, int(n), w)
    grad += (2.0 * float(penalty_coeff) * diff) * np.ones(int(n), dtype=float)

    _LAST_X = x.copy()
    _LAST_VAL = float(val)
    return val, grad


def lbfgsb_optimize(h, n, tau, deadline, penalty_coeff=1.0e5):
    global _LAST_X, _LAST_VAL

    n = int(n)
    h = project_simplex_box(h, n).copy()
    _LAST_X = h.copy()
    _LAST_VAL = float("inf")

    if minimize is None or time.time() >= deadline - 0.5:
        return h

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))

    try:
        res = minimize(
            lbfgsb_obj_grad,
            h,
            method="L-BFGS-B",
            jac=True,
            args=(n, float(tau), float(penalty_coeff), float(deadline)),
            options={
                "maxiter": 100000,
                "maxls": 30,
                "maxcor": 20,
                "ftol": 1.0e-15,
                "gtol": 1.0e-12,
                "iprint": -1,

    except TimeoutError:
        xopt = _LAST_X
    except Exception:
        if _LAST_X is not None:
            xopt = _LAST_X
        else:
            xopt = h

    if xopt is None:
        xopt = h

    return project_simplex_box(xopt, n)


def make_tau_list(n):
    n = float(max(1, int(n)))
    raw = [
        0.02 * n,
        0.005 * n,
        0.001 * n,
        0.00025 * n,
        max(0.02, 0.00005 * n),

    out = []
    prev = 1.0e100
    for t in raw:
        t = float(max(1.0e-4, t))
        if t < 0.5 * prev:
            out.append(t)
            prev = t

    while len(out) < 3:
        npass = float(max(1.0e-4, prev * 0.3))
        out.append(npass)
        prev = npass

    return out


def optimize_resolution(h, n, phase_deadline):
    n = int(n)
    h = project_simplex_box(h, n).copy()
    dx = 2.0 / n

    best_h = h.copy()
    c = compute_lags_fast(best_h, n)
    best_c = float(np.max(c) * dx)
    cur_h = best_h.copy()
    cur_c = best_c

    taus = make_tau_list(n)
    for idx, tau in enumerate(taus):
        rem = phase_deadline - time.time()
        if rem < 5.0:
            break

        n_left = len(taus) - idx
        alloc = min(rem / float(max(1, n_left)), 80.0)
        if alloc < 3.0:
            break

        dl = time.time() + alloc
        h_new = lbfgsb_optimize(cur_h, n, tau, dl)

        c_new = compute_lags_fast(h_new, n)
        v_new = float(np.max(c_new) * dx)

        if np.isfinite(v_new) and v_new < cur_c - 1.0e-14:
            cur_h = h_new
            cur_c = v_new
            if v_new < best_c:
                best_c = v_new
                best_h = h_new.copy()

    final_h = project_simplex_box(best_h, n)
    final_c = compute_c5(final_h, n)
    return final_h, float(final_c)


def build_active_gradient_matrix(h, n, sel_idx):
    h = np.asarray(h, dtype=float)
    n = int(n)
    p = n - 1
    K = int(len(sel_idx))
    G = np.zeros((K, n), dtype=float)

    for r, idx in enumerate(sel_idx):
        lag = int(idx) - p
        row = G[r]

        if lag == 0:
            row[:] = 1.0 - 2.0 * h
        elif lag > 0:
            m = lag
            row[m:] += 1.0 - h[:n - m]
            row[:n - m] += -h[m:]
        else:
            m = -lag
            row[:n - m] += 1.0 - h[m:]
            row[m:] += -h[:n - m]

    return G


def lp_polish(h, n, deadline, max_iters=20, caps=(0.05, 0.02, 0.01, 0.005)):

            else:

def gradient_pair_search(h, n, deadline, rng, max_passes=120):
    n = int(n)
    h = project_simplex_box(h, n).copy()

    c = compute_lags_fast(h, n)
    best_c_fast = float(np.max(c))
    best_h = h.copy()

    no_improve = 0

    for _ in range(int(max_passes)):
        if time.time() >= deadline - 5.0:
            break

        c = compute_lags_fast(h, n)
        cmax = float(np.max(c))

        if cmax < best_c_fast - 1.0e-14:
            best_c_fast = cmax
            best_h = h.copy()

        window = max(10, int(0.015 * (2 * n - 1)))
        if window < c.size:
            top = np.argpartition(c, -window)[-window:]
        else:
            top = np.arange(c.size)

        gamma = max(0.1, 0.0002 * n)
        near = np.where(c >= cmax - gamma)[0]
        if near.size > 0:
            active = np.unique(np.concatenate((top, near)))
        else:
            active = top

        w = np.zeros(c.size, dtype=float)
        w[active] = 1.0 / float(active.size)
        g = smooth_max_grad(h, n, w)
        g = g - float(np.mean(g))

        pos = np.where(g > 0.0)[0]
        neg = np.where(g < 0.0)[0]

        if pos.size:
            pos = pos[h[pos] > 1.0e-5]
        if neg.size:
            neg = neg[h[neg] < 1.0 - 1.0e-5]

        if pos.size:
            op = np.argsort(g[pos])[::-1]
            pos = pos[op[:min(40, pos.size)]]
        if neg.size:
            on = np.argsort(g[neg])
            neg = neg[on[:min(40, neg.size)]]

        trials = max(5, min(60, 40000 // max(1, n)))
        improved = False

        while trials > 0 and time.time() < deadline - 5.0:
            trials -= 1
            if pos.size == 0 or neg.size == 0:
                break

            i = int(pos[int(rng.randint(pos.size))])
            j = int(neg[int(rng.randint(neg.size))])
            if i == j:
                continue

            maxd = float(min(h[i], 1.0 - h[j], 0.05))
            if maxd <= 1.0e-8:
                continue

            best_m = best_c_fast
            best_h2 = None

            for fr in (0.05, 0.2, 0.5, 0.85):
                d = fr * maxd
                h2 = h.copy()
                h2[i] -= d
                h2[j] += d

                if h2[i] < -1.0e-12 or h2[j] > 1.0 + 1.0e-12:
                    continue

                c2 = compute_lags_fast(h2, n)
                m2 = float(np.max(c2))
                if np.isfinite(m2) and m2 < best_m:
                    best_m = m2
                    best_h2 = h2

            if best_h2 is not None and best_m < best_c_fast - 1.0e-14:
                h = project_simplex_box(best_h2, n)
                best_h = h.copy()
                best_c_fast = best_m
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
            sigma = 0.004 if n > 1000 else 0.008
            h = h + rng.normal(0.0, sigma, n)
            h = project_simplex_box(h, n)

    final_h = project_simplex_box(best_h, n)
    return final_h, compute_c5(final_h, n)

    target = n // 2
    arr = []
    val = 1.0
    while len(arr) < n:
        L = int(rng.randint(7, max(8, min(45, n - len(arr) + 1))))
        for _ in range(L):
            if len(arr) < n:
                arr.append(val)
        val = 1.0 if val < 0.5 else 0.0

    arr = np.asarray(arr[:n], dtype=float)
    count = int(np.sum(arr > 0.5))

    if count > target:
        one_idx = np.where(arr > 0.5)[0]
        order = np.argsort(arr[one_idx])
        for k in range(count - target):
            arr[one_idx[order[k]]] = 0.0
    elif count < target:
        zero_idx = np.where(arr <= 0.5)[0]
        order = np.argsort(-arr[zero_idx])
        for k in range(target - count):
            arr[zero_idx[order[k]]] = 1.0

    return arr


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(max(20.0, float(budget_s)))
    deadline = t0 + budget - 5.0
    rng = np.random.RandomState(int(seed))

    try:
        h_init = np.asarray(initial_h_values, dtype=float)
        if h_init.size == 0:
            raise ValueError("empty initial")
    except Exception:
        h_init = None
        n_init = 500
        h_init = make_random_binary(n_init, rng)

    if h_init is None:
        n_init = 500
        h_init = make_random_binary(n_init, rng)

    n_init = int(np.asarray(h_init).size)
    best_h = project_simplex_box(h_init, n_init)
    best_c = float(compute_c5(best_h, n_init))
    n_best = n_init

    if budget < 40.0:
        return best_h, best_c, n_best

    res = []
    n0 = max(2, n_best)
    if n0 % 2:
        n0 += 1
    res.append(n0)

    n1 = int(min(max(2 * n0, 700), 1500))
    if n1 % 2:
        n1 += 1
    if n1 > n0:
        res.append(n1)

    n2 = int(min(max(4 * n0, 1300), 2500))
    if n2 % 2:
        n2 += 1
    if n2 > res[-1]:
        res.append(n2)

    def add_candidate(h_cand, n_cand):
        nonlocal best_h, best_c, n_best
        if h_cand is None:
            return
        h_cand = np.asarray(h_cand, dtype=float)
        if h_cand.size != n_cand:
            return
        try:
            c_cand = float(compute_c5(h_cand, n_cand))
        except Exception:
            return
        if np.isfinite(c_cand) and c_cand < best_c - 1.0e-15:
            best_c = c_cand
            best_h = project_simplex_box(h_cand, n_cand)
            n_best = int(n_cand)

    # First stage at the parent/sample resolution.
    n_stage = res[0]
    h_stage = resize_h(best_h, n_best, n_stage)
    add_candidate(h_stage, n_stage)

    sec = min(100.0, max(30.0, 0.10 * budget))
    if time.time() + 20.0 < deadline:
        dl = min(deadline - 10.0, time.time() + sec)
        h_stage, c_stage = optimize_resolution(h_stage, n_stage, dl)
        add_candidate(h_stage, n_stage)

    if n_stage <= 1500 and time.time() + 70.0 < deadline:
        dl = min(deadline - 15.0, time.time() + min(90.0, max(25.0, 0.08 * budget)))
        h_stage, c_stage = lp_polish(h_stage, n_stage, dl, max_iters=18)
        add_candidate(h_stage, n_stage)

    # Limited diverse attempt, then refinement at a medium resolution.
    if budget > 450.0 and time.time() + 110.0 < deadline and len(res) > 1:
        n_coarse = 100
        if n0 >= 300:
            n_coarse = max(100, min(220, n0 // 3))
        if n_coarse % 2:
            n_coarse += 1

        h_c = make_random_binary(n_coarse, rng)
        h_c = project_simplex_box(h_c, n_coarse)

        if time.time() + 25.0 < deadline:
            h_c, c_c = optimize_resolution(h_c, n_coarse, min(deadline - 20.0, time.time() + 30.0))
            add_candidate(h_c, n_coarse)

        if n_coarse <= 1200 and time.time() + 30.0 < deadline:
            h_c, c_c = lp_polish(h_c, n_coarse, min(deadline - 20.0, time.time() + 18.0), max_iters=5)
            add_candidate(h_c, n_coarse)

        n_mid = res[1]
        if time.time() + 45.0 < deadline:
            h_mid = resize_h(h_c, n_coarse, n_mid)
            add_candidate(h_mid, n_mid)
            h_mid, c_mid = optimize_resolution(h_mid, n_mid, min(deadline - 20.0, time.time() + 35.0))
            add_candidate(h_mid, n_mid)

    # Medium resolution from the current best.
    if len(res) > 1 and time.time() + 70.0 < deadline:
        n_med = res[1]
        h_med = resize_h(best_h, n_best, n_med)
        add_candidate(h_med, n_med)

        sec = min(120.0, max(30.0, 0.13 * budget))
        if time.time() + 20.0 < deadline:
            dl = min(deadline - 15.0, time.time() + sec)
            h_med, c_med = optimize_resolution(h_med, n_med, dl)
            add_candidate(h_med, n_med)

        if n_med <= 1500 and time.time() + 70.0 < deadline:
            dl = min(deadline - 15.0, time.time() + min(90.0, max(25.0, 0.08 * budget)))
            h_med, c_med = lp_polish(h_med, n_med, dl, max_iters=16)
            add_candidate(h_med, n_med)

        if time.time() + 25.0 < deadline:
            taus = make_tau_list(n_med)
            h_med = lbfgsb_optimize(h_med, n_med, taus[-1], min(deadline - 10.0, time.time() + 20.0))
            add_candidate(h_med, n_med)

    # High resolution.
    if len(res) > 2 and time.time() + 80.0 < deadline:
        n_high = res[2]
        h_high = resize_h(best_h, n_best, n_high)
        add_candidate(h_high, n_high)

        sec = min(120.0, max(30.0, 0.14 * budget))
        if time.time() + 20.0 < deadline:
            dl = min(deadline - 15.0, time.time() + sec)
            h_high, c_high = optimize_resolution(h_high, n_high, dl)
            add_candidate(h_high, n_high)

        if n_high <= 1500 and time.time() + 55.0 < deadline:
            dl = min(deadline - 15.0, time.time() + min(50.0, max(15.0, 0.04 * budget)))
            h_high, c_high = lp_polish(h_high, n_high, dl, max_iters=10)
            add_candidate(h_high, n_high)
        elif time.time() + 55.0 < deadline:
            dl = min(deadline - 10.0, time.time() + min(60.0, max(20.0, 0.05 * budget)))
            h_high, c_high = gradient_pair_search(h_high, n_high, dl, rng)
            add_candidate(h_high, n_high)

    # Final polish at the best resolution found.
    if time.time() + 30.0 < deadline:
        dl = min(deadline - 5.0, time.time() + min(80.0, max(25.0, 0.07 * budget)))
        h_final, c_final = gradient_pair_search(best_h, n_best, dl, rng)
        add_candidate(h_final, n_best)

    if n_best <= 1500 and time.time() + 45.0 < deadline:
        dl = min(deadline - 5.0, time.time() + min(40.0, max(15.0, 0.04 * budget)))
        h_final, c_final = lp_polish(best_h, n_best, dl, max_iters=8)
        add_candidate(h_final, n_best)

    if time.time() + 20.0 < deadline:
        taus = make_tau_list(n_best)
        h_final = lbfgsb_optimize(best_h, n_best, taus[-1], min(deadline - 3.0, time.time() + 15.0))
        add_candidate(h_final, n_best)

    best_h = project_simplex_box(best_h, n_best)
    best_c = float(compute_c5(best_h, n_best))
    if not np.isfinite(best_c):
        fallback = project_simplex_box(np.full(max(2, n_best), 0.5), n_best)
        best_h = fallback
        n_best = int(fallback.size)
        best_c = float(compute_c5(best_h, n_best))

    return best_h, best_c, int(n_best)
