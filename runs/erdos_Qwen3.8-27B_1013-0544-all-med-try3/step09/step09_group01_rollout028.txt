def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    """Project h onto the feasible set: 0 <= h <= 1, sum(h) = n/2."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + diff / tot * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + diff / tot * h)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] = min(1.0, h[idx[0]] + diff)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] = max(0.0, h[idx[0]] + diff)
    return h

def soft_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    # Log-sum-exp for numerical stability
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        lse = (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha
    else:
        lse = np.log(np.sum(np.exp(shifted))) / alpha
    val = m + lse
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    # For each lag k, the gradient contribution is w[m] * dx * d(c[m])/d(h)
    # c[m] at lag k = m-(n-1) is sum_i h[i]*(1-h[i+k]) for valid i
    # d(c[m])/d(h[j]) = (1-h[j+k])*1[j+k<n] - h[j-k]*1[j-k>=0] (with proper bounds)
    
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            # c = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
            # d/dh[j] for j in [0, n-1]:
            #   if j <= n-1-k: contribution from h[j]*(1-h[j+k]) -> (1-h[j+k])
            #   if j >= k: contribution from h[j-k]*(1-h[j]) -> -h[j-k]
            nv = n - k
            # First part: j in [0, nv-1], grad += wm*(1-h[j+k])
            grad[0:nv] += wm * (1.0 - x[k:n])
            # Second part: j in [k, n-1], grad += wm*(-h[j-k])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            # c = sum_{i=0}^{n-1-kk} h[i+kk] * (1-h[i])
            # = sum_{j=kk}^{n-1} h[j] * (1-h[j-kk])
            # d/dh[j] for j in [kk, n-1]: (1-h[j-kk])
            # d/dh[j] for j in [0, n-1-kk]: -h[j+kk]
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def targeted_pair_search(h, n, deadline, max_lags=20, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        improved = True
        passes = 0
        while improved and time.time() < deadline and passes < 5:
            improved = False
            passes += 1
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            # Get active lags
            active = np.where(c_vec >= max_val - 1e-9)[0]
            if len(active) > max_lags:
                active = active[np.argsort(-c_vec[active])[:max_lags]]
            
            # For each active lag, try targeted swaps
            for lag_idx in active:
                if time.time() > deadline:
                    break
                k = lag_idx - (n - 1)
                if k == 0:
                    continue
                ak = abs(k)
                if ak >= n:
                    continue
                # Pairs: (i, i+ak) for i in [0, n-1-ak)
                n_pairs = n - ak
                for i in range(n_pairs):
                    j = i + ak
                    # Try moving mass from j to i (decrease h[j], increase h[i])
                    # This decreases c at lag k if h[j] was contributing
                    if h[j] > d and h[i] < 1.0 - d:
                        h[j] -= d
                        h[i] += d

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return h

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, max_iter_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = max_iter_list[i] if max_iter_list and i < len(max_iter_list) else max(500, int(remaining * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 10,
                                   'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

                h_new = project_feasible(res.x, n)
