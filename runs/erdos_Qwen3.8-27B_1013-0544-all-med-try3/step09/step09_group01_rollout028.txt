```
def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    if h.size != n:
        h = h[:n] if h.size>n else np.pad(h,(0,n-h.size))
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    for _ in range(1000):
       diff=target-h.sum()
       if abs(diff)<1e-15: break
       if diff>0:
          room=1.0-h; tot=room.sum()
          if tot<=1e-15: break
          step=diff/tot
          # if step <=1, no saturate; if >1, saturate
          h= h + step*room
       else:
          tot=h.sum()
          if tot<=1e-15: break
          h=h + (diff/tot)*h
       h=np.clip(h,0.0,1.0)
    diff=target-h.sum()
    if diff>0:
       order=np.argsort(1.0-h)
       for idx in order:
          add=min(1.0-h[idx], diff)
          h[idx]+=add; diff-=add
          if diff<=1e-15: break
    elif diff<0:
       order=np.argsort(h)
       for idx in order:
          sub=min(h[idx], -diff)
          h[idx]-=sub; diff+=sub
          if diff>=-1e-15: break
    h=np.clip(h,0.0,1.0)
    return h
```

```python
import numpy as np
import time
from scipy.optimize import minimize

try:
    from scipy.signal import correlate as _sig_correlate
except Exception:
    _sig_correlate = None


def c5_vec(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if _sig_correlate is not None and n >= 1200:
        c = _sig_correlate(h, 1.0 - h, mode='full', method='fft')
        c = np.real(c)
    else:
        c = np.correlate(h, 1.0 - h, mode='full')
    return c * (2.0 / float(n))


def eval_c5(h, n):
    return float(np.max(c5_vec(h, n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.size), mode='edge')
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(1200):
        diff = target - float(h.sum())
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)

    diff = target - float(h.sum())
    if diff > 1e-14:
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(float(1.0 - h[idx]), diff)
            if add <= 0.0:
                break
            h[idx] += add
            diff -= add
            if diff <= 1e-15:
                break
    elif diff < -1e-14:
        order = np.argsort(h)
        neg = -diff
        for idx in order:
            sub = min(float(h[idx]), neg)
            if sub <= 0.0:
                break
            h[idx] -= sub
            neg -= sub
            if neg <= 1e-15:
                break

    return np.clip(h, 0.0, 1.0)


def make_bounds(n):
    return [(0.0, 1.0)] * int(n)


def logsumexp_safe(v):
    v = np.asarray(v, dtype=float)
    mx = float(np.max(v))
    if mx > 500.0:
        return mx + float(np.log(np.sum(np.exp(v - mx))))
    return float(np.log(np.sum(np.exp(v))))


def select_active(cvec, top_k, tol_frac=2.0e-4):
    cvec = np.asarray(cvec, dtype=float)
    m = float(np.max(cvec))
    thr = m - max(1e-12, tol_frac * max(m, 1.0))
    idx = np.where(cvec >= thr)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(cvec))], dtype=int)
    if idx.size > int(top_k):
        order = np.argsort(-cvec[idx], kind='stable')
        idx = idx[order[:int(top_k)]]
    return idx.astype(int)


def accumulate_lag_grad(h, n, active, weights, dx, grad):
    h = np.asarray(h, dtype=float)
    for wi, mm in zip(weights, active):
        if wi <= 0.0:
            continue
        wm = float(wi) * dx
        k = int(mm) - (int(n) - 1)
        if k >= 0:
            nv = int(n) - k
            if nv <= 0:
                continue
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = int(n) - kk
            if nv <= 0:
                continue
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] -= wm * h[kk:]


def active_obj(x, n, active, alpha, mu):
    c = c5_vec(x, n)
    vals = c[active]
    mx = float(np.max(vals))
    sh = alpha * (vals - mx)
    val = mx + logsumexp_safe(sh) / alpha
    if mu > 0.0:
        s = float(x.sum()) - 0.5 * float(n)
        val += mu * s * s
    return float(val)


def active_grad(x, n, active, alpha, mu):
    c = c5_vec(x, n)
    vals = c[active]
    mx = float(np.max(vals))
    sh = alpha * (vals - mx)
    smx = float(np.max(sh))
    w = np.exp(sh - smx)
    w /= float(np.sum(w))

    grad = np.zeros(int(n), dtype=float)
    dx = 2.0 / float(n)
    accumulate_lag_grad(x, n, active, w, dx, grad)

    if mu > 0.0:
        grad += 2.0 * mu * (float(x.sum()) - 0.5 * float(n))
    return grad


def active_refine(h, n, deadline, rng=None, top_k=12, alphas=None, mu=0.002, max_outer=3):
    n = int(n)
    if alphas is None:
        alphas = (1000.0, 10000.0, 100000.0)

    h = project_feasible(h, n)
    best_c = eval_c5(h, n)
    bounds = make_bounds(n)
    stalled = 0

    for _outer in range(int(max_outer)):
        remaining = float(deadline) - time.time()
        if remaining < 20.0:
            break

        cvec = c5_vec(h, n)
        active = select_active(cvec, top_k, 2.0e-4)
        improved = False

        for alpha in alphas:
            remaining = float(deadline) - time.time()
            if remaining < 12.0:
                break

            if n <= 600:
                maxit = int(max(80, min(1200, int(remaining * 5))))
            elif n <= 1300:
                maxit = int(max(60, min(650, int(remaining * 3))))
            else:
                maxit = int(max(30, min(240, int(remaining * 2))))

            try:
                res = minimize(
                    active_obj,
                    h,
                    jac=active_grad,
                    args=(n, active, float(alpha), float(mu)),
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={
                        'maxiter': maxit,
                        'ftol': 1e-20,
                        'gtol': 1e-14,
                        'maxls': 40,
                        'maxcor': 50
                    }
                )
            except Exception:
                res = None

            if res is not None and hasattr(res, 'x'):
                h2 = project_feasible(res.x, n)
                c2 = eval_c5(h2, n)
                if c2 + 1e-14 < best_c:
                    h = h2
                    best_c = float(c2)
                    improved = True

            if time.time() > float(deadline) - 12.0:
                break

        if not improved:
            stalled += 1
            if rng is not None and stalled == 1:
                scale = 0.025 if n > 1200 else 0.04
                try:
                    h2 = project_feasible(h + scale * rng.standard_normal(n), n)
                    c2 = eval_c5(h2, n)
                    if c2 + 1e-14 < best_c:
                        h = h2
                        best_c = float(c2)
                        stalled = 0
                except Exception:
                    pass
            if stalled >= 3:
                break
        else:
            stalled = 0

    return h, best_c


def resample_1d(h, n_old, n_new):
    h = np.asarray(h, dtype=float)
    n_old = int(n_old)
    n_new = int(n_new)
    if n_old == n_new:
        return h.copy()
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)


def consider_candidate(h, n, best_h, best_c):
    h = project_feasible(h, int(n))
    c = eval_c5(h, n)
    if c < best_c:
        return h, float(c)
    return best_h, best_c


def symmetrize(h):
    h = np.asarray(h, dtype=float)
    return project_feasible(0.5 * (h + h[::-1]), len(h))


def make_smooth_start(n, rng, modes=12):
    n = int(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 * np.ones(n, dtype=float)
    for k in range(1, int(modes) + 1):
        amp = 0.5 / float(k)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * float(k) * x + ph)
    dev = float(np.max(np.abs(h - 0.5)))
    if dev > 0.5:
        h = 0.5 + (h - 0.5) * (0.5 / dev)
    return project_feasible(h, n)


def make_trig_start(n, radius, phi):
    n = int(n)
    radius = float(radius)
    phi = float(phi)
    if radius > 0.5:
        radius = 0.5
    elif radius < 0.0:
        radius = 0.0
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + radius * np.cos(np.pi * x + phi)
    return project_feasible(h, n)


def make_two_level_from_h(h, n, delta):
    h = project_feasible(h, n)
    n = int(n)
    order = np.argsort(-h, kind='stable')
    b = np.zeros(n, dtype=float)
    b[order[:n // 2]] = 1.0
    s = 2.0 * b - 1.0
    d = float(delta)
    if d > 0.5:
        d = 0.5
    elif d < 0.0:
        d = 0.0
    h2 = 0.5 + d * s
    return project_feasible(h2, n)


def expand_blocks(v, q):
    q = int(q)
    return np.repeat(np.asarray(v, dtype=float), q)


def block_active_obj(v, n, q, active, alpha, mu):
    h = expand_blocks(v, q)
    c = c5_vec(h, n)
    vals = c[active]
    mx = float(np.max(vals))
    sh = alpha * (vals - mx)
    val = mx + logsumexp_safe(sh) / alpha
    if mu > 0.0:
        s = float(h.sum()) - 0.5 * float(n)
        val += mu * s * s
    return float(val)


def block_active_grad(v, n, q, active, alpha, mu):
    n = int(n)
    q = int(q)
    h = expand_blocks(v, q)
    c = c5_vec(h, n)
    vals = c[active]
    mx = float(np.max(vals))
    sh = alpha * (vals - mx)
    smx = float(np.max(sh))
    w = np.exp(sh - smx)
    w /= float(np.sum(w))

    grad_full = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    accumulate_lag_grad(h, n, active, w, dx, grad_full)

    if mu > 0.0:
        grad_full += 2.0 * mu * (float(h.sum()) - 0.5 * float(n))

    m = int(np.ceil(float(n) / float(q)))
    if v.size == m and n == m * q:
        offsets = np.arange(0, n, q, dtype=int)
        grad_v = np.add.reduceat(grad_full, offsets)
    else:
        grad_v = expand_blocks(v, q).copy()
        tmp = np.reshape(grad_full, (m, q))
        grad_v = tmp.sum(axis=1)
    return np.asarray(grad_v, dtype=float)

                )
                v2 = v_trig if np.random.RandomState(0).rand() < 0.5 else v_rand
                vb2, hb2, cb2 = block_refine(
                    v2, m, q, cfg_end - 6.0, rng,
                    top_k=12,
                    alphas=(1000.0, 10000.0, 100000.0),
                    mu=0.01,
                    max_outer=1
                )
                best_h, best_c = consider_candidate(hb2, m * q, best_h, best_c)

    # Medium-high resolution refinement.
    rem = deadline - time.time()
    if rem > 300.0:
        n_mid = 1024
        if best_n == n_mid:
            h_mid = best_h.copy()
        else:
            h_mid = project_feasible(resample_1d(best_h, best_n, n_mid), n_mid)
        best_h, best_c = consider_candidate(h_mid, n_mid, best_h, best_c)

        end = phase_end(time.time(), 240.0, deadline, 120.0)
        if end > time.time() + 20.0:
            h_mid, c_mid = active_refine(
                h_mid, n_mid, end, rng,
                top_k=16,
                alphas=(500.0, 2000.0, 20000.0, 200000.0, 1000000.0),
                mu=0.001,
                max_outer=2
            )
            best_h, best_c = consider_candidate(h_mid, n_mid, best_h, best_c)

    # Very high resolution polish if budget allows.
    rem = deadline - time.time()
    if rem > 200.0:
        n_high = 2048
        if best_n == n_high:
            h_high = best_h.copy()
        else:
            h_high = project_feasible(resample_1d(best_h, best_n, n_high), n_high)
        best_h, best_c = consider_candidate(h_high, n_high, best_h, best_c)

        end = phase_end(time.time(), 170.0, deadline, 25.0)
        if end > time.time() + 20.0:
            h_high, c_high = active_refine(
                h_high, n_high, end, rng,
                top_k=14,
                alphas=(2000.0, 20000.0, 200000.0, 1000000.0),
                mu=0.0008,
                max_outer=2
            )
            best_h, best_c = consider_candidate(h_high, n_high, best_h, best_c)

    # Final transformations and resolution sweeps.
    if deadline - time.time() > 6.0:
        best_h, best_c = consider_candidate(symmetrize(best_h), best_n, best_h, best_c)

        for delta in (0.20, 0.28, 0.34, 0.40, 0.46):
            best_h, best_c = consider_candidate(
                make_two_level_from_h(best_h, best_n, delta),
                best_n, best_h, best_c
            )

        radius_candidates = (0.38, 0.43, 0.48)
        phi_candidates = (0.0, 0.35, 0.70, 1.05, 1.40, 1.90, 2.40, 2.90)
        for radius in radius_candidates:
            for phi in phi_candidates:
                best_h, best_c = consider_candidate(
                    make_trig_start(best_n, radius, phi),
                    best_n, best_h, best_c
                )

        for cand_n in (400, 500, 512, 600, 640, 700, 800, 1024, 1536, 2048):
            if cand_n == best_n or cand_n < 128 or cand_n > 3000:
                continue
            if time.time() > deadline - 3.0:
                break
            h_cand = project_feasible(resample_1d(best_h, best_n, cand_n), cand_n)
            best_h, best_c = consider_candidate(h_cand, cand_n, best_h, best_c)

    # Tiny last-moment polish.
    if deadline - time.time() > 12.0:
        h_last, c_last = active_refine(
            best_h, best_n, deadline - 1.0, rng,
            top_k=8,
            alphas=(10000.0, 100000.0),
            mu=0.001,
            max_outer=1
        )
        best_h, best_c = consider_candidate(h_last, best_n, best_h, best_c)

    best_n = int(best_n)
    best_h = project_feasible(best_h, best_n)
    best_c = eval_c5(best_h, best_n)
    return best_h, float(best_c), best_n
```
