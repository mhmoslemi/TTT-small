import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def soft_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)


def soft_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return w / s

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = soft_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx


def lbfgsb_step(h, n, alpha, deadline, maxiter=1000, m=20):
    h0 = project_feasible(h, n)

    def obj(x):
        return soft_max_val(compute_all_lags(x, n), alpha)

    def grad(x):
        return smooth_max_grad(x, n, alpha)

    try:
        res = minimize(obj, h0, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-9, 'maxls': 50, 'iprint': -1,
                                'm': m})
        return project_feasible(res.x, n)
    except Exception:
        return h0


def upsample(h, n_src, n_dst):
    x_src = np.arange(n_src) * (2.0 / n_src)
    x_dst = np.arange(n_dst) * (2.0 / n_dst)
    h_ext = np.concatenate([h[-1:], h, h[:1]])
    x_ext = np.concatenate([[-2.0 / n_src], x_src, [2.0 + 2.0 / n_src]])
    h_dst = np.interp(x_dst, x_ext, h_ext)
    return project_feasible(h_dst, n_dst)


def downsample(h, n_src, n_dst):
    x_src = np.arange(n_src) * (2.0 / n_src)
    x_dst = np.arange(n_dst) * (2.0 / n_dst)
    h_ext = np.concatenate([h[-1:], h, h[:1]])
    x_ext = np.concatenate([[-2.0 / n_src], x_src, [2.0 + 2.0 / n_src]])
    h_dst = np.interp(x_dst, x_ext, h_ext)
    return project_feasible(h_dst, n_dst)


def fourier_project(h, n, n_modes):
    H = np.fft.rfft(h)
    mag = np.abs(H).copy()
    mag[0] = 0
    if n_modes < len(mag):
        keep = np.argsort(mag)[::-1][:n_modes]
        mask = np.zeros(len(mag), dtype=bool)
        mask[keep] = True
        mask[0] = True
        H[~mask] = 0
    h_proj = np.real(np.fft.irfft(H, n=n))
    return project_feasible(h_proj, n)
